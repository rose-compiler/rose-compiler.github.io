136 template<class Filter>
137 Sawyer::Container::Trace<Address> trace(ThreadId tid, Filter &filter) {
138 Sawyer::Container::Trace<Address> retval;
139 while (!isTerminated()) {
140 Address va = executionAddress(tid);
141 FilterAction action = filter(va);
142 if (action.isClear(FilterActionFlag::REJECT))
143 retval.append(va);
144 if (action.isSet(FilterActionFlag::STOP))
145 return retval;
146 singleStep(tid);
147 }
148 return retval;
149 }
