149 void *
150 SgAsmPESectionTableEntry::encode(PESectionTableEntry_disk *disk) const
151 {
152  /* The file can hold up to eight characters of the name. The name is NUL-padded, not necessarily NUL-terminated. */
153  if (p_name.size()>8) {
154  mlog[WARN] <<"section name too long to store in PE file: \"" <<escapeString(p_name) <<"\" (truncated)\n";
155  }
156  memset(disk->name, 0, sizeof(disk->name));
157 
158 #ifdef USE_ROSE
159  // DQ (1/27/2010): std::min() does not appear to be handle different type of arguments for ROSE. Need to look into this later.
160  // memcpy(disk->name, p_name.c_str(), std::min(sizeof(disk->name), (size_t)p_name.size()));
161  memcpy(disk->name, p_name.c_str(), std::min( (size_t)(sizeof(disk->name)), (size_t)(p_name.size()) ));
162 #else
163  memcpy(disk->name, p_name.c_str(), std::min(sizeof(disk->name), p_name.size()));
164 #endif
165 
166  ByteOrder::hostToLe(p_virtual_size, &(disk->virtual_size));
167  ByteOrder::hostToLe(p_rva, &(disk->rva));
168  ByteOrder::hostToLe(p_physical_size, &(disk->physical_size));
169  ByteOrder::hostToLe(p_physical_offset, &(disk->physical_offset));
170  ByteOrder::hostToLe(p_coff_line_nums, &(disk->coff_line_nums));
171  ByteOrder::hostToLe(p_n_relocs, &(disk->n_relocs));
172  ByteOrder::hostToLe(p_n_coff_line_nums, &(disk->n_coff_line_nums));
173  ByteOrder::hostToLe(p_flags, &(disk->flags));
174 
175  return disk;
176 }

215 SgAsmPESectionTable*
216 SgAsmPESectionTable::parse()
217 {
218  SgAsmGenericSection::parse();
219 
220  SgAsmPEFileHeader *fhdr = dynamic_cast<SgAsmPEFileHeader*>(get_header());
221  ROSE_ASSERT(fhdr!=nullptr);
222 
223  /* Parse section table and construct section objects, but do not parse the sections yet. */
224  SgAsmGenericSectionPtrList pending;
225  const size_t entsize = sizeof(SgAsmPESectionTableEntry::PESectionTableEntry_disk);
226  for (size_t i=0; i<fhdr->get_e_nsections(); i++) {
227  SgAsmPESectionTableEntry::PESectionTableEntry_disk disk;
228  if (entsize!=readContentLocal(i * entsize, &disk, entsize, false))
229  mlog[WARN] <<"SgAsmPESectionTable::parse: section table entry " <<i
230  <<" at file offset " <<StringUtility::addrToString(get_offset()+i*entsize)
231  <<" extends beyond end of defined section table.\n";
232  SgAsmPESectionTableEntry *entry = new SgAsmPESectionTableEntry(&disk);
233 
234  SgAsmPESection *section = nullptr;
235  if (entry->get_name() == ".idata") {
236  // If the PAIR_IMPORTS rva/size pair has a non-zero pointer, then avoid creating an import table from this ".idata"
237  // section. Sometimes the rva/size pair will point to a different region in memory than ".idata", in which case the
238  // rva/size pair should be honored instead.
239  SgAsmPERVASizePair *pair = fhdr->get_rvaSizePairs()->get_pairs()[SgAsmPEFileHeader::PAIR_IMPORTS];
240  if (pair->get_e_rva().va().orElse(pair->get_e_rva().rva()) == 0) {
241  section = new SgAsmPEImportSection(fhdr); // treat .idata as an import table
242  } else {
243  section = new SgAsmPESection(fhdr); // construct the import table from the rva/size pair instead
244  }
245  } else {
246  section = new SgAsmPESection(fhdr);
247  }
248  section->initFromSectionTable(entry, i+1);
249  pending.push_back(section);
250  }
251 
252  /* Build the memory mapping like the real loader would do. This is the same code used by
253  * SgAsmExecutableFileFormat::parseBinaryFormat() except we're doing it here early because we need it in the rest of the
254  * PE parser. */
255  ROSE_ASSERT(nullptr==fhdr->get_loaderMap());
256  BinaryLoader::Ptr loader = BinaryLoader::lookup(fhdr); /*no need to clone; we're not changing any settings*/
257  ASSERT_not_null(loader);
258  MemoryMap::Ptr loader_map = MemoryMap::instance();
259  loader->remap(loader_map, fhdr);
260  fhdr->set_loaderMap(loader_map);
261 
262  /* Parse each section after the loader map is created */
263  for (size_t i=0; i<pending.size(); i++)
264  pending[i]->parse();
265 
266  return this;
267 }

276 SgAsmPESectionTable::addSection(SgAsmPESection *section)
277 {
278  ROSE_ASSERT(section!=nullptr);
279  ROSE_ASSERT(section->get_file()==get_file());
280  ROSE_ASSERT(section->get_header()==get_header());
281  ROSE_ASSERT(section->get_section_entry()==nullptr); /* must not be in the section table yet */
282 
283  SgAsmPEFileHeader *fhdr = dynamic_cast<SgAsmPEFileHeader*>(get_header());
284  ROSE_ASSERT(fhdr!=nullptr);
285 
286  /* Assign an ID if there isn't one yet. */
287  if (section->get_id()<0) {
288  SgAsmGenericSectionList *seclist = fhdr->get_sections();;
289  int max_id=0; /*assume zero is used so we start at one*/
290  for (size_t i=0; i<seclist->get_sections().size(); i++) {
291  SgAsmGenericSection *s = seclist->get_sections()[i];
292 
293  max_id = std::max(max_id, s->get_id());
294 
295  }
296  section->set_id(max_id+1);
297  }
298 
299  /* Create a new section table entry. */
300  SgAsmPESectionTableEntry *entry = new SgAsmPESectionTableEntry;
301  entry->updateFromSection(section);
302  section->set_section_entry(entry);
303 }

306 bool
307 SgAsmPESectionTable::reallocate()
308 {
309  bool reallocated = false;
310 
311  /* Resize based on section having largest ID */
312  SgAsmPEFileHeader *fhdr = dynamic_cast<SgAsmPEFileHeader*>(get_header());
313  ROSE_ASSERT(fhdr != nullptr);
314  SgAsmGenericSectionPtrList sections = fhdr->get_sections()->get_sections();
315  int max_id = 0;
316  for (size_t i=0; i<sections.size(); i++) {
317 
318  max_id = std::max(max_id, sections[i]->get_id());
319 
320  }
321 
322  size_t nsections = max_id; /*PE section IDs are 1-origin*/
323  size_t need = nsections * sizeof(SgAsmPESectionTableEntry::PESectionTableEntry_disk);
324  if (need < get_size()) {
325  if (isMapped()) {
326  ROSE_ASSERT(get_mappedSize()==get_size());
327  set_mappedSize(need);
328  }
329  set_size(need);
330  reallocated = true;
331  } else if (need > get_size()) {
332  get_file()->shiftExtend(this, 0, need-get_size(), SgAsmGenericFile::ADDRSP_ALL, SgAsmGenericFile::ELASTIC_HOLE);
333  reallocated = true;
334  }
335 
336  return reallocated;
337 }

340 void
341 SgAsmPESectionTable::unparse(std::ostream &f) const
342 {
343  SgAsmPEFileHeader *fhdr = dynamic_cast<SgAsmPEFileHeader*>(get_header());
344  ROSE_ASSERT(fhdr != nullptr);
345  SgAsmGenericSectionPtrList sections = fhdr->get_sections()->get_sections();
346 
347  for (size_t i = 0; i < sections.size(); i++) {
348  if (sections[i]->get_id()>=0) {
349  SgAsmPESection *section = isSgAsmPESection(sections[i]);
350  ROSE_ASSERT(section!=nullptr);
351 
352  /* Write the table entry */
353  ROSE_ASSERT(section->get_id() > 0); /*ID's are 1-origin in PE*/
354  size_t slot = section->get_id() - 1;
355  SgAsmPESectionTableEntry *shdr = section->get_section_entry();
356  SgAsmPESectionTableEntry::PESectionTableEntry_disk disk;
357  shdr->encode(&disk);
358  write(f, slot*sizeof(disk), sizeof disk, &disk);
359  }
360  }
361 }
