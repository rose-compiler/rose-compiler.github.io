665 int SgNodeHelper::scopeSequenceNumber(SgNode* node) {
666  /* this is quite an expensive function (searches the AST upwards and within one container). However, the purpose is to provide
667  a meaningful (=human readable) scope-sequence number. Alternatively it could also be provided for each scope as precomputed value
668  attached as ast-attribute.
669  1) find scope we are in
670  2) determine scope number from parent container
671  scopes are numbered 0 .. n
672  returns the scope-number of the provided node's scope within the outer scope
673  returns -1 if the provided node has no outer scope
674  */

688  if(isSgScopeStatement(child)) {
689  ++cnum;

692  }

694  throw CodeThorn::Exception("SgNodeHelper::scopeSequenceNumber: improper node operation.");
