9 void CollectDependences::CollectFromFile(std::istream& input_file) {
10  Log.push("Constructing DependenceTable");
11  while (input_file.good()) {
12  std::string next_string;
13  // Each line starts with the name of a component in the software.
14  if (input_file >> next_string) {
15  Log.push("reading " + next_string);
16  std::string dest = next_string;
17  input_file >> next_string;
18  if (next_string != ":") {
19  Log.fatal("Expecting `:' after dependence destination " + dest + " " + next_string);
20  }
21  std::string source, dep_type, attr;
22  // Read and process all the components that `dest' depends on immediately.
23  while ((input_file >> next_string).good()) {
24  Log.push("reading " + next_string);
25  if (next_string == ";") {
26  if (source != "") {
27  save_dependence(DependenceEntry(dest, source, dep_type, attr));
28  source = attr = "";
29  Log.push( source + "->" + dest + "[" + dep_type + "]");
30  }
31  break;
32  }
33  else if (next_string == "[") {
34  if (!(input_file >> next_string).good()) {
35  Log.fatal("Expecting strings after `[' but get " + next_string);
36  }
37  dep_type = next_string;
38  if (!(input_file >> next_string).good() || next_string != "]") {
39  Log.fatal("Expecting \"]\" but get " + next_string);
40  }
41  } else if (next_string == "<-" || next_string == "->" || next_string == "=") {
42  if (!(input_file >> next_string).good()) {
43  Log.fatal("Expecting strings after `=' but get " + next_string);
44  }
45  attr = next_string;
46  } else if (attr == "") {
47  source = source + next_string;
48  } else {
49  attr = attr + next_string;
50  }
51  }
52  if (next_string != ";") {
53  Log.fatal("Expecting `;' but getting " + next_string);
54  }
55  } else if (input_file.peek() == EOF) {
