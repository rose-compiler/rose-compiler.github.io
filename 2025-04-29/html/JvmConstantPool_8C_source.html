155 {

158  switch (get_tag()) {
159  case PoolEntry::CONSTANT_Class: // 4.4.1 CONSTANT_Class_info table entry

162  Jvm::read_value(pool, p_name_index);
163  break;
164 
165  case PoolEntry::CONSTANT_String: // 4.4.2 CONSTANT_String_info table entry
166  Jvm::read_value(pool, p_string_index);
167  break;
168 
169  case PoolEntry::CONSTANT_Fieldref: // 4.4.3 CONSTANT_Fieldref_info table entry
170  case PoolEntry::CONSTANT_Methodref: // 4.4.3 CONSTANT_Methodref_info table entry
171  case PoolEntry::CONSTANT_InterfaceMethodref: // 4.4.3 CONSTANT_InterfeceMethodref_info table entry
172  Jvm::read_value(pool, p_class_index);
173  Jvm::read_value(pool, p_name_and_type_index);
174  break;
175 
176  case PoolEntry::CONSTANT_Integer: // 4.4.4 CONSTANT_Integer_info table entry
177  case PoolEntry::CONSTANT_Float: // 4.4.4 CONSTANT_Float_info table entry
178  Jvm::read_value(pool, p_bytes);
179  break;
180 
181  case PoolEntry::CONSTANT_Long: // 4.4.5 CONSTANT_Long_info table entry
182  case PoolEntry::CONSTANT_Double: // 4.4.5 CONSTANT_Double_info table entry
183  Jvm::read_value(pool, p_hi_bytes);
184  Jvm::read_value(pool, p_low_bytes);
185  break;
186 
187  case PoolEntry::CONSTANT_NameAndType: // 4.4.6 CONSTANT_NameAndType_info table entry
188  Jvm::read_value(pool, p_name_index);
189  Jvm::read_value(pool, p_descriptor_index);
190  break;
191 
192  case PoolEntry::CONSTANT_Utf8: // 4.4.7 CONSTANT_Utf8_info table entry
193  {
194  char* bytes{nullptr};

197  set_utf8_bytes(bytes);
198  ASSERT_not_null(bytes);
199  break;
200  }
201 
202  case PoolEntry::CONSTANT_MethodHandle: // 4.4.8 CONSTANT_MethodHandle_info table entry
203  Jvm::read_value(pool, p_reference_kind);
204  Jvm::read_value(pool, p_reference_index);
205  break;
206

208  Jvm::read_value(pool, p_descriptor_index);
209  break;
210

212  case PoolEntry::CONSTANT_InvokeDynamic: // 4.4.10 CONSTANT_InvokeDynamic_info table entry
213  Jvm::read_value(pool, p_bootstrap_method_attr_index);
214  Jvm::read_value(pool, p_name_and_type_index);
215  break;
216 
217  default:
218  set_tag(PoolEntry::EMPTY);
219  }
220 
221  return this;

237  auto string_index = p_string_index;
238  hostToBe(p_string_index, &string_index);
239  os.write(reinterpret_cast<const char*>(&string_index), sizeof string_index);
240  break;
241  }
242  case PoolEntry::CONSTANT_Fieldref: // 4.4.3 CONSTANT_Fieldref_info table entry
243  case PoolEntry::CONSTANT_Methodref: // 4.4.3 CONSTANT_Methodref_info table entry
244  case PoolEntry::CONSTANT_InterfaceMethodref: { // 4.4.3 CONSTANT_InterfaceMethodref_info table entry
245  auto class_index = p_class_index;
246  auto name_and_type_index = p_name_and_type_index;
247  hostToBe(p_class_index, &class_index);
248  hostToBe(p_name_and_type_index, &name_and_type_index);
249  os.write(reinterpret_cast<const char*>(&class_index), sizeof class_index);
250  os.write(reinterpret_cast<const char*>(&name_and_type_index), sizeof name_and_type_index);
251  break;
252  }
253  case PoolEntry::CONSTANT_Integer: // 4.4.4 CONSTANT_Integer_info table entry
254  case PoolEntry::CONSTANT_Float: { // 4.4.4 CONSTANT_Float_info table entry
255  auto bytes = p_bytes;
256  hostToBe(p_bytes, &bytes);
257  os.write(reinterpret_cast<const char*>(&bytes), sizeof bytes);
258  break;
259  }
260  case PoolEntry::CONSTANT_Long: // 4.4.5 CONSTANT_Long_info table entry
261  case PoolEntry::CONSTANT_Double: { // 4.4.5 CONSTANT_Double_info table entry
262  auto hi_bytes = p_hi_bytes;
263  auto low_bytes = p_low_bytes;
264  hostToBe(p_hi_bytes, &hi_bytes);
265  hostToBe(p_low_bytes, &low_bytes);
266  os.write(reinterpret_cast<const char*>(&hi_bytes), sizeof hi_bytes);
267  os.write(reinterpret_cast<const char*>(&low_bytes), sizeof low_bytes);
268  break;
269  }
270  case PoolEntry::CONSTANT_NameAndType: { // 4.4.6 CONSTANT_NameAndType_info table entry
271  auto name_index = p_name_index;
272  auto descriptor_index = p_descriptor_index;
273  hostToBe(p_name_index, &name_index);
274  hostToBe(p_descriptor_index, &descriptor_index);
275  os.write(reinterpret_cast<const char*>(&name_index), sizeof name_index);
276  os.write(reinterpret_cast<const char*>(&descriptor_index), sizeof descriptor_index);
277  break;
278  }
279  case PoolEntry::CONSTANT_Utf8: // 4.4.7 CONSTANT_Utf8_info table entry
280  {
281  auto length = p_length;
282  hostToBe(p_length, &length);
283  os.write(reinterpret_cast<const char*>(&length), sizeof length);
284  os.write(p_utf8_bytes, p_length);
285  break;
286  }
287  case PoolEntry::CONSTANT_MethodHandle: { // 4.4.8 CONSTANT_MethodHandle_info table entry
288  auto reference_kind = p_reference_kind;
289  auto reference_index = p_reference_index;
290  hostToBe(p_reference_kind, &reference_kind);
291  hostToBe(p_reference_index, &reference_index);
292  os.write(reinterpret_cast<const char*>(&reference_kind), sizeof reference_kind);
293  os.write(reinterpret_cast<const char*>(&reference_index), sizeof reference_index);
294  break;
295  }
296  case PoolEntry::CONSTANT_MethodType: { // 4.4.9 CONSTANT_MethodType_info table entry
297  auto descriptor_index = p_descriptor_index;
298  hostToBe(p_descriptor_index, &descriptor_index);
299  os.write(reinterpret_cast<const char*>(&descriptor_index), sizeof descriptor_index);
300  break;
301  }
302  case PoolEntry::CONSTANT_Dynamic: // 4.4.10 CONSTANT_Dynamic_info table entry
303  case PoolEntry::CONSTANT_InvokeDynamic: { // 4.4.10 CONSTANT_InvokeDynamic_info table entry
304  auto bootstrap_method_attr_index = p_bootstrap_method_attr_index;
305  auto name_and_type_index = p_name_and_type_index;
306  hostToBe(p_bootstrap_method_attr_index, &bootstrap_method_attr_index);
307  hostToBe(p_name_and_type_index, &name_and_type_index);
308  os.write(reinterpret_cast<const char*>(&bootstrap_method_attr_index), sizeof bootstrap_method_attr_index);
309  os.write(reinterpret_cast<const char*>(&name_and_type_index), sizeof name_and_type_index);
310  break;
311  }
312  case PoolEntry::EMPTY: // An extra entry following CONSTANT_Long or CONSTANT_Double (a noop)
313  default:
314  break;
315  }

318 SgAsmJvmConstantPool* SgAsmJvmConstantPool::parse()
319 {
320  PoolEntry* entry{nullptr};
321  auto header = get_header();
322  ASSERT_not_null(header);
323 
324  /* Constant pool count */
325  uint16_t constant_pool_count;
326  Jvm::read_value(this, constant_pool_count);
327 
328  // A constant_pool index is considered valid if it is greater than zero and less than constant_pool_count
329  for (int ii = 1; ii < constant_pool_count; ii++) {
330  /* tag */
331  uint8_t tag;
332  Jvm::read_value(this, tag);
333 
334  // Create and initialize (parse) a new entry
335  auto kind = static_cast<PoolEntry::Kind>(tag);
336  entry = new PoolEntry(kind);
337  entry->parse(this);
338 
339  // Store the new entry
340  get_entries().push_back(entry);
341 
342  // If this is CONSTANT_Long or CONSTANT_Double, store index location with empty entry
343  // 4.4.5 "In retrospect, making 8-byte constants take two constant pool entries was a poor choice."
344  //
345  if (kind == PoolEntry::CONSTANT_Long || kind == PoolEntry::CONSTANT_Double) {
346  // Create and store an empty entry
347  entry = new PoolEntry(PoolEntry::EMPTY);
348  get_entries().push_back(entry);
349  ii += 1;
350  }
351  }
352  return this;
353 }

356 void SgAsmJvmConstantPool::unparse(std::ostream& os) const
357 {
358  uint16_t count = get_entries().size() + 1; // constant_pool_count = #entries + 1
359  hostToBe(count, &count);
360  os.write(reinterpret_cast<const char*>(&count), sizeof count);
361 
362  for (auto entry : get_entries()) {
363  uint8_t tag = entry->get_tag();
364  auto kind = static_cast<PoolEntry::Kind>(tag);
365 
366  if (kind != PoolEntry::EMPTY) {
367  hostToBe(tag, &tag);
368  os.write(reinterpret_cast<const char*>(&tag), sizeof tag);
369  entry->unparse(os);
370  }
371  }
