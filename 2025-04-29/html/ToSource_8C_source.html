218  std::string value = SValue::promote(sideEffect.expression)->ctext();

232  if (sideEffects[i].location) {

234  bool isLastOccurrence = true;

236  if (sideEffects[j].location) {

240  }

271 BinaryToSource::emitBasicBlock(const P2::Partitioner::ConstPtr &partitioner, const P2::BasicBlock::Ptr &bblock,
272  std::ostream &out) {
273  ASSERT_not_null(partitioner);
274  out <<" case " <<StringUtility::addrToString(bblock->address()) <<":\n";
275 
276  Address fallThroughVa = 0;
277  raisingOps_->resetState();
278  for (SgAsmInstruction *insn: bblock->instructions()) {
279  emitInstruction(insn, out);
280  fallThroughVa = insn->get_address() + insn->get_size();
281  }
282 
283  // If this bblock is a binary function call, then call the corresponding source function. We can't do this
284  // directly because the call might be indirect. Therefore all calls go through a function call dispatcher.
285  if (partitioner->basicBlockIsFunctionCall(bblock))
286  out <<" function_call();\n";
287 
288  bool needBreak = true;
289  P2::ControlFlowGraph::ConstVertexIterator placeholder = partitioner->findPlaceholder(bblock->address());
290  ASSERT_require(partitioner->cfg().isValidVertex(placeholder));
291  if (placeholder->nOutEdges() == 0) {
292  out <<" assert(!\"not reachable\");\n";
293  needBreak = false;
294  } else if (placeholder->nOutEdges() == 1) {
295  P2::ControlFlowGraph::ConstVertexIterator nextVertex = placeholder->outEdges().begin()->target();
296  if (nextVertex->value().type() == P2::V_BASIC_BLOCK && nextVertex->value().address() == fallThroughVa) {
297  out <<" /* fall through... */\n";
298  needBreak = false;
299  }
300  }
301  if (needBreak)
302  out <<" break;\n";
303 }

342 BinaryToSource::emitFunctionDispatcher(const P2::Partitioner::ConstPtr &partitioner, std::ostream &out) {
343  ASSERT_not_null(partitioner);
344  out <<"\n"
345  <<"void\n"
346  <<"function_call(void) {\n";
347 
348  // To whence does this call return?
349  Architecture::Base::ConstPtr arch = disassembler_->architecture();
350  const RegisterDescriptor IP = arch->registerDictionary()->instructionPointerRegister();
351  const RegisterDescriptor SP = arch->registerDictionary()->stackPointerRegister();
352  const RegisterDescriptor SS = arch->registerDictionary()->stackSegmentRegister();
353  raisingOps_->reset();
354  BaseSemantics::SValue::Ptr spDflt = raisingOps_->undefined_(SP.nBits());
355  BaseSemantics::SValue::Ptr returnTarget = raisingOps_->readMemory(SS,
356  raisingOps_->peekRegister(SP, spDflt),
357  raisingOps_->undefined_(IP.nBits()),
358  raisingOps_->boolean_(true));
359  emitEffects(out);
360  out <<" " <<SValue::unsignedTypeNameForSize(IP.nBits()) <<" returnTarget = "
361  <<SValue::promote(returnTarget)->ctext() <<";\n";
362 
363  // Emit the dispatch table
364  out <<" switch (" <<raisingOps_->registerVariableName(IP) <<") {\n";
365  for (const P2::Function::Ptr &function: partitioner->functions()) {
366  out <<" case " <<StringUtility::addrToString(function->address()) <<": ";
367  out <<" F_" <<StringUtility::addrToString(function->address()).substr(2) <<"(returnTarget); ";
368  out <<" break;";
369  std::string fname = function->name();
370  if (!fname.empty()) {
371  fname = StringUtility::cEscape(fname);
372  fname = boost::replace_all_copy(fname, "*/", "*\\/");
373  out <<" /* " <<fname <<" */";
374  }
375  out <<"\n";
376  }
377  out <<" }\n"
378  <<"}\n";

399  } else {

402  }
403  }
404 
405  // Initialize the memory array with the contents of the memory map
406  Address va = 0;
407  uint8_t buf[8192];
408  while (AddressInterval where = partitioner->memoryMap()->atOrAfter(va).limit(sizeof buf).read(buf)) {
409  uint8_t *bufptr = buf;
