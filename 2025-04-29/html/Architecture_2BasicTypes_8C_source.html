170 static std::tuple<std::vector<std::string>, std::string, std::string>
171 loadSharedLibrary(const std::string &name) {
172  const bool isFileName = name.find('/') != std::string::npos;
173  std::vector<std::string> newArchNames;
174 
175  // Load the library or return an error
176  std::string loadedName;
177  void *so = nullptr;
178  if (isFileName) {
179  so = dlopen(name.c_str(), RTLD_LAZY);
180  if (!so)
181  return {newArchNames, dlerror(), ""};
182  loadedName = name;
183  } else {
184  std::string firstError;
185  for (int i = 0; i < 4 && !so; ++i) {
186  loadedName = nameVariation(name, i);
187  so = dlopen(loadedName.c_str(), RTLD_LAZY);
188  if (!so && 0 == i)
189  firstError = dlerror();
190  }
191  if (!so)
192  return {newArchNames, firstError, ""};
193  }
194 
195  // Call the library's main architecture registration function
196  const std::set<std::string> oldArchNames = registeredNames();
197  ASSERT_not_null(so);
198  if (auto func = reinterpret_cast<void(*)()>(dlsym(so, "registerArchitectures"))) {
199  func();
200  } else {
201  dlclose(so);
202  return {
203  newArchNames,
204  "",
205  "no \"registerArchitecture\" function in \"" + StringUtility::cEscape(loadedName) + "\""
206  };
207  }
208 
209  // Show what architectures were registered
210  const std::set<std::string> allArchNames = registeredNames();
211  std::set_difference(allArchNames.begin(), allArchNames.end(), oldArchNames.begin(), oldArchNames.end(),
212  std::inserter(newArchNames, newArchNames.begin()));
213  if (newArchNames.empty()) {
214  return {
215  newArchNames,
216  "",
217  "library \"" + StringUtility::cEscape(loadedName) + "\" did not define any new architectures\n"
218  };
219  } else {
220  return {newArchNames, "", ""};
221  }
222 }

329 std::pair<Base::Ptr, size_t>
330 findBestByInterpretation(SgAsmInterpretation *interp) {
331  ASSERT_not_null(interp);
332  const SgAsmGenericHeaderPtrList &headers = interp->get_headers()->get_headers();
333 
334  Base::Ptr best; // best matching architecture
335  size_t bestN = 0; // number of matches for best architecture
336 
337  for (const Base::Ptr &arch: registeredDefinitions()) {
338  // How many headers does `arch` match?
339  size_t nMatched = 0;
340  for (SgAsmGenericHeader *header: headers) {
341  if (arch->matchesHeader(header))
342  ++nMatched;
343  }
344 
345  // Return or remember
346  if (nMatched == headers.size()) {
347  return {arch, nMatched};
348  } else if (nMatched > bestN) {
349  best = arch;
350  bestN = nMatched;
351  }
352  }
353 
354  return std::make_pair(best, bestN);
355 }
