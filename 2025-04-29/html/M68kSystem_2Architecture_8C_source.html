102 Sawyer::Optional<Address>
103 Architecture::entryAddress() {
104  ASSERT_not_null(partitioner());
105 
106  // First, try to get the entry address from the file header.
107  if (SgAsmInterpretation *interp = partitioner()->interpretation()) {
108  for (SgAsmGenericHeader *header: interp->get_headers()->get_headers()) {
109  ASSERT_not_null(header);
110  for (const RelativeVirtualAddress &rva: header->get_entryRvas())
111  return header->get_baseVa() + rva.rva();
112  }
113  }
114 
115  // Since file headers are not always present (e.g., firmware is often raw memory dumps without an ELF or PE container), we then
116  // try to look at the interrupt vector for the m68k.
117  if (const auto va = partitioner()->memoryMap()->readUnsigned<uint32_t>(4))
118  return ByteOrder::leToHost(*va);
119 
120  mlog[WARN] <<"M68kSystem::Architecture::entryAddress: unable to find a starting address\n";
121  return Sawyer::Nothing();
122 }
