29 static unsigned
30 daysInMonth(unsigned year, unsigned month) {
31  // Assume this formula is even correct for years before 1583 and after 9999.
32  ASSERT_require(month >= 1 && month <= 12);
33  if (2 == month) {
34  return isLeapYear(year) ? 29 : 28;
35  } else {
36  return 4 == month || 6 == month || 9 == month || 11 == month ? 30 : 31;
37  }
38 }

40 // Conditionally increment a value.
41 // Value CarryIn Return CarryOut
42 // -----------------------------
43 // Exists False Value False
44 // Exists True Value+1 Overflowed?
45 // Absent False zero True
46 // Absent True zero True
47 template<class T>
48 static std::pair<T, bool>
49 increment(const Optional<T> &value, const T maxValue, T base, bool carryIn) {
50  if (value) {
51  if (carryIn) {
52  const T next = *value >= maxValue ? base : *value + T{1};
53  return std::make_pair(next, next < *value);
54  } else {
55  return std::make_pair(*value, false);
56  }
57  } else {
58  const T next = base;
59  return std::make_pair(next, true);
60  }
61 }

91 Result<Time, std::string>
92 Time::parse(const std::string &origStr) {
93  // No template parameter deduction in constructors before C++17, so make aliases
94  using Error = Sawyer::Error<std::string>;
95  using Ok = Sawyer::Ok<Time>;
96 
97  Time t;
98  std::smatch found;
99  std::string str = origStr;
100 
101  // Parse the date part if present
102  std::regex basicDateRe("([0-9]{8})(.*)");
103  if (std::regex_match(str, found /*out*/, basicDateRe)) {
104  // yyyy (handled below)
105  // yyyymm (not allowed by standard)
106  // yyyymmdd
107  const std::string s = found.str(1);
108  ASSERT_require(s.size() == 8);
109  t.year_ = *Sawyer::parse<unsigned>(s.substr(0, 4));
110  t.month_ = *Sawyer::parse<unsigned>(s.substr(4, 2));
111  t.day_ = *Sawyer::parse<unsigned>(s.substr(6, 2));
112  str = found.str(2);
113  } else {
114  std::regex extendedDateRe("([0-9]{4}(-[0-9]{2}(-[0-9]{2})?)?)(.*)");
115  if (std::regex_match(str, found /*out*/, extendedDateRe)) {
116  // yyyy
117  // yyyy-mm
118  // yyyy-mm-dd
119  const std::string s = found.str(1);
120  ASSERT_require(s.size() == 4 || s.size() == 7 || s.size() == 10);
121  t.year_ = *Sawyer::parse<unsigned>(s.substr(0, 4));
122  if (s.size() >= 7) {
123  t.month_ = *Sawyer::parse<unsigned>(s.substr(5, 2));
124  if (s.size() >= 10)
125  t.day_ = *Sawyer::parse<unsigned>(s.substr(8, 2));
126  }
127  str = found.str(4);
128  }
129  }
130 
131  std::regex basicTimeRe1(std::string(t.hasDate() ? " ?T" : "T") + "([0-9]{4}([0-9]{2})?)(.*)");
132  if (std::regex_match(str, found /*out*/, basicTimeRe1)) {
133  // Thh (handled below)
134  // Thhmm
135  // Thhmmss
136  const std::string s = found.str(1);
137  ASSERT_require(s.size() == 4 || s.size() == 6);
138  t.hour_ = *Sawyer::parse<unsigned>(s.substr(0, 2));
139  t.minute_ = *Sawyer::parse<unsigned>(s.substr(2, 2));
140  if (s.size() >= 6)
141  t.second_ = *Sawyer::parse<unsigned>(s.substr(4, 2));
142  str = found.str(3);
143  } else {
144  std::regex extendedTimeRe1(std::string(t.hasDate() ? " ?T" : "T") + "([0-9]{2}(:[0-9]{2}(:[0-9]{2})?)?)(.*)");
145  if (std::regex_match(str, found /*out*/, extendedTimeRe1)) {
146  // Thh
147  // Thh:mm
148  // Thh:mm:ss
149  const std::string s = found.str(1);
150  ASSERT_require(s.size() == 2 || s.size() == 5 || s.size() == 8);
151  t.hour_ = *Sawyer::parse<unsigned>(s.substr(0, 2));
152  if (s.size() >= 5) {
153  t.minute_ = *Sawyer::parse<unsigned>(s.substr(3, 2));
154  if (s.size() >= 8)
155  t.second_ = *Sawyer::parse<unsigned>(s.substr(6, 2));
156  }
157  str = found.str(4);
158  } else {
159  std::regex extendedTimeRe2(std::string(t.hasDate() ? " " : "") + "([0-9]{4}(:[0-9]{2})?)(.*)");
160  if (std::regex_match(str, found /*out*/, extendedTimeRe2)) {
161  // hh (not allowed by standard)
162  // hh:mm
163  // hh:mm:ss
164  const std::string s = found.str(1);
165  ASSERT_require(s.size() == 5 || s.size() == 8);
166  t.hour_ = *Sawyer::parse<unsigned>(s.substr(0, 2));
167  t.minute_ = *Sawyer::parse<unsigned>(s.substr(3, 2));
168  if (s.size() >= 8)
169  t.second_ = *Sawyer::parse<unsigned>(s.substr(6, 2));
170  str = found.str(3);
171  }
172  }
173  }
174 
175  // Parse the time part if present
176  // 1 2 3 4
177  std::regex basicTimeRe("T([0-9]{2}([0-9]{2}([0-9]{2})?)?)(.*)");
178  if (std::regex_match(str, found /*out*/, basicTimeRe)) {
179  // Thhmmss
180  // hhmmss (not allowed)
181  // Thhmm
182  // hhmm (not allowed)
183  // Thh
184  // hh (handled below)
185  const std::string s = found.str(1);
186  ASSERT_require(s.size() == 2 || s.size() == 4 || s.size() == 6);
187  t.hour_ = *Sawyer::parse<unsigned>(s.substr(0, 2));
188  if (s.size() >= 5) {
189  t.minute_ = *Sawyer::parse<unsigned>(s.substr(4, 2));
190  if (s.size() >= 8)
191  t.second_ = *Sawyer::parse<unsigned>(s.substr(4, 2));
192  }
193  str = found.str(4);
194  } else {
195  // 1 1 2 3 4 5
196  std::regex extendedTimeRe(std::string(t.hasDate()?"([ T])":"(T?)") + "([0-9]{2}(:[0-9]{2}(:[0-9]{2})?)?)(.*)");
197  if (std::regex_match(str, found /*out*/, extendedTimeRe)) {
198  // Thh:mm:ss
199  // Thh:mm
200  // Thh
201  // hh:mm:ss
202  // hh:mm
203  // hh
204  const std::string s = found.str(2);
205  ASSERT_require(s.size() == 2 || s.size() == 5 || s.size() == 8);
206  t.hour_ = *Sawyer::parse<unsigned>(s.substr(0, 2));
207  if (s.size() >= 5) {
208  t.minute_ = *Sawyer::parse<unsigned>(s.substr(3, 2));
209  if (s.size() >= 8)
210  t.second_ = *Sawyer::parse<unsigned>(s.substr(6, 2));
211  }
212  str = found.str(5);
213  }
214  }
215 
216  // Parse time zone part if present
217  if (str.empty()) {
218  // no zone
219  } else if (str == "Z") {
220  // same as +0000
221  t.tz_hour_ = 0;
222  t.tz_minute_ = 0;
223  } else {
224  std::regex extendedZoneRe("([-+][0-9]{2}(:[0-9]{2})?)");
225  if (std::regex_match(str, found /*out*/, extendedZoneRe)) {
226  // +hh:mm
227  // -hh:mm
228  // +hh
229  // -hh
230  // -00 (not allowed, use +00)
231  // -00:00 (not allowed, use +00:00)
232  const std::string s = found.str(1);
233  ASSERT_require(s.size() == 3 || s.size() == 6);
234  t.tz_hour_ = *Sawyer::parse<int>(s.substr(0, 3));
235  if (s.size() >= 6)
236  t.tz_minute_ = *Sawyer::parse<int>(s.substr(0, 1) + s.substr(4, 2));
237  if ('-' == s[0] && 0 == *t.tz_hour_ && 0 == t.tz_minute_.orElse(0))
238  return Error("timezone cannot be \"" + s + "\"");
239 
240  } else {
241  std::regex basicZoneRe("([-+][0-9]{4})");
242  if (std::regex_match(str, found /*out*/, basicZoneRe)) {
243  // +hhmm
244  // -hhmm
245  // +hh (handled above)
246  // -hh (handled above)
247  // -0000 (not allowed, use +0000)
248  const std::string s = found.str(1);
249  ASSERT_require(s.size() == 5);
250  t.tz_hour_ = *Sawyer::parse<int>(s.substr(0, 3));
251  t.tz_minute_ = *Sawyer::parse<int>(s.substr(0, 1) + s.substr(3, 2));
252  if ("-0000" == s)
253  return Error("timezone cannot be \"-0000\"");
254  } else {
255  return Error("invalid time specification \"" + origStr + "\"");
256  }
257  }
258  }
259 
260  // Check ranges
261  if (t.year_.orElse(1900) < 1583)
262  return Error("year must be 1583 or later in \"" + origStr + "\"");
263  if (t.month_ && (*t.month_ < 1 || *t.month_ > 12))
264  return Error("month is out of range in \"" + origStr + "\"");
265  if (t.day_) {
266  const unsigned m = daysInMonth(*t.year_, *t.month_);
267  if (*t.day_ < 1 || *t.day_ > m)
268  return Error("day of month is out of range in \"" + origStr + "\"");
269  }
270  if (t.hour_.orElse(0) > 23)
271  return Error("hour is out of range in \"" + origStr + "\"");
272  if (t.minute_.orElse(0) > 59)
273  return Error("minute is out of range in \"" + origStr + "\"");
274  if (t.second_.orElse(0) > 60)
275  return Error("second is out of range in \"" + origStr + "\"");
276  if (t.tz_hour_ && (*t.tz_hour_ < -23 || *t.tz_hour_ > 23))
277  return Error("timezone hour is out of range in \"" + origStr + "\"");
278  if (t.tz_minute_ && (*t.tz_minute_ < -59 || *t.tz_minute_ > 59))
279  return Error("timezone minute is out of range in \"" + origStr + "\"");
280 
281  return Ok(t);
282 }

354 Result<Time, std::string>
355 Time::upperBound() const {
356  // No template parameter deduction in constructors before C++17, so make aliases
357  using Error = Sawyer::Error<std::string>;
358  using Ok = Sawyer::Ok<Time>;
359 
360  Time t;
361 
362  // Update the time part if present.
363  // 12:30:45 -> 12:30:46
364  // 12:30:59 -> 12:31:00
365  // 12:30:60 -> 12:31:00 (leap second)
366  //
367  // 12:30 -> 12:31:00
368  // 12:59 -> 13:00:00
369  // 23:59 -> 00:00:00 (and increment date if present)
370  //
371  // 12 -> 13:00:00
372  // 23 -> 00:00:00 (and increment date if present)
373  const auto seconds = increment(second_, 59u, 0u, true);
374  t.second_ = seconds.first;
375 
376  const auto minutes = increment(minute_, 59u, 0u, seconds.second);
377  t.minute_ = minutes.first;
378 
379  const auto hours = increment(hour_, 23u, 0u, minutes.second);
380  t.hour_ = hours.first;
381 
382  // Update the date part if present
383  if (hasDate()) {
384  const auto days = day_ ?
385  increment(day_, daysInMonth(*year_, *month_), 1u, hours.second) :
386  std::make_pair(1u, true);
387  t.day_ = days.first;
388 
389  const auto months = increment(month_, 12u, 1u, days.second);
390  t.month_ = months.first;
391 
392  const auto years = increment(year_, 9999u, 0u, months.second);
393  t.year_ = years.first;
394  if (years.second)
395  return Error("cannot represent years greater than 9999");
396  }
397 
398  t.tz_hour_ = tz_hour_;
399  t.tz_minute_ = tz_minute_;
400  return Ok(t);
401 }

477 // Normalize seconds by borrowing from or carrying to the minutes
478 void
479 Time::normalizeSecond() {
480  if (second_ && (int)*second_ < 0) {
481  ASSERT_require(minute_);
482  minute_ = (int)*minute_ + (int)*second_ / 60 - 1;
483  second_ = 60 + (int)*second_ % 60;
484  } else if (second_.orElse(0) >= 60) {
485  ASSERT_require(minute_);
486  minute_ = *minute_ + *second_ / 60;
487  second_ = *second_ % 60;
488  }
489  ASSERT_require(second_.orElse(0) < 60);
490 }

494 Time::normalizeMinute() {

507 // Normalize hours by borrowing from or carrying to the day (if any)
508 void
509 Time::normalizeHour() {
510  if (hour_ && (int)*hour_ < 0) {
511  if (day_)
512  day_ = (int)*day_ + (int)*hour_ / 24 - 1;
513  hour_ = 24 + (int)*hour_ % 24;
514  } else if (hour_.orElse(0) >= 24) {
515  if (day_)
516  day_ = *day_ + *hour_ / 24;
517  hour_ = *hour_ % 24;
518  }
519  ASSERT_require(hour_.orElse(0) < 24);

522 // Normalize the month by borrowing from or carrying to the year.
523 void
524 Time::normalizeMonth() {
525  if (month_ && (int)*month_ < 1) {
526  const int zmonth = (int)*month_ - 1;
527  ASSERT_require(year_);
528  year_ = (int)*year_ + zmonth / 12 - 1;
529  month_ = 13 + zmonth % 12;
530  } else if (month_.orElse(0) > 12) {
531  const int zmonth = *month_ - 1;
532  ASSERT_require(year_);
533  year_ = *year_ + zmonth / 12;
534  month_ = zmonth % 12 + 1;
535  }
536  ASSERT_require(month_.orElse(1) >= 1 && month_.orElse(1) <= 12);
537 }

539 // Normalize date and/or time. After calling this, the year might be out of range and should be checked.
540 void
541 Time::normalize() {
542  normalizeSecond();
543  normalizeMinute();
544  normalizeHour();
545 
546  // Normalize days
547  if (day_) {
548  ASSERT_require(month_);
549  ASSERT_require(year_);
550 
551  // Normalize negative days by borrowing from the month
552  while ((int)*day_ < 1) {
553  month_ = *month_ - 1;
554  normalizeMonth();
555  day_ = *day_ + daysInMonth(*year_, *month_);
556  }
557 
558  // Normalize overflow days by carrying to the month
559  while (true) {
560  normalizeMonth();
561  const unsigned dim = daysInMonth(*year_, *month_);
562  if (*day_ > dim) {
563  day_ = *day_ - dim;
564  month_ = *month_ + 1;
565  } else {
566  break;
567  }
568  }
569  } else {
570  normalizeMonth();
571  }
572 }

574 Result<Time, std::string>
575 Time::toZulu() const {
576  using Ok = Sawyer::Ok<Time>;
577  using Error = Sawyer::Error<std::string>;
578 
579  if (!hasTime() || !hasZone())
580  return Ok(*this);
581 
582  Time result = *this;
583  result.tz_minute_ = tz_minute_.andThen(0);
584  result.tz_hour_ = 0;
585 
586  if (result.minute_ && tz_minute_)
587  result.minute_ = (int)*result.minute_ - *tz_minute_;
588  if (result.hour_ && tz_hour_)
589  result.hour_ = (int)*result.hour_ - *tz_hour_;
590 
591  result.normalize();
592  if (result.year_.orElse(1583) < 1583)
593  return Error("cannot represent years before 1583");
594  if (result.year_.orElse(9999) > 9999)
595  return Error("cannot represent years after 9999");
596 
597  return Ok(result);
598 }

600 std::string
601 Time::toString() const {
602  std::string s;
603 
604  // Date
605  if (year_)
606  s += (boost::format("%04d") % *year_).str();
607  if (month_)
608  s += (boost::format("-%02d") % *month_).str();
609  if (day_)
610  s += (boost::format("-%02d") % *day_).str();
611 
612  // Date-time separator
613  if (hasDate() && hasTime()) {
614  if (!month_ && !minute_) {
615  s += "T";
616  } else if (!minute_) {
617  s += " T";
618  } else {
619  s += " ";
620  }
621  } else if (hasTime() && !minute_) {
622  s += "T";
623  }
624 
625  // Time
626  if (hour_)
627  s += (boost::format("%02d") % *hour_).str();
628  if (minute_)
629  s += (boost::format(":%02d") % *minute_).str();
630  if (second_)
631  s += (boost::format(":%02d") % *second_).str();
632 
633  // Zone
634  if (tz_hour_) {
635  if (0 == *tz_hour_ && tz_minute_ && 0 == *tz_minute_) {
636  s += "Z";
637  } else {
638  if (0 == *tz_hour_) {
639  s += tz_minute_ && *tz_minute_ < 0 ? "-00" : "+00";
640  } else {
641  s += (boost::format("%+03d") % *tz_hour_).str();
642  }
643  if (tz_minute_)
644  s += (boost::format(":%02d") % (*tz_minute_ < 0 ? -*tz_minute_ : *tz_minute_)).str();
645  }
646  }
647 
648  return s;
649 }

651 Result<time_t, std::string>
652 Time::toUnix() const {
653  // No template parameter deduction in constructors before C++17, so make aliases
654  using Error = Sawyer::Error<std::string>;
655  using Ok = Sawyer::Ok<time_t>;
656 
657  if (!hasSpecificTime())
658  return Error("cannot convert non-specific time to time_t");
659  if (*year_ < 1970)
660  return Error("time point is not representable as a time_t value");
661 
662  // Full years since 1970
663  const unsigned long nYears = *year_ - 1970;
664 
665  // Number of leap days since 1970-01-01.
666  const unsigned long nLeapDays = (*year_ / 4) - (*year_ / 100) + (*year_ / 400) - 477 -
667  (isLeapYear(*year_) && *month_ <= 2 ? 1 : 0); // don't count this year yet?
