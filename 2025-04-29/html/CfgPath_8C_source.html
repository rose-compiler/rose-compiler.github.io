440 findPathEdges(const ControlFlowGraph &graph, const CfgConstVertexSet &beginVertices,
441  const CfgConstVertexSet &avoidVertices, const CfgConstEdgeSet &avoidEdges,
442  bool avoidCallsAndReturns) {
443  using namespace Sawyer::Container::Algorithm;
444 
445  // Find all edges that are reachable with a forward traversal from any starting vertex, avoiding certain vertices and edges.
446  std::vector<bool> retval(graph.nEdges(), false);
447  for (const ControlFlowGraph::ConstVertexIterator &beginVertex: beginVertices.values()) {
448  ASSERT_require(graph.isValidVertex(beginVertex));
449  typedef DepthFirstForwardGraphTraversal<const ControlFlowGraph> ForwardTraversal;
450  for (ForwardTraversal t(graph, beginVertex, ENTER_EVENTS); t; ++t) {
451  switch (t.event()) {
452  case ENTER_VERTEX:
453  if (avoidVertices.exists(t.vertex()))
454  t.skipChildren();
455  break;
456  case ENTER_EDGE:
457  if (avoidCallsAndReturns &&
458  (t.edge()->value().type() == E_FUNCTION_CALL || t.edge()->value().type() == E_FUNCTION_RETURN)) {
459  t.skipChildren();
460  } else if (avoidEdges.exists(t.edge())) {
461  t.skipChildren();
462  } else {
463  retval[t.edge()->id()] = true;
464  }
465  break;
466  default:
467  break;
468  }
469  }
470  }
471  return retval;
472 }

544 size_t
545 eraseUnreachablePaths(ControlFlowGraph &graph /*in,out*/, CfgConstVertexSet &beginVertices /*in,out*/,
546  CfgConstVertexSet &endVertices /*in,out*/, CfgVertexMap &vmap /*in,out*/, CfgPath &path /*in,out*/) {
547  size_t origPathSize = path.nEdges();
548  if (beginVertices.empty() || endVertices.empty()) {
549  graph.clear();
550  vmap.clear();
551  path.clear();
552  return origPathSize;
553  }
554 
555  // Erase unreachable edges from the graph and path
556  CfgConstVertexSet avoidVertices;
557  CfgConstEdgeSet avoidEdges;
558  CfgConstEdgeSet badEdges = findPathUnreachableEdges(graph, beginVertices, endVertices, avoidVertices, avoidEdges, false);
559  CfgConstVertexSet incidentVertices = findIncidentVertices(badEdges);
560  path.truncate(badEdges);
561  eraseEdges(graph, badEdges);
562 
563  // This might leave some vertices having no incident edges, so remove them since they can't participate on any path. Avoid
564  // scanning the entire graph by considering only vertices that are incident to the edges we just removed. We also need to
565  // check the beginVertices and endVertices because they might not have had incident edges, yet we'll wwant to remove them.
566  CfgConstVertexSet badVertices;
567  for (const ControlFlowGraph::ConstVertexIterator &vertex: incidentVertices.values()) {
568  if (vertex->degree() == 0 &&
569  (!beginVertices.exists(vertex) || !endVertices.exists(vertex)))
570  badVertices.insert(vertex);
571  }
572  for (const ControlFlowGraph::ConstVertexIterator &vertex: beginVertices.values()) {
573  if (vertex->degree() == 0 && !endVertices.exists(vertex))
574  badVertices.insert(vertex);
575  }
576  for (const ControlFlowGraph::ConstVertexIterator &vertex: endVertices.values()) {
577  if (vertex->degree() == 0 && !beginVertices.exists(vertex))
578  badVertices.insert(vertex);
579  }
580  for (const ControlFlowGraph::ConstVertexIterator &vertex: badVertices.values()) {
581  beginVertices.erase(vertex);
582  endVertices.erase(vertex);
583  vmap.eraseTarget(vertex);
584  graph.eraseVertex(vertex);
585  }
586 
587  if (beginVertices.empty())
588  path.clear();
589 
590  return origPathSize - path.nEdges();
591 }

657  !avoidVertices.exists(beginVertex)) { // not an avoided vertex

707 bool
708 inlineOneCallee(ControlFlowGraph &paths /*in,out*/, const ControlFlowGraph::ConstVertexIterator &pathsCallSite,
709  const ControlFlowGraph &cfg, const ControlFlowGraph::ConstVertexIterator &cfgCallTarget,
710  const CfgConstVertexSet &cfgAvoidVertices, const CfgConstEdgeSet &cfgAvoidEdges,
711  std::vector<ControlFlowGraph::ConstVertexIterator> *newVertices /*=NULL*/) {
712  ASSERT_require(paths.isValidVertex(pathsCallSite));
713  ASSERT_require(cfg.isValidVertex(cfgCallTarget));
714  ASSERT_require2(pathsCallSite->value().type() == V_BASIC_BLOCK, "only basic blocks can call functions");
715 
716  // Most functions either don't return or have a single return target. Functions like longjmp might have multiple return
717  // targets. We need to know the return targets in the paths graph because we will create new edges from the inlined
718  // function's return sites to each return target.
719  CfgConstVertexSet pathsReturnTargets;
720  for (const ControlFlowGraph::Edge &edge: pathsCallSite->outEdges()) {
721  if (edge.value().type() == E_CALL_RETURN)
722  pathsReturnTargets.insert(edge.target());
723  }
724 
725  // If this is a call to some indeterminate function, just copy another indeterminate vertex into the paths graph. Normally
726  // a CFG will have only one indeterminate vertex and it will have no outgoing edges, but the paths graph is different. We
727  // need separate indeterminate vertices so that each has its own function-return edge(s).
728  if (cfgCallTarget->value().type() == V_INDETERMINATE) {
729  ControlFlowGraph::ConstVertexIterator indet = paths.insertVertex(CfgVertex(V_INDETERMINATE));
730  if (newVertices)
731  newVertices->push_back(indet);
732  paths.insertEdge(pathsCallSite, indet, CfgEdge(E_FUNCTION_CALL));
733  for (const ControlFlowGraph::ConstVertexIterator &returnTarget: pathsReturnTargets.values())
734  paths.insertEdge(indet, returnTarget, CfgEdge(E_FUNCTION_RETURN));
735  SAWYER_MESG(mlog[DEBUG]) <<"insertCalleePaths: indeterminate function call from "
736  <<pathsCallSite->value().bblock()->printableName() <<"\n";
737  return true;
738  }
739 
740  // Call to a normal function.
741  ASSERT_require2(cfgCallTarget->value().type() == V_BASIC_BLOCK, "non-basic block callees not implemented yet");
742 
743  // Find all paths through the callee that return and avoid certain vertices and edges.
744  CfgVertexMap vmap1; // relates CFG to calleePaths
745  CfgConstVertexSet cfgCallTargets; cfgCallTargets.insert(cfgCallTarget);
746  CfgConstVertexSet cfgReturns = findFunctionReturns(cfg, cfgCallTarget);
747  ControlFlowGraph calleePaths;
748  findFunctionPaths(cfg, calleePaths, vmap1, cfgCallTargets, cfgReturns, cfgAvoidVertices, cfgAvoidEdges);
749  if (calleePaths.isEmpty()) {
750  if (mlog[DEBUG]) {
751  std::string calleePrintableName;
752  if (cfgCallTarget->value().nOwningFunctions() > 1) {
753  for (const Function::Ptr &function: cfgCallTarget->value().owningFunctions().values()) {
754  if (!function->name().empty()) {
755  calleePrintableName = function->printableName();
756  break;
757  }
758  }
759  } else if (1 == cfgCallTarget->value().nOwningFunctions()) {
760  calleePrintableName = (*cfgCallTarget->value().owningFunctions().values().begin())->printableName();
761  } else {
762  calleePrintableName = cfgCallTarget->value().bblock()->printableName();
763  }
764  mlog[DEBUG] <<"insertCalleePaths: " <<calleePrintableName <<" has no paths to insert\n";
765  }
766  return false;
767  }
768 
769  // Insert the callee into the paths CFG
770  CfgVertexMap vmap2; // relates calleePaths to paths
771  insertCfg(paths, calleePaths, vmap2);
772  if (newVertices) {
773  for (ControlFlowGraph::ConstVertexIterator vertex: vmap2.forward().values()) {
774  ASSERT_require(paths.isValidVertex(vertex));
775  newVertices->push_back(vertex);
776  }
777  }
778  CfgVertexMap vmap(vmap1, vmap2); // composite map from the CFG to paths graph
779 
780  // Make an edge from call site to the entry block of the callee in the paths graph
781  if (vmap.forward().exists(cfgCallTarget)) {
782  ControlFlowGraph::ConstVertexIterator pathStart = vmap.forward()[cfgCallTarget];
783  paths.insertEdge(pathsCallSite, pathStart, CfgEdge(E_FUNCTION_CALL));
784  }
785 
786  // Make edges from the callee's return statements back to the return targets in the caller
787  for (const ControlFlowGraph::ConstVertexIterator &cfgReturnSite: cfgReturns.values()) {
788  if (vmap.forward().exists(cfgReturnSite)) {
789  ControlFlowGraph::ConstVertexIterator pathsReturnSite = vmap.forward()[cfgReturnSite];
790  for (const ControlFlowGraph::ConstVertexIterator &pathsReturnTarget: pathsReturnTargets.values())
791  paths.insertEdge(pathsReturnSite, pathsReturnTarget, CfgEdge(E_FUNCTION_RETURN));
792  }
793  }
794  return true;
795 }

869 void
870 Inliner::inlinePaths(const Partitioner::ConstPtr &partitioner, const CfgConstVertexSet &cfgBeginVertices,
871  const CfgConstVertexSet &cfgEndVertices, const CfgConstVertexSet &cfgAvoidVertices,
872  const CfgConstEdgeSet &cfgAvoidEdges) {
873  ASSERT_not_null(partitioner);
874  reset(partitioner, cfgBeginVertices, cfgEndVertices, cfgAvoidVertices, cfgAvoidEdges);
875 
876  // When finding paths through a called function, avoid the usual vertices and edges, but also avoid those vertices that
877  // mark the end of paths. We want paths that go all the way from the entry block of the called function to its returning
878  // blocks.
879  CfgConstVertexSet calleeCfgAvoidVertices = cfgAvoidVertices;
880  calleeCfgAvoidVertices.insert(cfgEndVertices);
881 
882  while (!workList_.empty()) {
883  CallSite work = workList_.back();
884  workList_.pop_back();
885  ControlFlowGraph::ConstVertexIterator cfgVertex = pathToCfg(partitioner, work.pathsVertex);
886  CfgConstEdgeSet callEdges = findCallEdges(cfgVertex);
887  for (const ControlFlowGraph::ConstEdgeIterator &cfgCallEdge: callEdges.values()) {
888  HowInline how = (*shouldInline_)(partitioner, cfgCallEdge, paths_, work.pathsVertex, work.callDepth);
889  switch (how) {
890  case INLINE_NONE:
891  break;
892  case INLINE_NORMAL: {
893  std::vector<ControlFlowGraph::ConstVertexIterator> insertedVertices;
894  inlineOneCallee(paths_, work.pathsVertex, partitioner->cfg(), cfgCallEdge->target(),
895  calleeCfgAvoidVertices, cfgAvoidEdges, &insertedVertices);
896  for (const ControlFlowGraph::ConstVertexIterator &vertex: insertedVertices) {
897  if (isFunctionCall(partitioner, vertex) && !findCallReturnEdges(vertex).empty())
898  workList_.push_back(CallSite(vertex, work.callDepth+1));
899  }
900  break;
901  }
902  case INLINE_USER: {
903  CfgVertex userdef(V_USER_DEFINED);
904  ControlFlowGraph::ConstVertexIterator callee = cfgCallEdge->target();
905  if (callee->value().type() == V_BASIC_BLOCK)
906  userdef.address(callee->value().address());
907  ControlFlowGraph::VertexIterator userVertex = paths_.insertVertex(userdef);
908  paths_.insertEdge(work.pathsVertex, userVertex);
909  CfgConstEdgeSet callReturnEdges = findCallReturnEdges(work.pathsVertex);
910  for (const ControlFlowGraph::ConstEdgeIterator &cre: callReturnEdges.values())
911  paths_.insertEdge(userVertex, cre->target(), E_FUNCTION_RETURN);
912  break;
913  }
914  }
915  }
916  eraseEdges(paths_, findCallReturnEdges(work.pathsVertex));
917  }
918 
919  // Remove edges and vertices that cannot be part of any path
920  CfgConstEdgeSet pathsUnreachableEdges = findPathUnreachableEdges(paths_, pathsBeginVertices_, pathsEndVertices_);
921  CfgConstVertexSet pathsIncidentVertices = findIncidentVertices(pathsUnreachableEdges);
922  CfgConstVertexSet pathsBadVertices;
923  eraseEdges(paths_, pathsUnreachableEdges);
924  for (const ControlFlowGraph::ConstVertexIterator &pathVertex: pathsIncidentVertices.values()) {
925  if (pathVertex->degree() == 0 && (!pathsBeginVertices_.exists(pathVertex) || !pathsEndVertices_.exists(pathVertex)))
926  pathsBadVertices.insert(pathVertex);
927  }
928  for (const ControlFlowGraph::ConstVertexIterator &pathVertex: pathsBeginVertices_.values()) {
929  if (pathVertex->degree() == 0 && !pathsEndVertices_.exists(pathVertex))
930  pathsBadVertices.insert(pathVertex);
931  }
932  for (const ControlFlowGraph::ConstVertexIterator &pathVertex: pathsEndVertices_.values()) {
933  if (pathVertex->degree() == 0 && !pathsBeginVertices_.exists(pathVertex))
934  pathsBadVertices.insert(pathVertex);
935  }
936  for (const ControlFlowGraph::ConstVertexIterator &pathVertex: pathsBadVertices.values()) {
937  pathsBeginVertices_.erase(pathVertex);
938  pathsEndVertices_.erase(pathVertex);
939  vmap_.eraseTarget(pathVertex);
940  paths_.eraseVertex(pathVertex);
941  }
942 }
