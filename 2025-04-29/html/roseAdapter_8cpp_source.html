375  // Go through all parent scopes until reach SgSourceFile. store them into a list
376  // iterate on the list to generate all abstract handles, with optional the parent handle information
377  abstract_handle * buildAbstractHandle(SgNode* snode)
378  {
379  ROSE_ASSERT (snode != NULL);
380  // simple nodes
381  if (isSgSourceFile (snode) || isSgProject(snode) ||isSgGlobal(snode))
382  return buildSingleAbstractHandle (snode);
383 
384  // all other nodes, trace back to SgGlobal, store all intermediate scope nodes
385  std::vector<SgNode*> scope_list;
386  SgScopeStatement* p_scope = SageInterface::getEnclosingScope(snode);
387  while (!isSgGlobal(p_scope))

389  scope_list.push_back(p_scope);
390  p_scope = SageInterface::getEnclosingScope(p_scope);
391  }
392 
393  ROSE_ASSERT (isSgGlobal(p_scope));
394  abstract_handle * p_handle = buildSingleAbstractHandle (p_scope);
395 
396  // Now go through the list to generate numbering handles
397  std::vector<SgNode*>::reverse_iterator riter;
398  for (riter = scope_list.rbegin(); riter!= scope_list.rend(); riter++)
399  {
