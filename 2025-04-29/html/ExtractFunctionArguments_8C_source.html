56 bool isVariableReference(SgExpression* expression)
57 {
58  if (isSgVarRefExp(expression))
59  {
60  return true;
61  }
62  else if (isSgThisExp(expression))
63  {
64  return true;
65  }
66  else if (isSgDotExp(expression))
67  {
68  SgDotExp* dotExpression = isSgDotExp(expression);
69  return isVariableReference(dotExpression->get_lhs_operand()) &&
70  isVariableReference(dotExpression->get_rhs_operand());
71  }
72  else if (isSgArrowExp(expression))
73  {
74  SgArrowExp* arrowExpression = isSgArrowExp(expression);
75  return isVariableReference(arrowExpression->get_lhs_operand()) &&
76  isVariableReference(arrowExpression->get_rhs_operand());
77  }
78  else if (isSgCommaOpExp(expression))
79  {
80  //Comma op where both the lhs and th rhs are variable references.
81  //The lhs would be semantically meaningless since it doesn't have any side effects
82  SgCommaOpExp* commaOp = isSgCommaOpExp(expression);
83  return isVariableReference(commaOp->get_lhs_operand()) &&
84  isVariableReference(commaOp->get_rhs_operand());
85  }
86  else if (isSgPointerDerefExp(expression) || isSgCastExp(expression) || isSgAddressOfOp(expression))
87  {
88  return isVariableReference(isSgUnaryOp(expression)->get_operand());
89  }
90  else
91  {
92  return false;
93  }
94 }
