55 DGraph::create_DFS_links (BaseGraph::Node* n)
56 {
57  SinkNodesIterator neigh(dynamic_cast<Node*>(n));
58  BaseGraph::Node* last = n;
59  while ((bool)neigh) {
60  if (neigh->dfs_succ == 0) {

64  ++neigh;
65  }

126 void
127 DGraph::remove (DGraph::Node* n)
128 {
129  BaseGraph::remove(n);
130 
131  IncomingEdgesIterator incoming(n);
132  while ((bool)incoming) {
133  Edge* e = incoming;
134  // remove this edge from the list of outgoing edges of the source node
135  e->source()->outgoing_edges.remove(e);
136  // remove this edge
137  BaseGraph::remove(e);
138  delete e; // an edge has no existence without its anchoring nodes
139  ++incoming;
140  }
141 
142  OutgoingEdgesIterator outgoing(n);
143  while ((bool)outgoing) {
144  Edge* e = outgoing;
145  // remove this edge from the list of outgoing edges of the source node
146  e->sink()->incoming_edges.remove(e);
147  // remove this edge
148  BaseGraph::remove(e);
149  delete e; // an edge has no existence without its anchoring nodes
150  ++outgoing;
151  }
152 }
