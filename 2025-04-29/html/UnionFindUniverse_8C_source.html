113 void UnionFindUniverse::Union(int i, int j, int k)
114 {
115  if ((i == j) && (j == k))
116  return;
117 
118  int large, small;
119  if (Count(Root(i)) > Count(Root(j))) {
120  large = Root(i);
121  small = Root(j);
122  } else {
123  large = Root(j);
124  small = Root(i);
125  }
126 
127  Parent(small) = large;
128  Count(large) += Count(small);
129  Name(large) = k;
130  Root(k) = large;
131 }
