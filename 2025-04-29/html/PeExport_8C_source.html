162 SgAsmPEExportSection::parse()
163 {
164  SgAsmPESection::parse();
165 
166  SgAsmPEFileHeader *fhdr = dynamic_cast<SgAsmPEFileHeader*>(get_header());
167  ROSE_ASSERT(fhdr!=NULL);
168 
169  p_exportDirectory = new SgAsmPEExportDirectory(this);
170 
171  // Check that the p_export_dir.p_nameptr_n is not out of range.
172  Address availBytes = fhdr->get_loaderMap()->at(*get_exportDirectory()->get_nameptr_rva().va()).available().size();
173  size_t availElmts = availBytes / sizeof(ExportNamePtr_disk);
174  if (get_exportDirectory()->get_nameptr_n() > availElmts) {
175  mlog[ERROR] <<"SgAsmPEExportSection::parse: number of entries indicated (" <<get_exportDirectory()->get_nameptr_n() <<")"
176  <<" exceeds available mapped memory (room for " <<StringUtility::plural(availElmts, "entries") <<")\n";
177  }
178  static const size_t maxNamePtrN = 10000;
179  if (get_exportDirectory()->get_nameptr_n() > maxNamePtrN) {
180  availElmts = std::min(availElmts, maxNamePtrN);
181  mlog[WARN] <<"SgAsmPEExportSection::parse: number of entries indicated (" <<get_exportDirectory()->get_nameptr_n() <<")"
182  <<" is large; resetting to " <<availElmts <<"\n";
183  }
184 
185  /* The export directory points to three parallel arrays:
186  * 1. An array of RVAs that point to NUL-terminated names.
187  * 2. An array of "ordinals" which serve as indices into the Export Address Table.
188  * 3. An array of export addresses (see note below). */
189  for (size_t i=0; i<std::min(get_exportDirectory()->get_nameptr_n(), availElmts); i++) {
190  /* Function name RVA (nameptr)*/
191  ExportNamePtr_disk nameptr_disk = 0;
192  Address nameptr_va = *get_exportDirectory()->get_nameptr_rva().va() + i*sizeof(nameptr_disk);
193  bool badFunctionNameVa = false;
194  try {
195  readContent(fhdr->get_loaderMap(), nameptr_va, &nameptr_disk, sizeof nameptr_disk);
196  } catch (const MemoryMap::NotMapped &e) {
197  badFunctionNameVa = true;
198  if (mlog[ERROR]) {
199  mlog[ERROR] <<"SgAsmPEExportSection::parse: export name #" <<i
200  <<" at va " <<StringUtility::addrToString(nameptr_va)
201  <<" contains unmapped va " <<StringUtility::addrToString(e.va) <<"\n";
202  if (e.map) {
203  mlog[ERROR] <<"Memory map in effect at time of error:\n";
204  e.map->dump(mlog[ERROR], " ");
205  }
206  }
207  nameptr_disk = 0;
208  }
209  Address fname_rva = ByteOrder::leToHost(nameptr_disk);
210  Address fname_va = fname_rva + fhdr->get_baseVa();
211 
212  /* Function name (fname) */
213  std::string s;
214  try {
215  s = readContentString(fhdr->get_loaderMap(), fname_va);
216  } catch (const MemoryMap::NotMapped &e) {
217  if (mlog[ERROR]) {
218  mlog[ERROR] <<"SgAsmPEExportSection::parse: export name #" <<i
219  <<" at va " <<StringUtility::addrToString(fname_va)
220  <<" contains unmapped va " <<StringUtility::addrToString(e.va) <<"\n";
221  if (e.map) {
222  mlog[ERROR] <<"Memory map in effect at time of error:\n";
223  e.map->dump(mlog[ERROR], " ");
224  }
225  }
226  }
227  SgAsmGenericString *fname = new SgAsmBasicString(s);
228 
229  /* Ordinal (an index into the Export Address Table) */
230  ExportOrdinal_disk ordinal_disk = 0;
231  Address ordinal_va = *get_exportDirectory()->get_ordinals_rva().va() + i*sizeof(ordinal_disk);
232  bool badOrdinalVa = false;
233  try {
234  readContent(fhdr->get_loaderMap(), ordinal_va, &ordinal_disk, sizeof ordinal_disk);
235  } catch (const MemoryMap::NotMapped &e) {
236  badOrdinalVa = true;
237  if (mlog[ERROR]) {
238  mlog[ERROR] <<"SgAsmPEExportSection::parse: ordinal #" <<i
239  <<" at va " <<StringUtility::addrToString(ordinal_va)
240  <<" contains unmapped va " <<StringUtility::addrToString(e.va) <<"\n";
241  if (e.map) {
242  mlog[ERROR] <<"Memory map in effect at time of error:\n";
243  e.map->dump(mlog[ERROR], " ");
244  }
245  }
246  ordinal_disk = 0;
247  }
248  const unsigned ordinal = ByteOrder::leToHost(ordinal_disk);
249 
250  /* If the function name pointer and ordinal pointer are both in unmapped memory then give up. */
251  if (badFunctionNameVa && badOrdinalVa) {
252  if (mlog[ERROR])
253  mlog[ERROR] <<"SgAsmPEExportSection::parse: giving up after trying to read entry #" <<i <<"\n";
254  break;
255  }
256 
257  // Read the address from the Export Address Table. This table is indexed by ordinal.
258  RelativeVirtualAddress expaddr = 0; // export address
259  const Address expaddr_va = *get_exportDirectory()->get_expaddr_rva().va() + ordinal * sizeof(ExportAddress_disk);
260  try {
261  ExportAddress_disk expaddr_disk;
262  readContent(fhdr->get_loaderMap(), expaddr_va, &expaddr_disk, sizeof expaddr_disk);
263  expaddr = ByteOrder::leToHost(expaddr_disk);
264  expaddr.bindBestSection(fhdr);
265  } catch (const MemoryMap::NotMapped &e) {
266  if (mlog[ERROR]) {
267  mlog[ERROR] <<"SgAsmPEExportSection::parse: export address #" <<i
268  <<" at va " <<StringUtility::addrToString(expaddr_va)
269  <<" contains unmapped va " <<StringUtility::addrToString(e.va) <<"\n";
270  if (e.map) {
271  mlog[ERROR] <<"Memory map in effect at time of error:\n";
272  e.map->dump(mlog[ERROR], " ");
273  }
274  }
275  expaddr = 0xffffffff; // ordinal out of range
276  }
277 
278  /* If export address is within this section then it points to a NUL-terminated forwarder name.
279  * FIXME: Is this the proper precondition? [RPM 2009-08-20] */
280  SgAsmGenericString *forwarder = NULL;
281  if (*expaddr.va()>=get_mappedActualVa() && *expaddr.va()<get_mappedActualVa()+get_mappedSize()) {
282  std::string s;
283  try {
284  s = readContentString(fhdr->get_loaderMap(), *expaddr.va());
285  } catch (const MemoryMap::NotMapped &e) {
286  if (mlog[ERROR]) {
287  mlog[ERROR] <<"SgAsmPEExportSection::parse: forwarder " <<i
288  <<" at rva " <<StringUtility::addrToString(expaddr.rva())
289  <<" contains unmapped va " <<StringUtility::addrToString(e.va) <<"\n";
290  if (e.map) {
291  mlog[ERROR] <<"Memory map in effect at time of error:\n";
292  e.map->dump(mlog[ERROR], " ");
293  }
294  }
295  }
296  forwarder = new SgAsmBasicString(s);
297  }
298 
299  SgAsmPEExportEntry *entry = new SgAsmPEExportEntry(fname, ordinal, expaddr, forwarder);
300  addEntry(entry);
301  }
302  return this;
303 }
