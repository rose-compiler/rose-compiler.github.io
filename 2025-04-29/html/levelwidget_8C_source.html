118 void LevelWidget::paintEvent( QPaintEvent* /*e*/ )
119 {
120  buffer.fill( /*backgroundColor()*/ );
121 
122  Q3TextParagraph *p = m_editor->getDocument()->firstParagraph();
123  QPainter painter( &buffer );
124  int yOffset = m_editor->contentsY();
125  while ( p ) {
126  if ( !p->isVisible() ) {
127  p = p->next();
128  continue;
129  }
130  if ( p->rect().y() + p->rect().height() - yOffset < 0 ) {
131  p = p->next();
132  continue;
133  }
134  if ( p->rect().y() - yOffset > height() )
135  break;
136 
137  ParagData* data = (ParagData*) p->extraData();
138  if( data ){
139 
140  int prevLevel = 0;
141  if( p->prev() ){
142  prevLevel = ((ParagData*) p->prev()->extraData())->level();
143  }
144 
145  if( data->isBlockStart() ){
146  if( data->isOpen() ){
147  painter.drawLine( minusPixmap->width() / 2 + 2,
148  p->rect().y() + p->rect().height() - yOffset,
149  minusPixmap->width() / 2 - 2,
150  p->rect().y() + p->rect().height() - yOffset );
151 
152  painter.drawPixmap( 0, p->rect().y() +
153  ( p->rect().height() - minusPixmap->height() ) / 2 -
154  yOffset, *minusPixmap );
155  } else {
156  painter.drawPixmap( 0, p->rect().y() +
157  ( p->rect().height() - plusPixmap->height() ) / 2 -
158  yOffset, *plusPixmap );
159  }
160  } else if( data->level() < prevLevel ){
161  painter.drawLine( plusPixmap->width() / 2, p->rect().y() - yOffset,
162  plusPixmap->width() / 2, p->rect().y() + p->rect().height() - yOffset );
163 
164  painter.drawLine( plusPixmap->width() / 2 + 2,
165  p->rect().y() + p->rect().height() - yOffset,
166  plusPixmap->width() / 2 - 2,
167  p->rect().y() + p->rect().height() - yOffset );
168  } else if( data->level() != 0 ){
169  painter.drawLine( plusPixmap->width() / 2, p->rect().y() - yOffset,
170  plusPixmap->width() / 2, p->rect().y() + p->rect().height() - yOffset );
171  }
172  }
173  p = p->next();
174  }
175 
176  painter.end();
177  bitBlt( this, 0, 0, &buffer );
178 }

192  ParagData *data = (ParagData*) p->extraData();
