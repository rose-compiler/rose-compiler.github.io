272  bool changed = false;
273  for (size_t i = 0; i < nOperands(); ++i) {
274  // match (+ (+ reg1 (* reg2 sz)) offset)
275  if (auto mre = isSgAsmMemoryReferenceExpression(operand(i))) {
276  if (auto add1 = isSgAsmBinaryAdd(mre->get_address())) {
277  if (auto add2 = isSgAsmBinaryAdd(add1->get_lhs())) {
278  if (/*auto reg1 = */isSgAsmDirectRegisterExpression(add2->get_lhs())) {
279  if (auto mul = isSgAsmBinaryMultiply(add2->get_rhs())) {
280  if (/*auto reg2 = */isSgAsmDirectRegisterExpression(mul->get_lhs())) {
281  if (/*auto sz = */isSgAsmIntegerValueExpression(mul->get_rhs())) {
282  if (auto offset = isSgAsmIntegerValueExpression(add1->get_rhs())) {
283  // Swap the offset and multiplication to result in
284  // (+ (+ reg1 offset) (* reg2 sz))
285 
286  // Remove offset and mul from the tree
287  add2->set_rhs(nullptr);
288  add1->set_rhs(nullptr);
289 
290  // Insert them back in the correct order
291  add2->set_rhs(offset);
292  offset->set_parent(add2);
293  add1->set_rhs(mul);
294  mul->set_parent(add1);
295  changed = true;
296  }
297  }
298  }
299  }
300  }
301  }
302  }
303  }
304  }
305  return changed;
