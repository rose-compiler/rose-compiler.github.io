18 class FlattenBlocksVisitor: public AstSimpleProcessing {
19  public:
20  virtual void visit(SgNode* node) {
21  if (isSgBasicBlock(node)) {
22  SgBasicBlock* c = isSgBasicBlock(node);
23  SgStatementPtrList newStatements;
24  for (SgStatementPtrList::const_iterator i = c->get_statements().begin();
25  i != c->get_statements().end(); ++i) {
26  if (isSgBasicBlock(*i)) {
27  SgBasicBlock* c2 = isSgBasicBlock(*i);
28  const SgStatementPtrList& c2Stmts = c2->get_statements();
29  // We need to prevent a declaration from immediately following a label, as that is illegal
30  if (!newStatements.empty() && isSgLabelStatement(newStatements.back()) && !c2Stmts.empty() && isSgVariableDeclaration(c2Stmts.front())) {
31  newStatements.push_back(SageBuilder::buildExprStatement(SageBuilder::buildNullExpression()));
32  }
33  newStatements.insert(newStatements.end(), isSgBasicBlock(*i)->get_statements().begin(), isSgBasicBlock(*i)->get_statements().end());
34  } else {
35  if (!newStatements.empty() && isSgLabelStatement(newStatements.back()) && isSgVariableDeclaration(*i)) {
36  newStatements.push_back(SageBuilder::buildExprStatement(SageBuilder::buildNullExpression()));
37  }
38  newStatements.push_back(*i);
39  }
40  }
41  if (!newStatements.empty() && isSgLabelStatement(newStatements.back())) {
42  // Prevent block from ending with a label
43  newStatements.push_back(SageBuilder::buildExprStatement(SageBuilder::buildNullExpression()));
44  }
45  for (SgStatementPtrList::const_iterator i = newStatements.begin();
46  i != newStatements.end(); ++i) {
47  (*i)->set_parent(c);
48  }
49  c->get_statements() = newStatements;
50  c->get_symbol_table()->get_table()->clear();
51  SageInterface::rebuildSymbolTable(c);
52  }
53  }
54 };

262 // Within sc, is the variable toCheck modified between the declaration of
263 // lifetime and its last use? This is used to determine whether, whenever
264 // toCheck and lifetime are equal, one can be used as a substitute for the
265 // other.
266 bool isPotentiallyModifiedDuringLifeOf(SgBasicBlock* sc,
267  SgInitializedName* toCheck,
268  SgInitializedName* lifetime) {
269  SgStatementPtrList& stmts = sc->get_statements();
270  bool inLiveRange = false;
271  for (SgStatementPtrList::reverse_iterator i = stmts.rbegin();
272  i != stmts.rend(); ++i) {
273  if (containsVariableReference(*i, lifetime))
274  inLiveRange = true;
275  SgVariableSymbol* toCheckSym = new SgVariableSymbol(toCheck);
276  SgVarRefExp* toCheckVr = new SgVarRefExp(SgNULL_FILE, toCheckSym);
277 
278  bool result = false;
279  if (inLiveRange && isPotentiallyModified(toCheckVr, *i)) {
280  result = true;
281  }
282  delete toCheckSym;
283  toCheckSym = NULL;
284  toCheckVr->set_symbol(NULL);
285  delete toCheckVr;
286  toCheckVr = NULL;
287  if (result) return true;
288 
289  if (isSgVariableDeclaration(*i) &&
