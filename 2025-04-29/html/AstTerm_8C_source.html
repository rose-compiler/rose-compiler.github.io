90  } else {

92  }

94  typeTerm+="type:null";

96  }
97  }
98  int arity=node->get_numberOfTraversalSuccessors();
99  if(arity>0) {
100  s+="(";
101  if(withTypes) {
102  if(typeTerm.size()>0)
103  s+=typeTerm+",";
104  }
105  for(int i=0; i<arity;i++) {
106  SgNode* child = node->get_traversalSuccessorByIndex(i);
107  if(i!=0)
108  s+=",";
109  s+=astTermWithNullValuesToString(child,withNumbers,withTypes,visited);
110  }
111  s+=")";
112  }
113  // unparse number leaf nodes if requested
114  if(arity==0 && withNumbers) {
115  if(SgExpression* exp=isSgExpression(node)) {
116  s+=":"+exp->unparseToString();
117  }
118  }
119  return s;
