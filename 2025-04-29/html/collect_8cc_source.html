138 bool
139 StmtCollectorByLoc::matches (const SgForInitStatement* node) const
140 {
141  if (node == NULL) return false;
142 
143  const SgForStatement* loop = isSgForStatement (node->get_parent ());
144  if (loop == NULL) return false;
145 
146  /* "Derive" line numbers for this node by taking the starting line
147  * of the surrounding loop as the starting line, and the starting
148  * line of the body minus 1 as the ending line.
149  */
150  string filename = RoseHPCT::getFilename (node);
151  int line_start = RoseHPCT::getLineStart (loop);
152  int line_end = RoseHPCT::getLineEnd (loop->get_loop_body ()) - 1;
153 
154  if (line_end < line_start) line_end = line_start;
155 
156  return filenamesMatch (filename) && linesMatch (line_start, line_end);
157 }
