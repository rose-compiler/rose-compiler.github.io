476 void
477 RiscOperators::insertDataFlowEdge(const AbstractLocation &source, const AbstractLocation &target,
478  DataFlowEdge::EdgeType edgeType) {
479  // Try to find existing vertices. This is a linear traversal because abstract locations for memory addresses
480  // don't have relational operators (they might be symbolic expressions or other user-defined types).
481  DataFlowGraph::VertexIterator sourceIter = dflow_.vertices().end();
482  DataFlowGraph::VertexIterator targetIter = dflow_.vertices().end();
483  for (DataFlowGraph::VertexIterator iter=dflow_.vertices().begin(); iter!=dflow_.vertices().end(); ++iter) {
484  const AbstractLocation &vertex = iter->value();
485  if (sourceIter == dflow_.vertices().end()) {
486  if (vertex.mustAlias(source))
487  sourceIter = iter;
488  }
489  if (targetIter == dflow_.vertices().end()) {
490  if (vertex.mustAlias(target))
491  targetIter = iter;
492  }
493  }
494 
495  // Add new vertices if we didn't find existing vertices
496  if (sourceIter == dflow_.vertices().end())
497  sourceIter = dflow_.insertVertex(source);
498  if (targetIter == dflow_.vertices().end())
499  targetIter = dflow_.insertVertex(target);
500 
501  // Add the edge. Parallel edges are allowed (even desired) and they are numbered consecutively. We can't rely on the
502  // edge ID numbers provided by the graph because those ID numbers are not stable over edge erasure, so we store the
503  // sequence number as part of the edge itself.
504  dflow_.insertEdge(sourceIter, targetIter, DataFlowEdge(dflow_.nEdges(), edgeType));
505 }

507 void
508 RiscOperators::insertDataFlowEdges(const BaseSemantics::SValue::Ptr &svalue_, const AbstractLocation &target) {
509  SValue::Ptr svalue = SValue::promote(svalue_);
510  DataFlowEdge::EdgeType edgeType = DataFlowEdge::CLOBBER;
511  if (svalue->sources().empty()) {
512  insertDataFlowEdge(AbstractLocation(), target, edgeType);
513  } else {
514  for (const AbstractLocation &source: svalue->sources()) {
515  insertDataFlowEdge(source, target, edgeType);
516  edgeType = DataFlowEdge::AUGMENT;
517  }
518  }

552 BaseSemantics::SValue::Ptr
553 RiscOperators::readOrPeekMemory(RegisterDescriptor segreg, const BaseSemantics::SValue::Ptr &addr_,
554  const BaseSemantics::SValue::Ptr &dflt_, const BaseSemantics::SValue::Ptr &cond,
555  bool allowSideEffects) {
556  TemporarilyDeactivate deactivate(this, innerDomainId_);
557  MultiSemantics::SValue::Ptr addr = MultiSemantics::SValue::promote(addr_);
558  MultiSemantics::SValue::Ptr dflt = MultiSemantics::SValue::promote(dflt_);
559  MultiSemantics::SValue::Ptr result;
560  if (allowSideEffects) {
561  result = MultiSemantics::SValue::promote(Super::readMemory(segreg, addr, dflt, cond));
562  } else {
563  result = MultiSemantics::SValue::promote(Super::peekMemory(segreg, addr, dflt));
564  }
565 
566  size_t addrWidth = addr->nBits();
567  size_t valueWidth = dflt->nBits();
568  ASSERT_require(0 == valueWidth % 8);
569 
570  BaseSemantics::RiscOperators::Ptr userOps = get_subdomain(userDomainId_);
571  BaseSemantics::RiscOperators::Ptr innerOps = get_subdomain(innerDomainId_);
572 
573  SValue::Ptr definers = SValue::promote(dflt->get_subvalue(innerDomainId_)->copy());
574  for (size_t bytenum=0; bytenum<valueWidth/8; ++bytenum) {
575  BaseSemantics::SValue::Ptr byteAddr = userOps->add(addr->get_subvalue(userDomainId_),
576  userOps->number_(addrWidth, bytenum));
577  definers->insert(AbstractLocation(byteAddr));
578  }
579  result->set_subvalue(innerDomainId_, definers);
580  return result;
581 }
