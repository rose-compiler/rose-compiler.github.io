97 BaseGraph::DFSIterator::DFSIterator (BaseGraph& g)
98 {
99  if (g.root_node != 0) {
100  if (g.DFS_needed) {
101  // reset all the DFS succ links and call the recursive routine to re-create them
102  std::set<Node*>::iterator ni = g.node_set.begin();
103  while (ni != g.node_set.end()) {
104  Node* n = *ni;
105  n->dfs_succ = 0;
106  ++ni;
107  }
108  g.create_DFS_links(g.root_node);
109  g.DFS_needed = false;
110  }
111  }
112  p = g.root_node;
113  cerr << "BaseGraph DFSIterator " << p << endl;
114 }

145 void
146 BaseGraph::add (BaseGraph::Edge* e)
147 #if __cplusplus < 201103L
148  throw (BaseGraph::DuplicateEdge, BaseGraph::EdgeInUse, BaseGraph::EmptyEdge,
149  /* indirect: from BaseGraph::add(BaseGraph::Node) */
150  BaseGraph::DuplicateNode, BaseGraph::NodeInUse, BaseGraph::EmptyNode)
151 #endif
152 {
153  if (e == 0)
154  throw EmptyEdge();
155  if (edge_set.find(e) != edge_set.end())
156  throw DuplicateEdge(e);
157  if (e->in_use)
158  throw EdgeInUse(e);
159 
160  // insert the nodes if they don't already exist in the graph
161  if (node_set.find(e->n1) == node_set.end())
162  add(e->n1);
163  if (node_set.find(e->n2) == node_set.end())
164  add(e->n2);
165  e->in_use = true;
166  edge_set.insert(e);
167  DFS_needed = BFS_needed = true;
168 }

175 BaseGraph::add (BaseGraph::Node* n)
176 #if __cplusplus < 201103L
177  throw (BaseGraph::DuplicateNode, BaseGraph::NodeInUse, BaseGraph::EmptyNode)
178 #endif
179 {
180  if (n == 0)
181  throw EmptyNode();
182  if (node_set.find(n) != node_set.end())
183  throw DuplicateNode(n);
184  if (n->in_use)
185  throw NodeInUse(n);
186  if (root_node == 0)
187  root_node = n;
188  n->in_use = true;
189  node_set.insert(n);
190  DFS_needed = BFS_needed = true;
191 }

216 void
217 BaseGraph::remove (BaseGraph::Node* n)
218 #if __cplusplus < 201103L
219  throw (BaseGraph::NonexistentNode, BaseGraph::DeletingRootOfNonSingletonGraph, BaseGraph::EmptyNode)
220 #endif
221 {
222  if (n == 0)
223  throw EmptyNode();
224  if ((n == root_node) && (!edge_set.empty() || (node_set.size() > 1)))
225  throw DeletingRootOfNonSingletonGraph(n);
226  if (node_set.erase(n) == 0)
227  throw NonexistentNode(n);
228  n->in_use = false;
229  DFS_needed = BFS_needed = true;
230  if (n == root_node)
231  root_node = 0;
232 }
