48 Partitioner::basicBlockOptionalMayReturn(const BasicBlock::Ptr &bb) const {
49  ASSERT_not_null(bb);
50 
51  bool retval;
52  if (bb->mayReturn().getOptional().assignTo(retval))
53  return retval; // already cached
54  ControlFlowGraph::ConstVertexIterator startVertex = findPlaceholder(bb->address());
55  if (startVertex != cfg_.vertices().end())
56  return basicBlockOptionalMayReturn(startVertex); // full CFG-based analysis
57 
58  if (basicBlockIsFunctionReturn(bb)) {
59  bb->mayReturn() = true;
60  return true;
61  }
62 
63  BasicBlock::Successors successors;
64  if (!bb->successors().getOptional().assignTo(successors))
65  return Sawyer::Nothing();
66 
67  // If the basic block has any significant successor that may return then this basic block may return.
68  bool successorIsIndeterminate=false, calleeMayReturn=false, calleeIsIndeterminate=false;
69  std::vector<ControlFlowGraph::ConstVertexIterator> callReturnSuccessors;
70  for (const BasicBlock::Successor &edge: successors) {
71  if (!edge.expr()->isConcrete()) {
72  successorIsIndeterminate = true; // sucessor vertex is (or would be) the indeterminate vertex
73  continue;
74  }
75  Address successorVa = edge.expr()->toUnsigned().get();
76  if (successorVa == bb->address())
77  continue; // a self edge is not significant to the analysis result
78  ControlFlowGraph::ConstVertexIterator successorVertex = findPlaceholder(successorVa);
79  if (successorVertex == cfg_.vertices().end()) {
80  successorIsIndeterminate = true; // a not-present successor has indeterminate may-return property
81  continue;
82  }
83 
84  bool b;
85  if (edge.type() == E_CALL_RETURN) {
86  callReturnSuccessors.push_back(successorVertex); // skip for now; they need to be processed last
87  } else if (edge.type() == E_FUNCTION_CALL) {
88  if (!calleeMayReturn) {
89  if (!basicBlockOptionalMayReturn(successorVertex).assignTo(b)) {
90  calleeIsIndeterminate = true;
91  } else if (b) {
92  calleeMayReturn = true;
93  }
94  }
95  } else {
96  // This is a significant edge
97  if (!basicBlockOptionalMayReturn(successorVertex).assignTo(b)) {
98  successorIsIndeterminate = true;
99  } else if (b) {
100  bb->mayReturn() = true;
101  return true; // bb may return if any significant successor may return
102  }
103  }
104  }
105 
106  // Call-return edges can be processed now.
107  if (calleeMayReturn) {
108  for (const ControlFlowGraph::ConstVertexIterator &successor: callReturnSuccessors) {
109  bool b;
110  if (!basicBlockOptionalMayReturn(successor).assignTo(b)) {
111  successorIsIndeterminate = true;
112  } else if (b) {
113  bb->mayReturn() = true;
114  return true; // call-ret is a significant successor that may return
115  }
116  }
117  }
118 
119  // If there is no call-return edge and a callee is positive or indeterminate then this block is also indeterminate.
120  if (callReturnSuccessors.empty() && (calleeMayReturn || calleeIsIndeterminate))
121  return Sawyer::Nothing();
122 
123  // None of the significant successors has a positive may-return property. If they were all negative (no indeterminates)
124  // then we can say that this block does not return.
125  if (!successorIsIndeterminate) {
126  bb->mayReturn() = false;
127  return false;
128  }
129 
130  return Sawyer::Nothing();
131 }

155 // Internal: Is edge significant for may-return analysis?
156 // + Function call edges are never significant (a return from a function call does not mean that the calling function also
157 // returns.
158 // + A self edge is not significant; it doesn't change the result of the analysis
159 // + Call-ret edges are significant only if at least one of the sibling function calls returns or if any of the sibling
160 // function calls has an indeterminate may-return.
161 // + Other edges are significant, including E_FUNCTION_XFER edges.
162 bool
163 Partitioner::mayReturnIsSignificantEdge(const ControlFlowGraph::ConstEdgeIterator &edge,
164  std::vector<MayReturnVertexInfo> &vertexInfo) const {
165  if (edge == cfg_.edges().end())
166  return false;
167  if (edge->value().type() == E_FUNCTION_CALL)
168  return false;
169  if (edge->target() == edge->source())
170  return false;
171  if (edge->value().type() == E_CALL_RETURN) {
172  boost::logic::tribool tb = mayReturnDoesCalleeReturn(edge->source(), vertexInfo);
173  return tb || boost::logic::indeterminate(tb) ? true : false;
174  }
175  return true;
176 }

180 boost::logic::tribool

209 // Internal: Returns non-negative (positive or indeterminate) if any significant successor of this vertex has non-negative
210 // may-return. Simulates a phantom call-return edge if necessary since such edges might not yet be present in the CFG. For
211 // instance, if vertex V calls vertex U but V has no call-return edge then a phantom one will be created unless U has negative
212 // may-return. The phantom call-return edge is assumed to point to a phantom vertex with indeterminate may-return.
213 boost::logic::tribool
214 Partitioner::mayReturnDoesSuccessorReturn(const ControlFlowGraph::ConstVertexIterator &vertex,
215  std::vector<MayReturnVertexInfo> &vertexInfo) const {
216  ASSERT_require(vertex->value().type() == V_BASIC_BLOCK);
217 
218  bool hasIndeterminateSuccessor = false; // does any significant successor have indeterminate may-return?
219  bool isFunctionCall = false; // does vertex have any function call edges?
220  bool hasIndeterminateCallee = false; // does any callee have indeterminate may-return?
221  bool hasPositiveCallee = false; // does any callee have positive may-return
222  bool hasCallReturnEdge = false; // does vertex have any call-return edges?
223 
224  for (ControlFlowGraph::ConstEdgeIterator edge = vertex->outEdges().begin();
225  edge != vertex->outEdges().end(); ++edge) {
226  ControlFlowGraph::ConstVertexIterator successor = edge->target();
227  if (mayReturnIsSignificantEdge(edge, vertexInfo)) {
228  // Significant successors
229  if (edge->value().type() == E_CALL_RETURN)
230  hasCallReturnEdge = true;
231 
232  if (vertexInfo[successor->id()].result) {
233  return true;
234  } else if (boost::logic::indeterminate(vertexInfo[successor->id()].result)) {
235  hasIndeterminateSuccessor = true;
236  }
237  } else if (edge->value().type() == E_FUNCTION_CALL) {
238  // Callees
239  isFunctionCall = true;
240  bool mayReturn = false;
241  if (!basicBlockOptionalMayReturn(successor, vertexInfo).assignTo(mayReturn)) {
242  hasIndeterminateCallee = true;
243  } else if (mayReturn) {
244  hasPositiveCallee = true;
245  }
246  }
247  }
248 
249  if (hasIndeterminateSuccessor)
250  return boost::logic::indeterminate;
251 
252  if (isFunctionCall && !hasCallReturnEdge && (hasIndeterminateCallee || hasPositiveCallee))
253  return boost::logic::indeterminate; // assume phantom call-return edge is indeterminate
254 
255  return false;
256 }

261 Partitioner::basicBlockOptionalMayReturn(const ControlFlowGraph::ConstVertexIterator &start,
262  std::vector<MayReturnVertexInfo> &vertexInfo) const {

274 
275  typedef DepthFirstForwardGraphTraversal<const ControlFlowGraph> Traversal;
276  for (Traversal t(cfg_, start, ENTER_EDGE|ENTER_VERTEX|LEAVE_VERTEX); t; ++t) {
277  switch (t.event()) {
278  case ENTER_EDGE: {
279  // The call to mayReturnIsSignificantEdge will be recursive if t.vertex is a function call
280  if (!mayReturnIsSignificantEdge(t.edge(), vertexInfo)) {
281  SAWYER_MESG(debug) <<"[" <<depth <<"] skipping edge " <<edgeName(t.edge()) <<"\n";
282  t.skipChildren();
283  } else {
284  SAWYER_MESG(debug) <<"[" <<depth <<"] following edge " <<edgeName(t.edge()) <<"\n";
285  }
286  break;
287  }
288 
289  case ENTER_VERTEX: {
290  SAWYER_MESG(debug) <<"[" <<depth <<"] enter vertex " <<vertexName(t.vertex())
291  <<" via " <<edgeNameSrc(t.edge()) <<"\n";
292 
293  // Recursion termination
294  switch (vertexInfo[t.vertex()->id()].state) {
295  case MayReturnVertexInfo::INIT:
296  vertexInfo[t.vertex()->id()].state = MayReturnVertexInfo::CALCULATING;
297  break;
298  case MayReturnVertexInfo::CALCULATING:
299  SAWYER_MESG(debug) <<"[" <<depth <<"] recursive query; returning nothing\n";
300  t.skipChildren();
301  return Sawyer::Nothing();
302  case MayReturnVertexInfo::FINISHED:
303  SAWYER_MESG(debug) <<"[" <<depth <<"] already calculated: may-return is "
304  <<toString(vertexInfo[t.vertex()->id()].result) <<"\n";
305  t.skipChildren();
306  if (vertexInfo[t.vertex()->id()].result)
307  return true;
308  if (!vertexInfo[t.vertex()->id()].result)
309  return false;
310  return Sawyer::Nothing();
311  }
312 
313  // Can we get the may-return value immediately, or do we need to follow outgoing edges first?
314  if (t.vertex()->value().type() == V_BASIC_BLOCK) {
315  BasicBlock::Ptr bb = t.vertex()->value().bblock();
316 
317  // Properties of functions that own this block.
318  Function::Ptr isBlackListed, isWhiteListed, isDynamicLinked;
319  for (const Function::Ptr &function: t.vertex()->value().owningFunctions().values()) {
320  if (function->address() == t.vertex()->value().address()) {
321  bool b = false;
322  if (config_.functionMayReturn(function).assignTo(b)) {
323  if (b) {
324  isWhiteListed = function;
325  } else {
326  isBlackListed = function;
327  }
328  } else if (boost::ends_with(function->name(), "@plt") ||
329  boost::ends_with(function->name(), "@iat")) {
330  isDynamicLinked = function;
331  }
332  }
333  }
334 
335  if (isWhiteListed && isBlackListed) {
336  // Block is owned by functions that are both white and black listed for may-return. Assume white.
337  SAWYER_MESG(debug) <<"[" <<depth <<"] block "
338  <<StringUtility::addrToString(t.vertex()->value().address())
339  <<" belongs to white- and black-listed functions (assuming white).\n";
340  SAWYER_MESG(debug) <<"[" <<depth <<"] example white: " <<isWhiteListed->printableName() <<"\n"
341  <<"[" <<depth <<"] example black: " <<isBlackListed->printableName() <<"\n";
342  vertexInfo[t.vertex()->id()].result = true;
343  t.skipChildren();
344  } else if (isWhiteListed) {
345  // Block is whitelisted by some owning function.
346  SAWYER_MESG(debug) <<"[" <<depth <<"] " <<isWhiteListed->printableName() <<" is whitelisted\n";
347  vertexInfo[t.vertex()->id()].result = true;
348  t.skipChildren();
349  } else if (isBlackListed) {
350  // Block is blacklisted by some owning function.
351  SAWYER_MESG(debug) <<"[" <<depth <<"] " <<isBlackListed->printableName() <<" is blacklisted\n";
352  vertexInfo[t.vertex()->id()].result = false;
353  t.skipChildren();
354  } else if (isDynamicLinked) {
355  // Dynamically linked functions return or not by definition
356  bool b = assumeFunctionsReturn_;
357  SAWYER_MESG(debug) <<"[" <<depth <<"] " <<isDynamicLinked->printableName()
358  <<" is " <<(b?"whitelisted":"blacklisted") <<" by /@plt$/ pattern\n";
359  vertexInfo[t.vertex()->id()].result = b;
360  t.skipChildren();
361  } else if (t.vertex()->nOutEdges()==1 && t.vertex()->outEdges().begin()->target()==nonexistingVertex_) {
362  // Non-existing vertex returns or not by definition
363  SAWYER_MESG(debug) <<"[" <<depth <<"] " <<vertexName(t.vertex())
364  <<(assumeFunctionsReturn_?"returns":"does not return")
365  <<" by virtue of not existing\n";
366  vertexInfo[t.vertex()->id()].result = assumeFunctionsReturn_;
367  t.skipChildren();
368  } else if (bb && bb->mayReturn().isCached()) {
369  // Basic block may-return is already calculated
370  bool b = bb->mayReturn().get();
371  SAWYER_MESG(debug) <<"[" <<depth <<"] already cached: may-return is " <<(b?"yes":"no") <<"\n";
372  vertexInfo[t.vertex()->id()].result = b;
373  t.skipChildren();
374  } else if (bb && basicBlockIsFunctionReturn(bb)) {
375  // This is a function return statement, so it obviously returns
376  SAWYER_MESG(debug) <<"[" <<depth <<"] block is a function return; may-return is yes\n";
377  bb->mayReturn() = true;
378  vertexInfo[t.vertex()->id()].result = true;
379  t.skipChildren();
380  } else if (bb && basicBlockIsFunctionCall(bb)) {
381  // Function calls handled by recursively computing may-return in the callee
382  SAWYER_MESG(debug) <<"[" <<depth <<"] block is a function call; process callees...\n";
383  boost::logic::tribool tb = mayReturnDoesCalleeReturn(t.vertex(), vertexInfo);
384  SAWYER_MESG(debug) <<"[" <<depth <<"] callees for " <<vertexName(t.vertex()) <<" processed"
385  <<"; " <<(tb || boost::logic::indeterminate(tb)?"at least one":"none")
386  <<" may return\n";
387  } else if (t.vertex()->nOutEdges()==1 && t.vertex()->outEdges().begin()->target()==indeterminateVertex_ &&
388  assumeFunctionsReturn_) {
389  // Indeterminate successor is likely a permanent condition
390  SAWYER_MESG(debug) <<"[" <<depth <<"] " <<vertexName(t.vertex())
391  <<" returns by virtue of having only an indeterminate successor\n";
392  vertexInfo[t.vertex()->id()].result = assumeFunctionsReturn_;
393  t.skipChildren();
394  } else if (!addressIsExecutable(t.vertex()->value().address())) {
395  // Non-executable address is perhaps not possible, but such returns or not by definition
396  SAWYER_MESG(debug) <<"[" <<depth <<"] " <<vertexName(t.vertex())
397  <<(assumeFunctionsReturn_?"returns":"does not return")
398  <<" by virtue of being at a non-executable address\n";
399  vertexInfo[t.vertex()->id()].result = assumeFunctionsReturn_;
400  t.skipChildren();
401  } else if (BasicBlock::Ptr bb = t.vertex()->value().bblock()) {
402  // FIXME[Robb P. Matzke 2014-12-17]: Need to handle graph cycles, but for now we'll just say that the
403  // may-return for a cycle is indeterminate. E.g., see test7_no in i386-may-return-tests.
404  SAWYER_MESG(debug) <<"[" <<depth <<"] must process vertex successors first...\n";
405  }
406  }
407  break;
408  }
409 
410  case LEAVE_VERTEX: {
411  ASSERT_require(vertexInfo[t.vertex()->id()].state == MayReturnVertexInfo::CALCULATING);
412  SAWYER_MESG(debug) <<"[" <<depth <<"] leaving vertex " <<vertexName(t.vertex()) <<"\n";

416  // may-return. Otherwise it has indeterminate may-return if any of its significant successors is
417  // indeterminate. Otherwise, it has negative may-return.

427  } else if (!tb) {
428  bblock->mayReturn() = false;
429  } else {

432  }

443  }
