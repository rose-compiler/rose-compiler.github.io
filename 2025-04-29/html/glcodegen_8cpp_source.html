68 // tell DotViz mouse position to OpenGL display
69 void GlCodegenState::set_mousepos(int x, int y)
70 {
71  // remember position
72  MouseX = x;
73  MouseY = y;
74  if((x<0)||(y<0)) return;
75 
76  // calculate intersection with graph plane
77  ntlVec3d camera( -CamX, -CamY, -CamZ );
78  ntlVec3d lookat( -LookatX, -LookatY, 0.0 );
79  ntlVec3d direction = lookat - camera;
80  double fovy = 90.0;
81  double aspect = (double)ViewSizeX/ViewSizeY;
82  /* calculate width of screen using perpendicular triangle diven by
83  * viewing direction and screen plane */
84  double screenWidth = direction.getNorm()*tan( (fovy*0.5/180.0)*M_PI );
85 
86  /* calculate vector orthogonal to up and viewing direction */
87  ntlVec3d upVec(0.0, 1.0, 0.0);
88  ntlVec3d rightVec( upVec.crossProd(direction) );
89  rightVec.normalize();
90 
91  /* calculate screen plane up vector, perpendicular to viewdir and right vec */
92  upVec = ntlVec3d( rightVec.crossProd(direction) );
93  upVec.normalize();
94 
95  /* check if vectors are valid FIXME what to do if not? */
96  if( (upVec==ntlVec3d(0.0)) || (rightVec==ntlVec3d(0.0)) ) {
97  return;
98  }
99 
100  /* length from center to border of screen plane */
101  rightVec *= (screenWidth*aspect * -1.0);
102  upVec *= (screenWidth * -1.0);
103 
104  /* calc edges positions */
105  double zplane = 0.0;
106  ntlVec3d mousedir = direction - rightVec + upVec;
107  mousedir += rightVec*2.0* (float)x/(float)ViewSizeX;
108  mousedir -= upVec*2.0* (float)y/(float)ViewSizeY;
109 
110  if((zplane-mousedir[2])>1.0) { // only treat negative directions
111  mousedir[0] /= (zplane-mousedir[2]);
112  mousedir[1] /= (zplane-mousedir[2]);
113  }
114  MousePlaneX = camera[0] + mousedir[0]*camera[2];
115  MousePlaneY = camera[1] + mousedir[1]*camera[2];
116  //cout << gMousePlaneX<<","<<gMousePlaneY<<" "<<gCamX<<","<<gCamY << endl;
117 }

301 void GlCodegenState::select_node(graph_t *pGraph, node_t *node)
302 {
303  SelNodeEdges.clear();
304  // first find all successors
305  for (Agedge_t *edge = agfstout(pGraph,node); edge!=NULL; edge = agnxtout(pGraph,edge)) {
306  SelNodeEdges.push_back( edge );
307  }
308  // find all edges pointing to this one
309  int parentCnt = 0;
310  SelNodeParent = NULL;
311  for (Agnode_t *inode = agfstnode(pGraph); inode!=NULL; inode = agnxtnode(pGraph,inode)) {
312  for (Agedge_t *edge = agfstout(pGraph,inode); edge!=NULL; edge = agnxtout(pGraph,edge)) {
313  if(edge->head == node) {
314  SelNodeEdges.push_back( edge );
315  SelNodeParent = edge->tail;
316  parentCnt++;
317  }
318  }
319  }
320  EdgeSelected = 0;
321  NodeSelected = node;
322  if(mpGraphInfo) {
323  mpGraphInfo->showInfo( NodeSelected );
324  }
325 }
