705 void ClangToDotTranslator::VisitTemplateArgument(const clang::TemplateArgument & template_argument, ClangToDotTranslator::NodeDescriptor & node_desc, std::string prefix) {
706  std::ostringstream oss;
707  oss << prefix;
708  switch (template_argument.getKind()) {
709  case clang::TemplateArgument::Null:
710  oss << " null";
711  node_desc.attributes.push_back(std::pair<std::string, std::string>(oss.str(), ""));
712  break;
713  case clang::TemplateArgument::Type:
714  oss << " type";
715  node_desc.successors.push_back(std::pair<std::string, std::string>(oss.str(), Traverse(template_argument.getAsType().getTypePtr())));
716  break;
717  case clang::TemplateArgument::Declaration:
718  oss << " declaration";
719  node_desc.successors.push_back(std::pair<std::string, std::string>(oss.str(), Traverse(template_argument.getAsDecl())));
720  break;
721  case clang::TemplateArgument::Integral:
722  oss << " integral";
723  // assert(DEBUG_TODO == 0); // TODO
724  ROSE_ASSERT(FAIL_TODO == 0); // TODO
725  break;
726  case clang::TemplateArgument::Template:
727  oss << " template";
728  // assert(DEBUG_TODO == 0); // TODO
729  ROSE_ASSERT(FAIL_TODO == 0); // TODO
730  break;
731  case clang::TemplateArgument::TemplateExpansion:
732  oss << " template_expansion";
733  // assert(DEBUG_TODO == 0); // TODO
734  ROSE_ASSERT(FAIL_TODO == 0); // TODO
735  break;
736  case clang::TemplateArgument::Expression:
737  oss << " expression";
738  node_desc.successors.push_back(std::pair<std::string, std::string>(oss.str(), Traverse(template_argument.getAsExpr())));
739  break;
740  case clang::TemplateArgument::Pack:
741  {
742  clang::TemplateArgument::pack_iterator pack_it;
