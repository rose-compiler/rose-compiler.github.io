168 //-----------------------------------------------------------------------------
170 void VizGroup::vizLayout(VizGroupLayout slayout)
171 {
172  int x=this->x(), y=this->y();
173 
174  for(int i=0; i<MAX_VIZ; i++) {
175  if((mVizActive[i])&&(mVizVisible[i])) {
176  mpGroup[i]->remove( mpViz[i] );
177  // hide them - this is important, otherwise they will still be drawn somewhere...
178  mpViz[i]->hide();
179  }
180  mVizVisible[i] = false;
181  mpGroup[i] = NULL;
182  }
183  clear();
184  current(this);
185 
186  switch(slayout) {
187  case single:
188  addViz( x, y, w(), h() ,0,NULL);
189  break;
190  case splitHoriz:
191  addViz( x, y, w(), h()/2 ,0,NULL);
192  addViz( x, y+h()/2, w(), h()/2 ,1,NULL);
193  break;
194  case splitVert:
195  addViz( x, y, w()/2, h() ,0,NULL);
196  addViz( x+w()/2, y, w()/2, h() ,1,NULL);
197  break;
198  case triple:
199  addViz( x+w()/3*1, y, w()/3*2, h() ,0,NULL);
200  addViz( x+w()/3*0, y+h()/2*0, w()/3, h()/2 ,1,NULL);
201  addViz( x+w()/3*0, y+h()/2*1, w()/3, h()/2 ,2,NULL);
202  break;
203  case quad:
204  addViz( x , y , w()/2, h()/2 ,0,NULL);
205  addViz( x+w()/2, y , w()/2, h()/2 ,1,NULL);
206  addViz( x , y+h()/2, w()/2, h()/2 ,2,NULL);
207  addViz( x+w()/2, y+h()/2, w()/2, h()/2 ,3,NULL);
208  break;
209  case threeone:
210  addViz( x+w()/4*1, y, w()/4*3, h() ,0,NULL);
211  addViz( x , y+h()/3*0, w()/4, h()/3 ,1,NULL);
212  addViz( x , y+h()/3*1, w()/4, h()/3 ,2,NULL);
213  addViz( x , y+h()/3*2, w()/4, h()/3 ,3,NULL);
214  break;
215  default:
216  break;
217  }
218 
219  end();
220  focus( mpViz[0] );
221 }
