105 SgType* AttributeRec::computeEntityType()
106 {
107  if (charLenExpr)
108  {
109  // convert entityType to string type with char length and the existing type kind
110  SgExpression* typeKind = entityType->get_type_kind();
111  entityType = SgTypeString::createType(charLenExpr, typeKind);
112  }
113  if (hasDimension)
114  {
115  entityType = buildArrayType(dimExp);
116  }
117  if (hasCodimension)
118  {
119  entityType = buildArrayType(codimExp);
120  entityType->set_isCoArray(true);
121  }
122  if (isCopointer)
123  {
124  entityType = new SgPointerType(entityType);
125  entityType->set_isCoArray(true); // a copointer is a pointer whose isCoArray flag is true.
126  }
127  if (isPointer)
128  {
129  entityType = new SgPointerType(entityType);
130  }
131  return entityType;

308 SgExprListExp* AttributeRec::buildDimensionInfo()
309 {
310  // transfer the array/coarray spec from the astExpressionStack to this AttrSpec record.
311  SgIntVal * intVal = isSgIntVal(astExpressionStack.front());
312  ROSE_ASSERT(intVal); // astExpressionStack.front() contains the number of array spec elements for the dimension info
313  int count = intVal->get_value();
314  delete intVal; // must remove it from AST
315  astExpressionStack.pop_front();
316  processMultidimensionalSubscriptsIntoExpressionList(count); // the dimension info is now on top of astExpressionStack
317  SgExprListExp* dimInfo = isSgExprListExp(astExpressionStack.front());
318  ROSE_ASSERT(dimInfo);
319 // setDimExp(dimInfo);
320 // setHasDimension(true);
321  astExpressionStack.pop_front();
322  return dimInfo;
