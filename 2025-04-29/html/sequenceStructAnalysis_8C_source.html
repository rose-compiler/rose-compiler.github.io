392 bool SeqStructAnalysis::transfer(const Function& func, const DataflowNode& n, NodeState& state, const vector<Lattice*>& dfInfo)
393 {
394  bool modified=false;
395 
396  ConstrGraph* cg = dynamic_cast<ConstrGraph*>(state.getLatticeAbove(cgAnalysis, 0));
397 
398  ROSE_ASSERT(dfInfo.size() == 1);
399  FiniteVarsExprsProductLattice* prodLat = dynamic_cast<FiniteVarsExprsProductLattice*>(*(dfInfo.begin()));
400 
401  // Make sure that all the lattices are initialized
402  const vector<Lattice*>& lattices = prodLat->getLattices();
403  for(vector<Lattice*>::const_iterator it = lattices.begin(); it!=lattices.end(); it++)
404  (dynamic_cast<DivLattice*>(*it))->initialize();
405 
406 
407  if(isSgPntrArrRefExp(n.getNode()) || isSgAssignOp(n.getNode())) {
408  // Variable that corresponds to the value referenced by the array index or the new value of the assigned variable
409  varID refVar;
410  // The SeqStructLattice associated with refVar
411  SeqStructLattice* ssL = 0;
412 
413  if(isSgPntrArrRefExp(n.getNode())) {
414  ssL = dynamic_cast<SeqStructLattice*>(prodLat->getVarLattice(SgExpr2Var(isSgPntrArrRefExp(n.getNode()))));
415  ROSE_ASSERT(ssL);
416  refVar = SgExpr2Var(isSgPntrArrRefExp(n.getNode())->get_rhs_operand());
417  } else if(isSgAssignOp(n.getNode())) {
418  ssL = dynamic_cast<SeqStructLattice*>(prodLat->getVarLattice(SgExpr2Var(isSgAssignOp(n.getNode())->get_lhs_operand())));
419  // If the lhs of this assignment is live
420  if(ssL)
421  refVar = SgExpr2Var(isSgPntrArrRefExp(n.getNode())->get_rhs_operand());
422  // Otherwise, we're done
423  else
424  goto DONE;
425  }
426 
427  if(ssL->getLevel() == SeqStructLattice::bottom) {
428  modified = ssL->setToStartKnown(refVar, 0) || modified;
429  } else if(ssL->getLevel() == SeqStructLattice::top) {
430  // Do nothing, we can't represent this sequence
431  } else if(ssL->getLevel() == SeqStructLattice::startKnown) {
432  // If the last-known access was a fixed stride away from this access
433  int a, b, c;
434  if(cg->isEqVars(refVar, ssL->vInit, a, b, c) && a==1 && b==1) {
435  modified = ssL->setToSeqKnown(refVar, 0, c) || modified;
436  // Otherwise, this sequence is too complex to be represented
437  } else
438  modified = ssL->setToTop() || modified;
439  // If we've identified a sequence after analyzing multiple instances of this array access
440  } else if(ssL->getLevel() == SeqStructLattice::seqKnown) {
441  // If the stride of this access is equal to what was previously computed
442  int a, b, c;
443  if(cg->isEqVars(refVar, ssL->vInit, a, b, c) && a==1 && b==1 && c==ssL->s) {
444  modified = ssL->setToSeqKnown(refVar, 0, ssL->s) || modified;
445  // If the stride is different from what we've identified, this sequence is too complex to be represented
446  } else
447  modified = ssL->setToTop() || modified;
448  }
449  }
450 
451  DONE:
452 
453  // Finally, call the transfer function of the ConstrGraph to make sure that the current statement's effects are incorporated
454  modified = cgAnalysis->transfer(func, n, state, dfInfo) || modified;
455 
456  return modified;
