89 SAWYER_EXPORT std::string
90 escape(const std::string &s) {
91  std::string retval;
92  for (size_t i=0; i<s.size(); ++i) {
93  switch (s[i]) {
94  case '\a': retval += "\\a"; break;
95  case '\b': retval += "\\b"; break;
96  case '\t': retval += "\\t"; break;
97  case '\n': retval += "\\n"; break;
98  case '\v': retval += "\\v"; break;
99  case '\f': retval += "\\f"; break;
100  case '\r': retval += "\\r"; break;
101  case '\\': retval += "\\\\"; break;
102  default:
103  if (isprint(s[i])) {
104  retval += s[i];
105  } else {
106  char buf[8];
107 #include <Sawyer/WarningsOff.h>
108  snprintf(buf, sizeof(buf), "\\%03o", (unsigned)s[i]);
109 #include <Sawyer/WarningsRestore.h>
110  retval += buf;
111  }
112  break;
113  }
114  }
115  return retval;
116 }

119 SAWYER_EXPORT double
120 now() {
121 #if defined(SAWYER_HAVE_BOOST_CHRONO)
122  // Boost::chrono is thread-safe since we're using only stack allocated objects
123  boost::chrono::system_clock::time_point curtime = boost::chrono::system_clock::now();
124  boost::chrono::system_clock::time_point epoch;
125  boost::chrono::duration<double> diff = curtime - epoch;
126  return diff.count();
127 #elif defined(BOOST_WINDOWS)
128  FILETIME ft;
129  GetSystemTimeAsFileTime(&ft);
130  unsigned __int64 t = ft.dwHighDateTime;
131  t <<= 32;
132  t |= ft.dwLowDateTime;
133  t /= 10; // convert into microseconds
134  //t -= 11644473600000000Ui64; // convert file time to microseconds since Unix epoch
135  return t / 1e6;
136 #else // POSIX
137  struct timeval t;
138  if (-1==gettimeofday(&t, NULL))
139  return 0.0;
140  return t.tv_sec + 1e-6 * t.tv_usec;
141 #endif
142 }

1131 std::streamsize

1134 
1135  // This is called from std::ostream::operator<< (and possibly others), so we need to acquire a lock. Since this object is
1136  // owned by exactly one Stream object, the mutex we lock is in the stream object.
1137  assert(stream_!=NULL);

1139

1141  if (termination_symbol==s[i]) {

1143  } else if ('\r'!=s[i]) {
1144  message_->insert(s[i]);
1145  for (std::streamsize i=0; i<n; ++i) {
1146  if (isgraph(s[i])) {
1147  bake();
1148  break;
1149  }
1150  }
1151  }
1152  }
1153  post();
1154  return n;
1155 }
