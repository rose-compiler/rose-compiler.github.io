53 /* Parse a PE Import Section. This parses an entire PE Import Section, by recursively parsing the section's Import
54  * Directories. An Import Section is a sequence of Import Directories terminated by a zero-filled Import Directory struct.
55  * The terminating entry is not stored explicitly in the section's list of directories. */
56 SgAsmPEImportSection*
57 SgAsmPEImportSection::parse()
58 {
59  importMessageReset();
60  SgAsmPESection::parse();
61 
62  SgAsmPEFileHeader *fhdr = isSgAsmPEFileHeader(get_header());
63  ROSE_ASSERT(fhdr!=NULL);
64 
65  ROSE_ASSERT(isMapped());
66  Rose::BinaryAnalysis::Address idir_va = get_mappedActualVa();
67 
68  // Some compilers don't actually terminate the Import Directory list with an all-zero entry as specified in Microsoft's PE
69  // format documenttion. Instead, they apparently assume that that loader uses the size of the import section to calculate
70  // the length of the list and that it never actually reads the final entry which is normally all zero. These compilers
71  // seem to have garbage in the last entry.
72  Rose::BinaryAnalysis::Address nBytes = get_size();
73  ASSERT_require(nBytes > 0);
74  size_t nEntries = nBytes / sizeof(SgAsmPEImportDirectory::PEImportDirectory_disk);
75 
76  /* Parse each Import Directory. The list of directories is terminated with a zero-filled entry, which is not added to this
77  * import section. */
78  for (size_t i = 0; i < nEntries; i++) {
79  /* Import directory entry */
80  SgAsmPEImportDirectory *idir = new SgAsmPEImportDirectory(this);
81  if (NULL==idir->parse(idir_va, i+1 == nEntries)) {
82  /* We've reached the zero entry. Remove this directory from the section and delete it. */
83  removeImportDirectory(idir);
84  SageInterface::deleteAST(idir);
85  break;
86  }
87  idir_va += sizeof(SgAsmPEImportDirectory::PEImportDirectory_disk);
88 #if 1 /* FIXME: Do we really want this stuff duplicated in the AST? [RPM 2008-12-12] */
89  SgAsmGenericString *name2 = new SgAsmBasicString(idir->get_dllName()->get_string());
90  fhdr->addDll(new SgAsmGenericDLL(name2));
91 #endif
92  }
93  return this;
94 }

133 /* Reallocate space for the import section if necessary. This will likely not work correctly because it moves the import
134  * address tables. Unlike ELF, which tends to have dynamic linking trampolines and tables (PLT and GOT) in well defined
135  * locations, PE executables seem to have them haphazardly scattered throughout virtual memory. That means there's no easy
136  * way to fix things if we have to move the import address table. This whole subject of Import reallocation needs much
137  * improvement. */
138 bool
139 SgAsmPEImportSection::reallocate()
140 {
141  importMessageReset();
142  bool reallocated = SgAsmPESection::reallocate();
143  Rose::BinaryAnalysis::RelativeVirtualAddress end_rva(this->get_mappedPreferredRva(), this);
144  SgAsmPEImportDirectoryPtrList &dirlist = get_importDirectories()->get_vector();
145 
146  /* Space needed for the list of import directory structs. The list is terminated with a zero entry. */
147  size_t nimports = dirlist.size();
148  end_rva.increment((1 + nimports) * sizeof(SgAsmPEImportDirectory::PEImportDirectory_disk));
149 
150  /* Space needed for the data of each import directory. */
151  for (size_t i=0; i<nimports; i++)
152  end_rva.increment(dirlist[i]->reallocate(end_rva));
153 
154  /* Adjust the section size */
155  Rose::BinaryAnalysis::Address need = end_rva.boundOffset().orElse(end_rva.rva());
156  if (need < get_size()) {
157  if (isMapped())
158  set_mappedSize(need);
159  set_size(need);
160  reallocated = true;
161  } else if (need > get_size()) {
162  get_file()->shiftExtend(this, 0, need-get_size(), SgAsmGenericFile::ADDRSP_ALL, SgAsmGenericFile::ELASTIC_HOLE);
163  reallocated = true;
164  }
165 
166  return reallocated;
167 }

192 SgAsmPEImportSection::unparse(std::ostream &f) const
193 {
194  importMessageReset();
195 #if 1 /* DEBUGGING [RPM 2010-11-09] */

203  unparseHoles(f);
204 
205  /* Import Directory Entries and all they point to (even in other sections) */
206  for (size_t i=0; i<get_importDirectories()->get_vector().size(); i++) {
207  SgAsmPEImportDirectory *idir = get_importDirectories()->get_vector()[i];
208  try {
209  idir->unparse(f, this, i);

218  memset(&zero, 0, sizeof zero);
