91 LineVector::cacheLines(size_t nLineFeeds) const {
92  size_t n = nCharacters();
93  size_t i = lineFeeds_.empty() ? (size_t)0 : lineFeeds_.back() + 1;
94  i = std::max(i, nextCharToScan_);
95  while (lineFeeds_.size() < nLineFeeds && i < n) {
96  if ('\n' == charBuf_[i])
