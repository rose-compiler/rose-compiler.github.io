199 void
200 rebaseImportAddressTables(const Partitioner::Ptr &partitioner, const ImportIndex &index) {
201  ASSERT_not_null(partitioner);
202  const size_t wordSize = partitioner->instructionProvider().instructionPointerRegister().nBits() / 8;
203  if (wordSize > 8) {
204  mlog[WARN] <<"ModulesPe::rebaseImportAddressTable does not support a word size of "
205  <<StringUtility::plural(wordSize, "bytes") <<"\n";
206  return;
207  }
208 
209  // We need to make sure the underlying buffers are writable before we start writing to them. The easiest way to do this is to
210  // just make new buffers and map them over the top of any existing read-only buffers. First, gather the addresses so we're
211  // creating as few new buffers as possible.
212  AddressIntervalSet toWrite;
213  for (const ImportIndex::Node &node: index.nodes()) {
214  if (node.value()->get_iat_written())
215  continue;
216  const Address iatVa = node.value()->get_iatEntryVa();
217  toWrite |= AddressInterval::baseSize(iatVa, wordSize);
218  }
219  makeMemoryWritable(partitioner->memoryMap(), toWrite);
220 
221  // Write IAT entries into the newly mapped IATs
222  for (const ImportIndex::Node &node: index.nodes()) {
223  //First check if the loader already set the address
224  if (node.value()->get_iat_written())
225  continue;
226 
227  // Pack it as little-endian
228  uint8_t packed[8];
229  memset(packed, 0, 8);
230  for (size_t i=0; i<wordSize; ++i)
231  packed[i] = (node.key() >> (8*i)) & 0xff;
232 
233  // Then reorder bytes for other sexes
234  switch (partitioner->instructionProvider().defaultByteOrder()) {
235  case ByteOrder::ORDER_LSB:
236  break;
237 
238  case ByteOrder::ORDER_MSB:
239  std::reverse(packed+0, packed+wordSize);
240  break;
241 
242  case ByteOrder::ORDER_UNSPECIFIED:
243  ASSERT_not_reachable("unknown default byte order");
244  }
245 
246  Address iatVa = node.value()->get_iatEntryVa();
247  if (wordSize != partitioner->memoryMap()->at(iatVa).limit(wordSize).write(packed).size()){
248  ASSERT_not_reachable("write failed to map we just created");
249  }else{
250  node.value()->set_iat_written(true);
251  }
252  }

255 void
256 nameImportThunks(const Partitioner::ConstPtr &partitioner, SgAsmInterpretation *interp) {
257  ASSERT_not_null(partitioner);
258  if (interp == nullptr)
259  return;
260  std::vector<Function::Ptr> functions = partitioner->functions();
261 
262  // Get the addresses for the PE Import Address Tables
263  AddressIntervalSet iatExtent;
264  for (SgAsmGenericHeader *fileHeader: interp->get_headers()->get_headers()) {
265  SgAsmGenericSectionPtrList iatSections = fileHeader->get_sectionsByName("Import Address Table");
266  for (SgAsmGenericSection *section: iatSections) {
267  if (section->get_id()==-1 && section->isMapped())
268  iatExtent.insert(AddressInterval::baseSize(section->get_mappedActualVa(), section->get_mappedSize()));
269  }
270  }
271  if (iatExtent.isEmpty())
272  return;
273 
274  // Build an index that maps addresses to entries in the import tables. The addresses are the addresses where the imported
275  // functions are expected to be mapped.
276  ImportIndex importIndex = getImportIndex(partitioner, interp);
277 
278  // Process each function that's attached to the CFG/AUM
279  for (const Function::Ptr &function: functions) {
280  if (!function->name().empty())
281  continue; // no need to name functions that already have a name
282  if (function->basicBlockAddresses().size()!=1)
283  continue; // thunks have only one basic block...
284  BasicBlock::Ptr bblock = partitioner->basicBlockExists(function->address());
285  ASSERT_not_null(bblock);
286  if (bblock->nInstructions()!=1)
287  continue; // ...that contains only one instruction...
288  SgAsmX86Instruction *insn = isSgAsmX86Instruction(bblock->instructions().front());
289  if (!insn) {
290  static bool warned;
291  if (!warned) {
292  mlog[WARN] <<"ModulesPe::nameImportThunks does not yet support this ISA\n";
293  warned = true;
294  }
295  return;
296  }
297  if (insn->get_kind()!=x86_jmp || insn->nOperands() != 1)
298  continue; // ...that is a JMP...
299  SgAsmMemoryReferenceExpression *mre = isSgAsmMemoryReferenceExpression(insn->operand(0));
300  SgAsmIntegerValueExpression *addr = mre ? isSgAsmIntegerValueExpression(mre->get_address()) : nullptr;
301  if (!addr)
302  continue; // ...with addressing mode [C] where C is a constant...
303  AddressInterval indirectionAddress = AddressInterval::baseSize(addr->get_absoluteValue(), 4);
304  if (!iatExtent.contains(indirectionAddress))
305  continue; // ...and where C is inside an Import Address Table...
306  bool isComplete = true;
307  std::vector<Address> successors = partitioner->basicBlockConcreteSuccessors(bblock, &isComplete);
308  if (!isComplete || successors.size()!=1)
309  continue; // ...and the JMP has a single successor that is concrete...
310  SgAsmPEImportItem *importItem = nullptr;
311  if (!importIndex.getOptional(successors.front()).assignTo(importItem))
312  continue; // ...and is a known address for an imported function.
313 
314  // Merge the new name into the function
315  std::string importName = importItem->get_name()->get_string();
316  SgAsmPEImportDirectory *importDir = AST::Traversal::findParentTyped<SgAsmPEImportDirectory>(importItem);
317  if (importDir && !importDir->get_dllName()->get_string().empty())
318  importName += "@" + importDir->get_dllName()->get_string();
319  function->name(importName);
320  function->insertReasons(SgAsmFunction::FUNC_THUNK | SgAsmFunction::FUNC_IMPORT);
321  }

382 Sawyer::Optional<Address>
383 PeDescrambler::findCalleeAddress(const Partitioner::ConstPtr &partitioner, Address returnVa) {
384  ASSERT_not_null(partitioner);
385  size_t tableIdx = 0;
386  bool hitNullEntry = false;
387  while (1) {
388  if (tableIdx >= dispatchTable_.size() && !reachedEndOfTable_) {
389  // suck in some more table entries if possible
390  static const size_t nEntriesToRead = 512; // arbitrary
391  static const size_t wordsPerEntry = 2;
392  static const size_t bytesPerEntry = wordsPerEntry * 4;
393  static const size_t nWordsToRead = nEntriesToRead * wordsPerEntry;
394  static uint32_t buf[nWordsToRead];
395  Address batchVa = dispatchTableVa_ + dispatchTable_.size() * sizeof(DispatchEntry);
396  size_t nReadBytes = partitioner->memoryMap()->at(batchVa).limit(sizeof buf).read((uint8_t*)buf).size();
397  reachedEndOfTable_ = nReadBytes < bytesPerEntry;
398  for (size_t i=0; 4*(i+1)<nReadBytes; i+=2)
399  dispatchTable_.push_back(DispatchEntry(ByteOrder::leToHost(buf[i+0]), ByteOrder::leToHost(buf[i+1])));
400  SAWYER_MESG(mlog[DEBUG]) <<"read " <<StringUtility::plural(nReadBytes, "bytes") <<" from PEScrambler dispatch "
401  <<"table at " <<StringUtility::addrToString(batchVa) <<"\n";
402  }
403  if (tableIdx >= dispatchTable_.size())
404  return Sawyer::Nothing(); // couldn't find a suitable entry
405  if (dispatchTable_[tableIdx].returnVa == returnVa) {
406  Address va = dispatchTable_[tableIdx].calleeVa;
407  if (hitNullEntry) {
408  uint32_t va2;
409  if (4 != partitioner->memoryMap()->at(va).limit(4).read((uint8_t*)&va2).size())
410  return Sawyer::Nothing(); // couldn't dereference table entry
411  va = ByteOrder::leToHost(va2);
412  }
413  return va;
414  }
415  if (dispatchTable_[tableIdx].returnVa == 0)
416  hitNullEntry = true;
417  ++tableIdx;
418  }
