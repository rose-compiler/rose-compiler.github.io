111 void QEditor::gotoLine( int line, bool highlight )
112 {
113  setCursorPosition( line, 0 );
114  Q3TextParagraph *p = getDocument()->paragAt( line );
115  if ( !p )
116  return;
117  Q3TextCursor c( getDocument() );
118  emit ensureTextIsVisible( p );
119  c.setParagraph( p );
120  c.setIndex( 0 );
121  if (highlight) {
122  getDocument()->removeSelection( sel0 );
123  getDocument()->setSelectionStart( sel0, c );
124  }
125  c.gotoLineEnd();
126  if (highlight) {
127  getDocument()->setSelectionEnd( sel0, c );
128  }
129  viewport()->repaint( FALSE );
130 }

260 void QEditor::refresh()
261 {
262  getDocument()->invalidate();
263  Q3TextParagraph* p = getDocument()->firstParagraph();
264  while( p ){
265  p->format();
266  p = p->next();
267  }
268  getDocument()->removeSelection( ParenMatcher::Match );
269  getDocument()->removeSelection( ParenMatcher::Mismatch );
270  ensureCursorVisible();
271  repaintContents( false );
272 }
