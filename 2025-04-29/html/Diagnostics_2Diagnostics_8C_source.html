212 int StreamPrintf::operator()(const char *fmt, ...) {
213  char buf_[1024]; // arbitrary size; most strings will be smaller
214  char *buf = buf_;
215  int bufsz = sizeof buf_, need = 0;
216 
217  while (1) {
218  va_list ap;
219  va_start(ap, fmt);
220  need = vsnprintf(buf, bufsz, fmt, ap); // "need" does not include NUL termination
221  va_end(ap);
222  if (need >= bufsz) {
223  if (buf!=buf_)
224  delete[] buf;
225  bufsz = need + 1; // +1 for NUL termination
226  buf = new char[bufsz];
227  } else {
228  break;
229  }
230  }
231  stream <<buf;
232  if (buf!=buf_)
233  delete[] buf;
234 
235  return need;
236 }
