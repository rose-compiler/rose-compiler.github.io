131  const SmtSolverPtr&) const {

158 SValue::Ptr
159 SValue::instance_from_bits(size_t nbits, uint64_t possible_bits)
160 {
161  Intervals retval;
162  possible_bits &= IntegerOps::genMask<uint64_t>(nbits);
163 
164  size_t nset = 0, lobit=nbits, hibit=0;
165  for (size_t i=0; i<nbits; ++i) {
166  if (0 != (possible_bits & IntegerOps::shl1<uint64_t>(i))) {
167  ++nset;
168  lobit = std::min(lobit, i);
169  hibit = std::max(hibit, i);
170  }
171  }
172 
173  if (possible_bits == IntegerOps::genMask<uint64_t>(nset)) {
174  // The easy case: all possible bits are grouped together at the low end.
175  retval.insert(Interval::hull(0, possible_bits));
176  } else {
177  // Low-order bit of result must be clear, so the rangemap will have 2^nset ranges
178  uint64_t nranges = IntegerOps::shl1<uint64_t>(nset); // 2^nset
179  if (nranges>maxComplexity) {
180  uint64_t hi = IntegerOps::genMask<uint64_t>(hibit+1) ^ IntegerOps::genMask<uint64_t>(lobit);
181  retval.insert(Interval::hull(0, hi));
182  } else {
183  for (uint64_t i=0; i<nranges; ++i) {
184  uint64_t lo=0, tmp=i;
185  for (uint64_t j=0; j<nbits; ++j) {
186  uint64_t bit = IntegerOps::shl1<uint64_t>(j);
187  if (0 != (possible_bits & bit)) {
188  lo |= (tmp & 1) << j;
189  tmp = tmp >> 1;
190  }
191  }
192  retval.insert(lo);
193  }
194  }
195  }
196  return SValue::instance_intervals(nbits, retval);
197 }

240 uint64_t
241 SValue::possible_bits() const
242 {
243  uint64_t bits = 0;
244  for (const Interval &interval: intervals_.intervals()) {
245  uint64_t lo = interval.least(), hi = interval.greatest();
246  bits |= lo | hi;
247  for (uint64_t bitno=0; bitno<nBits(); ++bitno) {
248  uint64_t bit = IntegerOps::shl1<uint64_t>(bitno);
249  if (0 == (bits & bit)) {
250  uint64_t base = lo & ~IntegerOps::genMask<uint64_t>(bitno);
251  if (interval.contains(base+bit))
252  bits |= bit;
253  }
254  }
255  }
256  return bits;
257 }

453 RiscOperators::and_(const BaseSemantics::SValue::Ptr &a_, const BaseSemantics::SValue::Ptr &b_)
454 {
455  SValue::Ptr a = SValue::promote(a_);
456  SValue::Ptr b = SValue::promote(b_);
457  ASSERT_require(a->nBits()==b->nBits());
458  if (a->isBottom() || b->isBottom())
459  return bottom_(a->nBits());

462  uint64_t au = a->isConcrete() ? 0 : a->possible_bits();
463  uint64_t bu = b->isConcrete() ? 0 : b->possible_bits();
464  uint64_t r = ak & bk & au & bu;
465  if (au || bu)
466  return svalue_from_bits(a->nBits(), r);
467  return number_(a->nBits(), r);

522 BaseSemantics::SValue::Ptr
523 RiscOperators::extract(const BaseSemantics::SValue::Ptr &a_, size_t begin_bit, size_t end_bit)
524 {
525  using namespace IntegerOps;
526  SValue::Ptr a = SValue::promote(a_);
527  ASSERT_require(end_bit<=a->nBits());
528  ASSERT_require(begin_bit<end_bit);
529  if (a->isBottom())
530  return bottom_(a->nBits());
531  Intervals result;
532  uint64_t discard_mask = ~genMask<uint64_t>(end_bit); // hi-order bits being discarded
533  uint64_t src_mask = genMask<uint64_t>(a->nBits()); // significant bits in the source operand
534  uint64_t dst_mask = genMask<uint64_t>(end_bit-begin_bit); // significant bits in the result
535  for (const Interval &iv: a->get_intervals().intervals()) {
536  uint64_t d1 = shiftRightLogical2(iv.least() & discard_mask, end_bit, a->nBits()); // discarded part, lo
537  uint64_t d2 = shiftRightLogical2(iv.greatest() & discard_mask, end_bit, a->nBits()); // discarded part, hi
538  uint64_t k1 = shiftRightLogical2(iv.least() & src_mask, begin_bit, a->nBits()) & dst_mask; // keep part, lo
539  uint64_t k2 = shiftRightLogical2(iv.greatest() & src_mask, begin_bit, a->nBits()) & dst_mask; // keep part, hi
540  if (d1==d2) { // no overflow in the kept bits
541  ASSERT_require(k1<=k2);
542  result.insert(Interval::hull(k1, k2));
543  } else if (d1+1<d2 || k1<k2) { // complete overflow
544  result.insert(Interval::hull(0, dst_mask));
545  break;
546  } else { // partial overflow
547  result.insert(Interval::hull(0, k2));
548  result.insert(Interval::hull(k1, dst_mask));
549  }
550  }
551  return svalue_from_intervals(end_bit-begin_bit, result);
552 }

554 BaseSemantics::SValue::Ptr
555 RiscOperators::concat(const BaseSemantics::SValue::Ptr &a_, const BaseSemantics::SValue::Ptr &b_)
556 {
557  SValue::Ptr a = SValue::promote(a_);
558  SValue::Ptr b = SValue::promote(b_);
559  size_t retsize = a->nBits() + b->nBits();
560  if (a->isBottom() || b->isBottom())
561  return bottom_(a->nBits() + b->nBits());
562  Intervals result;
563  uint64_t mask_a = IntegerOps::genMask<uint64_t>(a->nBits());
564  uint64_t mask_b = IntegerOps::genMask<uint64_t>(b->nBits());
565  for (const Interval &av: a->get_intervals().intervals()) {
566  for (const Interval &bv: b->get_intervals().intervals()) {
567  uint64_t lo = (IntegerOps::shiftLeft2(bv.least() & mask_b, retsize, a->nBits()) |
568  (av.least() & mask_a));
569  uint64_t hi = (IntegerOps::shiftLeft2(bv.greatest() & mask_b, retsize, a->nBits()) |
570  (av.greatest() & mask_a));
571  result.insert(Interval::hull(lo, hi));
572  }
573  }
574  return svalue_from_intervals(retsize, result);

577 BaseSemantics::SValue::Ptr
578 RiscOperators::leastSignificantSetBit(const BaseSemantics::SValue::Ptr &a_)
579 {
580  SValue::Ptr a = SValue::promote(a_);
581  size_t nbits = a->nBits();
582  if (a->isBottom())
583  return bottom_(nbits);
584  if (auto av = a->toUnsigned()) {
585  if (*av) {
586  for (size_t i=0; i<nbits; ++i) {
587  if (*av & IntegerOps::shl1<uint64_t>(i))
588  return number_(nbits, i);
589  }
590  }
591  return number_(nbits, 0);
592  }
593 
594  uint64_t abits = a->possible_bits();
595  ASSERT_require(abits!=0); // or else the value would be known to be zero and handled above
596  uint64_t lo=nbits, hi=0;
597  for (size_t i=0; i<nbits; ++i) {

871 BaseSemantics::SValue::Ptr
872 RiscOperators::add(const BaseSemantics::SValue::Ptr &a_, const BaseSemantics::SValue::Ptr &b_)
873 {
874  SValue::Ptr a = SValue::promote(a_);
875  SValue::Ptr b = SValue::promote(b_);
876  ASSERT_require(a->nBits()==b->nBits());
877  size_t nbits = a->nBits();
878  if (a->isBottom() || b->isBottom())
879  return bottom_(nbits);
880 
881  const Intervals &aints=a->get_intervals(), &bints=b->get_intervals();
882  Intervals result;
883  for (const Interval &av: aints.intervals()) {
884  for (const Interval &bv: bints.intervals()) {
885  uint64_t lo = (av.least() + bv.least()) & IntegerOps::genMask<uint64_t>(nbits);
886  uint64_t hi = (av.greatest() + bv.greatest()) & IntegerOps::genMask<uint64_t>(nbits);
887  if (lo < av.least() || lo < bv.least()) {
888  // lo and hi both overflow
889  result.insert(Interval::hull(lo, hi));
890  } else if (hi < av.greatest() || hi < bv.greatest()) {
891  // hi overflows, but not lo
892  result.insert(Interval::hull(lo, IntegerOps::genMask<uint64_t>(nbits)));
893  result.insert(Interval::hull(0, hi));
894  } else {
895  // no overflow
896  result.insert(Interval::hull(lo, hi));
897  }
898  }
899  }
900  return svalue_from_intervals(nbits, result);
901 }

903 BaseSemantics::SValue::Ptr
904 RiscOperators::addWithCarries(const BaseSemantics::SValue::Ptr &a_, const BaseSemantics::SValue::Ptr &b_,
905  const BaseSemantics::SValue::Ptr &c_, BaseSemantics::SValue::Ptr &carry_out/*out*/)
906 {
907  SValue::Ptr a = SValue::promote(a_);
908  SValue::Ptr b = SValue::promote(b_);
909  ASSERT_require(a->nBits()==b->nBits());
910  size_t nbits = a->nBits();
911  SValue::Ptr c = SValue::promote(c_);
912  ASSERT_require(c->nBits()==1);
913  if (a->isBottom() || b->isBottom() || c->isBottom())
914  return bottom_(nbits);
915 
916  SValue::Ptr wide_carry = SValue::promote(unsignedExtend(c, nbits));
917  SValue::Ptr retval = SValue::promote(add(add(a, b), wide_carry));
918 
919  uint64_t known1 = a->toUnsigned().orElse(0);
920  uint64_t known2 = b->toUnsigned().orElse(0);
921  uint64_t known3 = c->toUnsigned().orElse(0);
922  uint64_t unkwn1 = a->toUnsigned() ? 0 : a->possible_bits();
923  uint64_t unkwn2 = b->toUnsigned() ? 0 : b->possible_bits();
924  uint64_t unkwn3 = c->toUnsigned() ? 0 : c->possible_bits();
925  enum Carry { C_FALSE, C_TRUE, C_UNKNOWN };
926  Carry cin = C_FALSE; // carry propagated across loop iterations
927  uint64_t known_co=0, unkwn_co=0; // known or possible carry-out bits
928  for (size_t i=0; i<nbits; ++i, known1>>=1, known2>>=1, known3>>=1, unkwn1>>=1, unkwn2>>=1, unkwn3>>=1) {
929  int known_sum = (known1 & 1) + (known2 & 1) + (known3 & 1) + (C_TRUE==cin ? 1 : 0);
930  int unkwn_sum = (unkwn1 & 1) + (unkwn2 & 1) + (unkwn3 & 1) + (C_UNKNOWN==cin ? 1 : 0);
931  if (known_sum>1) {
932  known_co |= IntegerOps::shl1<uint64_t>(i);
933  cin = C_TRUE;
934  } else if (known_sum + unkwn_sum > 1) {
935  unkwn_co |= IntegerOps::shl1<uint64_t>(i);
936  cin = C_UNKNOWN;
937  }
938  }
939  if (unkwn_co) {
940  carry_out = svalue_from_bits(nbits, known_co | unkwn_co);
941  } else {
942  carry_out = number_(nbits, known_co);
943  }
944  return retval;
945 }
