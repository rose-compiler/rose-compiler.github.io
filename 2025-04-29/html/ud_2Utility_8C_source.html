123 // Returns the Damerau-Levenshtein edit distance.
124 size_t
125 editDistance(const std::string &src, const std::string &tgt) {
126  // Based on the C# implementation on the wikipedia page
127  if (src.empty() || tgt.empty())
128  return std::max(src.size(), tgt.size());
129 
130  const size_t x = src.size();
131  const size_t y = tgt.size();
132  std::vector<std::vector<size_t> > score(x+2, std::vector<size_t>(y+2, 0));
133  size_t score_ceil = x + y;
134  score[0][0] = score_ceil;
135  for (size_t i=0; i<=x; ++i) {
136  score[i+1][1] = i;
137  score[i+1][0] = score_ceil;
138  }
139  for (size_t j=0; j<=y; ++j) {
140  score[1][j+1] = j;
141  score[0][j+1] = score_ceil;
142  }
143 
144  LevenshteinStack dict;
145  for (size_t i=0; i<x; ++i)
146  dict.unique_push_zero(src[i]);
147  for (size_t j=0; j<y; ++j)
148  dict.unique_push_zero(tgt[j]);
149 
150  for (size_t i=1; i<=x; ++i) {
151  size_t db = 0;
152  for (size_t j=1; j<=y; ++j) {
153  size_t i1 = dict[tgt[j-1]];
154  size_t j1 = db;
155  if (src[i-1]==tgt[j-1]) {
156  score[i+1][j+1] = score[i][j];
157  db = j;
158  } else {
159  score[i+1][j+1] = std::min(score[i][j], std::min(score[i+1][j], score[i][j+1])) + 1;
160  }
161  // swaps
162  score[i+1][j+1] = std::min(score[i+1][j+1], score[i1][j1] + (i-i1-1) + 1 + (j-j1-1));
163  }
164  dict[src[i-1]] = i;
165  }
166 
167  return score[x+1][y+1];
168 }

439 appendToDoxygen(const std::string &comment, const std::string &newText) {
440  std::vector<std::string> commentLines = splitIntoLines(comment);
441  const std::vector<std::string> newTextLines = splitIntoLines(newText);
442 
443  // Remove the last line of the comment if it contains only the C-style closing. Otherwise just remove the C-style closing from
444  // the lat line.
445  const std::regex closingOnlyRe("[^a-zA-Z0-9]*\\*/[ \t]*");
446  const std::regex closingRe("(.*)\\*/[ \t]*");
447  if (!commentLines.empty()) {
448  std::smatch parts;
449  if (std::regex_match(commentLines.back(), closingOnlyRe)) {
450  commentLines.pop_back();
451  } else if (std::regex_match(commentLines.back(), parts, closingRe)) {
452  commentLines.back() = parts.str(1);
453  }
454  }
455 
456  // What is the prefix for the previous lines of the comment that contain text.
457  const std::string prefix = [&commentLines]() -> std::string {
458  if (commentLines.empty()) {
459  return "/** ";
460  } else {
461  const std::regex nonEmptyPrefixRe("^([^a-zA-Z0-9@\\\\]*)[a-zA-Z0-9@\\\\]");
462  const std::regex startOfCommentRe("^([ \\t]*)/\\*");
463  std::smatch parts;
464  for (size_t i = commentLines.size(); i > 0; --i) {
465  const std::string &line = commentLines[i-1];
466  if (1 == i && std::regex_search(line, parts, startOfCommentRe)) {
467  return parts.str(1) + " * ";
468  } else if (std::regex_search(line, parts, nonEmptyPrefixRe)) {
469  return parts.str(1);
470  }
471  }
472  return " * ";
473  }
474  }();
475 
476  // Build the new comment and add it to the end of the existing comment.
477  const std::vector<std::string> newCommentLines = makeBlockComment(newTextLines, prefix);
478  commentLines.insert(commentLines.end(), newCommentLines.begin(), newCommentLines.end());
479  return boost::join(commentLines, "\n");
480 }

633 checkClassHierarchy(Hierarchy &h) {
634  // Any cycles in the graph?
635  using namespace Sawyer::Container::Algorithm;
636  std::vector<bool> visited(h.nVertices(), false);
637  for (size_t rootId = 0; rootId < h.nVertices(); ++rootId) {
638  if (visited[rootId])
639  continue;

641  path.reserve(h.nVertices());
642  path.push_back(rootId);
643  std::vector<bool> onPath(h.nVertices(), false);
644  onPath[rootId] = true;
645  using Traversal = DepthFirstForwardGraphTraversal<Hierarchy>;
646  for (Traversal t(h, h.findVertex(rootId), EDGE_EVENTS); t; ++t) {
647  const auto target = t.edge()->target();
648  if (t.event() == ENTER_EDGE) {
649  if (onPath[target->id()]) {
650  auto curFile = target->value()->findAncestor<Ast::File>();
651  message(ERROR, curFile, target->value()->nameToken, "cycle in class hierarchy");
652  for (size_t nextId: boost::adaptors::reverse(path)) {
653  auto next = h.findVertex(nextId);
654  ASSERT_require(next != h.vertices().end());
655  auto nextFile = next->value()->findAncestor<Ast::File>();
656  message(INFO, nextFile, next->value()->nameToken, "inherits from here");
657  }
658  }
659  onPath[target->id()] = true;
660  path.push_back(target->id());
661  if (visited[target->id()]) {

663  } else {

837  std::smatch found;

839  filtered += prior;

841  } else {

843  }
844  tokens.consume();
845  } else {

847  break;
848  }
849  }

851  }
