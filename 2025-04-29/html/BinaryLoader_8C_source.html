195 void
196 BinaryLoader::link(SgAsmInterpretation* interp) {
197  ASSERT_not_null(interp);
198  SgBinaryComposite *composite = SageInterface::getEnclosingNode<SgBinaryComposite>(interp);
199  ASSERT_not_null(composite);
200 
201  /* Make sure the pre-load objects are parsed and linked into the AST. */
202  for (std::vector<std::string>::const_iterator pi=preloads_.begin(); pi!=preloads_.end(); ++pi) {
203  mlog[TRACE] <<"preload object " <<*pi <<"\n";
204  std::string filename = findSoFile(*pi);
205  if (isLinked(composite, filename)) {
206  mlog[TRACE] <<filename <<" is already parsed.\n";
207  } else {
208  Stream m1(mlog[TRACE] <<"parsing " <<filename);
209  createAsmAST(composite, filename);
210  m1 <<"... done.\n";
211  }
212  }
213 
214  /* Bootstrap */
215  std::list<SgAsmGenericHeader*> unresolved_hdrs;
216  unresolved_hdrs.insert(unresolved_hdrs.end(),
217  interp->get_headers()->get_headers().begin(),
218  interp->get_headers()->get_headers().end());
219 
220  /* Process unresolved headers from the beginning of the queue and push new ones onto the end. */
221  while (!unresolved_hdrs.empty()) {
222  SgAsmGenericHeader *header = unresolved_hdrs.front();
223  unresolved_hdrs.pop_front();
224  std::string header_name = header->get_file()->get_name();
225  std::vector<std::string> deps = dependencies(header);
226  for (std::vector<std::string>::iterator di=deps.begin(); di!=deps.end(); ++di) {
227  mlog[TRACE] <<"library " <<*di <<" needed by " <<header_name <<"\n";
228  std::string filename = findSoFile(*di);
229  if (isLinked(composite, filename)) {
230  mlog[TRACE] <<filename <<" is already parsed.\n";
231  } else {
232  Stream m1(mlog[TRACE] <<"parsing " <<filename);
233  SgAsmGenericFile *new_file = createAsmAST(composite, filename);
234  m1 <<"... done.\n";
235  ASSERT_not_null2(new_file, "createAsmAST failed");
236  SgAsmGenericHeaderPtrList new_hdrs = findSimilarHeaders(header, new_file->get_headers()->get_headers());
237  unresolved_hdrs.insert(unresolved_hdrs.end(), new_hdrs.begin(), new_hdrs.end());
238  }
239  }
240  }

329 /* Maps the sections of a single header. */
330 void
331 BinaryLoader::remap(MemoryMap::Ptr &map, SgAsmGenericHeader *header) {
332  SgAsmGenericFile *file = header->get_file();
333  ASSERT_not_null(file);
334 
335  Stream trace(mlog[TRACE]);
336 
337  trace <<"remapping sections of " <<header->get_file()->get_name() <<"\n";
338  SgAsmGenericSectionPtrList sections = getRemapSections(header);
339 
340  Address old_base_va = header->get_baseVa();
341  Address new_base_va = rebase(map, header, sections);
342  if (new_base_va != old_base_va) {
343  trace <<" temporarily rebasing header from " <<StringUtility::addrToString(old_base_va)
344  <<" to " <<StringUtility::addrToString(new_base_va) <<"\n";
345  header->set_baseVa(new_base_va);
346  }
347 
348  try {
349  for (auto section: sections) {
350  section->set_mappedActualVa(0); /*reset in case previously mapped*/
351  unsigned mapperms = mappingPermissions(section);
352 
353  if (trace) {
354  trace <<" mapping section [" <<section->get_id() <<"] \"" <<section->get_name()->get_string(true) <<"\"";
355  if (section->get_baseVa()!=0)
356  trace <<" with base va " <<StringUtility::addrToString(section->get_baseVa());
357  trace <<"\n";
358  trace <<" specified RVA: " <<StringUtility::addrToString(section->get_mappedPreferredRva())
359  <<" + " <<StringUtility::addrToString(section->get_mappedSize()) <<" bytes"
360  <<" = " <<StringUtility::addrToString(section->get_mappedPreferredRva()+section->get_mappedSize())
361  <<"\n";
362  if (section->get_baseVa()!=0) {
363  trace <<" specified VA: "
364  <<StringUtility::addrToString(section->get_baseVa() + section->get_mappedPreferredRva()) <<" + "
365  <<StringUtility::addrToString(section->get_mappedSize()) <<" bytes = "
366  <<StringUtility::addrToString(section->get_baseVa() + section->get_mappedPreferredRva() +
367  section->get_mappedSize()) <<"\n";
368  }
369  trace <<" specified offset: "
370  <<StringUtility::addrToString(section->get_offset()) <<" + "
371  <<StringUtility::addrToString(section->get_size()) <<" bytes = "
372  <<StringUtility::addrToString(section->get_offset()+section->get_size()) <<"\n";
373  trace <<" specified alignment: memory="
374  <<StringUtility::addrToString(section->get_mappedAlignment()) <<", file="
375  <<StringUtility::addrToString(section->get_fileAlignment()) <<"\n";
376  }
377 
378  // Initial guesses
379  Address malign_lo = std::max(section->get_mappedAlignment(), (Address)1);
380  Address malign_hi = std::min(std::max(section->get_mappedAlignment(), (Address)1), (Address)4096);
381  Address va = header->get_baseVa() + section->get_mappedPreferredRva();
382  Address mem_size = section->get_mappedSize();
383  Address offset = section->get_offset();
384  Address file_size = section->get_size();
385 
386  /* Figure out alignment, etc. */
387  Address va_offset=0;
388  bool anon_lo=true, anon_hi=true, map_private=false;
389  ConflictResolution resolve = RESOLVE_THROW;
390  MappingContribution contrib = alignValues(section, map, /* inputs */
391  &malign_lo, &malign_hi, /* alignment outputs */
392  &va, &mem_size, /* memory location outputs */
393  &offset, &file_size, &map_private, /* file location outputs */
394  &va_offset, &anon_lo, &anon_hi, /* internal location outputs */
395  &resolve); /* conflict resolution output */
396  Address falign_lo = std::max(section->get_fileAlignment(), (Address)1);
397  Address falign_hi = falign_lo;
398 
399  if (trace) {
400  if (CONTRIBUTE_NONE==contrib || 0==mem_size) {
401  trace <<" does not contribute to map\n";
402  } else {
403  trace <<" adjusted alignment: memory=["
404  <<StringUtility::addrToString(malign_lo) <<".."
405  <<StringUtility::addrToString(malign_hi) <<"], file=["
406  <<StringUtility::addrToString(falign_lo) <<".."
407  <<StringUtility::addrToString(falign_hi) <<"]\n";
408  trace <<" aligned VA: "
409  <<StringUtility::addrToString(va) <<" + "
410  <<StringUtility::addrToString(mem_size) <<" bytes = "
411  <<StringUtility::addrToString(va+mem_size);
412  if (section->get_baseVa()+section->get_mappedPreferredRva()==va &&
413  section->get_mappedSize()==mem_size) {
414  trace <<" (no change)\n";
415  } else {
416  trace <<"\n";
417  }
418  if (va < new_base_va) {
419  trace <<" WARNING: aligned va " <<StringUtility::addrToString(va) <<" is less than "
420  <<(new_base_va==old_base_va?"":"temporary ") <<"base va "
421  <<StringUtility::addrToString(new_base_va) <<"\n";
422  }
423  if (CONTRIBUTE_ADD==contrib) {
424  trace <<" aligned offset: " <<StringUtility::addrToString(offset) <<" + "
425  <<StringUtility::addrToString(file_size) <<" bytes = "
426  <<StringUtility::addrToString(offset+file_size)
427  <<(section->get_offset()==offset && section->get_size()==file_size ? " (no change)\n" : "\n");
428  trace <<" permissions: "
429  <<((mapperms & MemoryMap::READABLE) ? "r" : "-")
430  <<((mapperms & MemoryMap::WRITABLE) ? "w" : "-")
431  <<((mapperms & MemoryMap::EXECUTABLE) ? "x" : "-");
432  if ((mapperms & ~MemoryMap::READ_WRITE_EXECUTE) != 0)
433  mfprintf(trace)("0x%x", (mapperms & ~MemoryMap::READ_WRITE_EXECUTE));
434  trace <<"\n";
435  trace <<" internal offset: " <<StringUtility::addrToString(va_offset)
436  <<" (va " <<StringUtility::addrToString(va+va_offset) <<")\n";
437  }
438  }
439  }
440 
441  /* Sanity checks */
442  if (CONTRIBUTE_NONE==contrib || 0==mem_size)
443  continue;
444  ASSERT_require(va_offset<mem_size);
445  if (file_size>mem_size) file_size = mem_size;
446  ASSERT_require(va + va_offset >= header->get_baseVa());
447  if (trace) {
448  trace <<" current memory map (before we map this section)\n";
449  map->dump(trace, " ");
450  }
451 
452  /* Erase part of the mapping? */
453  if (CONTRIBUTE_SUB==contrib) {
454  trace <<" subtracting contribution\n";
455  map->erase(AddressInterval::baseSize(va, mem_size));
456  continue;
457  }
458 
459  // Resole mapping conflicts
460  if (map->within(AddressInterval::baseSize(va, mem_size)).exists()) {
461  switch (resolve) {
462  case RESOLVE_THROW:
463  throw MemoryMap::NotMapped("cannot map segment", map, va);
464  case RESOLVE_OVERMAP:
465  trace <<" conflict: resolved by making a hole\n";
466  map->erase(AddressInterval::baseSize(va, mem_size));
467  break;
468  case RESOLVE_REMAP:
469  case RESOLVE_REMAP_ABOVE: {
470  trace <<" unable to map entire desired region.\n";
471  AddressInterval where = AddressInterval::hull(RESOLVE_REMAP_ABOVE==resolve ? va : 0,
472  AddressInterval::whole().greatest());
473  Address new_va = 0;
474  if (!map->findFreeSpace(mem_size, malign_lo, where).assignTo(new_va)) {
475  throw MemoryMap::NoFreeSpace("unable to allocate space in specimen memory map",
476  map, mem_size);
477  }
478  ASSERT_require2(0 == (new_va+mem_size) % malign_hi, "FIXME: not handled yet [RPM 2010-09-03]");
479  va = new_va;
480  trace <<" relocated to VA: " <<StringUtility::addrToString(va) <<" + "
481  <<StringUtility::addrToString(mem_size) <<" bytes = "
482  <<StringUtility::addrToString(va + mem_size) <<"\n";
483  break;
484  }
485  }
486  }
487 
488  /* Save the virtual address where this section is (will be) mapped. When a section is mapped more than once
489  * (perfectly legal to do so) only the last mapping is saved. */
490  section->set_mappedActualVa(va + va_offset);
491 
492  /* Segment name for debugging. This is the file base name and section name concatenated. */
493  std::string::size_type file_basename_pos = file->get_name().find_last_of("/");
494  file_basename_pos = file_basename_pos==file->get_name().npos ? 0 : file_basename_pos+1;
495  std::string melmt_name = file->get_name().substr(file_basename_pos) + "(" + section->get_name()->get_string() + ")";
496  trace <<" map element name: " <<escapeString(melmt_name) <<"\n";
497 
498  /* Anonymously map the part of memory beyond the physical end of the file */
499  SgAsmGenericFile *file = section->get_file();
500  Address total = file->get_data().size(); /*total size of file*/
501  if (offset+mem_size > total) {
502  Address n, a;
503  if (offset >= total) {
504  /* starts beyond EOF */
505  n = mem_size;
506  a = va;
507  } else {
508  /* overlaps EOF */
509  n = (offset + mem_size) - total;
510  a = va + total - offset;
511  }
512  trace <<" mapping part beyond EOF(" <<StringUtility::addrToString(total) <<"): "
513  <<"va=" <<StringUtility::addrToString(a) <<" + " <<StringUtility::addrToString(n) <<" = "
514  <<StringUtility::addrToString(a+n) <<"\n";
515  map->insert(AddressInterval::baseSize(a, n),
516  MemoryMap::Segment::anonymousInstance(n, mapperms|MemoryMap::PRIVATE, melmt_name));
517  mem_size -= n;
518  file_size = std::min(file_size, mem_size);
519  }
520 
521  /* Anonymously map the part of memory beyond the part of file */
522  if (anon_hi && mem_size>file_size) {
523  Address n = mem_size - file_size;
524  Address a = va + file_size;
525  trace <<" mapping part beyond end of section: va="
526  <<StringUtility::addrToString(a) <<" + " <<StringUtility::addrToString(n) <<" = "
527  <<StringUtility::addrToString(a+n) <<"\n";
528  map->insert(AddressInterval::baseSize(a, n),
529  MemoryMap::Segment::anonymousInstance(n, mapperms|MemoryMap::PRIVATE, melmt_name));
530  mem_size -= n;
531  }
532 
533  /* Anonymously map the part of memory before the section */
534  if (anon_lo && va_offset>0 && mem_size>0) {
535  Address n = va_offset - va;
536  Address a = va;
537  trace <<" mapping part before beginning of section: va="
538  <<StringUtility::addrToString(a) <<" + " <<StringUtility::addrToString(n) <<" = "
539  <<StringUtility::addrToString(a+n) <<"\n";
540  map->insert(AddressInterval::baseSize(a, n),
541  MemoryMap::Segment::anonymousInstance(n, mapperms|MemoryMap::PRIVATE, melmt_name));
542  mem_size -= n;
543  file_size -= n;
544  va += n;
545  offset += n;
546  }
547 
548  /* Map the section. We use the file content as the underlying storage of the map because we might be mapping parts of
549  * the file left and right of the actual section. */
550  if (mem_size>0) {
551  trace <<" mapping section: va="
552  <<StringUtility::addrToString(va) <<" + " <<StringUtility::addrToString(mem_size) <<" = "
553  <<StringUtility::addrToString(va+mem_size) <<" "
554  <<(map_private?"private":"shared") <<"\n";

599 int64_t
600 BinaryLoader::gcd(int64_t a, int64_t b, int64_t *xout/*=NULL*/, int64_t *yout/*=NULL*/) {
601  uint64_t x=0, xprev=1, y=1, yprev=0;
602  bool swapped = false;
603  if (b>a) {
604  std::swap(a, b);
605  swapped = true;
606  }
607 
608  while (b!=0) {
609  uint64_t quotient = a / b;
610  uint64_t temp;
611 
612  temp = b;
613  b = a % b;
614  a = temp;
615 
616  if (xout) {
617  temp = x;
618  x = xprev - quotient * x;
619  xprev = temp;
620  }
621 
622  if (yout) {
623  temp = y;
624  y = yprev - quotient * y;
625  yprev = temp;
626  }
627  }
628 
629  if (swapped)
630  std::swap(xprev, yprev);
631 
632  if (xout) *xout = xprev;
633  if (yout) *yout = yprev;
634  return a;
635 }

637 Address
638 BinaryLoader::bialign(Address val1, Address align1, Address val2, Address align2) {
639  using namespace StringUtility;
640 
641  if (0==val1 % align1 && 0==val2 % align2)
642  return 0;
643  Stream trace(mlog[TRACE]);
644 
645  /* Minimum amount by which the addresses must be adjusted downward to independently meet their alignment constraint. */
646  int64_t Ma = val1 - alignDown(val1, align1);
647  int64_t Mb = val2 - alignDown(val2, align2);
648  if (Ma > Mb) {
649  std::swap(val1, val2);
650  std::swap(align1, align2);
651  std::swap(Ma, Mb);
652  }
653 
654  SAWYER_MESG(trace) <<" aligning " <<addrToString(val1) <<" to " <<addrToString(align1)
655  <<" and " <<addrToString(val2) <<" to " <<addrToString(align2) <<"\n"
656  <<" misalignment: Ma=" <<addrToString(Ma) <<", Mb=" <<addrToString(Mb) <<"\n";
657 
658  /* Alignment constraints that must both be satisfied. */
659  int64_t a = align1;
660  int64_t b = align2;
661  int64_t c = Mb - Ma;
662  SAWYER_MESG(trace) <<" constraints: a=" <<addrToString(a) <<", b=" <<addrToString(b) <<", c=" <<addrToString(c) <<"\n";
663 
664  /* Alignment is satsfied when:
665  *
666  * x a + Ma = y b + Mb (1)
667  *
668  * Rearranging, we get the linear Biophantine equation:
669  *
670  * x a + y (-b) = Mb - Ma = c (2)
671  *
672  * This has solutions if and only if c is a multiple of the greatest common divisor of a and b. */
673  int64_t t, u;
674  int64_t g = gcd(a, b, &t, &u);
675  SAWYER_MESG(trace) <<" bezout coef: t=" <<addrToString(t) <<", u=" <<addrToString(u)
676  <<", gcd(a,b)=" <<addrToString(g) <<"\n";
677  if (c % g) {
678  trace <<" no solutions (Mb-Ma not a multiple of gcd(a,b))\n";
679  throw Exception("no solutions to alignment constraints");
680  }
681  ASSERT_require(t*a + u*b == g);
682 
683  /* The t and u are Bezout coefficients that satisfy
684  *
685  * t a + u b = g (3)
686  *
687  * By multiplying both sides by c/g we get a solution for (2) above:
688  *
689  * m = c/g
690  * (m t) (a) + (m u) (-b) = c
691  *
692  * Substituting:
693  * x = m t
694  * y = m u
695  * c = Mb - Ma
696  *
697  * and rearranging gives us (1) above. Therefore, both alignment constraints can be satisfied simultaneously
698  * by subtracting a common amount, namely:
699  *
700  * x a + Ma or
701  * y b + Mb
702  */
703  ASSERT_require(c % g == 0);
704  int64_t m = c/g;
705 
706  /* Calculate adjustment */
707  int64_t Aa = m * t * a + Ma;
708  int64_t Ab = m * u * b + Mb;
709  SAWYER_MESG(trace) <<" adjustment: " <<addrToString(Aa) <<"\n";
710  ASSERT_always_require(Aa==Ab);
711  ASSERT_always_require2(Aa>0, "FIXME[Robb Matzke 2010-09-07]: add multiples of lcm(a,b) to make this positive");
712  return (Address)Aa;
713 }

715 BinaryLoader::MappingContribution
716 BinaryLoader::alignValues(SgAsmGenericSection *section, const MemoryMap::Ptr&,
717  Address *malign_lo_p, Address *malign_hi_p,
718  Address *va_p, Address *mem_size_p,
719  Address *offset_p, Address *file_size_p, bool *map_private_p,
720  Address *va_offset_p, bool *anon_lo_p, bool *anon_hi_p,
721  ConflictResolution *resolve_p) {
722  ASSERT_not_null(section);
723  ASSERT_require2(section->isMapped(), "section must be mapped to virtual memory");
724  ASSERT_not_null(malign_lo_p);
725  ASSERT_not_null(malign_hi_p);
726  ASSERT_not_null(va_p);
727  ASSERT_not_null(mem_size_p);
728  ASSERT_not_null(offset_p);
729  ASSERT_not_null(file_size_p);
730  ASSERT_not_null(anon_lo_p);
731  ASSERT_not_null(anon_hi_p);
732  ASSERT_not_null(resolve_p);
733 
734  SgAsmGenericHeader *header = isSgAsmGenericHeader(section);
735  if (!header) header = section->get_header();
736  ASSERT_not_null(header);
737 
738  /* Initial guesses */
739  Address malign_lo = *malign_lo_p;
740  Address malign_hi = *malign_hi_p;
741  Address va = *va_p;
742  Address mem_size = *mem_size_p;
743  Address offset = *offset_p;
744  Address falign_lo = std::max(section->get_fileAlignment(), (Address)1);
745  Address file_size = *file_size_p;
746 
747  /* Align lower end of mapped region to satisfy both memory and file alignment constraints. */
748  Address va_offset = bialign(va, malign_lo, offset, falign_lo);
749  if (va_offset>va || va_offset>offset) {
750  mlog[TRACE] <<" adjustment " <<va_offset <<" exceeds va or offset (va=" <<va
751  <<", offset=" <<offset <<")\n";
752  throw Exception("no solutions to memory/file alignment constraints");
753  }
754  ASSERT_require((va - va_offset) % malign_lo == 0);
755  ASSERT_require((offset - va_offset) % falign_lo == 0);
756  va -= va_offset;
757  mem_size += va_offset;
758  offset -= va_offset;
759  file_size += va_offset;
760 
761  /* Extend mapped region to satisfy high alignment; but do not extend the file region */
762  if ((va + mem_size) % malign_hi != 0) {
763  uint64_t extend_by = alignUp(va+mem_size, malign_hi) - (va+mem_size);
764  mem_size += extend_by;
765  }
766 
767  /* Return values */
768  *malign_lo_p = malign_lo;
769  *malign_hi_p = malign_hi;
770  *va_p = va;
771  *mem_size_p = mem_size;
772  *offset_p = offset;
773  *file_size_p = file_size;
774  *map_private_p = false;
775  *va_offset_p = va_offset;
776  *anon_lo_p = true;
777  *anon_hi_p = true;
778  *resolve_p = RESOLVE_THROW;
779  return CONTRIBUTE_ADD;
780 }
