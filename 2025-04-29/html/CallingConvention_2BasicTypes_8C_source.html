29 readArgument(const RiscOperators::Ptr &ops, const Definition::Ptr &ccDef, size_t argNumber) {
30  ASSERT_not_null(ops);
31  ASSERT_not_null(ccDef);
32  ops->comment("reading function argument #" + boost::lexical_cast<std::string>(argNumber) +
33  " using calling convention " + ccDef->name());
34  const size_t nBits = ccDef->bitsPerWord();
35  SValue::Ptr retval;
36  if (argNumber < ccDef->inputParameters().size()) {
37  // Argument is explicit in the definition
38  const ConcreteLocation &loc = ccDef->inputParameters()[argNumber];
39 
40  switch (loc.type()) {
41  case ConcreteLocation::REGISTER:
42  retval = ops->readRegister(loc.reg());
43  break;
44 
45  case ConcreteLocation::RELATIVE: {
46  const auto base = ops->readRegister(loc.reg());
47  const auto offset = ops->signExtend(ops->number_(64, loc.offset()), base->nBits());
48  const auto address = ops->add(base, offset);
49  const auto dflt = ops->undefined_(nBits);
50  retval = ops->readMemory(RegisterDescriptor(), address, dflt, ops->boolean_(true));
51  break;
52  }
53 
54  case ConcreteLocation::ABSOLUTE: {
55  auto address = ops->number_(64, loc.address());
56  const auto dflt = ops->undefined_(nBits);
57  retval = ops->readMemory(RegisterDescriptor(), address, dflt, ops->boolean_(true));
58  break;
59  }
60 
61  case ConcreteLocation::NO_LOCATION:
62  break;
63  }
64 
65  } else {
66  // Argument is at an implied stack location
67  argNumber = argNumber - ccDef->inputParameters().size();
68  switch (ccDef->stackParameterOrder()) {
69  case StackParameterOrder::RIGHT_TO_LEFT:
70  break;
71  case StackParameterOrder::LEFT_TO_RIGHT:
72  ASSERT_not_implemented("we need to know how many parameters were pushed");
73  case StackParameterOrder::UNSPECIFIED:
74  ASSERT_not_implemented("invalid stack paramter order");
75  }
76  int64_t stackOffset = ccDef->nonParameterStackSize() + argNumber * nBits/8;
77  switch (ccDef->stackDirection()) {
78  case StackDirection::GROWS_DOWN:
79  break;
80  case StackDirection::GROWS_UP:
81  stackOffset = -stackOffset - nBits/8;
82  break;
83  }
84 
85  const RegisterDescriptor reg_sp = ccDef->stackPointerRegister();
86  const auto base = ops->readRegister(reg_sp);
87  const auto offset = ops->signExtend(ops->number_(64, stackOffset), base->nBits());
88  const auto address = ops->add(base, offset);
89  const auto dflt = ops->undefined_(nBits);
90  retval = ops->readMemory(RegisterDescriptor(), address, dflt, ops->boolean_(true));
91  }
92 
93  ASSERT_always_not_null2(retval, "invalid parameter location");
94  ops->comment("argument value is " + retval->toString());
95  return retval;

247 void
248 simulateFunctionReturn(const RiscOperators::Ptr &ops, const Definition::Ptr &ccDef) {
249  ASSERT_not_null(ops);
250  ASSERT_not_null(ccDef);
251  ops->comment("simulating function return using calling convention " + ccDef->name());
252 
253  // Assume that the current stack pointer is the same as it was when this function was entered.
254  const RegisterDescriptor SP = ccDef->stackPointerRegister();
255  ASSERT_require(SP);
256  const auto originalSp = ops->readRegister(SP);
257 
258  // How many input parameters to be popped from the stack by the callee?
259  //
260  // FIXME[Robb Matzke 2022-07-13]: This assumes that each parameter is one word, which is often not the case, but
261  // we don't currently have any information about parameter sizes.
262  int64_t nArgBytes = 0;
263  if (StackCleanup::BY_CALLEE == ccDef->stackCleanup()) {
264  for (const ConcreteLocation &loc: ccDef->inputParameters()) {
265  if (ConcreteLocation::RELATIVE == loc.type())
266  nArgBytes += ccDef->bitsPerWord() / 8;
267  }
268  }
269 
270  // How many other things need to be popped from the stack by the callee as part of returning?
271  int64_t nNonArgBytes = ccDef->nonParameterStackSize();
272 
273  // Obtain the return address
274  const ConcreteLocation &retVaLoc = ccDef->returnAddressLocation();
275  SValue::Ptr retVa;
276  switch (retVaLoc.type()) {
277  case ConcreteLocation::REGISTER:
278  retVa = ops->readRegister(retVaLoc.reg());
279  break;
280 
281  case ConcreteLocation::RELATIVE: {
282  const auto base = ops->readRegister(retVaLoc.reg());
283  const auto offset = ops->signExtend(ops->number_(64, retVaLoc.offset()), base->nBits());
284  const auto address = ops->add(base, offset);
285  const auto dflt = ops->undefined_(ccDef->bitsPerWord());
286  retVa = ops->readMemory(RegisterDescriptor(), address, dflt, ops->boolean_(true));
287  break;
288  }
289 
290  case ConcreteLocation::ABSOLUTE: {
291  auto address = ops->number_(64, retVaLoc.address());
292  const auto dflt = ops->undefined_(ccDef->bitsPerWord());
293  retVa = ops->readMemory(RegisterDescriptor(), address, dflt, ops->boolean_(true));
294  break;
295  }
296 
297  case ConcreteLocation::NO_LOCATION:
298  break;
299  }
300  ASSERT_not_null2(retVa, "unknown location for fuction return address");
301 
302  // Pop things from the stack
303  SValue::Ptr newSp;
304  switch (ccDef->stackDirection()) {
305  case StackDirection::GROWS_DOWN:
306  newSp = ops->add(originalSp, ops->number_(originalSp->nBits(), nArgBytes + nNonArgBytes));
307  break;
308  case StackDirection::GROWS_UP:
309  newSp = ops->subtract(originalSp, ops->number_(originalSp->nBits(), nArgBytes + nNonArgBytes));
310  break;
311  }
312  ASSERT_not_null2(newSp, "invalid stack growth direction");
313  ops->writeRegister(SP, newSp);
314 
315  // Change the instruction pointer to be the return address.
316  const RegisterDescriptor IP = ccDef->instructionPointerRegister();
317  ops->writeRegister(IP, retVa);
318  ops->comment("function return has been simulated");
319 }
