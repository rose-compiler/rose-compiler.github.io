186  if (m_Sequence.size() == 0) {
187  Node *pNode = new Node;
188  m_Sequence.insert({0, pNode});
189  return pNode;
190  }
191 
192  if (index >= m_Sequence.size()) {
193  auto it = m_Sequence.end();
194  --it;
195  Node *pNode = new Node;
196  m_Sequence.insert({it->first, pNode});
197  return pNode;
198  }
199 
200  auto it = m_Sequence.cbegin();
201  while (it != m_Sequence.cend()) {
202  m_Sequence[it->first+1] = it->second;
203  if (it->first == index)
204  break;
205  }
206 
207  Node *pNode = new Node;
208  m_Sequence.insert({index, pNode});
209  return pNode;
210  }
211 
212  virtual Node* PushFront() {
213  for (auto it = m_Sequence.cbegin(); it != m_Sequence.cend(); it++)
214  m_Sequence[it->first+1] = it->second;
215 
216  Node *pNode = new Node;
217  m_Sequence.insert({0, pNode});
218  return pNode;
219  }
220 
221  virtual Node* PushBack() {
222  size_t index = 0;

224  auto it = m_Sequence.end();
225  --it;
226  index = it->first + 1;
227  }
228 
229  Node *pNode = new Node;
230  m_Sequence.insert({index, pNode});
231  return pNode;
232  }
233 
234  virtual void Erase(const size_t index) {
235  auto it = m_Sequence.find(index);
236  if (it == m_Sequence.end())
237  return;
238  delete it->second;
239  m_Sequence.erase(index);

1299  // Read all lines, ignoring: empty lines, comments, and document start/end.
1300  void ReadLines(std::istream &stream) {
1301  std::string line = "";
1302  size_t lineNo = 0;
1303  size_t first_unprintable = 0;
1304  bool documentStartFound = false;
1305  bool foundFirstNotEmpty = false;
1306  std::streampos streamPos = 0;
1307 
1308  // Read all lines, as long as the stream is ok.
1309  while (!stream.eof() && !stream.fail()) {
1310  // Read line
1311  streamPos = stream.tellg();
1312  std::getline(stream, line);
1313  lineNo++;
1314 
1315  // Remove comment
1316  const size_t commentPos = FindNotCited(line, '#');
1317  if (commentPos != std::string::npos)
1318  line.resize(commentPos);
1319 
1320  // Start of document.
1321  if (documentStartFound == false && line == "---") {
1322  // Erase all lines before this line.
1323  ClearLines();
1324  documentStartFound = true;
1325  continue;
1326  }
1327 
1328  // End of document.
1329  if (line == "...") {
1330  break;
1331  } else if (line == "---") {
1332  stream.seekg(streamPos);
1333  break;
1334  }
1335 
1336  // Remove trailing return.
1337  if (line.size()) {
1338  if (line[line.size() - 1] == '\r')
1339  line.resize(line.size() - 1);
1340  }
1341 
1342  first_unprintable = find_unprintable(line);
1343 
1344  if (first_unprintable < utf8_chars(line))
1345  throw ParsingException(ExceptionMessage(g_ErrorInvalidCharacter, lineNo, first_unprintable));
1346 
1347  // Validate tabs
1348  const size_t firstTabPos = line.find_first_of('\t');
1349  size_t startOffset = line.find_first_not_of(" \t");
1350 
1351  // Make sure no tabs are in the very front.
1352  if (startOffset != std::string::npos) {
1353  if (firstTabPos < startOffset)
1354  throw ParsingException(ExceptionMessage(g_ErrorTabInOffset, lineNo, firstTabPos));
1355 
1356  // Remove front spaces.
1357  line = line.substr(startOffset);
1358  } else {
1359  startOffset = 0;
1360  line = "";
1361  }
1362 
1363  // Add line.
1364  if (foundFirstNotEmpty == false) {
1365  if (line.size()) {
1366  foundFirstNotEmpty = true;
1367  } else {
1368  continue;
1369  }
1370  }
1371 
1372  ReaderLine *pLine = new ReaderLine(line, lineNo, startOffset);
1373  m_Lines.push_back(pLine);
1374  }
1375  }

1377  // Run post-processing on all lines. Basically split lines into multiple lines if needed, to follow the parsing algorithm.
1378  void PostProcessLines() {
1379  for (auto it = m_Lines.begin(); it != m_Lines.end(); /*void*/) {
1380  // Sequence.
1381  if (PostProcessSequenceLine(it) == true)
1382  continue;
1383 
1384  // Mapping.
1385  if (PostProcessMappingLine(it) == true)
1386  continue;
1387 
1388  // Scalar.
1389  PostProcessScalarLine(it);
1390  }
1391 
1392  // Set next line of all lines.
1393  if (m_Lines.size()) {
1394  if (m_Lines.back()->Type != Node::ScalarType)
1395  throw ParsingException(ExceptionMessage(g_ErrorUnexpectedDocumentEnd, *m_Lines.back()));
1396 
1397  if (m_Lines.size() > 1) {
1398  auto prevEnd = m_Lines.end();
1399  --prevEnd;
1400 
1401  for (auto it = m_Lines.begin(); it != prevEnd; it++) {
1402  auto nextIt = it;
1403  ++nextIt;
1404  (*it)->NextLine = *nextIt;
1405  }

1410  // Run post-processing and check for sequence. Split line into two lines if sequence token is not on it's own line. Return
1411  // true if line is a sequence and does not need more postprocessing, otherwise return false.
1412  bool PostProcessSequenceLine(std::list<ReaderLine *>::iterator &it) {
1413  ReaderLine *pLine = *it;
1414 
1415  // Sequence split
1416  if (!IsSequenceStart(pLine->Data))
1417  return false;
1418 
1419  pLine->Type = Node::SequenceType;
1420 
1421  ClearTrailingEmptyLines(++it);
1422 
1423  const size_t valueStart = pLine->Data.find_first_not_of(" \t", 1);
1424  if (valueStart == std::string::npos) {
1425  pLine->Data = "";
1426  return true;
1427  }
1428 
1429  // Create new line and insert
1430  std::string newLine = pLine->Data.substr(valueStart);
1431  size_t child_offset = pLine->Offset + valueStart;
1432 
1433  it = m_Lines.insert(it, new ReaderLine(newLine, pLine->No, child_offset));
1434  pLine->Data = "";
1435 
1436  // Recursively handle nested sequences
1437  if (IsSequenceStart(newLine))
1438  return PostProcessSequenceLine(it);
1439 
1440  return false;
1441  }

1443  // Run post-processing and check for mapping. Split line into two lines if mapping value is not on it's own line. Return
1444  // true if line is mapping, else move on to scalar parsing.
1445  bool PostProcessMappingLine(std::list<ReaderLine*>::iterator &it) {
1446  ReaderLine *pLine = *it;
1447 
1448  // Find map key.
1449  size_t preKeyQuotes = 0;
1450  size_t tokenPos = FindNotCited(pLine->Data, ':', preKeyQuotes);
1451  if (tokenPos == std::string::npos)
1452  return false;
1453  if (preKeyQuotes > 1)
1454  throw ParsingException(ExceptionMessage(g_ErrorKeyIncorrect, *pLine));
1455 
1456  pLine->Type = Node::MapType;
1457 
1458  // Get key
1459  std::string key = pLine->Data.substr(0, tokenPos);
1460  const size_t keyEnd = key.find_last_not_of(" \t");
1461  if (keyEnd == std::string::npos)
1462  throw ParsingException(ExceptionMessage(g_ErrorKeyMissing, *pLine));
1463  key.resize(keyEnd + 1);
1464 
1465  // Handle cited key.
1466  if (preKeyQuotes == 1) {
1467  if (key.front() != '"' || key.back() != '"')
1468  throw ParsingException(ExceptionMessage(g_ErrorKeyIncorrect, *pLine));
1469  key = key.substr(1, key.size() - 2);
1470  }
1471  RemoveAllEscapeTokens(key);
1472 
1473  // Get value
1474  std::string value = "";
1475  size_t valueStart = std::string::npos;
1476  if (tokenPos + 1 != pLine->Data.size()) {
1477  valueStart = pLine->Data.find_first_not_of(" \t", tokenPos + 1);
1478  if (valueStart != std::string::npos)
1479  value = pLine->Data.substr(valueStart);
1480  }
1481 
1482  // Make sure the value is not a sequence start.
1483  if (IsSequenceStart(value) == true)
1484  throw ParsingException(ExceptionMessage(g_ErrorBlockSequenceNotAllowed, *pLine, valueStart));
1485 
1486  pLine->Data = key;
1487 
1488  // Remove all empty lines after map key.
1489  ClearTrailingEmptyLines(++it);
1490 
1491  // Add new empty line?
1492  size_t newLineOffset = valueStart;
1493  if (newLineOffset == std::string::npos) {
1494  // There are three options:
1495  // 1. The value is present at a deeper level than the map key.
1496  // 2. The value is empty.
1497  // 3. The next line is a compact sequence at the same level as the map key.
1498 
1499  if (it != m_Lines.end()) {
1500  // Case 1
1501  if ((*it)->Offset > pLine->Offset)
1502  return true;
1503 
1504  // Case 3, compact sequence
1505  if ((*it)->Offset == pLine->Offset && IsSequenceStart((*it)->Data))
1506  return true;
1507  }
1508 
1509  // Must be Case 2 otherwise
1510  newLineOffset = tokenPos + 2;
1511  } else {
1512  newLineOffset += pLine->Offset;
1513  }
