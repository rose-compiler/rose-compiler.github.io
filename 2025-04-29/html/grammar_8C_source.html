539 void Grammar::editStringList ( vector<GrammarString *> & targetList, const vector<GrammarString *> & excludeList )
540 {
541  // Remove the elements in the excludeList from the elements in targetList
542  // The match is determined by the use of the operator= on the elements!
543  // list is modified!
544  vector<GrammarString*> newList;
545  for (vector<GrammarString*>::const_iterator i = targetList.begin(); i != targetList.end(); ++i) {
546  for (vector<GrammarString*>::const_iterator j = i + 1; j != targetList.end(); ++j) {
547  if (**i == **j)
548  goto skipThisElement;
549  }
550  for (vector<GrammarString*>::const_iterator j = excludeList.begin(); j != excludeList.end(); ++j) {
551  if (**i == **j)
552  goto skipThisElement;
553  }
554  newList.push_back(*i);
555  skipThisElement:;
556  }
557  targetList.swap(newList);
558 }
