66 void LineNumberWidget::paintEvent( QPaintEvent* /*e*/ )
67 {
68  buffer.fill();
69 
70  Q3TextParagraph *p = m_editor->getDocument()->firstParagraph();
71  QPainter painter( &buffer );
72  QFont f(painter.font());
73  f.setPointSize(m_editor->pointSize());
74  painter.setFont(f);
75  int yOffset = m_editor->contentsY();
76  while ( p ) {
77  if ( !p->isVisible() ) {
78  p = p->next();
79  continue;
80  }
81  if ( p->rect().y() + p->rect().height() - yOffset < 0 ) {
82  p = p->next();
83  continue;
84  }
85  if ( p->rect().y() - yOffset > height() )
86  break;
87  //ParagData *paragData = (ParagData*)p->extraData();
88 
89  painter.drawText( 0, p->rect().y() - yOffset+3,
90  buffer.width() - 10, p->rect().height(),
91  Qt::AlignRight | Qt::AlignBottom,
92  QString::number(p->paragId()+1) );
93  p = p->next();
94  }
95 
96  painter.end();
97  bitBlt( this, 0, 0, &buffer );
98 }
