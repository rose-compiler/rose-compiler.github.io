150  }

185 void
186 ExternalFunctionUnit::clearReturnValues(const BS::Dispatcher::Ptr &cpu) {
187  ASSERT_not_null(cpu);
188  BS::RiscOperators::Ptr ops = cpu->operators();
189  RegisterDictionary::Ptr regDict = cpu->registerDictionary();
190 
191  // Set return value locations to unknown values. Using calling convention analysis is the right way,
192  // but if that's not available fall back to the stupid closed list of dynamic casts.
193  if (CC::Definition::Ptr ccDefn = function_->callingConventionDefinition()) {
194  for (const ConcreteLocation &param: ccDefn->outputParameters()) {
195  size_t nBits = 0;
196  switch (param.type()) {
197  case ConcreteLocation::REGISTER:
198  if (param.reg() != cpu->stackPointerRegister() &&
199  param.reg() != cpu->stackFrameRegister() &&
200  param.reg() != cpu->callReturnRegister()) {
201  nBits = param.reg().nBits();
202  }
203  break;
204  case ConcreteLocation::RELATIVE:
205  case ConcreteLocation::ABSOLUTE:
206  nBits = cpu->stackPointerRegister().nBits();
207  break;
208  case ConcreteLocation::NO_LOCATION:
209  ASSERT_not_reachable("invalid parameter location");
210  }
211  if (nBits != 0) {
212  if (mlog[DEBUG]) {
213  mlog[DEBUG] <<" clearing output value ";
214  param.print(mlog[DEBUG], regDict);
215  mlog[DEBUG] <<" (" <<ccDefn->name() <<" calling convention)\n";
216  }
217  BS::SValue::Ptr undef = ops->undefined_(nBits);
218  writeLocation(cpu, param, undef);
219  }
220  }
221  } else {
222  std::vector<RegisterDescriptor> retRegs;
223  if (as<IS::DispatcherX86>(cpu)) {
224  RegisterDescriptor r;
225  if ((r = regDict->find("rax")) || (r = regDict->find("eax")) || (r = regDict->find("ax")))
226  retRegs.push_back(r);
227  } else if (as<IS::DispatcherM68k>(cpu)) {
228  // FIXME[Robb Matzke 2021-04-15]: m68k also typically has other return registers
229  if (RegisterDescriptor r = regDict->find("d0"))
230  retRegs.push_back(r);
231  } else if (as<IS::DispatcherPowerpc>(cpu)) {
232  if (RegisterDescriptor r = regDict->find("r3"))
233  retRegs.push_back(r);
234  if (RegisterDescriptor r = regDict->find("r4"))
235  retRegs.push_back(r);
236 #ifdef ROSE_ENABLE_ASM_AARCH32
237  } else if (as<IS::DispatcherAarch32>(cpu)) {
238  if (RegisterDescriptor r = regDict->find("r0"))
239  retRegs.push_back(r);
240 #endif
241 #ifdef ROSE_ENABLE_ASM_AARCH64
242  } else if (as<IS::DispatcherAarch64>(cpu)) {
243  if (RegisterDescriptor r = regDict->find("x0"))
244  retRegs.push_back(r);
245 #endif
246  } else {
247  ASSERT_not_implemented("function return value register is not implemented for this ISA/ABI");
248  }
249 
250  for (RegisterDescriptor reg: retRegs) {
251  RegisterNames regNames(regDict);
252  SAWYER_MESG(mlog[DEBUG]) <<" clearing return value " <<regNames(reg) <<" (hard coded)\n";
253  BS::SValue::Ptr undef = ops->undefined_(reg.nBits());
254  ops->writeRegister(reg, undef);
255  }
256  }

259 void
260 ExternalFunctionUnit::simulateReturn(const BS::Dispatcher::Ptr &cpu) {
261  ASSERT_not_null(cpu);
262  BS::RiscOperators::Ptr ops = cpu->operators();
263 
264  // FIXME[Robb Matzke 2021-04-02]: dynamic casting the CPU to each known architecture is certainly not the right
265  // way to do this! These operations should be abstracted in the BaseSemantics::Dispatcher class as virtual functions.
266  // Most of this has been copied from the FeasiblePath class.
267  if (as<IS::DispatcherPowerpc>(cpu)) {
268  // PowerPC calling convention stores the return address in the link register (LR)
269  const RegisterDescriptor LR = cpu->callReturnRegister();
270  ASSERT_forbid(LR.isEmpty());
271  BS::SValue::Ptr returnTarget = ops->readRegister(LR, ops->undefined_(LR.nBits()));
272  ops->writeRegister(cpu->instructionPointerRegister(), returnTarget);
273 
274  } else if (as<IS::DispatcherX86>(cpu) || as<IS::DispatcherM68k>(cpu)) {
275  // x86, amd64, and m68k store the return address at the top of the stack
276  const RegisterDescriptor SP = cpu->stackPointerRegister();
277  ASSERT_forbid(SP.isEmpty());
278  BS::SValue::Ptr stackPointer = ops->readRegister(SP, ops->undefined_(SP.nBits()));
279  BS::SValue::Ptr returnTarget = ops->readMemory(RegisterDescriptor(), stackPointer,
280  ops->undefined_(stackPointer->nBits()),
281  ops->boolean_(true));
282  ops->writeRegister(cpu->instructionPointerRegister(), returnTarget);
283  stackPointer = ops->add(stackPointer, ops->number_(stackPointer->nBits(), stackPointer->nBits()/8));
284  ops->writeRegister(cpu->stackPointerRegister(), stackPointer);
285 #ifdef ROSE_ENABLE_ASM_AARCH64
286  } else if (as<IS::DispatcherAarch64>(cpu)) {
287  // Return address is in the link register, lr
288  const RegisterDescriptor LR = cpu->callReturnRegister();
289  ASSERT_forbid(LR.isEmpty());
290  BS::SValue::Ptr returnTarget = ops->readRegister(LR, ops->undefined_(LR.nBits()));
291  ops->writeRegister(cpu->instructionPointerRegister(), returnTarget);
292 #endif
293 #ifdef ROSE_ENABLE_ASM_AARCH32
294  } else if (as<IS::DispatcherAarch32>(cpu)) {
295  // Return address is in the link register, lr
296  const RegisterDescriptor LR = cpu->callReturnRegister();
297  ASSERT_forbid(LR.isEmpty());
298  BS::SValue::Ptr returnTarget = ops->readRegister(LR, ops->undefined_(LR.nBits()));
299  ops->writeRegister(cpu->instructionPointerRegister(), returnTarget);
300 #endif
301  }
