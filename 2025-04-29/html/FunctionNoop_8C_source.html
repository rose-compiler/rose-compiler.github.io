27 bool
28 Partitioner::functionIsNoop(const Function::Ptr &function) const {
29  Sawyer::Message::Stream debug(mlog[DEBUG] <<"Partitioner::functionIsNoop(" <<function->printableName() <<")\n");
30  ASSERT_not_null(function);
31  bool retval = false;
32 
33  // If an analysis has already been run then return its value.
34  if (function->isNoop().getOptional().assignTo(retval)) {
35  SAWYER_MESG(debug) <<" " <<function->printableName() <<" is " <<(retval?"":"not ") <<"no-op (cached)\n";
36  return retval;
37  }
38 
39  // No-op analyser. We use an arbitrary concrete initial stack pointer because it allows the analysis to assume stack
40  // semantics. That is, items popped from the stack are no longer considered to be valid memory. Using an odd value for the
41  // initial stack reduces the number of false positives for stack-aligning instructions being no-ops (e.g., "AND SP,
42  // 0xfffffff0" will not be a no-op).
43  NoOperation noOpAnalyzer(newDispatcher(newOperators()));
44  noOpAnalyzer.initialStackPointer(0xdddd0001);
45 
46  // If any vertex of this function (other than its return vertices) has successors that are in some other function and that
47  // other function is not a proven no-op, then assume that this function is not a no-op either. We do this separately from
48  // the data-flow because this loop is much faster than that one.
49  for (Address bbVa: function->basicBlockAddresses()) {
50  ControlFlowGraph::ConstVertexIterator bbVertex = findPlaceholder(bbVa);
51  ASSERT_require(cfg().isValidVertex(bbVertex));
52  BasicBlock::Ptr bb = bbVertex->value().bblock();
53  ASSERT_not_null(bb);
54  if (!basicBlockIsFunctionReturn(bb)) {
55  for (const ControlFlowGraph::Edge &edge: bbVertex->outEdges()) {
56  for (const Function::Ptr &targetFunc: edge.target()->value().owningFunctions().values()) {
57  if (targetFunc != function && !targetFunc->isNoop().isCached()) {
58  // target function is not a no-op, or no-op analysis has not been run on it yet.
59  SAWYER_MESG(debug) <<" " <<function->printableName() <<" calls " <<targetFunc->printableName()
60  <<" which is not a no-op"
61  <<(targetFunc->isNoop().isCached()?" (proven)":" (untested)") <<"\n";
62  retval = false;
63  goto done;
64  }
65  }
66  }
67  }
68  }
69 
70  // Now use data-flow to look at each basic block. If any basic block of this function is not a noop then assume the
71  // function as a whole is not a no-op.
72  for (Address bbVa: function->basicBlockAddresses()) {
73  if (BasicBlock::Ptr bb = basicBlockExists(bbVa)) {
74  // Get the instructions for this block, excluding the final instruction if this is a function return.
75  std::vector<SgAsmInstruction*> insns = bb->instructions();
76  if (basicBlockIsFunctionReturn(bb)) {
77  insns.pop_back();
78  } else {
79  // If the basic block is not a function return and its CFG successor is undiscovered or indeterminate then we
80  // must assume that its successor imparts some effect to this function.
81  ControlFlowGraph::ConstVertexIterator placeholder = findPlaceholder(bbVa);
82  ASSERT_require(cfg().isValidVertex(placeholder));
83  for (const ControlFlowGraph::Edge &edge: placeholder->outEdges()) {
84  if (edge.target()->value().type() != V_BASIC_BLOCK) {
85  SAWYER_MESG(debug) <<" " <<function->printableName() <<" is not a no-op because the successor of "
86  <<bb->printableName() <<" is missing: " <<vertexName(edge.target()) <<"\n";
87  retval = false;
88  goto done;
89  }
90  }
91  }
92  if (!noOpAnalyzer.isNoop(insns)) {
93  SAWYER_MESG(debug) <<" " <<function->printableName() <<" is not a no-op because "
94  <<bb->printableName() <<" is not a no-op\n";
95  retval = false;
96  goto done;
97  }
98  } else {
99  SAWYER_MESG(debug) <<" " <<function->printableName() <<" is not a no-op because it is missing a basic block at "
100  <<StringUtility::addrToString(bbVa);
101  retval = false;
102  goto done;
103  }
104  }
105 
106  // Must be a no-op since all basic blocks are no-ops.
107  SAWYER_MESG(debug) <<" " <<function->printableName() <<" is a no-op\n";
108  retval = true;
109 
110 done:
111  function->isNoop() = retval;
112  return retval;
113 }
