13 static string escapeStringDFA(const string& s) {
14  string result;
15  for (unsigned int i = 0; i < s.length(); ++i) {
16  if (isprint(s[i]) && s[i] != '"' && s[i] != '\\') {
17  // if (isprint(s[i])) {
18  result.push_back(s[i]);
19  } else if (s[i] == '\a') {
20  result += "\\a";
21  } else if (s[i] == '\f') {
22  result += "\\f";
23  } else if (s[i] == '\n') {
24  result += "\\n";
25  } else if (s[i] == '\r') {
26  result += "\\r";
27  } else if (s[i] == '\t') {
28  result += "\\t";
29  } else if (s[i] == '\v') {
30  result += "\\v";
31  } else {
32  ostringstream stream;
33  stream << '\\';
34  stream << setw(3) << setfill('0') << (unsigned int)(s[i]);

233  template <typename NodeT, typename EdgeT>
234  void DfaToDotImpl<NodeT, EdgeT>::explore(NodeT n) {
235  ROSE_ASSERT (n.getNode());
236  //std::cerr << n.toStringForDebugging() << std::endl;
237 
238  // search through the multimap to make sure CFGNode is not visited before, if so, return.
239  pair<typename multimap<SgNode*, NodeT>::const_iterator,
240  typename multimap<SgNode*, NodeT>::const_iterator> ip = exploredNodes.equal_range(n.getNode());
241  for (typename multimap<SgNode*, NodeT>::const_iterator i = ip.first; i != ip.second; ++i) {
242  if (i->second == n) return;
243  }
244  // save visited CFG nodes into the multimap
245  exploredNodes.insert(make_pair(n.getNode(), n));
246 
247  // explore out edges' targets
248  vector<EdgeT> outEdges = n.outEdges();
249  for (unsigned int i = 0; i < outEdges.size(); ++i) {
250  ROSE_ASSERT (outEdges[i].source() == n);
251  explore(outEdges[i].target());
252  }
253  // explore in edges' targets
254  vector<EdgeT> inEdges = n.inEdges();
255  for (unsigned int i = 0; i < inEdges.size(); ++i) {
256  ROSE_ASSERT (inEdges[i].target() == n);
257  explore(inEdges[i].source());
258  }
259  if (getenv("BUGGY_INTERPROCEDURAL_CFG") && isSgFunctionCallExp(n.getNode()) && n.getIndex() == 2) explore(CFGNode(n.getNode(), 3));
260  }
