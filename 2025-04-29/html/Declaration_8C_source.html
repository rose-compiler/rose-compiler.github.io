34 // Parse a type and return the type or an error message.
35 static Sawyer::Result<SgAsmType*, std::string>
36 parseType(TokenStream &tokens, const Definition::Ptr &cc) {
37  ASSERT_not_null(cc);
38  using namespace Sawyer::Language::Clexer;
39  SgAsmType *type = nullptr;
40 
41  // Required type name, like 'void', 'u8', 'i32', etc.
42  if (tokens[0].type() != TOK_WORD)
43  return Sawyer::makeError("type name expected");
44  if (tokens.matches(tokens[0], "void")) {
45  type = SgAsmVoidType::instance();
46  } else if (tokens.matches(tokens[0], "u8")) {
47  type = SgAsmIntegerType::instanceUnsigned(cc->byteOrder(), 8);
48  } else if (tokens.matches(tokens[0], "u16")) {
49  type = SgAsmIntegerType::instanceUnsigned(cc->byteOrder(), 16);
50  } else if (tokens.matches(tokens[0], "u32")) {
51  type = SgAsmIntegerType::instanceUnsigned(cc->byteOrder(), 32);
52  } else if (tokens.matches(tokens[0], "u64")) {
53  type = SgAsmIntegerType::instanceUnsigned(cc->byteOrder(), 64);
54  } else if (tokens.matches(tokens[0], "i8")) {
55  type = SgAsmIntegerType::instanceSigned(cc->byteOrder(), 8);
56  } else if (tokens.matches(tokens[0], "i16")) {
57  type = SgAsmIntegerType::instanceSigned(cc->byteOrder(), 16);
58  } else if (tokens.matches(tokens[0], "i32")) {
59  type = SgAsmIntegerType::instanceSigned(cc->byteOrder(), 32);
60  } else if (tokens.matches(tokens[0], "i64")) {
61  type = SgAsmIntegerType::instanceSigned(cc->byteOrder(), 64);
62  } else if (tokens.matches(tokens[0], "f32")) {
63  type = SgAsmFloatType::instanceIeee32(cc->byteOrder());
64  } else if (tokens.matches(tokens[0], "f64")) {
65  type = SgAsmFloatType::instanceIeee64(cc->byteOrder());
66  } else {
67  return Sawyer::makeError("unrecognized type \"" + tokens.lexeme(tokens[0]) + "\"");
68  }
69  tokens.consume();
70 
71  // Optional '*' to indicate pointer
72  while (tokens.matches(tokens[0], "*")) {
73  type = SgAsmPointerType::instance(cc->byteOrder(), cc->bitsPerWord(), type);
74  tokens.consume();
75  }
76 
77  return Sawyer::makeOk(type);
78 }

122 Declaration::Ptr
123 Declaration::instance(const Definition::Ptr &cc, const std::string &sourceCode) {
124  Ptr decl = Ptr(new Declaration(cc));
125  using namespace Sawyer::Language::Clexer;
126  decl->sourceCode_ = sourceCode;
127 
128  auto buffer = Sawyer::Container::StaticBuffer<size_t, char>::instance(sourceCode.data(), sourceCode.size());
129  TokenStream tokens("declaration", buffer);
130 
131  // Return type and optional function name
132  const auto retType = parseTypeNamePair(tokens, cc);
133  if (!retType)
134  throw parseError(retType.unwrapError(), tokens);
135  decl->returnType_ = retType->first;
136  decl->name(retType->second);
137 
138  // Start of argument list
139  if (!tokens.matches(tokens[0], "("))
140  throw parseError("\"(\" expected before argument list", tokens);
141  tokens.consume();
142 
143  // Argument list
144  if (!tokens.matches(tokens[0], ")")) {
145  while (true) {
146  const auto argType = parseArgTypeNamePair(tokens, cc);
147  if (!argType) {
148  throw parseError(argType.unwrapError() + " for argument #" +
149  boost::lexical_cast<std::string>(decl->arguments().size()), tokens);
150  }
151 
152  // Check that argument name is unique
153  if (!argType->second.empty()) {
154  for (const auto &otherArg: decl->arguments()) {
155  if (otherArg.second == argType->second)
156  throw parseError("duplicate argument name", tokens);
157  }
158  }
159 
160  decl->arguments_.push_back(*argType);
161  if (!tokens.matches(tokens[0], ","))
162  break;
163  tokens.consume(); // the comma
164  }
165  }
166 
167  // End of argument list
168  if (!tokens.matches(tokens[0], ")"))
169  throw parseError("\")\" expected at end of argument list", tokens);
170  tokens.consume();
171  if (tokens[0].type() != TOK_EOF)
172  throw parseError("extra text after end of argument list", tokens);
173 
174  decl->computeArgumentLocations();
175  decl->computeReturnLocations();
176  return decl;
177 }
