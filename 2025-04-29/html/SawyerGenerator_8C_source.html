37  sg.doc("This backend generates Sawyer::Tree code. The goal is to use a more modern style of tree data structures in ROSE, such "
38  "as for abstract syntax trees (ASTs)."
39 
40  "@bullet{Automatic parent pointers. It is impossible to mess up the parent/child relationships in a tree.}"
41  "@bullet{Immediately enforced tree invariants. No need for a separate consistency checker.}"
42  "@bullet{Reference counted vertices. The nodes of the tree have clear ownership rules that are enforced.}"
43  "@bullet{Simple traversals. Only two traversals upon which all others can be built.}"
44  "@bullet{Event driven. Observers can register to be notified when parent/child edges change.}"
45  "@bullet{Exception safe. Tree connectivity will be consistent even when exceptions are thrown.}"

47  "@bullet{Well tested. Sawyer trees aim for 100% testing coverage through unit tests.}"

50  "used for new nodes that don't integrate into the same class hierarchy as the traditional AST node types.");

232  ASSERT_not_null(p);

281 void
282 SawyerGenerator::genPropertyMutators(std::ostream &header, std::ostream &impl, const Ast::Property::Ptr &p) {
283  ASSERT_not_null(p);
284 
285  // Vector of tree edges never has a mutator because changes to the vector are made through its accessor. For example, a property
286  // definition like this:
287  // [[Rosebud::property]]
288  // EdgeVector<Thing> things;
289  //
290  // is modified by the user like this:
291  // node.things().push_back(thing);
292  // or
293  // node.things().back();
294  // or
295  // node.things()[i];
296  // etc.
297  if (isTreeVectorEdge(p))
298  return;
299 
300  const std::string type = propertyMutatorArgumentType(p);
301  const std::string constRefType = constRef(type);
302 
303  // The normal mutators for all users of this class.
304  for (const std::string &mutatorName: propertyMutatorNames(p))
305  genPropertyMutator(header, impl, p, Access::PUBLIC, mutatorName, constRef(type));
306 
307  // If no mutators are given, we still want the class itself to be able to modify the data member. This is hard to do if the data
308  // member names are hidden by a random name generator.
309  if (propertyMutatorNames(p).empty())
310  genPropertyMutator(header, impl, p, Access::PRIVATE, p->name, constRef(type));

366 SawyerGenerator::genConstructorBody(std::ostream &impl, const Ast::Class::Ptr &c) {
367  ASSERT_not_null(c);
368 
369  for (const auto &p: c->properties) {
370  if (p->findAttribute("Rosebud::not_null")) {
371  const std::string t = valueType(p()) + " const&";
372  if (isTreeVectorEdge(p())) {
373  // Whenever a new edge is added to the vector we need to make sure the child pointer is not null. Also, we need
374  // to make sure that the edge cannot be set to null later.
375  impl <<"\n"
376  <<THIS_LOCATION <<locationDirective(p(), p->startToken)
377  <<" // Tree edge vector member cannot be set to null initially or later due to Rosebud::not_null attribute\n"
378  <<" " <<propertyDataMemberName(p()) <<".beforeResize([](int delta, " <<t <<" childPtr) {\n"
379  <<" if (1 == delta && !childPtr)\n"
380  <<" throw Rose::Exception(\"property \\\"" + p->name + "\\\" cannot be set to null\");\n"
381  <<" });\n"
382  <<" " <<propertyDataMemberName(p()) <<".afterResize([this](int delta, " <<t <<") {\n"
383  <<" if (1 == delta) {\n"
384  <<" " <<propertyDataMemberName(p()) <<".back().beforeChange([](" <<t <<", " <<t <<" childPtr) {\n"
385  <<" if (!childPtr)\n"
386  <<" throw Rose::Exception(\"property \\\"" + p->name + "\\\" cannot be set to null\");\n"
387  <<" });\n"
388  <<" }\n"
389  <<" });\n";
390  } else if (isTreeScalarEdge(p())) {
391  // The new value cannot be null
392  impl <<"\n"
393  <<THIS_LOCATION <<locationDirective(p(), p->startToken)
394  <<" // Tree edge cannot be set to null due to Rosebud::not_null attribute\n"
395  <<" " <<propertyDataMemberName(p()) <<".beforeChange([](" <<t <<", " <<t <<" childPtr) {\n"
396  <<" if (!childPtr)\n"
397  <<" throw Rose::Exception(\"property \\\"" + p->name + "\\\" cannot be set to null\");\n"
398  <<" });\n";
399  }
400  }
401  }

499 SawyerGenerator::genClass(const Ast::Class::Ptr &c, const Hierarchy &h) {
500  ASSERT_not_null(c);
501 
502  if (c->findAttribute("rosebud::suppress"))
503  return;
504  std::vector<Serializer::Ptr> serializers = Serializer::lookup(settings.serializers);
505 
506  // Open files
507  auto file = notnull(c->findAncestor<Ast::File>());
508  const boost::filesystem::path headerName = generatedDir / toPath(c->name, ".h");
509  const boost::filesystem::path implName = generatedDir / toPath(c->name, ".C");
510  std::ofstream header(headerName.c_str());
511  std::ofstream impl(implName.c_str());
512  if (!header) {
513  message(ERROR, "cannot open file for writing: \"" + headerName.string() + "\"");
514  return;
515  }
516  if (!impl) {
517  message(ERROR, "cannot open file for writing: \"" + implName.string() + "\"");
518  return;
519  }
520 
521  // Emit stuff before the beginning of the class
522  genHeaderOpen(header, c);
523  genImplOpen(impl, c);
524  for (const Serializer::Ptr &serializer: serializers)
525  serializer->genPrologue(header, impl, c, h, *this);
526  if (!c->priorText.empty()) {
527  static const std::regex implRe("#\\s*(ifdef\\s+ROSE_IMPL|if\\s+defined\\s*\\(\\s*ROSE_IMPL\\s*\\))");
528  const std::string cppImplSymbol = toCppSymbol(c->qualifiedNamespace + "::" + c->name) + "_IMPL";
529  const std::string s = std::regex_replace(c->priorText, implRe, "#ifdef " + cppImplSymbol);
530 
531  header <<"\n"
532  <<THIS_LOCATION <<locationDirective(c, c->priorTextToken)
533  <<s <<"\n";
534  impl <<"\n"
535  <<THIS_LOCATION <<"#define " <<cppImplSymbol <<"\n"
536  <<"#include <" <<toPath(c->qualifiedNamespace + "::" + c->name, ".h").string() <<">\n"
537  <<"#include <Rose/Exception.h>\n"
538  <<locationDirective(c, c->priorTextToken)
539  <<s <<"\n";
540  }
541 
542  // Class documentation
543  if (!c->doc.empty()) {
544  header <<"\n"
545  <<THIS_LOCATION <<locationDirective(c, c->docToken)
546  <<c->doc;
547  }
548 
549  // Class opening line: "class NAME [: BASES] {"
550  header <<"\n"
551  <<locationDirective(c, c->nameToken)
552  <<"class " <<c->name;
553  if (c->inheritance.empty()) {
554  header <<": public Sawyer::Tree::Vertex<" <<c->name <<">";
555  } else {
556  for (auto base = c->inheritance.begin(); base != c->inheritance.end(); ++base)
557  header <<(base == c->inheritance.begin() ? ": " : ", ") <<base->first <<" " <<base->second;
558  }
559  header <<" {\n";
560 
561  // Emit generated types at the start of the class
562  header <<THIS_LOCATION <<locationDirective(c, c->startToken)
563  <<"public:\n"
564  <<" // Automatically generated; do not modify!\n"
565  <<" /** Shared-ownership pointer to @ref " <<c->name <<".\n"
566  <<" *\n"
567  <<" * @{ */\n"
568  <<" using Ptr = " <<c->name <<"Ptr;\n"
569  <<" using ConstPtr = " <<c->name <<"ConstPtr;\n"
570  <<" /** @} */\n";
571 
572  // Properties
573  for (const auto &p: c->properties)
574  genProperty(header, impl, p());
575 
576  // User-defined class members
577  if (!c->endText.empty())
578  header <<"\n"
579  <<THIS_LOCATION <<locationDirective(c, c->endTextToken)
580  <<c->endText;
581 
582  // Other generated members
583  genClassConstructors(header, impl, c, h);
584  for (const Serializer::Ptr &serializer: serializers)
585  serializer->genBody(header, impl, c, h, *this);
