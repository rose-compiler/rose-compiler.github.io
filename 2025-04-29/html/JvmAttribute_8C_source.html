358 SgAsmJvmStackMapFrame* SgAsmJvmStackMapFrame::parse(SgAsmJvmConstantPool* pool)
359 {
360  uint16_t numStack{0}, numLocals{0};
361 
362  Jvm::read_value(pool, p_frame_type);
363 
364  switch (p_frame_type) {
365  case 0 ... 63: // frame_type = SAME
366  break;
367  case 64 ... 127: // frame_type = SAME_LOCALS_1_STACK_ITEM
368  numStack = 1;
369  break;
370  case 247: // frame_type = SAME_LOCALS_1_STACK_ITEM_EXTENDED
371  numStack = 1;
372  Jvm::read_value(pool, p_offset_delta);
373  break;
374  case 248 ... 250: // frame_type = CHOP
375  Jvm::read_value(pool, p_offset_delta);
376  break;
377  case 251: // frame_type = SAME_FRAME_EXTENDED
378  Jvm::read_value(pool, p_offset_delta);
379  break;
380  case 252 ... 254: // frame_type = SAME_FRAME_EXTENDED
381  Jvm::read_value(pool, p_offset_delta);
382  numLocals = p_frame_type - 251;
383  break;
384  case 255: { // frame_type = FULL_FRAME
385  Jvm::read_value(pool, p_offset_delta);
386  Jvm::read_value(pool, numLocals);
387 
388  // Unfortunately, locals are read before number_of_stack_items
389  for (int ii = 0; ii < numLocals; ii++) {
390  auto type = new SgAsmJvmStackMapVerificationType(this);
391  type->parse(pool);
392  get_locals().push_back(type);
393  }
394  numLocals = 0;
395 
396  Jvm::read_value(pool, numStack);
397  break;
398  }
399  default:
400  mlog[ERROR] << "SgAsmJvmStackMapFrame::parse: illegal frame type\n";
401  ROSE_ABORT();
402  break;
403  }
404 
405  for (int ii = 0; ii < numLocals; ii++) {
406  auto type = new SgAsmJvmStackMapVerificationType(this);
407  type->parse(pool);
408  get_locals().push_back(type);
409  }
410 
411  for (int ii = 0; ii < numStack; ii++) {
412  auto type = new SgAsmJvmStackMapVerificationType(this);
413  type->parse(pool);
414  get_stack().push_back(type);
415  }
416  return this;
