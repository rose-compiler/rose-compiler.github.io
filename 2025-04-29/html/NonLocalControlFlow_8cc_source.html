99 createJumpTable (SgVariableSymbol* jump_var,
100  ASTtools::JumpMap_t::const_iterator i,
101  ASTtools::JumpMap_t::const_iterator end)
102 {
103  if (i == end)
104  return 0;
105 
106  // Extract information for current iteration (jump statement) i.
107  const SgStatement* jump_stmt = i->first;
108  int jump_val = (int)i->second;
109 
110  // Build 'if' condition
111  SgVarRefExp* jump_var_ref = SageBuilder::buildVarRefExp (jump_var);
112  ROSE_ASSERT (jump_var_ref);
113  SgIntVal* val = SageBuilder::buildIntVal (jump_val);
114  ROSE_ASSERT (val);
115  SgEqualityOp* cond = SageBuilder::buildEqualityOp (jump_var_ref, val);
116  ROSE_ASSERT (cond);
117  SgExprStatement* cond_stmt =
118  SageBuilder::buildExprStatement (cond);
119  ROSE_ASSERT (cond_stmt);
120 
121  // The 'true' branch executes the original jump statement.
122  SgBasicBlock* true_body = SageBuilder::buildBasicBlock ();
123  ROSE_ASSERT (true_body);
124  SgStatement* true_stmt =
125  const_cast<SgStatement *> (isSgStatement (ASTtools::deepCopy (jump_stmt)));
126  ROSE_ASSERT (true_stmt);
127  true_body->append_statement (true_stmt);
128 
129  // The 'false' branch executes any other jump statements.
130  SgBasicBlock* false_body = SageBuilder::buildBasicBlock ();
131  SgStatement* false_stmt = createJumpTable (jump_var, ++i, end);
132  if (false_stmt)
133  false_body->append_statement (false_stmt);
134 
135  // Assemble final 'if'
136  SgIfStmt* new_jump = SageBuilder::buildIfStmt (cond_stmt, true_body, false_body);
137  ROSE_ASSERT (new_jump);
138  return new_jump;

297 
298  bool needNormalization=false;
