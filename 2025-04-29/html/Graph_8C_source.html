123 void
124 Graph::remove (Graph::Node* n)
125 {
126  BaseGraph::remove(n);
127 
128  // remove all the edges incident on this node
129  IncidentEdgesIterator incident(n);
130  while ((bool)incident) {
131  Edge* e = incident;
132  // remove this edge from the neighboring node's list of incident edges
133  if (e->node1() == n)
134  e->node2()->incident_edges.remove(e);
135  else
136  e->node1()->incident_edges.remove(e);
137  // remove this edge
138  BaseGraph::remove(e);
139  delete e; // an edge has no existence without its anchoring nodes
140  ++incident;
141  }
142 }
