249 POETCode* TraceEval(POETCode* vars, POETCode* res)
250 {
251  std::vector<LocalVar*> vec;
252  switch (vars->get_enum()) {
253  case SRC_LVAR: {
254  LocalVar* lvar = static_cast<LocalVar*>(vars);
255  vec.push_back(lvar);
256  break;
257  }
258  case SRC_LIST: {
259  POETList* l = static_cast<POETList*>(vars);
260  while (l != 0) {
261  POETCode* cur = l->get_first();
262  if (cur->get_enum() != SRC_LVAR) INCORRECT_TRACE_HANDLE(cur->toString());
263  vec.push_back(static_cast<LocalVar*>(cur));
264  l = dynamic_cast<POETList*>(l->get_rest());
265  }
266  }
267  break;
268  default: INCORRECT_TRACE_HANDLE(vars->toString());
269  }
270  for (int i = vec.size()-1; i >= 0; --i) {
271  LocalVar* lvar = vec[i];
272  LvarSymbolTable::Entry entry = lvar->get_entry();
273  entry.push(true);
274  entry.set_entry_type(LVAR_TRACE);
275  }
276  res = eval_AST(res);
277  for (int i = vec.size()-1; i >= 0; --i) {
278  LocalVar* lvar = dynamic_cast<LocalVar*>(vec[i]);
279  assert(lvar != 0);
280  LvarSymbolTable::Entry entry = lvar->get_entry();
281  POETCode* val = entry.get_code();
282  entry.pop();
283  entry.set_code(val);
284  }
285  return res;
286 }

348  POETCode* eval_cvar_attr(CodeVar* cv, LocalVar* lv)
349  {
350  CvarSymbolTable::Entry cv_e = cv->get_entry();
351  LvarSymbolTable* symTable = cv_e.get_symTable();
352  LvarSymbolTable::Entry e = lv->get_entry();
353  if (symTable->find(e.get_name()) != lv) return 0;
354 
355  POETCode* res = 0;
356  switch (e.get_entry_type()) {
357  case LVAR_CODEPAR:
358  res = eval_tuple_access(cv->get_args(), e.get_code());
359  break;
360  case LVAR_ATTR:
361  for (int i = 0; i < cv_e.attr_size(); ++i) {
362  if (cv_e.get_attr(i) == lv) {
363  POETCode* tuple = cv->get_attr();
364  if (tuple == 0) res = e.get_code();
365  else res = eval_tuple_access(tuple,i);
366  break;
367  }
368  }
369  break;
370  default: /*QY: access is a local variable. treat this as tuple access*/
371  res = eval_tuple_access(cv, lv);
372  }
373  return res;
374  }

376  POETCode* eval_cvar_access(CodeVar* cv, POETCode* access)
377  {
378  POETCode* res = 0;
379  CvarSymbolTable::Entry cv_e = cv->get_entry();
380  LvarSymbolTable* symTable = cv_e.get_symTable();
381 
382  symTable->push_table(true); /* push symbol table for evaluation of access*/
383  if (access->get_enum() != SRC_LVAR)
384  access= apply(access);
385  switch (access->get_enum()) {
386  case SRC_LVAR: {
387  cv->get_entry().get_symTable()->pop_table();
388  LocalVar* lv = static_cast<LocalVar*>(access);
389  res = eval_cvar_attr(cv, lv);
390  if (res == 0) {
391 
392  CVAR_ACC_MISMATCH( cv,access);
393  }
394  break;
395  }
396  case SRC_OP: {
397  POETOperator* op = static_cast<POETOperator*>(access);
398  if (op->get_op() == TYPE_TOR) {
399  cv->get_entry().get_symTable()->pop_table();
400  for (int i = 0; i < op->numOfArgs(); ++i) {
401  POETCode* cur = apply(op->get_arg(i));
402  LocalVar* lv = dynamic_cast<LocalVar*>(cur);
403  if (lv == 0) { CVAR_ACC_MISMATCH( cv,access); }
404  res = eval_cvar_attr(cv, lv);
405  if (res != 0) return res;
406  }
407  CVAR_ACC_MISMATCH( cv,access);
408  }
409  }
410  default:
411  if (cv->get_args() != 0) CVAR_ACC_MISMATCH( cv, access);
412  /*QY: access is used to set optional attribute values of code template;
413  return a new code template object with the given attribute*/
414  res = ASTFactory::inst()->build_codeRef(cv->get_entry(),0);
415  cv->get_entry().get_symTable()->pop_table();
416  break;
417  }
418  return res;

511 void XformEvalVisitor::visitOperator(POETOperator* op)
512  {
513  POETCode *r1 = op->get_arg(0), *r2 = op->get_arg(1), *r3 = op->get_arg(2), *r4 = op->get_arg(3);
514  switch(op->get_op()) {
515  case POET_OP_IFELSE: case POET_OP_CASE: case POET_OP_FOR: case POET_OP_FOREACH: case POET_OP_CONTINUE: case POET_OP_BREAK: case POET_OP_ERROR:
516  case POET_OP_CLEAR: case POET_OP_ASTMATCH: case POET_OP_TYPEMATCH: case POET_OP_ANNOT: case POET_OP_TYPEMATCH_Q:
517  case POET_OP_TRACE: case POET_OP_DELAY: case POET_OP_AND: case POET_OP_TUPLE: case POET_OP_LIST: case POET_OP_LIST1:
518  case POET_OP_DEBUG: case POET_OP_PRINT: case POET_OP_OR: case POET_OP_ERASE:
519  break;
520  default:
521  if (r1 != 0) { r1->visit(this); r1 = res; }
522  if (r2 != 0) { r2->visit(this); r2 = res; }
523  if (r3 != 0) { r3->visit(this); r3 = res; }
524  if (r4 != 0) { r4->visit(this); r4 = res; }
525  }
526 
527  if (op->get_op() == POET_OP_ERROR)
528  {
529  std::cerr << "User Error: ";
530  print_AST(std::cerr, r1);
531  USER_EXIT();
532  }
533  try {
534  switch (op->get_op()) {
535  case POET_OP_VAR:
536  case POET_OP_XFORM:
537  case POET_OP_CODE:
538  case POET_OP_EXP:
539  assert (r1 == 0);
540  case TYPE_LIST: case TYPE_LIST1:
541  case POET_OP_RANGE:
542  case POET_OP_TUPLE:
543  case POET_OP_LIST:
544  case POET_OP_LIST1:
545  case POET_OP_ANNOT:
546  res = op; return;
547  case TYPE_TOR:
548  r1->visit(this); r1 = res;
549  for (int i = 1; i < op->numOfArgs();++i) {
550  r2 = op->get_arg(i);
551  r2->visit(this); r2 = res;
552  r1 = POETProgram::make_typeTor(r1,r2);
553  }
554  res = r1;
555 //std::cerr << "MAKING TOR :" << res->toString() << " FROM " << op->toString() << "\n";
556  return;
557  case POET_OP_ASSERT:
558  assert(r1 != 0);
559  if (r1 == ZERO) ASSERT_FAIL(op->get_arg(0));
560  res = EMPTY;
561  return;
562  case POET_OP_PRINT:
563  assert(r1 != 0);
564  print_AST(std::cerr, r1);
565  res = EMPTY;
566  return;
567  case POET_OP_DEBUG: {
568  assert(r1 != 0);
569  int userDebugSave = user_debug;
570  user_debug = AST2Int(r1);
571  std::cerr << "\nDEBUGGING at level " << user_debug << "\n";
572  r2->visit(this);
573  std::cerr << "END DEBUGGING at level " << user_debug << "\n";
574  POETOperator* r2op = dynamic_cast<POETOperator*>(r2);
575  if (r2op == 0 || r2op->get_op() < POET_OP_SEQ)
576  std::cerr << "DEBUGGING result: " << r2->toString() << "===>" << res->get_className() << ":" << res->toString() << "\n";
577  user_debug = userDebugSave;
578  return;
579  }
580  case POET_OP_NOT: res = fac->new_iconst(!AST2Bool(res)); return;
581  case POET_OP_AND:
582  r1->visit(this); r1 = res;
583  if (AST2Bool(r1)) {
584  r2->visit(this); r2 = res;
585  res = fac->new_iconst(AST2Bool(r2));
586  }
587  else
588  res = fac->new_iconst(0);
589  return;
590  case POET_OP_OR:
591  r1->visit(this); r1 = res;
592  if (!AST2Bool(r1)) {
593  r2->visit(this); r2 = res;
594  res = fac->new_iconst(AST2Bool(r2));
595  }
596  else
597  res = fac->new_iconst(1);
598  return;
599  case POET_OP_UMINUS: res = build_Uop(op->get_op(), res, true); break;
600  case POET_OP_MINUS:
601  case POET_OP_PLUS:
602  case POET_OP_MULT:
603  case POET_OP_DIVIDE:
604  case POET_OP_MOD:
605  res = build_Bop(op->get_op(), r1,r2, true);
606  return;
607  case POET_OP_LT: case POET_OP_LE: case POET_OP_GT: case POET_OP_GE: case POET_OP_EQ: case POET_OP_NE:
608  res = build_Bop(op->get_op(), r1, r2, false); return;
609  case POET_OP_DOT: res = ASTFactory::inst()->make_attrAccess(r1,r2); return;
610  case POET_OP_CAR:

613  else if (r1->get_enum() == SRC_LIST)
614  res = static_cast<POETList*>(r1)->get_first();
615  break;
616  case POET_OP_CDR:
617  r1 = EvalTrace(r1);
618  if (r1 == EMPTY_LIST) { INCORRECT_LIST(r1); }
619  else if (r1->get_enum() == SRC_LIST) {
620  res = static_cast<POETList*>(r1)->get_rest();

622  }
623  else res = EMPTY_LIST;
624  break;
625  case POET_OP_CONS: {

627  POETList* l2 = 0;
628  if (r2 != 0) l2 = dynamic_cast<POETList*>(EvalTrace(r2));
629  if (l2 != 0 || r2 == 0 || r2 == EMPTY_LIST)
630  res = fac->new_list(r1,r2);
631  else
632  res = fac->new_list(r1, fac->new_list(r2,0));
633  return;
634  }
635  case POET_OP_POND: {
636  assert(r1 != 0 && r2 != 0 && r3 == 0);
637  res = build_codeRef(r1,r2,false);
638  return;
639  }
640  case POET_OP_DELAY: res = r1; return;
641  case POET_OP_APPLY: r1 = EvalTrace(r1); r1->visit(this); return;
642  case POET_OP_ASTMATCH: {
643  r1->visit(this); r1 = res;
644  res = match_AST(r1, r2, MATCH_AST_PATTERN);
645  res = fac->new_iconst(res != 0);
646  break;
647  }
648  case POET_OP_CLEAR:

651  if (r1->get_enum() != SRC_LVAR)
652  INCORRECT_VAR(r1->toString());

654  res = r1;
655  return;
656  case POET_OP_MAP: {
657  POETMap* m =new POETMap();
658  res = m;
659  if (r1 != 0 && r2 == 0 && r3 == 0) {
660  POETList* p1 = dynamic_cast<POETList*>(r1);

662  POETTuple* cur = dynamic_cast<POETTuple*>(r1);

673  }

678  }
679  case POET_OP_SPLIT: {

688  break;
689  }
690  case SRC_STRING:

694  default:

700  case POET_OP_REPLACE:

705  case POET_OP_DUPLICATE: res = eval_duplicate(r1,r2,r3); break;
706  case POET_OP_COPY: { res = eval_copy(r1); break; }
707  case POET_OP_CONCAT:
708  assert(r1 != 0 && r2 != 0 && r3 == 0);

718  else XFORM_PARAM_INCORRECT("PERMUTE", r1->toString());

732  targets[j-1] = cur;
733  }

735  break;
736  }

818  case POET_OP_BREAK:

844  while (AST2Bool(rr2)) {
845  if (r4 != 0)
846  try {r4->visit(this); }
847  catch (POETOperator* c) {
848  if (c->get_op() == POET_OP_BREAK)
849  break;
850  }
851  if (r3 != 0) r3->visit(this);
852  if (user_debug)
853  std::cerr << "FOR incr: " << r3->toString(DEBUG_NO_VAR) << "===>" << SHORT(res->toString(DEBUG_NO_VAR),500) << "\n";
854  r2->visit(this); rr2 = res;
855  if (user_debug)
856  std::cerr << "FOR cond: " << r2->toString(DEBUG_NO_VAR) << "===>" << SHORT(rr2->toString(DEBUG_NO_VAR),500) << "\n";
857  }
858  res = EMPTY;
859  return;
860  }
861  case POET_OP_CASE: {
862  assert(r1 != 0 && r2 != 0 && r3 == 0);
863  r1->visit(this);
864  if (user_debug)
865  std::cerr << "CASE cond: " << r1->toString(DEBUG_NO_VAR) << "===>" << res->get_className() << ":" << res->toString(DEBUG_NO_VAR) << "\n";
866  r1 = res;
867  POETTuple* v2 = dynamic_cast<POETTuple*>(r2);
868  if (v2 == 0) v2 = ASTFactory::inst()->append_tuple(0, r2);
869  bool succ=false;
870  for (int i = 0; i < v2->size(); ++i) {
871  POETList* cur = dynamic_cast<POETList*>(v2->get_entry(i));
872  if (cur == 0) {
873  std::cerr << "v2[i]=" << v2->get_entry(i)->toString() << "\n";
874  std:: cerr << OpName[op->get_op()] << "\n";
875  assert(false);
876  }
877  if (match_AST(r1, cur->get_first(), MATCH_AST_PATTERN)) {
878  succ = true;
879  get_head(cur->get_rest())->visit(this);
880  if (user_debug)
881  std::cerr << "CASE " << r1->toString() << " matched label: " << cur->get_first()->toString(DEBUG_NO_VAR) << "===>" << res->toString(DEBUG_NO_VAR) << "\n";
882  break;
883  }
884  }
885  if (!succ) {
886  CASE_NOTFOUND( r1);
887  }
888  return;
889  }
890  default:
891  std::cerr << "ERROR: undefined operator: " << op->toString() << "\n";
892  assert(0);
893  }
894  }
895  catch (Error err) {
896  std::cerr << "From operator " << SHORT(op->toString(DEBUG_NO_VAR),500) << "\n";
897  throw err;
898  }
899  if (user_debug) {
900  std::cerr << op->toString() << "\n===>" << res->toString() << "\n";
901  }
