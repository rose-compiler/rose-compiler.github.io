545 // Updates progress information during the main partitioning phase. The progress is the ratio of the number of bytes
546 // represented in the CFG to the number of executable bytes in the memory map.
547 void
548 Partitioner::updateCfgProgress() {
549  {
550  SAWYER_THREAD_TRAITS::LockGuard lock(mutex_);
551  if (!progress_)
552  return;
553  }
554 
555  // How many bytes are mapped with execute permission
556  if (0 == cfgProgressTotal_) {
557  for (const MemoryMap::Node &node: memoryMap_->nodes()) {
558  if (0 != (node.value().accessibility() & MemoryMap::EXECUTABLE))
559  cfgProgressTotal_ += node.key().size();
560  }
561  }
562 
563  double completion = cfgProgressTotal_ > 0 ? (double)nBytes() / cfgProgressTotal_ : 0.0;
564  updateProgress("partition", completion);
565 
566  // All partitioners share a single progress bar for the CFG completion amount
567  static Sawyer::ProgressBar<size_t, ProgressBarSuffix> *bar = nullptr;
568  if (!bar)
569  bar = new Sawyer::ProgressBar<size_t, ProgressBarSuffix>(cfgProgressTotal_, mlog[MARCH], "cfg");
570 
571  // Update the progress bar. If multiple partitioners are sharing the progress bar then also make sure that the lower and
572  // upper limits are appropriate for THIS partitioner. However, changing the limits is a configuration change, which also
573  // immediately updates the progress bar (we don't want that, so update only if necessary).
574  if (cfgProgressTotal_) {
575  bar->suffix(ProgressBarSuffix(this));
576  if (bar->domain().first!=0 || bar->domain().second!=cfgProgressTotal_) {
577  bar->value(0, nBytes(), cfgProgressTotal_);
578  } else {
579  bar->value(nBytes());
580  }
581  }

611 Partitioner::instructionCrossReferences(const AddressIntervalSet &restriction) const {

614  struct Accumulator: AstSimpleProcessing {
615  const AddressIntervalSet &restriction;
616  CrossReferences &xrefs;
617  Reference to;
618  Accumulator(const AddressIntervalSet &restriction, CrossReferences &xrefs): restriction(restriction), xrefs(xrefs) {}
619  void target(const Reference &to) { this->to = to; }
620  void visit(SgNode *node) {
621  if (SgAsmIntegerValueExpression *ival = isSgAsmIntegerValueExpression(node)) {
622  Address n = ival->get_absoluteValue();
623  if (restriction.exists(n))
624  xrefs.insertMaybeDefault(Reference(n)).insert(to);

626  }
627  } accumulator(restriction, xrefs);
628 
629  for (const BasicBlock::Ptr &bblock: basicBlocks()) {
630  std::vector<Function::Ptr> functions = functionsOwningBasicBlock(bblock);
631  for (SgAsmInstruction *insn: bblock->instructions()) {
632  for (const Function::Ptr &function: functions) {
633  accumulator.target(Reference(function, bblock, insn));
634  accumulator.traverse(insn, preorder);
635  }
636  }
637  }
638  return xrefs;
639 }

862 BasicBlock::Ptr
863 Partitioner::discoverBasicBlockInternal(Address startVa) const {
864  // If the first instruction of this basic block already exists (in the middle of) some other basic blocks then the other
865  // basic blocks are called "conflicting blocks". This only applies for the first instruction of this block, but is used in
866  // the termination conditions below.
867  AddressUser startVaOwners = instructionExists(startVa);
868  if (startVaOwners)
869  ASSERT_forbid(startVaOwners.isBlockEntry()); // handled in discoverBasicBlock
870 
871  // Keep adding instructions until we reach a termination condition. The termination conditions are enumerated in detail in
872  // the doxygen documentation for this function. READ IT AND KEEP IT UP TO DATE!!!
873  BasicBlock::Ptr retval = BasicBlock::instance(startVa, sharedFromThis());
874  Address va = startVa;
875  while (1) {
876  SgAsmInstruction *insn = discoverInstruction(va);
877  if (insn==nullptr) // case: no instruction available
878  goto done;
879  retval->append(sharedFromThis(), insn);
880  if (architecture()->isUnknown(insn) && !settings_.ignoringUnknownInsns) // case: "unknown" instruction
881  goto done;
882 
883  // Give user chance to adjust basic block successors and/or pre-compute cached analysis results
884  BasicBlockCallback::Results userResult;
885  basicBlockCallbacks_.apply(true, BasicBlockCallback::Args(sharedFromThis(), retval, userResult));
886 
887  for (Address successorVa: basicBlockConcreteSuccessors(retval)) {
888  if (successorVa!=startVa && retval->instructionExists(successorVa)) { // case: successor is inside our own block
889  retval->pop();
890  goto done;
891  }
892  }
893 
894  if (userResult.terminate == BasicBlockCallback::TERMINATE_NOW) // case: user wants us to terminate block here
895  goto done;
896  if (userResult.terminate == BasicBlockCallback::TERMINATE_PRIOR) {
897  retval->pop(); // case: user wants to terminate at prior insn
898  goto done;
899  }
900 
901  if (config_.basicBlockFinalInstructionVa(startVa).orElse(va+1)==va) // case: terminated by configuration
902  goto done;
903 
904  if (basicBlockIsFunctionCall(retval)) // case: bb looks like a function call
905  goto done;
906  BasicBlock::Successors successors = basicBlockSuccessors(retval);
907 
908  if (successors.size()!=1) // case: not exactly one successor
909  goto done;
910  SValue::Ptr successorExpr = successors.front().expr();
911 
912  if (!successorExpr->isConcrete()) // case: successor is indeterminate
913  goto done;
914  Address successorVa = successorExpr->toUnsigned().get();
915 
916  if (successorVa == startVa) // case: successor is our own basic block
917  goto done;
918 
919  if (findPlaceholder(successorVa)!=cfg_.vertices().end()) // case: successor is an existing block
920  goto done;
921 
922  AddressUser succVaOwners = instructionExists(successorVa);
923  if (succVaOwners && // case: successor is inside an existing block that
924  !isSupersetUnique(startVaOwners.basicBlocks(), // doesn't own startVa
925  succVaOwners.basicBlocks(),
926  sortBasicBlocksByAddress)) {
927  goto done;
928  }
929 
930  va = successorVa;
931  }
932 done:
933 
934  // We're terminating the basic block. If the configuration specifies successors for this block at this instruction then
935  // use them instead of what we might have already calculated.
936  if (!retval->instructions().empty()) {
937  Address finalInsnVa = retval->instructions().back()->get_address();
938  if (config_.basicBlockFinalInstructionVa(startVa).orElse(finalInsnVa+1)==finalInsnVa) {
939  retval->clearSuccessors();
940  size_t nBits = instructionProvider_->instructionPointerRegister().nBits();
941  std::set<Address> successorVas = config_.basicBlockSuccessorVas(startVa);
942  for (Address successorVa: successorVas)
943  retval->insertSuccessor(successorVa, nBits);
944  }
945  }
946 
947  retval->freeze();
948  return retval;
949 }

1009 void
1010 Partitioner::attachBasicBlock(const ControlFlowGraph::ConstVertexIterator &constPlaceholder, const BasicBlock::Ptr &bblock) {
1011  ASSERT_require(cfg_.isValidVertex(constPlaceholder));
1012  ASSERT_require(constPlaceholder->value().type() == V_BASIC_BLOCK);
1013  ASSERT_not_null(bblock);
1014  ControlFlowGraph::VertexIterator placeholder = cfg_.findVertex(constPlaceholder->id());
1015 
1016  if (placeholder->value().address() != bblock->address()) {
1017  throw PlaceholderError(placeholder->value().address(),
1018  "placeholder " + StringUtility::addrToString(placeholder->value().address()) +
1019  " cannot hold " + basicBlockName(bblock));
1020  }
1021  if (placeholder->value().bblock() == bblock)
1022  return; // nothing to do since basic block is already in the CFG
1023  if (placeholder->value().bblock() != nullptr) {
1024  throw PlaceholderError(placeholder->value().address(),
1025  "placeholder " + StringUtility::addrToString(placeholder->value().address()) +
1026  " already holds a different basic block");
1027  }
1028 
1029  if (bblock->sourceLocation().isEmpty())
1030  bblock->sourceLocation(sourceLocations_(bblock->address()));
1031 
1032  // Adjust the basic block according to configuration information. Configuration overrides automatically-detected values.
1033  const BasicBlockConfiguration &c = config_.basicBlock(bblock->address());
1034  if (!c.comment().empty())
1035  bblock->comment(c.comment());
1036  if (!c.sourceLocation().isEmpty())
1037  bblock->sourceLocation(c.sourceLocation());
1038 
1039  bblock->freeze();
1040 
1041  // Are any edges marked as function calls? If not, and this is a function call, then we'll want to convert all the normal
1042  // edges to function call edges. Similarly for function return edges.
1043  bool hasFunctionCallEdges = false, hasFunctionReturnEdges = false, hasCallReturnEdges = false;
1044  for (const BasicBlock::Successor &successor: basicBlockSuccessors(bblock)) {
1045  switch (successor.type()) {
1046  case E_FUNCTION_CALL: hasFunctionCallEdges = true; break;
1047  case E_FUNCTION_RETURN: hasFunctionReturnEdges = true; break;
1048  case E_CALL_RETURN: hasCallReturnEdges = true; break;
1049  default: break;
1050  }
1051  }
1052  bool convertEdgesToFunctionCalls = !hasFunctionCallEdges && basicBlockIsFunctionCall(bblock);
1053  bool convertEdgesToFunctionReturns = !hasFunctionReturnEdges && basicBlockIsFunctionReturn(bblock);
1054 
1055  // Make sure placeholders exist for the concrete successors
1056  bool hadIndeterminate = false;
1057  typedef std::pair<ControlFlowGraph::VertexIterator, CfgEdge> VertexEdgePair;
1058  std::vector<VertexEdgePair> successors;
1059  for (const BasicBlock::Successor &successor: basicBlockSuccessors(bblock)) {
1060  EdgeType edgeType = successor.type();
1061  if (edgeType == E_NORMAL) {
1062  if (convertEdgesToFunctionCalls) {
1063  edgeType = E_FUNCTION_CALL;
1064  } else if (convertEdgesToFunctionReturns) {
1065  edgeType = E_FUNCTION_RETURN;
1066  }
1067  }
1068 
1069  CfgEdge edge(edgeType, successor.confidence());
1070  if (auto sucval = successor.expr()->toUnsigned()) {
1071  successors.push_back(VertexEdgePair(insertPlaceholder(*sucval), edge));
1072  } else if (!hadIndeterminate) {
1073  successors.push_back(VertexEdgePair(indeterminateVertex_, edge));
1074  hadIndeterminate = true;
1075  }
1076  }
1077 
1078  // Call-return edges indicate where a function call eventually returns since the CFG doesn't have such edges from function
1079  // return statements (return statements usually have an indeterminate successor). We'll only add the call-return edge if
1080  // the user didn't already specify one and if the may-return analysis is positive.
1081  if (autoAddCallReturnEdges_ && !hasCallReturnEdges && basicBlockIsFunctionCall(bblock)) {
1082  for (const VertexEdgePair &successor: successors) {
1083  if (successor.second.type() == E_FUNCTION_CALL) {
1084  bool mayReturn = false;
1085  if (basicBlockOptionalMayReturn(bblock).assignTo(mayReturn)) {
1086  // may-return is provably true or false
1087  CfgEdge edge(E_CALL_RETURN, PROVED);
1088  successors.push_back(VertexEdgePair(insertPlaceholder(bblock->fallthroughVa()), edge));
1089  break;
1090  } else if (assumeFunctionsReturn_) {
1091  // assume functions return without proving it
1092  CfgEdge edge(E_CALL_RETURN, ASSUMED);
1093  successors.push_back(VertexEdgePair(insertPlaceholder(bblock->fallthroughVa()), edge));
1094  break;
1095  }
1096  }
1097  }
1098  }
1099 
1100  // Make CFG edges
1101  cfg_.clearOutEdges(placeholder);
1102  for (const VertexEdgePair &pair: successors)
1103  cfg_.insertEdge(placeholder, pair.first, pair.second);
1104 
1105  // Insert the basic block instructions into the AUM
1106  placeholder->value().bblock(bblock);
1107  for (SgAsmInstruction *insn: bblock->instructions())
1108  aum_.insertInstruction(insn, bblock);
1109  if (bblock->isEmpty())
1110  adjustNonexistingEdges(placeholder);
1111 
1112  // Insert the basic block static data. If the AUM contains an equivalent data block already, then use that one instead.
1113  for (const DataBlock::Ptr &dblock: bblock->dataBlocks()) {
1114  ASSERT_not_null(dblock);
1115  DataBlock::Ptr insertedDb = attachDataBlock(dblock);
1116  ASSERT_not_null(insertedDb);
1117  if (insertedDb != dblock)
1118  bblock->replaceOrInsertDataBlock(insertedDb);
1119  insertedDb->insertOwner(bblock);
1120  }
1121 
1122  if (basicBlockSemanticsAutoDrop())
1123  bblock->dropSemantics(sharedFromThis());
1124 
1125  bblockAttached(placeholder);
1126 }

1148 BasicBlock::Successors
1149 Partitioner::basicBlockSuccessors(const BasicBlock::Ptr &bb, Precision::Level precision) const {
1150  ASSERT_not_null(bb);
1151  BasicBlock::Successors successors;
1152 
1153  if (bb->isEmpty() || bb->successors().getOptional().assignTo(successors))
1154  return successors;
1155 
1156  SgAsmInstruction *lastInsn = bb->instructions().back();
1157 
1158  BasicBlockSemantics sem = bb->semantics();
1159  BaseSemantics::State::Ptr state;
1160  if (settings_.ignoringUnknownInsns && architecture()->isUnknown(lastInsn)) {
1161  // Special case for "unknown" instructions... the successor is assumed to be the fall-through address.
1162  Address va = lastInsn->get_address() + lastInsn->get_size();
1163  size_t nBits = instructionProvider_->wordSize();
1164  BaseSemantics::RiscOperators::Ptr ops = newOperators();
1165  successors.push_back(BasicBlock::Successor(Semantics::SValue::promote(ops->number_(nBits, va))));
1166  } else if (precision > Precision::LOW && (state = sem.finalState())) {
1167  // Use our own semantics if we have them.
1168  ASSERT_not_null(sem.dispatcher);
1169  ASSERT_not_null(sem.operators);
1170  RegisterDescriptor REG_IP = instructionProvider_->instructionPointerRegister();
1171  std::vector<Semantics::SValue::Ptr> worklist(1, Semantics::SValue::promote(sem.operators->peekRegister(REG_IP)));
1172  while (!worklist.empty()) {
1173  Semantics::SValue::Ptr pc = worklist.back();
1174  worklist.pop_back();
1175 
1176  // Special handling for if-then-else expressions
1177  if (SymbolicExpression::InteriorPtr ifNode = pc->get_expression()->isInteriorNode()) {
1178  if (ifNode->getOperator()==SymbolicExpression::OP_ITE) {
1179  Semantics::SValue::Ptr expr = Semantics::SValue::promote(sem.operators->undefined_(ifNode->nBits()));
1180  expr->set_expression(ifNode->child(1));
1181  worklist.push_back(expr);
1182  expr = Semantics::SValue::promote(sem.operators->undefined_(ifNode->nBits()));
1183  expr->set_expression(ifNode->child(2));
1184  worklist.push_back(expr);
1185  continue;
1186  }
1187  }
1188 
1189  successors.push_back(BasicBlock::Successor(pc));
1190  }
1191  } else {
1192  // We don't have semantics, so naively look at just the last instruction.
1193  bool complete = true;
1194  size_t nBits = instructionProvider_->wordSize();
1195  AddressSet successorVas = architecture()->getSuccessors(lastInsn, complete/*out*/);
1196 
1197  BaseSemantics::RiscOperators::Ptr ops = newOperators();
1198  for (Address va: successorVas.values())
1199  successors.push_back(BasicBlock::Successor(Semantics::SValue::promote(ops->number_(nBits, va))));
1200  if (!complete)
1201  successors.push_back(BasicBlock::Successor(Semantics::SValue::promote(ops->undefined_(nBits))));
1202  }
1203 
1204  // We don't want parallel edges in the CFG, so remove duplicates.
1205  std::sort(successors.begin(), successors.end(), sortByExpression);
1206  for (size_t i=1; i<successors.size(); /*void*/) {
1207  if (successors[i-1].expr()->get_expression()->isEquivalentTo(successors[i].expr()->get_expression())) {
1208  ASSERT_require(successors[i-1].type() == successors[i].type());
1209  successors.erase(successors.begin()+i);
1210  } else {
1211  ++i;
1212  }
1213  }
1214 
1215  // Cache the result, but not for low-precision results since that would mean a later call for high-precision results would
1216  // only return low-precision results.
1217  if (precision > Precision::LOW)
1218  bb->successors() = successors;
1219  return successors;
1220 }

1223 Partitioner::basicBlockGhostSuccessors(const BasicBlock::Ptr &bb) const {
1224  ASSERT_not_null(bb);

1226 
1227  if (bb->isEmpty() || bb->ghostSuccessors().getOptional().assignTo(ghosts))

1278 Partitioner::basicBlockPopsStack(const BasicBlock::Ptr &bb) const {
1279  ASSERT_not_null(bb);
1280 
1281  do {
1282  if (bb->popsStack().isCached())
1283  break;
1284 
1285  // We need instruction semantics, or return false
1286  BasicBlockSemantics sem = bb->undropSemantics(sharedFromThis());
1287  if (!sem.dispatcher) {
1288  bb->popsStack() = false;
1289  break;
1290  }
1291  ASSERT_not_null(sem.operators);
1292 
1293  // Get the block initial and final states
1294  BaseSemantics::State::Ptr state0 = sem.initialState;
1295  BaseSemantics::State::Ptr stateN = sem.finalState();
1296  if (!state0 || !stateN) {
1297  bb->popsStack() = false;
1298  break;
1299  }
1300 
1301  // Get initial and final stack pointer values
1302  const RegisterDescriptor REG_SP = instructionProvider_->stackPointerRegister();
1303  BaseSemantics::SValue::Ptr sp0 =
1304  state0->peekRegister(REG_SP, sem.operators->undefined_(REG_SP.nBits()), sem.operators.get());
1305  BaseSemantics::SValue::Ptr spN =
1306  stateN->peekRegister(REG_SP, sem.operators->undefined_(REG_SP.nBits()), sem.operators.get());
1307 
1308  // Did the basic block pop the return value from the stack? This impossible to determine unless we assume that the stack
1309  // has an initial value that's not near the minimum or maximum possible value. Therefore, we'll substitute a concrete
1310  // value for the stack pointer.
1311  SymbolicExpression::Ptr sp0ExprOrig = Semantics::SValue::promote(sp0)->get_expression();
1312  SymbolicExpression::Ptr sp0ExprNew = SymbolicExpression::makeIntegerConstant(REG_SP.nBits(), 0x8000); // arbitrary
1313  SymbolicExpression::Ptr spNExpr =
1314  Semantics::SValue::promote(spN)->get_expression()->substitute(sp0ExprOrig, sp0ExprNew, sem.operators->solver());
1315 
1316  // FIXME[Robb P Matzke 2016-11-15]: assumes stack grows down.
1317  // SPn > SP0 == true implies at least one byte popped.
1318  SymbolicExpression::Ptr cmpExpr = SymbolicExpression::makeGt(spNExpr, sp0ExprNew, sem.operators->solver());
1319  bb->popsStack() = cmpExpr->mustEqual(SymbolicExpression::makeBooleanConstant(true));
1320  } while (0);
1321 
1322 #if 0 // [Robb Matzke 2019-01-16]: commented out to debug race
1323  if (sem.wasDropped)
1324  bb->dropSemantics(*this);
1325 #endif
1326  ASSERT_require(bb->popsStack().isCached());
1327  return bb->popsStack().get();
1328 }

1330 bool
1331 Partitioner::basicBlockIsFunctionCall(const BasicBlock::Ptr &bb, Precision::Level precision) const {
1332  ASSERT_not_null(bb);
1333  bool retval = false;
1334 
1335  if (bb->isEmpty() || bb->isFunctionCall().getOptional().assignTo(retval))
1336  return retval; // already cached
1337 
1338  // Avoid infinite recursion
1339  static size_t recursionDepth = 0;
1340  RecursionCounter recursion(recursionDepth);
1341  if (recursionDepth > 2 /*arbitrary*/)
1342  return false; // if we can't prove it, assume false
1343 
1344  SgAsmInstruction *lastInsn = bb->instructions().back();
1345 
1346  // Use our own semantics if we have them.
1347  if (precision > Precision::LOW) {
1348  BasicBlockSemantics sem = bb->semantics();
1349  if (BaseSemantics::State::Ptr state = sem.finalState()) {
1350  ASSERT_not_null(sem.dispatcher);
1351  ASSERT_not_null(sem.operators);
1352  Address returnVa = bb->fallthroughVa();
1353  const RegisterDescriptor REG_IP = instructionProvider_->instructionPointerRegister();
1354 
1355  // Check whether the last instruction is a CALL (or similar) instruction.
1356  bool isInsnCall = architecture()->isFunctionCallFast(bb->instructions(), nullptr, nullptr);
1357 
1358  // Check whether the basic block has the semantics of a function call.
1359  //
1360  // For stack-based calling, after the call the top of the stack will contain the address of the instruction
1361  // immediately following the call. Depending on the memory state, if the stack pointer is not a concrete value
1362  // then reading the top of the stack might not return the same thing we just wrote there (due to trying to resolve
1363  // aliasing in the memory state).
1364  //
1365  // FIXME[Robb P Matzke 2016-11-15]: This only works for stack-based calling conventions.
1366  // Is the block fall-through address equal to the value on the top of the stack?
1367  bool isSemanticCall = false;
1368  if (!isInsnCall) {
1369  const RegisterDescriptor REG_SP = instructionProvider_->stackPointerRegister();
1370  const RegisterDescriptor REG_SS = instructionProvider_->stackSegmentRegister();
1371  BaseSemantics::SValue::Ptr returnExpr = sem.operators->number_(REG_IP.nBits(), returnVa);
1372  BaseSemantics::SValue::Ptr sp = sem.operators->peekRegister(REG_SP);
1373  BaseSemantics::SValue::Ptr topOfStack = sem.operators->undefined_(REG_IP.nBits());
1374  topOfStack = sem.operators->peekMemory(REG_SS, sp, topOfStack);
1375  BaseSemantics::SValue::Ptr z =
1376  sem.operators->equalToZero(sem.operators->add(returnExpr,
1377  sem.operators->negate(topOfStack)));
1378  isSemanticCall = z->isTrue();
1379  }
1380 
1381  // Defintely not a function call if it neither has semantics or a call or looks like a call.
1382  if (!isInsnCall && !isSemanticCall) {
1383  bb->isFunctionCall() = false;
1384  return false;
1385  }
1386 
1387  // If the only successor is also the fall-through address then this isn't a function call. This case handles code
1388  // that obtains the code address in position independent code. For example, x86 "A: CALL B; B: POP EAX" where A and
1389  // B are consecutive instruction addresses.
1390  BasicBlock::Successors successors = basicBlockSuccessors(bb);
1391  if (1 == successors.size() && successors[0].expr()->toUnsigned().isEqual(returnVa)) {
1392  bb->isFunctionCall() = false;
1393  return false;
1394  }
1395 
1396  // If all callee blocks pop the return address without returning, then this perhaps isn't a function call after
1397  // all.
1398  if (checkingCallBranch()) {
1399  bool allCalleesPopWithoutReturning = true; // all callees pop return address but don't return?
1400  for (const BasicBlock::Successor &successor: successors) {
1401  // Find callee basic block
1402  if (!successor.expr() || !successor.expr()->toUnsigned()) {
1403  allCalleesPopWithoutReturning = false;
1404  break;
1405  }
1406  Address calleeVa = successor.expr()->toUnsigned().get();
1407  BasicBlock::Ptr calleeBb = basicBlockExists(calleeVa);
1408  if (!calleeBb)
1409  calleeBb = discoverBasicBlock(calleeVa); // could cause recursion
1410  if (!calleeBb) {
1411  allCalleesPopWithoutReturning = false;
1412  break;
1413  }
1414 
1415  // If the called block is also a function return (i.e., we're calling a function which is only one block
1416  // long), then of course the block will pop the return address even though it's a legitimate function.
1417  if (basicBlockIsFunctionReturn(calleeBb)) {
1418  allCalleesPopWithoutReturning = false;
1419  break;
1420  }
1421 
1422  if (!basicBlockPopsStack(calleeBb)) {
1423  allCalleesPopWithoutReturning = false;
1424  break;
1425  }
1426 
1427  // Did the callee return to somewhere other than caller's return address?
1428  if (BaseSemantics::State::Ptr calleeStateN = calleeBb->semantics().finalState()) {
1429  BaseSemantics::SValue::Ptr ipN =
1430  calleeStateN->peekRegister(REG_IP, sem.operators->undefined_(REG_IP.nBits()), sem.operators.get());
1431  if (ipN->toUnsigned().isEqual(returnVa)) {
1432  allCalleesPopWithoutReturning = false;
1433  break;
1434  }
1435  }
1436  }
1437  if (allCalleesPopWithoutReturning) {
1438  bb->isFunctionCall() = false;
1439  return false;
1440  }
1441  }
1442 
1443  // This appears to be a function call
1444  bb->isFunctionCall() = true;
1445  return true;
1446  }
1447  }
1448 
1449  // An x86 idiom is two consecutive instructions "call next: pop ebx" where "next" is the address of the POP instruction
1450  // that immediately follows the CALL instruction.
1451  if (SgAsmX86Instruction *i1 = isSgAsmX86Instruction(lastInsn)) {
1452  if (i1->get_kind() == x86_call) {
1453  Address callFallThroughVa = i1->get_address() + i1->get_size();
1454  bool complete = true;
1455  AddressSet callSuccVas = architecture()->getSuccessors(i1, complete/*out*/);
1456  if (callSuccVas.size() == 1 && callSuccVas.least() == callFallThroughVa) {
1457  if (SgAsmX86Instruction *i2 = isSgAsmX86Instruction(discoverInstruction(callFallThroughVa))) {
1458  if (i2->get_kind() == x86_pop) {
1459  bb->isFunctionCall() = false;
1460  return false;
1461  }
1462  }
1463  }
1464  }
1465  }
1466 
1467  // We don't have semantics, so delegate to the SgAsmInstruction subclass.
1468  retval = architecture()->isFunctionCallFast(bb->instructions(), nullptr, nullptr);
1469  bb->isFunctionCall() = retval;
1470  return retval;
1471 }

1814 // We have a number of choices for the algorithm:
1815 // (1) iterate over all functions, compute each function's extent and compare it to the interval. Probably not too fast since
1816 // most functions would probably be excluded (intervals are often small, even single addresses), and also not fast because
1817 // it computes the full function extent even when only part of it could cause the function to be selected.
1818 // (2) iterate over all functions, descending into basic blocks and data blocks. This is faster because we don't have to
1819 // compute a full function extent and we can short circuit once we find an overlap. But it requires looking up each basic
1820 // block by address, which will be O(n log(n)) in total.
1821 // (3) Use the fact that the interval is probably small and therefore the list of basic blocks and data blocks that overlap
1822 // with it is small, and that the list can be returned quite quickly from the AUM. For each instruction and data block
1823 // returned by the AUM, look at its function ownership list and merge it into the return value. This is the approach we
1824 // take here.

1826 Partitioner::functionsOverlapping(const AddressInterval &interval) const {
1827  std::vector<Function::Ptr> functions;

2397  // reproducible.
2398  Sawyer::Container::Map<size_t, ControlFlowGraph::ConstVertexIterator> neighbors;
2399  for (const ControlFlowGraph::Edge &edge: vertex.outEdges()) {
2400  if (edge.value().type() != E_FUNCTION_CALL &&
2401  edge.value().type() != E_FUNCTION_XFER &&
2402  edge.value().type() != E_FUNCTION_RETURN && !edge.isSelfEdge() &&
2403  !edge.target()->value().isEntryBlock()) {
2404  SAWYER_MESG(debug) <<" following edge " <<edgeName(edge) <<"\n";
2405  neighbors.insert(edge.target()->id(), edge.target());
2406  }
2407  }
2408  for (const ControlFlowGraph::Edge &edge: vertex.inEdges()) {
2409  if (edge.value().type() != E_FUNCTION_CALL &&
2410  edge.value().type() != E_FUNCTION_XFER &&
2411  edge.value().type() != E_FUNCTION_RETURN && !edge.isSelfEdge() &&
2412  !edge.source()->value().isEntryBlock()) {
2413  SAWYER_MESG(debug) <<" following edge " <<edgeName(edge) <<"\n";
2414  neighbors.insert(edge.target()->id(), edge.target());
2415  }
2416  }
2417

2419  for (const ControlFlowGraph::ConstVertexIterator &neighbor: neighbors.values()) {
2420  if (neighbor->value().type() != V_BASIC_BLOCK)
2421  continue; // functions don't own things like the indeterminate vertex
2422  if (ownership.exists(neighbor->id()))
2423  continue; // we already know about this neighbor
2424  ownership.insertMaybe(neighbor->id(), Function::OWN_PROVISIONAL);
2425  worklist.push(neighbor->id());
2426  }
2427  }

2432  function->insertBasicBlock(cfg_.findVertex(node.key())->value().address());
