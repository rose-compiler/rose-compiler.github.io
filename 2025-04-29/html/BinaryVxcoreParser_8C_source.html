38 VxcoreParser::parseUrl(const std::string &spec) {
39  std::vector<std::string> parts = StringUtility::split(":", spec, 3);
40  if (parts.size() != 3)
41  throw Exception("URL", 0, "not enough colons in vxcore URL (expected \"[MEM_ATTRS]:[FILE_ATTRS]:FILENAME\")");
42 
43  // Memory protection
44  settings_.protDefault = MemoryMap::READ_WRITE_EXECUTE;
45  settings_.protOverride = Sawyer::Nothing();
46  for (const char *s = parts[0].c_str(); *s; /*void*/) {
47  if ('=' == *s) {
48  unsigned perms = 0;
49  for (++s; *s && strchr("rwx", *s); ++s) {
50  switch (*s) {
51  case 'r': perms |= MemoryMap::READABLE; break;
52  case 'w': perms |= MemoryMap::WRITABLE; break;
53  case 'x': perms |= MemoryMap::EXECUTABLE; break;
54  }
55  }
56  settings_.protOverride = perms;
57  } else {
58  throw Exception("URL", 0, "invalid memory attribute for vxcore URL: \"" + StringUtility::cEscape(s) + "\"");
59  }
60  }
61 
62  // File attributes
63  settings_.version = 1;
64  if (!parts[1].empty()) {
65  boost::regex versionRe("version=(\\d+)");
66  boost::smatch matched;
67  if (boost::regex_match(parts[1], matched, versionRe)) {
68  settings_.version = boost::lexical_cast<unsigned>(matched[1]);
69  } else {
70  throw Exception("URL", 0, "invalid file attribute for vxcore URL: \"" + StringUtility::cEscape(parts[1]) + "\"");
71  }
72  }
73  if (settings_.version != 1 && settings_.version != 2)
74  throw Exception("URL", 0, "vxcore version " + boost::lexical_cast<std::string>(settings_.version) + " is not supported");
75 
76  // Error checking
77  if (2 == settings_.version && settings_.protOverride)
78  throw Exception("URL", 0, "vxcore version 2 does not support memory protection override (\"=\" attribute)");
79 
80  return parts[2];
81 }

101 void
102 VxcoreParser::parse(std::istream &input, const MemoryMap::Ptr &memory, const BaseSemantics::RegisterState::Ptr &registers,
103  const BaseSemantics::RiscOperators::Ptr &ops, const std::string &inputName) {
104  Sawyer::Message::Stream debug(mlog[DEBUG]);
105  if (1 == settings_.version) {
106  for (size_t segmentIdx = 0; input; ++segmentIdx) {
107  size_t headerOffset = input.tellg();
108  std::string header = rose_getline(input);
109  if (header.empty())
110  break; // EOF
111 
112  std::string name = inputName + " segment #" + boost::lexical_cast<std::string>(segmentIdx);
113  if (!parseMemory(header, input, memory, name, headerOffset) &&
114  !parseRegisters(header, input, registers, ops, name, headerOffset)) {
115  throw Exception(inputName, input.tellg(), "invalid header: \"" + StringUtility::cEscape(header.substr(0, 30)) + "\"" +
116  (header.size() > 30 ? "..." : ""));
117  }
118  }
119  } else if (2 == settings_.version) {
120  MemoryMap::Ptr tempMemory = memory ? MemoryMap::instance() : MemoryMap::Ptr();
121  while (true) {
122  // Read the message header
123  HeaderVersion2 header;
124  const size_t headerOffset = input.tellg();
125  input.read((char*)&header, sizeof header);
126  const size_t nHeader = input.gcount();
127  header.payloadSize = BitOps::fromBigEndian(header.payloadSize);
128  header.addr = BitOps::fromBigEndian(header.addr);
129 
130  if (0 == nHeader) {
131  break;
132  } else if (nHeader != sizeof header) {
133  throw Exception(inputName, headerOffset,
134  (boost::format("short read (expected %1%, got only %2%) at %3%")
135  % sizeof(header) % nHeader % headerOffset).str());
136  } else if (2 != header.version) {
137  throw Exception(inputName, headerOffset,
138  (boost::format("invalid message version (expected %1%, got %2%) at %3%")
139  % settings_.version % (unsigned)header.version % headerOffset).str());
140  } else if (header.unused0 || header.unused1) {
141  throw Exception(inputName, headerOffset, (boost::format("unused fields must be zero at %1%") % headerOffset).str());
142  } else if (header.mapFlags & ~MemoryMap::READ_WRITE_EXECUTE) {
143  throw Exception(inputName, headerOffset, (boost::format("invalid map flags at %1%") % headerOffset).str());
144  } else if (header.payloadSize > 0) {
145  std::vector<uint8_t> buf(header.payloadSize);
146  input.read((char*)buf.data(), header.payloadSize);
147  const size_t nPayload = input.gcount();
148  if (nPayload != header.payloadSize) {
149  throw Exception(inputName, headerOffset,
150  (boost::format("short payload read (expected %1%, got only %2%) at %3%)")
151  % header.payloadSize % nPayload % (headerOffset + sizeof header)).str());
152  } else if (tempMemory) {
153  const auto where = AddressInterval::baseSize(header.addr, header.payloadSize);
154  SAWYER_MESG(debug) <<"vxcore: addresses " <<StringUtility::addrToString(where) <<" at " <<headerOffset <<"\n";
155  tempMemory->insert(where,
156  MemoryMap::Segment::anonymousInstance(header.payloadSize, header.mapFlags, inputName));
157  const size_t nCopied = tempMemory->at(header.addr).limit(header.payloadSize).write(buf.data()).size();
158  ASSERT_always_require(nCopied == header.payloadSize);
159  }
160  }
161  }
162  if (tempMemory) {
163  tempMemory->combineAdjacentSegments();
164  ASSERT_not_null(memory);
165  memory->linkTo(tempMemory, tempMemory->hull());
166  }
167  } else {
168  ASSERT_not_implemented("vxcore version " + boost::lexical_cast<std::string>(settings_.version));
169  }
170 }

172 bool
173 VxcoreParser::parseMemory(const std::string &header, std::istream &input, const MemoryMap::Ptr &memory,
174  const std::string &inputName, size_t headerOffset) {
175  Sawyer::Message::Stream debug(mlog[DEBUG]);
176 
177  // Parse the header
178  // address size protection
179  boost::regex headerRe("\\s*([0-9a-fA-F]+)\\s+([0-9a-fA-F]+)(?:\\s+=([-RWX]*))\\s*:\\s*");
180  boost::smatch matched;
181  if (!boost::regex_match(header, matched, headerRe))
182  return false;
183 
184  std::string vaStr = matched[1];
185  Address va = rose_strtoull(vaStr.c_str(), NULL, 16);
186 
187  std::string nBytesStr = matched[2];
188  Address nBytes = rose_strtoull(nBytesStr.c_str(), NULL, 16);
189 
190  std::string memProtStr = matched[3];
191  unsigned memProt = settings_.protDefault;
192  if (!memProtStr.empty()) {
193  memProt = 0;
194  BOOST_FOREACH (char ch, memProtStr) {
195  switch (ch) {
196  case 'R': memProt |= MemoryMap::READABLE; break;
197  case 'W': memProt |= MemoryMap::WRITABLE; break;
198  case 'X': memProt |= MemoryMap::EXECUTABLE; break;
199  case '-': break;
200  }
201  }
202  }
203  settings_.protOverride.assignTo(memProt);
204 
205  // Read the binary data from the input stream
206  struct Resources {
207  uint8_t *bytes;
208  Resources(): bytes(NULL) {}
209  ~Resources() { delete[] bytes; }
210  } r;
211  r.bytes = new uint8_t[nBytes];
212  input.read((char*)r.bytes, nBytes);
213  size_t nRead = input.gcount();
214  if (nRead != nBytes) {
215  throw Exception(inputName, headerOffset,
216  (boost::format("short read (expected %1%, got only %2%) at %3%") % nBytes % nRead % headerOffset).str());
217  }
218 
219  // Place the binary data into the map
220  if (memory) {
221  AddressInterval where = AddressInterval::baseSize(va, nBytes);
222  SAWYER_MESG(debug) <<"vxcore: addresses " <<StringUtility::addrToString(where) <<" at " <<headerOffset <<"\n";
223  memory->insert(where, MemoryMap::Segment::anonymousInstance(nBytes, memProt, inputName));
224  size_t nCopied = memory->at(va).limit(nBytes).write(r.bytes).size();
225  ASSERT_always_require(nCopied == nBytes);
226  }
227 
228  return true;
229 }

232 VxcoreParser::parseRegisters(const std::string &header, std::istream &input, const BaseSemantics::RegisterState::Ptr &registers,
233  const BaseSemantics::RiscOperators::Ptr &ops, const std::string &inputName, size_t headerOffset) {
234  // Match the register section's header
235  // ISA
236  boost::regex headerRe("\\s*registers\\s+(\\w+)\\s*");
237  boost::smatch matched;
238  if (!boost::regex_match(header, matched, headerRe))
239  return false;
240  isaName_ = matched[1];
241 
242  // Parse all the lines in the register section. This section doesn't have a size, but rather ends at either the end of the file or
243  // after the line-feed of a line containing only the word "end".
244  boost::regex registerRe("\\s*(\\S+)\\s*:\\s*(0x[0-9a-fA-F]+)\\s*");
245  boost::regex endRe("\\s*end\\s*");
246  while (true) {
247  std::string line = rose_getline(input);
248  if (line.empty() || boost::regex_match(line, endRe)) {
249  return true;
250  } else if (!boost::regex_match(line, matched, registerRe)) {
251  throw Exception(inputName, headerOffset, "syntax error in register specification: "
252  "\"" + StringUtility::cEscape(line.substr(0, 30)) + "\"" +
253  (line.size() > 30 ? "..." : ""));
254  } else if (registers != NULL) {
255  ASSERT_always_not_null(ops);
256  // FIXME[Robb Matzke 2020-02-12]: check register name, parse value, and save value in register state.
257  mlog[ERROR] <<"saving of register values is not implemented: " <<line <<"\n";
258  }
259  }
260 
261  return true;
262 }

276 void
277 VxcoreParser::unparse(std::ostream &out, const MemoryMap::Ptr &memory, const AddressInterval &memoryLimit,
278  const BaseSemantics::RegisterState::Ptr &registers, const BaseSemantics::RiscOperators::Ptr &ops,
279  const std::string &outputName) {
280  if (memory && !memoryLimit.isEmpty()) {
281  Address va = memoryLimit.least();
282  const size_t maxPayload = 0xffffffff;
283  while (const AddressInterval selected = memory->atOrAfter(va).limit(maxPayload).singleSegment().available() & memoryLimit) {
284  MemoryMap::ConstNodeIterator inode = memory->at(selected.least()).nodes().begin();
285  ASSERT_forbid(inode == memory->nodes().end()); // because of the while loop's condition
286  ASSERT_require(inode->key().contains(selected));
287  const MemoryMap::Segment &segment = inode->value();
288 
289  // Header
290  if (1 == settings_.version) {
291  out <<StringUtility::addrToString(selected.least()).substr(2)
292  <<" " <<StringUtility::addrToString(selected.size()).substr(2)
293  <<" =" <<(0 != (segment.accessibility() & MemoryMap::READABLE) ? "R" : "-")
294  <<(0 != (segment.accessibility() & MemoryMap::WRITABLE) ? "W" : "-")
295  <<(0 != (segment.accessibility() & MemoryMap::EXECUTABLE) ? "X" : "-")
296  <<"\n";
297  } else if (2 == settings_.version) {
298  HeaderVersion2 header;
299  memset(&header, 0, sizeof header);
300  header.version = settings_.version;
301  header.mapFlags = segment.accessibility() & MemoryMap::READ_WRITE_EXECUTE;
302  header.payloadSize = BitOps::toBigEndian(boost::numeric_cast<uint32_t>(selected.size()));
303  header.addr = BitOps::toBigEndian(boost::numeric_cast<uint64_t>(selected.least()));
304  out.write((const char*)&header, sizeof header);
305  if (!out.good())
306  throw Exception(outputName, out.tellp(), "write failed");
307  } else {
308  ASSERT_not_implemented("vxcore version " + boost::lexical_cast<std::string>(settings_.version));
309  }
310 
311  // Data output one buffer-full at a time since the memory map' underlying buffer might not be storing the bytes
312  // contiguously, but we need contiguous bytes for std::ostream::write.
313  uint8_t buffer[4096];
314  AddressInterval remaining = selected;
315  while (!remaining.isEmpty()) {
316  // What to read
317  Address bufferVa = remaining.least();
318  size_t nBytes = std::min(remaining.size(), (Address)sizeof(buffer));
319  AddressInterval toRead = AddressInterval::baseSize(bufferVa, nBytes);
320 
321  // Read from map, write to std::ostream
322  size_t nRead = memory->at(toRead).read(buffer).size();
323  size_t offset = out.tellp();
324  ASSERT_always_require(nRead == nBytes); // cannot fail since the extent is from mmNode.key
325  out.write((const char*)buffer, nBytes);
326  if (!out.good())
327  throw Exception(outputName, offset, "write failed");
328 
329  // What remains to transfer
330  remaining = remaining.split(toRead.greatest()).second;
331  }
332 
333  if (selected.greatest() == memoryLimit.greatest())
334  break;
335  va = selected.greatest() + 1;
336  }
337  }
338 
339  if (registers) {
340  if (1 == settings_.version) {
341  ASSERT_not_null(ops);
342  out <<"registers " <<registers->registerDictionary()->name() <<"\n";
343  RegisterDictionary::RegisterDescriptors regs = registers->registerDictionary()->getLargestRegisters();
344  RegisterNames registerName(registers->registerDictionary());
345  BOOST_FOREACH (RegisterDescriptor reg, regs) {
346  BaseSemantics::SValue::Ptr val = registers->peekRegister(reg, ops->undefined_(reg.nBits()), ops.get());
347  if (auto number = val->toUnsigned())
348  out <<(boost::format("%s 0x%x\n") % registerName(reg) % *number);
349  }
350  out <<"end\n";
351  } else if (2 == settings_.version) {
352  // Registers are not stored for version 2
353  } else {
354  ASSERT_not_implemented("vxcore version " + boost::lexical_cast<std::string>(settings_.version));
355  }
356  }
357 }
