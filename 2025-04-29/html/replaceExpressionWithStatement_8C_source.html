313 // Rewrites a while or for loop so that the official test is changed to
314 // "true" and what had previously been the test is now an if-break
315 // combination (with an inverted condition) at the beginning of the loop
316 // body
317 //
318 // For example, "while (a < 5) ++a;" becomes:
319 // "while (true) {bool temp = (a < 5); if (!temp) break; ++a;}"
320 // "for (init; test; step) body;" becomes:
321 // "for (init; true; step) {bool temp = test; if (!temp) break; body;}"
322 // "do body; while (test);" becomes:
323 // "do {body; bool temp = test; if (!temp) break;} while (true);"
324 void pushTestIntoBody(SgScopeStatement* loopStmt) {

331  new_body->set_parent(loopStmt);
332  AstPostProcessing(loopStmt);
333  SgStatement* cond = SageInterface::getLoopCondition(loopStmt);
334  ROSE_ASSERT (isSgExprStatement(cond));
335  SgExpression* root = isSgExprStatement(cond)->get_expression();
336  SgCastExp* cast = buildCastExp(root, SageInterface::getBoolType(loopStmt));
337  // Name does not need to be unique, but must not be used in user code anywhere
338  AstPostProcessing(loopStmt);
339  SgVariableDeclaration* new_decl =
340  buildVariableDeclaration("rose__temp",
341  SageInterface::getBoolType(loopStmt),
342  buildAssignInitializer(cast),

345  SageInterface::appendStatement(new_decl, new_body);
346  AstPostProcessing(loopStmt);
