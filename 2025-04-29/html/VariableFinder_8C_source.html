161  // Search the `haystack` to find a subexpression structurally equivalent to `needle` and return true iff found.
162  static bool findInExpr(const SymbolicExpression::Ptr &needle, const SymbolicExpression::Ptr &haystack) {
163  struct Finder: SymbolicExpression::Visitor {
164  std::set<SymbolicExpression::Hash> seen; // expressions in which we've already looked
165  SymbolicExpression::Ptr needle; // the thing we're looking for, by symbolic equivalence
166  SymbolicExpression::Ptr found; // the (first) place where we found something equivalent to the needle
167 
168  explicit Finder(const SymbolicExpression::Ptr &needle)
169  : needle(needle) {}
170 
171  SymbolicExpression::VisitAction preVisit(const SymbolicExpression::Ptr &node) override {
172  if (!seen.insert(node->hash()).second) {
173  return SymbolicExpression::TERMINATE; // already searched here
174  } else if (node->isEquivalentTo(needle)) {
175  found = node;
176  return SymbolicExpression::TERMINATE; // found, so no need to search further
177  } else {
178  return SymbolicExpression::CONTINUE; // not found, so keep searching
179  }
180  }
181 
182  SymbolicExpression::VisitAction postVisit(const SymbolicExpression::Ptr&) override {
183  return SymbolicExpression::CONTINUE;
184  }
185  } finder(needle);
186  haystack->depthFirstTraversal(finder);
187  return finder.found != nullptr;
188  }

199  // Called when reading or writing to the memory state.
200  void lookForVariable(const BS::SValue::Ptr &addrSVal, const Access access) {
201  SgAsmInstruction *insn = currentInstruction();
202  if (!insn)
203  return;
204 
205  const SymbolicExpression::Ptr absAddr = SValue::promote(addrSVal)->get_expression();
206  SAWYER_MESG(mlog[DEBUG]) <<" memory address: " <<*absAddr <<"\n";
207 
208  // Compute a relative address by subtracting the initial stack pointer from when this RiscOperators object was created.
209  // Equivalently, we can replace all occurrances of the original stack pointer with zero, which might be faster an more
210  // easily simplified by the symbolic expression layer.
211  if (findInExpr(expr(initialSp_), absAddr)) {
212  const SymbolicExpression::Ptr zero = SymbolicExpression::makeIntegerConstant(initialSp_->nBits(), 0);
213  const SymbolicExpression::Ptr relAddr = absAddr->substitute(expr(initialSp_), zero);
214  SAWYER_MESG(mlog[DEBUG]) <<" address w.r.t. initial stack pointer: " <<*relAddr <<"\n";
215  if (const auto relAddrConcrete = relAddr->toSigned()) {
216  StackVariable::insertBoundary(boundaries_, *relAddrConcrete, InstructionAccess(insn->get_address(), access));
217  SAWYER_MESG(mlog[DEBUG]) <<" concrete offset: " <<*relAddrConcrete <<"\n";
218  }
219  }

254  struct: SymbolicExpression::Visitor {
255  std::set<Address> constants;
256  std::vector<SymbolicExpression::Ptr> path;
257 
258  SymbolicExpression::VisitAction preVisit(const SymbolicExpression::Ptr &node) {
259  const SymbolicExpression::Interior *parent = nullptr;
260  if (!path.empty()) {
261  parent = path.back()->isInteriorNodeRaw();
262  ASSERT_not_null(parent);
263  }
264  path.push_back(node);
265 
266  // Some constants are never addresses
267  if (parent && parent->getOperator() == SymbolicExpression::OP_EXTRACT &&
268  (parent->child(0) == node || parent->child(1) == node))
269  return SymbolicExpression::CONTINUE; // first two args of extract are never addresses
270 
271  // If we found a constant, perhaps treat it like an address
272  if (Sawyer::Optional<uint64_t> n = node->toUnsigned())
273  constants.insert(*n);
274 
275  return SymbolicExpression::CONTINUE;
276  }

279  ASSERT_forbid(path.empty());

282  }
283  } visitor;
284

363 StackFrame
364 VariableFinder::detectFrameAttributes(const P2::Partitioner::ConstPtr &partitioner, const P2::Function::Ptr &function,
365  StackVariable::Boundaries &boundaries /*in,out*/) {
366  ASSERT_not_null(partitioner);
367  ASSERT_not_null(function);
368  Sawyer::Message::Stream debug(mlog[DEBUG]);
369  SAWYER_MESG(debug) <<"detect frame attributes for " <<function->printableName() <<"\n";
370 
371  const StackDelta::Analysis &stackDeltas = function->stackDeltaAnalysis();
372  SAWYER_MESG(debug) <<" consistent frame pointer? " <<(stackDeltas.hasConsistentFramePointer() ? "yes" : "no") <<"\n";
373 
374  SgAsmInstruction *firstInsn = partitioner->instructionProvider()[function->address()];
375  const size_t bytesPerWord = partitioner->architecture()->bytesPerWord();
376  StackFrame frame;
377 
378  if (isSgAsmX86Instruction(firstInsn)) {
379  //-------------+-------------+---------------------------+---------------------------------------------------
380  // Stack ptr | Frame ptr | Stack contents | Notes
381  // word offset | word offset | |
382  //-------------+-------------+---------------------------+---------------------------------------------------
383  //
384  // : :
385  // : (part of parent frame) :
386  // : :
387  // +1 +1 | callee's actual arguments | Variable size, 1st arg at lowest address
388  // +---(current frame)---------+
389  // 0 0 | return address | 1 word
390  // -1 -1 | addr of parent frame | 1 word
391  // -2 -2 | callee saved registers | Optional, variable size, multiple of word size
392  // : :
393  // : :
394  //
395  //-------------+-------------+---------------------------+---------------------------------------------------
396 
397  frame.growthDirection = StackFrame::GROWS_DOWN;
398  frame.maxOffset = bytesPerWord-1; // top of frame
399  frame.minOffset = Sawyer::Nothing(); // bottom of frame is unknown
400  frame.size = Sawyer::Nothing(); // frame size is unknown
401  frame.rule = "x86: general";
402 
403  StackVariable::Boundary &returnPtr = StackVariable::insertBoundaryImplied(boundaries, 0);
404  returnPtr.purpose = StackVariable::Purpose::RETURN_ADDRESS;
405 
406  StackVariable::Boundary &parentPtr = StackVariable::insertBoundaryImplied(boundaries, -bytesPerWord);
407  parentPtr.purpose = StackVariable::Purpose::FRAME_POINTER;
408 
409  } else if (isSgAsmPowerpcInstruction(firstInsn)) {
410  //-------------+-------------+---------------------------+---------------------------------------------------
411  // Stack ptr | Frame ptr | Stack contents | Notes
412  // word offset | word offset | |
413  //-------------+-------------+---------------------------+---------------------------------------------------
414  // From documentation for PPC calling conventions:
415  // : :
416  // : (part of parent frame) :
417  // : :
418  // +1 ? | LR saved | 1 word
419  // 0 ? | addr of grandparent frame | 1 word Caller's frame pointer points here
420  // +---(current frame)---------+
421  // ? ? | saved FP register area | optional, variable size
422  // ? ? | saved GP register area | optional, multiple of 4/8 bytes
423  // ? ? | CR saved | 0 or 1 word
424  // ? ? | Local variables | optional, variable size
425  // ? ? | Function parameter area | optional, variable size for callee args not fitting in registers
426  // ? +2 | Padding | 0 to 7 bytes, although I'm not sure when this is used
427  // ? +1 | LR saved by callees | 1 word
428  // ? 0 | addr of parent frame | 1 word callees's frame pointer points here
429  // +---------------------------+
430  //
431  // From experience with GCC PowerPC-32 big-endian:
432  // +---(current frame)---------+
433  // -1 ? | saved link register |
434  // ? ? : :
435  // ? ? : :
436  // ? 0 | saved stack pointer |
437  // +---------------------------+
438  //
439  //-------------+-------------+---------------------------+---------------------------------------------------
440  //
441  frame.growthDirection = StackFrame::GROWS_DOWN;
442 
443  // If this function starts with "stwu r1, u32 [r1 - N]" then the frame size is N.
444  if (isSgAsmPowerpcInstruction(firstInsn)->get_kind() == powerpc_stwu) {
445  static RegisterDescriptor REG_R1;
446  if (REG_R1.isEmpty())
447  REG_R1 = partitioner->architecture()->registerDictionary()->findOrThrow("r1");
448  SgAsmDirectRegisterExpression *firstRegister = isSgAsmDirectRegisterExpression(firstInsn->operand(0));
449  SgAsmMemoryReferenceExpression *secondArg = isSgAsmMemoryReferenceExpression(firstInsn->operand(1));
450  SgAsmBinaryAdd *memAddr = secondArg ? isSgAsmBinaryAdd(secondArg->get_address()) : NULL;
451  SgAsmDirectRegisterExpression *secondRegister = memAddr ? isSgAsmDirectRegisterExpression(memAddr->get_lhs()) : NULL;
452  SgAsmIntegerValueExpression *constant = memAddr ? isSgAsmIntegerValueExpression(memAddr->get_rhs()) : NULL;
453  int64_t n = constant ? constant->get_signedValue() : int64_t(0);
454  if (firstRegister && secondRegister && constant && n < 0 &&
455  firstRegister->get_descriptor() == REG_R1 && secondRegister->get_descriptor() == REG_R1) {
456  frame.size = -n;
457  frame.maxOffset = -1;
458  frame.minOffset = -*frame.size;
459  frame.rule = "ppc: stwu r1, u32 [r1 - N]";
460  }
461 
462  } else {
463  frame.size = 8;
464  frame.maxOffset = -1;
465  frame.minOffset = -8;
466  frame.rule = "ppc: general";
467  }
468 
469  if (frame.minOffset) {
470  // Saved link register is at the top of the frame
471  StackVariable::Boundary &parentPtr = StackVariable::insertBoundaryImplied(boundaries, -bytesPerWord);
472  parentPtr.purpose = StackVariable::Purpose::FRAME_POINTER;
473 
474  // Saved stack register is at the bottom of the frame
475  StackVariable::Boundary &returnPtr = StackVariable::insertBoundaryImplied(boundaries, *frame.minOffset);
476  returnPtr.purpose = StackVariable::Purpose::STACK_POINTER;
477  }
478  } else if (auto m68k = isSgAsmM68kInstruction(firstInsn)) {
479  //-------------+-------------+---------------------------+---------------------------------------------------
480  // Stack ptr | Frame ptr | Stack contents | Notes
481  // word offset | word offset | |
482  //-------------+-------------+---------------------------+---------------------------------------------------
483  //
484  // : :
485  // : (part of parent frame) :
486  // : :
487  // +---(current frame)---------+
488  // 0 +1 | return address | 4 bytes
489  // -1 0 | addr of parent frame | 4 bytes
490  // : :
491  //
492  //-------------+-------------+---------------------------+---------------------------------------------------
493  frame.growthDirection = StackFrame::GROWS_DOWN;
494 
495  // If this m68k function starts with "link.w fp, -N" then the frame size is 4 + 4 + N since the caller has pushed the
496  // return address and this function pushes the old frame pointer register r6 and then reserves N additional bytes in
497  // the frame.
498  const RegisterDescriptor REG_FP = partitioner->architecture()->registerDictionary()->stackFrameRegister();
499  if (REG_FP &&
500  m68k->get_kind() == m68k_link &&
501  m68k->nOperands() == 2 &&
502  isSgAsmDirectRegisterExpression(m68k->operand(0)) &&
503  isSgAsmDirectRegisterExpression(m68k->operand(0))->get_descriptor() == REG_FP &&
504  isSgAsmIntegerValueExpression(m68k->operand(1))) {
505  int64_t n = isSgAsmIntegerValueExpression(m68k->operand(1))->get_signedValue();
506  ASSERT_require(n <= 0);
507  frame.size = 4 /*pushed frame pointer*/ + (-n);
508  frame.maxOffset = 3;
509  frame.minOffset = n - 4;
510  frame.rule = "m68k: link a6, N";
511  } else {
512  frame.maxOffset = 3;
513  frame.rule = "m68k: general";
514  }
515 
516  StackVariable::Boundary &parentPtr = StackVariable::insertBoundaryImplied(boundaries, -bytesPerWord);
517  parentPtr.purpose = StackVariable::Purpose::FRAME_POINTER;
518 
519  StackVariable::Boundary &returnPtr = StackVariable::insertBoundaryImplied(boundaries, 0);
520  returnPtr.purpose = StackVariable::Purpose::RETURN_ADDRESS;
521 
522  if (frame.minOffset && *frame.minOffset < 0) {
523  StackVariable::Boundary &bottom = StackVariable::insertBoundaryImplied(boundaries, *frame.minOffset);
524  bottom.purpose = StackVariable::Purpose::UNKNOWN;
525  }
526 
527 #ifdef ROSE_ENABLE_ASM_AARCH32
528  } else if (isSgAsmAarch32Instruction(firstInsn)) {
529  //-------------+-------------+---------------------------+---------------------------------------------------
530  // Stack ptr | Frame ptr | Stack contents | Notes
531  // word offset | word offset | |
532  //-------------+-------------+---------------------------+---------------------------------------------------
533  //
534  // : :
535  // : (part of parent frame) :
536  // 0 : :
537  // +---(current frame)---------+
538  // -1 0 | addr of parent frame | 4 bytes
539  // -2 -1 | saved link register | optional, 4 bytes
540  // -3 -2 | local variables | variable size
541  // +---------------------------+
542  frame.growthDirection = StackFrame::GROWS_DOWN;
543 
544  const RegisterDescriptor REG_FP = partitioner->architecture()->registerDictionary()->findOrThrow("fp");
545  const RegisterDescriptor REG_SP = partitioner->architecture()->registerDictionary()->findOrThrow("sp");
546  const RegisterDescriptor REG_LR = partitioner->architecture()->registerDictionary()->findOrThrow("lr");
547 
548  // If the first three instructions are:
549  // push fp, lr
550  // add fp, sp, 4
551  // sub sp, N
552  // then the frame size is N + 8 bytes
553  SgAsmAarch32Instruction* insns[3] = {nullptr, nullptr, nullptr};
554  insns[0] = isSgAsmAarch32Instruction(firstInsn);
555  insns[1] = isSgAsmAarch32Instruction(partitioner->instructionProvider()[function->address() + 4]);
556  insns[2] = isSgAsmAarch32Instruction(partitioner->instructionProvider()[function->address() + 8]);
557  if (// push fp, lr
558  insns[0] && insns[0]->get_kind() == Aarch32InstructionKind::ARM_INS_PUSH &&
559  insns[0]->nOperands() == 2 &&
560  isSgAsmDirectRegisterExpression(insns[0]->operand(0)) &&
561  isSgAsmDirectRegisterExpression(insns[0]->operand(0))->get_descriptor() == REG_FP &&
562  isSgAsmDirectRegisterExpression(insns[0]->operand(1)) &&
563  isSgAsmDirectRegisterExpression(insns[0]->operand(1))->get_descriptor() == REG_LR &&
564  // add fp, sp, 4
565  insns[1] && insns[1]->get_kind() == Aarch32InstructionKind::ARM_INS_ADD &&
566  insns[1]->nOperands() == 3 &&
567  isSgAsmDirectRegisterExpression(insns[1]->operand(0)) &&
568  isSgAsmDirectRegisterExpression(insns[1]->operand(0))->get_descriptor() == REG_FP &&
569  isSgAsmDirectRegisterExpression(insns[1]->operand(1)) &&
570  isSgAsmDirectRegisterExpression(insns[1]->operand(1))->get_descriptor() == REG_SP &&
571  isSgAsmIntegerValueExpression(insns[1]->operand(2)) &&
572  isSgAsmIntegerValueExpression(insns[1]->operand(2))->get_absoluteValue() == 4 &&
573  // sub sp, sp, N
574  insns[2] && insns[2]->get_kind() == Aarch32InstructionKind::ARM_INS_SUB &&
575  insns[2]->nOperands() == 3 &&
576  isSgAsmDirectRegisterExpression(insns[2]->operand(0)) &&
577  isSgAsmDirectRegisterExpression(insns[2]->operand(0))->get_descriptor() == REG_SP &&
578  isSgAsmDirectRegisterExpression(insns[2]->operand(1)) &&
579  isSgAsmDirectRegisterExpression(insns[2]->operand(1))->get_descriptor() == REG_SP &&
580  isSgAsmIntegerValueExpression(insns[2]->operand(2))) {
581  // The frame pointer will point to one past the top (highest address) of the frame
582  int64_t n = boost::numeric_cast<int64_t>(isSgAsmIntegerValueExpression(insns[2]->operand(2))->get_absoluteValue());
583  frame.size = n + 8;
584  frame.maxOffset = -1;
585  frame.minOffset = -(n + 4); // entire frame except the lr_0 which appears on the stack at the fp
586  // Note that the string "<saved-lr>" is important and used by initializeFrameBoundaries
587  frame.rule = "aarch32: push fp, lr <saved-lr>; add fp, sp, 4; sub sp N";
588 
589  } else if (// str fp, u32 [sp (after sp -= 4)]
590  insns[0] &&
591  insns[0]->get_kind() == Aarch32InstructionKind::ARM_INS_STR &&
592  insns[0]->nOperands() == 2 &&
593  isSgAsmDirectRegisterExpression(insns[0]->operand(0)) &&
594  isSgAsmDirectRegisterExpression(insns[0]->operand(0))->get_descriptor() == REG_FP &&
595  isSgAsmMemoryReferenceExpression(insns[0]->operand(1)) &&
596  isSgAsmBinaryPreupdate(isSgAsmMemoryReferenceExpression(insns[0]->operand(1))->get_address()) &&
597  isSgAsmDirectRegisterExpression(isSgAsmBinaryPreupdate(isSgAsmMemoryReferenceExpression(insns[0]->operand(1))->get_address())->get_lhs()) &&
598  isSgAsmDirectRegisterExpression(isSgAsmBinaryPreupdate(isSgAsmMemoryReferenceExpression(insns[0]->operand(1))->get_address())->get_lhs())->get_descriptor() == REG_SP &&
599  // add fp, sp, 0
600  insns[1] &&
601  insns[1]->get_kind() == Aarch32InstructionKind::ARM_INS_ADD &&
602  insns[1]->nOperands() == 3 &&
603  isSgAsmDirectRegisterExpression(insns[1]->operand(0)) &&
604  isSgAsmDirectRegisterExpression(insns[1]->operand(0))->get_descriptor() == REG_FP &&
605  isSgAsmDirectRegisterExpression(insns[1]->operand(1)) &&
606  isSgAsmDirectRegisterExpression(insns[1]->operand(1))->get_descriptor() == REG_SP &&
607  isSgAsmIntegerValueExpression(insns[1]->operand(2)) &&
608  isSgAsmIntegerValueExpression(insns[1]->operand(2))->get_absoluteValue() == 0 &&
609  // sub sp, sp, N
610  insns[2] &&
611  insns[2]->get_kind() == Aarch32InstructionKind::ARM_INS_SUB &&
612  insns[2]->nOperands() == 3 &&
613  isSgAsmDirectRegisterExpression(insns[2]->operand(0)) &&
614  isSgAsmDirectRegisterExpression(insns[2]->operand(0))->get_descriptor() == REG_SP &&
615  isSgAsmDirectRegisterExpression(insns[2]->operand(1)) &&
616  isSgAsmDirectRegisterExpression(insns[2]->operand(1))->get_descriptor() == REG_SP &&
617  isSgAsmIntegerValueExpression(insns[2]->operand(2))) {
618  int64_t n = boost::numeric_cast<int64_t>(isSgAsmIntegerValueExpression(insns[2]->operand(2))->get_absoluteValue());
619  frame.size = n + 4;
620  frame.maxOffset = -1;
621  frame.minOffset = -(n + 4);
622  frame.rule = "aarch32: str fp, u32 [sp (after sp -= 4)]; add fp, sp, 0; sub sp, sp N";
623 
624  } else {
625  frame.size = 8;
626  frame.maxOffset = -1;
627  frame.minOffset = -8;
628  frame.rule = "aarch32: general";
629  }
630 
631  if (boost::contains(frame.rule, "<saved-lr>")) {
632  // The "push fp, lr" pushes 8 bytes calculated from the stack pointer, so we don't want this eight
633  // bytes to cross a frame variable boundary.
634  StackVariable::Boundary &returnPtr = StackVariable::insertBoundaryImplied(boundaries, -4);
635  returnPtr.purpose = StackVariable::Purpose::RETURN_ADDRESS;
636  } else {
637  StackVariable::Boundary &parentPtr = StackVariable::insertBoundaryImplied(boundaries, 0);
638  parentPtr.purpose = StackVariable::Purpose::FRAME_POINTER;
639  }
640 #endif
641  }
642 
643  if (debug) {
644  debug <<" frame initialization rule: architecture \"" <<StringUtility::cEscape(partitioner->architecture()->name()) <<"\""
645  <<", rule ";
646  if (!frame.rule.empty()) {
647  debug <<"\"" <<StringUtility::cEscape(frame.rule) <<"\"\n";
648  } else {
649  debug <<"unspecified\n";
650  }
651 
652  switch (frame.growthDirection) {
653  case StackFrame::GROWS_UP:
654  debug <<" stack direction: grows up\n";
655  break;
656  case StackFrame::GROWS_DOWN:
657  debug <<" stack direction: grows down\n";
658  break;
659  }
660 
661  debug <<" frame boundaries w.r.t initial function SP:\n";
662  debug <<" maximum: ";
663  if (frame.maxOffset) {
664  debug <<*frame.maxOffset <<"\n";
665  } else {
666  debug <<"unknown\n";
667  }
668  debug <<" mininum: ";
669  if (frame.minOffset) {
670  debug <<*frame.minOffset <<"\n";
671  } else {
672  debug <<"unknown\n";
673  }
674 
675  debug <<" frame size: ";
676  if (frame.size) {
677  debug <<StringUtility::plural(*frame.size, "bytes") <<"\n";
678  } else {
679  debug <<"unknown\n";
680  }
681 
682  debug <<" boundaries:\n";
683  for (const StackVariable::Boundary &boundary: boundaries)
684  StackVariable::printBoundary(debug, boundary, " ");
685  }

690 void
691 VariableFinder::removeOutliers(const StackFrame &frame, StackVariable::Boundaries &boundaries /*in,out,sorted*/) {
692 #ifndef NDEBUG
693  for (size_t i = 1; i < boundaries.size(); ++i)
694  ASSERT_require(boundaries[i-1].stackOffset < boundaries[i].stackOffset);
695 #endif
696 
697  if (frame.minOffset) {
698  // Remove all boundaries that are below the bottom of the frame, but don't yet remove the greatest one that's below the
699  // frame.
700  size_t i = 0;
701  while (i+1 < boundaries.size() && boundaries[i+1].stackOffset < *frame.minOffset)
702  ++i;
703  boundaries.erase(boundaries.begin(), boundaries.begin() + i);
704 
705  // Adjust the lowest bounded area so it starts at the start of the frame unless there's already a boundary there (in which
706  // case, remove it instead of moving it).
707  if (!boundaries.empty() && boundaries[0].stackOffset < *frame.minOffset) {
708  if (boundaries.size() == 1 || boundaries[1].stackOffset > *frame.minOffset) {
709  boundaries[0].stackOffset = *frame.minOffset;
710  } else {
711  boundaries.erase(boundaries.begin());
712  }
713  }
714  }
715 
716  if (frame.maxOffset) {
717  // Remove all boundaries that start above the top of the frame.
718  while (!boundaries.empty() && boundaries.back().stackOffset > *frame.maxOffset)
719  boundaries.pop_back();
720  }
721 }

723 StackVariables
724 VariableFinder::findStackVariables(const P2::Partitioner::ConstPtr &partitioner, const P2::Function::Ptr &function) {
725  ASSERT_not_null(partitioner);
726  ASSERT_not_null(function);
727 
728  // Return cached local variable information if available.
729  if (function->attributeExists(ATTR_LOCAL_VARS))
730  return function->getAttribute<StackVariables>(ATTR_LOCAL_VARS);
731 
732  Sawyer::Message::Stream debug(mlog[DEBUG]);
733  StackVariable::Boundaries boundaries;
734  const StackFrame frame = detectFrameAttributes(partitioner, function, boundaries /*in,out*/);
735 
736  // Look for stack offsets semantically. We could do a complete data-flow analysis, but it turns out that processing each basic
737  // block independently is usually just as good, and much faster.
738  SAWYER_MESG(debug) <<"searching for local variables semantically in " <<function->printableName() <<"\n";
739  const StackDelta::Analysis &stackDeltas = function->stackDeltaAnalysis();
740  if (!stackDeltas.hasResults()) {
741  SAWYER_MESG(mlog[WARN]) <<"no stack delta information; variable synthesis skipped for " <<function->printableName() <<"\n";
742  return {};
743  }
744  for (Address bblockVa: function->basicBlockAddresses()) {
745  P2::BasicBlock::Ptr bb = partitioner->basicBlockExists(bblockVa);
746  ASSERT_not_null(bb);
747  auto ops = LVarSemantics::RiscOperators::instance(partitioner, function->stackDeltaAnalysis(), boundaries /*in,out*/);
748  BS::Dispatcher::Ptr cpu = partitioner->newDispatcher(ops);
749  for (SgAsmInstruction *insn: bb->instructions()) {
750  try {
751  cpu->processInstruction(insn);
752  } catch (const BS::Exception &e) {
753  if (!partitioner->architecture()->isUnknown(insn))
754  debug <<" semantic failure for " <<insn->toString() <<": " <<e.what() <<"\n";
755  break;
756  } catch (...) {
757  if (!partitioner->architecture()->isUnknown(insn))
758  debug <<" semantic failure for " <<insn->toString() <<"\n";
759  }
760  }
761  }
762 
763  // Sort and prune the boundaries so we have just those that are in the frame.
764  std::sort(boundaries.begin(), boundaries.end(), [](const StackVariable::Boundary &a, const StackVariable::Boundary &b) {
765  return a.stackOffset < b.stackOffset;
766  });
767  if (debug) {
768  debug <<" sorted boundary list:\n";
769  for (const StackVariable::Boundary &boundary: boundaries)
770  StackVariable::printBoundary(debug, boundary, " ");
771  }
772  removeOutliers(frame, boundaries);
773  if (debug) {
774  debug <<" boundary list after pruning:\n";
775  for (const StackVariable::Boundary &boundary: boundaries)
776  StackVariable::printBoundary(debug, boundary, " ");
777  }
778 
779  // Now that we know the boundaries between parts of the frame, create the variables that live between those boundaries.
780  StackVariables lvars;
781  for (size_t i = 0; i < boundaries.size(); ++i) {
782  const StackVariable::Boundary &boundary = boundaries[i];
783 
784  // We know the low offset of this variable, but what is the high offset.
785  int64_t maxOffset = boundary.stackOffset;
786  if (i + 1 < boundaries.size()) {
787  maxOffset = boundaries[i+1].stackOffset - 1;
788  } else if (frame.maxOffset) {
789  maxOffset = *frame.maxOffset;
790  } else {
791  maxOffset = (int64_t)BitOps::lowMask<uint64_t>(partitioner->architecture()->bitsPerWord() - 1);
792  }
793 
794  // Create the variable
795  ASSERT_require2(maxOffset >= boundary.stackOffset,
796  "maxOffset=" + boost::lexical_cast<std::string>(maxOffset) +
797  ", boundary.stackOffset=" + boost::lexical_cast<std::string>(boundary.stackOffset));
798  const OffsetInterval where = OffsetInterval::hull(boundary.stackOffset, maxOffset);
799  Address varMaxSize = (uint64_t)maxOffset + 1u - (uint64_t)boundary.stackOffset;
800  StackVariable lvar(function, boundary.stackOffset, varMaxSize, boundary.purpose, boundary.definingInsns);
801  lvar.setDefaultName();
802  lvars.insert(where, lvar);
803  }
804 
805  if (debug) {
806  debug <<" local variables:\n";
807  print(lvars, partitioner, debug, " ");
808  }
809  function->setAttribute(ATTR_LOCAL_VARS, lvars);
810  return lvars;
811 }

836 // For x86 ELF executables, global variables live in a particular section, such as .bss. The compiler typically generates code that
837 // computes the global variable address as an offset from the function's own address. It does this by calling __x86.get_pc_thunk.ax
838 // (or similar) which returns the address of the instruction following the call, and then adding a constant to the returned
839 // value. This computes the starting address of the section (base address), to which additional offsets are added to get to the
840 // various global variables. Since the base address is computed in one basic block and used in subsequent basic blocks, we use a
841 // data flow analysis to propagate the base address to the other blocks.
842 struct FindGlobalVariableVasMethod1Worker {
843  P2::Partitioner::ConstPtr partitioner;
844  Sawyer::ProgressBar<size_t> &progress;
845  VariableFinder &self;
846 
847  // The following data members are synchronized with this mutex
848  SAWYER_THREAD_TRAITS::Mutex mutex_;
849  AddressToAddresses result;
850 
851  explicit FindGlobalVariableVasMethod1Worker(const P2::Partitioner::ConstPtr &partitioner, Sawyer::ProgressBar<size_t> &progress,
852  VariableFinder &self)
853  : partitioner(partitioner), progress(progress), self(self) {
854  ASSERT_not_null(partitioner);
855  }
856 
857  void operator()(size_t /*workId*/, const P2::Function::Ptr &function) {
858  ++progress;
859  using Df = P2::DataFlow::Engine;
860  using Cfg = Df::Cfg;
861 
862  Sawyer::Message::Stream debug(mlog[DEBUG]);
863  SAWYER_MESG(debug) <<" analyzing " <<function->printableName() <<"\n";
864 
865  // Create the data-flow CFG
866  P2::ControlFlowGraph::ConstVertexIterator cfgRoot = partitioner->findPlaceholder(function->address());
867  ASSERT_require(cfgRoot != partitioner->cfg().vertices().end());
868  FollowNamedEdges predicate(".+\\.get_pc_thunk\\..+");
869  Cfg cfg = P2::DataFlow::buildDfCfg(partitioner, partitioner->cfg(), cfgRoot, predicate);
870 
871  // Create the data-flow transfer and merge functors
872  BS::Dispatcher::Ptr cpu = partitioner->newDispatcher(partitioner->newOperators());
873  Df::TransferFunction xfer(cpu);
874  xfer.ignoringSemanticFailures(true);
875  Df::MergeFunction merger(cpu);
876 
877  // Create the initial state, and cause it to use a simple merge function that doesn't account for aliasing
878  BS::State::Ptr initialState = xfer.initialState();
879  initialState->memoryState()->merger(BS::Merger::instance());
880  initialState->memoryState()->merger()->memoryAddressesMayAlias(false);
881 
882  // Run the dataflow with a low number of iterations
883  Df df(cfg, xfer, merger);
884  df.insertStartingVertex(0, initialState);
885 
886  Sawyer::Stopwatch timer;
887  const size_t maxSteps = cfg.nVertices() * 3 + 1; // arbitrary, but positive
888  for (size_t i = 0; i < maxSteps; ++i) {
889  if (!df.runOneIteration())
890  break;
891  if (timer.report() > self.settings().gvarMethod1MaxTimePerFunction.count()) {
892  mlog[WARN] <<"max time exceeded for gvar method 1 for " <<function->printableName()
893  <<(boost::format(" (%d of %d steps; %1.0f%%)") % i % maxSteps % (100.0 * i / maxSteps)) <<"\n";
894  break;
895  }
896  }
897 
898  // Find all constants that appear in memory address expressions and organize them according to what instruction(s) wrote to
899  // that address. We could organize by memory reads also, but alas, the memory state does not track readers--only writers.
900  using Map = Sawyer::Container::IntervalSetMap<AddressInterval, Sawyer::Container::Set<Address>>;
901  struct Accumulator: BS::MemoryCell::Visitor {
902  VariableFinder &variableFinder;
903  Sawyer::Message::Stream &debug;
904  Map map;
905  Accumulator(VariableFinder &variableFinder, Sawyer::Message::Stream &debug)
906  : variableFinder(variableFinder), debug(debug) {}
907  void operator()(BS::MemoryCell::Ptr &cell) override {
908  SymbolicExpression::Ptr address = IS::SymbolicSemantics::SValue::promote(cell->address())->get_expression();
909  if (debug) {
910  debug <<" memory address: " <<*address <<"\n"
911  <<" writers = {";
912  for (Address va: cell->getWriters().values())
913  debug <<" " <<StringUtility::addrToString(va);
914  debug <<" }\n";
915  }
916  std::set<Address> constants = variableFinder.findConstants(address);
917  for (Address c: constants) {
918  SAWYER_MESG(debug) <<" constant = " <<StringUtility::addrToString(c) <<"\n";
919  map.insert(c, cell->getWriters());
920  }
921  }
922  };
923 
924  Accumulator accumulator(self, debug);
925  for (size_t i = 0; i < cfg.nVertices(); ++i) {
926  SAWYER_MESG(debug) <<" final state for vertex " <<i <<": " <<cfg.findVertex(i)->value().toString() <<"\n";
927  if (auto state = df.getFinalState(i)) {
928  SAWYER_MESG(debug) <<(*state+" ") <<"\n";
929  auto mem = BS::MemoryCellState::promote(state->memoryState());
930  mem->traverse(accumulator);
931  }
932  }
933 
934  // Compute the result for this single function
935  AddressToAddresses functionResult;
936  SAWYER_MESG(debug) <<" summary for " <<function->printableName() <<":\n";
937  for (const auto &node: accumulator.map.nodes()) {
938  if (debug) {
939  debug <<" variable va: " <<StringUtility::addrToString(node.key()) <<"; insns:";
940  for (Address va: node.value().values())
941  debug <<" " <<StringUtility::addrToString(va);
942  debug <<"\n";
943  }
944 
945  if (self.regionContainsInstructions(partitioner, node.key())) {
946  SAWYER_MESG(debug) <<" skipped because region overlaps with instructions\n";
947  continue;
948  }
949  if (!self.regionIsFullyMapped(partitioner, node.key())) {
950  SAWYER_MESG(debug) <<" skipped because sub-region is not fully mapped\n";
951  continue;
952  }
953  if (!self.regionIsFullyReadWrite(partitioner, node.key())) {
954  SAWYER_MESG(debug) <<" skipped because sub-region is not fully read+write\n";
955  continue;
956  }
957 
958  functionResult[node.key().least()].insert(node.value());
959  }
960 
961  // Merge this function's result into the global result
962  SAWYER_THREAD_TRAITS::LockGuard lock(mutex_);
963  for (const auto &node: functionResult)
964  result[node.first] |= node.second;
965  }
966 };
