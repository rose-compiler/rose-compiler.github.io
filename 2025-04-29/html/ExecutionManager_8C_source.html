167 void
168 ExecutionManager::run() {
169  ASSERT_not_null(database_);
170  ASSERT_require(config_);
171  Sawyer::Message::Stream debug(mlog[DEBUG]);
172 
173  // Configuration -- these are documented and checked in `instance`
174  const std::string architectureName = config_["architecture"].as<std::string>();
175  const std::string concreteName = architectureName + "::" + config_["concrete"].as<std::string>();
176  const size_t concolicStride =
177  std::max(StringUtility::toNumber<size_t>(config_["concolic-stride"].as<std::string>()).ok().orElse(1),
178  size_t{1});
179 
180  SAWYER_MESG(debug) <<"execution manager running\n"
181  <<" architecture = \"" <<StringUtility::cEscape(architectureName) <<"\"\n"
182  <<" concrete = \"" <<StringUtility::cEscape(concreteName) <<"\"\n";
183 
184  while (!isFinished()) {
185  // Run as many test cases concretely as possible.
186  while (TestCaseId testCaseId = pendingConcreteResult()) {
187  TestCase::Ptr testCase = database()->object(testCaseId);
188  std::cerr <<testCase->printableName(database()) <<" running concretely...\n";
189  auto concreteExecutor = ConcreteExecutor::forge(database_, concreteName);
190  if (!concreteExecutor)
191  throw Exception("cannot instantiate concrete executor \"" + StringUtility::cEscape(concreteName) + "\"");
192  auto concreteResult = concreteExecutor->execute(testCase);
193  ASSERT_not_null(concreteResult);
194  insertConcreteResults(testCase, concreteResult);
195  }
196 
197  // Now that all the test cases have run concretely, run a few of the "best" ones concolically. The "best" is defined
198  // either by the ranks returned from the concrete executor, or by this class overriding pendingConcolicResult (which we
199  // haven't done).
200  for (TestCaseId testCaseId: pendingConcolicResults(concolicStride)) {
201  TestCase::Ptr testCase = database()->object(testCaseId);
202  std::cerr <<testCase->printableName(database()) <<" running concolically...\n";
203  auto concolicExecutor = ConcolicExecutor::instance();
204  concolicExecutor->settings(concolicExecutorSettings_);
205  concolicExecutor->configureExecution(database_, testCase, config_);
206  std::vector<TestCase::Ptr> newTestCases = concolicExecutor->execute();
207  insertConcolicResults(testCase, newTestCases);
208  }
209  }
210 }
