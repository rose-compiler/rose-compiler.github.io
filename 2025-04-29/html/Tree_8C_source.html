117 Tree::create_preorder_links (Tree::Node* n)
118 {
119  Node* last;
120 
121  ChildNodesIterator iter(n);
122  if (!(bool)iter) {
123  n->next_preorder = 0;
124  last = n;
125  }
126  else {
127  last = n;
128  do {
129  last->next_preorder = (Node*)iter;

141 void
142 Tree::add (Tree::Edge* e)
143  throw (Tree::DuplicateEdge, Tree::EdgeInUse, Tree::EmptyEdge, Tree::SecondParent, Tree::EmptyEndPoint)
144 {
145  if (e == 0)
146  throw EmptyEdge();
147  if (edge_set.find(e) != edge_set.end())
148  throw DuplicateEdge(e);
149  if (e->in_use)
150  throw EdgeInUse(e);
151  if ((e->child_node == 0) || (e->parent_node == 0))
152  throw EmptyEndPoint(e);
153  if (e->child_node->incoming != 0)
154  throw SecondParent(e);
155 
156  // insert the nodes if they don't already exist in the graph
157  if (node_set.find(e->parent_node) == node_set.end())
158  add(e->parent_node);
159  if (node_set.find(e->child_node) == node_set.end())
160  add(e->child_node);
161 
162  // insert the edge in the corresponding lists of the two nodes
163  e->parent_node->outgoing.push_back(e);
164  e->child_node->incoming = e;
165 
166  // finally, insert the edge itself in the tree
167  e->in_use = true;
168  edge_set.insert(e);
169  preorder_needed = postorder_needed = rpostorder_needed = true;
170 }

211 Tree::remove (Tree::Edge* e)
212  throw (Tree::NonexistentEdge, Tree::EmptyEdge)
213 {
214  if (e == 0)
215  throw EmptyEdge();
216  if (edge_set.erase(e) == 0)
217  throw NonexistentEdge(e);
218  preorder_needed = postorder_needed = rpostorder_needed = true;
219  e->in_use = false;
220  deque_erase<Edge*>(e->parent_node->outgoing, e);
221  e->child_node->incoming = 0;
222 }

227 void
228 Tree::remove (Tree::Node* n)
229  throw (Tree::NonexistentNode, Tree::DeletingRootOfNonSingletonTree, Tree::EmptyNode)
230 {
231  if (n == 0)
232  throw EmptyNode();
233  if ((n == root_node) && (!edge_set.empty() || (node_set.size() > 1)))
234  throw DeletingRootOfNonSingletonTree(n);
235  if (node_set.erase(n) == 0)
236  throw NonexistentNode(n);
237  preorder_needed = postorder_needed = rpostorder_needed = true;
238  n->in_use = false;
239  if (n == root_node)
240  root_node = 0;
241 
242  if (n->incoming != 0) {
243  edge_set.erase(n->incoming);
244  deque_erase<Edge*>(n->incoming->parent_node->outgoing, n->incoming);
245  delete n->incoming; // an edge has no existence without its anchoring nodes
246  }
247 
248  OutEdgesIterator out(n);
249  while ((bool)out) {
250  Edge* e = (Edge*)out;
251  e->child_node->incoming = 0;
252  edge_set.erase(e);
253  delete e; // an edge has no existence without its anchoring nodes
254  ++out;
255  }
256 }

264 deque_erase (std::deque<T> d, T elt)
265 {
266  typename std::deque<T>::iterator iter = d.begin();
267  while (iter != d.end()) {
268  T k = *iter;

284 Tree::Node*
285 Tree::clone (Tree::Node* subtree)
286 {
287  Node* new_root = subtree->new_copy();
288  OutEdgesIterator edge_iter(subtree);
289  while ((bool)edge_iter) {
290  Edge* out_edge = edge_iter;
291  if (out_edge != 0) {
292  Node* new_child = clone(out_edge->sink());
293  add(out_edge->new_copy(new_root, new_child));
294  }
295  else
296  new_root->outgoing.push_back(0);
297  ++edge_iter;
298  }
299  return new_root;
300 }
