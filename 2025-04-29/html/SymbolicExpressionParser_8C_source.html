139  unsigned startColumn = columnNumber_;
140  consumeCharacter();
141  switch (nextCharacter()) {
142  case EOF:
143  throw SymbolicExpressionParser::SyntaxError("end of input reached in an escape sequence",
144  name_, startLine, startColumn);
145  case 'a':
146  consumeCharacter();
147  return '\a';
148  case 'b':
149  consumeCharacter();
150  return '\b';
151  case 't':
152  consumeCharacter();
153  return '\t';
154  case 'n':
155  consumeCharacter();
156  return '\n';
157  case 'v':
158  consumeCharacter();
159  return '\v';
160  case 'f':
161  consumeCharacter();
162  return '\f';
163  case 'r':
164  consumeCharacter();
165  return '\r';
166  default:
167  if (strchr("01234567", nextCharacter())) {
168  int c = 0;
169  for (size_t i=0; i<3; ++i) {
170  if (EOF == nextCharacter() || !strchr("01234567", nextCharacter())) {
171  throw SymbolicExpressionParser::SyntaxError("octal escape sequences need three digits",
172  name_, startLine, startColumn);
173  }
174  c = (8 * c) + (nextCharacter() - '0');
175  consumeCharacter();
176  }
177  return c;
178  }
179  break;
180  }
181  return consumeCharacter(); // escaping any other character just returns that character

253 SymbolicExpression::Type
254 SymbolicExpressionParser::TokenStream::consumeType() {
255  // '['
256  consumeWhiteSpaceAndComments();
257  unsigned startLine = lineNumber_;
258  unsigned startColumn = columnNumber_;
259  if (nextCharacter() != '[') {
260  throw SymbolicExpressionParser::SyntaxError("expected '[' to start a type specification",
261  name_, startLine, startColumn);
262  }
263  consumeCharacter();
264 
265  // Type name or width constant
266  consumeWhiteSpaceAndComments();
267  SymbolicExpression::Type retval;
268  if (isdigit(nextCharacter())) {
269  // integer width
270  std::string s;
271  while (isdigit(nextCharacter()))
272  s += consumeCharacter();
273  // optional "->N"
274  if (nextCharacter() == '-') {
275  size_t width1 = boost::lexical_cast<size_t>(s);
276  consumeCharacter();
277  if (nextCharacter() != '>')
278  throw SymbolicExpressionParser::SyntaxError("expected '->' in type specification",
279  name_, lineNumber_, columnNumber_);
280  consumeCharacter();
281  consumeWhiteSpaceAndComments();
282  s = "";
283  unsigned ln = lineNumber_, cn = columnNumber_;
284  while (isdigit(nextCharacter()))
285  s += consumeCharacter();
286  if (s.empty())
287  throw SymbolicExpressionParser::SyntaxError("expected decimal integer domain width after '->'", name_, ln, cn);
288  size_t width2 = boost::lexical_cast<size_t>(s);
289  consumeWhiteSpaceAndComments();
290  retval = SymbolicExpression::Type::memory(width1, width2);
291  } else {
292  retval = SymbolicExpression::Type::integer(boost::lexical_cast<size_t>(s));
293  }
294  } else if (nextCharacter() == 'u') {
295  // integer followed by width
296  consumeCharacter();
297  if (!isdigit(nextCharacter()))
298  throw SymbolicExpressionParser::SyntaxError("invalid type specification", name_, startLine, startColumn);
299  std::string s;
300  while (isdigit(nextCharacter()))
301  s += consumeCharacter();
302  retval = SymbolicExpression::Type::integer(boost::lexical_cast<size_t>(s));
303  } else if (nextCharacter() == 'f') {
304  // floating point followed by width
305  consumeCharacter();
306  if (!isdigit(nextCharacter()))
307  throw SymbolicExpressionParser::SyntaxError("invalid type specification", name_, startLine, startColumn);
308  std::string s;
309  while (isdigit(nextCharacter()))
310  s += consumeCharacter();
311  if ("32" == s) {
312  retval = SymbolicExpression::Type::floatingPoint(8, 24);
313  } else if ("64" == s) {
314  retval = SymbolicExpression::Type::floatingPoint(11, 53);
315  } else {
316  throw SymbolicExpressionParser::SyntaxError("invalid type specification", name_, startLine, startColumn);
317  }
318  } else {
319  throw SymbolicExpressionParser::SyntaxError("invalid type specification", name_, startLine, startColumn);
320  }
321  consumeWhiteSpaceAndComments();
322 
323  // ']'
324  if (nextCharacter() == EOF) {
325  throw SymbolicExpressionParser::SyntaxError("end of input reached inside '[' type expression",
326  name_, startLine, startColumn);
327  }
328  if (nextCharacter() != ']')
329  throw SymbolicExpressionParser::SyntaxError("missing closing ']' in type specification", name_, startLine, startColumn);
330  consumeCharacter();
331  return retval;
332 }

334 SymbolicExpressionParser::Token
335 SymbolicExpressionParser::TokenStream::scan() {
336  consumeWhiteSpaceAndComments();
337  unsigned startLine = lineNumber_;
338  unsigned startColumn = columnNumber_;
339  switch (nextCharacter()) {
340  case EOF:
341  return Token();
342  case '<':
343  ASSERT_not_reachable("'<' should have been consumed by consumeWhiteSpaceAndComments");
344  case '>':
345  throw SymbolicExpressionParser::SyntaxError("found '>' outside inline comment", name_, startLine, startColumn);
346  case '(':
347  consumeCharacter();
348  return Token(Token::LTPAREN, SymbolicExpression::Type(), "(", startLine, startColumn);
349  case ')':
350  consumeCharacter();
351  return Token(Token::RTPAREN, SymbolicExpression::Type(), ")", startLine, startColumn);
352  case '[':
353  throw SymbolicExpressionParser::SyntaxError("unexpected width specification", name_, startLine, startColumn);
354  case ']':
355  throw SymbolicExpressionParser::SyntaxError("found ']' outside width specification", name_, startLine, startColumn);
356  default: {
357  std::string s = consumeTerm();
358  boost::smatch matches;
359  if (boost::regex_match(s, matches, hexLiteralRe)) {
360  SymbolicExpression::Type exprType = consumeType();
361  if (exprType.typeClass() == SymbolicExpression::Type::MEMORY)
362  throw SymbolicExpressionParser::SyntaxError("type of hex literal must be scalar", name_, startLine, startColumn);
363  Sawyer::Container::BitVector bv(exprType.nBits());
364  bv.fromHex(matches.str(1)); // hex digits without leading "0x"
365  return Token(bv, exprType, s, startLine, startColumn);
366  } else if (boost::regex_match(s, signedDecimalLiteralRe)) {
367  int n = boost::lexical_cast<int>(s);
368  consumeWhiteSpaceAndComments();
369  SymbolicExpression::Type exprType = nextCharacter() == '[' ? consumeType() : SymbolicExpression::Type::integer(8*sizeof(int));
370  if (exprType.typeClass() == SymbolicExpression::Type::MEMORY)
371  throw SymbolicExpressionParser::SyntaxError("type of decimal literal must be scalar", name_, startLine, startColumn);
372  Sawyer::Container::BitVector bv(exprType.nBits());
373  bv.fromInteger(IntegerOps::signExtend2((uint64_t)n, 8*sizeof(int), 64));
374  return Token(bv, exprType, s, startLine, startColumn);
375  } else {
376  consumeWhiteSpaceAndComments();
377  SymbolicExpression::Type exprType = nextCharacter() == '[' ? consumeType() : SymbolicExpression::Type::none();
378  return Token(Token::SYMBOL, exprType, s, startLine, startColumn);
379  }
380  }
381  }
382  return Token();
383 }

959 SymbolicExpression::Ptr
960 SymbolicExpressionParser::RegisterSubstituter::delayedExpansion(const SymbolicExpression::Ptr &src, const SymbolicExpressionParser *parser) {
961  ASSERT_not_null(src);
962  ASSERT_always_not_null(parser);
963  ASSERT_not_null(ops_);
964 
965  namespace SS = Rose::BinaryAnalysis::InstructionSemantics::SymbolicSemantics;
966  namespace BS = Rose::BinaryAnalysis::InstructionSemantics::BaseSemantics;
967  Sawyer::Message::Stream debug(SymbolicExpressionParser::mlog[DEBUG]);
968 
969  RegisterDescriptor reg;
970  if (reg2var_.reverse().getOptional(src).assignTo(reg)) {
971  // Earlier (in immediateExpansion), we set the temporary variable's comment to be the original variable (register)
972  // name including any "_0" suffix. Now, if we're expanding an "_0" register we should read from the original state
973  // rather than the current state.
974  BS::RegisterState::Ptr regState;
975  if (boost::ends_with(src->comment(), "_0")) {
976  if (!ops_->initialState()) {
977  std::ostringstream ss;
978  ss <<"no initial state from which to read register"

982  regState = ops_->initialState()->registerState();
983  } else {
984  regState = ops_->currentState()->registerState();
985  }

987  // Read the register
988  SS::SValue::Ptr regval = SS::SValue::promote(regState->readRegister(reg, ops_->undefined_(reg.nBits()), ops_.get()));

991  }
