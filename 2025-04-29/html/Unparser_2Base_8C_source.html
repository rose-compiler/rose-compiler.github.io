266 State::reachabilityName(Reachability::ReasonFlags value) const {
267  std::string s;
268  if (reachabilityNames_.getOptional(value.vector()).assignTo(s))
269  return s;
270 
271  std::vector<std::string> names;
272  for (size_t i = 0; i < 8*sizeof(Reachability::ReasonFlags::Vector); ++i) {
273  Reachability::ReasonFlags bit(1U << i);
274  if (value.isAllSet(bit)) {
275  if (reachabilityNames_.getOptional(bit.vector()).assignTo(s)) {
276  // void
277  } else if (bit.vector() >= Reachability::USER_DEFINED_0) {
278  s = "user-defined";

280  if (bit.vector() >> j == Reachability::USER_DEFINED_0) {
281  s += "-" + StringUtility::numberToString(j);

284  }
285  } else if (const char *cs = stringify::Rose::BinaryAnalysis::Reachability::Reason(bit.vector())) {
286  s = cs;
287  for (char &ch: s)
288  ch = tolower(ch);
289  } else {
290  s = StringUtility::toHex2(bit.vector(), 8*sizeof(Reachability::ReasonFlags::Vector), false, false);

696 std::string
697 ArrowMargin::render() {
698  std::string retval;
699  if (atEndpoint_) {
700  if (whichEnd(*atEndpoint_) == CfgEndpoint::End::Target) {
701  retval = arrows_.render(*atEndpoint_, EdgeArrows::FIRST_LINE); // arrows terminate at the beginning of a basic block
702  nextPart_ = EdgeArrows::MIDDLE_LINE; // and now we're inside the basic block
703  } else {
704  retval = arrows_.render(*atEndpoint_, EdgeArrows::LAST_LINE); // arrows originate at the end of a basic block
705  nextPart_ = EdgeArrows::INTER_LINE; // and now we're between basic blocks
706  }
707  latestEndpoint_ = atEndpoint_;
708  atEndpoint_ = Sawyer::Nothing();
709  } else if (latestEndpoint_) {
710  retval = arrows_.render(*latestEndpoint_, nextPart_); // continue with what we had before
711  } else {
712  retval = arrows_.renderBlank(); // we haven't seen any endpoints yet
713  }
714  return retval;
715 }

749

751  std::string noEscapes = StringUtility::removeAnsiEscapes(s);

758 // class method
759 std::string
760 Base::juxtaposeColumns(const std::vector<std::string> &parts, const std::vector<size_t> &minWidths,
761  const std::vector<std::pair<std::string, std::string> > &colorEscapes,
762  const std::string &columnSeparator) {
763  ASSERT_require(minWidths.size() == parts.size());
764  ASSERT_require(colorEscapes.size() == parts.size());
765 
766  std::string retval;
767 
768  // Split each part into separate lines
769  size_t nlines = 0;
770  std::vector<std::vector<std::string> > partLines(parts.size());
771  for (size_t i=0; i<parts.size(); ++i) {
772  partLines[i] = StringUtility::split('\n', parts[i]);
773  nlines = std::max(nlines, partLines[i].size());
774  }
775 
776  // Emit one line at a time across all columns
777  const std::string empty;
778  for (size_t i=0; i<nlines; ++i) {
779  if (i != 0)
780  retval += "\n";
781  std::ostringstream ss;
782  for (size_t j=0; j<partLines.size(); ++j) {
783  const std::string &s = i < partLines[j].size() ? partLines[j][i] : empty;
784  if (j > 0)
785  ss <<columnSeparator;
786  ss <<colorEscapes[j].first <<leftJustify(s, minWidths[j]) <<colorEscapes[j].second;
787  }
788  retval += boost::trim_right_copy(ss.str());
789  }
790  return retval;
791 }

1325 void
1326 Base::computeGutterArrows(const P2::Function::Ptr &function, State &state) const {
1327  ASSERT_not_null(function);
1328  ArrowMargin &margin = state.intraFunctionCfgArrows();
1329  margin.reset();
1330 
1331  P2::Partitioner::ConstPtr partitioner = state.partitioner();
1332 
1333  // Process basic blocks in the order that they will be emitted by the unparser.
1334  const std::vector<Address> bbAddrs(function->basicBlockAddresses().begin(), function->basicBlockAddresses().end());
1335  for (size_t i = 0; i < bbAddrs.size(); ++i) {
1336  const Address bbAddr = bbAddrs[i];
1337  const P2::ControlFlowGraph::ConstVertexIterator bbVertex = partitioner->findPlaceholder(bbAddr);
1338  ASSERT_require(bbVertex != partitioner->cfg().vertices().end());
1339  const P2::BasicBlock::Ptr curBb = bbVertex->value().bblock();
1340  ASSERT_not_null(curBb);
1341 
1342  const P2::BasicBlock::Ptr prevBb = i > 0 ? partitioner->basicBlockExists(bbAddrs[i-1]) : P2::BasicBlock::Ptr();
1343  const P2::BasicBlock::Ptr nextBb = i + 1 < bbAddrs.size() ?
1344  partitioner->basicBlockExists(bbAddrs[i+1]) : P2::BasicBlock::Ptr();
1345 
1346  //----------------------------------------------------
1347  // Arrows coming into the top of this basic block
1348  //----------------------------------------------------
1349 
1350  const auto incomingEdges = orderedBlockPredecessors(partitioner, curBb);
1351  for (const auto &incomingEdge: incomingEdges) {
1352  // We don't care about inter-function edges or suppressed fallthrough edges.
1353  const auto sourceAddr = incomingEdge->source()->value().optionalAddress();
1354  if (sourceAddr && function->ownsBasicBlock(*sourceAddr) &&
1355  !suppressFallThroughEdge(*incomingEdge, prevBb, curBb, state)) {
1356  const EdgeArrows::EndpointId arrowSrc = margin.getEndpoint(ArrowMargin::CfgEndpoint::End::Source, *incomingEdge);
1357  const EdgeArrows::EndpointId arrowTgt = margin.getEndpoint(ArrowMargin::CfgEndpoint::End::Target, *incomingEdge);
1358  margin.maybeInsertArrow(arrowSrc, arrowTgt, incomingEdge->id());
1359  margin.maybeInsertEndpoint(arrowTgt);
1360  }
1361  }
1362 
1363  //----------------------------------------------------
1364  // Arrows going out of the bottom of this basic block
1365  //----------------------------------------------------
1366 
1367  const auto outgoingEdges = orderedBlockSuccessors(partitioner, curBb);
1368  for (const auto &outgoingEdge: outgoingEdges) {
1369  // We don't care about inter-function edges or suppressed fallthrough edges.
1370  const auto targetAddr = outgoingEdge->target()->value().optionalAddress();
1371  if (targetAddr && function->ownsBasicBlock(*targetAddr) &&
1372  !suppressFallThroughEdge(*outgoingEdge, curBb, nextBb, state)) {
1373  const EdgeArrows::EndpointId arrowSrc = margin.getEndpoint(ArrowMargin::CfgEndpoint::End::Source, *outgoingEdge);
1374  const EdgeArrows::EndpointId arrowTgt = margin.getEndpoint(ArrowMargin::CfgEndpoint::End::Target, *outgoingEdge);
1375  margin.maybeInsertArrow(arrowSrc, arrowTgt, outgoingEdge->id());
1376  margin.maybeInsertEndpoint(arrowSrc);
1377  }
1378  }
1379  }
1380 
1381  margin.computeLayout();
1382 }

1500 struct EmitBlockVisitor: public boost::static_visitor<> {
1501  std::ostream &out;
1502  P2::Function::Ptr function;
1503  Address &fallThroughAddr; // tracks the fall-through address for each emitted block
1504  const InsnsOrData *prevBlock, *nextBlock; // optional pointers to the previous and next blocks to be emitted
1505  State &state;
1506 
1507  EmitBlockVisitor(std::ostream &out, const P2::Function::Ptr &function, Address &fallThroughAddr, const InsnsOrData *prevBlock,
1508  const InsnsOrData *nextBlock, State &state)
1509  : out(out), function(function), fallThroughAddr(fallThroughAddr), prevBlock(prevBlock), nextBlock(nextBlock),
1510  state(state) {}
1511 
1512  void operator()(const P2::BasicBlock::Ptr &bb) const {
1513  if (state.frontUnparser().settings().bblock.cfg.showingFallThroughEdges) {
1514  state.frontUnparser().emitLinePrefix(out, state);
1515  out <<"\n";
1516  }
1517 
1518  BasicBlockGuard push(state, getBasicBlockPtr(prevBlock), bb, getBasicBlockPtr(nextBlock));
1519  if (bb->address() != *function->basicBlockAddresses().begin()) {
1520  if (bb->address() > fallThroughAddr) {
1521  state.frontUnparser().emitLinePrefix(out, state);
1522  StyleGuard style(state.styleStack(), state.frontUnparser().settings().comment.line.style);
1523  out <<style.render();
1524  out <<";;; skip forward " <<StringUtility::plural(bb->address() - fallThroughAddr, "bytes") <<"\n";
1525  out <<style.restore();
1526  } else if (bb->address() < fallThroughAddr) {
1527  state.frontUnparser().emitLinePrefix(out, state);
1528  StyleGuard style(state.styleStack(), state.frontUnparser().settings().comment.line.style);
1529  out <<style.render();
1530  out <<";;; skip backward " <<StringUtility::plural(fallThroughAddr - bb->address(), "bytes") <<"\n";
1531  out <<style.restore();
1532  }
1533  }
1534  state.frontUnparser().emitBasicBlock(out, bb, state);
1535  if (bb->nInstructions() > 0) {
1536  fallThroughAddr = bb->fallthroughVa();
1537  } else {
1538  fallThroughAddr = bb->address();
1539  }
1540  }
1541 
1542  void operator()(const P2::DataBlock::Ptr &db) const {
1543  state.frontUnparser().emitLinePrefix(out, state);
1544  out <<"\n";
1545  if (db->address() != *function->basicBlockAddresses().begin()) {
1546  if (db->address() > fallThroughAddr) {
1547  state.frontUnparser().emitLinePrefix(out, state);
1548  StyleGuard style(state.styleStack(), state.frontUnparser().settings().comment.line.style);
1549  out <<style.render();
1550  out <<";;; skip forward " <<StringUtility::plural(db->address() - fallThroughAddr, "bytes") <<"\n";
1551  out <<style.restore();

1555  out <<style.render();

1561  fallThroughAddr = db->address() + db->size();

1565 void
1566 Base::emitFunctionBody(std::ostream &out, const P2::Function::Ptr &function, State &state) const {
1567  if (nextUnparser()) {
1568  nextUnparser()->emitFunctionBody(out, function, state);
1569  } else {
1570  // If we're not emitting instruction addresses then we need some other way to identify basic blocks for branch targets.
1571  if (settings().insn.address.showing && settings().insn.address.useLabels) {
1572  state.basicBlockLabels().clear();
1573  for (Address bbVa: function->basicBlockAddresses()) {
1574  std::string label = "L" + boost::lexical_cast<std::string>(state.basicBlockLabels().size()+1);
1575  state.basicBlockLabels().insertMaybe(bbVa, label);
1576  }
1577  }
1578 
1579  // Get all the basic blocks and data blocks and sort them by starting address. Be careful because data blocks might
1580  // appear more than once since they can be owned by both the function and by multiple basic blocks.
1581  std::set<P2::DataBlock::Ptr> dblocks;
1582  std::vector<InsnsOrData> blocks;
1583  blocks.reserve(function->nBasicBlocks() + function->nDataBlocks());
1584  ASSERT_not_null(state.partitioner());
1585  for (Address bbVa: function->basicBlockAddresses()) {
1586  if (P2::BasicBlock::Ptr bb = state.partitioner()->basicBlockExists(bbVa)) {
1587  blocks.push_back(bb);
1588  for (const P2::DataBlock::Ptr &db: bb->dataBlocks())
1589  dblocks.insert(db);
1590  }
1591  }
1592  for (const P2::DataBlock::Ptr &db: function->dataBlocks())
1593  dblocks.insert(db);
1594  blocks.insert(blocks.end(), dblocks.begin(), dblocks.end());
1595  std::sort(blocks.begin(), blocks.end(), increasingAddress);
1596 
1597  // Emit each basic- or data-block
1598  Address fallThroughAddr = function->address();
1599  for (size_t i = 0; i < blocks.size(); ++i) {
1600  const InsnsOrData &curBlock = blocks[i];
1601  const InsnsOrData *prevBlock = i > 0 ? &blocks[i-1] : nullptr;
1602  const InsnsOrData *nextBlock = i + 1 < blocks.size() ? &blocks[i+1] : nullptr;

1604  }
1605  }

1932 void
1933 Base::emitBasicBlockBody(std::ostream &out, const P2::BasicBlock::Ptr &bb, State &state) const {
1934  ASSERT_not_null(bb);
1935  if (nextUnparser()) {
1936  nextUnparser()->emitBasicBlockBody(out, bb, state);
1937  } else {
1938  if (0 == bb->nInstructions()) {
1939  state.thisIsBasicBlockFirstInstruction();

1947  state.thisIsBasicBlockFirstInstruction();

1950  for (SgAsmInstruction *insn: bb->instructions()) {

1955  // \u{21aa}. Anyway, we need to be able to support ASCII-art arrows. Therfore, if a basic block has an arrow
1956  // that originates and terminates at itself and the block has only one instruction, we need to delay the
1957  // outgoing arrows and place them on a line following the last instruction.

1960  // call `atPossibleEndpoint` twice for the same line of output.
1961  ArrowMargin &margin = state.intraFunctionCfgArrows();

1963  margin.findEndpoint(ArrowMargin::CfgEndpoint::End::Target, *bbVertex) &&
1964  margin.findEndpoint(ArrowMargin::CfgEndpoint::End::Source, *bbVertex)) {
1965  arrowsNeedExtraLine = true;

1967  margin.atPossibleEndpoint(ArrowMargin::CfgEndpoint::End::Source, *bbVertex);

1969  }

1979  if (arrowsNeedExtraLine) {
1980  state.intraFunctionCfgArrows().atPossibleEndpoint(ArrowMargin::CfgEndpoint::End::Source, *bbVertex);
1981  arrowsNeedExtraLine = false;
1982  }
1983

1985  std::ostringstream ss;
1986  state.isPostInstruction(true);
1987  emitInstruction(ss, insn, state);
1988  state.isPostInstruction(false);
1989  const bool hasNonSpace = [](const std::string &s) {
1990  for (char ch: s) {
1991  if (!isspace(ch))
1992  return true;
1993  }
1994  return false;
1995  }(ss.str());
1996  if (hasNonSpace)
1997  out <<ss.str() <<"\n";
1998  }

2001  if (arrowsNeedExtraLine) {
2002  state.intraFunctionCfgArrows().atPossibleEndpoint(ArrowMargin::CfgEndpoint::End::Source, *bbVertex);
2003  state.frontUnparser().emitLinePrefix(out, state);
