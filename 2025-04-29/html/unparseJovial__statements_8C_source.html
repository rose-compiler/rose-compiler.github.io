270 UnparseJovial::unparseProcDeclStmt(SgStatement* stmt, SgUnparse_Info& info)
271  {
272  SgUnparse_Info ninfo(info);
273 
274  SgBasicBlock* func_body{nullptr};
275  SgScopeStatement* param_scope{nullptr};
276 
277  SgProcedureHeaderStatement* func = isSgProcedureHeaderStatement(stmt);
278  ASSERT_not_null(func);
279 
280  bool is_defining_decl = (func->get_declarationModifier().isJovialRef() == false);
281 
282  SgFunctionDefinition* func_def = func->get_definition();
283  if (func_def) {
284  param_scope = func_body = func_def->get_body();
285  ASSERT_not_null(func_body);
286  }
287  else {
288  param_scope = func->get_functionParameterScope();
289  }
290  ASSERT_not_null(param_scope);
291 
292  // unparse the declaration modifiers
293  if (func->get_declarationModifier().isJovialDef()) curprint_indented("DEF ", info);
294  if (func->get_declarationModifier().isJovialRef()) curprint_indented("REF ", info);
295 
296  curprint("PROC ");
297  curprint(func->get_name());
298 
299  // unparse the function modifiers
300  if (func->get_functionModifier().isRecursive()) curprint(" REC");
301  if (func->get_functionModifier().isReentrant()) curprint(" RENT");
302 
303  // unparse function arguments
304  SgFunctionParameterList* params = func->get_parameterList();
305  SgInitializedNamePtrList & args = params->get_args();
306 
307  const std::vector<SgInitializedName*> in_params = SageInterface::getInParameters(args);
308  const std::vector<SgInitializedName*> out_params = SageInterface::getOutParameters(args);
309 
310  curprint("(");
311 
312  bool print_comma = false;
313  for (SgInitializedName* param : in_params) {
314  if (print_comma) curprint(",");
315  curprint(param->get_name());
316  print_comma = true;
317  }
318 
319  print_comma = false;
320  for (SgInitializedName* param : out_params) {
321  if (print_comma) curprint(",");
322  else curprint(":");
323  curprint(param->get_name());
324  print_comma = true;
325  }
326 
327  curprint(")");
328 
329  // unparse function type
330  SgType* type = func->get_type();
331  unparseType(type, ninfo);
332 
333  curprint(";\n");
334 
335  if (is_defining_decl) {
336  auto block_size = func_body->get_statements().size();
337  info.inc_nestingLevel();
338 
339  // Don't add extra BEGIN/ENDs (if block_size > 1, SgBasicBlock adds BEGIN/END)
340  if (block_size == 1) {
341  curprint_indented("BEGIN\n", info);
342  unparseStatement(func_def, ninfo);
343  curprint_indented("END\n", info);
344  }
345  else {
346  unparseStatement(func_def, ninfo);
347  }
348 
349  info.dec_nestingLevel();
350  }
351  else {
352  for (SgDeclarationStatement* decl : param_scope->getDeclarationList()) {
353  if (isSgJovialDirectiveStatement(decl)) {
354  unparseStatement(decl, ninfo);
355  }
356  }
357 
358  // There still needs to be at least a BEGIN and END
359  curprint_indented("BEGIN\n", info);
360 
361  info.inc_nestingLevel();
362  for (SgInitializedName* arg : args) {
363  SgVariableSymbol* var_sym = SageInterface::lookupVariableSymbolInParentScopes(arg->get_name(), param_scope);
364  SgInitializedName* var_init_name = var_sym->get_declaration();
365  ASSERT_not_null(var_init_name);
366  SgVariableDeclaration* var_decl = isSgVariableDeclaration(var_init_name->get_declaration());
367  ASSERT_not_null(var_decl);
368 
369  unparseVarDeclStmt(var_decl, info);
370  }
371  info.dec_nestingLevel();
372 
373  curprint_indented("END\n", info);
374  } // !is_defining_decl
375  }

684  // may be the last statement. The other statement (if present) will be the compiler generated control
685  // variable declaration. However a SimpleStatement for the loop body won't have the extra basic block.
686  //
687  // In any case, comments may be attached before outer block loop body (not to actual loop body, see note
688  // above about about weird construction and code below finding loop_body).
689  unparseCommentsBefore(loop_body, info);

692  loop_body = isSgBasicBlock(for_stmt->get_loop_body()->get_statements()[0]);
693  }

697  if (var_decl) SageInterface::removeStatement(var_decl);
698  loop_body = isSgBasicBlock(for_stmt->get_loop_body()->get_statements()[1]);
699  }
700 
701  if (!loop_body) {
702  loop_body = isSgBasicBlock(for_stmt->get_loop_body());
703  }
704  ROSE_ASSERT(loop_body);
705 
706  unparseStatement(loop_body, info);
707  unp->cur.insert_newline(1);
