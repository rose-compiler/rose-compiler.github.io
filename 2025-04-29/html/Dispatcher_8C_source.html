146 Dispatcher::advanceInstructionPointer(SgAsmInstruction *insn) {
147  operators()->comment("advancing instruction pointer");
148  RegisterDescriptor ipReg = instructionPointerRegister();
149  size_t nBits = ipReg.nBits();
150  BaseSemantics::SValue::Ptr ipValue;
151  if (!autoResetInstructionPointer_ && operators()->currentState() && operators()->currentState()->registerState()) {
152  auto grState = as<BaseSemantics::RegisterStateGeneric>(operators()->currentState()->registerState());
153  if (grState && grState->is_partly_stored(ipReg))
154  ipValue = operators()->readRegister(ipReg);
155  }
156  if (!ipValue)
157  ipValue = operators()->number_(nBits, insn->get_address());
158  ipValue = operators()->add(ipValue, operators()->number_(nBits, insn->get_size()));
159  operators()->writeRegister(ipReg, ipValue);

162 void
163 Dispatcher::processCommon() {
164  SgAsmInstruction *insn = notnull(operators()->currentInstruction());
165  InsnProcessor *iproc = iprocLookup(insn);
166  try {
167  if (!iproc)
168  throw NotImplemented("no semantics for \"" + insn->get_mnemonic() + "\" instruction", insn);
169  iproc->process(shared_from_this(), insn);
170  } catch (Exception &e) {
171  // If the exception was thrown by something that didn't have an instruction available, then add the instruction
172  if (!e.insn)
173  e.insn = insn;
174  e.insn->incrementSemanticFailure();
175  throw;
176  }
177 }

190 void
191 Dispatcher::processDelaySlot(SgAsmInstruction *insn) {
192  if (insn) {
193  // Remember the outer instruction -- the one that we were previously processing and which has a delay slot. We'll eventually
194  // return to processing the outer instruction.
195  SgAsmInstruction *outerInsn = operators()->currentInstruction();
196  ASSERT_not_null(outerInsn);
197 
198  // Raise an interrupt if an illegal (control transfer) instruction is found in the delay slot
199  SValue::Ptr raise{operators()->boolean_(false)};
200  if (architecture()->isControlTransfer(notnull(insn))) {
201  raise = operators()->boolean_(true);
202  }
203  operators()->raiseInterrupt(mips_signal_exception, mips_reserved_instruction, raise);
204 
205  // Execute the inner instruction -- this is the delay slot and is executing within the context of the outer instruction.
206  // Before returning (either normally or via exception) make sure we restore the outer instruction. In the case of an
207  // exception, the outer instruction might catch the exception from the inner instruction and resume its own processing.
208  operators()->comment("starting delay slot");
209  operators()->startInstruction(insn);
210  BOOST_SCOPE_EXIT(outerInsn, this_) {
211  this_->operators()->currentInstruction(outerInsn);
212  this_->operators()->comment("resuming after delay slot");
213  } BOOST_SCOPE_EXIT_END;
214 
215  processCommon();
216  operators()->finishInstruction(insn);
217  }

283  forwardPre<SgAsmRegisterReferenceExpression>(e, [&ops](SgAsmRegisterReferenceExpression *rre) {

312  using namespace Rose::AST::Traversal;
313  forwardPre<SgAsmBinaryPreupdate>(e, [this, &enabled](SgAsmBinaryPreupdate *op) {
314  if (enabled->isTrue()) {
315  // Definitely updating, therefore not necessary to read old value
316  BaseSemantics::SValue::Ptr newValue = read(op->get_rhs());
317  write(op->get_lhs(), newValue);
318  } else if (!enabled->isConcrete() || enabled->toUnsigned().get() != 0) {
319  // Maybe updating (but not "definitely not updating")
320  BaseSemantics::SValue::Ptr oldValue = read(op->get_lhs());
321  BaseSemantics::SValue::Ptr newValue = read(op->get_rhs());
322  BaseSemantics::SValue::Ptr maybe = operators()->ite(enabled, newValue, oldValue);
323  write(op->get_lhs(), maybe);
324  }

346 SValue::Ptr
347 Dispatcher::effectiveAddress(SgAsmExpression *e, size_t nbits/*=0*/)
348 {
349  BaseSemantics::SValue::Ptr retval;
350 #if 1 // DEBUGGING [Robb P. Matzke 2015-08-04]
351  ASSERT_always_require(retval==NULL);
352 #endif
353  if (0==nbits)
354  nbits = addressWidth();
355 
356  if (SgAsmMemoryReferenceExpression *mre = isSgAsmMemoryReferenceExpression(e)) {
357  retval = effectiveAddress(mre->get_address(), nbits);
358  } else if (SgAsmRegisterReferenceExpression *rre = isSgAsmRegisterReferenceExpression(e)) {
359  retval = read(rre, nbits);
360  } else if (SgAsmBinaryAdd *op = isSgAsmBinaryAdd(e)) {
361  BaseSemantics::SValue::Ptr lhs = effectiveAddress(op->get_lhs(), nbits);
362  BaseSemantics::SValue::Ptr rhs = effectiveAddress(op->get_rhs(), nbits);
363  retval = operators()->add(lhs, rhs);
364  } else if (SgAsmBinarySubtract *op = isSgAsmBinarySubtract(e)) {
365  BaseSemantics::SValue::Ptr lhs = effectiveAddress(op->get_lhs(), nbits);
366  BaseSemantics::SValue::Ptr rhs = effectiveAddress(op->get_rhs(), nbits);
367  retval = operators()->subtract(lhs, rhs);
368  } else if (SgAsmBinaryPreupdate *op = isSgAsmBinaryPreupdate(e)) {
369  // The add operation is ignored here, but performed by preUpdate.
370  retval = effectiveAddress(op->get_lhs(), nbits);
371  } else if (SgAsmBinaryPostupdate *op = isSgAsmBinaryPostupdate(e)) {
372  // The add operation is ignored here, but performed by postUpdate.
373  retval = effectiveAddress(op->get_lhs(), nbits);
374  } else if (SgAsmBinaryMultiply *op = isSgAsmBinaryMultiply(e)) {
375  BaseSemantics::SValue::Ptr lhs = effectiveAddress(op->get_lhs(), nbits);
376  BaseSemantics::SValue::Ptr rhs = effectiveAddress(op->get_rhs(), nbits);
377  retval = operators()->unsignedMultiply(lhs, rhs);
378  } else if (SgAsmIntegerValueExpression *ival = isSgAsmIntegerValueExpression(e)) {
379  retval = operators()->number_(ival->get_significantBits(), ival->get_value());
380  } else if (SgAsmUnaryUnsignedExtend *op = isSgAsmUnaryUnsignedExtend(e)) {
381  BaseSemantics::SValue::Ptr lhs = effectiveAddress(op->get_operand(), op->get_operand()->get_nBits());
382  retval = operators()->unsignedExtend(lhs, op->get_nBits());
383  } else if (SgAsmUnarySignedExtend *op = isSgAsmUnarySignedExtend(e)) {
384  BaseSemantics::SValue::Ptr lhs = effectiveAddress(op->get_operand(), op->get_operand()->get_nBits());
385  retval = operators()->signExtend(lhs, op->get_nBits());
386  } else if (SgAsmUnaryTruncate *op = isSgAsmUnaryTruncate(e)) {
387  BaseSemantics::SValue::Ptr lhs = effectiveAddress(op->get_operand(), op->get_operand()->get_nBits());
388  retval = operators()->unsignedExtend(lhs, op->get_nBits()); // yes, can be used for truncation
389  } else if (SgAsmBinaryLsl *op = isSgAsmBinaryLsl(e)) {
390  BaseSemantics::SValue::Ptr lhs = effectiveAddress(op->get_lhs(), nbits);
391  BaseSemantics::SValue::Ptr rhs = effectiveAddress(op->get_rhs(), nbits);
392  retval = operators()->shiftLeft(lhs, rhs);
393  } else if (SgAsmBinaryLsr *op = isSgAsmBinaryLsr(e)) {
394  BaseSemantics::SValue::Ptr lhs = effectiveAddress(op->get_lhs(), nbits);
395  BaseSemantics::SValue::Ptr rhs = effectiveAddress(op->get_rhs(), nbits);
396  retval = operators()->shiftRight(lhs, rhs);
397  } else if (SgAsmBinaryAsr *op = isSgAsmBinaryAsr(e)) {
398  BaseSemantics::SValue::Ptr lhs = effectiveAddress(op->get_lhs(), nbits);
399  BaseSemantics::SValue::Ptr rhs = effectiveAddress(op->get_rhs(), nbits);
400  retval = operators()->shiftRightArithmetic(lhs, rhs);
401  } else if (SgAsmBinaryRor *op = isSgAsmBinaryRor(e)) {
402  BaseSemantics::SValue::Ptr lhs = effectiveAddress(op->get_lhs(), nbits);
403  BaseSemantics::SValue::Ptr rhs = effectiveAddress(op->get_rhs(), nbits);
404  retval = operators()->rotateRight(lhs, rhs);
405  } else if (SgAsmUnaryUnsignedExtend *op = isSgAsmUnaryUnsignedExtend(e)) {
406  BaseSemantics::SValue::Ptr lhs = effectiveAddress(op->get_operand(), op->get_operand()->get_nBits());
407  retval = operators()->unsignedExtend(lhs, op->get_nBits());
408  }
409 
410  ASSERT_not_null(retval);
411  if (retval->nBits() < nbits) {
412  retval = operators()->signExtend(retval, nbits);
413  } else if (retval->nBits() > nbits) {
414  retval = operators()->extract(retval, 0, nbits);
415  }
416  return retval;
417 }

430 void
431 Dispatcher::write(SgAsmExpression *e, const SValue::Ptr &value, size_t addr_nbits/*=0*/)
432 {
433  ASSERT_not_null(e);
434  ASSERT_not_null(value);
435  if (SgAsmDirectRegisterExpression *re = isSgAsmDirectRegisterExpression(e)) {
436  // If the expression type is different than the register lvalue, we need to adjust what part of the register we're writing.
437  RegisterDescriptor reg = re->get_descriptor();
438  if (reg.nBits() > e->get_nBits())
439  reg = RegisterDescriptor(reg.majorNumber(), reg.minorNumber(), reg.offset(), e->get_nBits());
440  operators()->writeRegister(reg, value);
441  } else if (SgAsmIndirectRegisterExpression *re = isSgAsmIndirectRegisterExpression(e)) {
442  SValue::Ptr offset = operators()->readRegister(re->get_offset());
443  if (!offset->isConcrete()) {
444  std::string offset_name = registerDictionary()->lookup(re->get_offset());
445  offset_name = offset_name.empty() ? "" : "(" + offset_name + ") ";
446  throw Exception("indirect register offset " + offset_name + "must have a concrete value", NULL);
447  }
448  size_t idx = (offset->toUnsigned().get() + re->get_index()) % re->get_modulus();
449  RegisterDescriptor reg = re->get_descriptor();
450  reg.majorNumber(reg.majorNumber() + re->get_stride().majorNumber() * idx);
451  reg.minorNumber(reg.minorNumber() + re->get_stride().minorNumber() * idx);
452  reg.offset(reg.offset() + re->get_stride().offset() * idx);
453  reg.nBits(reg.nBits() + re->get_stride().nBits() * idx);
454  operators()->writeRegister(reg, value);
455  } else if (SgAsmMemoryReferenceExpression *mre = isSgAsmMemoryReferenceExpression(e)) {
456  SValue::Ptr addr = effectiveAddress(mre, addr_nbits);
457  ASSERT_require(addr->nBits() == architecture()->bitsPerWord());
458  operators()->writeMemory(segmentRegister(mre), addr, value, operators()->boolean_(true));
459  } else {
460  ASSERT_not_implemented(e->class_name());
461  }
462 }

464 SValue::Ptr
465 Dispatcher::read(SgAsmExpression *e, size_t value_nbits/*=0*/, size_t addr_nbits/*=0*/)
466 {
467  ASSERT_not_null(e);
468  if (0 == value_nbits) {
469  SgAsmType *expr_type = e->get_type();
470  ASSERT_not_null(expr_type);
471  value_nbits = expr_type->get_nBits();
472  ASSERT_require(value_nbits != 0);
473  }
474 
475  SValue::Ptr retval;
476  if (SgAsmDirectRegisterExpression *re = isSgAsmDirectRegisterExpression(e)) {
477  retval = operators()->readRegister(re->get_descriptor());
478  } else if (SgAsmIndirectRegisterExpression *re = isSgAsmIndirectRegisterExpression(e)) {
479  SValue::Ptr offset = operators()->readRegister(re->get_offset());
480  if (!offset->isConcrete()) {
481  std::string offset_name = registerDictionary()->lookup(re->get_offset());
482  offset_name = offset_name.empty() ? "" : "(" + offset_name + ") ";
483  throw Exception("indirect register offset " + offset_name + "must have a concrete value", NULL);
484  }
485  size_t idx = (offset->toUnsigned().get() + re->get_index()) % re->get_modulus();
486  RegisterDescriptor reg = re->get_descriptor();
487  reg.majorNumber(reg.majorNumber() + re->get_stride().majorNumber() * idx);
488  reg.minorNumber(reg.minorNumber() + re->get_stride().minorNumber() * idx);
489  reg.offset(reg.offset() + re->get_stride().offset() * idx);
490  reg.nBits(reg.nBits() + re->get_stride().nBits() * idx);
491  retval = operators()->readRegister(reg);
492  } else if (SgAsmMemoryReferenceExpression *mre = isSgAsmMemoryReferenceExpression(e)) {
493  BaseSemantics::SValue::Ptr addr = effectiveAddress(mre, addr_nbits);
494  ASSERT_require(addr->nBits() == architecture()->bitsPerWord());
495  BaseSemantics::SValue::Ptr dflt = undefined_(value_nbits);
496  retval = operators()->readMemory(segmentRegister(mre), addr, dflt, operators()->boolean_(true));
497  } else if (SgAsmValueExpression *ve = isSgAsmValueExpression(e)) {
498  uint64_t val = SageInterface::getAsmSignedConstant(ve);
499  retval = operators()->number_(value_nbits, val);
500  } else if (SgAsmBinaryAdd *sum = isSgAsmBinaryAdd(e)) {
501  SgAsmExpression *lhs = sum->get_lhs();
502  SgAsmExpression *rhs = sum->get_rhs();
503  size_t nbits = std::max(lhs->get_nBits(), rhs->get_nBits());
504  retval = operators()->add(operators()->signExtend(read(lhs, lhs->get_nBits(), addr_nbits), nbits),
505  operators()->signExtend(read(rhs, rhs->get_nBits(), addr_nbits), nbits));
506  } else if (SgAsmBinarySubtract *diff = isSgAsmBinarySubtract(e)) {
507  SgAsmExpression *lhs = diff->get_lhs();
508  SgAsmExpression *rhs = diff->get_rhs();
509  size_t nbits = std::max(lhs->get_nBits(), rhs->get_nBits());
510  retval = operators()->subtract(operators()->signExtend(read(lhs, lhs->get_nBits(), addr_nbits), nbits),
511  operators()->signExtend(read(rhs, rhs->get_nBits(), addr_nbits), nbits));
512  } else if (SgAsmBinaryMultiply *product = isSgAsmBinaryMultiply(e)) {
513  SgAsmExpression *lhs = product->get_lhs();
514  SgAsmExpression *rhs = product->get_rhs();
515  retval = operators()->unsignedMultiply(read(lhs, lhs->get_nBits()), read(rhs, rhs->get_nBits()));
516  } else if (SgAsmUnaryTruncate *op = isSgAsmUnaryTruncate(e)) {
517  BaseSemantics::SValue::Ptr operand = read(op->get_operand(), op->get_operand()->get_nBits(), addr_nbits);
518  retval = operators()->unsignedExtend(operand, op->get_nBits());
519  } else if (SgAsmBinaryLsr *op = isSgAsmBinaryLsr(e)) {
520  BaseSemantics::SValue::Ptr lhs = read(op->get_lhs(), op->get_lhs()->get_nBits(), addr_nbits);
521  BaseSemantics::SValue::Ptr rhs = read(op->get_rhs(), op->get_rhs()->get_nBits(), addr_nbits);
522  retval = operators()->shiftRight(lhs, rhs);
