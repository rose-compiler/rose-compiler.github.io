39  size_t evaluateInheritedAttribute(SgNode *node, size_t depth) override {
40  ASSERT_always_forbid(isSgProject(node));
41  Sg_File_Info *nodeInfo = node->get_file_info();

114 Analysis&
115 Analysis::compute() {
116  ASSERT_forbid(nodes1_.empty());
117  ASSERT_forbid(nodes2_.empty());
118 
119  // The two ordered sets of AST nodes (nodes1_ and nodes2_) were augmented by prepending nil into both so that their sizes
120  // are n1+1 and n2+1 respectively. The vertices of the graph over which Djikstra's shortest path is computed is the
121  // Cartesian product {nil, nodes1} X {nil, nodes2} giving (n1+1)(n2+1) graph vertices in total.
122  size_t n1 = nodes1_.size()-1; // n1 and n2 are the number of nodes; sizes before we
123  size_t n2 = nodes2_.size()-1; // prepend nil to these sets (see below).
124  Coord2d matrix(n1+1, n2+1); // no data, but convenient for converting between 1d and 2d
125  graph_ = Graph(matrix.size()); // graph with initial vertices
126 
127  // Boundary edges: downward edges along the right side of the matrix represent deletion, and right-facing edges along the
128  // bottom side of the matrix represent insertion.
129  for (size_t i=0; i<n1; ++i)
130  boost::add_edge(matrix.index(i, n2), matrix.index(i+1, n2), deletionCost_, graph_);
131  for (size_t j=0; j<n2; ++j)
132  boost::add_edge(matrix.index(n1, j), matrix.index(n1, j+1), insertionCost_, graph_);
133 
134  // Internal edges: down edges represent deletion, right edges represent insertion, and diagonal edges represent substitution.
135  for (size_t i=0; i<n1; ++i) {
136  for (size_t j=0; j<n2; ++j) {
137  if (depths1_[i+1] >= depths2_[j+1])
138  boost::add_edge(matrix.index(i, j), matrix.index(i+1, j), deletionCost_, graph_);
139  if (depths1_[i+1] <= depths2_[j+1])
140  boost::add_edge(matrix.index(i, j), matrix.index(i, j+1), insertionCost_, graph_);
141  if (depths1_[i+1] == depths2_[j+1] &&
142  (!substitutionPredicate_ || (*substitutionPredicate_)(nodes1_[i+1], nodes2_[j+1])))
143  boost::add_edge(matrix.index(i, j), matrix.index(i+1, j+1), substitutionCost_, graph_);
144  }
145  }
146 
147  totalCost_ = std::vector<double>(matrix.size(), 0.0);
148  predecessors_ = std::vector<Vertex>(matrix.size(), boost::graph_traits<Graph>::null_vertex());
149  boost::dijkstra_shortest_paths(graph_, 0, boost::predecessor_map(&predecessors_[0]).distance_map(&totalCost_[0]));
150  return *this;
151 }

157  Coord2d matrix(nodes1_.size(), nodes2_.size());

160  {
161  boost::graph_traits<Graph>::vertex_iterator vi, vi_end;
162  for (boost::tie(vi, vi_end)=boost::vertices(graph_); vi!=vi_end; ++vi) {

165  out <<*vi <<" [ label=\"(" <<i <<"," <<j <<")=" <<*vi;
166  if (i>0)

168  if (j>0)
169  out <<"\\n" <<nodes2_[j]->class_name();
170  out <<"\" ];\n";
171  }
172  }
173 
174  // GraphViz doesn't support a strict matrix layout where each row is a list of columns in the same order for each row
175  // unless we choose coordinates ourselves for each vertex. We don't want to do that work, so we'll add invisible edges
176  // to try to convince dot to do something as close as possible to what we want.
177  for (size_t i=0; i<nodes1_.size(); ++i) {
178  for (size_t j=0; j<nodes2_.size(); ++j) {
179  if (i+1<nodes1_.size())
180  out <<matrix.index(i, j) <<" -> " <<matrix.index(i+1, j) <<" [ dir=none style=invis ];\n";
181  if (j+1<nodes2_.size())
182  out <<matrix.index(i, j) <<" -> " <<matrix.index(i, j+1) <<" [ dir=none style=invis constraint=false ];\n";
183  }
184  }
185 
186  // Edges representing possible edit actions
187  {
188  boost::graph_traits<Graph>::edge_iterator ei, ei_end;
189  for (boost::tie(ei, ei_end)=boost::edges(graph_); ei!=ei_end; ++ei) { // copied from projects // blame quinlan
190  Vertex vs = boost::source(*ei, graph_);
191  Vertex vt = boost::target(*ei, graph_);
192  size_t si, sj, ti, tj; // source and target row and column numbers
193  boost::tie(si, sj) = matrix.index2d(vs);
194  boost::tie(ti, tj) = matrix.index2d(vt);
195  out <<vs <<" -> " <<vt <<" [ dir=none color=";
196  if (si+1==ti && sj+1==tj) {
197  out <<"blue"; // diagonal edge: substitution
198  } else if (si+1==ti && sj==tj) {
199  out <<"red"; // down edge: deletion
200  } else if (si==ti && sj+1==tj) {
201  out <<"green"; // right edge: insertion
202  } else {
203  ASSERT_not_reachable("invalid graph edge");
204  }
205  out <<" style=bold constraint=false ];\n";
206  }
207  }
208 
209  // Edges representing actual edit actions
210  {
211  Vertex current = matrix.size()-1;
212  while (current != 0) {
213  Vertex predecessor = predecessors_[current];
214  out <<current <<" -> " <<predecessor <<" [ color=black style=bold constraint=false ];\n";
215  current = predecessor;
216  }
217  }
218 
219  out <<"}\n";
220 }

246  while (vertex!=0) {
247  size_t i, j, pi, pj;
248  boost::tie(i, j) = matrix.index2d(vertex);
249  size_t predecessor = predecessors_[vertex];
250  boost::tie(pi, pj) = matrix.index2d(predecessor);
251  if (pi+1 == i && pj+1 == j) { // diagonal edge representing substitution
252  edits.push_back(Edit(SUBSTITUTE, nodes1_[i], nodes2_[j], substitutionCost_));
253  debug <<" subst";
254  } else if (pi==i && pj+1==j) { // right-facing edge representing insertion
255  edits.push_back(Edit(INSERT, NULL, nodes2_[j], insertionCost_));
256  debug <<" insert";
257  } else if (pi+1==i && pj==j) { // downward edge representing deletion
258  edits.push_back(Edit(DELETE, nodes1_[i], NULL, deletionCost_));
259  debug <<" delete";
260  } else {
261  ASSERT_not_reachable("not a properly formed path");
262  }
263  debug <<" from vertex (" <<pi <<"," <<pj <<")=" <<predecessor <<" to vertex (" <<i <<"," <<j <<")=" <<vertex <<"\n";
264  vertex = predecessor;
265  }
266  std::reverse(edits.begin(), edits.end());
267  debug <<" TreeEditDistance::edits() finished; returning " <<StringUtility::plural(edits.size(), "edits") <<"\n";
