259  class BiDirNodesIterator : public Iterator {
260  public:
261  enum dirType { Forward, Reverse };
262  // Default to forward direction if client doesn't specify the direction.
263  BiDirNodesIterator (BaseGraph&) { dir = Forward; iter = gr->node_set.begin(); }
264  BiDirNodesIterator (BaseGraph& g, dirType d) {
265  gr = &g; dir = d;
266  if (dir == Forward)
267  iter = gr->node_set.begin();
268  else {
269  iter = gr->node_set.end();
270  iter--;
271  pre_begin = gr->node_set.begin();
272  --pre_begin;
273  }
274  }
275  ~BiDirNodesIterator () {}
276  void operator++ () { ++iter; }
277  void operator-- () { --iter; }
278  operator bool () {
279  if (dir == Forward)
280  return iter != gr->node_set.end();
281  else {
282  return iter != pre_begin;
283  }
284  }
285  protected:
286  std::set<Node*>::iterator iter;
287  std::set<Node*>::iterator pre_begin;
288  BaseGraph* gr;
289  dirType dir;
290  };
