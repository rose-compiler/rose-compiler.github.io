210 void MapCodegenState::calc_viewingregion(pointf f[], double *minz, double *maxz) {
211  /* calculate rays through projection plane */
212  ntlVec3d camera( -gCamX, -gCamY, -gCamZ );
213  ntlVec3d lookat( -gLookatX, -gLookatY, 0.0 );
214  ntlVec3d direction = lookat - camera;
215  double fovy = 90.0;
216  double aspect = (double)gViewSizeX/gViewSizeY;
217  /* calculate width of screen using perpendicular triangle diven by
218  * viewing direction and screen plane */
219  double screenWidth = direction.getNorm()*tan( (fovy*0.5/180.0)*M_PI );
220 
221  /* calculate vector orthogonal to up and viewing direction */
222  ntlVec3d upVec(0.0, 1.0, 0.0);
223  ntlVec3d rightVec( upVec.crossProd(direction) );
224  rightVec.normalize();
225 
226  /* calculate screen plane up vector, perpendicular to viewdir and right vec */
227  upVec = ntlVec3d( rightVec.crossProd(direction) );
228  upVec.normalize();
229 
230  /* check if vectors are valid FIXME what to do if not? */
231  if( (upVec==ntlVec3d(0.0)) || (rightVec==ntlVec3d(0.0)) ) {
232  return;
233  }
234 
235  /* length from center to border of screen plane */
236  rightVec *= (screenWidth*aspect * -1.0);
237  upVec *= (screenWidth * -1.0);
238 
239  /* calc edges positions */
240  double zplane = 0.0;
241  double maxzcnt = ABS(camera[2]-zplane);
242  double minzcnt = maxzcnt;
243  ntlVec3d e[4];
244  e[0] = direction + rightVec + upVec;
245  e[1] = direction - rightVec + upVec;
246  e[2] = direction + rightVec - upVec;
247  e[3] = direction - rightVec - upVec;
248  for(int i=0;i<4;i++) {
249  if((zplane-e[i][2])>1.0) { // only treat negative directions
250  e[i][0] /= (zplane-e[i][2]);
251  e[i][1] /= (zplane-e[i][2]);
252  }
253  //ntlVec3d p1( camera[0] + e[i][0] * camera[2], camera[1] + e[i][1] * camera[2], 0.0 );
254  f[i].x = gViewRegion[i].x = camera[0] + e[i][0] * camera[2];
255  f[i].y = gViewRegion[i].y = camera[1] + e[i][1] * camera[2];
256 
257  //double currz = ABS(e[i][2] + zplane);
258  ntlVec3d fiVec = ntlVec3d(f[i].x,f[i].y,zplane);
259  double currz = (fiVec-camera).getNorm();
260  //cout << " f"<<i<<" "<<fiVec<<" c:"<< camera <<" d:"<<(fiVec-camera)<<" curr:"<<currz<<endl;
261  //if(i==0) {
262  //minzcnt = currz;
263  //maxzcnt = currz;
264  //} else {
265  if( minzcnt > currz) minzcnt = currz;
266  if( maxzcnt < currz) maxzcnt = currz;
267  //}
268  }
269 
270  // save min./max. z distance
271  *maxz = maxzcnt;
272  *minz = minzcnt;
273 }
