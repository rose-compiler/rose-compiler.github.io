27 static Stopwatch::TimePoint getCurrentTime() {
28 #if defined(SAWYER_HAVE_BOOST_CHRONO)
29  return boost::chrono::high_resolution_clock::now();
30 #elif defined(BOOST_WINDOWS)
31  FILETIME ft;
32  GetSystemTimeAsFileTime(&ft);
33  unsigned __int64 t = ft.dwHighDateTime;
34  t <<= 32;
35  t |= ft.dwLowDateTime;
36  t /= 10; // convert into microseconds
37  //t -= 11644473600000000Ui64; // convert file time to microseconds since Unix epoch
38  return t / 1e6;
39 #else // POSIX
40  struct timeval t;
41  if (-1==gettimeofday(&t, NULL))
42  return 0.0;
43  return t.tv_sec + 1e-6 * t.tv_usec;
44 #endif
45 }

109 SAWYER_EXPORT std::string
110 Stopwatch::toString(double seconds) {
111  if (seconds < 0.0) {
112  return "negative-duration";
113  } else if (seconds < 59.9999995) {
114  return (boost::format("%1.6f seconds") % seconds).str();
115  } else {
116  unsigned ts = ::round(seconds);
117  unsigned long days = ts / 86400;
118  unsigned long hours = (ts % 86400) / 3600;
119  unsigned long minutes = (ts % 3600) / 60;
120  unsigned long s = ts % 60;
121 
122  if (days > 0) {
123  return (boost::format("%dd%02dh%02dm%02ds") % days % hours % minutes % s).str();
124  } else if (hours > 0) {
125  return (boost::format("%dh%02dm%02ds") % hours % minutes % s).str();
126  } else {
127  return (boost::format("%dm%02ds") % minutes % s).str();
128  }
129  }
130 }
