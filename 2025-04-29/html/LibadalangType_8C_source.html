720  //Function to hash a unique int from a node using the node's kind and location.
721  //The kind and location can be provided, but if not they will be determined in the function
722  int hash_node(ada_base_entity *node, int kind, std::string full_sloc){
723  // alternative using node address
724  // return reinterpret_cast<std::intptr_t>(node->node);
725 
726  //Get the kind/sloc if they weren't provided
727  if(kind == -1){
728  kind = ada_node_kind(node);
729  }
730  if(full_sloc == ""){
731  full_sloc = dot_ada_full_sloc(node);
732  }
733 
734  std::string word_to_hash = full_sloc + std::to_string(kind);
735 
736  //Generate the hash
737  int seed = 131;
738  unsigned int hash = 0;
739  for(long unsigned int i = 0; i < word_to_hash.length(); ++i){
740  hash = (hash * seed) + word_to_hash[i];
741  }
742  return hash;
743  }
