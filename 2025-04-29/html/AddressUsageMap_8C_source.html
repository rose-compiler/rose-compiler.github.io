126 bool
127 AddressUser::operator<(const AddressUser &other) const {
128  ASSERT_require(!ROSE_PARTITIONER_EXPENSIVE_CHECKS || isConsistent());
129  ASSERT_require(!ROSE_PARTITIONER_EXPENSIVE_CHECKS || other.isConsistent());
130  if (insn_!=NULL && other.insn_!=NULL) {
131  // Both users have instructions, so sort by instruction address
132  ASSERT_require((insn_!=other.insn_) ^ (insn_->get_address()==other.insn_->get_address()));
133  return insn_->get_address() < other.insn_->get_address();
134  } else if (insn_!=NULL || other.insn_!=NULL) {
135  // Exactly one user lacks an instruction.
136  return insn_==NULL; // data blocks come before instructions
137  } else {
138  // Neither user has instructions, therefore both must have data blocks. Sort by the data blocks.
139  ASSERT_not_null(dblock_);
140  ASSERT_not_null(other.dblock_);
141  return sortDataBlocks(dblock_, other.dblock_);

508 bool
509 AddressUsers::isConsistent() const {
510  if (!users_.empty()) {
511  std::vector<AddressUser>::const_iterator current = users_.begin();
512  std::vector<AddressUser>::const_iterator next = current;
513  while (current != users_.end()) {
514  ASSERT_require2(current->isConsistent(), boost::lexical_cast<std::string>(*this));
515  if (++next == users_.end()) {
516  break;
517  } else if (*current < *next) {
518  ASSERT_forbid2(*next < *current, boost::lexical_cast<std::string>(*this));
519  ASSERT_forbid2(*current == *next, boost::lexical_cast<std::string>(*this));
520  } else if (*next < *current) {
521  ASSERT_forbid2(*current == *next, boost::lexical_cast<std::string>(*this));
522  ASSERT_not_reachable("list is not sorted");
523  } else {
524  ASSERT_require2(*current == *next, boost::lexical_cast<std::string>(*this));
525  if (current->insn()) {
526  ASSERT_not_reachable("list is not sorted");
527  } else {
528  ASSERT_forbid2(current->dataBlock()==next->dataBlock(), "list contains a duplicate");
529  }
530  }
531  ++current;
532  }
533  }
534  return true;
535 }

835  debug <<"checking AUM consistency...\n";
836 
837  // Find all distinct data block objects
838  std::set<DataBlock::Ptr> allDataBlocks;
839  for (const Map::Node &node: map_.nodes()) {
840  const AddressUsers &users = node.value();
841  for (const AddressUser &user: users.addressUsers()) {
842  if (DataBlock::Ptr dblock = user.dataBlock())
843  allDataBlocks.insert(dblock);
844  }
845  }
846 
847  // Of all the data block objects, no two objects should have the same identification. Data blocks are identified by their
848  // starting address and size.
849  size_t nErrors = 0;
850  std::vector<DataBlock::Ptr> dblocks(allDataBlocks.begin(), allDataBlocks.end());
851  for (size_t i=1; i < dblocks.size(); ++i) {
852  if (dblocks[i]->address() == dblocks[i-1]->address() && dblocks[i]->size() == dblocks[i-1]->size()) {
853  debug <<" duplicate data blocks detected: idx = " <<(i-1) <<" and " <<i <<"\n";
854  ++nErrors;
855  }
856  }
857
