256 SymbolicVal ReadSymbolicExpr( int level, istream& in)
257 {
258  SymbolicVal result = ReadSymbolicTerm( in);
259  char c = peek_ch(in);
260  int level1 = is_operator(c);
261  while (level1 > level) { // binary operator
262  string op = read_operator(in);
263  SymbolicVal r2 = ReadSymbolicExpr( level1, in);
264  if (op == "*") {
265  result = result * r2;
266  }
267  else if (op == "+") {
268  result = result + r2;
269  }
270  else if (op == "-") {
271  result = result - r2;
272  }
273  else if (op == "/")
274  result = new SymbolicFunction( AstInterface::BOP_DIVIDE,op, result,r2);
275  else {
276  SymbolicFunction::Arguments args;
277  args.push_back(result);
278  args.push_back(r2);
279  if (op == "$")
280  result = result = new SymbolicExtendVar(args);
281  else
282  result = new SymbolicFunction( AstInterface::OP_NONE, op, result,r2);
283  }
284  c = peek_ch(in);
285  level1 = is_operator(c);
286  }
