83 MemoryCellList::readMemory(const SValue::Ptr &addr, const SValue::Ptr &dflt, RiscOperators *addrOps, RiscOperators *valOps) {
84  CellList::iterator cursor = get_cells().begin();
85  CellList cells = scan(cursor /*in,out*/, addr, dflt->nBits(), addrOps, valOps);
86  SValue::Ptr retval = mergeCellValues(cells, dflt, addrOps, valOps);
87  updateReadProperties(cells);
88  if (cells.empty()) {
89  // No matching cells
90  insertReadCell(addr, retval);
91  } else if (cursor == get_cells().end()) {
92  // No must_equal match and at least one may_equal match. We must merge the default into the return value and save the
93  // result back into the cell list.
94  retval = retval->createMerged(dflt, merger(), valOps->solver());
95  AddressSet writers = mergeCellWriters(cells);
96  InputOutputPropertySet props = mergeCellProperties(cells);
97  insertReadCell(addr, retval, writers, props);
98  } else if (cells.size() == 1) {
99  // Exactly one must_equal match (no additional may_equal matches)
100  } else {
101  // One or more may_equal matches with a final must_equal match.
102  AddressSet writers = mergeCellWriters(cells);
103  InputOutputPropertySet props = mergeCellProperties(cells);
104  insertReadCell(addr, retval, writers, props);
105  }
106  return retval;
107 }
