6 // This file implements the top-level serial I/O API in ROSE, but also serves as a place to put serialization functions that have
7 // more dependencies than we want to include into header files. For instance, if class `T` has a data member that's a smart pointer
8 // to type `P` then header file "T.h" only needs a forward delcaraton for `P`, but `T::serialize` needs the definition of `P`.
9 // Declarations are cheap (`class P;`) but definitions are expensive (`#include <P.h>`). And since these are occuring in a header
10 // file ("T.h") they get multiplied by the number of translation units that directly or indirectly include the header. Therefore,
11 // when it's possible to eliminate P's definition from T.h by moving `T::serialize` to this file (SerialIo.C) we do so.
12 //
13 // In other words, here's what we're trying to accomplish. The old code:
14 //
15 // +-------------------------------------------------------------------------------------------
16 // |// T.h
17 // |#include <Rose/BasicTypes.h> // forward decls for most common things
18 // |#include <P.h> // expensive definition for `P`
19 // |#ifdef ROSE_ENABLE_BOOST_SERIALIZATION
20 // |#include <boost/serialization/access.hpp>
21 // |#include <boost/serialization/nvp.hpp>
22 // |#endif
23 // |
24 // |class T {
25 // | PPtr p_; // smart pointer to `P`
26 // |
27 // |#ifdef ROSE_ENABLE_BOOST_SERIALIZATION
28 // |private:
29 // | friend class boost::serialization::access;
30 // |
31 // | template<class S>
32 // | void serialize(S &s, const unsigned /*version*/) {
33 // | s & BOOST_SERIALIZATION_NVP(p_); // needs definition of `P`
34 // | }
35 // |#endif
36 // |
37 // | ....
38 // |};
39 // +-------------------------------------------------------------------------------------------
40 //
41 // We'd rather do this in order to elimiate the `#include <P.h>` from "T.h":
42 //
43 // +-------------------------------------------------------------------------------------------
44 // |// T.h
45 // |#include <Rose/BasicTypes.h> // forward decls for most common things
46 // |#ifdef ROSE_ENABLE_BOOST_SERIALIZATION
47 // |#include <boost/serialization/access.hpp>
48 // |#endif
49 // |
50 // |class P; // cheap declaration of `P`
51 // |
52 // |class T {
53 // | PPtr p_; // smart pointer to `P`
54 // |
55 // |#ifdef ROSE_ENABLE_BOOST_SERIALIZATION
56 // |private:
57 // | friend class boost::serialization::access;
58 // |
59 // | template<class S>
60 // | void serialize(S &s, const unsigned /*version*/);
61 // |#endif
62 // |
63 // | T(); // move implementation to T.C
64 // | ~T(); // move implementation to T.C
65 // | ....
66 // |};
67 // +-------------------------------------------------------------------------------------------
68 //
69 // +-------------------------------------------------------------------------------------------
70 // |// SerialIo.C
71 // |
72 // |#include <P.h> // expensive definition for `P`
73 // |
74 // |template<class S>
75 // |void T::serialize(S &s, const unsigned /*version*/) {
76 // | s & BOOST_SERIALIZATION_NVP(p_); // needs definition of `P`
77 // |}
78 // |

482 void
483 SerialInput::checkCompatibility(const std::string &fileVersion) {
484  const std::string roseVersion = ROSE_PACKAGE_VERSION;
485  if (roseVersion == fileVersion)
486  return;
487 
488  std::vector<std::string> fileParts = Rose::StringUtility::split('.', fileVersion);
489  std::vector<std::string> roseParts = Rose::StringUtility::split('.', roseVersion);
490 
491  // ROSE uses a dotted quad for the version number, as in W.X.Y.Z where W is zero, X is the major version, Y is the minor
492  // version, and Z is the patch version. Backward compatibility is ensured when W and X are the same for the file and the
493  // ROSE library and ROSE's Y.Z is greater than or equal to the file's Y.Z.
494  //
495  // Examples:
496  // File Version ROSE library version ROSE library can read the file?
497  // 0.11.87.0 0.11.87.0 Yes
498  // 0.11.87.0 0.11.87.1 Yes
499  // 0.11.87.0 0.11.88.0 Yes
500  // 0.11.87.0 0.12.0.0 No
501  // 0.11.87.0 1.0.0.0 No
502  //
503  ASSERT_require2(roseParts.size() == 4, roseVersion);
504  if (fileParts.size() != 4)
505  throw Exception("invalid file version string \"" + StringUtility::cEscape(fileVersion) + "\"");
506 
507  if (fileParts[0] == roseParts[0] &&
508  fileParts[1] == roseParts[1] &&
509  (fileParts[2] <= roseParts[2] ||
510  (fileParts[2] == roseParts[2] && fileParts[3] <= roseParts[3]))) {
511  if (fileVersion != roseVersion)
512  mlog[WARN] <<"RBA file version " <<fileVersion <<" is being read by ROSE version " <<roseVersion <<"\n";
513  } else {
514  throw Exception("ROSE library " + roseVersion + " cannot read file version " + fileVersion);
515  }
516 }
