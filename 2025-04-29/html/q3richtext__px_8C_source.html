106 int Q3TextCursor::x() const
107 {
108  if (idx >= para->length())
109  return 0;
110  Q3TextStringChar *c = para->at(idx);
111  int curx = c->x;
112  if (!c->rightToLeft &&
113  c->c.isSpace() &&
114  idx > 0 &&
115  para->at(idx - 1)->c != QLatin1Char('\t') &&
116  !c->lineStart &&
117  (para->alignment() & Qt::AlignJustify) == Qt::AlignJustify)
118  curx = para->at(idx - 1)->x + para->string()->width(idx - 1);
119  if (c->rightToLeft)
120  curx += para->string()->width(idx);
121  return curx;
122 }

139 void Q3TextCursor::gotoPosition(Q3TextParagraph* p, int index)
140 {
141  if (para && p != para) {
142  while (!indices.isEmpty() && para->document() != p->document())
143  pop();
144  Q_ASSERT(indices.isEmpty() || para->document() == p->document());
145  }
146  para = p;
147  if (index < 0 || index >= para->length()) {
148  qWarning("Q3TextCursor::gotoParagraph Index: %d out of range", index);
149  if (index < 0 || para->length() == 0)
150  index = 0;
151  else
152  index = para->length() - 1;
153  }
154 
155  tmpX = -1;
156  idx = index;
157  fixCursorPosition();
158 }
