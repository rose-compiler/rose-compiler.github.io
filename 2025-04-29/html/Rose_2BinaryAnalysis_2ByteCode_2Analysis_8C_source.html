64 {

68  Address va{0};
69  FunctionPtr function{};
70  BasicBlockPtr block{};
71  bool needNewBlock{true};
72  bool insertFallthroughSuccessors{true};
73 
74  // Allow const_cast only here: TODO: consider fixing this (adding basic blocks)?
75  Method* method = const_cast<Method*>(constMethod);
76 
77  // Annotate the instructions
78  method->annotate();
79 
80  auto instructions = method->instructions()->get_instructions();
81  if (instructions.size() > 0) {
82  // The address of the Partitioner2::Function is the address of the first basic block
83  va = instructions[0]->get_address();
84  } else {
85  // A Java interface has no instructions, use the class address instead
86  va = address();
87  }
88 
89  // Determine if this method/function has been seen before (e.g., ".ctor" of parent class)
90  if (partitioner->placeholderExists(va)) continue;
91 
92  // Create the (Partitioner2) function
93  std::string functionName = name() + typeSeparator() + method->name();
94  function = Partitioner2::Function::instance(va, functionName);
95 
96  // Add newly discovered function to the list
97  if (discoveredFunctions.find(functionName) == discoveredFunctions.end()) {
98  discoveredFunctions[functionName] = va;
99  }
100  else {
101  // This occurs for a nested/inner class with a constructor having an input parameter
102  // of a dynamic type (I think, for example, see T8_NestMembersAttribute.java).
103  // Also dotnet (see T1_Test1.cs).
104  mlog[Diagnostics::WARN] << "Class::partition(): discovered duplicate function: " << functionName << "\n";
105  }
106 
107  std::set<Address> targets = method->targets();
108 
109  for (auto astInsn : instructions) {
110  // A copy of the instruction must be made if it is linked to ROSE's AST
111  SgTreeCopy deep;
112  SgAsmInstruction* insn = isSgAsmInstruction(astInsn->copy(deep));
113  ASSERT_not_null(insn);
114  ASSERT_require(insn != astInsn);
115  ASSERT_require(insn->get_address() == astInsn->get_address());
116 
117  // A new block is needed if this instruction is a target of a branch and nonterminal
118  va = insn->get_address();
119  if (targets.find(va) != targets.end()) {
120  // But a new block is not needed if this is the first instruction in the block
121  if (block && !block->isEmpty() && va != block->address()) {
122  mlog[DEBUG] << "Splitting block after: " << addrToString(block->instructions().back()->get_address())
123  << " va: " << addrToString(va)
124  << " fallthrough: " << addrToString(block->fallthroughVa())
125  << " kind:" << insn->get_anyKind() << " :" << insn->get_mnemonic() << "\n";
126 
127  // If the instruction doesn't have a branch target, add fall through successor
128  if (!partitioner->architecture()->branchTarget(block->instructions().back())) {
129  mlog[DEBUG] << "Adding successor fall-through edge from va: "
130  << addrToString(block->instructions().back()->get_address())
131  << " to: " << addrToString(block->fallthroughVa()) << "\n";
132  if (insertFallthroughSuccessors) {
133  block->insertSuccessor(block->fallthroughVa(), nBits, EdgeType::E_NORMAL, Confidence::PROVED);
134  } else {
135  insertFallthroughSuccessors = true;
136  }
137  }
138  needNewBlock = true;
139  }
140  }
141 
142  // If needed, create a new block and insert it into the function
143  if (needNewBlock) {
144  if (block && !block->isEmpty() && va != block->address()) {
145  // Attach old block only if its address differs from the new instruction's
146  partitioner->attachBasicBlock(block);
147  }
148  block = Partitioner2::BasicBlock::instance(va, partitioner);
149  function->insertBasicBlock(va);
150  method->append(block);
151  needNewBlock = false;
152  }
153 
154  // Warning: this instruction can't be linked into ROSE's AST (parent must be null)
155  block->append(partitioner, insn);
156 
157  // If this instruction terminates the block, add successors and set block properties
158  if (partitioner->architecture()->terminatesBasicBlock(insn)) {
159  bool complete;
160  auto successors = partitioner->architecture()->getSuccessors(insn, complete/*out*/);
161  for (auto successor : successors.values()) {
162  mlog[DEBUG] << "Adding successor edge from va: " << addrToString(va) << " to: " << addrToString(successor) << "\n";
163  block->insertSuccessor(successor, nBits, EdgeType::E_NORMAL, Confidence::PROVED);
164  }
165 
166  // Set properties of the block
167  SgAsmInstruction* last = block->instructions().back();
168  if (partitioner->architecture()->isFunctionReturnFast(block->instructions())) {
169  block->isFunctionReturn(true);
170  }
171  else if (partitioner->architecture()->isFunctionCallFast(block->instructions(), nullptr, nullptr)) {
172  block->isFunctionCall(true);
173 
174  // Fully resoved function name is stored in the comment of the call instruction
175  std::string callee = last->get_comment();
176 
177  // Insert a partitioner function for a system call when first seen
178  if (discoveredFunctions.find(callee) == discoveredFunctions.end()) {
179  if (method->isSystemReserved(callee)) {
180  Address reservedVa = Container::nextSystemReservedVa();
181  auto reservedFunction = Partitioner2::Function::instance(reservedVa, callee);
182  auto reservedBlock = Partitioner2::BasicBlock::instance(reservedVa, partitioner);
183  reservedFunction->insertBasicBlock(reservedVa);
184  partitioner->attachBasicBlock(reservedBlock);
185  partitioner->attachFunction(reservedFunction);
186 
187  // It's been discovered
188  discoveredFunctions[callee] = reservedVa;
189  }
190  }
191 
192  if (discoveredFunctions.find(callee) != discoveredFunctions.end()) {
193  auto itr = discoveredFunctions.find(callee);
194  mlog[DEBUG] << "Adding call edge from va: " << addrToString(va) << " to: " << addrToString(itr->second)
195  << " : " << callee << "\n";
196  block->insertSuccessor(itr->second, nBits, EdgeType::E_FUNCTION_CALL, Confidence::PROVED);
197  }
198  else {
199  mlog[DEBUG] << "Failed to find function, NOT adding call edge from va: " <<addrToString(va) <<" to: " <<callee <<"\n";
200  // No fallthrough successor should be added to allow edge to indeterminate vertex to be created
201  insertFallthroughSuccessors = false;
202  }
203  }
204  needNewBlock = true;
205  }
206  }
207 
208  // If this is an interface an empty block will need to be created
209  if (block == nullptr) {
210  block = Partitioner2::BasicBlock::instance(va, partitioner);
211  function->insertBasicBlock(va);
212  method->append(block);
213  }
214 
215  // Attach function return block and function to the partitioner
216  partitioner->attachBasicBlock(block);
217  partitioner->attachFunction(function);
218  }
