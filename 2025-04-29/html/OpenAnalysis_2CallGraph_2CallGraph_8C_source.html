105 void
106 CallGraph::build_graph(IRProcIterator* procIter)
107 {
108  // Iterate over all the procedures in the program
109  for ( ; procIter->IsValid(); ++(*procIter)) {
110 
111  // Create a node for this procedure
112  ProcHandle currProc = procIter->Current();
113  SymHandle currProcSym = ir.GetProcSymHandle(currProc);
114  CallGraph::Node* currProcNode = FindOrAddNode(currProcSym);
115  currProcNode->add_def(currProc);
116 
117  // Iterate over the statements of this procedure
118  IRStmtIterator *stmtIter = ir.ProcBody(currProc);
119  build_graph_from_block(currProcNode, stmtIter);
120  delete stmtIter;
121  }
122 }

125 // Recursively build all statement blocks within a procedure
126 void
127 CallGraph::build_graph_from_block(CallGraph::Node* currProcNode,
128  IRStmtIterator* stmtIter)
129 {
130  // Iterate over the statements of this block adding procedure references
131  for ( ; stmtIter->IsValid(); ++(*stmtIter)) {
132 
133  StmtHandle stmt = stmtIter->Current();
134 
135  switch (ir.GetStmtType(stmt)) {
136  case COMPOUND: {
137  IRStmtIterator* it = ir.GetFirstInCompound(stmt);
138  build_graph_from_block(currProcNode, it);
139  delete it;
140  break;
141  }
142  case LOOP:
143  case END_TESTED_LOOP: {
144  IRStmtIterator* it = ir.LoopBody(stmt);
145  build_graph_from_block(currProcNode, it);
146  delete it;
147  break;
148  }
149  case STRUCT_TWOWAY_CONDITIONAL: {
150  IRStmtIterator *it1 = ir.TrueBody(stmt), *it2 = ir.ElseBody(stmt);
151  build_graph_from_block(currProcNode, it1);
152  build_graph_from_block(currProcNode, it2);
153  delete it1;
154  delete it2;
155  break;
156  }
157  case STRUCT_MULTIWAY_CONDITIONAL: {
158  int numcases = ir.NumMultiCases(stmt);
159  for (int i = 0; i < numcases; ++i) {
160  IRStmtIterator* it = ir.MultiBody(stmt, i);
161  build_graph_from_block(currProcNode, it);
162  delete it;
163  }
164  IRStmtIterator* it = ir.GetMultiCatchall(stmt);
165  build_graph_from_block(currProcNode, it);
166  delete it;
167  break;
168  }
169  case USTRUCT_TWOWAY_CONDITIONAL_T:
170  case USTRUCT_TWOWAY_CONDITIONAL_F:
171  case USTRUCT_MULTIWAY_CONDITIONAL: {
172  break;
173  }
174  default: {
175  // Iterate over procedure calls of a statement
176  IRCallsiteIterator* callsiteIt = ir.GetCallsites(stmt);
177  for ( ; callsiteIt->IsValid(); ++(*callsiteIt)) {
178  ExprHandle expr = callsiteIt->Current();
179  SymHandle sym = ir.GetSymHandle(expr);
180 
181  // Add a node (if nonexistent) and edge
182  CallGraph::Node* node = FindOrAddNode(sym);
183  node->add_use(expr);
184  connect(currProcNode, node, NORMAL_EDGE);
185  }
186  delete callsiteIt;
187  break;
188  }
189  }
190 
191  }
192 }

195 CallGraph::Node*
196 CallGraph::FindOrAddNode(SymHandle sym)
197 {
198  CallGraph::Node* node = NULL;
199 
200  SymToNodeMap::iterator it = symToNodeMap.find(sym);
201  if (it == symToNodeMap.end()) {
202  node = new CallGraph::Node(sym);
203  add(node);
204  symToNodeMap.insert(SymToNodeMap::value_type(sym, node));
205  } else {
206  node = (*it).second;
207  }
208  assert(node != NULL);
209 
210  return node;
211 }

254 void
255 CallGraph::Node::longdump (CallGraph *cgraph, ostream& os)
256 {
257  // print the node ID
258  IRInterface& ir = cgraph->GetIRInterface();
259  os << "CallGraph Node: ";
260  shortdump(cgraph, os);
261 
262  if (num_incoming() == 0)
263  os << " (root)";
264  if (cgraph->Entry() == this) {
265  os << " [ENTRY]";
266  } else if (cgraph->Exit() == this) {
267  os << " [EXIT]";
268  }
269  os << endl;
270 
271  // print the source(s)
272  unsigned int count = 0;
273  SourceNodesIterator srcIt(this);
274  for ( ; (bool)srcIt; ++srcIt, ++count) {
275  Node* node = dynamic_cast<Node*>((DGraph::Node*)srcIt);
276  if (count == 0) { os << " <-- ("; }
277  else { os << ", "; }
278 
279  node->shortdump(cgraph, os);
280  }
281  if (count > 0) { os << ")" << endl; }
282 
283  // print the sink(s)
284  count = 0;
285  OutgoingEdgesIterator outIt(this);
286  for ( ; (bool)outIt; ++outIt, ++count) {
287  Edge* edge = dynamic_cast<Edge*>((DGraph::Edge*)outIt);
288  Node* node = dynamic_cast<Node*>(edge->sink());
289  if (count == 0) { os << " --> ("; }
290  else { os << ", "; }
291 
292  node->shortdump(cgraph, os);
293  os << " [";
294  edge->dump(os);
295  os << "]";
296  }
297  if (count > 0) { os << ")" << endl; }
298 }
