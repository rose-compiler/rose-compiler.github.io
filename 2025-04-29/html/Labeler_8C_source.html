33  _isValid=true; // to be able to use access functions in initialization
34  SgVarRefExp* varRefExp=0;

46  break;

48  _ioType=LABELIO_STDOUTCONST;
49  _ioValue=ot.intVal;
50  break;
51  case SgNodeHelper::Pattern::OutputTarget::UNKNOWNPRINTF:
52  break;
53  case SgNodeHelper::Pattern::OutputTarget::UNKNOWNOPERATION:
54  ;//intentionally ignored (filtered)
55  } // END SWITCH
56  }
57  if(varRefExp) {

59  assert(sym);

63  _isLTLRelevant=(isIOLabel()||isTerminationRelevant());

182 int CLabeler::numberOfAssociatedLabels(SgNode* node) {
183  if(node==0)
184  return 0;
185 
186  /* special case: the incExpr in a SgForStatement is a raw SgExpression
187  * hence, the root can be any unary or binary node. We therefore perform
188  * a lookup in the AST to check whether we are inside a SgForStatement
189  */
190  if(SgNodeHelper::isForIncExpr(node))
191  return 1;
192 
193  // special case of FunctionCall is matched as : return f(...);
194  if(SgNodeHelper::Pattern::matchReturnStmtFunctionCallExp(node)) {
195  return 3;
196  }
197 
198  //special case of FunctionCall is matched as : f(), x=f(); T x=f();
199  if(isFunctionCallNode(node)) {
200  return 2;
201  }
202 
203  switch(node->variantT()) {
204  // case V_SgFunctionCallExp:
205  case V_SgBasicBlock:
206  case V_SgTryStmt:
207  return 1;
208  case V_SgFunctionDefinition:
209  return 2;
210  case V_SgExprStatement:
211  return 1;
212  case V_SgIfStmt:
213  case V_SgWhileStmt:
214  case V_SgDoWhileStmt:
215  case V_SgForStatement:
216  // case V_SgForInitStatement: // not necessary
217  case V_SgBreakStmt:
218  case V_SgContinueStmt:
219  case V_SgGotoStatement:
220  case V_SgLabelStatement:
221  case V_SgNullStatement:
222  case V_SgSwitchStatement:
223  case V_SgDefaultOptionStmt:
224  case V_SgCaseOptionStmt:
225  return 1;
226 
227  // declarations
228  case V_SgPragmaDeclaration:
229  case V_SgFunctionDeclaration:
230  case V_SgVariableDeclaration:
231  case V_SgClassDeclaration:
232  case V_SgEnumDeclaration:
233  case V_SgTypedefDeclaration:
234  case V_SgStaticAssertionDeclaration:
235  case V_SgUsingDeclarationStatement:
236  return 1;
237 
238  // All OpenMP constructs are in the CFG
239  // omp parallel results in fork/join nodes.
240  // for and sections (and currently also SIMD) results in workshare/barrier nodes
241  case V_SgOmpForSimdStatement:
242  case V_SgOmpSimdStatement:
243  case V_SgOmpForStatement:
244  case V_SgOmpSectionsStatement:
245  case V_SgOmpParallelStatement:
246  return 2;
247  // OpenMP barrier results in an actual barrier node
248  case V_SgOmpBarrierStatement:
249  case V_SgOmpCriticalStatement:
250  case V_SgOmpAtomicStatement:
251  case V_SgOmpDoStatement:
252  case V_SgOmpFlushStatement:
253  case V_SgOmpMasterStatement:
254  case V_SgOmpOrderedStatement:
255  case V_SgOmpSectionStatement:
256  case V_SgOmpSingleStatement:
257  case V_SgOmpTargetDataStatement:
258  case V_SgOmpTargetStatement:
259  case V_SgOmpTaskStatement:
260  case V_SgOmpTaskwaitStatement:
261  case V_SgOmpThreadprivateStatement:
262  case V_SgOmpWorkshareStatement:
263  return 1;
264 
265  case V_SgReturnStmt:
266  return 1;
267 
268  case V_SgAsmStmt:
269  return 1;
270 
271 #if OBSOLETE_CODE
272  case V_SgCtorInitializerList:
273  return countLabels(isSgCtorInitializerList(node));
274 #endif /* OBSOLETE_CODE */
275 
276  default:
277  return 0;
278  }

287 void CLabeler::createLabels(SgNode* root) {
288  RoseAst ast(root);
289  for(RoseAst::iterator i=ast.begin();i!=ast.end();++i) {
290  if(int num=numberOfAssociatedLabels(*i)) {
291  if(isFunctionCallNode(*i)) {
292  if(SgNodeHelper::Pattern::matchReturnStmtFunctionCallExp(*i)) {
293  assert(num==3);
294  registerLabel(LabelProperty(*i,LabelProperty::LABEL_FUNCTIONCALL));
295  registerLabel(LabelProperty(*i,LabelProperty::LABEL_FUNCTIONCALLRETURN));
296  registerLabel(LabelProperty(*i)); // return-stmt-label
297  } else {
298  assert(num==2);
299  registerLabel(LabelProperty(*i,LabelProperty::LABEL_FUNCTIONCALL));
300  registerLabel(LabelProperty(*i,LabelProperty::LABEL_FUNCTIONCALLRETURN));
301  }
302  } else if(isSgFunctionDefinition(*i)) {
303  assert(num==2);
304  registerLabel(LabelProperty(*i,LabelProperty::LABEL_FUNCTIONENTRY));
305  registerLabel(LabelProperty(*i,LabelProperty::LABEL_FUNCTIONEXIT));
306  } else if(isSgBasicBlock(*i)) {
307  assert(num==1);
308  registerLabel(LabelProperty(*i,LabelProperty::LABEL_BLOCKBEGIN));
309  } else if(isSgOmpParallelStatement(*i)){
310  assert(num == 2);
311  registerLabel(LabelProperty(*i, LabelProperty::LABEL_FORK));
312  registerLabel(LabelProperty(*i, LabelProperty::LABEL_JOIN));
313  } else if(isSgOmpForStatement(*i) || isSgOmpSectionsStatement(*i)
314  || isSgOmpSimdStatement(*i) || isSgOmpForSimdStatement(*i)) {
315  assert(num == 2);
316  registerLabel(LabelProperty(*i, LabelProperty::LABEL_WORKSHARE));
317  registerLabel(LabelProperty(*i, LabelProperty::LABEL_BARRIER));
318  } else if(isSgOmpBarrierStatement(*i)) {
319  assert(num == 1);
320  registerLabel(LabelProperty(*i, LabelProperty::LABEL_BARRIER));
321  } else {
322  // all other cases
323  for(int j=0;j<num;j++) {
324  registerLabel(LabelProperty(*i));
325  }
326  }
327  }
328  // schroder3 (2016-07-12): We can not skip the children of a variable declaration
329  // because there might be a member function definition inside the variable declaration.
330  // Example:
331  // int main() {
332  // class A {
333  // public:
334  // void mf() {
335  // int i = 2;
336  // }
337  // } a; // Var decl
338  // }
339  if(isSgExprStatement(*i)||isSgReturnStmt(*i)/*||isSgVariableDeclaration(*i)*/)
340  i.skipChildrenOnForward();
341  // MS 2018: skip templates (only label template instantiations)
342  if(isSgTemplateClassDeclaration(*i)||isSgTemplateClassDefinition(*i)) {
343  i.skipChildrenOnForward();
344  }
345  }
346 }

389 void CLabeler::computeNodeToLabelMapping() {
390  mappingNodeToLabel.clear();
391  for(Label i=0;i<mappingLabelToLabelProperty.size();++i) {
392  SgNode* node=mappingLabelToLabelProperty[i.getId()].getNode();
393  assert(node);
394  // There exist nodes with multiple associated labels (1-3 labels). The labels are guaranteed
395  // to be in consecutive increasing order. We only store the very first associated label for each
396  // node and the access functions adjust the label as necessary.
397  // This allows to provide an API where the user does not need to operate on sets but on
398  // distinct labels only.
399  if(mappingNodeToLabel.find(node)!=mappingNodeToLabel.end())
400  continue;
401  else
402  mappingNodeToLabel[node]=i;
403  }
404  _isValidMappingNodeToLabel=true;
405 }
