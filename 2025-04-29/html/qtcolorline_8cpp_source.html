268 void QtColorLinePrivate::checkColor()
269 {
270  switch (m_component) {
271  case QtColorLine::Red:
272  case QtColorLine::Green:
273  case QtColorLine::Blue:
274  if (m_color.spec() != QColor::Rgb)
275  m_color = m_color.toRgb();
276  break;
277  case QtColorLine::Hue:
278  case QtColorLine::Saturation:
279  case QtColorLine::Value:
280  if (m_color.spec() != QColor::Hsv)
281  m_color = m_color.toHsv();
282  break;
283  default:
284  break;
285  }
286  if (m_color.spec() == QColor::Hsv) {
287  if (m_color.hue() == 360 || m_color.hue() == -1) {
288  m_color.setHsvF(0.0, m_color.saturationF(), m_color.valueF(), m_color.alphaF());
289  }
290  }
291 }

293 bool QtColorLinePrivate::isMainPixmapValid() const
294 {
295  if (m_mainPixmap.isNull()) {
296  if (m_pixmapSize.isEmpty())
297  return true;
298  else
299  return false;
300  }
301  if (m_lastValidMainPixmapData.component != m_component)
302  return false;
303  if (m_lastValidMainPixmapData.size != m_pixmapSize)
304  return false;
305  if (m_lastValidMainPixmapData.flipped != m_flipped)
306  return false;
307  if (m_lastValidMainPixmapData.orientation != m_orientation)
308  return false;
309  if (m_lastValidMainPixmapData.color == m_color)
310  return true;
311  switch (m_component) {
312  case QtColorLine::Red:
313  if (m_color.green() == m_lastValidMainPixmapData.color.green() &&
314  m_color.blue() == m_lastValidMainPixmapData.color.blue() &&
315  (!m_combiningAlpha || m_color.alpha() == m_lastValidMainPixmapData.color.alpha()))
316  return true;
317  break;
318  case QtColorLine::Green:
319  if (m_color.red() == m_lastValidMainPixmapData.color.red() &&
320  m_color.blue() == m_lastValidMainPixmapData.color.blue() &&
321  (!m_combiningAlpha || m_color.alpha() == m_lastValidMainPixmapData.color.alpha()))
322  return true;
323  break;
324  case QtColorLine::Blue:
325  if (m_color.red() == m_lastValidMainPixmapData.color.red() &&
326  m_color.green() == m_lastValidMainPixmapData.color.green() &&
327  (!m_combiningAlpha || m_color.alpha() == m_lastValidMainPixmapData.color.alpha()))
328  return true;
329  break;
330  case QtColorLine::Hue:
331  if (m_color.saturation() == m_lastValidMainPixmapData.color.saturation() &&
332  m_color.value() == m_lastValidMainPixmapData.color.value() &&
333  (!m_combiningAlpha || m_color.alpha() == m_lastValidMainPixmapData.color.alpha()))
334  return true;
335  break;
336  case QtColorLine::Saturation:
337  if (m_color.hue() == m_lastValidMainPixmapData.color.hue() &&
338  m_color.value() == m_lastValidMainPixmapData.color.value() &&
339  (!m_combiningAlpha || m_color.alpha() == m_lastValidMainPixmapData.color.alpha()))
340  return true;
341  break;
342  case QtColorLine::Value:
343  if (m_color.hue() == m_lastValidMainPixmapData.color.hue() &&
344  m_color.saturation() == m_lastValidMainPixmapData.color.saturation() &&
345  (!m_combiningAlpha || m_color.alpha() == m_lastValidMainPixmapData.color.alpha()))
346  return true;
347  break;
348  case QtColorLine::Alpha:
349  if (m_color.hue() == m_lastValidMainPixmapData.color.hue() &&
350  m_color.saturation() == m_lastValidMainPixmapData.color.saturation() &&
351  m_color.value() == m_lastValidMainPixmapData.color.value())
352  return true;
353  }
354  return false;
355 }

373 QPixmap QtColorLinePrivate::gradientPixmap(int size, Qt::Orientation orientation,
374  const QColor &begin, const QColor &end, bool flipped) const
375 {
376  int gradW = size;
377  int gradH = size;
378  int w = size;
379  int h = size;
380  if (orientation == Qt::Horizontal) {
381  gradH = 0;
382  h = 1;
383  } else {
384  gradW = 0;
385  w = 1;
386  }
387  QColor c1 = begin;
388  QColor c2 = end;
389  if (flipped) {
390  c1 = end;
391  c2 = begin;
392  }
393  QLinearGradient lg(0, 0, gradW, gradH);
394  lg.setColorAt(0, c1);
395  lg.setColorAt(1, c2);
396  QImage img(w, h, QImage::Format_ARGB32);
397  QPainter p(&img);
398  p.setCompositionMode(QPainter::CompositionMode_Source);
399  p.fillRect(QRect(0, 0, w, h), lg);
400  return QPixmap::fromImage(img);
401 }

412 QPixmap QtColorLinePrivate::hueGradientPixmap(int size, Qt::Orientation orientation, bool flipped,
413  int saturation, int value, int alpha) const
414 {
415  int gradW = size + 1;
416  int gradH = size + 1;
417  int w = size;
418  int h = size;
419  if (orientation == Qt::Horizontal) {
420  gradH = 0;
421  h = 1;
422  } else {
423  gradW = 0;
424  w = 1;
425  }
426  QList<QColor> colorList;
427  colorList << QColor::fromHsv(0, saturation, value, alpha);
428  colorList << QColor::fromHsv(60, saturation, value, alpha);
429  colorList << QColor::fromHsv(120, saturation, value, alpha);
430  colorList << QColor::fromHsv(180, saturation, value, alpha);
431  colorList << QColor::fromHsv(240, saturation, value, alpha);
432  colorList << QColor::fromHsv(300, saturation, value, alpha);
433  colorList << QColor::fromHsv(0, saturation, value, alpha);
434  QLinearGradient lg(0, 0, gradW, gradH);
435  for (int i = 0; i <= 6; i++)
436  lg.setColorAt((double)i / 6.0, flipped ? colorList.at(6 - i) : colorList.at(i));
437  QImage img(w, h, QImage::Format_ARGB32);
438  QPainter p(&img);
439  p.setCompositionMode(QPainter::CompositionMode_Source);
440  p.fillRect(QRect(0, 0, w, h), lg);
441  return QPixmap::fromImage(img);
442 }

444 void QtColorLinePrivate::recreateMainPixmap()
445 {
446  m_lastValidMainPixmapData.size = m_pixmapSize;
447  m_lastValidMainPixmapData.component = m_component;
448  m_lastValidMainPixmapData.color = m_color;
449  m_lastValidMainPixmapData.flipped = m_flipped;
450  m_lastValidMainPixmapData.orientation = m_orientation;
451 
452  if (m_pixmapSize.isEmpty()) {
453  m_mainPixmap = QPixmap();
454  m_alphalessPixmap = QPixmap();
455  m_semiAlphaPixmap = QPixmap();
456  return;
457  }
458 
459  if (m_mainPixmap.size() != m_pixmapSize) {
460  m_mainPixmap = QPixmap(m_pixmapSize);
461  m_alphalessPixmap = QPixmap(m_pixmapSize);
462  m_semiAlphaPixmap = QPixmap(m_pixmapSize);
463  }
464 
465  Qt::Orientation orient = m_orientation;
466  const bool flip = m_flipped;
467 
468  const int r = m_color.red();
469  const int g = m_color.green();
470  const int b = m_color.blue();
471  const int h = m_color.hue();
472  const int s = m_color.saturation();
473  const int v = m_color.value();
474  const int a = m_color.alpha();
475  const double coef = 0.5;
476  const int semi = qRound(a * coef + 0xFF * (1.0 - coef));
477 
478  if (m_component == QtColorLine::Hue) {
479  m_alphalessPixmap = hueGradientPixmap(orient, flip, s, v, 0xFF);
480  if (m_combiningAlpha) {
481  m_mainPixmap = hueGradientPixmap(orient, flip, s, v, a);
482  m_semiAlphaPixmap = hueGradientPixmap(orient, flip, s, v, semi);
483  }
484  } else if (m_component == QtColorLine::Saturation) {
485  m_alphalessPixmap = gradientPixmap(orient, QColor::fromHsv(h, 0, v, 0xFF), QColor::fromHsv(h, 0xFF, v, 0xFF), flip);
486  if (m_combiningAlpha) {
487  m_mainPixmap = gradientPixmap(orient, QColor::fromHsv(h, 0, v, a), QColor::fromHsv(h, 0xFF, v, a), flip);
488  m_semiAlphaPixmap = gradientPixmap(orient, QColor::fromHsv(h, 0, v, semi), QColor::fromHsv(h, 0xFF, v, semi), flip);
489  }
490  } else if (m_component == QtColorLine::Value) {
491  m_alphalessPixmap = gradientPixmap(orient, QColor::fromRgb(0, 0, 0, 0xFF), QColor::fromHsv(h, s, 0xFF, 0xFF), flip);
492  if (m_combiningAlpha) {
493  m_mainPixmap = gradientPixmap(orient, QColor::fromRgb(0, 0, 0, a), QColor::fromHsv(h, s, 0xFF, a), flip);
494  m_semiAlphaPixmap = gradientPixmap(orient, QColor::fromRgb(0, 0, 0, semi), QColor::fromHsv(h, s, 0xFF, semi), flip);
495  }
496  } else if (m_component == QtColorLine::Red) {
497  m_alphalessPixmap = gradientPixmap(orient, QColor::fromRgb(0, g, b, 0xFF), QColor::fromRgb(0xFF, g, b, 0xFF), flip);
498  if (m_combiningAlpha) {
499  m_mainPixmap = gradientPixmap(orient, QColor::fromRgb(0, g, b, a), QColor::fromRgb(0xFF, g, b, a), flip);
500  m_semiAlphaPixmap = gradientPixmap(orient, QColor::fromRgb(0, g, b, semi), QColor::fromRgb(0xFF, g, b, semi), flip);
501  }
502  } else if (m_component == QtColorLine::Green) {
503  m_alphalessPixmap = gradientPixmap(orient, QColor::fromRgb(r, 0, b, 0xFF), QColor::fromRgb(r, 0xFF, b, 0xFF), flip);
504  if (m_combiningAlpha) {
505  m_mainPixmap = gradientPixmap(orient, QColor::fromRgb(r, 0, b, a), QColor::fromRgb(r, 0xFF, b, a), flip);
506  m_semiAlphaPixmap = gradientPixmap(orient, QColor::fromRgb(r, 0, b, semi), QColor::fromRgb(r, 0xFF, b, semi), flip);
507  }
508  } else if (m_component == QtColorLine::Blue) {
509  m_alphalessPixmap = gradientPixmap(orient, QColor::fromRgb(r, g, 0, 0xFF), QColor::fromRgb(r, g, 0xFF, 0xFF), flip);
510  if (m_combiningAlpha) {
511  m_mainPixmap = gradientPixmap(orient, QColor::fromRgb(r, g, 0, a), QColor::fromRgb(r, g, 0xFF, a), flip);
512  m_semiAlphaPixmap = gradientPixmap(orient, QColor::fromRgb(r, g, 0, semi), QColor::fromRgb(r, g, 0xFF, semi), flip);
513  }
514  } else if (m_component == QtColorLine::Alpha) {
515  m_mainPixmap = gradientPixmap(orient, QColor::fromRgb(r, g, b, 0), QColor::fromRgb(r, g, b, 0xFF), flip);
516 
517 // m_alphalessPixmap = gradientPixmap(orient, QColor::fromRgb(r, g, b, 0xFF), QColor::fromRgb(r, g, b, 0xFF), flip);
518 // m_semiAlphaPixmap = gradientPixmap(orient, QColor::fromRgb(r, g, b, semi), QColor::fromRgb(r, g, b, semi), flip);
519  }
520  if (!m_combiningAlpha && m_component != QtColorLine::Alpha)
521  m_mainPixmap = m_alphalessPixmap;
522 }

536 QColor QtColorLinePrivate::colorFromPoint(const QPointF &point) const
537 {
538  QPointF p = point;
539  if (p.x() < 0)
540  p.setX(0.0);
541  else if (p.x() > 1)
542  p.setX(1.0);
543  if (p.y() < 0)
544  p.setY(0.0);
545  else if (p.y() > 1)
546  p.setY(1.0);
547 
548  double pos = p.x();
549  if (m_orientation == Qt::Vertical)
550  pos = p.y();
551  if (m_flipped)
552  pos = 1.0 - pos;
553  QColor c;
554  qreal hue;
555  switch (m_component) {
556  case QtColorLine::Red:
557  c.setRgbF(pos, m_color.greenF(), m_color.blueF(), m_color.alphaF());
558  break;
559  case QtColorLine::Green:
560  c.setRgbF(m_color.redF(), pos, m_color.blueF(), m_color.alphaF());
561  break;
562  case QtColorLine::Blue:
563  c.setRgbF(m_color.redF(), m_color.greenF(), pos, m_color.alphaF());
564  break;
565  case QtColorLine::Hue:
566  hue = pos;
567  hue *= 35999.0 / 36000.0;
568  c.setHsvF(hue, m_color.saturationF(), m_color.valueF(), m_color.alphaF());
569  break;
570  case QtColorLine::Saturation:
571  c.setHsvF(m_color.hueF(), pos, m_color.valueF(), m_color.alphaF());
572  break;
573  case QtColorLine::Value:
574  c.setHsvF(m_color.hueF(), m_color.saturationF(), pos, m_color.alphaF());
575  break;
576  case QtColorLine::Alpha:
577  c.setHsvF(m_color.hueF(), m_color.saturationF(), m_color.valueF(), pos);
578  break;
579  }
580  return c;
581 }

583 QPointF QtColorLinePrivate::pointFromColor(const QColor &color) const
584 {
585  qreal hue = color.hueF();
586  if (color.hue() == 360)
587  hue = 0.0;
588  else
589  hue *= 36000.0 / 35999.0;
590 
591  double pos = 0.0;
592  switch (m_component) {
593  case QtColorLine::Red:
594  pos = color.redF();
595  break;
596  case QtColorLine::Green:
597  pos = color.greenF();
598  break;
599  case QtColorLine::Blue:
600  pos = color.blueF();
601  break;
602  case QtColorLine::Hue:
603  pos = hue;
604  break;
