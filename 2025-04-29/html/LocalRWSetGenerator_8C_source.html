296 VarType LocalRWSetGenerator::determineType(SgType* curType) {
297  //Aside from cases where we know exactly what the type is, we're going to have
298  //to match along looking for the worst possible type. (For example, is a struct a
299  //simple struct or a complex one? We can only tell by marching over the whole classTree
300  //tooking for a pointer.
301  //So we start with the lowest possible type, and everytime we see a worse one, we upgrade it.
302  VarType returnType = PRIMITIVES;
303  if(isSgArrayType(curType)){
304  SgArrayType* arrayType = isSgArrayType(curType);
305  VarType elemType = determineType(arrayType->get_base_type());
306  if(elemType <= ARRAYS) {
307  return ARRAYS;
308  }
309  return ARRAYS_OF_STRUCTS;
310  }
311  else if(isSgModifierType(curType)){
312  SgType* baseType = curType->stripType(SgType::STRIP_MODIFIER_TYPE);
313  return determineType(baseType);
314  }
315  else if(isSgReferenceType(curType)){
316  SgType* baseType = curType->stripType(SgType::STRIP_REFERENCE_TYPE);
317  return determineType(baseType);
318  }
319  else if(isSgTypedefType(curType)){
320  SgType* baseType = curType->stripType(SgType::STRIP_TYPEDEF_TYPE);
321  return determineType(baseType);
322  }
323 
324  else if(isSgClassType(curType)) {
325 
326  SgClassType* classType = isSgClassType(curType);
327  SgDeclarationStatement* sgDeclStmt = classType->get_declaration();
328  SgClassDeclaration* sgClassDecl = isSgClassDeclaration(sgDeclStmt);
329  ROSE_ASSERT(sgClassDecl);
330  SgClassDefinition* sgClassDef = getClassDefinitionFromDeclaration(sgClassDecl);
331  if(sgClassDef == NULL) {
332  mlog[WARN] << "In determineType: Unable to get class definition: " << sgClassDecl->get_qualified_name().getString() << std::endl;
333  return VARTYPE_UNKNOWN; //If we can't define the class, it's unmockable
334  }
335  return STRUCTS;
336 
337  }
338  else if(isSgPointerType(curType)) {
339  //Saw a pointer in class, but it wasn't dereferenced, so it's a primative?
340  return POINTERS;
341  }
342  else if(curType->isPrimativeType()) {
343  return PRIMITIVES;
344  }
345  else if(isSgEnumType(curType)) {
346  return PRIMITIVES;
347  }
348  else {
349  // This happens with ellipses (...) in function arguments. We
350  // don't support those yet, so there's nothing to do about it.
351  // The function used with ellipses (va_start etc.) produce
352  // SgTypeDefault, which I can't do anything with. So leave this
353  // warning here for now. -Jim 2023/06/14
354  // tests: ellipses_1.cxx ellipses_2.c
355  mlog[WARN] << "In determineType got unexpected type " << curType->class_name() << std::endl;
356  Sg_File_Info* finfo = curType->get_file_info();
357  if(finfo) {
358  mlog[WARN] << VxUtilFuncs::getNodeRelativePath(curType) << ":" << finfo->get_line() << std::endl;
359  } else {
360  mlog[WARN] << "FileInfo was null" << std::endl;
361  }
362  return VarType::VARTYPE_UNKNOWN;
363  }
364  return returnType; //Return whatever the worst type was.
365 
366 }
