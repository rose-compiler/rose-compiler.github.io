151 SgAsmStringStorage *
152 SgAsmElfStrtab::createStorage(Address offset, bool shared)
153 {
154  ROSE_ASSERT(offset!=SgAsmGenericString::unallocated);
155 
156  /* Has this string already been created? If so, return previous storage object. However, never share the empty_string at
157  * offset zero created when this string table was constructed because the ELF spec says it needs to stay there whether
158  * referenced or not. */
159  if (shared) {
160  for (referenced_t::iterator i=p_storageList.begin(); i!=p_storageList.end(); i++) {
161  if ((*i)->get_offset()==offset && (*i) != get_dontFree())
162  return *i;
163  }
164  }
165 
166  /* Create a new storage object at this offset. */
167  SgAsmStringStorage *storage = NULL;
168  if (0==offset && 0==get_container()->get_data().size()) {
169  ROSE_ASSERT(get_container()->get_size()>=1);
170  storage = new SgAsmStringStorage(this, "", 0);
171  } else {
172  std::string s = get_container()->readContentLocalString(offset);
173  storage = new SgAsmStringStorage(this, s, offset);
174  }
175 
176  /* It's a bad idea to free (e.g., modify) strings before we've identified all the strings in the table. Consider
177  * the case where offset 1 is "domain" and offset 3 is "main" (i.e., they overlap). If we modify "main" before knowing
178  * about "domain" then we'll end up freeing the last part of "domain" (and possibly replacing it with something else)!
179  *
180  * The only time we can guarantee this is OK is when the new storage points to the same file location as "dontFree"
181  * since the latter is guaranteed to never be freed or shared. This exception is used when creating a new, unallocated
182  * string (see SgAsmStoredString(SgAsmGenericStrtab,const std::string&)). */
183  if (get_numberFreed()>0 && (!get_dontFree() || offset!=get_dontFree()->get_offset())) {
184  mlog[WARN] <<"SgAsmElfStrtab::createStorage(" <<StringUtility::addrToString(offset) <<"): "
185  <<StringUtility::plural(p_numberFreed, "other strings")
186  <<" (of " <<p_storageList.size() <<" created)"
187  <<" in [" <<get_container()->get_id() <<"] \"" <<get_container()->get_name()->get_string(true) <<"\" "
188  <<(1==get_numberFreed()?"has":"have") <<" been modified and/or reallocated\n";
189  ROSE_ASSERT(0==get_numberFreed());
190  }
191 
192  p_storageList.push_back(storage);
193  set_isModified(true);
194  return storage;
195 }

222 void
223 SgAsmElfStrtab::allocateOverlap(SgAsmStringStorage *storage)
224 {
225  ROSE_ASSERT(storage->get_offset()==SgAsmGenericString::unallocated);
226  size_t need = storage->get_string().size();
227  for (size_t i=0; i<p_storageList.size(); i++) {
228  SgAsmStringStorage *existing = p_storageList[i];
229  if (existing->get_offset()!=SgAsmGenericString::unallocated) {
230  size_t have = existing->get_string().size();
231  if (need<=have && 0==existing->get_string().compare(have-need, need, storage->get_string())) {
232  /* An existing string ends with the new string. */
233  storage->set_offset(existing->get_offset() + (have-need));
234  return;
235  } else if (need>have && existing->get_offset()>=need-have &&
236  0==storage->get_string().compare(need-have, have, existing->get_string())) {
237  /* New string ends with an existing string. Check for, and allocate, free space. */
238  Address offset = existing->get_offset() - (need-have); /* positive diffs checked above */
239  AddressInterval allocationRequest = AddressInterval::baseSize(offset, need-have);
240  if (get_freeList().contains(allocationRequest)) {
241  get_freeList().erase(allocationRequest);
242  storage->set_offset(offset);
243  return;
244  }
245  }
246  }
247  }

250 void
251 SgAsmElfStrtab::unparse(std::ostream &f) const
252 {
253  SgAsmGenericSection *container = get_container();
254 
255  /* Write strings with NUL termination. Shared strings will be written more than once, but that's OK. */
256  for (size_t i=0; i<p_storageList.size(); i++) {
257  SgAsmStringStorage *storage = p_storageList[i];
258  if (storage->get_offset() == SgAsmGenericString::unallocated) {
259  mlog[WARN] <<"during unparsing, this string is unallocated in the string table and will be discarded: "
260  <<"\"" <<StringUtility::cEscape(storage->get_string()) <<"\"\n";
261  } else {
262  Address at = container->write(f, storage->get_offset(), storage->get_string());
263  container->write(f, at, '\0');
264  }
265  }
266 
267  /* Fill free areas with zero */
268  BOOST_FOREACH (const AddressInterval &interval, get_freeList().intervals())
269  container->write(f, interval.least(), std::string(interval.size(), '\0'));
270 }
