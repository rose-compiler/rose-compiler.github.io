335 void QRTreeBox::loadTree(bool doListView) {
336  Q3ListView *lv = doListView? m_wgListView : m_wgCacheView;
337 
338  map<QRListViewItem *, SgNode *> *vi = doListView? (&m_viewItems) : (&m_viewCacheItems) ;
339  map<SgNode *, QRListViewItem *> *ni = doListView? (&m_nodeItems) : (&m_nodeCacheItems);
340  map<string, unsigned> *cn = doListView? (&m_capturedNames) : (&m_capturedCacheNames);
341  map<unsigned, string> *ci = doListView? (&m_capturedIndices) : (&m_capturedCacheIndices);
342 
343  SgNode *rootNode = doListView? m_rootNode : m_project;
344 
345  if (!doListView && !rootNode) {
346  // we will assume that the universe is the project of the first cache node
347  set<SgNode *> *cache = getCache();
348  if (!cache->empty()) {
349  SgNode *node = *(cache->begin());
350  // if a node is not associated to a project (e.g. types), then
351  // getProject() aborts the program. This is a small fix to avoid this.
352  if (node->get_parent()) {
353  rootNode = TransformationSupport::getProject(node);
354  }
355  }
356  }
357 
358  vi->clear();
359  ni->clear();
360  cn->clear();
361  ci->clear();
362  lv->clear();
363  while (lv->columns())
364  lv->removeColumn(0);
365 
366  lv->addColumn("Nodes", 600);
367  lv->setAllColumnsShowFocus (true);
368  lv->setSorting(0);
369  lv->setShowSortIndicator(false);
370  lv->setRootIsDecorated(true);
371  m_sortColumn = 0;
372  m_sortCacheColumn = 0;
373 
374  QRTreeVisitor visitor(this);
375 
376 
377  visitor.setArguments(!doListView, lv, vi, ni, cn, ci, m_detailLevel, m_cacheShowChildren);
378  emit setInfo("rendering tree... please wait...", 0);
379  QROSE::processEvents();
380  visitor.performTraversal(rootNode);
381 
382  emit setInfo("done!", statusTimer);
383 
384 }

414 int QRTreeBox::compareItems(QRListViewItem *item1, QRListViewItem *item2, int col, bool ascending) {
415  if (col == 0) {
416  // sorting for column 0 (Project Tree) is just the order in which they were created
417  if (item1->getId() < item2->getId()) {
418  return -1;
419  } else if (item1->getId() == item2->getId()) {
420  return 0;
421  } else {
422  return 1;
423  }
424  } else {
425  // find attribute name corresponding to 'col' index
426  std::map<unsigned, std::string>::iterator iter;
427  if (isCacheView()) iter = m_capturedCacheIndices.find(col); else
428  iter = m_capturedIndices.find(col);
429  eAssert(iter != m_capturedIndices.end(), ("internal error: cannot find attribute!"));
430 
431  string attrib_name = iter->second;
432  QRAttribute *qrattrib = getAttribute(attrib_name);
433 
434  AstAttribute *att_ast1 = item1->getAstAttribute(attrib_name);
435  AstAttribute *att_ast2 = item2->getAstAttribute(attrib_name);
436 
437  return qrattrib->compare(att_ast1, att_ast2, ascending);
438  }

515 void QRTreeBox::popupMenuHandler(int id) {
516 
517  QRListViewItem *selected_item = (QRListViewItem *) getCurrentView()->selectedItem();
518  SgNode *selected_node = selected_item->getNode();
519  switch (id) {
520  case pm_set_root:
521  m_rootNode = selected_item->getNode();
522  loadTree(true);
523  selected_item = m_nodeItems[selected_node];
524  m_wgListView->setSelected(selected_item, true);
525  m_wgListView->setOpen(selected_item, true);
526  break;
527  case pm_set_default_root:
528  m_rootNode = m_defaultRootNode;
529  loadTree(true);
530  selected_item = m_nodeItems[selected_node];
531  m_wgListView->setSelected(selected_item, true);
532  m_wgListView->ensureItemVisible(selected_item);
533  break;
534  case pm_remove:
535  removeNodeFromCache(selected_node);
536  break;
537  case pm_expand:
538  case pm_collapse:
539  case pm_expand_code:
540  {
541  list<QRListViewItem *> items;
542  items.push_back(selected_item);
543 
544  while (items.size() != 0) {
545  QRListViewItem *item;
546  item = items.front(); items.erase(items.begin());
547  if ((item == selected_item) && (id != pm_collapse)) {
548  item->setOpen(true);
549  } else {
550  item->setOpen((id == pm_expand_code)?
551  QRAstInfo::isRelevant(item->getNode()) : id == pm_expand);
552  }
553  QRListViewItem *child_item = (QRListViewItem *) item->firstChild();
554  while (child_item) {
555  items.push_back(child_item);
556  child_item = (QRListViewItem *) child_item->nextSibling();
557  }
558  }
559  }
560  break;
561  }
