60 /* Class method comparing two extents. The return value is one of the following letters, depending on how extent A is related
61  * to extent B:
62  * C (congruent): A and B are congruent
63  * L (left): A is left of B
64  * R (right): A is right of B
65  * O (outer): A contains B, but A and B are not congruent
66  * I (inner): A is contained by B, but A and B are not congruent
67  * B (beginning): A overlaps with the beginning of B but does not contain B
68  * E (ending): A overlaps with the end of B but does not contain B */
69 char
70 ExtentMap::category(const Extent &a, const Extent &b)
71 {
72  if (a.relaxed_first()==b.relaxed_first() && a.size()==b.size())
73  return 'C';
74  if (a.relaxed_first()+a.size() <= b.relaxed_first())
75  return 'L';
76  if (a.relaxed_first() >= b.relaxed_first()+b.size())
77  return 'R';
78  if (a.relaxed_first() <= b.relaxed_first() && a.relaxed_first()+a.size() >= b.relaxed_first()+b.size())
79  return 'O';
80  if (a.relaxed_first() >= b.relaxed_first() && a.relaxed_first()+a.size() <= b.relaxed_first()+b.size())
81  return 'I';
82  if (a.relaxed_first() <= b.relaxed_first()) /*already know a.first+a.size > b.first*/
83  return 'B';
84  return 'E';
85 }
