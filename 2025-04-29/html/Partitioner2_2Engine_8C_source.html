163 // If the block is a function return (e.g., ends with an x86 RET instruction) to an indeterminate location, then that successor
164 // type should be E_FUNCTION_RETURN instead of E_NORMAL.
165 void
166 Engine::BasicBlockFinalizer::fixFunctionReturnEdge(const Args &args) {
167  if (args.partitioner->basicBlockIsFunctionReturn(args.bblock)) {
168  bool hadCorrectEdge = false, edgeModified = false;
169  BasicBlock::Successors successors = args.partitioner->basicBlockSuccessors(args.bblock);
170  for (size_t i = 0; i < successors.size(); ++i) {
171  if (!successors[i].expr()->isConcrete() ||
172  (successors[i].expr()->get_expression()->flags() & SymbolicExpression::Node::INDETERMINATE) != 0) {
173  if (successors[i].type() == E_FUNCTION_RETURN) {
174  hadCorrectEdge = true;
175  break;
176  } else if (successors[i].type() == E_NORMAL && !edgeModified) {
177  successors[i].type(E_FUNCTION_RETURN);
178  edgeModified = true;
179  }
180  }
181  }
182  if (!hadCorrectEdge && edgeModified) {
183  args.bblock->clearSuccessors();
184  args.bblock->successors(successors);
185  SAWYER_MESG(mlog[DEBUG]) <<args.bblock->printableName() <<": fixed function return edge type\n";
186  }
187  }
188 }

190 // If the block is a function call (e.g., ends with an x86 CALL instruction) then change all E_NORMAL edges to E_FUNCTION_CALL
191 // edges.
192 void
193 Engine::BasicBlockFinalizer::fixFunctionCallEdges(const Args &args) {
194  if (args.partitioner->basicBlockIsFunctionCall(args.bblock)) {
195  BasicBlock::Successors successors = args.partitioner->basicBlockSuccessors(args.bblock);
196  bool changed = false;
197  for (BasicBlock::Successor &successor: successors) {

204  args.bblock->clearSuccessors();
205  args.bblock->successors(successors);
206  SAWYER_MESG(mlog[DEBUG]) <<args.bblock->printableName() <<": fixed function call edge(s) type\n";
207  }

211 // Should we add an indeterminate CFG edge from this basic block? For instance, a "JMP [ADDR]" instruction should get an
212 // indeterminate edge if ADDR is a writable region of memory. There are two situations: ADDR is non-writable, in which case
213 // RiscOperators::peekMemory would have returned a free variable to indicate an indeterminate value, or ADDR is writable but
214 // its MemoryMap::INITIALIZED bit is set to indicate it has a valid value already, in which case RiscOperators::peekMemory
215 // would have returned the value stored there but also marked the value as being INDETERMINATE. The
216 // SymbolicExpression::TreeNode::INDETERMINATE bit in the expression should have been carried along so that things like "MOV
217 // EAX, [ADDR]; JMP EAX" will behave the same as "JMP [ADDR]".
218 void
219 Engine::BasicBlockFinalizer::addPossibleIndeterminateEdge(const Args &args) {
220  BasicBlockSemantics sem = args.bblock->semantics();
221  if (sem.finalState() == nullptr)
222  return;
223  ASSERT_not_null(sem.operators);
224 
225  bool addIndeterminateEdge = false;
226  size_t addrWidth = 0;
227  for (const BasicBlock::Successor &successor: args.partitioner->basicBlockSuccessors(args.bblock)) {
228  if (!successor.expr()->isConcrete()) { // BB already has an indeterminate successor?
229  addIndeterminateEdge = false;
230  break;
231  } else if (!addIndeterminateEdge &&
232  (successor.expr()->get_expression()->flags() & SymbolicExpression::Node::INDETERMINATE) != 0) {
233  addIndeterminateEdge = true;
234  addrWidth = successor.expr()->nBits();
235  }
236  }
237 
238  // Add an edge
239  if (addIndeterminateEdge) {
240  ASSERT_require(addrWidth != 0);
241  BaseSemantics::SValue::Ptr addr = sem.operators->undefined_(addrWidth);
242  EdgeType type = args.partitioner->basicBlockIsFunctionReturn(args.bblock) ? E_FUNCTION_RETURN : E_NORMAL;
243  args.bblock->insertSuccessor(addr, type);
244  SAWYER_MESG(mlog[DEBUG]) <<args.bblock->printableName()
245  <<": added indeterminate successor for initialized, non-constant memory read\n";
246  }
247 }

263 // Add basic block to worklist(s)
264 bool
265 Engine::BasicBlockWorkList::operator()(bool chain, const AttachedBasicBlock &args) {
266  if (chain) {
267  ASSERT_not_null(args.partitioner);
268  Partitioner::ConstPtr partitioner = args.partitioner;
269 
270  // Basic block that is not yet discovered. We could use the special "undiscovered" CFG vertex, but there is no ordering
271  // guarantee for its incoming edges. We want to process undiscovered vertices in a depth-first manner, which is why we
272  // maintain our own list instead. The reason for depth-first discovery is that some analyses are recursive in nature
273  // and we want to try to have children discovered and analyzed before we try to analyze the parent. For instance,
274  // may-return analysis for one vertex probably depends on the may-return analysis of its successors.
275  if (args.bblock == nullptr) {
276  undiscovered_.pushBack(args.startVa);
277  return chain;
278  }
279 
280  // If a new function call is inserted and it has no E_CALL_RETURN edge and at least one of its callees has an
281  // indeterminate value for its may-return analysis, then add this block to the list of blocks for which we may need to
282  // later add a call-return edge. The engine can be configured to also just assume that all function calls may return
283  // (or never return).
284  if (partitioner->basicBlockIsFunctionCall(args.bblock)) {
285  ControlFlowGraph::ConstVertexIterator placeholder = partitioner->findPlaceholder(args.startVa);
286  boost::logic::tribool mayReturn;
287  switch (engine_->settings().partitioner.functionReturnAnalysis) {
288  case MAYRETURN_ALWAYS_YES:
289  mayReturn = true;
290  break;
291  case MAYRETURN_ALWAYS_NO:
292  mayReturn = false;
293  break;
294  case MAYRETURN_DEFAULT_YES:
295  case MAYRETURN_DEFAULT_NO: {
296  ASSERT_require(placeholder != partitioner->cfg().vertices().end());
297  mayReturn = hasAnyCalleeReturn(partitioner, placeholder);
298  break;
299  }
300  }
301  if (!hasCallReturnEdges(placeholder) && (mayReturn || boost::logic::indeterminate(mayReturn)))
302  pendingCallReturn_.pushBack(args.startVa);
303  }
304  }
305  return chain;
306 }
307 
308 // Remove basic block from worklist(s). We probably don't really need to erase things since the items are revalidated when
309 // they're consumed from the list, and avoiding an erase will keep the list in its original order when higher level functions
310 // make a minor adjustment with a detach-adjust-attach sequence. On the other hand, perhaps such adjustments SHOULD move the
311 // block to the top of the stack.
312 bool
313 Engine::BasicBlockWorkList::operator()(bool chain, const DetachedBasicBlock &args) {
314  if (chain) {
315  pendingCallReturn_.erase(args.startVa);
316  processedCallReturn_.erase(args.startVa);
317  finalCallReturn_.erase(args.startVa);
318  undiscovered_.erase(args.startVa);
319  }
320  return chain;
321 }

343 // Move pendingCallReturn items into the finalCallReturn list and (re)sort finalCallReturn items according to the CFG so that
344 // descendents appear after their ancestors (i.e., descendents will be processed first since we always use popBack). This is a
345 // fairly expensive operation: O((V+E) ln N) where V and E are the number of edges in the CFG and N is the number of addresses
346 // in the combined lists.
347 void
348 Engine::BasicBlockWorkList::moveAndSortCallReturn(const Partitioner::ConstPtr &partitioner) {
349  using namespace Sawyer::Container::Algorithm; // graph traversals
350 
351  if (processedCallReturn().isEmpty())
352  return; // nothing to move, and finalCallReturn list was previously sorted
353 
354  if (maxSorts_ == 0) {
355  for (Address va: processedCallReturn_.items())
356  finalCallReturn().pushBack(va);
357  processedCallReturn_.clear();
358 
359  } else {
360  if (0 == --maxSorts_)
361  mlog[WARN] <<"may-return sort limit reached; reverting to unsorted analysis\n";
362 
363  // Get the list of virtual addresses that need to be processed
364  std::vector<AddressOrder> pending;
365  pending.reserve(finalCallReturn_.size() + processedCallReturn_.size());
366  for (Address va: finalCallReturn_.items())
367  pending.push_back(AddressOrder(va, (size_t)0));
368  for (Address va: processedCallReturn_.items())
369  pending.push_back(AddressOrder(va, (size_t)0));
370  finalCallReturn_.clear();
371  processedCallReturn_.clear();
372 
373  // Find the CFG vertex for each pending address and insert its "order" value. Blocks that are leaves (after arbitrarily
374  // breaking cycles) have lower numbers than blocks higher up in the global CFG.
375  std::vector<size_t> order = graphDependentOrder(partitioner->cfg());
376  for (AddressOrder &pair: pending) {
377  ControlFlowGraph::ConstVertexIterator vertex = partitioner->findPlaceholder(pair.first);
378  if (vertex != partitioner->cfg().vertices().end() && vertex->value().type() == V_BASIC_BLOCK) {
379  pair.second = order[vertex->id()];
380  }
381  }
382 
383  // Sort the pending addresses based on their calculated "order", skipping those that aren't CFG basic blocks, and save
384  // the result.
385  pending.erase(std::remove_if(pending.begin(), pending.end(), isSecondZero), pending.end());
386  std::sort(pending.begin(), pending.end(), sortBySecond);
387  for (const AddressOrder &pair: pending)
388  finalCallReturn().pushBack(pair.first);
389  }
390 }

437 Sawyer::Optional<Address>
438 Engine::CodeConstants::nextConstant(const Partitioner::ConstPtr &partitioner) {
439  if (!constants_.empty()) {
440  Address constant = constants_.back();
441  constants_.pop_back();
442  return constant;
443  }
444 
445  while (!toBeExamined_.empty()) {
446  inProgress_ = *toBeExamined_.begin();
447  toBeExamined_.erase(inProgress_);
448  if (SgAsmInstruction *insn = partitioner->instructionExists(inProgress_).insn()) {
449

453  constants.insert(ival->get_absoluteValue());
454  });

456  }
457

460  constants_.pop_back();

569 Engine::Ptr
570 Engine::forge(const std::vector<std::string> &args, Sawyer::CommandLine::Parser &switchParser,
571  const PositionalArgumentParser &positionalParser, const Settings &settings) {
572  const auto factories = registeredFactories();
573  for (const Engine::Ptr &factory: boost::adaptors::reverse(factories)) {
574  // Figure out what arguments describe the specimen
575  auto tempSwitchParser = switchParser;
576  factory->addAllToParser(tempSwitchParser);
577  auto parserResult = tempSwitchParser.parse(args);
578  std::vector<std::string> specimen = positionalParser.specimen(parserResult.unreachedArgs());
579 
580  // If the factory can handle the situation, use it to create a new engine and add the engine's command-line parsing to the
581  // supplied switch parser. Save the specimen in the engine, although it can be adjusted after the return.
582  if (factory->matchFactory(parserResult, specimen)) {
583  Engine::Ptr engine = factory->instanceFromFactory(settings);
584  engine->addAllToParser(switchParser);
585  engine->specimen(specimen);
586  return engine;
587  }
588  }
589  return {};
590 }

688 std::list<Sawyer::CommandLine::SwitchGroup>
689 Engine::allCommandLineSwitches() {
690  std::list<Sawyer::CommandLine::SwitchGroup> retval;
691 
692  // Gather switch groups for `this`. It's important to grab these because they're linked to this object (parsing those switches
693  // updates this object's settings).
694  {
695  const auto list = commandLineSwitches();
696  retval.insert(retval.end(), list.begin(), list.end());
697  }
698 
699  // Gather all switch groups for factories. It's okay if the factory has switch groups that are duplicates of this object's
700  // switch groups because we'll erase the duplicates next.
701  const auto factories = registeredFactories();
702  for (const Engine::Ptr &factory: boost::adaptors::reverse(factories)) {
703  const auto list = factory->commandLineSwitches();
704  retval.insert(retval.end(), list.begin(), list.end());
705  }
706 
707  // Erase duplicates, keeping only the first occurrance of each switch group.
708  Sawyer::Container::Set<std::string> titles, names;
709  for (auto iter = retval.begin(); iter != retval.end(); /*void*/) {
710  if (titles.exists(iter->title()) || names.exists(iter->name())) {
711  iter = retval.erase(iter);
712  } else {
713  titles.insert(iter->title());
714  names.insert(iter->name());
715  ++iter;
716  }
717  }
718 
719  return retval;
720 }

724  using List = std::list<std::pair<std::string, std::string>>;
725 
726  // Accumulate all documentation
727  List retval;

730  retval.push_back(factory->specimenNameDocumentation());
731

738  ++iter;
739  } else {

741  iter = retval.erase(iter);
742  }
743  }
744  return retval;

911 Partitioner::Ptr
912 Engine::createBarePartitioner() {
913  Sawyer::Message::Stream info(mlog[MARCH]);
914 
915  checkCreatePartitionerPrerequisites();
916  auto partitioner = Partitioner::instance(architecture(), memoryMap());
917  if (partitioner->memoryMap() && partitioner->memoryMap()->byteOrder() == ByteOrder::ORDER_UNSPECIFIED)
918  partitioner->memoryMap()->byteOrder(architecture()->byteOrder());
919  partitioner->settings(settings().partitioner.base);
920  partitioner->progress(progress());
921 
922  // Load configuration files
923  if (!settings().engine.configurationNames.empty()) {
924  Sawyer::Stopwatch timer;
925  info <<"loading configuration files";
926  for (const std::string &configName: settings().engine.configurationNames)
927  partitioner->configuration().loadFromFile(configName);
928  info <<"; took " <<timer <<"\n";
929  }
930 
931  // Build the may-return blacklist and/or whitelist. This could be made specific to the type of interpretation being
932  // processed, but there's so few functions that we'll just plop them all into the lists.
933 //TODO: Not for JVM?
934  ModulesPe::buildMayReturnLists(partitioner);
935  ModulesElf::buildMayReturnLists(partitioner);
936 
937  // Make sure the basicBlockWorkList gets updated when the partitioner's CFG is adjusted.
938  ASSERT_not_null(basicBlockWorkList());
939  partitioner->cfgAdjustmentCallbacks().prepend(basicBlockWorkList());
940 
941  // Make sure the stream of constants found in instruction ASTs is updated whenever the CFG is adjusted.
942  if (settings().partitioner.findingCodeFunctionPointers) {
943  codeFunctionPointers(CodeConstants::instance());
944  partitioner->cfgAdjustmentCallbacks().prepend(codeFunctionPointers());
945  }
946 
947  // Perform some finalization whenever a basic block is created. For instance, this figures out whether we should add an
948  // extra indeterminate edge for indirect jump instructions that go through initialized but writable memory.
949  partitioner->basicBlockCallbacks().append(BasicBlockFinalizer::instance());
950 
951  // If the may-return analysis is run and cannot decide whether a function may return, should we assume that it may or
952  // cannot return? The engine decides whether to actually invoke the analysis -- this just sets what to do if it's
953  // invoked.
954  switch (settings().partitioner.functionReturnAnalysis) {
955  case MAYRETURN_ALWAYS_YES:
956  case MAYRETURN_DEFAULT_YES:
957  partitioner->assumeFunctionsReturn(true);
958  break;
959  case MAYRETURN_ALWAYS_NO:
960  case MAYRETURN_DEFAULT_NO:
961  partitioner->assumeFunctionsReturn(false);
962  }
963 
964  // Should the partitioner favor list-based or map-based containers for semantic memory states?
965  partitioner->semanticMemoryParadigm(settings().partitioner.semanticMemoryParadigm);
966 
967  // Miscellaneous settings
968  if (!settings().partitioner.ipRewrites.empty()) {
969  std::vector<Modules::IpRewriter::AddressPair> rewrites;
970  for (size_t i = 0; i+1 < settings().partitioner.ipRewrites.size(); i += 2)
971  rewrites.push_back(std::make_pair(settings().partitioner.ipRewrites[i+0], settings().partitioner.ipRewrites[i+1]));
972  partitioner->basicBlockCallbacks().append(Modules::IpRewriter::instance(rewrites));
973  }
974  if (settings().partitioner.followingGhostEdges)
975  partitioner->basicBlockCallbacks().append(Modules::AddGhostSuccessors::instance());
976  if (!settings().partitioner.discontiguousBlocks)
977  partitioner->basicBlockCallbacks().append(Modules::PreventDiscontiguousBlocks::instance());
978  if (settings().partitioner.maxBasicBlockSize > 0)
979  partitioner->basicBlockCallbacks().append(Modules::BasicBlockSizeLimiter::instance(settings().partitioner.maxBasicBlockSize));
980 
981  // PEScrambler descrambler
982  if (settings().partitioner.peScramblerDispatcherVa) {
983  ModulesPe::PeDescrambler::Ptr cb = ModulesPe::PeDescrambler::instance(settings().partitioner.peScramblerDispatcherVa);
984  cb->nameKeyAddresses(partitioner); // give names to certain PEScrambler things
985  partitioner->basicBlockCallbacks().append(cb);
986  partitioner->attachFunction(Function::instance(settings().partitioner.peScramblerDispatcherVa,
987  partitioner->addressName(settings().partitioner.peScramblerDispatcherVa),
988  SgAsmFunction::FUNC_PESCRAMBLER_DISPATCH));
989  }
990 
991  return partitioner;
992 }
