114 void
115 TokenStream::emit(std::ostream &out, const std::string &fileName, const Token &begin, const Token &locus, const Token &end,
116  const std::string &message) const {
117 
118  std::pair<size_t, size_t> loc1 = content_.location(begin.begin_);
119  std::pair<size_t, size_t> loc2 = content_.location(locus.begin_);
120  //std::pair<size_t, size_t> loc3 = content_.location(locus.end_);
121  std::pair<size_t, size_t> loc4 = content_.location(end.end_);
122 
123  // Emit "NAME:LINE:COL: MESG" to show the beginning of the locus
124  if (!message.empty())
125  out <<fileName <<":" <<(loc2.first+1) <<":" <<(loc2.second+1) <<": " <<message <<"\n";
126 
127  // Emit context matched lines
128  for (size_t lineIdx = loc1.first; lineIdx <= loc4.first; ++lineIdx) {
129  const char *line = content_.lineChars(lineIdx);
130  if (line) {
131  // Line number right justified in a field of 7 characters
132  std::string lineNumStr = boost::lexical_cast<std::string>(lineIdx+1);
133  if (lineNumStr.size() > 7)
134  lineNumStr = "+" + boost::lexical_cast<std::string>(lineIdx-loc1.first);
135  if (lineNumStr.size() < 7)
136  lineNumStr = std::string(7 - lineNumStr.size(), ' ') + lineNumStr;
137 
138  // The line from the source file
139  std::string str(line, content_.nCharacters(lineIdx));
140  boost::trim_right(str);
141  out <<lineNumStr <<"|" <<str <<"\n"; // indentation preserves TABS when traditional stops are assumed
142 
143  // The matching part of the line is underlined with "~" characters except "^" is used at the start of the locus.
144  out <<" |";
145  size_t col0 = content_.characterIndex(lineIdx);
146  size_t colN = col0 + str.size();
147  size_t cur = col0;
148 
149  // white space before first '~'
150  if (col0 < begin.begin_) {
151  size_t n = begin.begin_ - col0;
152  out <<std::string(n, ' ');
153  cur += n;
154  }
155 
156  // '~' characters, up to '^' or EOL
157  if (lineIdx < loc2.first) {
158  size_t n = colN - cur;
159  out <<std::string(n, '~'); // '~' to eol
160  cur += n;
161  } else if (lineIdx == loc2.first) {
162  ASSERT_require(locus.begin_ >= cur);
163  size_t n = locus.begin_ - cur;
164  out <<std::string(n, '~'); // '~' to '^'
165  cur += n;
166  }
167 
168  // '^' at beginning of locus
169  if (lineIdx == loc2.first) {
170  out <<"^";
171  ++cur;
172  }
173 
174  // '~' characters, up to end of match or EOL
175  if (lineIdx < loc4.first) {
176  size_t n = colN - cur;
177  out <<std::string(n, '~'); // '~' to eol
178  cur += n;
179  } else {
180  ASSERT_require(lineIdx == loc4.first);
181  ASSERT_require(end.end_ >= cur);
182  size_t n = end.end_ - cur;
183  out <<std::string(n, '~'); // '~' to end of match
184  cur += n;
185  }
186 
187  out <<"\n";
188  }
189  }
190 }

215 void
216 TokenStream::makeNextToken() {
217  if (!tokens_.empty() && tokens_.back().type() == TOK_EOF)
218  return;
219  while (isspace(getChar(at_)))
220  ++at_;
221  int c = getChar(at_);
222  if (EOF == c) {
223  tokens_.push_back(Token(TOK_EOF, prior_, at_, at_));
224  prior_ = at_;
225  } else if ('\'' == c || '"' == c) {
226  const size_t begin = at_;
227  scanString();
228  tokens_.push_back(Token('"'==c ? TOK_STRING : TOK_CHAR, prior_, begin, at_));
229  prior_ = at_;
230  } else if ('/' == c && '/' == getChar(at_+1)) {
231  const size_t begin = at_;
232  at_ = content_.characterIndex(content_.lineIndex(at_) + 1);
233  if (skipCommentTokens_) {
234  makeNextToken();
235  } else {
236  tokens_.push_back(Token(TOK_COMMENT, prior_, begin, at_));
237  prior_ = at_;
238  }
239  } else if ('/' == c && '*' == getChar(at_+1)) {
240  const size_t begin = at_;
241  at_ += 2;
242  while (EOF != (c = getChar(at_))) {
243  if (getChar(at_) == '*' && getChar(at_+1) == '/') {
244  at_ = at_ + 2;
245  break;
246  }
247  ++at_;
248  }
249  if (skipCommentTokens_) {
250  makeNextToken();
251  } else {
252  tokens_.push_back(Token(TOK_COMMENT, prior_, begin, at_));
253  prior_ = at_;
254  }
255  } else if (isalpha(c) || c=='_') {
256  const size_t begin = at_++;
257  while (isalnum(c = getChar(at_)) || '_'==c)
258  ++at_;
259  tokens_.push_back(Token(TOK_WORD, prior_, begin, at_));
260  prior_ = at_;
261  } else if ('('==c || '{'==c || '['==c) {
262  ++at_;
263  tokens_.push_back(Token(TOK_LEFT, prior_, at_-1, at_));
264  prior_ = at_;
265  } else if (')'==c || '}'==c || ']'==c) {
266  ++at_;
267  tokens_.push_back(Token(TOK_RIGHT, prior_, at_-1, at_));
268  prior_ = at_;
269  } else if (isdigit(c) || (('-'==c || '+'==c) && isdigit(getChar(at_+1)))) {
270  const size_t begin = at_;
271  if (!isdigit(c))
272  ++at_;
273  if ('0' == getChar(at_) && 'x' == getChar(at_+1)) {
274  at_ += 2;
275  while (isxdigit(getChar(at_)))
276  ++at_;
277  } else if ('0' == getChar(at_) && 'b' == getChar(at_+1)) {
278  at_ += 2;
279  while (strchr("01", getChar(at_)))
280  ++at_;
281  } else if ('0' == getChar(at_)) {
282  ++at_;
283  while ((c = getChar(at_)) >= '0' && c <= '7')
284  ++at_;
285  } else {
286  ++at_;
287  while (isdigit(getChar(at_)))
288  ++at_;
289  }
290  tokens_.push_back(Token(TOK_NUMBER, prior_, begin, at_));
291  prior_ = at_;
292  } else if ('#' == c) {
293  const size_t begin = at_;
294  at_ = content_.characterIndex(content_.lineIndex(at_) + 1);
295  while (at_>=2 && at_ < content_.nCharacters() && getChar(at_-2)=='\\' && getChar(at_-1)=='\n')
296  at_ = content_.characterIndex(content_.lineIndex(at_) + 1);
297  if (skipPreprocessorTokens_) {
298  makeNextToken();
299  } else {
300  tokens_.push_back(Token(TOK_CPP, prior_, begin, at_));
301  prior_ = at_;
302  }
303  } else if (('<' == c && getChar(at_+1) == '<' && getChar(at_+2) == '=') ||
304  ('>' == c && getChar(at_+1) == '>' && getChar(at_+2) == '=') ||
305  ('<' == c && getChar(at_+1) == '=' && getChar(at_+2) == '>') ||
306  ('-' == c && getChar(at_+1) == '>' && getChar(at_+2) == '*')) {
307  tokens_.push_back(Token(TOK_OTHER, prior_, at_, at_+3));
308  at_ += 3;
309  prior_ = at_;
310  } else if ((getChar(at_+1) == '=' && strchr("|&^*/%+-!<>=", c)) ||
311  ('|' == c && getChar(at_+1) == '|') ||
312  ('&' == c && getChar(at_+1) == '&') ||
313  ('<' == c && getChar(at_+1) == '<') ||
314  ('>' == c && getChar(at_+1) == '>') ||
315  ('.' == c && getChar(at_+1) == '*') ||
316  ('+' == c && getChar(at_+1) == '+') ||
317  ('-' == c && getChar(at_+1) == '-') ||
318  ('-' == c && getChar(at_+1) == '>') ||
319  (':' == c && getChar(at_+1) == ':')) {
320  tokens_.push_back(Token(TOK_OTHER, prior_, at_, at_+2));
321  at_ += 2;
322  prior_ = at_;
323  } else {
324  tokens_.push_back(Token(TOK_OTHER, prior_, at_, at_+1));
325  ++at_;
326  prior_ = at_;
327  }
328 }
