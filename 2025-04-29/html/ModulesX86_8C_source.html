152 bool
153 MatchRetPadPush::match(const Partitioner::ConstPtr &partitioner, Address anchor) {
154  ASSERT_not_null(partitioner);
155 
156  // RET (prior to anchor) must already exist in the CFG/AUM
157  // The RET instruction can be 1 or 3 bytes.
158  SgAsmX86Instruction *ret = NULL;
159  if (partitioner->instructionExists(anchor-1) &&
160  (ret = isSgAsmX86Instruction(partitioner->discoverInstruction(anchor-1))) &&
161  ret->get_kind() == x86_ret && ret->get_size()==1) {
162  // found RET
163  } else if (partitioner->instructionExists(anchor-3) &&
164  (ret = isSgAsmX86Instruction(partitioner->discoverInstruction(anchor-3))) &&
165  ret->get_kind() == x86_ret && ret->get_size()==3) {
166  // found RET x
167  } else {
168  return false;
169  }
170 
171  // Optional padding (NOP; or INT3; or MOV EDI,EDI)
172  Address padVa = anchor;
173  if (partitioner->instructionExists(padVa))
174  return false;
175  SgAsmX86Instruction *pad = isSgAsmX86Instruction(partitioner->discoverInstruction(padVa));
176  if (!pad)
177  return false;
178  if (pad->get_kind() != x86_nop && pad->get_kind() != x86_int3 && !matchMovDiDi(partitioner, pad))
179  pad = NULL;
180 
181  // PUSH x
182  Address pushVa = padVa + (pad ? pad->get_size() : 0);
183  if (partitioner->instructionExists(pushVa))
184  return false;
185  SgAsmX86Instruction *push = isSgAsmX86Instruction(partitioner->discoverInstruction(pushVa));
186  if (!push || push->get_kind()!=x86_push)
187  return false;
188 
189  // Looks good
190  function_ = Function::instance(pushVa, SgAsmFunction::FUNC_PATTERN);
191  function_->reasonComment("matched RET [x]; <padding>; PUSH <y>");
192  return true;

195 bool
196 FunctionReturnDetector::operator()(bool chain, const Args &args) {
197  if (chain) {
198  if (args.bblock->isFunctionReturn().isCached()) // property is already computed?
199  return chain;
200  if (args.bblock->isEmpty()) {
201  args.bblock->isFunctionReturn() = false; // empty blocks are never considered returns
202  return chain;
203  }
204  SgAsmX86Instruction *lastInsn = isSgAsmX86Instruction(args.bblock->instructions().back());
205  if (NULL==lastInsn)
206  return chain; // defer if not x86
207  if (lastInsn->get_kind()!=x86_ret && lastInsn->get_kind()!=x86_retf)
208  return chain; // defer if not a return instruction
209 
210  // A RET/RETF that has a single successor that is concrete probably isn't a real function return. Sometimes these
211  // instructions are used to hide unconditional branches, like "PUSH label; RET".
212  bool isComplete = false;
213  std::vector<Address> concreteSuccessors = args.partitioner->basicBlockConcreteSuccessors(args.bblock, &isComplete);
214  if (1==concreteSuccessors.size() && isComplete) {
215  args.bblock->isFunctionReturn() = false;
216  return chain;
217  }
218 
219  // Must be a function return
220  args.bblock->isFunctionReturn() = true;
221  }
222  return chain;

407 static const char*
408 isInvalidTarget(Address target, const AddressInterval &targetLimits, const Partitioner::ConstPtr &partitioner) {
409  ASSERT_not_null(partitioner);
410  if (!targetLimits.contains(target)) {
411  return "target is not within target limits";
412  } else if (!partitioner->memoryMap()->at(target).require(MemoryMap::EXECUTABLE).exists()) {
413  return "target is not in executable memory";
414  } else {
415  SgAsmInstruction *targetInsn = partitioner->instructionProvider()[target];
416  if (!targetInsn || partitioner->architecture()->isUnknown(targetInsn)) {
417  return "target does not point to a valid instruction";
418  } else {
419  std::vector<SgAsmInstruction*> overlappingInsns = partitioner->instructionsOverlapping(target);
420  for (SgAsmInstruction *insn: overlappingInsns) {
421  if (insn->get_address() != target)
422  return "target overlaps existing instruction";
423  }
424  }
425  }
426  return nullptr;

429 std::vector<Address>
430 scanCodeAddressTable(const Partitioner::ConstPtr &partitioner, AddressInterval &tableLimits /*in,out*/,
431  const AddressInterval &targetLimits, const SwitchSuccessors::EntryType tableEntryType,
432  const size_t tableEntrySizeBytes, const Sawyer::Optional<Address> probableStartVa, const size_t nSkippable) {
433  ASSERT_require(tableEntrySizeBytes > 0 && tableEntrySizeBytes <= sizeof(Address));
434  Sawyer::Message::Stream debug(mlog[DEBUG]);
435  if (debug) {
436  namespace Str = stringify::Rose::BinaryAnalysis::Partitioner2::ModulesX86::SwitchSuccessors;
437  debug <<"scanCodeAddressTable: tableLimits = " <<StringUtility::addrToString(tableLimits)
438  <<", targetLimits = " <<StringUtility::addrToString(targetLimits)
439  <<", entry size = " <<StringUtility::plural(tableEntrySizeBytes, "bytes")
440  <<", entry type " <<Str::EntryType(tableEntryType)
441  <<(probableStartVa ? ", probable start = " + StringUtility::addrToString(*probableStartVa) : "")
442  <<", nSkippable = " <<nSkippable <<"\n";
443  }
444 
445  std::vector<Address> tableEntries; // decoded entries in the table
446 
447  if (tableLimits.isEmpty() || targetLimits.isEmpty())
448  return tableEntries;
449  const MemoryMap::Ptr map = partitioner->memoryMap();
450 
451  // Look forward from the probable start address, but possibly allow the table to start with some addresses that are out of
452  // range (which are skipped and not officially part of the table).
453  Address actualStartVa = probableStartVa.orElse(tableLimits.least()); // adjusted later if entries are skipped
454  size_t nSkippedEntries = 0;
455  while (1) {
456  // Read table entry to get target address
457  uint8_t bytes[sizeof(Address)];
458  Address tableEntryVa = actualStartVa + tableEntries.size() * tableEntrySizeBytes;
459  if (!tableLimits.contains(AddressInterval::baseSize(tableEntryVa, tableEntrySizeBytes))) {
460  SAWYER_MESG(debug) <<" entry at " <<StringUtility::addrToString(tableEntryVa) <<" falls outside table boundary\n";
461  break;
462  }
463  if (tableEntrySizeBytes != (map->at(tableEntryVa).limit(tableEntrySizeBytes)
464  .require(MemoryMap::READABLE).prohibit(MemoryMap::WRITABLE).read(bytes).size())) {
465  SAWYER_MESG(debug) <<" entry at " <<StringUtility::addrToString(tableEntryVa) <<" is not read-only\n";
466  break;
467  }
468  Address target = 0;
469  for (size_t i=0; i<tableEntrySizeBytes; ++i)
470  target |= bytes[i] << (8*i); // x86 is little endian
471  switch (tableEntryType) {
472  case SwitchSuccessors::ABSOLUTE:
473  break;
474  case SwitchSuccessors::RELATIVE:
475  // target is the table entry plus the table address, careful for overflow
476  SAWYER_MESG(debug) <<" entry at " <<StringUtility::addrToString(tableEntryVa)
477  <<" raw value = " <<StringUtility::addrToString(target) <<"\n";
478  target = (BitOps::signExtend<Address>(target, 8*tableEntrySizeBytes) + actualStartVa) &
479  BitOps::lowMask<Address>(partitioner->instructionProvider().instructionPointerRegister().nBits());
480  break;
481  }
482 
483  // Save or skip the table entry
484  const char *invalidReason = isInvalidTarget(target, targetLimits, partitioner);
485  if (!invalidReason) {
486  tableEntries.push_back(target);
487  } else if (tableEntries.empty() && nSkippedEntries < nSkippable) {
488  ++nSkippedEntries;
489  actualStartVa += tableEntrySizeBytes;
490  SAWYER_MESG(debug) <<" entry at " <<StringUtility::addrToString(tableEntryVa) <<" is skipped"
491  <<": target is " <<StringUtility::addrToString(target)
492  <<" (" <<invalidReason <<")\n";
493  } else {
494  SAWYER_MESG(debug) <<" entry at " <<StringUtility::addrToString(tableEntryVa) <<" is invalid"
495  <<": target is " <<StringUtility::addrToString(target)
496  <<" (" <<invalidReason <<")\n";
497  break;
498  }
499  }
500 
501  // Look backward from the start address to see if there are valid table entries at negative indexes, even if we didn't find
502  // any by looking forward from the probable start address. Be careful of over/under flows.
503  size_t nBackwardEntries = 0;
504  if (0 == nSkippedEntries) {
505  while (actualStartVa >= tableEntrySizeBytes &&
506  tableLimits.contains(AddressInterval::baseSize(actualStartVa-tableEntrySizeBytes, tableEntrySizeBytes))) {
507  uint8_t bytes[sizeof(Address)];
508  Address tableEntryVa = actualStartVa - tableEntrySizeBytes;
509  if (tableEntrySizeBytes != (map->at(tableEntryVa).limit(tableEntrySizeBytes)
510  .require(MemoryMap::READABLE).prohibit(MemoryMap::WRITABLE).read(bytes).size()))
511  break;
512  Address target = 0;
513  for (size_t i=0; i<tableEntrySizeBytes; ++i)
514  target |= bytes[i] << (8*i); // x86 is little endian
515 
516  // Save entry if valid, otherwise we've reached the beginning of the table
517  const char *invalidReason = isInvalidTarget(target, targetLimits, partitioner);
518  if (invalidReason) {
519  SAWYER_MESG(debug) <<" entry at " <<StringUtility::addrToString(tableEntryVa) <<" is invalid"
520  <<": target is " <<StringUtility::addrToString(target)
521  <<" (" <<invalidReason <<")\n";
522  break;
523  } else {
524  tableEntries.insert(tableEntries.begin(), target);
525  actualStartVa -= tableEntrySizeBytes;
526  ++nBackwardEntries;
527  SAWYER_MESG(debug) <<" entry at " <<StringUtility::addrToString(tableEntryVa) <<" found by backward search\n";
528  }
529  }
530  }
531 
532  if (debug) {
533  if (tableEntries.empty()) {
534  debug <<" no valid table entries found\n";
535  } else {
536  if (nSkippedEntries > 0)
537  debug <<" skipped " <<StringUtility::plural(nSkippedEntries, "leading table entries") <<"\n";
538  if (tableEntries.size() > nBackwardEntries) {
539  debug <<" found " <<StringUtility::plural(tableEntries.size() - nBackwardEntries, "entries")
540  <<" by scanning forward\n";
541  }
542  if (nBackwardEntries > 0)
543  debug <<" found " <<StringUtility::plural(nBackwardEntries, "entries") <<" by scanning backward\n";
544  debug <<" total entries found: " <<tableEntries.size() <<"\n";
545  int idx = -nBackwardEntries;
546  for (Address target: tableEntries) {
547  debug <<" entry[" <<boost::format("%4d") % idx <<"]"
548  <<" at " <<StringUtility::addrToString(actualStartVa + idx * tableEntrySizeBytes)
549  <<" = " <<StringUtility::addrToString(target) <<"\n";
550  ++idx;
551  }
552  }
553  }

556  // the table. For an example, look at tetris.exe compiled with MSVC 2010 (md5sum 30f1442a16d0275c2db4f52e9c78b5cd): eax is
557  // the zero-origin value of the switch expression, which is looked up in the byte array at 0x00401670, which in turn is
558  // used to index into the jump address array at 0x004165c. Only do this for small tables, otherwise it will eat up all
559  // kinds of stuff.
560  // 0x0040150a: 3d a2 00 00 00 |=.... | cmp eax, 0x000000a2
561  // 0x0040150f: 0f 87 86 00 00 00 |...... | ja 0x0040159b
562  // 0x00401515: 0f b6 90 70 16 40 00 |...p.@. | movzx edx, BYTE PTR ds:[eax + 0x00401670]
563  // 0x0040151c: ff 24 95 5c 16 40 00 |.$.\.@. | jmp DWORD PTR ds:[0x0040165c + edx*0x04]
564  //
565  // [0x0040165c,0x0040166f]: uint32_t addresses[5] = { <target addresses> };
566  //
567  // [0x00401670,0x00401712]: uint8_t index[0xa3] = { <values 0..4> };
568  //
569  // Let's hope that the compiler doesn't combine the offset table technique with negative offsets.
570  size_t nIndexes = 0;
571  if (!tableEntries.empty() && 0 == nBackwardEntries) {
572  Address indexArrayStartVa = actualStartVa + tableEntries.size() * tableEntrySizeBytes;
573  Address indexArrayCurrentVa = indexArrayStartVa;
574  if (tableEntries.size() <= 16 /*arbitrarily small tables*/) {
575  while (indexArrayCurrentVa <= tableLimits.greatest()) {
576  uint8_t byte;
577  if (!map->at(indexArrayCurrentVa).limit(1).require(MemoryMap::READABLE).prohibit(MemoryMap::WRITABLE).read(&byte))
578  break;
579  if (byte >= tableEntries.size())
580  break;
581  if (indexArrayCurrentVa == tableLimits.greatest())
582  break; // avoid overflow
583  ++indexArrayCurrentVa;
584  ++nIndexes;
585  }
586  }
587  if (nIndexes > 0)
588  SAWYER_MESG(debug) <<" found " <<StringUtility::plural(nIndexes, "post table indexes") <<"\n";
589  }
590 
591  // Return values
592  AddressInterval actualTableLocation =
593  AddressInterval::baseSize(actualStartVa, tableEntries.size() * tableEntrySizeBytes + nIndexes);
594  ASSERT_require(tableLimits.contains(actualTableLocation));
595  tableLimits = actualTableLocation;
596  SAWYER_MESG(debug) <<" actual table location = " <<StringUtility::addrToString(actualTableLocation) <<"\n";
597  return tableEntries;
598 }

697 // Matches:
698 // LEA reg1, [RIP+const1] where RIP+const1 is the address of the table
699 // MOV EDI, EDI or other optional no-op
700 // MOVSXD reg2, [reg1 + ...]
701 // ADD reg2, reg1
702 // JMP reg2
703 //
704 // This pattern is used for position independent AMD64 code. The table, starting at base, contains offsets from the start of
705 // the table to the target basic block. Representative specimen: wget2 email-20201209 df4ea624fc0d2a890c5cf6d0344ece52
706 bool
707 SwitchSuccessors::matchPattern3(const Partitioner::ConstPtr &partitioner, const BasicBlock::Ptr &bb, SgAsmInstruction *jmp) {
708  ASSERT_not_null(bb);
709  ASSERT_always_not_null(jmp);
710  size_t nInsns = bb->nInstructions();
711  if (nInsns < 4)
712  return false;
713 
714  // ADD reg2, reg1
715  SgAsmX86Instruction *add = isSgAsmX86Instruction(bb->instructions()[nInsns-2]);
716  if (!add || add->get_kind() != x86_add)
717  return false;
718  const SgAsmExpressionPtrList &addArgs = add->get_operandList()->get_operands();
719  if (addArgs.size() != 2)
720  return false;
721  SgAsmDirectRegisterExpression *addDst = isSgAsmDirectRegisterExpression(addArgs[0]);
722  SgAsmDirectRegisterExpression *addSrc = isSgAsmDirectRegisterExpression(addArgs[1]);
723  if (!addDst || !addSrc)
724  return false;
725 
726  // MOVSXD reg2, [reg1 + ...]
727  SgAsmX86Instruction *movsxd = isSgAsmX86Instruction(bb->instructions()[nInsns-3]);
728  if (!movsxd || movsxd->get_kind() != x86_movsxd)
729  return false;
730  const SgAsmExpressionPtrList &movsxdArgs = movsxd->get_operandList()->get_operands();
731  if (movsxdArgs.size() != 2)
732  return false;
733  SgAsmDirectRegisterExpression *movsxdDst = isSgAsmDirectRegisterExpression(movsxdArgs[0]);
734  SgAsmMemoryReferenceExpression *movsxdSrc = isSgAsmMemoryReferenceExpression(movsxdArgs[1]);
735  if (!movsxdDst || !movsxdSrc || movsxdDst->get_descriptor() != addDst->get_descriptor())
736  return false;
737  SgAsmBinaryAdd *movsxdAddr = isSgAsmBinaryAdd(movsxdSrc->get_address());
738  if (!movsxdAddr)
739  return false;
740  SgAsmDirectRegisterExpression *movsxdSrcReg = isSgAsmDirectRegisterExpression(movsxdAddr->get_lhs());
741  if (!movsxdSrcReg)
742  movsxdSrcReg = isSgAsmDirectRegisterExpression(movsxdAddr->get_rhs());
743  if (!movsxdSrcReg || movsxdSrcReg->get_descriptor() != addSrc->get_descriptor())
744  return false;
745 
746  // MOV EDI EDI (optional)
747  bool foundNop = false;
748  do {
749  SgAsmX86Instruction *mov = isSgAsmX86Instruction(bb->instructions()[nInsns-4]);
750  if (!mov || mov->get_kind() != x86_mov)
751  break;
752  const SgAsmExpressionPtrList &movArgs = mov->get_operandList()->get_operands();
753  if (movArgs.size() != 2)
754  break;
755  SgAsmDirectRegisterExpression *movDst = isSgAsmDirectRegisterExpression(movArgs[0]);
756  SgAsmDirectRegisterExpression *movSrc = isSgAsmDirectRegisterExpression(movArgs[1]);
757  if (!movDst || !movSrc)
758  break;
759  if (movDst->get_descriptor() != movSrc->get_descriptor())
760  break;
761  foundNop = true;
762  } while (false);
763 
764  // LEA reg1 [RIP + const1]
765  if (foundNop && nInsns < 5)
766  return false;
767  SgAsmX86Instruction *lea = isSgAsmX86Instruction(bb->instructions()[nInsns - (foundNop ? 5 : 4)]);
768  if (!lea || lea->get_kind() != x86_lea)
769  return false;
770  const SgAsmExpressionPtrList &leaArgs = lea->get_operandList()->get_operands();
771  if (leaArgs.size() != 2)
772  return false;
773  SgAsmDirectRegisterExpression *leaDst = isSgAsmDirectRegisterExpression(leaArgs[0]);
774  if (!leaDst || leaDst->get_descriptor() != addSrc->get_descriptor())
775  return false;
776  SgAsmMemoryReferenceExpression *leaSrc = isSgAsmMemoryReferenceExpression(leaArgs[1]);
777  if (!leaSrc)
778  return false;
779  SgAsmBinaryAdd *leaSrcAdd = isSgAsmBinaryAdd(leaSrc->get_address());
780  SgAsmDirectRegisterExpression *leaSrcRip = isSgAsmDirectRegisterExpression(leaSrcAdd->get_lhs());
781  SgAsmIntegerValueExpression *leaSrcOffset = NULL;
782  if (leaSrcRip && leaSrcRip->get_descriptor() == partitioner->instructionProvider().instructionPointerRegister()) {
783  leaSrcOffset = isSgAsmIntegerValueExpression(leaSrcAdd->get_rhs());
784  } else {
785  leaSrcRip = isSgAsmDirectRegisterExpression(leaSrcAdd->get_rhs());
786  if (!leaSrcRip || leaSrcRip->get_descriptor() != partitioner->instructionProvider().instructionPointerRegister())
787  return false;
788  leaSrcOffset = isSgAsmIntegerValueExpression(leaSrcAdd->get_lhs());
789  }
790  if (!leaSrcOffset)
791  return false;
792 
793  tableVa_ = (lea->get_address() + lea->get_size() + leaSrcOffset->get_absoluteValue()) &
794  BitOps::lowMask<Address>(partitioner->instructionProvider().instructionPointerRegister().nBits());
795  if (!tableVa_)
796  return false;
797 
798  entryType_ = RELATIVE;
799  entrySizeBytes_ = 4;
800  return true;
801 }

803 // Match a basic block whose final instruction pointer has a symbolic value matching:
804 // (add[u64] (signextend[u64] 64, v1[u32]) c1[u64])
805 // where
806 // v1 is any symbolic variable
807 // c1 is a constant that is inside a section of memory mapped read-only no-execute no-write (typically named ".rodata")
808 bool
809 SwitchSuccessors::matchPattern4(const Partitioner::ConstPtr &partitioner, const BasicBlock::Ptr &bblock) {
810  ASSERT_not_null(partitioner);
811  ASSERT_not_null(bblock);
812  using namespace SymbolicExpression;
813 
814  const BasicBlockSemantics &semantics = bblock->semantics();
815  const InstructionSemantics::BaseSemantics::State::Ptr state = semantics.finalState();
816  if (!semantics.operators || !state)
817  return false; // only works when instruction semantics are enabled
818 
819  const RegisterDescriptor IP = partitioner->architecture()->registerDictionary()->instructionPointerRegister();
820  const auto ip = semantics.operators->peekRegister(IP);
821  const auto add = InstructionSemantics::SymbolicSemantics::SValue::promote(ip)->get_expression();
822 
823  if (!add || !add->isOperator(OP_ADD) || add->nBits() != 64 || !add->isIntegerExpr())
824  return false;
825 
826  const auto sext = add->child(0);
827  if (!sext || !sext->isOperator(OP_SEXTEND) || sext->nBits() != 64)
828  return false;
829 
830  const auto sixtyfour = sext->child(0);
831  if (!sixtyfour || sixtyfour->toUnsigned().orElse(0) != 64)
832  return false;
833 
834  const auto v1 = sext->child(1);
835  if (!v1 || !v1->isIntegerVariable() || v1->nBits() != 32)
836  return false;
837 
838  const auto c1 = add->child(1);
839  if (!c1 || !c1->isIntegerConstant() || c1->nBits() != 64)
840  return false;
841  Address tableAddr = *c1->toUnsigned();
842 
843  auto segments = partitioner->memoryMap()->at(tableAddr)
844  .require(MemoryMap::READABLE)
845  .prohibit(MemoryMap::WRITABLE | MemoryMap::EXECUTABLE)
846  .segments();
847  if (segments.begin() == segments.end())
848  return false; // table addr is not mapped, or not read-only
849 
850  // Matched
851  tableVa_ = tableAddr;
852  entryType_ = RELATIVE;
853  entrySizeBytes_ = 4;
854  return true;
855 }

857 bool
858 SwitchSuccessors::matchPatterns(const Partitioner::ConstPtr &partitioner, const BasicBlock::Ptr &bblock) {
859  ASSERT_not_null(bblock);
860 
861  // Block always ends with JMP with one argument.
862  size_t nInsns = bblock->nInstructions();
863  SgAsmX86Instruction *jmp = isSgAsmX86Instruction(bblock->instructions()[nInsns-1]);
864  if (!jmp || jmp->get_kind() != x86_jmp)
865  return false;
866  const SgAsmExpressionPtrList &jmpArgs = jmp->get_operandList()->get_operands();
867  if (jmpArgs.size() != 1)
868  return false;
869 
870  // JMP [const1 + ...] where base is assumed to be the table address
871  if (matchPattern1(jmpArgs[0]))
872  return true;
873 
874  // MOV reg1, ...
875  // JMP reg1
876  if (matchPattern2(bblock, jmp))
877  return true;
878 
879  // LEA reg1, [RIP+const1] where RIP+const1 is the address of the table
880  // MOV EDI, EDI or other optional no-op
881  // MOVSXD reg2, [reg1 + ...]
882  // ADD reg2, reg1
883  // JMP reg2
884  if (matchPattern3(partitioner, bblock, jmp))
885  return true;
886 
887  // Symbolic expression for instruction pointer:
888  // (add[u64] (signextend[u64] 0x40[u32] v1[u32]) c1[u64])
889  if (matchPattern4(partitioner, bblock))
890  return true;
891 
892  // no matching pattern
893  return false;
894 }

896 // A "switch" statement is a computed jump consisting of a base address and a register offset.
897 bool
898 SwitchSuccessors::operator()(const bool chain, const Args &args) {
899  ASSERT_not_null(args.bblock);
900  if (!chain)
901  return false;
902  const size_t nInsns = args.bblock->nInstructions();
903  if (nInsns < 1)
904  return chain;
905  if (!matchPatterns(args.partitioner, args.bblock))
906  return chain;
907  ASSERT_require(tableVa_);
908  Sawyer::Message::Stream debug(mlog[DEBUG]);
909 
910  // Set some limits on the location of the target address table, besides those restrictions that will be imposed during the
911  // table-reading loop (like table is mapped read-only).
912  const size_t wordSizeBytes = args.partitioner->architecture()->bytesPerWord();
913  const AddressInterval whole = AddressInterval::hull(0, BitOps::lowMask<Address>(8*wordSizeBytes));
914  AddressInterval tableLimits;
915  SgAsmInstruction *lastInsn = args.bblock->instructions().back();
916  if (*tableVa_ > lastInsn->get_address()) {
917  // table is after the jmp instruction
918  tableLimits = AddressInterval::hull(std::min(lastInsn->get_address() + lastInsn->get_size(), *tableVa_), whole.greatest());
919  } else {
920  // table is before the jmp instruction
921  tableLimits = AddressInterval::hull(0, lastInsn->get_address());
922  }
923 
924  // Set some limits on allowable target addresses contained in the table, besides those restrictions that will be imposed
925  // during the table-reading loop (like targets must be mapped with execute permission).
926  AddressInterval targetLimits = AddressInterval::whole();
927  std::vector<Function::Ptr> functions = args.partitioner->functions();
928  {
929  // If there's a function that precedes this basic block (possibly the one that will eventually own this block) then the
930  // switch statement targets are almost certainly not before the beginning of that function.
931  Function::Ptr needle = Function::instance(args.bblock->address());
932  std::vector<Function::Ptr>::iterator prevFunctionIter = std::lower_bound(functions.begin(), functions.end(),
933  needle, sortFunctionsByAddress);
934  if (prevFunctionIter != functions.end()) {
935  while (prevFunctionIter != functions.begin() && (*prevFunctionIter)->address() > needle->address())
936  --prevFunctionIter;
937  Function::Ptr prevFunction = *prevFunctionIter;
938  if (prevFunction->address() <= needle->address()) {
939  targetLimits = targetLimits & AddressInterval::hull(prevFunction->address(), targetLimits.greatest());
940  } else {
941  // no prior function, so assume that the "cases" are at higher addresses than the "switch"
942  targetLimits = AddressInterval::hull(args.bblock->fallthroughVa(), AddressInterval::whole().greatest());
943  }
944  }
945  }
946  {
947  // If there's a function that follows this basic block then the switch targets are almost certainly not after the
948  // beginning of that function.
949  Function::Ptr needle = Function::instance(args.bblock->fallthroughVa());
950  std::vector<Function::Ptr>::iterator nextFunctionIter = std::lower_bound(functions.begin(), functions.end(),
951  needle, sortFunctionsByAddress);
952  if (nextFunctionIter != functions.end()) {
953  Function::Ptr nextFunction = *nextFunctionIter;
954  targetLimits = targetLimits & AddressInterval::hull(0, nextFunction->address()-1);
955  }
956  }
957  SAWYER_MESG(debug) <<"switch table target limits: " <<StringUtility::addrToString(targetLimits) <<"\n";
958  if (targetLimits.isEmpty())
959  return chain; // not even room for one case label
960 
961  // Read the table
962  static const size_t maxSkippable = 1; // max number of invalid table entries to skip; arbitrary
963  std::vector<Address> tableEntries = scanCodeAddressTable(args.partitioner, tableLimits /*in,out*/,
964  targetLimits, entryType_, entrySizeBytes_,
965  *tableVa_, maxSkippable);
966  if (tableEntries.empty())
967  return chain;
968 
969  // Replace basic block's successors with the new ones we found.
970  std::set<Address> successors(tableEntries.begin(), tableEntries.end());
971  args.bblock->successors().clear();
972  for (Address va: successors)
973  args.bblock->insertSuccessor(va, wordSizeBytes*8);
974 
975  // Create a data block for the offset table and attach it to the basic block
976  size_t nTableEntries = tableLimits.size() / entrySizeBytes_;
977  SgAsmType *tableEntryType = SageBuilderAsm::buildTypeU(8*entrySizeBytes_);
978  SgAsmType *tableType = SageBuilderAsm::buildTypeVector(nTableEntries, tableEntryType);
979  DataBlock::Ptr addressTable = DataBlock::instance(tableLimits.least(), tableType);
980  addressTable->comment("x86 'switch' statement's 'case' address table");
981  args.bblock->insertDataBlock(addressTable);
982 
983  // Debugging
984  if (debug) {
985  using namespace StringUtility;
986  debug <<"ModulesX86::SwitchSuccessors: found \"switch\" statement\n";
987  debug <<" basic block: " <<addrToString(args.bblock->address()) <<"\n";
988  debug <<" instruction: " <<args.bblock->instructions()[nInsns-1]->toString() <<"\n";
989  debug <<" table va: " <<addrToString(tableLimits.least()) <<"\n";
990  debug <<" table size: " <<plural(tableEntries.size(), "entries")
991  <<", " <<plural(tableLimits.size(), "bytes") <<"\n";
992  debug <<" successors: " <<plural(successors.size(), "distinct addresses") <<"\n";
993  debug <<" ";
994  for (Address va: successors)
995  debug <<" " <<addrToString(va);
996  debug <<"\n";
997  }
998 
999  return chain;
1000 }
