35 static QString escapeCpp(const QString& s, bool rep)
36 {
37  if ( !rep )
38  return s;
39 
40  QString es;
41 
42  for ( int i = 0; i < s.count(); ++i )
43  {
44  if ( (s.at(i) == '\\') && ((i + 1) < s.count()) )
45  {
46  QChar c = s.at(++i);
47 
48  if ( c == '\\' )
49  es += '\\';
50  else if ( c == 't' )
51  es += '\t';
52  else if ( c == 'n' )
53  es += '\n';
54  else if ( c == 'r' )
55  es += '\r';
56  else if ( c == '0' )
57  es += '\0';
58 
59  } else {
60  es += s.at(i);
61  }
62  }
63 
64  //qDebug("\"%s\" => \"%s\"", qPrintable(s), qPrintable(es));
65 
66  return es;
67 }

216 bool QSearchReplacePanel::eventFilter(QObject *o, QEvent *e)
217 {
218  int kc;
219 
220  if ( o == leFind || o == leReplace )
221  {
222  switch ( e->type() )
223  {
224  /*
225  case QEvent::FocusIn :
226  leFind->grabKeyboard();
227  break;
228 
229  case QEvent::FocusOut :
230  leFind->releaseKeyboard();
231  break;
232  */
233 
234  case QEvent::KeyPress :
235 
236  kc = static_cast<QKeyEvent*>(e)->key();
237 
238  if ( (kc == Qt::Key_Enter) || (kc == Qt::Key_Return) )
239  {
240  //on_leFind_returnPressed();
241  on_leFind_returnPressed(Qt::ShiftModifier & static_cast<QKeyEvent*>(e)->modifiers());
242  return true;
243  } else if ( kc == Qt::Key_Escape) {
244  if ( cbReplace->isChecked() )
245  display(1,false);
246  else
247  display(0,false);
248  return true;
249  } else if ( kc == Qt::Key_Tab || kc == Qt::Key_Backtab ) {
250  if ( cbReplace->isChecked() )
251  {
252  if ( leFind->hasFocus() )
253  leReplace->setFocus();
254  else
255  leFind->setFocus();
256  }
257  return true;
258  }
259  break;
260 
261  default:
262  break;
263  }
264  }
265 
266  return QWidget::eventFilter(o, e);
267 }

269 void QSearchReplacePanel::on_leFind_textEdited(const QString& text)
270 {

277 
278  m_search->setSearchText(text);

281  {
282  QDocumentCursor c = cur;
283  c.setColumnNumber(qMin(c.anchorColumnNumber(), c.columnNumber()));
284 
285  m_search->setCursor(c);
286  }
287  } else {
288  // TODO : make incremental search optional
289  init();
290  }
291 
292  if ( text.isEmpty() )
293  {
294  leFind->setStyleSheet(QString());
295  return;
296  }
297 
298  m_search->setOption(QDocumentSearch::Silent, true);
299 
300  find(0);
301 
302  m_search->setOption(QDocumentSearch::Silent, false);
303 
304  if ( m_search->cursor().isNull() )
305  {
306  leFind->setStyleSheet("QLineEdit { background: red; color : white; }");
307 
308  if ( hadSearch )
309  {
310  m_search->setCursor(cur);
311 
312  // figure out whether other matches are availables
313  QDocumentSearch::Options opts = m_search->options();
314  opts &= ~QDocumentSearch::HighlightAll;
315  opts |= QDocumentSearch::Silent;
316 
317  QDocumentSearch temp(editor(), text, opts);
318  temp.setOrigin(QDocumentCursor());
319  temp.setScope(m_search->scope());
320  temp.next(true);
321 
322  if ( temp.cursor().isValid() )
323  {
324  // other match found from doc start
325  leFind->setStyleSheet("QLineEdit { background: yellow; color : black; }");
326  m_search->setCursor(cur.document()->cursor(0,0));
327  find(0);
328  }
329  }
330  } else {
331  leFind->setStyleSheet(QString());
332  editor()->setCursor(m_search->cursor());
333  }
