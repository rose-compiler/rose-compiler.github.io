373  // Warning: Caller should create a nullPtrSolver transaction if necessary
374  bool isNullDeref(const SymbolicExpression::Ptr &expr) {
375  ASSERT_not_null(expr);
376  Sawyer::Message::Stream debug(fpAnalyzer_->mlog[DEBUG]);
377  SAWYER_MESG(debug) <<" isNullDeref addr = " <<*expr <<"\n";
378  SmtSolver::Ptr solver = nullPtrSolver();
379 
380  // Instead of using SymbolicExpression::mustEqual and SymbolicExpression::mayEqual, we always call the SMT solver. This
381  // is so that if a null pointer is found, the callback will be able to get the evidence from the solver. The problem
382  // with mustEqual and mayEqual is threefold: (1) these functions can sometimes make the determination themselves
383  // without invoking an SMT solver thus the solver would not be updated with evidence, (2) these functions might invoke
384  // a user-defined callback that does almost anything, (3) when these functions invoke an SMT solver they protect it
385  // with a transaction so the evidence would be gone by time we get control back.
386  bool retval = false;
387  switch (fpAnalyzer_->settings().nullDeref.mode) {
388  case FeasiblePath::MAY:
389  if (solver) {
390  SymbolicExpression::Ptr assertion =
391  SymbolicExpression::makeLt(expr,
392  SymbolicExpression::makeIntegerConstant(expr->nBits(),
393  fpAnalyzer_->settings().nullDeref.minValid));
394  solver->insert(assertion);
395  retval = SmtSolver::SAT_YES == solver->check();
396  } else {
397  retval =
398  SymbolicExpression::makeLt(expr,
399  SymbolicExpression::makeIntegerConstant(expr->nBits(),
400  fpAnalyzer_->settings().nullDeref.minValid))->mayEqual(SymbolicExpression::makeIntegerConstant(expr->nBits(), 1));
401  }
402  break;
403 
404  case FeasiblePath::MUST:
405  // Check for constness of the address expression first since that doesn't involve an SMT solver and we might
406  // be able to return quickly.
407  if (!fpAnalyzer_->settings().nullDeref.constOnly || isConstExpr(expr)) {
408  if (solver) {
409  SymbolicExpression::Ptr assertion = SymbolicExpression::makeGe(expr, SymbolicExpression::makeIntegerConstant(expr->nBits(), fpAnalyzer_->settings().nullDeref.minValid));
410  solver->insert(assertion);
411  retval = SmtSolver::SAT_NO == solver->check();
412  } else {
413  retval = SymbolicExpression::makeLt(expr, SymbolicExpression::makeIntegerConstant(expr->nBits(), fpAnalyzer_->settings().nullDeref.minValid))->mustEqual(SymbolicExpression::makeIntegerConstant(expr->nBits(), 1));
414  }
415  } else {
416  SAWYER_MESG(debug) <<" isNullDeref address is not constant as required by settings\n";
417  }
418  break;
419 
420  default:
421  ASSERT_not_reachable("invalid null-dereference mode");
422  }
423 
424  if (debug) {
425  debug <<" isNullDeref result = " <<(retval ? "is null" : "not null");
426  if (solver)
427  debug <<" using SMT solver \"" <<StringUtility::cEscape(solver->name()) <<"\"";
428  debug <<"\n";
429  }
430 
431  return retval;
432  }

486  if (segreg.isEmpty()) {

490  adjustedVa = add(addr_, signExtend(segregValue, addr_->nBits()));

584  if (fpAnalyzer_->settings().nullDeref.check && pathProcessor_ && !breakRequested()) {
585  SmtSolver::Transaction transaction(nullPtrSolver());
586  if (isNullDeref(adjustedVa)) {
587  ASSERT_not_null(fpAnalyzer_);
588  ASSERT_not_null(path_);
589  BaseSemantics::RiscOperators::Ptr cpu = shared_from_this();
590  ASSERT_not_null(cpu);
591  merge(pathProcessor_->nullDeref(*fpAnalyzer_, *path_, currentInstruction(), cpu, nullPtrSolver(),
592  FeasiblePath::WRITE, adjustedVa));
593  }
594  }
595 
596  // Save a description of the variable
597  SymbolicExpression::Ptr valExpr = SValue::promote(value)->get_expression();
598  if (valExpr->isLeafNode() && valExpr->isLeafNode()->isIntegerVariable())
599  State::promote(currentState())->varDetail(valExpr->isLeafNode()->toString(), detailForVariable(adjustedVa, "write"));
600 
601  // Save a description for its addresses
602  size_t nBytes = value->nBits() / 8;
603  for (size_t i=0; i<nBytes; ++i) {
604  SValue::Ptr va = SValue::promote(add(adjustedVa, number_(adjustedVa->nBits(), i)));
605  if (va->get_expression()->isLeafNode()) {
606  State::promote(currentState())->varDetail(va->get_expression()->isLeafNode()->toString(),
607  detailForVariable(adjustedVa, "read", i, nBytes));
608  }
609  }
610 
611  // Callback for the memory access
612  if (pathProcessor_ && !breakRequested()) {
613  ASSERT_not_null(fpAnalyzer_);
614  ASSERT_not_null(path_);
615  SmtSolver::Ptr s = nullPtrSolver();
616  SmtSolver::Transaction tx(s);
617  BaseSemantics::RiscOperators::Ptr cpu = shared_from_this();
618  ASSERT_not_null(cpu);
619  merge(pathProcessor_->memoryIo(*fpAnalyzer_, *path_, currentInstruction(), cpu, s, FeasiblePath::WRITE,
620  adjustedVa, value));
621  }
622  }
623 };

977 BaseSemantics::Dispatcher::Ptr
978 FeasiblePath::buildVirtualCpu(const P2::Partitioner::ConstPtr &partitioner, const P2::CfgPath *path, PathProcessor *pathProcessor,
979  const SmtSolver::Ptr &solver) {
980  ASSERT_not_null(partitioner);
981 
982  // Augment the register dictionary with a "path" register that holds the expression describing how the location is
983  // reachable along some path.
984  if (NULL == registers_) {
985  registers_ = RegisterDictionary::instance("Rose::BinaryAnalysis::FeasiblePath");
986  registers_->insert(partitioner->instructionProvider().registerDictionary());
987  ASSERT_require(REG_PATH.isEmpty());
988  REG_PATH = RegisterDescriptor(registers_->firstUnusedMajor(), 0, 0, 1);
989  registers_->insert("path", REG_PATH);
990 
991  // Where are return values stored? See also, end of this function. FIXME[Robb Matzke 2015-12-01]: We need to support
992  // returning multiple values. We should be using the new calling convention analysis to detect these.
993  ASSERT_require(REG_RETURN_.isEmpty());
994  Disassembler::Base::Ptr dis = partitioner->instructionProvider().disassembler();
995  ASSERT_not_null(dis);
996  RegisterDescriptor r;
997  if (as<Disassembler::X86>(dis)) {
998  if ((r = registers_->find("rax")) || (r = registers_->find("eax")) || (r = registers_->find("ax")))
999  REG_RETURN_ = r;
1000  } else if (as<Disassembler::M68k>(dis)) {
1001  if ((r = registers_->find("d0")))
1002  REG_RETURN_ = r; // m68k also typically has other return registers
1003  } else if (as<Disassembler::Powerpc>(dis)) {
1004  if ((r = registers_->find("r3")))
1005  REG_RETURN_ = r; // PowerPC also returns via r4
1006 #ifdef ROSE_ENABLE_ASM_AARCH32
1007  } else if (as<Disassembler::Aarch32>(dis)) {
1008  if ((r = registers_->find("r0")))
1009  REG_RETURN_ = r;
1010 #endif
1011 #ifdef ROSE_ENABLE_ASM_AARCH64
1012  } else if (as<Disassembler::Aarch64>(dis)) {
1013  if ((r = registers_->find("r0")))
1014  REG_RETURN_ = r;
1015 #endif
1016  } else {
1017  ASSERT_not_implemented("function return value register is not implemented for this ISA/ABI");
1018  }
1019  }
1020 
1021  // Create the RiscOperators and Dispatcher.
1022  RiscOperators::Ptr ops = RiscOperators::instance(partitioner, registers_, this, path, pathProcessor);
1023  ops->trimThreshold(settings_.maxExprSize);
1024  ops->initialState(ops->currentState()->clone());
1025  ops->nullPtrSolver(solver);
1026  for (size_t i = 0; i < settings().ipRewrite.size(); i += 2) {
1027  const RegisterDescriptor REG_IP = partitioner->instructionProvider().instructionPointerRegister();
1028  BaseSemantics::SValue::Ptr oldValue = ops->number_(REG_IP.nBits(), settings().ipRewrite[i+0]);
1029  BaseSemantics::SValue::Ptr newValue = ops->number_(REG_IP.nBits(), settings().ipRewrite[i+1]);
1030  SAWYER_MESG(mlog[DEBUG]) <<"ip-rewrite hot-patch from " <<*oldValue <<" to " <<*newValue <<"\n";
1031  ops->hotPatch().append(HotPatch::Record(REG_IP, oldValue, newValue, HotPatch::Record::MATCH_BREAK));
1032  }
1033 
1034  BaseSemantics::Dispatcher::Ptr cpu = partitioner->architecture()->newInstructionDispatcher(ops);
1035  ASSERT_not_null(cpu);
1036 
1037  // More return value stuff, continued from above
1038 #ifdef ROSE_ENABLE_ASM_AARCH64
1039  if (as<InstructionSemantics::DispatcherAarch64>(cpu)) {
1040  REG_RETURN_ = registers_->find("x0");
1041  }
1042 #endif
1043 
1044  return cpu;
1045 }

1047 void
1048 FeasiblePath::setInitialState(const BaseSemantics::Dispatcher::Ptr &cpu,
1049  const P2::ControlFlowGraph::ConstVertexIterator &pathsBeginVertex) {
1050  ASSERT_not_null(cpu);
1051  ASSERT_forbid(REG_PATH.isEmpty());
1052  checkSettings();
1053 
1054  // Create the new state from an existing state and make the new state current.
1055  BaseSemantics::State::Ptr state = cpu->currentState()->clone();
1056  state->clear();
1057  cpu->initializeState(state);
1058  BaseSemantics::RiscOperators::Ptr ops = cpu->operators();
1059  ops->currentState(state);
1060 
1061  // Start of path is always feasible.
1062  ops->writeRegister(REG_PATH, ops->boolean_(true));
1063 
1064  // Initialize instruction pointer register
1065  if (pathsBeginVertex->value().type() == P2::V_INDETERMINATE) {
1066  ops->writeRegister(cpu->instructionPointerRegister(),
1067  ops->undefined_(cpu->instructionPointerRegister().nBits()));
1068  } else {
1069  ops->writeRegister(cpu->instructionPointerRegister(),
1070  ops->number_(cpu->instructionPointerRegister().nBits(), pathsBeginVertex->value().address()));
1071  }
1072 
1073  // Initialize stack pointer register
1074  if (settings_.initialStackPtr) {
1075  const RegisterDescriptor REG_SP = cpu->stackPointerRegister();
1076  ops->writeRegister(REG_SP, ops->number_(REG_SP.nBits(), *settings_.initialStackPtr));
1077  }
1078 
1079  // Direction flag (DF) is always set
1080  if (const RegisterDescriptor REG_DF = cpu->registerDictionary()->find("df"))
1081  ops->writeRegister(REG_DF, ops->boolean_(true));
1082 
1083  initialState_ = ops->currentState()->clone();
1084 }

1139  }

1182  retval = functionSummarizer_->returnValue(*this, summary, fpOperators(ops));

1203  ops->boolean_(true));

1208  summary.stackDelta :
1209  returnTarget->nBits() / 8;
1210  stackPointer = ops->add(stackPointer, ops->number_(stackPointer->nBits(), sd));
1211  ops->writeRegister(cpu->stackPointerRegister(), stackPointer);

1214  // Return address is in the link register, lr
1215  const RegisterDescriptor LR = cpu->callReturnRegister();
1216  ASSERT_forbid(LR.isEmpty());

1218  ops->writeRegister(cpu->instructionPointerRegister(), returnTarget);
1219 #endif
1220 #ifdef ROSE_ENABLE_ASM_AARCH32
1221  } else if (as<InstructionSemantics::DispatcherAarch32>(cpu)) {
1222  // Return address is in the link register, lr
1223  const RegisterDescriptor LR = cpu->callReturnRegister();
1224  ASSERT_forbid(LR.isEmpty());
1225  BaseSemantics::SValue::Ptr returnTarget = ops->readRegister(LR, ops->undefined_(LR.nBits()));
1226  ops->writeRegister(cpu->instructionPointerRegister(), returnTarget);
1227 #endif
1228  }
1229  }
1230

1232  VarDetail varDetail;
1233  varDetail.returnFrom = summary.address;
1234  varDetail.firstAccessIdx = fpOperators(ops)->pathInsnIndex();

1236  }

1320 SymbolicExpression::Ptr
1321 FeasiblePath::pathEdgeConstraint(const P2::ControlFlowGraph::ConstEdgeIterator &pathEdge, const BaseSemantics::Dispatcher::Ptr &cpu) {
1322  ASSERT_not_null(cpu);
1323  BaseSemantics::RiscOperators::Ptr ops = cpu->operators();
1324  static const char *prefix = " ";
1325 
1326  const RegisterDescriptor IP = partitioner()->instructionProvider().instructionPointerRegister();
1327  BaseSemantics::SValue::Ptr ip = ops->peekRegister(IP, ops->undefined_(IP.nBits()));
1328  if (!settings_.nonAddressIsFeasible && !hasVirtualAddress(pathEdge->target())) {
1329  SAWYER_MESG(mlog[DEBUG]) <<prefix <<"unfeasible at edge " <<partitioner()->edgeName(pathEdge)
1330  <<" because settings().nonAddressIsFeasible is false\n";
1331  return SymbolicExpression::Ptr(); // trivially unfeasible
1332  } else if (auto ipval = ip->toUnsigned()) {
1333  if (!hasVirtualAddress(pathEdge->target())) {
1334  // If the IP register is pointing to an instruction but the path vertex is indeterminate (or undiscovered or
1335  // nonexisting) then consider this path to be not-feasible. If the CFG is accurate then there's probably
1336  // a sibling edge that points to the correct vertex.
1337  SAWYER_MESG(mlog[DEBUG]) <<prefix <<"unfeasible at edge " <<partitioner()->edgeName(pathEdge) <<" because IP = "
1338  <<StringUtility::addrToString(*ipval) <<" and edge target has no address\n";
1339  return SymbolicExpression::Ptr(); // trivially unfeasible
1340  } else if (*ipval != virtualAddress(pathEdge->target())) {
1341  // Executing the path forces us to go a different direction than where the path indicates we should go. We
1342  // don't need an SMT solver to tell us that when the values are just integers.
1343  SAWYER_MESG(mlog[DEBUG]) <<prefix <<"unfeasible at edge " <<partitioner()->edgeName(pathEdge) <<" because IP = "
1344  <<StringUtility::addrToString(*ipval) <<" and edge target is "
1345  <<StringUtility::addrToString(virtualAddress(pathEdge->target())) <<"\n";
1346  return SymbolicExpression::Ptr(); // trivially unfeasible
1347  }
1348  } else if (hasVirtualAddress(pathEdge->target())) {
1349  SymbolicExpression::Ptr targetVa = SymbolicExpression::makeIntegerConstant(ip->nBits(), virtualAddress(pathEdge->target()));
1350  SymbolicExpression::Ptr constraint = SymbolicExpression::makeEq(targetVa,
1351  SymbolicSemantics::SValue::promote(ip)->get_expression());
1352  constraint->comment("cfg edge " + partitioner()->edgeName(pathEdge));
1353  SAWYER_MESG(mlog[DEBUG]) <<prefix <<"constraint at edge " <<partitioner()->edgeName(pathEdge)
1354  <<": " <<*constraint <<"\n";
1355  return constraint;
1356  }
1357 
1358  SAWYER_MESG(mlog[DEBUG]) <<prefix <<"trivially feasible at edge " <<partitioner()->edgeName(pathEdge) <<"\n";
1359  return SymbolicExpression::makeBooleanConstant(true); // trivially feasible
1360 }

1509 bool
1510 FeasiblePath::shouldSummarizeCall(const P2::ControlFlowGraph::ConstVertexIterator &/*pathVertex*/,
1511  const P2::ControlFlowGraph &cfg,
1512  const P2::ControlFlowGraph::ConstVertexIterator &cfgCallTarget) {

1514  return false;

1516  if (!callee)
1517  return false;
1518  if (boost::ends_with(callee->name(), "@plt") || boost::ends_with(callee->name(), ".dll"))
1519  return true;

1523  settings_.summarizeFunctions.end())

1527  // the Partitioner2 not being able to find reasonable function boundaries, especially for GNU libc.
1528  static const size_t maxCallsAllowed = 10; // arbitrary
1529  size_t nCalls = 0;

1532  for (Traversal t(cfg, cfgCallTarget); t; ++t) {
1533  if (t.edge()->value().type() == P2::E_FUNCTION_CALL) {
1534  if (++nCalls > maxCallsAllowed)
1535  return true;
1536  t.skipChildren();
1537  } else if (t.edge()->value().type() == P2::E_FUNCTION_RETURN) {
1538  t.skipChildren();
1539  }
1540  }
1541

1543 }

1545 bool
1546 FeasiblePath::shouldInline(const P2::CfgPath &path, const P2::ControlFlowGraph::ConstVertexIterator &cfgCallTarget) {
1547  // We must inline indeterminte functions or else we'll end up removing the call and return edge, which is another way
1548  // of saying "we know there's no path through here" when in fact there could be.
1549  if (cfgCallTarget->value().type() == P2::V_INDETERMINATE)
1550  return true;
1551 
1552  // Don't let the call depth get too deep
1553  ssize_t callDepth = path.callDepth();
1554  ASSERT_require(callDepth >= 0);
1555  if ((size_t)callDepth >= settings_.maxCallDepth) {
1556  SAWYER_THREAD_TRAITS::LockGuard lock(statsMutex_);
1557  ++stats_.maxCallDepthHits;
1558  return false;
1559  }
1560 
1561  // Don't inline if there's no function
1562  if (cfgCallTarget->value().type() != P2::V_BASIC_BLOCK)
1563  return false;
1564  P2::Function::Ptr callee = cfgCallTarget->value().isEntryBlock();
1565  if (!callee)
1566  return false;
1567 
1568  // Don't let recursion get too deep
1569  if (settings_.maxRecursionDepth < UNLIMITED) {
1570  ssize_t callDepth = path.callDepth(callee);
1571  ASSERT_require(callDepth >= 0);
1572  if ((size_t)callDepth >= settings_.maxRecursionDepth) {
1573  SAWYER_THREAD_TRAITS::LockGuard lock(statsMutex_);
1574  ++stats_.maxRecursionDepthHits;
1575  return false;
1576  }
1577  }
1578 
1579  // Don't inline imported functions that aren't linked -- we'd just get bogged down deep inside the
1580  // dynamic linker without ever being able to resolve the actual function's instructions.
1581  if (P2::ModulesElf::isUnlinkedImport(partitioner_, callee))
1582  return false;
1583 
1584  return true;
1585 }

1940 BaseSemantics::State::Ptr
1941 FeasiblePath::evaluate(P2::CfgPath &path, int position, const Semantics &sem) {
1942  Stream debug(mlog[DEBUG]);
1943  ASSERT_require2(positionToIndex(position, path.nVertices()), "position is out of range");
1944  size_t goalIdx = positionToIndex(position, path.nVertices()).orElse(0);
1945 
1946  // Find the most recent path vertex that has been evaluated, and get a copy of its outgoing state.
1947  size_t idx = goalIdx;
1948  BaseSemantics::State::Ptr state = incomingState(path, idx, sem.originalState);
1949  while (!state && idx > 0)
1950  state = incomingState(path, --idx, sem.originalState);
1951  ASSERT_not_null(state);
1952 
1953  // Evaluate each vertex up to and including the goal vertex, saving the results per vertex.
1954  for (/*void*/; idx <= goalIdx; ++idx) {
1955  state = state->clone(); // need to make a copy before we modify it
1956  sem.ops->currentState(state);
1957  try {
1958  processVertex(sem.cpu, path.vertices()[idx], incomingStepCount(path, idx));
1959  } catch (...) {
1960  SAWYER_MESG(debug) <<" path semantics failed, path idx = " <<idx <<"\n";
1961  state = BaseSemantics::State::Ptr();
1962  break;
1963  }
1964  path.vertexAttributes(idx).setAttribute(POST_STATE, state);
1965  }
1966 
1967  sem.ops->currentState(BaseSemantics::State::Ptr());
1968  return state;
1969 }

1971 boost::logic::tribool
1972 FeasiblePath::isFeasible(P2::CfgPath &path, const Substitutions &subst, const Semantics &sem,
1973  const SmtSolver::Ptr &solver) {
1974  Stream debug(mlog[DEBUG]);
1975  ASSERT_forbid(path.isEmpty());
1976  ASSERT_require(solver->nLevels() == path.nVertices());
1977  boost::logic::tribool retval = false;
1978 
1979  bool atEndOfPath = pathsEndVertices_.exists(path.backVertex()); // true if path has reached one of the end goals
1980  BaseSemantics::State::Ptr oldState = sem.ops->currentState(); // probably null
1981 
1982  if (path.nVertices() == 1) {
1983  switch (solvePathConstraints(solver, path, SymbolicExpression::makeBooleanConstant(true), subst, atEndOfPath)) {
1984  case SmtSolver::SAT_YES:
1985  SAWYER_MESG(debug) <<" = is feasible\n";
1986  retval = true;
1987  break;
1988  case SmtSolver::SAT_NO:
1989  SAWYER_MESG(debug) <<" = not feasible\n";
1990  retval = false;
1991  break;
1992  case SmtSolver::SAT_UNKNOWN:
1993  SAWYER_MESG(debug) <<" = unknown\n";
1994  retval = boost::logic::indeterminate;
1995  break;
1996  }
1997  } else {
1998  // pathEdgeConstraint needs the incoming state of the last vertex, which is the outgoing state of the penultimate
1999  // vertex. It doesn't modify the state.
2000  BaseSemantics::State::Ptr curState = evaluate(path, -2, sem);
2001  sem.ops->currentState(curState);
2002 
2003  if (solver->nAssertions(solver->nLevels()-1) > 0) {
2004  SAWYER_MESG(debug) <<" = is feasible (previously computed)\n";
2005  retval = true;
2006  } else if (SymbolicExpression::Ptr edgeConstraint = pathEdgeConstraint(path.edges().back(), sem.cpu)) {
2007  switch (solvePathConstraints(solver, path, edgeConstraint, subst, atEndOfPath)) {
2008  case SmtSolver::SAT_YES:
2009  SAWYER_MESG(debug) <<" = is feasible\n";
2010  retval = true;
2011  break;
2012  case SmtSolver::SAT_NO:
2013  SAWYER_MESG(debug) <<" = not feasible\n";
2014  retval = false;
2015  break;
2016  case SmtSolver::SAT_UNKNOWN:
2017  SAWYER_MESG(debug) <<" = unknown\n";
2018  retval = boost::logic::indeterminate;
2019  break;
2020  }
2021  } else {
2022  SAWYER_MESG(debug) <<" = not feasible (trivial)\n";
2023  retval = false;
2024  }
2025  }
2026 
2027  // Restore the previous state
2028  sem.ops->currentState(oldState);
2029  return retval;
2030 }

2094 void
2095 FeasiblePath::summarizeOrInline(P2::CfgPath &path, const Semantics &sem) {
2096  Stream debug(mlog[DEBUG]);
2097  std::string indent = debug ? " " : "";
2098 
2099  P2::ControlFlowGraph::ConstVertexIterator backVertex = path.backVertex();
2100  P2::ControlFlowGraph::ConstVertexIterator cfgBackVertex = pathToCfg(backVertex);
2101  ASSERT_require(partitioner()->cfg().isValidVertex(cfgBackVertex));
2102  P2::CfgConstEdgeSet callEdges = P2::findCallEdges(cfgBackVertex);
2103  P2::CfgConstEdgeSet erasableEdges;
2104 
2105  for (const P2::ControlFlowGraph::ConstEdgeIterator &cfgCallEdge: callEdges.values()) {
2106  if (shouldSummarizeCall(path.backVertex(), partitioner()->cfg(), cfgCallEdge->target())) {
2107  SAWYER_MESG(debug) <<indent <<"summarizing function for edge " <<partitioner()->edgeName(cfgCallEdge) <<"\n";
2108  insertCallSummary(backVertex, partitioner()->cfg(), cfgCallEdge);
2109  erasableEdges.insert(cfgCallEdge);
2110  } else if (shouldInline(path, cfgCallEdge->target())) {
2111  if (cfgCallEdge->target()->value().type() == P2::V_INDETERMINATE &&
2112  cfgBackVertex->value().type() == P2::V_BASIC_BLOCK) {
2113  // If the CFG has a vertex to an indeterminate function (e.g., from "call eax"), then instead of inlining the
2114  // indeterminate vertex, see if we can inline an actual function by using the instruction pointer register. The
