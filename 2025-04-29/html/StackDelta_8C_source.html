107 // Augment the base data-flow transfer function because we need to keep track of the stack for every instruction and basic
108 // block.
109 class TransferFunction: public P2::DataFlow::TransferFunction {
110  Analysis *analysis_;
111 public:
112  explicit TransferFunction(Analysis *analysis)
113  : P2::DataFlow::TransferFunction(analysis->cpu()), analysis_(analysis) {}
114 
115  // Override the base class by initializing only the stack pointer register.
116  BaseSemantics::State::Ptr initialState() const {
117  BaseSemantics::RiscOperators::Ptr ops = cpu()->operators();
118  BaseSemantics::State::Ptr newState = ops->currentState()->clone();
119  newState->clear();
120  cpu()->initializeState(newState);
121 
122  BaseSemantics::RegisterStateGeneric::Ptr regState =
123  BaseSemantics::RegisterStateGeneric::promote(newState->registerState());
124 
125  const RegisterDescriptor SP = cpu()->stackPointerRegister();
126  Address initialSp = 0;
127  if (analysis_->initialConcreteStackPointer().assignTo(initialSp)) {
128  newState->writeRegister(SP, ops->number_(SP.nBits(), initialSp), ops.get());
129  } else {
130  newState->writeRegister(SP, ops->undefined_(SP.nBits()), ops.get());
131  }
132  return newState;
133  }
134 
135  // Required by data-flow engine: compute next state from current state and dfCfg vertex
136  BaseSemantics::State::Ptr
137  operator()(const P2::DataFlow::DfCfg &dfCfg, size_t vertexId, const BaseSemantics::State::Ptr &incomingState) const {
138  P2::DataFlow::DfCfg::ConstVertexIterator vertex = dfCfg.findVertex(vertexId);
139  ASSERT_require(dfCfg.isValidVertex(vertex));
140  if (P2::DataFlow::DfCfgVertex::BBLOCK == vertex->value().type()) {
141  BaseSemantics::State::Ptr retval = incomingState->clone();
142  BaseSemantics::RiscOperators::Ptr ops = analysis_->cpu()->operators();
143  ops->currentState(retval);
144  ASSERT_not_null(vertex->value().bblock());
145  const RegisterDescriptor SP = cpu()->stackPointerRegister();
146  const RegisterDescriptor FP = cpu()->stackFrameRegister(); // not all architectures have this
147 #if 1 // DEBUGGING [Robb Matzke 2024-10-09]
148  ASSERT_always_require2(FP, "no frame pointer register");
149 #endif
150  BaseSemantics::SValue::Ptr oldSp = retval->peekRegister(SP, ops->undefined_(SP.nBits()), ops.get());
151  BaseSemantics::SValue::Ptr oldFp = FP ?
152  retval->peekRegister(FP, ops->undefined_(FP.nBits()), ops.get()) :
153  BaseSemantics::SValue::Ptr();
154  SAWYER_MESG(mlog[DEBUG]) <<"transfer function for " <<vertex->value().bblock()->printableName() <<"\n";
155  for (SgAsmInstruction *insn: vertex->value().bblock()->instructions()) {
156  SAWYER_MESG(mlog[DEBUG]) <<" executing " <<insn->toString() <<"\n";
157  cpu()->processInstruction(insn);
158 
159  // Stack and frame pointers after the instruction executes
160  BaseSemantics::SValue::Ptr newSp = retval->peekRegister(SP, ops->undefined_(SP.nBits()), ops.get());
161  BaseSemantics::SValue::Ptr newFp = FP ?
162  retval->peekRegister(FP, ops->undefined_(FP.nBits()), ops.get()) :
163  BaseSemantics::SValue::Ptr();
164 
165  // Stack delta is the net change to the stack pointer by this instruction.
166  BaseSemantics::SValue::Ptr spDelta = ops->subtract(newSp, oldSp);
167 
168  // Save and update loop variables
169  analysis_->adjustInstruction(insn, oldSp, newSp, spDelta, oldFp, newFp);
170  oldSp = newSp;
171  oldFp = newFp;
172  }
173  return retval;
174  }
175 
176  return P2::DataFlow::TransferFunction::operator()(dfCfg, vertexId, incomingState);
177  }

180 void
181 Analysis::analyzeFunction(const P2::Partitioner::ConstPtr &partitioner, const P2::FunctionPtr &function,
182  Partitioner2::DataFlow::InterproceduralPredicate &ipPredicate) {
183  ASSERT_not_null(partitioner);
184  Sawyer::Message::Stream debug(mlog[DEBUG]);
185  SAWYER_MESG(debug) <<"analyzeFunction(" <<function->printableName() <<")\n";
186  clearResults();
187  const RegisterDescriptor SP = cpu_->stackPointerRegister();
188 
189  // Build the CFG used by the data-flow: dfCfg. The dfCfg includes only those vertices that are reachable from the entry
190  // point for the function we're analyzing and which belong to that function. All return points in the function will flow
191  // into a special CALLRET vertex (which is absent if there are no returns).
192  typedef P2::DataFlow::DfCfg DfCfg;
193  DfCfg dfCfg = P2::DataFlow::buildDfCfg(partitioner, partitioner->cfg(), partitioner->findPlaceholder(function->address()),
194  ipPredicate);
195  size_t startVertexId = 0;
196  DfCfg::ConstVertexIterator returnVertex = dfCfg.vertices().end();
197  for (const DfCfg::Vertex &vertex: dfCfg.vertices()) {
198  if (vertex.value().type() == P2::DataFlow::DfCfgVertex::FUNCRET) {
199  returnVertex = dfCfg.findVertex(vertex.id());
200  break;
201  }
202  }
203  if (returnVertex == dfCfg.vertices().end()) {
204  SAWYER_MESG(debug) <<" function CFG has no return vertex\n";
205  // continue anyway, to get stack delta info for blocks and instructions...
206  }
207 
208  // Build the dataflow engine. If an instruction dispatcher is already provided then use it, otherwise create one and store
209  // it in this analysis object.
210  typedef DataFlow::Engine<DfCfg, BaseSemantics::State::Ptr, TransferFunction, DataFlow::SemanticsMerge> DfEngine;
211  if (!cpu_ && NULL==(cpu_ = partitioner->newDispatcher(partitioner->newOperators()))) {
212  SAWYER_MESG(debug) <<" no instruction semantics\n";
213  return;
214  }
215  const CallingConvention::Dictionary &ccDefs = partitioner->instructionProvider().callingConventions();
216  P2::DataFlow::MergeFunction merge(cpu_);
217  TransferFunction xfer(this);
218  xfer.defaultCallingConvention(ccDefs.empty() ? CallingConvention::Definition::Ptr() : ccDefs.front());
219  DfEngine dfEngine(dfCfg, xfer, merge);
220  dfEngine.name("stack-delta");
221  size_t maxIterations = dfCfg.nVertices() * 5; // arbitrary
222  dfEngine.maxIterations(maxIterations);
223  BaseSemantics::RiscOperators::Ptr ops = cpu_->operators();
224 
225  // Build the initial state
226  BaseSemantics::State::Ptr initialState = xfer.initialState();
227  BaseSemantics::RegisterStateGeneric::Ptr initialRegState =
228  BaseSemantics::RegisterStateGeneric::promote(initialState->registerState());
229 
230  // Run data flow analysis
231  bool converged = true;
232  try {
233  // Use this rather than runToFixedPoint because it lets us show a progress report
234  Sawyer::ProgressBar<size_t> progress(maxIterations, mlog[MARCH], function->printableName());
235  progress.suffix(" iterations");
236  dfEngine.reset(BaseSemantics::State::Ptr());
237  dfEngine.insertStartingVertex(startVertexId, initialState);
238  while (dfEngine.runOneIteration())
239  ++progress;
240  } catch (const DataFlow::NotConverging &e) {
241  mlog[WARN] <<e.what() <<" for " <<function->printableName() <<"\n";
242  converged = false; // didn't converge, so just use what we have
243  } catch (const BaseSemantics::NotImplemented &e) {
244  mlog[WHERE] <<e.what() <<" for " <<function->printableName() <<"\n";
245  } catch (const BaseSemantics::Exception &e) {
246  mlog[WARN] <<e.what() <<" for " <<function->printableName() <<"\n";
247  converged = false;
248  } catch (const SmtSolver::Exception &e) {
249  mlog[WARN] <<e.what() <<" for " <<function->printableName() <<"\n";
250  converged = false;
251  }
252 
253  // Get the final dataflow state
254  BaseSemantics::State::Ptr finalState;
255  BaseSemantics::RegisterStateGeneric::Ptr finalRegState;
256  if (dfCfg.isValidVertex(returnVertex)) {
257  finalState = dfEngine.getInitialState(returnVertex->id());
258  if (finalState == NULL) {
259  SAWYER_MESG(debug) <<" data flow analysis did not reach final state\n";
260  // continue anyway for stack delta info for blocks and instructions
261  }
262  if (debug) {
263  if (!converged) {
264  debug <<" data flow analysis did not converge to a solution (using partial solution)\n";
265  } else if (finalState) {
266  SymbolicSemantics::Formatter fmt;
267  fmt.set_line_prefix(" ");
268  fmt.expr_formatter.max_depth = 10; // prevent really long output
269  debug <<" final state:\n" <<(*finalState+fmt);
270  }
271  }
272  if (finalState)
273  finalRegState = BaseSemantics::RegisterStateGeneric::promote(finalState->registerState());
274  }
275 
276  // Get stack pointers for each basic block
277  for (const DfCfg::Vertex &vertex: dfCfg.vertices()) {
278  if (vertex.value().type() == P2::DataFlow::DfCfgVertex::BBLOCK) {
279  P2::BasicBlock::Ptr bblock = vertex.value().bblock();
280  ASSERT_not_null(bblock);
281  BaseSemantics::SValue::Ptr sp0, sp1;
282  if (BaseSemantics::State::Ptr state = dfEngine.getInitialState(vertex.id()))
283  sp0 = state->peekRegister(SP, ops->undefined_(SP.nBits()), ops.get());
284  if (BaseSemantics::State::Ptr state = dfEngine.getFinalState(vertex.id()))
285  sp1 = state->peekRegister(SP, ops->undefined_(SP.nBits()), ops.get());
286  bblockStackPtrs_.insert(bblock->address(), SValuePair(sp0, sp1));
287 
288  if (sp0 && sp1) {
289  BaseSemantics::SValue::Ptr delta = ops->subtract(sp1, sp0);
290  bblockDeltas_.insert(bblock->address(), delta);
291  }
292  }
293  }
294 
295  // Functon stack delta is final stack pointer minus initial stack pointer. This includes popping the return address from
296  // the stack (if the function did that) and popping arguments (if the function did that).
297  functionStackPtrs_.first = initialRegState->peekRegister(SP, ops->undefined_(SP.nBits()), ops.get());
298  SAWYER_MESG(debug) <<" function initial stack pointer is " <<*functionStackPtrs_.first <<"\n";
299  if (finalRegState) {
300  functionStackPtrs_.second = finalRegState->peekRegister(SP, ops->undefined_(SP.nBits()), ops.get());
301  functionDelta_ = ops->subtract(functionStackPtrs_.second, functionStackPtrs_.first);
302  SAWYER_MESG(debug) <<" function final stack pointer is " <<*functionStackPtrs_.second <<"\n";
303  SAWYER_MESG(debug) <<" function stack delta is " <<*functionDelta_ <<"\n";
304  } else {
305  SAWYER_MESG(debug) <<" no final state, thus no stack delta\n";
306  }
307 
308  // Does this function use a frame pointer register consistently as if it were a frame pointer?
309  hasConsistentFramePointer_ = [this, &dfCfg]() {
310  Sawyer::Optional<int64_t> fpWrtFunctionSp;
311  for (const DfCfg::Vertex &vertex: dfCfg.vertices()) {
312  if (vertex.value().type() == P2::DataFlow::DfCfgVertex::BBLOCK) {
313  const P2::BasicBlock::Ptr bblock = vertex.value().bblock();
314  ASSERT_not_null(bblock);
315  for (SgAsmInstruction *insn: bblock->instructions()) {
316  const Sawyer::Optional<int64_t> spDelta = toInt(instructionInputStackDeltaWrtFunction(insn));
317  const Sawyer::Optional<int64_t> fpDelta = toInt(instructionInputFrameDelta(insn));
318  if (spDelta && fpDelta) {
319  if (!fpWrtFunctionSp) {
320  fpWrtFunctionSp = *spDelta + *fpDelta;
321  } else if (*fpWrtFunctionSp != *spDelta + *fpDelta) {
322  return false;
323  }
324  }
325  }
326  }
327  }
328  return true;
329  }();
330 
331  hasResults_ = true;
332  didConverge_ = converged;
333 }
