192 | This function read font file and store information in memory
193 | Return: GLF_OK - if all OK
194 | Return: GLF_ERROR - if any error
195 */
196 static int ReadFont(char *font_name, struct glf_font *glff)
197 {
198 
199  FILE *fontf;
200  char buffer[64];
201  int i, j;
202  unsigned char temp, code, verts, fcets, lns;
203  float tempfx, tempfy;
204  unsigned char *tp;
205  int LEndian; /* True if little endian machine */
206 
207  fontf = fopen(font_name, "rb");
208  if (fontf == NULL) return GLF_ERROR;
209 
210  fread(buffer, 3, 1, fontf);
211  buffer[3] = 0;
212  if (strcmp(buffer, "GLF"))
213  {
214  /* If header is not "GLF" */
215  if (console_msg) printf("Error reading font file: incorrect file format\n");
216  return GLF_ERROR;
217  }
218 
219  /* Check for machine type */
220  LEndian = LittleEndian();
221 
222  fread(glff->font_name, 96, 1, fontf);
223  glff->font_name[96] = 0;
224 
225  fread(&glff->sym_total, 1, 1, fontf); /* Read total symbols in font */
226 
227  for (i=0; i<MAX_FONTS; i++) glff->symbols[i] = NULL;
228 
229  for (i=0; i<28; i++) fread(&temp, 1, 1, fontf); /* Read unused data */
230 
231  /* Now start to read font data */
232 
233  for (i=0; i<glff->sym_total; i++)
234  {
235  fread(&code, 1, 1, fontf); /* Read symbol code */
236  fread(&verts, 1, 1, fontf); /* Read vertexs count */
237  fread(&fcets, 1, 1, fontf); /* Read facets count */
238  fread(&lns, 1, 1, fontf); /* Read lines count */
239 
240  if (glff->symbols[code] != NULL)
241  {
242  if (console_msg) printf("Error reading font file: encountered symbols in font\n");
243  return GLF_ERROR;
244  }
245 
246  glff->symbols[code] = (struct one_symbol *)malloc(sizeof(struct one_symbol));
247  glff->symbols[code]->vdata = (float *)malloc(8*verts);
248  glff->symbols[code]->fdata = (unsigned char *)malloc(3*fcets);
249  glff->symbols[code]->ldata = (unsigned char *)malloc(lns);
250 
251  glff->symbols[code]->vertexs = verts;
252  glff->symbols[code]->facets = fcets;
253  glff->symbols[code]->lines = lns;
254 
255  /* Read vertexs data */
256  glff->symbols[code]->leftx = 10;
257  glff->symbols[code]->rightx = -10;
258  glff->symbols[code]->topy = 10;
259  glff->symbols[code]->bottomy = -10;
260 
261  for (j=0; j<verts; j++)
262  {
263  fread(&tempfx, 4, 1, fontf);
264  fread(&tempfy, 4, 1, fontf);
265 
266  /* If machine is bigendian -> swap low and high words in
267  tempfx and tempfy */
268  if (!LEndian)
269  {
270  tp = (unsigned char *)&tempfx;
271  temp = tp[0]; tp[0] = tp[3]; tp[3] = temp;
272  temp = tp[1]; tp[1] = tp[2]; tp[2] = temp;
273  tp = (unsigned char *)&tempfy;
274  temp = tp[0]; tp[0] = tp[3]; tp[3] = temp;
275  temp = tp[1]; tp[1] = tp[2]; tp[2] = temp;
276  }
277  glff->symbols[code]->vdata[j*2] = tempfx;
278  glff->symbols[code]->vdata[j*2+1] = tempfy;
279 
280  if (tempfx < glff->symbols[code]->leftx) glff->symbols[code]->leftx = tempfx;
281  if (tempfx > glff->symbols[code]->rightx) glff->symbols[code]->rightx = tempfx;
282  if (tempfy < glff->symbols[code]->topy) glff->symbols[code]->topy = tempfy;
283  if (tempfy > glff->symbols[code]->bottomy) glff->symbols[code]->bottomy = tempfy;
284  }
285  for (j=0; j<fcets; j++)
286  fread(&glff->symbols[code]->fdata[j*3], 3, 1, fontf);
287  for (j=0; j<lns; j++)
288  fread(&glff->symbols[code]->ldata[j], 1, 1, fontf);
289  }
290  fclose(fontf);
291  return GLF_OK;
292 }

572 static void DrawString(char *s, void (*funct) (char s))
573 {
574  int i;
575  float sda, sdb;
576  float distance = 0.0;
577 
578  if (!s) return;
579  if (!*s) return;
580  if (curfont == -1) return;
581 
582  /* Calculate correction (if string centering enabled) */
583  if (m_string_center)
584  {
585  distance = 0;
586  for (i=0; i<(int)strlen(s); i++)
587  {
588  if ((fonts[curfont]->symbols[s[i]] == NULL) || (s[i] == ' '))
589  {
590  if (m_direction == GLF_LEFT || m_direction == GLF_UP) distance += SpaceSize;
591  else distance -= SpaceSize;
592  }
593  else
594  if (i < ((int)strlen(s)-1))
595  if (s[i+1] == ' ')
596  {
597  if (m_direction == GLF_LEFT || m_direction == GLF_UP) distance += SymbolDist;
598  else distance -= SymbolDist;
599  }
600  else
601  {
602  if (fonts[curfont]->symbols[s[i+1]] == NULL) continue;
603 
604  if (m_direction == GLF_LEFT || m_direction == GLF_RIGHT)
605  {
606  sda = (float)fabs(fonts[curfont]->symbols[s[i]]->rightx);
607  sdb = (float)fabs(fonts[curfont]->symbols[s[i+1]]->leftx);
608  if (m_direction == GLF_LEFT) distance += sda+sdb+SymbolDist;
609  else distance -= sda+sdb+SymbolDist;
610  }
611  else
612  {
613  sda = (float)fabs(fonts[curfont]->symbols[s[i]]->topy);
614  sdb = (float)fabs(fonts[curfont]->symbols[s[i]]->bottomy);
615  if (m_direction == GLF_DOWN) distance -= sda+sdb+SymbolDist;
616  else distance += sda+sdb+SymbolDist;
617  }
618  }
619  }
620  }
621 
622  glPushMatrix();
623 
624  /* Rotate if needed */
625  if (RotateAngle != 0.0f) glRotatef(RotateAngle, 0, 0, 1);
626 
627  /* Correct string position */
628  if (m_string_center)
629  {
630  switch (m_direction)
631  {
632  case GLF_LEFT : glTranslatef(-distance/2, 0, 0); break;
633  case GLF_RIGHT : glTranslatef(distance/2, 0, 0); break;
634  case GLF_UP : glTranslatef(0, distance/2, 0); break;
635  case GLF_DOWN : glTranslatef(0, -distance/2, 0); break;
636  }
637  }
638  else if (s[0] != ' ')
639  {
640  switch (m_direction)
641  {
642  case GLF_LEFT : glTranslatef(-(1-(float)fabs(fonts[curfont]->symbols[s[0]]->leftx)), 0, 0); break;
643  case GLF_RIGHT : glTranslatef((1-(float)fabs(fonts[curfont]->symbols[s[0]]->rightx)), 0, 0); break;
644  case GLF_UP : glTranslatef(0, (1-(float)fabs(fonts[curfont]->symbols[s[0]]->topy)), 0); break;
