262 {
263  panel->attach(m_editor);

265  QAction *a = new QAction(panel->type(), m_editor);
266  a->setCheckable(true);

268 
269  QObject::connect(a , SIGNAL( toggled(bool) ),
270  panel , SLOT ( setVisible(bool) ) );
271 
272  m_layout->addWidget(panel, QPanelLayout::Position(pos));
273  m_layout->update();
274 
275  m_actions << a;

284  m_panelsMenu->menuAction()->setObjectName("panels");
285  m_editor->addAction(m_panelsMenu->menuAction(), QEditor::tr("&View"), QString());
286  }
287 
288  m_panelsMenu->addAction(a);
289  }
290
