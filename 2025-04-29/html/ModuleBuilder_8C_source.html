107  }
108

111  if (namespace_defn) namespace_symbols = namespace_defn->get_symbol_table();
112

114  if (namespace_symbols) {

128  // Import all namespace symbols

130  }

141  loadSymbol(symbol, namespace_symbols, file_scope);
142  }
143  }
144  }

246 void ModuleBuilder::loadTypeSymbol(SgType* type, SgSymbolTable* symbol_table, SgGlobal* file_scope)
247 {
248  if (SgNamedType* named_type = isSgNamedType(type)) {
249  SgName type_name = named_type->get_name();
250 
251  // Ensure the symbol does not exist, otherwise could lead to an infinite recursion
252  if (!file_scope->symbol_exists(type_name)) {

281
