80 processFortranComment(SgLocatedNode* node)
81 {
82  SgStatement* target = NULL;
83  ROSE_ASSERT(node);
84  AttachedPreprocessingInfoType *comments =
85  node->getAttachedPreprocessingInfo ();
86  if (comments==NULL)
87  return 0;
88  AttachedPreprocessingInfoType::iterator i;
89  std::vector< PreprocessingInfo* > removeList;
90  for (i = comments->begin (); i != comments->end (); i++)
91  {
92  if ((*i)->getTypeOfDirective() == PreprocessingInfo::FortranStyleComment)
93  {
94  string commentString = (*i)->getString();
95  boost::algorithm::trim(commentString);
96  if ( (commentString == "!$"+PRAGMA_OUTLINE)
97  || (commentString == "c$"+PRAGMA_OUTLINE)
98  || (commentString == "*$"+PRAGMA_OUTLINE))
99  {
100  target = isSgStatement(node);
101  if (target==NULL)
102  {
103  cerr<<"Unhandled case when a Fortran statement is attached to a non-statement node!!"<<endl;
104  ROSE_ASSERT(false);
105  }
106  removeList.push_back(*i);
107  }
108  } // end if Fortran comment
109  } // end for-loop
110 
111  // remove those special comments
112  for (std::vector<PreprocessingInfo* >::iterator j = removeList.begin();
113  j!=removeList.end();j++)
114  {
115  comments->erase(find(comments->begin(), comments->end(),*j));
116 
117  // DQ (2/27/2009): Don't mix free with C++ code (malloc/free for C and new/delete for C++)
118  // free memory also
119  // free(*j);
120  delete(*j);
121  }
122 
123  return target; // const_cast<SgStatement *> (target);
124 }
