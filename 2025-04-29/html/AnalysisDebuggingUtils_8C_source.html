171 streamsize dbgBuf::xsputn(const char * s, streamsize n)
172 {
173  // If the owner is printing, output their text exactly
174  if(ownerAccess) {
175  return baseBuf->sputn(s, n);
176  // Otherwise, replace all line-breaks with <BR>'s
177  } else {
178  int ret;
179  int i=0;
180  char br[]="<BR>\n";
181  char space[]=" ";
182  char spaceHTML[]="&nbsp;";
183  char tab[]="\t";
184  char tabHTML[]="&#09;";
185  while(i<n) {
186  int j;
187  for(j=i; j<n && s[j]!='\n' && s[j]!=' ' && s[j]!='\t'/* && s[j]!='<' && s[j]!='>'*/; j++) {
188  if(s[j]=='<') numOpenAngles++;
189  else if(s[j]=='>') numOpenAngles--;
190  }
191  // Send out all the bytes from the start of the string or the
192  // last line-break until this line-break
193  if(j-i>0) {
194  ret = baseBuf->sputn(&(s[i]), j-i);
195  if(ret != (j-i)) return 0;
196  }
197 
198  if(j<n) {
199  // If we're at at line-break Send out the line-break
200  if(s[j]=='\n') {
201  ret = baseBuf->sputn(br, sizeof(br)-1);
202  if(ret != sizeof(br)-1) return 0;
203  } else if(s[j]==' ') {
204  // If we're at a space and not inside an HTML tag, replace it with an HTML space escape code
205  if(numOpenAngles==0) {
206  ret = baseBuf->sputn(spaceHTML, sizeof(spaceHTML)-1);
207  if(ret != sizeof(spaceHTML)-1) return 0;
208  // If we're inside an HTML tag, emit a regular space character
209  } else {
210  ret = baseBuf->sputn(space, sizeof(space)-1);
211  if(ret != sizeof(space)-1) return 0;
212  }
213  } else if(s[j]=='\t') {
214  // If we're at a tab and not inside an HTML tag, replace it with an HTML tab escape code
215  if(numOpenAngles==0) {
216  ret = baseBuf->sputn(tabHTML, sizeof(tabHTML)-1);
217  if(ret != sizeof(tabHTML)-1) return 0;
218  // If we're inside an HTML tag, emit a regular tab character
219  } else {
220  ret = baseBuf->sputn(tab, sizeof(tab)-1);
221  if(ret != sizeof(tab)-1) return 0;
222  }
223  }/* else if(s[j]=='<') {
224  ret = baseBuf->sputn(lt, sizeof(lt)-1);
225  if(ret != sizeof(lt)-1) return 0;
226  } else if(s[j]=='>') {
227  ret = baseBuf->sputn(gt, sizeof(gt)-1);

229  }*/
230  //cout << " printing <BR>\n";

711  // Given a string, returns a version of the string with all the control characters that may appear in the
712  // string escaped to that the string can be written out to Dbg::dbg with no formatting issues.
713  // This function can be called on text that has already been escaped with no harm.
714  std::string escape(std::string s)
715  {
716  string out;
717  for(size_t i=0; i<s.length(); i++) {
718  // Manage HTML tags
719  if(s[i] == '<') out += "&lt;";
720  else if(s[i] == '>') out += "&gt;";
721  // Manage hashes, since they confuse the C PreProcessor CPP
722  else if(s[i] == '#') out += "&#35;";
723  else out += s[i];
724  }
725  return out;
726  }
