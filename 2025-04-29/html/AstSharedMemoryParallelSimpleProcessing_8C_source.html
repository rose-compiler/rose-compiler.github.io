87 void AstSharedMemoryParallelProcessingSynchronizationBase::synchronize()
88 {
89  // (Safely) decrement the shared variable counting the threads that are
90  // still working, then wait until that value is 0 (i.e. all threads are
91  // done) and we are signalled to go ahead.
92  pthread_mutex_lock(syncInfo.mutex);
93  *syncInfo.workingThreads -= 1;
94  if (*syncInfo.workingThreads == 0)
95  {
96  // If the counter reached 0, this is the last thread to get here; tell
97  // the others to wake up and then just go ahead.
98  *syncInfo.workingThreads = numberOfThreads;
99  pthread_cond_broadcast(syncInfo.synchronizationEvent);
100  }
101  else
102  {
103  // Not all threads are done yet, wait for them to get to this
104  // synchronization point.
105  pthread_cond_wait(syncInfo.synchronizationEvent, syncInfo.mutex);
106  // This loop guards against spurious wake ups which are apparently
107  // allowed by the pthreads standard under some circumstances. I have
108  // never observed them, but this doesn't cost anything, so why not
109  // have it here.
110  while (*syncInfo.workingThreads == 0)
111  {
112  std::cout << this << ": spurious wake up" << std::endl;
113  pthread_cond_wait(syncInfo.synchronizationEvent, syncInfo.mutex);
114  }
115  }
116  pthread_mutex_unlock(syncInfo.mutex);
117 }

214 void AstSharedMemoryParallelSimpleProcessing::traverseInParallel(SgNode *basenode,
215  t_traverseOrder treeTraverseOrder)
216 {
217  size_t numberOfTraversals = traversals.size();
218  size_t i;
219 
220  AstSharedMemoryParallelProcessingSynchronizationInfo syncInfo(numberOfThreads, synchronizationWindowSize);
221 
222  // Chop the flat list of traversals apart and distribute them into a few
223  // parallelizable traversals.
224  // std::cerr << "AstSharedMemoryParallelSimpleProcessing : starting " << numberOfThreads << " threads. " << std::endl;
225  ParallelizableTraversalPtrList parallelTraversals(numberOfThreads);
226  size_t begin = 0, end;
227  for (i = 0; i < numberOfThreads; i++)
228  {
229  end = begin + numberOfTraversals / numberOfThreads + 1;
230  if (end > numberOfTraversals)
231  end = numberOfTraversals;
232 
233  parallelTraversals[i]
234  = new AstSharedMemoryParallelizableSimpleProcessing(syncInfo,
235  std::vector<TraversalPtr>(traversals.begin() + begin, traversals.begin() + end));
236  begin = end;
237  }
238 
239  // Start a thread for each of the parallelizable traversals with its share
240  // of the initial inherited attributes.
241  pthread_t *threads = new pthread_t[numberOfThreads];
242  for (i = 0; i < numberOfThreads; i++)
243  {
244  pthread_create(&threads[i], NULL,
245  parallelSimpleProcessingThread,
246  new AstSharedMemoryParallelSimpleThreadArgs(parallelTraversals[i], basenode, treeTraverseOrder));
247  // std::cerr << " starting " << i << " thread. " << std::endl;
248  }
249 
250  // Main "event loop" for the "master" thread: Simply wait for the
251  // condition that is signalled when a thread is completely done with its
252  // traversal. The counter tells us when we are finished.
253  pthread_mutex_lock(syncInfo.mutex);
254  while (*syncInfo.finishedThreads < numberOfThreads)
255  pthread_cond_wait(syncInfo.threadFinishedEvent, syncInfo.mutex);
256  pthread_mutex_unlock(syncInfo.mutex);
257 
258  // Join all threads. (Strictly speaking this is not necessary since we
259  // know that they are all done, but it doesn't really const anything).
260  void *resultDummy;
261  for (i = 0; i < numberOfThreads; i++)
262  pthread_join(threads[i], &resultDummy);
263  delete [] threads;
264  // Done!
265 }
