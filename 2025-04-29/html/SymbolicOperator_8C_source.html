684  virtual void VisitExpr( const SymbolicExpr& v)
685  {
686  SymbolicVal in1, fr1;
687  bool _hasfrac = hasfrac;
688  SymbolicExpr::OpdIterator opds = v.GetOpdIterator();
689  for ( ; !opds.ReachEnd(); ++opds) {
690  if (operator()(v.Term2Val(*opds), inp, frp)) {
691  _hasfrac = true;
692  break;
693  }
694  }
695  if (opds.ReachEnd()) {
696  if (inp != 0) *inp = v;
697  }
698  else {
699  if (inp == 0 && frp == 0)
700  return;
701  SymbolicExpr* inv = (inp == 0)? 0 : v.DistributeExpr(SYMOP_NIL, SymbolicVal());
702  SymbolicExpr* frv = (frp == 0)? 0 : v.DistributeExpr(SYMOP_NIL, SymbolicVal());
703  SymbolicExpr::OpdIterator opd1 = v.GetOpdIterator();
704  for ( ; opd1 != opds ; ++opd1) {
705  if (inv != NULL)
706  inv->AddOpd(*opd1);
707  if (frv != 0)
708  frv->AddOpd(*opd1);
709  }
710  if (inv != NULL)
711  inv->ApplyOpd(*inp);
712  if (frv != NULL)
713  frv->ApplyOpd(*frp);
714  for (++opd1; !opd1.ReachEnd(); ++opd1) {
715  SymbolicVal cur = v.Term2Val(*opd1);
716  if (operator()(cur, inp, frp)) {
717  if (inv != NULL)
718  inv->ApplyOpd(*inp);
719  if (frv != NULL)
720  frv->ApplyOpd(*frp);
721  }
722  else {
723  if (inv != NULL)
724  inv->AddOpd(*opd1);
725  if (frv != NULL)
726  frv->AddOpd(*opd1);
727  }
728  }
729  if (inp != NULL) {
730  assert(inv != NULL);
731  *inp = GetExprVal(inv);
732  }
733  if (frp != NULL) {
734  assert(frv != NULL);
735  *frp = GetExprVal(frv);
736  }
737  }
738  hasfrac = _hasfrac;
739  }

746  if (frp != 0) *frp = 0;
747  hasfrac = false;
748  v.Visit(this);
