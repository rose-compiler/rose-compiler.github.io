102  bool modified = magic_tainted(node, prodLat); // some values are automatically tainted based on their name
103 
104  // Process AST nodes that transfer taintedness. Most of these operations have one or more inputs from which a result
105  // is always calculated the same way. So we just gather up the inputs and do the calculation at the very end of this
106  // function. The other operations are handled individually within their "if" bodies.
107  TaintLattice *result = NULL; // result pointer into the taint lattice
108  std::vector<TaintLattice*> inputs; // input pointers into the taint lattice
109  if (isSgAssignInitializer(node)) {
110  // as in "int a = b"
111  SgAssignInitializer *xop = isSgAssignInitializer(node);
112  TaintLattice *in1 = dynamic_cast<TaintLattice*>(prodLat->getVarLattice(SgExpr2Var(xop->get_operand())));
113  inputs.push_back(in1);
114 
115  } else if (isSgAggregateInitializer(node)) {
116  // as in "int a[1] = {b}"
117  SgAggregateInitializer *xop = isSgAggregateInitializer(node);
118  const SgExpressionPtrList &exprs = xop->get_initializers()->get_expressions();
119  for (size_t i=0; i<exprs.size(); ++i) {
120  varID in_id = SgExpr2Var(exprs[i]);
121  TaintLattice *in = dynamic_cast<TaintLattice*>(prodLat->getVarLattice(in_id));
122  inputs.push_back(in);
123  }
124 
125  } else if (isSgInitializedName(node)) {
126  SgInitializedName *xop = isSgInitializedName(node);
127  if (xop->get_initializer()) {
128  varID in1_id = SgExpr2Var(xop->get_initializer());
129  TaintLattice *in1 = dynamic_cast<TaintLattice*>(prodLat->getVarLattice(in1_id));
130  inputs.push_back(in1);
131  }
132 
133  } else if (isSgValueExp(node)) {
134  // numeric and character constants
135  SgValueExp *xop = isSgValueExp(node);
136  result = dynamic_cast<TaintLattice*>(prodLat->getVarLattice(SgExpr2Var(xop)));
137  if (result)
138  modified = result->set_vertex(TaintLattice::VERTEX_UNTAINTED);
139 
140  } else if (isSgAddressOfOp(node)) {
141  // as in "&x". The result taintedness has nothing to do with the value in x.
142  /*void*/
143 
144  } else if (isSgBinaryOp(node)) {
145  // as in "a + b"
146  SgBinaryOp *xop = isSgBinaryOp(node);
147  varID in1_id = SgExpr2Var(isSgExpression(xop->get_lhs_operand()));
148  TaintLattice *in1 = dynamic_cast<TaintLattice*>(prodLat->getVarLattice(in1_id));
149  inputs.push_back(in1);
150  varID in2_id = SgExpr2Var(isSgExpression(xop->get_rhs_operand()));
151  TaintLattice *in2 = dynamic_cast<TaintLattice*>(prodLat->getVarLattice(in2_id));
152  inputs.push_back(in2);
153 
154  if (isSgAssignOp(node)) { // copy the rhs lattice to the lhs lattice (as well as the entire '=' expression result)
155  assert(in1 && in2);
156  modified = in1->meetUpdate(in2);
157  }
158 
159  } else if (isSgUnaryOp(node)) {
160  // as in "-a"
161  SgUnaryOp *xop = isSgUnaryOp(node);
162  varID in1_id = SgExpr2Var(xop->get_operand());
163  TaintLattice *in1 = dynamic_cast<TaintLattice*>(prodLat->getVarLattice(in1_id));
164  inputs.push_back(in1);
165 
166  } else if (isSgReturnStmt(node)) {
167  // as in "return a". The result will always be dead, so we're just doing this to get some debugging output. Most
168  // of our test inputs are functions, and the test examines the function's returned taintedness.
169  SgReturnStmt *xop = isSgReturnStmt(node);
170  varID in1_id = SgExpr2Var(xop->get_expression());
171  TaintLattice *in1 = dynamic_cast<TaintLattice*>(prodLat->getVarLattice(in1_id));
172  inputs.push_back(in1);

174  }

177  // Update the result lattice (unless dead) with the inputs (unless dead) by using the meedUpdate() method. All this
178  // means is that the new result will be the maximum of the old result and all inputs, where "maximum" is defined such
179  // that "tainted" is greater than "untainted" (and both of them are greater than bottom/unknown).
180  for (size_t i=0; i<inputs.size(); ++i)
181  if (debug)
182  *debug <<"TaintAnalysis::transfer: input " <<(i+1) <<" is " <<lattice_info(inputs[i]) <<"\n";
183  if (!result && varID::isValidVarExp(node)) {
184  varID result_id(node); // NOTE: constructor doesn't handle all SgExpression nodes, thus the next "if"
185  result = dynamic_cast<TaintLattice*>(prodLat->getVarLattice(result_id));
186  }
187  if (!result && isSgExpression(node)) {
188  varID result_id = SgExpr2Var(isSgExpression(node));
189  result = dynamic_cast<TaintLattice*>(prodLat->getVarLattice(result_id));
190  }
191  if (result) {

193  if (inputs[i])

196  }
