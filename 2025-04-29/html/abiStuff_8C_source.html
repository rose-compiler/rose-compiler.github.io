13 StructLayoutInfo ChainableTypeLayoutGenerator::layoutType(SgType* t) const {
14  // Default implementation just passes everything (except typedefs and
15  // modifiers) on to next
16  switch (t->variantT()) {
17  case V_SgTypedefType: {
18  SgType* t2 = isSgTypedefType(t)->get_base_type();
19  return this->beginning->layoutType(t2);
20  }
21  case V_SgModifierType: {
22  SgType* t2 = isSgModifierType(t)->get_base_type();
23  return this->beginning->layoutType(t2);
24  }
25  default: {
26  if (this->next) {
27  return this->next->layoutType(t);
28  } else {
29  std::cerr << "Reached end of chain when trying to lay out a " << t->class_name() << std::endl;
30  abort();
31  }
32  }
33  }

35  // but it not clear how to modify this code to avoid that warning.
36 }

38 void NonpackedTypeLayoutGenerator::layoutOneField(SgType* fieldType, SgNode* decl, bool isUnion, size_t& currentOffset, StructLayoutInfo& layout) const {
39  StructLayoutInfo fieldInfo = this->beginning->layoutType(fieldType);
40  if (fieldInfo.alignment > layout.alignment) {
41  layout.alignment = fieldInfo.alignment;
42  }
43  if (currentOffset % fieldInfo.alignment != 0) {
44  size_t paddingNeeded = fieldInfo.alignment - (currentOffset % fieldInfo.alignment);
45  if (!isUnion) {
46  layout.fields.push_back(StructLayoutEntry(NULL, currentOffset, paddingNeeded));
47  }
48  currentOffset += paddingNeeded;
49  }
50  layout.fields.push_back(StructLayoutEntry(decl, currentOffset, fieldInfo.size));
51  currentOffset += fieldInfo.size; // We need to do this even for unions to set the object size
52  if (currentOffset > layout.size) {
53  layout.size = currentOffset;
54  }
55  if (isUnion) {
56  ROSE_ASSERT (currentOffset == fieldInfo.size);
57  currentOffset = 0;
58  }
59 }

61 StructLayoutInfo NonpackedTypeLayoutGenerator::layoutType(SgType* t) const {
62  switch (t->variantT()) {
63  case V_SgClassType: { // Also covers structs and unions
64  SgClassDeclaration* decl = isSgClassDeclaration(isSgClassType(t)->get_declaration());
65  ROSE_ASSERT (decl);
66  decl = isSgClassDeclaration(decl->get_definingDeclaration());
67  ROSE_ASSERT (decl);
68  SgClassDefinition* def = decl->get_definition();
69  ROSE_ASSERT (def);
70  StructLayoutInfo layout;
71  size_t currentOffset = 0;
72  const SgBaseClassPtrList& bases = def->get_inheritances();
73  for (SgBaseClassPtrList::const_iterator i = bases.begin();
74  i != bases.end(); ++i) {
75  SgBaseClass* base = *i;
76  SgClassDeclaration* basecls = base->get_base_class();
77  layoutOneField(basecls->get_type(), base, false, currentOffset, layout);
78  }
79  const SgDeclarationStatementPtrList& body = def->get_members();
80  bool isUnion = (decl->get_class_type() == SgClassDeclaration::e_union);
81  for (SgDeclarationStatementPtrList::const_iterator i = body.begin();
82  i != body.end(); ++i) {
83  SgDeclarationStatement* mem = *i;
84  SgVariableDeclaration* vardecl = isSgVariableDeclaration(mem);
85  SgClassDeclaration* classdecl = isSgClassDeclaration(mem);
86  bool isUnnamedUnion = classdecl ? classdecl->get_isUnNamed() : false;
87  if (vardecl) {
88  if (!vardecl->get_declarationModifier().isDefault()) continue; // Static fields and friends
89  ROSE_ASSERT (!vardecl->get_bitfield());
90  const SgInitializedNamePtrList& vars = isSgVariableDeclaration(mem)->get_variables();
91  for (SgInitializedNamePtrList::const_iterator j = vars.begin();
92  j != vars.end(); ++j) {
93  SgInitializedName* var = *j;
94  layoutOneField(var->get_type(), var, isUnion, currentOffset, layout);
95  }
96  } else if (isUnnamedUnion) {
97  layoutOneField(classdecl->get_type(), classdecl, isUnion, currentOffset, layout);
98  } // else continue;
99  }
100  if (layout.alignment != 0 && layout.size % layout.alignment != 0) {
101  size_t paddingNeeded = layout.alignment - (layout.size % layout.alignment);
102  if (!isUnion) {
103  layout.fields.push_back(StructLayoutEntry(NULL, layout.size, paddingNeeded));
104  }
105  layout.size += paddingNeeded;
106  }
107  return layout;
108  }
109  case V_SgArrayType: {
110  StructLayoutInfo layout = this->beginning->layoutType(isSgArrayType(t)->get_base_type());
111  layout.fields.clear();
112  SgExpression* numElements = isSgArrayType(t)->get_index();
113 
114  //Adjustment for UPC array like a[100*THREADS],treat it as a[100]
115  // Liao, 8/7/2008
116  if (isUpcArrayWithThreads(isSgArrayType(t)))
117  {
118  SgMultiplyOp* multiply = isSgMultiplyOp(isSgArrayType(t)->get_index());
119  ROSE_ASSERT(multiply);
120 
121  // DQ (9/26/2011): Do constant folding if required.
122  // SageInterface::constantFolding(multiply);
123 
124  numElements = multiply->get_lhs_operand();
125  }
126  SgValueExp* valuexpr = isSgValueExp(numElements);
127  if (valuexpr == NULL) {
128  cerr << "Error: trying to compute static size of an array with non-constant size" << endl;
129  abort();
130  }
131  layout.size *= SageInterface::getIntegerConstantValue(valuexpr);
132  return layout;
133  }
134  case V_SgTypeComplex: {
135  //"Each complex type has the same representation and alignment requirements as
136  //an array type containing exactly two elements of the corresponding real type"
137  StructLayoutInfo layout = this->beginning->layoutType(isSgTypeComplex(t)->get_base_type());
138  layout.size *= 2;
139  return layout;
140  }
141  case V_SgTypeImaginary: {
142  StructLayoutInfo layout = this->beginning->layoutType(isSgTypeImaginary(t)->get_base_type());
143  return layout;
144  }
145 
146  default: return ChainableTypeLayoutGenerator::layoutType(t);
147  }
148 }
