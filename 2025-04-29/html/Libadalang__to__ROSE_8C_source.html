122 {
123  // \todo does there exist a libadalang function to convert UTF32 to Unicode???
124  // If not, it would be preferable to use built in C++ mechanisms for the
125  // conversion of UTF32 to Unicode as supported by C++ strings.
126  // For some reason, the code below does not generate linkable files as
127  // the instantiation of std::codecvt_utf8<uint32_t, 0x10ffff, (std::codecvt_mode)0>
128  // remains missing. Even with the added explicit instantiation, many
129  // member functions remain missing.
130  //
131  //~ template std::codecvt_utf8<uint32_t, 0x10ffff, (std::codecvt_mode)0>;
132  //~ using Utf32Converter = std::codecvt_utf8<uint32_t, 0x10ffff, (std::codecvt_mode)0>;
133 
134  //~ if (text_type == nullptr || text_type->items == nullptr) return {};
135 
136  //~ std::wstring_convert<Utf32Converter, uint32_t> convert;
137  //~ return convert.to_bytes(text_type->items, text_type->items + text_type->n);
138  //
139  // => use handcrafted conversion for now.
140  struct AppendUTF32ToString
141  {
142  // see also: https://stackoverflow.com/questions/42012563/convert-unicode-code-points-to-utf-8-and-utf-32
143  std::string operator()(std::string s, uint32_t code) const
144  {
145  if (code <= 0x7F) {
146  s.append(1, code);
147  } else if (code <= 0x7FF) {
148  s.append(1, 0xC0 | (code >> 6)); /* 110xxxxx */
149  s.append(1, 0x80 | (code & 0x3F)); /* 10xxxxxx */
150  } else if (code <= 0xFFFF) {
151  s.append(1, 0xE0 | (code >> 12)); /* 1110xxxx */
152  s.append(1, 0x80 | ((code >> 6) & 0x3F)); /* 10xxxxxx */
153  s.append(1, 0x80 | (code & 0x3F)); /* 10xxxxxx */
154  } else if (code <= 0x10FFFF) {
155  s.append(1, 0xF0 | (code >> 18)); /* 11110xxx */
156  s.append(1, 0x80 | ((code >> 12) & 0x3F)); /* 10xxxxxx */
157  s.append(1, 0x80 | ((code >> 6) & 0x3F)); /* 10xxxxxx */
158  s.append(1, 0x80 | (code & 0x3F)); /* 10xxxxxx */
159  } else {
160  logError() << "Unknown unicode codepoint: " << code << std::endl;
161  }
162 
163  return s;
164  }
165  };
166 
167  if (text_type == nullptr)
168  {
169  logWarn() << "text_type = nullptr" << std::endl;
170  return std::string{};
171  }
172 
173  const int N = text_type->n;
174 
175  return std::accumulate( text_type->items, text_type->items+N,
176  std::string{},
177  AppendUTF32ToString{}
178  );
179 }
