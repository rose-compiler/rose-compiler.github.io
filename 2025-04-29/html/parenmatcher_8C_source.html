33 bool ParenMatcher::match( Q3TextCursor *cursor )
34 {
35  if ( !enabled )
36  return FALSE;
37  bool ret = FALSE;
38 
39  QChar c( cursor->paragraph()->at( cursor->index() )->c );
40  bool ok1 = FALSE;
41  bool ok2 = FALSE;
42  if ( c == '{' || c == '(' || c == '[' ) {
43  ok1 = checkOpenParen( cursor );
44  ret = ok1 || ret;
45  } else if ( cursor->index() > 0 ) {
46  c = cursor->paragraph()->at( cursor->index() - 1 )->c;
47  if ( c == '}' || c == ')' || c == ']' ) {
48  ok2 = checkClosedParen( cursor );
49  ret = ok2 || ret;
50  }
51  }
52  return ret;
53 }

55 bool ParenMatcher::checkOpenParen( Q3TextCursor *cursor )
56 {
57  if ( !cursor->paragraph()->extraData() )
58  return FALSE;
59  Q3ValueList<Symbol> parenList = ( (ParagData*)cursor->paragraph()->extraData() )->symbolList();
60 
61  Symbol openParen, closedParen;
62  Q3TextParagraph *closedParenParag = cursor->paragraph();
63 
64  int i = 0;
65  int ignore = 0;
66  bool foundOpen = FALSE;
67  QChar c = cursor->paragraph()->at( cursor->index() )->c;
68  for (;;) {
69  if ( !foundOpen ) {
70  if ( i >= (int)parenList.count() )
71  goto bye;
72  openParen = parenList[ i ];
73  if ( openParen.pos() != cursor->index() ) {
74  ++i;
75  continue;
76  } else {
77  foundOpen = TRUE;
78  ++i;
79  }
80  }
81 
82  if ( i >= (int)parenList.count() ) {
83  for (;;) {
84  closedParenParag = closedParenParag->next();
85  if ( !closedParenParag )
86  goto bye;
87  if ( closedParenParag->extraData() &&
88  ( (ParagData*)closedParenParag->extraData() )->symbolList().count() > 0 ) {
89  parenList = ( (ParagData*)closedParenParag->extraData() )->symbolList();
90  break;
91  }
92  }
93  i = 0;
94  }
95 
96  closedParen = parenList[ i ];
97  if ( closedParen.type() == Symbol::Left ) {
98  ignore++;
99  ++i;
100  continue;
101  } else {
102  if ( ignore > 0 ) {
103  ignore--;
104  ++i;
105  continue;
106  }
107 
108  int id = Match;
109  if ( c == '{' && closedParen.ch() != '}' ||
110  c == '(' && closedParen.ch() != ')' ||
111  c == '[' && closedParen.ch() != ']' )
112  id = Mismatch;
113  cursor->document()->setSelectionStart( id, *cursor );
114  int tidx = cursor->index();
115  Q3TextParagraph *tstring = cursor->paragraph();
116  cursor->setParagraph( closedParenParag );
117  cursor->setIndex( closedParen.pos() + 1 );
118  cursor->document()->setSelectionEnd( id, *cursor );
119  cursor->setParagraph( tstring );
120  cursor->setIndex( tidx );
121  return TRUE;
122  }
123  }
124 
125  bye:
126  return FALSE;
127 }

137 
138  int i = parenList.count() - 1;

142  for (;;) {
143  if ( !foundClosed ) {
144  if ( i < 0 )
