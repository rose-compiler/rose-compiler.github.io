303 void
304 FormattedTable::printRow(std::ostream &out, const std::vector<size_t> &widths, const std::vector<CellProperties> &props,
305  const std::vector<std::string> &row) const {
306  switch (format_) {
307  case Format::PLAIN: {
308  std::vector<std::vector<std::string>> columnLines;
309  size_t maxCellHeight = 0;
310  for (const std::string &cell: row) {
311  columnLines.push_back(StringUtility::split("\n", cell));
312  maxCellHeight = std::max(maxCellHeight, columnLines.back().size());
313  }
314 
315  // i == 0 means we print a line even if all columns of the row are empty
316  for (size_t i = 0; i == 0 || i < maxCellHeight; ++i) {
317  out <<indentation_ <<"|";
318  for (size_t j = 0; j < widths.size(); ++j) {
319  const std::string s = i < columnLines[j].size() ? columnLines[j][i] : "";
320  out <<" " <<cellPropertiesBegin(props[j]);
321  switch (props[j].alignment().orElse(Alignment::LEFT)) {
322  case Alignment::LEFT:
323  out <<StringUtility::leftJustify(s, widths[j]);
324  break;
325  case Alignment::RIGHT:
326  out <<StringUtility::rightJustify(s, widths[j]);
327  break;
328  case Alignment::CENTER:
329  out <<StringUtility::centerJustify(s, widths[j]);
330  break;
331  }
332  out <<cellPropertiesEnd(props[j]) <<" |";
333  }
334  out <<"\n";
335  }
336  break;
337  }
338 
339  case Format::HTML: {
340  out <<"<tr>\n";
341  for (size_t j = 0; j < widths.size(); ++j) {
342  out <<"<td style=\"";
343  switch (props[j].alignment().orElse(Alignment::LEFT)) {
344  case Alignment::LEFT:
345  out <<"text-align: left;";
346  break;
347  case Alignment::RIGHT:
348  out <<"text-align: right;";
349  break;
350  case Alignment::CENTER:
351  out <<"text-align: center;";
352  break;
353  }
354  out <<cellPropertiesBegin(props[j]);
355  out <<"\">";
356  const std::string s = row[j].empty() ? "&nbsp;" : row[j];
357  out <<s <<"</td>\n";
358  }
359  out <<"</tr>\n";
360  break;
361  }
362 
363  case Format::CSV: {
364  // See RFC 4180 [https://datatracker.ietf.org/doc/html/rfc4180]
365  for (size_t j = 0; j < widths.size(); ++j)
366  out <<(j?",":"") <<StringUtility::csvEscape(row[j]);
367  out <<"\n";
368  break;
369  }
370 
371  case Format::SHELL: {
372  // See enum for documentation
373  for (size_t j = 0; j < widths.size(); ++j) {
374  bool needQuotes = false;
375  for (char ch: row[j]) {
376  if (!::isalnum(ch) && !strchr("_-+./", ch)) {
377  needQuotes = true;
378  break;
379  }
380  }
381  out <<(j ? "\t" : "") <<(needQuotes ? "$'" + StringUtility::cEscape(row[j],'\'') + "'" : row[j]);
382  }
383  out <<"\n";
384  break;
385  }
386  }

389 void
390 FormattedTable::print(std::ostream &out) const {
391  std::vector<size_t> widths = computeColumnWidths();
392  if (widths.empty())
393  return;
394 
395  if (Format::HTML == format_)
396  out <<"<table border=\"1\">\n";
397 
398  // Headers
399  if (!columnHeaders_.empty() && format_ != Format::CSV && format_ != Format::SHELL) {
400  printHorizontalRule(out, widths);
401  for (size_t i = 0; i < columnHeaders_.size(); ++i) {
402  std::vector<CellProperties> props;
403  props.reserve(widths.size());
404  for (size_t j = 0; j < widths.size(); ++j)
405  props.push_back(columnHeaderProperties(i, j));
406  printRow(out, widths, props, columnHeaders_[i]);
407  }
408  }
409 
410  // Body
411  printHorizontalRule(out, widths);
412  for (size_t i = 0; i < cells_.size(); ++i)
413  printRow(out, widths, props_[i], cells_[i]);
414  printHorizontalRule(out, widths);
415 
416  if (Format::HTML == format_)
417  out <<"<table>\n";
418 }
