456 bool
457 SyscallBase::operator()(bool /*handled*/, SyscallContext &ctx) {
458  // Since a system call might have multiple callbacks, and since all of them inherit from this class, they will all end up
459  // calling this code. Some things (such as creating new events and single stepping the concrete process) should only be
460  // done once, so be careful!
461  hello("linux i386 base class", ctx);
462  Sawyer::Message::Stream debug(mlog[DEBUG]);
463  auto i386 = as<Architecture>(ctx.architecture);
464  ASSERT_not_null(i386);
465  const RegisterDescriptor SYS_RET = i386->systemCallReturnRegister();
466  penultimateReturnEvent_ = latestReturnEvent_;
467  latestReturnEvent_ = ExecutionEvent::Ptr();
468 
469  if (ConcolicPhase::REPLAY == ctx.phase) {
470  // Pick out the return value and save it for later. The return value will have been created as an event that describes
471  // the side effect of writing the return value to the return register. When replaying, the return value will be
472  // concrete.
473  if (!ctx.returnEvent) {
474  for (const ExecutionEvent::Ptr &relatedEvent: ctx.relatedEvents) {
475  if (relatedEvent->action() == ExecutionEvent::Action::REGISTER_WRITE &&
476  relatedEvent->registerDescriptor() == SYS_RET) {
477  ctx.returnEvent = relatedEvent;
478  break;
479  }
480  }
481  }
482  ASSERT_not_null(ctx.returnEvent);
483  latestReturnEvent_ = ctx.returnEvent;
484 
485  // Allow the child class to do its thing.
486  playback(ctx);
487 
488  } else {
489  // For non-replay (i.e., concolic execution) we should make sure that there's a return event if no other callback has
490  // already created one. If we create one here, then also give it an input variable to affect the behavior of future
491  // test cases for this same specimen (subsequent callbacks can remove this variable if desired).
492  if (!ctx.returnEvent) {
493  handlePreSyscall(ctx);
494 
495  // after this, we're in the syscall-exit-stop state
496  i386->debugger()->stepIntoSystemCall(Debugger::ThreadId::unspecified());
497 
498  // If the syscall terminated the program, we should still allow the subclass to handle things even though we're
499  // not creating a return event or input variable for the return.
500  if (i386->debugger()->isTerminated()) {
501  ASSERT_require(ctx.returnEvent == nullptr);
502  handlePostSyscall(ctx);
503  ASSERT_not_reachable("unexpected concrete termination in system call");
504  }
505 
506  // Create the input variable and execution event for this return value.
507  if (!ctx.symbolicReturn) {
508  ctx.symbolicReturn = SymbolicExpression::makeIntegerVariable(SYS_RET.nBits(), ctx.syscallEvent->name() + "_return");
509  SAWYER_MESG(debug) <<" created input variable " <<*ctx.symbolicReturn <<"\n";
510  } else {
511  SAWYER_MESG(debug) <<" using existing variable " <<*ctx.symbolicReturn <<"\n";
512  ASSERT_require(ctx.symbolicReturn->nBits() == SYS_RET.nBits());
513  }
514  uint64_t retConcrete = i386->debugger()->readRegister(Debugger::ThreadId::unspecified(), SYS_RET).toInteger();
515  SymbolicExpression::Ptr retValue = SymbolicExpression::makeIntegerConstant(SYS_RET.nBits(), retConcrete);
516  SAWYER_MESG(debug) <<" concrete return value: " <<*retValue <<"\n";
517  ctx.returnEvent = ExecutionEvent::registerWrite(i386->testCase(), i386->nextEventLocation(When::POST),
518  ctx.syscallEvent->instructionPointer(), SYS_RET,
519  ctx.symbolicReturn, retValue, ctx.symbolicReturn);
520  ctx.ops->inputVariables()->activate(ctx.returnEvent, InputType::SYSCALL_RET);
521  SAWYER_MESG(debug) <<" created " <<ctx.returnEvent->printableName(i386->database()) <<"\n";
522 
523  // Update the symbolic state
524  BS::SValue::Ptr retSValue = ctx.ops->svalueExpr(ctx.symbolicReturn);
525  ctx.ops->writeRegister(SYS_RET, retSValue);
526  SAWYER_MESG(debug) <<" return value saved in symbolic state: " <<*retSValue <<"\n";
527  }
528 
529  latestReturnEvent_ = ctx.returnEvent;
530  handlePostSyscall(ctx);
531  }
532  return true; // handled
533 }

672 void
673 SyscallReturn::handlePostSyscall(SyscallContext &ctx) {
674  hello("syscall-return-constraint", ctx);
675  Sawyer::Message::Stream debug(mlog[DEBUG]);
676  auto i386 = as<Architecture>(ctx.architecture);
677  ASSERT_not_null(i386);
678  const RegisterDescriptor SYS_RET = i386->systemCallReturnRegister();
679  showRecentReturnValues(debug, ctx);
680 
681  // If this is the first invocation of this syscall, then there's nothing we need to do.
682  if (!penultimateReturnEvent())
683  return;
684 
685  // Build the SMT solver constraint that the current return value must be equal to the previous return value.
686  std::pair<SymbolicExpression::Ptr, Sawyer::Optional<uint64_t>> x = makeReturnConstraint(ctx);
687  SymbolicExpression::Ptr constraint = x.first;
688  Sawyer::Optional<uint64_t> concreteReturn = x.second;
689 
690  if (constraint) {
691  SAWYER_MESG(mlog[DEBUG]) <<" return value constraint: " <<*constraint <<"\n";
692  ctx.ops->solver()->insert(constraint);
693  } else {
694  SAWYER_MESG(mlog[DEBUG]) <<" return value constraint: none\n";
695  }
696 
697  if (concreteReturn) {
698  SAWYER_MESG(mlog[DEBUG]) <<" modifying concrete state to have returned "
699  <<StringUtility::toHex2(*concreteReturn, SYS_RET.nBits()) <<"\n";
700  i386->debugger()->writeRegister(Debugger::ThreadId::unspecified(), SYS_RET, *concreteReturn);
701  } else {
702  SAWYER_MESG(mlog[DEBUG]) <<" not modifying concrete return value\n";
703  }
704 }

1036  ASSERT_forbid(isFactory());

1127 std::vector<ExecutionEvent::Ptr>
1128 Architecture::createMemoryAdjustEvents(const MemoryMap::Ptr &oldMap, Address insnVa) {
1129  ASSERT_forbid(isFactory());
1130  std::vector<ExecutionEvent::Ptr> retval;
1131  SAWYER_MESG(mlog[DEBUG]) <<"saving memory map adjustments\n";
1132  std::vector<ExecutionEvent::Ptr> events;
1133  auto newMap = MemoryMap::instance();
1134  std::vector<MemoryMap::ProcessMapRecord> segments = disposableMemory();
1135  newMap->insertProcessPid(*debugger()->processId(), segments);
1136 
1137 #if 1 // DEBUGGING [Robb Matzke 2021-12-20]
1138  if (mlog[DEBUG]) {
1139  mlog[DEBUG] <<" old map:\n";
1140  oldMap->dump(mlog[DEBUG], " ");
1141  mlog[DEBUG] <<" new map:\n";
1142  newMap->dump(mlog[DEBUG], " ");
1143  }
1144 #endif
1145 
1146  // Addresses that exist in the old map but not the new map are removed from the old map and unmap events are generated.
1147  Address oldVa = 0;
1148  while (AddressInterval oldWhere = oldMap->atOrAfter(oldVa).available()) {
1149  AddressInterval newWhere = newMap->atOrAfter(oldWhere.least()).available();
1150  Address lastProcessedVa = 0;
1151 
1152  if (!newWhere) {
1153  // No addresses here or higher in new map, so delete the rest of the old map
1154  SAWYER_MESG(mlog[DEBUG]) <<" erasing " <<StringUtility::addrToString(oldWhere)
1155  <<" " <<StringUtility::plural(oldWhere.size(), "bytes") <<"\n";
1156  oldMap->erase(oldWhere);
1157  lastProcessedVa = oldWhere.greatest();
1158  auto event = ExecutionEvent::bulkMemoryUnmap(testCase(), nextEventLocation(When::POST), insnVa, oldWhere);
1159  event->name("unmap " + StringUtility::addrToString(oldWhere));
1160  retval.push_back(event);
1161 
1162  } else if (newWhere.least() > oldWhere.least()) {
1163  // The next mapped interval in the new address map starts after the current address, so delete all parts of the old
1164  // map up to where the new map starts.
1165  auto toErase = AddressInterval::hull(oldWhere.least(), newWhere.least() - 1);
1166  SAWYER_MESG(mlog[DEBUG]) <<" erasing " <<StringUtility::addrToString(toErase)
1167  <<" " <<StringUtility::plural(toErase.size(), "bytes") <<"\n";
1168  oldMap->erase(toErase);
1169  lastProcessedVa = toErase.greatest();
1170  auto event = ExecutionEvent::bulkMemoryUnmap(testCase(), nextEventLocation(When::POST), insnVa, toErase);
1171  event->name("unmap " + StringUtility::addrToString(toErase));
1172  retval.push_back(event);
1173 
1174  } else {
1175  // The new map has data at the old map's current address, so we can fast forward across the new addresses.
1176  ASSERT_require(newWhere.least() == oldWhere.least());
1177  lastProcessedVa = newWhere.greatest();
1178  }
1179 
1180  // Increment the address, but be careful of overflow.
1181  if (lastProcessedVa < AddressInterval::whole().greatest()) {
1182  oldVa = lastProcessedVa + 1;
1183  } else {
1184  break;
1185  }
1186  }
1187 
1188  // Addresses that exist in the new map but not the old map are added to the old map and map events are generated.
1189  Address newVa = 0;
1190  while (AddressInterval newWhere = newMap->atOrAfter(newVa).singleSegment().available()) {
1191  AddressInterval oldWhere = oldMap->atOrAfter(newWhere.least()).singleSegment().available();
1192  Address lastProcessedVa = 0;
1193 
1194 
1195  if (!oldWhere) {
1196  // Old map doesn't contain any subsequent address from the new map.
1197  SAWYER_MESG(mlog[DEBUG]) <<" inserting " <<StringUtility::addrToString(newWhere) <<"\n";
1198  lastProcessedVa = newWhere.greatest();
1199  auto events = copyMemory(newMap, oldMap, newWhere, insnVa);
1200  retval.insert(retval.end(), events.begin(), events.end());
1201 
1202  } else if (oldWhere.least() > newWhere.least()) {
1203  // New map contains some addresses not in the old map.
1204  auto toInsert = AddressInterval::hull(newWhere.least(), std::min(newWhere.greatest(), oldWhere.least() - 1));
1205  SAWYER_MESG(mlog[DEBUG]) <<" inserting " <<StringUtility::addrToString(toInsert) <<"\n";
1206  lastProcessedVa = toInsert.greatest();
1207  auto events = copyMemory(newMap, oldMap, toInsert, insnVa);
1208  retval.insert(retval.end(), events.begin(), events.end());
1209 
1210  } else {
1211  // Both maps have some of the same data.
1212  ASSERT_require(newWhere.least() == oldWhere.least());
1213  lastProcessedVa = oldWhere.least();
1214  }
1215 
1216  // Increment the address, but be careful of overflow.
1217  if (lastProcessedVa < AddressInterval::whole().greatest()) {
1218  newVa = lastProcessedVa + 1;
1219  } else {
1220  break;
1221  }
1222  }
1223 
1224 #if 1 // DEBUGGING [Robb Matzke 2021-12-20]
1225  if (mlog[DEBUG]) {
1226  mlog[DEBUG] <<" adjusted map:\n";
1227  oldMap->dump(mlog[DEBUG], " ");
1228  }
1229 #endif
1230  return retval;
1231 }

1378 void
1379 Architecture::createInputVariables(const P2::PartitionerConstPtr &partitioner, const Emulation::RiscOperators::Ptr &ops,
1380  const SmtSolver::Ptr &solver) {
1381  ASSERT_forbid(isFactory());
1382  ASSERT_not_null(ops);
1383  ASSERT_not_null(solver);
1384 
1385  // For Linux ELF x86, the argc and argv values are on the stack, not in registers. Also note that there is no return
1386  // address on the stack (i.e., the stack pointer is pointing at argc, not a return address). This means all the stack
1387  // argument offsets are four (or eight) bytes less than usual.
1388  //
1389  // The stack looks like this:
1390  // +-------------------------------------------------+
1391  // | argv[argc] 4-byte zero |
1392  // +-------------------------------------------------+
1393  // | ... |
1394  // +-------------------------------------------------+
1395  // | argv[1], 4-byte address for start of a C string |
1396  // +-------------------------------------------------+
1397  // | argv[0], 4-byte address for start of a C string |
1398  // +-------------------------------------------------+
1399  // | argc value, 4-byte integer, e.g., 2 | <--- ESP points here
1400  // +-------------------------------------------------+
1401  Sawyer::Message::Stream debug(mlog[DEBUG]);
1402  SAWYER_MESG(debug) <<"creating program arguments\n";
1403  const RegisterDescriptor SP = partitioner->instructionProvider().stackPointerRegister();
1404  size_t wordSizeBytes = SP.nBits() / 8;
1405  SymbolicExpression::Formatter fmt;
1406  fmt.show_comments = SymbolicExpression::Formatter::CMT_AFTER;
1407 
1408  //---------------------------------------------------------------------------------------------------------------------------
1409  // argc
1410  //---------------------------------------------------------------------------------------------------------------------------
1411  ASSERT_require(SP.nBits() == 32); // we only handle 32-bit for now
1412  ASSERT_require(ops->currentState()->memoryState()->get_byteOrder() == memoryByteOrder());
1413  Address argcVa = readRegister(SP).toInteger();
1414  size_t argc = readMemoryUnsigned(argcVa, wordSizeBytes);
1415 
1416  // Event and input variable
1417  SymbolicExpression::Ptr argcVariable = SymbolicExpression::makeIntegerVariable(SP.nBits(), "argc");
1418  SymbolicExpression::Ptr argcValue = SymbolicExpression::makeIntegerConstant(SP.nBits(), argc);
1419  auto argcEvent = ExecutionEvent::memoryWrite(testCase(), nextEventLocation(When::PRE), ip(),
1420  AddressInterval::baseSize(argcVa, SP.nBits()/8),
1421  argcVariable, argcValue, argcVariable);
1422  inputVariables()->activate(argcEvent, InputType::ARGC);
1423 
1424  // Adjust symbolic state
1425  ops->writeMemory(RegisterDescriptor(), ops->number_(SP.nBits(), argcVa), ops->svalueExpr(argcVariable), ops->boolean_(true));
1426  ExecutionEventId argcEventId = database()->id(argcEvent);
1427  SAWYER_MESG(debug) <<" argc @" <<StringUtility::addrToString(argcVa) <<" = " <<argc
1428  <<"; symbolic = " <<(*argcVariable + fmt)
1429  <<"; event = " <<*argcEventId <<"\n";
1430 
1431  // The argc value cannot be less than 1 since it always points to at least the program name.
1432  SymbolicExpression::Ptr argcMinConstraint =
1433  SymbolicExpression::makeSignedGt(argcVariable, SymbolicExpression::makeIntegerConstant(SP.nBits(), 0));
1434  solver->insert(argcMinConstraint);
1435 
1436  // The argc value cannot be greater than its current concrete value since making it greater would mean that the address
1437  // of the environment variable list would need to change, potentially affecting many other things in the program.
1438  SymbolicExpression::Ptr argcMaxConstraint = SymbolicExpression::makeSignedLe(argcVariable, argcValue);
1439  solver->insert(argcMaxConstraint);
1440 
1441  //---------------------------------------------------------------------------------------------------------------------------
1442  // argv
1443  //---------------------------------------------------------------------------------------------------------------------------
1444 
1445  // The characters argv[0][...] are not inputs because we don't have much control over them.
1446  Address argvVa = argcVa + wordSizeBytes;
1447  SAWYER_MESG(debug) <<" argv @" <<StringUtility::addrToString(argvVa) <<" = [\n";
1448  for (size_t i = 1; i < argc; ++i) {
1449  Address ptrVa = argvVa + i * wordSizeBytes; // address of string pointer
1450  Address strVa = readMemoryUnsigned(ptrVa, wordSizeBytes);
1451  std::string s = readCString(strVa);
1452  SAWYER_MESG(debug) <<" " <<i <<": @" <<StringUtility::addrToString(strVa)
1453  <<" \"" <<StringUtility::cEscape(s) <<"\"\n";
1454 
1455  if (markingArgvAsInput_) {
1456  SymbolicExpression::Ptr anyPreviousCharIsNul; // is any previous char of this argument an ASCII NUL character?
1457  for (size_t j = 0; j <= s.size(); ++j) {
1458  Address charVa = strVa + j;
1459 
1460  // Event and input variable
1461  std::string name = (boost::format("argv_%d_%d") % i % j).str();
1462  SymbolicExpression::Ptr charValue = SymbolicExpression::makeIntegerConstant(8, s.c_str()[j]);
1463  SymbolicExpression::Ptr charVariable = SymbolicExpression::makeIntegerVariable(8, name);
1464  auto charEvent = ExecutionEvent::memoryWrite(testCase(), nextEventLocation(When::PRE), ip(),
1465  charVa, charVariable, charValue, charVariable);
1466  inputVariables()->activate(charEvent, InputType::ARGV, i, j);
1467 
1468  // Adjust symbolic state
1469  ops->writeMemory(RegisterDescriptor(), ops->number_(SP.nBits(), charVa), ops->svalueExpr(charVariable),
1470  ops->boolean_(true));
1471  ExecutionEventId charEventId = database()->id(charEvent);
1472  SAWYER_MESG(debug) <<" byte " <<j <<" @" <<StringUtility::addrToString(charVa)
1473  <<"; symbolic = " <<(*charVariable + fmt)
1474  <<"; event = " <<*charEventId <<"\n";
1475 
1476  SymbolicExpression::Ptr currentCharIsNul =
1477  SymbolicExpression::makeEq(charVariable, SymbolicExpression::makeIntegerConstant(8, 0));
1478  if (s.size() == j) {
1479  // Final byte of the argument's buffer must always be NUL
1480  solver->insert(currentCharIsNul);
1481 
1482  } else if (j > 0) {
1483  // Linux doesn't allow NUL characters to appear inside program arguments. A NUL terminates the argument and
1484  // the next argument starts immediately thereafter. For concolic testing, if an argument is shortened (by
1485  // making one of it's non-ending bytes NUL) we don't want to have to adjust the addresses of all following
1486  // arguments, environment variables, and the auxv vector because that would end up being a lot of changes
1487  // to the the program. Instead, we reserve some amount of space for each argument (based on the root test
1488  // case) and write NULs into the interior of arguments to shorten them. In order to prevent some
1489  // impossible inputs (arguments with interior NUL characters) we add solver constraints so that any
1490  // character of an argument after a NUL is also a NUL.
1491  ASSERT_not_null(anyPreviousCharIsNul);
1492  auto bothNul = SymbolicExpression::makeAnd(anyPreviousCharIsNul, currentCharIsNul);
1493  auto assertion = SymbolicExpression::makeOr(SymbolicExpression::makeInvert(anyPreviousCharIsNul), bothNul);
1494  solver->insert(assertion);
1495  } else {
1496  // argv[i] must be empty (i.e., first byte is NUL character) if argc <= i
1497  auto argcGreaterThanI = SymbolicExpression::makeGt(argcVariable,
1498  SymbolicExpression::makeIntegerConstant(SP.nBits(), i));
1499  auto assertion = SymbolicExpression::makeOr(argcGreaterThanI, currentCharIsNul);
1500  solver->insert(assertion);
1501  }
1502 
1503  // Extend or create the expression for any previous character being NUL
1504  if (anyPreviousCharIsNul) {
1505  anyPreviousCharIsNul = SymbolicExpression::makeOr(anyPreviousCharIsNul, currentCharIsNul);
1506  } else {
1507  anyPreviousCharIsNul = currentCharIsNul;
1508  }
1509  }
1510  }
1511  }
1512  SAWYER_MESG(debug) <<" " <<argc <<": @" <<StringUtility::addrToString(argvVa + argc * wordSizeBytes) <<" null\n"
1513  <<" ]\n";
1514 
1515  //---------------------------------------------------------------------------------------------------------------------------
1516  // envp
1517  //---------------------------------------------------------------------------------------------------------------------------
1518

1523  while (true) {

1525  Address strVa = readMemoryUnsigned(ptrVa, wordSizeBytes);

1528  ++envc;
1529  }
1530

1534  Address strVa = readMemoryUnsigned(ptrVa, wordSizeBytes);
1535  std::string s = readCString(strVa);
1536  SAWYER_MESG(debug) <<" " <<i <<": @" <<StringUtility::addrToString(strVa)
1537  <<" \"" <<StringUtility::cEscape(s) <<"\"\n";
1538 
1539  if (markingEnvpAsInput_) {
1540  SymbolicExpression::Ptr anyPreviousCharIsNul; // is any previous char of this env an ASCII NUL character?
1541  for (size_t j = 0; j <= s.size(); ++j) {
1542  Address charVa = strVa + j;
1543

1545  std::string name = (boost::format("envp_%d_%d") % i % j).str();

1689 void
1690 Architecture::systemCall(const P2::PartitionerConstPtr &partitioner, const BS::RiscOperators::Ptr &ops_) {
1691  // A system call has been encountered. The INT instruction has been processed symbolically (basically a no-op other than
1692  // to adjust the instruction pointer), and the concrete execution has stepped into the system call but has not yet executed
1693  // it (i.e., the subordinate process is in the syscall-enter-stop state).
1694  ASSERT_forbid(isFactory());
1695 
1696  auto ops = Emulation::RiscOperators::promote(ops_);
1697  ASSERT_not_null(ops);
1698  Sawyer::Message::Stream debug(mlog[DEBUG]);
1699  const Address ip = debugger()->executionAddress(Debugger::ThreadId::unspecified());
1700 
1701  //-------------------------------------
1702  // Create system call execution event.
1703  //-------------------------------------
1704 
1705  // Gather info about the system call such as its arguments. On Linux, system calls have up to six arguments stored
1706  // in registers, so we just grab all six for now since we don't want to maintain a big switch statement to say how
1707  // many arguments each system call actually uses.
1708  std::vector<uint64_t> argsConcrete;
1709  uint64_t functionNumber = systemCallFunctionNumber(partitioner, ops);
1710  for (size_t i = 0; i < 6; ++i) {
1711  BS::SValue::Ptr argSymbolic = systemCallArgument(partitioner, ops, i);
1712  if (auto argConcrete = argSymbolic->toUnsigned()) {
1713  argsConcrete.push_back(*argConcrete);
1714  } else {
1715  ASSERT_not_implemented("non-concrete system call argument");
1716  }
1717  }
1718  if (debug) {
1719  debug <<" " <<syscallName(functionNumber) <<" system call (sys" <<functionNumber <<"), potential args:\n";
1720  for (uint64_t arg: argsConcrete)
1721  debug <<" " <<StringUtility::toHex(arg) <<"\n";
1722  }
1723 
1724  // The execution event records the system call number and arguments, but not any side effects (because side effects haven't
1725  // happened yet). Since the side effect events (created shortly) are general things like "write this value to this
1726  // register", the fact that they're preceded by this syscall event is what marks them as being side effects of this system
1727  // call.
1728  auto syscallEvent = ExecutionEvent::osSyscall(testCase(), nextEventLocation(When::PRE), ip, functionNumber, argsConcrete);
1729  syscallEvent->name(syscallName(functionNumber) + "_" + boost::lexical_cast<std::string>(syscallEvent->location().primary()));
1730  database()->save(syscallEvent);
1731  SAWYER_MESG(debug) <<" created " <<syscallEvent->printableName(database()) <<"\n";
1732 
1733  //-------------------------------------
1734  // Process the system call
