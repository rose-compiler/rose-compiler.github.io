386 void
387 Architecture::sharedMemory(const SharedMemoryCallback::Ptr &callback, const AddressInterval &where) {
388  ASSERT_forbid(where.isEmpty());
389  ASSERT_not_null(callback);
390 
391  AddressInterval remaining = where;
392  while (!remaining.isEmpty()) {
393  SharedMemoryCallbacks callbacks;
394  auto iter = sharedMemory_.findFirstOverlap(remaining);
395  AddressInterval whereToInsert;
396  if (iter == sharedMemory_.nodes().end()) {
397  whereToInsert = remaining;
398  } else if (remaining.least() < iter->key().least()) {
399  whereToInsert = AddressInterval::hull(remaining.least(), iter->key().least() - 1);
400  } else {
401  whereToInsert = remaining & iter->key();
402  callbacks = sharedMemory_.get(whereToInsert.least());
403  }
404 
405  callbacks.append(callback);
406  sharedMemory_.insert(whereToInsert, callbacks);
407 
408  if (whereToInsert == remaining)
409  break;
410  remaining = AddressInterval::hull(whereToInsert.greatest() + 1, remaining.greatest());
411  }
412 }

465 bool
466 Architecture::playEvent(const ExecutionEvent::Ptr &event) {
467  ASSERT_forbid(isFactory());
468  ASSERT_not_null(event);
469  Sawyer::Message::Stream debug(mlog[DEBUG]);
470 
471  SAWYER_MESG(debug) <<"processing " <<event->printableName(database())
472  <<" ip=" <<StringUtility::addrToString(event->instructionPointer()) <<"\n";
473  inputVariables()->bind(event);
474 
475  switch (event->action()) {
476  case ExecutionEvent::Action::NONE:
477  SAWYER_MESG(debug) <<" no action necessary\n";
478  return true;
479 
480  case ExecutionEvent::Action::BULK_MEMORY_MAP: {
481  AddressInterval where = event->memoryLocation();
482  ASSERT_forbid(where.isEmpty());
483  if (debug) {
484  debug <<" map " <<where.size() <<" bytes at " <<StringUtility::addrToString(where) <<", prot=";
485  if ((event->permissions() & MemoryMap::READABLE) != 0)
486  debug <<"r";
487  if ((event->permissions() & MemoryMap::WRITABLE) != 0)
488  debug <<"w";
489  if ((event->permissions() & MemoryMap::EXECUTABLE) != 0)
490  debug <<"x";
491  if ((event->permissions() & MemoryMap::IMMUTABLE) != 0)
492  debug <<"i";
493  if ((event->permissions() & MemoryMap::PRIVATE) != 0)
494  debug <<"p";
495  const unsigned otherMask = ~(MemoryMap::READ_WRITE_EXECUTE | MemoryMap::IMMUTABLE | MemoryMap::PRIVATE);
496  if (const unsigned other = event->permissions() & otherMask)
497  debug <<(boost::format("+0x%08x") % (event->permissions() & otherMask));
498  debug <<"\n";
499  }
500  mapMemory(where, event->permissions());
501  return true;
502  }
503 
504  case ExecutionEvent::Action::BULK_MEMORY_UNMAP: {
505  AddressInterval where = event->memoryLocation();
506  ASSERT_forbid(where.isEmpty());
507  unmapMemory(where);
508  return true;
509  }
510 
511  case ExecutionEvent::Action::BULK_MEMORY_WRITE: {
512  AddressInterval where = event->memoryLocation();
513  SAWYER_MESG(debug) <<" write memory " <<StringUtility::plural(where.size(), "bytes")
514  <<" at " <<StringUtility::addrToString(where) <<"\n";
515  ASSERT_forbid(where.isEmpty());
516  ASSERT_require(where.size() == event->bytes().size());
517  size_t nWritten = writeMemory(where.least(), event->bytes());
518  if (nWritten != where.size())
519  mlog[ERROR] <<"failed to write to memory\n";
520  return true;
521  }
522 
523  case ExecutionEvent::Action::MEMORY_WRITE: {
524  AddressInterval where = event->memoryLocation();
525  SAWYER_MESG(debug) <<" write memory " <<StringUtility::plural(where.size(), "bytes")
526  <<" at " <<StringUtility::addrToString(where) <<"\n";
527  ASSERT_forbid(where.isEmpty());
528  SymbolicExpression::Ptr value = event->calculateResult(inputVariables()->bindings());
529  ASSERT_require(8*where.size() == value->nBits());
530 
531  // We do it this way because the value could, theoretically, be very wide.
532  std::vector<uint8_t> bytes;
533  bytes.reserve(where.size());
534  Sawyer::Container::BitVector bits = value->isLeafNode()->bits();
535  for (size_t i = 0; i < where.size(); ++i) {
536  uint8_t byte = bits.toInteger(Sawyer::Container::BitVector::BitRange::baseSize(8*i, 8));
537  bytes.push_back(byte);
538  }
539  size_t nWritten = writeMemory(where.least(), bytes);
540  if (nWritten != where.size())
541  mlog[ERROR] <<"failed to write to memory\n";
542  return true;
543  }
544 
545  case ExecutionEvent::Action::BULK_MEMORY_HASH: {
546  AddressInterval where = event->memoryLocation();
547  SAWYER_MESG(debug) <<" hash memory " <<StringUtility::plural(where.size(), "bytes")
548  <<" at " <<StringUtility::addrToString(where) <<"\n";
549  ASSERT_forbid(where.isEmpty());
550  std::vector<uint8_t> buf = readMemory(where.least(), where.size());
551  if (buf.size() != where.size()) {
552  mlog[ERROR] <<"memory hash comparison failed at " <<StringUtility::addrToString(where) <<": read error\n";
553  } else {
554  Combinatorics::HasherSha256Builtin hasher;
555  hasher.insert(buf);
556  Combinatorics::Hasher::Digest currentDigest = hasher.digest();
557  const Combinatorics::Hasher::Digest &savedDigest = event->bytes();
558  ASSERT_require(currentDigest.size() == savedDigest.size());
559  if (!std::equal(currentDigest.begin(), currentDigest.end(), savedDigest.begin()))
560  mlog[ERROR] <<"memory hash comparison failed at " <<StringUtility::addrToString(where) <<": hash differs\n";
561  }
562  return true;
563  }
564 
565  case ExecutionEvent::Action::REGISTER_WRITE: {
566  const RegisterDescriptor reg = event->registerDescriptor();
567  const uint64_t concreteValue = event->calculateResult(inputVariables()->bindings())->toUnsigned().get();
568  SAWYER_MESG(debug) <<" write register "
569  <<reg <<" = " <<toString(concreteValue, reg.nBits()) <<"\n";
570  writeRegister(reg, concreteValue);
571  return true;
572  }
573 
574  case ExecutionEvent::Action::BULK_REGISTER_WRITE: {
575  SAWYER_MESG(mlog[DEBUG]) <<" restore registers\n";
576  Sawyer::Container::BitVector allRegisters = event->registerValues();
577  debugger()->writeAllRegisters(Debugger::ThreadId::unspecified(), allRegisters);
578  return true;
579  }
580 
581  case ExecutionEvent::Action::OS_SYSCALL: {
582  // This is only the start of a system call. Additional following events for the same instruction will describe the
583  // effects of the system call.
584  const uint64_t functionNumber = event->syscallFunction();
585  SyscallCallbacks callbacks = systemCalls().getOrDefault(functionNumber);
586  SyscallContext ctx(sharedFromThis(), event, getRelatedEvents(event));
587  return callbacks.apply(false, ctx);
588  }
589 
590  case ExecutionEvent::Action::OS_SHARED_MEMORY: {
591  // This is only the start of a shared memory read. Additional following events for the same instruction will
592  // describe the effects of the read.
593  SharedMemoryCallbacks callbacks = sharedMemory().getOrDefault(event->memoryLocation().least());
594  SharedMemoryContext ctx(sharedFromThis(), event);
595  return callbacks.apply(false, ctx);
596  }
597 
598  default:
599  // Not handled here, so may need to be handled by a subclass.
600  return false;
601  }
602 }

623 uint64_t
624 Architecture::readMemoryUnsigned(Address va, size_t nBytes) {
625  ASSERT_forbid(isFactory());
626  ASSERT_require(nBytes >= 1 && nBytes <= 8);
627  std::vector<uint8_t> bytes = readMemory(va, nBytes);
628  ASSERT_require2(bytes.size() == nBytes,
629  "va=" + StringUtility::addrToString(va) +
630  ", requested=" + boost::lexical_cast<std::string>(nBytes) +
631  ", read_size=" + boost::lexical_cast<std::string>(bytes.size()));
632  uint64_t retval = 0;
633  switch (memoryByteOrder()) {
634  case ByteOrder::ORDER_LSB:
635  for (size_t i = 0; i < nBytes; ++i)
636  retval |= (uint64_t)bytes[i] << (8*i);
637  break;
638  case ByteOrder::ORDER_MSB:
639  for (size_t i = 0; i < nBytes; ++i)
640  retval = (retval << 8) | (uint64_t)bytes[i];
641  break;
642  default:
643  ASSERT_not_implemented("byte order is not LSB or MSB");
644  }
645  return retval;
646 }

719 Architecture::sharedMemoryRead(const SharedMemoryCallbacks &callbacks, const P2::Partitioner::ConstPtr&,
720  const Emulation::RiscOperators::Ptr &ops, Address addr, size_t nBytes) {
721  // A shared memory read has just been encountered, and we're in the middle of executing the instruction that caused it.
722  ASSERT_forbid(isFactory());
723  ASSERT_not_null(ops);
724  ASSERT_not_null2(ops->currentInstruction(), "must be called during instruction execution");
725  Sawyer::Message::Stream debug(mlog[DEBUG]);
726 
727  const Address ip = ops->currentInstruction()->get_address();
728  SAWYER_MESG(debug) <<" shared memory read at instruction " <<StringUtility::addrToString(ip)
729  <<" from memory address " <<StringUtility::addrToString(addr)
730  <<" for " <<StringUtility::plural(nBytes, "bytes") <<"\n";
731 
732  // Create an input variable for the value read from shared memory, and bind it to a new event that indicates that this
733  // instruction is reading from shared memory.
734  ExecutionLocation loc = nextEventLocation(When::PRE);
735  std::string name = (boost::format("shm_read_%s_%d") % StringUtility::addrToString(addr).substr(2) % loc.primary()).str();
736  auto valueRead = SymbolicExpression::makeIntegerVariable(8 * nBytes, name);
737  auto sharedMemoryEvent = ExecutionEvent::osSharedMemory(testCase(), loc, ip,
738  AddressInterval::baseSize(addr, nBytes), valueRead,
739  SymbolicExpression::Ptr(), /*concrete value not known yet*/
740  valueRead);
741  inputVariables()->activate(sharedMemoryEvent, InputType::SHMEM_READ);
742  database()->save(sharedMemoryEvent);
743  SAWYER_MESG(debug) <<" created input variable " <<*valueRead <<" (v" <<*valueRead->variableId() <<")"
744  <<" for " <<sharedMemoryEvent->printableName(database()) <<"\n";
745 
746  // Invoke the callbacks
747  SharedMemoryContext ctx(sharedFromThis(), ops, sharedMemoryEvent);
748  bool handled = callbacks.apply(false, ctx);
749  ASSERT_require(ctx.sharedMemoryEvent == sharedMemoryEvent);
750 
751  if (!handled) {
752  mlog[ERROR] <<" shared memory read not handled by any callbacks; treating it as normal memory\n";
753  return {ExecutionEvent::Ptr(), SymbolicExpression::Ptr()};
754  } else if (!ctx.valueRead) {
755  SAWYER_MESG(debug) <<" shared memory read did not return a special value; doing a normal read\n";
756  } else {
757  SAWYER_MESG(debug) <<" shared memory read returns " <<*ctx.valueRead <<"\n";
758  ASSERT_require(ctx.valueRead->nBits() == 8 * nBytes);
759  if (ctx.valueRead->isScalarConstant())
760  sharedMemoryEvent->value(ctx.valueRead);
761  }
762 
763  // Post-callback actions
764  database()->save(sharedMemoryEvent); // just in case the user modified it.
765  return {ctx.sharedMemoryEvent, ctx.valueRead};
766 }

803 void
804 Architecture::fixupSharedMemoryEvents(const ExecutionEvent::Ptr &sharedMemoryEvent, const Emulation::RiscOperators::Ptr &ops) {
805  ASSERT_forbid(isFactory());
806  ASSERT_not_null(sharedMemoryEvent);
807  ASSERT_not_null(ops);
808  Sawyer::Message::Stream debug(mlog[DEBUG]);
809  SAWYER_MESG(debug) <<" fixup shared memory events...\n";
810 
811  // Update the related events, and at the same time create assertions that can be used to solve for the memory read event's
812  // value. The register write for a shared-memory read instruction has a symbolic value that depends on the shared memory
813  // variable. Therefore, we can set the expression equal to the concrete value stored in this register and solve to get the
814  // value that was read from memory represented by the sharedMemoryEvent.
815  SmtSolver::Transaction tx(ops->solver());
816  std::vector<SymbolicExpression::Ptr> newAssertions;
817  std::vector<ExecutionEvent::Ptr> relatedEvents = getRelatedEvents(sharedMemoryEvent);
818  for (const ExecutionEvent::Ptr &relatedEvent: relatedEvents) {
819  if (relatedEvent->action() == ExecutionEvent::Action::REGISTER_WRITE) {
820  ASSERT_not_null(relatedEvent->expression());
821  ASSERT_require(relatedEvent->value() == nullptr);
822 
823  const RegisterDescriptor REG = relatedEvent->registerDescriptor();
824  relatedEvent->value(SymbolicExpression::makeIntegerConstant(readRegister(REG)));
825 
826  if (!sharedMemoryEvent->value()) {
827  SymbolicExpression::Ptr eq = SymbolicExpression::makeEq(relatedEvent->expression(), relatedEvent->value());
828  SAWYER_MESG(debug) <<" from " <<relatedEvent->printableName(database()) <<":\n";
829  SAWYER_MESG(debug) <<" asserting: (eq[u1] " <<*relatedEvent->expression()
830  <<" " <<*relatedEvent->value() <<")\n";
831  if (debug && eq->getOperator() != SymbolicExpression::OP_EQ)
832  debug <<" simplified: " <<*eq <<"\n";
833  newAssertions.push_back(eq);
834  }
835  }
836  }
837  ops->solver()->insert(newAssertions);
838 
839  // Figure out a concrete value for the shared memory read event based on the assertions we added above.
840  if (!sharedMemoryEvent->value()) {
841  std::string varName = "v" + boost::lexical_cast<std::string>(*sharedMemoryEvent->variable()->variableId());
842  SymbolicExpression::Ptr concreteRead; // value read concretely
843 
844  // First, we try to figure out the value that was concretely read from memory, but we do this in the context of all the
845  // previous assertions also, so it might fail. For instance, if we're reading shared memory that represents a timer,
846  // then there's probably an assertion that says the value is monotonically increasing. But if we've already replayed
847  // events for this memory, then the replayed value might be higher than the one we just read concretely, causing the
848  // assertions to be unsatisfiable.
849  if (debug) {
850  debug <<" all assertions:\n";
851  for (const SymbolicExpression::Ptr &assertion: ops->solver()->assertions())
852  debug <<" asserting: " <<*assertion <<"\n";
853  }
854  SmtSolver::Satisfiable isSatisfied = ops->solver()->check();
855 
856  // If we couldn't figure out a value with all the assertions, then try again with only the assertion for the memory
857  // read instruction: the shared memory event and any register writes.
858  if (SmtSolver::SAT_YES == isSatisfied) {
859  concreteRead = ops->solver()->evidenceForName(varName);
860  } else {
861  SAWYER_MESG(debug) <<" not satisified; trying again with assertions for just this instruction\n";
862  SmtSolver::Ptr extraSolver = ops->solver()->create();
863  extraSolver->insert(newAssertions);
864  isSatisfied = extraSolver->check();
865  if (SmtSolver::SAT_YES == isSatisfied)
866  concreteRead = extraSolver->evidenceForName(varName);
867  }
868 
869  switch (isSatisfied) {
870  case SmtSolver::SAT_NO:
871  SAWYER_MESG(debug) <<" not satisfiable\n";
872  ASSERT_not_implemented("how to recover?"); // [Robb Matzke 2021-09-14]
873  case SmtSolver::SAT_UNKNOWN:
874  SAWYER_MESG(debug) <<" unknown satisfiability (timed out?)\n";
875  ASSERT_not_implemented("how to recover?"); // [Robb Matzke 2021-09-14]
876  case SmtSolver::SAT_YES: {
877  ASSERT_not_null2(concreteRead, "no evidence for " + varName);
878  SAWYER_MESG(debug) <<" presumptive concrete value read from memory: " <<*concreteRead <<"\n";
879  ASSERT_require(concreteRead->isScalarConstant());
880  sharedMemoryEvent->value(concreteRead);
881  }
882  }
883  }
884  ASSERT_not_null(sharedMemoryEvent->value());
885  if (sharedMemoryEvent->variable())
886  inputVariables()->bindVariableValue(sharedMemoryEvent->variable(), sharedMemoryEvent->value());
887 
888 
889  // Make sure the concrete register state is what we think it should be
890  SAWYER_MESG(debug) <<" fixing up concrete state\n";
891  for (const ExecutionEvent::Ptr &relatedEvent: relatedEvents) {
892  if (relatedEvent->action() == ExecutionEvent::Action::REGISTER_WRITE) {
893  SymbolicExpression::Ptr registerValue = relatedEvent->calculateResult(inputVariables()->bindings());
894  ASSERT_require(registerValue->isScalarConstant());
895  SAWYER_MESG(debug) <<" for " <<relatedEvent->printableName(database()) <<"\n"
896  <<" writing " <<*registerValue <<" to register " <<relatedEvent->registerDescriptor() <<"\n";
897  writeRegister(relatedEvent->registerDescriptor(), registerValue->isLeafNode()->bits());
898  }
899  }
900 
901  // If the shared memory event is not a test case input variable, then make the subsequent register updates constant.
902  if (!sharedMemoryEvent->inputVariable()) {
903  for (const ExecutionEvent::Ptr &relatedEvent: relatedEvents) {
904  if (relatedEvent->action() == ExecutionEvent::Action::REGISTER_WRITE) {
905  if (relatedEvent->expression()->isConstant()) {
906  std::cerr <<"ROBB: relatedEvent->expression() = " <<*relatedEvent->expression() <<"\n";
907  } else {
908  relatedEvent->expression(relatedEvent->value());
909  ops->writeRegister(relatedEvent->registerDescriptor(), ops->svalueExpr(relatedEvent->value()));
910  }
911  SAWYER_MESG(debug) <<" " <<relatedEvent->printableName(database()) <<" is constant; "
912  <<"value = "<<*relatedEvent->value() <<"\n";
913  }
914  }
915  }
