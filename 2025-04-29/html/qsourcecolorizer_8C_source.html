106 void QSourceColorizer::process( Q3TextDocument* doc, Q3TextParagraph* parag, int,
107  bool invalidate )
108 {
109  int state = 0;
110  int startLevel = 0;
111  int startState = 0;
112 
113  if( parag->prev() ){
114  if( parag->prev()->endState() == -1 )
115  process( doc, parag->prev(), 0, FALSE );
116  startState = parag->prev()->endState();
117  startLevel = ((ParagData*) parag->prev()->extraData())->level();
118  }
119  state = startState;
120 
121  ParagData* extra = (ParagData*) parag->extraData();
122  if( extra ){
123  extra->clear();
124  } else {
125  extra = new ParagData();
126  parag->setExtraData( extra );
127  }
128 
129  HLItemCollection* ctx = m_items.at( state );
130  QString s = m_editor->text( parag->paragId() );
131  const QChar* buffer = s.unicode();
132  int length = s.length();
133 
134  int pos = 0;
135  while( pos < length ){
136  int attr = 0;
137  int next = state;
138 
139  int npos = ctx->checkHL( buffer, pos, s.length(), &attr, &next );
140  if( npos > pos ){
141  state = next;
142  ctx = m_items.at( state );
143  parag->setFormat( pos, npos, format(attr) );
144  pos = npos;
145  } else {
146  const QChar& ch = buffer[ pos ];
147  int a = ctx->attr();
148  if( !a ){
149  if( m_left.find(ch) != -1 ){
150  extra->add( Symbol::Left, ch, pos );
151  } else if( m_right.find(ch) != -1 ){
152  extra->add( Symbol::Right, ch, pos );
153  }
154  }
155  parag->setFormat( pos, pos+1, format(a) );
156  ++pos;
157  }
158  }
159 
160  int oldState = parag->endState();
161 
162  if( state != oldState ){
163  parag->setEndState( state );
164  }
165 
166  int oldLevel = extra->level();
167  int level = computeLevel( parag, startLevel );
168  if( level != oldLevel ){
169  extra->setLevel( level > 0 ? level : 0 );
170  }
171 
172  parag->setFirstPreProcess( FALSE );
173 
174  Q3TextParagraph *p = parag->next();
175 
176  if( (oldLevel != level
177  || (oldState == -1 && parag->prev() && parag->endState() != parag->prev()->endState())
178  || (oldState != -1 && oldState != state))
179  && invalidate && p && !p->firstPreProcess() && p->endState() != -1 )
180  {
181  //kdDebug(9032) << "invalidate!!! line = " << parag->paragId() << endl;
182  int count = 0;
183  while ( p ) {
184  if ( p->endState() == -1 )
185  return;
186 
187  ++count;
188  p->setEndState( -1 );
189  p = p->next();
190  }
191  //kdDebug(9032) << "invalidated " << count+1 << " lines" << endl;
192  }
193 }
