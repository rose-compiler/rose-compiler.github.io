490 struct Hasher: Visitor {
491  virtual VisitAction preVisit(const Node *node) override {
492  ASSERT_not_null(node);
493  return 0 == node->isHashed() ? CONTINUE : TRUNCATE;
494  }
495 
496  virtual VisitAction postVisit(const Node *node) override {
497  ASSERT_not_null(node);
498  if (!node->isHashed()) { // probably true, but some other thread may have beaten us here.
499  Hash h = hash(hash(node->isMemoryExpr() ? node->domainWidth() : 0, node->nBits()), node->flags());
500  if (const Leaf *leaf = node->isLeafNodeRaw()) {
501  if (leaf->isConstant()) {
502  if (leaf->isIntegerConstant() && leaf->nBits() <= 64) {
503  h = hash(h, leaf->toUnsigned().get());
504  } else {
505  for (size_t i=0; i<leaf->nBits(); i+=64) {
506  size_t j = std::min(i+64, leaf->nBits());
507  h = hash(h, leaf->bits().toInteger(BitRange::hull(i, j-1)));
508  }
509  }
510  } else {
511  // It's okay to hash only the variable ID and not the type because one variable should never be more than
512  // one type.
513  h = hash(h, leaf->nameId());
514  }
515  } else {
516  const Interior *inode = node->isInteriorNodeRaw();
517  ASSERT_not_null(inode);
518  h = hash(h, inode->getOperator());
519  for (const Ptr &child: inode->children()) {
520  ASSERT_require(child->isHashed());
521  h = hash(h, child->hash());
522  }
523  }
524  node->hash(h);
525  }
526  return CONTINUE;
527  }

529  // Incorporates data into the existing hash, h, and returns a new hash. This is no particular well-known algorithm, but
530  // testing showed that it gives pretty well-distributed results for close values, particularly when called on two or more
531  // pieces of data.
532  Hash hash(Hash h, uint64_t data) {
533  ASSERT_require(sizeof(h) == 8);
534  for (size_t i=0; i<64-6; i += 6) {
535  unsigned sa = ((data >> i) ^ h ^ i) & 0x3f;
536  h = (h >> (64-sa)) | (h << sa);
537  h ^= data;
538  }
539  return h;
540  }

761 void
762 Interior::adjustWidth(const Type &type) {
763  if (children_.empty())
764  throw Exception(toStr(op_) + " operator requires argument(s)");
765  switch (op_) {
766  case OP_ASR:
767  case OP_ROL:
768  case OP_ROR:
769  case OP_SHL0:
770  case OP_SHL1:
771  case OP_SHR0:
772  case OP_SHR1: {
773  if (nChildren() != 2)
774  throw Exception(toStr(op_) + " operator expects two arguments");
775  if (!childRaw(0)->isIntegerExpr())
776  throw Exception(toStr(op_) + " operator's first argument (shift amount) must be integer");
777  if (!childRaw(1)->isIntegerExpr())
778  throw Exception(toStr(op_) + " operator's second argument (value to shift) must be integer");
779  type_ = Type::integer(childRaw(1)->nBits());
780  break;
781  }
782  case OP_CONCAT: {
783  // When concatenating, the inputs can be any scalar type but the output will always be integer. E.g, concatenating
784  // two floating-point values results in an integer (i.e., bit vector) since it wouldn't make sense interpretted as
785  // floating-point. If you want a floating-point result you must wrap the concatenation in a reinterpret operation.
786  size_t totalWidth = 0;
787  for (const Ptr &child: children_)
788  totalWidth += child->nBits();
789  type_ = Type::integer(totalWidth);
790  break;
791  }
792  case OP_CONVERT: {
793  if (nChildren() != 1)
794  throw Exception(toStr(op_) + " operator expects one argument");
795  type_ = type;
796  break;
797  }
798  case OP_REINTERPRET: {
799  if (nChildren() != 1)
800  throw Exception(toStr(op_) + " operator expects one argument");
801  if (childRaw(0)->type().nBits() == type.nBits())
802  type_ = type;
803  break;
804  }
805  case OP_EQ:
806  case OP_NE:
807  case OP_SGE:
808  case OP_SGT:
809  case OP_SLE:
810  case OP_SLT:
811  case OP_UGE:
812  case OP_UGT:
813  case OP_ULE:
814  case OP_ULT: {
815  if (nChildren() != 2)
816  throw Exception(toStr(op_) + " operator expects two arguments");
817  if (!childRaw(0)->isIntegerExpr())
818  throw Exception(toStr(op_) + " operator's first argument (shift amount) must be integer");
819  if (!childRaw(1)->isIntegerExpr())
820  throw Exception(toStr(op_) + " operator's second argument (value to shift) must be integer");
821  if (childRaw(0)->nBits() != childRaw(1)->nBits())
822  throw Exception(toStr(op_) + " operator's arguments must both be the same width");
823  type_ = Type::integer(1);
824  break;
825  }
826  case OP_EXTRACT: {
827  if (nChildren() != 3)
828  throw Exception(toStr(op_) + " operator expects three arguments");
829  if (!childRaw(0)->isIntegerConstant())
830  throw Exception(toStr(op_) + " operator's first argument (begin bit) must be an integer constant");
831  if (!childRaw(1)->isIntegerConstant())
832  throw Exception(toStr(op_) + " operator's second argument (end bit) must be an integer constant");
833  if (childRaw(0)->nBits() > 64)
834  throw Exception(toStr(op_) + " operator's first argument (begin bit) must be 64 bits or narrower");
835  if (childRaw(1)->nBits() > 64)
836  throw Exception(toStr(op_) + " operator's second argument (end bit) must be 64 bits or narrower");
837  if (!childRaw(2)->isScalarExpr())
838  throw Exception(toStr(op_) + " operator's third argument must be scalar");
839  if (*childRaw(0)->toUnsigned() >= *childRaw(1)->toUnsigned())
840  throw Exception(toStr(op_) + " operator's first argument must be less than the second");
841  size_t totalSize = *childRaw(1)->toUnsigned() - *childRaw(0)->toUnsigned();
842  // Result will be integer even if arg is floating-point. Example: extracting the bytes of a floating-point value in
843  // order to store it in memory results in integer-valued bytes since it doesn't make sense to interpret the bytes
844  // as floating-point values.
845  type_ = Type::integer(totalSize);
846  break;
847  }
848  case OP_ITE: {
849  if (nChildren() != 3)
850  throw Exception(toStr(op_) + " operator expects three arguments");
851  if (!childRaw(0)->isIntegerExpr())
852  throw Exception(toStr(op_) + " operator's first argument (condition) must be integer");
853  if (childRaw(0)->nBits() != 1)
854  throw Exception(toStr(op_) + " operator's first argument (condition) must be Boolean");
855  if (childRaw(1)->type() != childRaw(2)->type())
856  throw Exception(toStr(op_) + " operator's second and third arguments must have the same type");
857  type_ = childRaw(1)->type();
858  break;
859  }
860  case OP_LET: {
861  if (nChildren() != 3)
862  throw Exception(toStr(op_) + " operator expects three arguments");
863  type_ = childRaw(2)->type();
864  break;
865  }
866  case OP_LSSB:
867  case OP_MSSB:
868  case OP_NEGATE: {
869  if (nChildren() != 1)
870  throw Exception(toStr(op_) + " operator expects one argument");
871  if (!childRaw(0)->isIntegerExpr())
872  throw Exception(toStr(op_) + " argument must be integer");
873  type_ = childRaw(0)->type();
874  break;
875  }
876  case OP_READ: {
877  if (nChildren() != 2)
878  throw Exception(toStr(op_) + " operator expects two arguments");
879  if (!childRaw(0)->isMemoryExpr())
880  throw Exception(toStr(op_) + " operator's first argument must be a memory state");
881  if (!childRaw(1)->isIntegerExpr())
882  throw Exception(toStr(op_) + " operator's second argument (address) must be integer");
883  if (childRaw(0)->domainWidth() != childRaw(1)->nBits())
884  throw Exception(toStr(op_) + " operator's arguments have mismatched sizes");
885  type_ = Type::integer(childRaw(0)->nBits()); // size of values stored in memory
886  break;
887  }
888  case OP_SDIV:
889  case OP_UDIV: {
890  if (nChildren() != 2)
891  throw Exception(toStr(op_) + " operator expects two arguments");
892  if (!childRaw(0)->isIntegerExpr())
893  throw Exception(toStr(op_) + " operator's first argument must be integer");
894  if (!childRaw(1)->isIntegerExpr())
895  throw Exception(toStr(op_) + " operator's second argument must be integer");
896  type_ = childRaw(0)->type();
897  break;
898  }
899  case OP_SEXTEND: {
900  if (nChildren() != 2)
901  throw Exception(toStr(op_) + " operator expects two arguments");
902  if (!childRaw(0)->isIntegerConstant())
903  throw Exception(toStr(op_) + " operator's first argument (size) must be an integer constant");
904  if (childRaw(0)->nBits() > 64)
905  throw Exception(toStr(op_) + " operator's first argument (size) must be 64 bits or narrower");
906  if (!childRaw(1)->isIntegerExpr())
907  throw Exception(toStr(op_) + " operator's second argument must be integer");
908  type_ = Type::integer(childRaw(0)->toUnsigned().get());
909  break;
910  }
911  case OP_UEXTEND: {
912  // This operator can make the argument wider or narrower. The argument can be any scalar type and the result will
913  // always be an integer (bit vector) type. For instance, narrowing a floating-point value results in an integer
914  // bit vector type since it wouldn't make sense anymore as a floating point type.
915  if (nChildren() != 2)
916  throw Exception(toStr(op_) + " operator expects two arguments");
917  if (!childRaw(0)->isIntegerConstant())
918  throw Exception(toStr(op_) + " operator's first argument (size) must be an integer constant");
919  if (childRaw(0)->nBits() > 64)
920  throw Exception(toStr(op_) + " operator's first argument (size) must be 64 bits or narrower");
921  if (!childRaw(1)->isScalarExpr())
922  throw Exception(toStr(op_) + " operator's second argument must be scalar");
923  type_ = Type::integer(childRaw(0)->toUnsigned().get());
924  break;
925  }
926  case OP_SMOD:
927  case OP_UMOD: {
928  if (nChildren() != 2)
929  throw Exception(toStr(op_) + " operator expects two arguments");
930  if (!childRaw(0)->isIntegerExpr())
931  throw Exception(toStr(op_) + " operator's first argument must be integer");
932  if (!childRaw(1)->isIntegerExpr())
933  throw Exception(toStr(op_) + " operator's second argument must be integer");
934  type_ = childRaw(1)->type();
935  break;
936  }
937  case OP_SMUL:
938  case OP_UMUL: {
939  if (nChildren() < 1)
940  throw Exception(toStr(op_) + " operator expects at least one argument");
941  size_t totalWidth = 0;
942  for (size_t i=0; i<nChildren(); ++i) {
943  if (!childRaw(i)->isIntegerExpr())
944  throw Exception(toStr(op_) + " operator's arguments must all be integer");
945  totalWidth += childRaw(i)->nBits();
946  }
947  type_ = Type::integer(totalWidth);
948  break;
949  }
950  case OP_WRITE: {
951  if (nChildren() != 3)
952  throw Exception(toStr(op_) + " operator expects three arguments");
953  if (!childRaw(0)->isMemoryExpr())
954  throw Exception(toStr(op_) + " operator's first operand must be a memory state");
955  if (!childRaw(1)->isIntegerExpr())
956  throw Exception(toStr(op_) + " operator's second argument (address) must be integer");
957  if (!childRaw(2)->isIntegerExpr())
958  throw Exception(toStr(op_) + " operator's third argument (value to write) must be integer");
959  if (childRaw(1)->nBits() != childRaw(0)->domainWidth())
960  throw Exception(toStr(op_) + " operator's second argument (address) has incorrect width");
961  if (childRaw(2)->nBits() != childRaw(0)->nBits())
962  throw Exception(toStr(op_) + " operator's third argument (value to write) has incorrect width");
963  type_ = childRaw(0)->type();
964  break;
965  }
966  case OP_ZEROP: {
967  if (nChildren() != 1)
968  throw Exception(toStr(op_) + " operator expects one argument");
969  if (!childRaw(0)->isIntegerExpr())
970  throw Exception(toStr(op_) + " operator's argument must be integer");
971  type_ = Type::integer(1);
972  break;
973  }
974  case OP_FP_LE:
975  case OP_FP_LT:
976  case OP_FP_GE:
977  case OP_FP_GT:
978  case OP_FP_EQ: {
979  if (nChildren() != 2)
980  throw Exception(toStr(op_) + " operator expects two arguments");
981  if (!childRaw(0)->isFloatingPointExpr())
982  throw Exception(toStr(op_) + " operator's first argument must be floating-point");
983  if (!childRaw(1)->isFloatingPointExpr())
984  throw Exception(toStr(op_) + " operator's second argument must be floating-point");
985  type_ = Type::integer(1);
986  break;
987  }
988  case OP_FP_ISNORM:
989  case OP_FP_ISSUBNORM:
990  case OP_FP_ISZERO:
991  case OP_FP_ISINFINITE:
992  case OP_FP_ISNAN:
993  case OP_FP_ISNEG:
994  case OP_FP_ISPOS: {
995  if (nChildren() != 1)
996  throw Exception(toStr(op_) + " operator expects one argument");
997  if (!childRaw(0)->isFloatingPointExpr())
998  throw Exception(toStr(op_) + " operator's argument must be floating-point");
999  type_ = Type::integer(1);
1000  break;
1001  }
1002  default: {
1003  // All children must have the same type, which is the type of this expression. This is suitable for things like
1004  // bitwise operators, add, etc.
1005  ASSERT_require(nChildren() > 0);
1006  for (size_t i = 1; i < nChildren(); ++i) {
1007  if (childRaw(i)->type() != childRaw(0)->type())
1008  throw Exception(toStr(op_) + " operator's arguments must all be the same type");
1009  }
1010  type_ = childRaw(0)->type();
1011  break;
1012  }
1013  }
1014  ASSERT_require(nBits() != 0);
1015 }

1142 bool
1143 Interior::mayEqual(const Ptr &other, const SmtSolver::Ptr &solver/*NULL*/) {
1144  // Fast comparison of literally the same expression pointer
1145  if (this == getRawPointer(other))
1146  return true;
1147 
1148  // Give the user a chance to decide.
1149  if (mayEqualCallback) {
1150  boost::logic::tribool result = (mayEqualCallback)(sharedFromThis(), other, solver);
1151  if (true == result || false == result)
1152  return result ? true : false;
1153  }
1154 
1155  // Two addition operations of the form V + C1 and V + C2 where V is a variable and C1 and C2 are constants, are equal if
1156  // and only if C1 = C2.
1157  LeafPtr variableA, variableB, constantA, constantB;
1158  if (matchAddVariableConstant(variableA/*out*/, constantA/*out*/)) {
1159  if (other->matchAddVariableConstant(variableB/*out*/, constantB/*out*/)) {
1160  // Comparing V + C1 with V + C2; return true iff C1 == C2
1161  if (variableA->nameId() == variableB->nameId()) {
1162  ASSERT_require(variableA->nBits() == variableB->nBits());
1163  ASSERT_require(constantA->nBits() == constantB->nBits());
1164  return constantA->bits().compare(constantB->bits()) == 0;
1165  }
1166  } else if ((variableB = other->isLeafNode()) && variableB->isIntegerVariable()) {
1167  // Comparing V + C with V; return true iff C == 0 (which it shouldn't or else the additive identity rule would have
1168  // already kicked in and removed it.
1169  if (variableA->nameId() == variableB->nameId()) {
1170  ASSERT_require(variableA->nBits() == variableB->nBits());
1171  ASSERT_require(constantA->nBits() == variableA->nBits());
1172  ASSERT_forbid2(constantA->bits().isEqualToZero(), "additive identity should have been simplified");
1173  return false;
1174  }
1175  }
1176  }
1177 
1178  // Two expressions that are structurally equivalent are also equal.
1179  if (isEquivalentTo(other))
1180  return true;
1181 
1182  // It is difficult to prove that arbitrary expressions are equal or not equal. Ask the solver to find a solution where
1183  // they are equal. If the solver finds a solution, then obviously they can be equal. If the solver can't prove the
1184  // equality assertion, we want to assume that they can be equal. Thus only if the solver can prove that they cannot be
1185  // equal do we have anything to return here.
1186  if (solver) {
1187  SmtSolver::Transaction transaction(solver);
1188  Ptr assertion = makeEq(sharedFromThis(), other, solver);
1189  solver->insert(assertion);
1190  if (SmtSolver::SAT_NO == solver->check())
1191  return false;
1192  }
1193 
1194  // If we couldn't prove that the two expressions are unequal, then assume they could be equal.
1195  return true;
1196 }

1230 bool
1231 Interior::isEquivalentHelper(Node *other_, EquivPairs &matched) {
1232  if (!other_)
1233  return false;
1234  const Interior *other = other_->isInteriorNodeRaw();
1235  if (this == other) {
1236  return true;
1237  } else if (!other || type() != other->type() || flags() != other->flags()) {
1238  return false;
1239  } else if (hashval_ != 0 && other->hashval_ != 0 && hashval_ != other->hashval_) {
1240  // Unequal hashvals imply non-equivalent expressions. The converse is not necessarily true due to possible
1241  // collisions.
1242  return false;
1243  } else if (op_ != other->op_ || children_.size() != other->children_.size()) {
1244  return false;
1245  } else {
1246  // Did we test this pair of nodes already?
1247  auto node = matched.find(this);
1248  if (node != matched.end()) {
1249  for (const auto &pair: node->second) {
1250  if (pair.first == other)
1251  return pair.second; // previously tested pair of node pointers
1252  }
1253  }
1254 
1255  // Test equivalence
1256  bool retval = true;
1257  for (size_t i = 0; retval && i < children_.size(); ++i)
1258  retval = children_[i]->isEquivalentHelper(other->children_[i].getRawPointer(), matched);
1259 
1260  // Cache hash values. There's no need to compute a hash value if we've determined that the two expressions are
1261  // equivalent because it wouldn't save us any work--two equal hash values doesn't necessarily mean that two expressions
1262  // are equivalent. However, if we already know one of the hash values then we can cache that hash value in the other
1263  // expression too.
1264  if (retval) {
1265  if (hashval_ != 0 && other->hashval_ == 0) {
1266  other->hashval_ = hashval_;
1267  } else if (hashval_ == 0 && other->hashval_ != 0) {
1268  hashval_ = other->hashval_;
1269  } else {
1270  ASSERT_require(hashval_ == other->hashval_);
1271  }
1272  }
1273 
1274  // Remember equivalence results for this pair of nodes.
1275  matched.insert(std::make_pair(this, EquivPairs::value_type::second_type{{other_, retval}}));
1276  return retval;
1277  }
1278 }

1305 Interior::associative() {
1306  Nodes newOperands;
1307  std::list<Ptr> worklist(children_.begin(), children_.end());
1308  bool modified = false;

1637  // A and B are duals if they have one of the following forms:
1638  // (1) A = x AND B = (negate x)
1639  // (2) A = x AND B = (invert x) [adjust constant]
1640  // (3) A = (negate x) AND B = x
1641  // (4) A = (invert x) AND B = x [adjust constant]
1642  //
1643  // This makes use of the relationship:
1644  // (add (negate x) -1) == (invert x)
1645  // by decrementing adjustment. The adjustment, whose width is the same as A and B, is allowed to overflow. For example,
1646  // consider the expression, where all values are two bits wide:
1647  // (add v1 (invert v1) v2 (invert v2) v3 (invert v3)) by substitution for invert:
1648  // (add v1 (negate v1) -1 v2 (negate v2) -1 v3 (negate v3) -1) canceling duals gives:
1649  // (add -1 -1 -1) rewriting as 2's complement (2 bits wide):
1650  // (add 3 3 3) constant folding modulo 4:
1651  // 1
1652  // compare with v1=0, v2=1, v3=2 (i.e., -2 in two's complement):
1653  // (add 0 3 1 2 2 1) == 1 mod 4
1654  struct are_duals {
1655  bool operator()(Ptr a, Ptr b, Sawyer::Container::BitVector &adjustment/*in,out*/) {
1656  ASSERT_not_null(a);
1657  ASSERT_not_null(b);
1658  ASSERT_require(a->nBits() == b->nBits());
1659 
1660  // swap A and B if necessary so we have form (1) or (2).
1661  if (b->isInteriorNodeRaw() == nullptr)
1662  std::swap(a, b);
1663  if (b->isInteriorNodeRaw() == nullptr)
1664  return false;
1665 
1666  const Interior *bi = b->isInteriorNodeRaw();
1667  if (bi->getOperator() == OP_NEGATE) {
1668  // form (3)
1669  ASSERT_require(1 == bi->nChildren());
1670  return a->isEquivalentTo(bi->child(0));
1671  } else if (bi->getOperator() == OP_INVERT) {
1672  // form (4) and ninverts is small enough
1673  if (a->isEquivalentTo(bi->child(0))) {
1674  adjustment.decrement();
1675  return true;
1676  }
1677  }
1678  return false;
1679  }

2619 Ptr
2620 ShiftSimplifier::combine_strengths(Ptr strength1, Ptr strength2, size_t value_width, const SmtSolver::Ptr &solver) const {
2621  if (!strength1 || !strength2)
2622  return Ptr();
2623 
2624  // Calculate the width for the sum of the strengths. If the width of the value being shifted isn't a power of two then we
2625  // need to avoid overflow in the sum, otherwise overflow doesn't matter. The sum should be wide enough to hold a shift
2626  // amount that's the same as the width of the value, otherwise we wouldn't be able to distinguish between the case where
2627  // modulo addition produced a shift amount that's large enough to decimate the value, as opposed to a shift count of zero
2628  // which is a no-op.
2629  size_t sum_width = std::max(strength1->nBits(), strength2->nBits());
2630  if (IntegerOps::isPowerOfTwo(value_width)) {
2631  sum_width = std::max(sum_width, IntegerOps::log2max(value_width)+1);
2632  } else {
2633  sum_width = std::max(sum_width+1, IntegerOps::log2max(value_width)+1);
2634  }
2635  if (sum_width > 64)
2636  return Ptr();
2637 
2638  // Zero-extend the strengths if they're not as wide as the sum. This is because the ADD operator requires that its
2639  // operands are the same width, and the result will also be that width.
2640  if (strength1->nBits() < sum_width)
2641  strength1 = makeExtend(makeIntegerConstant(32, sum_width), strength1, solver);
2642  if (strength2->nBits() < sum_width)
2643  strength2 = makeExtend(makeIntegerConstant(32, sum_width), strength2, solver);
2644 
2645  return makeAdd(strength1, strength2, solver);
2646 }

2654  Sawyer::Optional<uint64_t> sa;
2655  if (sa_leaf && sa_leaf->isIntegerConstant()) {
2656  if (sa_leaf->nBits() <= 64) {
2657  sa = sa_leaf->toUnsigned();
2658  } else {
2659  Sawyer::Optional<size_t> msbSetIdx = sa_leaf->bits().mostSignificantSetBit();
2660  if (!msbSetIdx) {
2661  sa = 0;
2662  } else if (*msbSetIdx < 64) {
2663  sa = sa_leaf->bits().toInteger(); // truncate to 64 bits
2664  } else {
2665  sa = inode->nBits();
2666  }
2667  }
2668  }
2669  if (sa)
2670  sa = std::min((uint64_t)inode->nBits(), *sa);

2673  if (val_leaf && val_leaf->isIntegerConstant() && sa) {
2674  Sawyer::Container::BitVector result = val_leaf->bits();
2675  result.shiftLeft(*sa, newbits);
2676  return makeIntegerConstant(result, inode->comment(), inode->flags());
2677  }
2678 
2679  // If the shifted operand is itself a shift of the same kind, then simplify by combining the strengths:
2680  // (shl AMT1 (shl AMT2 X)) ==> (shl (add AMT1 AMT2) X)
2681  const Interior *val_inode = inode->childRaw(1)->isInteriorNodeRaw();
2682  if (val_inode && val_inode->getOperator()==inode->getOperator()) {
2683  if (Ptr strength = combine_strengths(inode->child(0), val_inode->child(0), inode->childRaw(1)->nBits(), solver)) {

2690  Sawyer::Container::BitVector result(inode->nBits(), newbits);
2691  return makeIntegerConstant(result, inode->comment());
2692  }
2693

2695  if (sa && 0==*sa)
2696  return inode->child(1);
2697

2699  // (shl0[N] AMT X) ==> (concat (extract 0 N-AMT X)<hiBits> 0[AMT]<loBits>)
2700  // (shl1[N] AMT X) ==> (concat (extract 0 N-AMT X)<hiBits> -1[AMT]<loBits>)

2702  ASSERT_require(*sa > 0 && *sa < inode->nBits());// handled above
2703  size_t nHiBits = inode->nBits() - *sa;
2704  Ptr hiBits = makeExtract(makeIntegerConstant(32, 0), makeIntegerConstant(32, nHiBits), inode->child(1), solver);
2705  Ptr loBits = makeIntegerConstant(*sa, newbits?uint64_t(-1):uint64_t(0));

2707  }
