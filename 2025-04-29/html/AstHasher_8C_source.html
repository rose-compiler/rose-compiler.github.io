51 AstHasher::hash(SgNode *ast) {
52  if (ast) {
53  ASSERT_not_null(hasher_);
54  AST::Traversal::forwardPre<SgNode>(ast, [this](SgNode *node) {
55  // Always include the type of each node in the hash. We include the type name rather than the enum constant because we
56  // want the hash to be as stable as possible across different ROSE versions. The type name never changes, but the enum
57  // constant can change whenever someone adds a new SgNode type even if that type is completely unrelated to binary
58  // analysis.
59  hasher_->insert(node->class_name());
60 
61  if (auto asmInstruction = isSgAsmInstruction(node)) {
62  // Include the mnemonic, and maybe the address.
63  hasher_->insert(asmInstruction->get_mnemonic());
64  if (hashingAddresses_)
65  hasher_->insert(asmInstruction->get_address());
66 
67  } else if (auto regRef = isSgAsmRegisterReferenceExpression(node)) {
68  // Always include register references

72  // Maybe include constants (integers, floats, pointers)
73  if (hashingConstants_)
74  hasher_->insert(constExpr->get_bitVector().toHex());
