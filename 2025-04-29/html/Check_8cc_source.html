66 isHiddenType (const SgType* type)
67 {
68  if (!type) return false;
69  const SgType* base_type = type->findBaseType ();
70  if (base_type)
71  {
72  const SgNamedType* n = isSgNamedType (base_type);
73  if (n)
74  { // Get the declaration of a named type
75  const SgDeclarationStatement* decl = n->get_declaration ();
76  if (decl)
77  {
78  const SgScopeStatement* scope = decl->get_scope ();
79  while (!isSgGlobal (scope))
80  {
81  // Non-global scoped declaration, not within a class or name space,
82  // a hidden type!
83  if (!isSgClassDefinition (scope)
84  && !isSgNamespaceDefinitionStatement (scope))
85  return true;
86  scope = scope->get_scope ();
87  }
88  }
89  }
90  }
91  return false;
92 }
