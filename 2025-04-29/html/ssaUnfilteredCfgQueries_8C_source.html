17 SgExpression* SSA_UnfilteredCfg::buildVariableReference(const VarName& var, SgScopeStatement* scope)
18 {
19  ROSE_ASSERT(var.size() > 0);
20 
21  SgExpression* varsSoFar = SageBuilder::buildVarRefExp(var.front(), scope);
22 
23  for (size_t i = 0; i < var.size(); i++)
24  {
25  SgInitializedName* initName = var[i];
26  if (initName == var.back())
27  {
28  break;
29  }
30 
31  SgVarRefExp* nextVar = SageBuilder::buildVarRefExp(var[i + 1], scope);
32 
33  if (SageInterface::isPointerType(initName->get_type()))
34  {
35  varsSoFar = SageBuilder::buildArrowExp(varsSoFar, nextVar);
36  }
37  else
38  {
39  varsSoFar = SageBuilder::buildDotExp(varsSoFar, nextVar);
40  }
41  }
42 
43  return varsSoFar;
44 }
