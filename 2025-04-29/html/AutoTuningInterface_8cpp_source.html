655 {
656  const std::vector<LocalVar*>& tophandles=handleMap.GetTopTraceHandles();
657  LocalVar* top = handleMap.GenTraceCommand(*poet,tophandles, target,lineNo);
658  ++lineNo;
659 
660  /* insert xform decls*/
661  int lineold = lineNo;
662  POETCode* traceMod = 0;
663  for (std::vector<OptSpec*>::const_iterator
664  p = optvec.begin(); p != optvec.end(); ++p) {
665  OptSpec* curopt = *p;
666  curopt->insert_xformDecl(*poet,top,traceMod,lineNo);
667  }
668  if (lineNo > lineold) ++lineNo;
669 
670  /* insert xform parameter declarations */
671  int applyOpt = configure_optimization();
672  POETCode* cond = 0;
673  for (int i = 1; i <= OptSpec::OPT_LEVEL_MAX; i <<= 1) {
674  OptSpec::OptLevel curLevel = (OptSpec::OptLevel)i;
675  if (!(curLevel & applyOpt)) continue;
676  for (std::vector<OptSpec*>::const_iterator
677  p = optvec.begin(); p != optvec.end(); ++p) {
678  OptSpec* curopt = *p;
679  if (curopt->get_opt_level() & curLevel)

682  if (cur_cond != 0)
683  cond = (cond == 0)? cur_cond : poet->make_Bop(POET_OP_AND,cond,cur_cond);
684  }
685  }
686  }
687  ++lineNo;
688  if (cond != 0) {
689  poet->insert_condDecl(cond, lineNo++);
690  ++lineNo;
691  }
692 
693  /* gen xform invocations*/
694  std::vector<LocalVar*> delay_opts;
695  for (int i = 1; i <= OptSpec::OPT_LEVEL_MAX; i <<= 1) {
696  OptSpec::OptLevel curLevel = (OptSpec::OptLevel)i;
697  if (!(curLevel & applyOpt)) continue;
698  for (std::vector<OptSpec*>::const_iterator
699  p = optvec.begin(); p != optvec.end(); ++p) {
700  OptSpec* curopt = *p;
701  if (curopt->get_opt_level() & curLevel) {

703  POETCode* cureval=curopt->gen_xformEval(*poet,top,traceMod,curLevel, lineNo);
704  if (cureval != 0) {
705  LocalVar* optvar = EVAL_VAR(*poet, curopt->get_opt_prefix(curLevel)+"_"+curopt->get_targetName());
706  delay_opts.push_back(optvar);
707  MAIN_EVAL(*poet, ASSIGN(optvar,DELAY(cureval)), linenum);
708  lineNo+=2;
709  }
710  }
711  }
712  }
713  ++lineNo;
714 
715  /*QY: apply delayed optimiations*/
716  std::vector<LocalVar*>::const_iterator p_opt = delay_opts.begin();
717  POETCode* res = APPLY(*p_opt);
718  for ( ++p_opt; p_opt != delay_opts.end(); ++p_opt) {
719  res = SEQ(*poet,res, APPLY(*p_opt));
720  }
721  MAIN_EVAL(*poet, res, ++lineNo);
722 }
