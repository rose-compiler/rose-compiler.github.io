73  // Build the Capstone context object, which must be explicitly closed in the destructor. Furthermore, since capstone is a
74  // shared library, it might not be possible to call cs_close (in the Aarch32 destructor) after libcapstone has been shut
75  // down during program termination. For instance, if you store any static pointers to Aarch32 objects that have open
76  // capstone connections, then its likely that the C++ runtime will close the capstone library before calling the
77  // destructors for those Aarch32 objects.
78 
79  // Warning: the "mode" constants are not orthogonal with each other or the "arch" values.
80  cs_mode mode = (cs_mode)modes_.vector();
81 
82  if (CS_ERR_OK != cs_open(CS_ARCH_ARM64, mode, &capstone_))
83  throw Exception("capstone cs_open failed");
84  capstoneOpened_ = true;
85  if (CS_ERR_OK != cs_option(capstone_, CS_OPT_DETAIL, CS_OPT_ON))
86  throw Exception("capstone cs_option failed");
87 }

97 Aarch64::disassembleOne(const MemoryMap::Ptr &map, Address va, AddressSet *successors/*=nullptr*/) {
98  openCapstone();
99 
100  // Resources that must be explicitly reclaimed before returning.
101  struct Resources {
102  cs_insn *csi = nullptr;
103  size_t nInsns = 0;
104  ~Resources() {
105  if (csi)
106  cs_free(csi, nInsns);
107  }
108  } r;
109 
110  // Read the encoded instruction bytes into a temporary buffer to be used by capstone
111  if (va % instructionAlignment_ != 0)
112  throw Exception("instruction pointer not aligned", va);
113  uint8_t bytes[4]; // largest possible instruction is 4 bytes
114  size_t nRead = map->at(va).limit(sizeof bytes).require(MemoryMap::EXECUTABLE).read(bytes).size();
115  if (0 == nRead)
116  throw Exception("short read", va);
117 
118  // Disassemble the instruction with capstone
119  r.nInsns = cs_disasm(capstone_, bytes, nRead, va, 1, &r.csi);
120  if (0 == r.nInsns)
121  return makeUnknownInstruction(Exception("unable to decode instruction", va, SgUnsignedCharList(bytes+0, bytes+nRead), 0));
122 
123  ASSERT_require(1 == r.nInsns);
124  ASSERT_not_null(r.csi);
125  ASSERT_require(r.csi->address == va);
126  ASSERT_not_null(r.csi->detail);
127 
128  // Convert disassembled capstone instruction to ROSE AST
129  SgAsmInstruction *retval = nullptr;
130  const cs_arm64 &detail = r.csi->detail->arm64;
131 #if 0 // DEBGUGGING: show the disassembly string from capstone itself
132  std::cerr <<"ROBB: capstone disassembly:"
133  <<" " <<StringUtility::addrToString(va) <<":"
134  <<" " <<StringUtility::toHex2(bytes[0], 8, false, false).substr(2)
135  <<" " <<StringUtility::toHex2(bytes[1], 8, false, false).substr(2)
136  <<" " <<StringUtility::toHex2(bytes[2], 8, false, false).substr(2)
137  <<" " <<StringUtility::toHex2(bytes[3], 8, false, false).substr(2)
138  <<" " <<r.csi->mnemonic <<" " <<r.csi->op_str <<"\n";
139 #endif
140  auto operands = new SgAsmOperandList;
141  for (uint8_t i = 0; i < detail.op_count; ++i) {
142  SgAsmExpression *operand = makeOperand(*r.csi, detail.operands[i]);
143  ASSERT_not_null(operand);
144  ASSERT_not_null(operand->get_type());
145  operands->get_operands().push_back(operand);
146  operand->set_parent(operands);
147  }
148  wrapPrePostIncrement(operands, detail);
149  auto insn = new SgAsmAarch64Instruction(va, *architecture()->registrationId(), (Aarch64InstructionKind)r.csi->id, detail.cc);
150  insn->set_rawBytes(SgUnsignedCharList(r.csi->bytes, r.csi->bytes + r.csi->size));
151  insn->set_operandList(operands);
152  insn->set_updatesFlags(detail.update_flags);
153  operands->set_parent(insn);
154  retval = insn;
155 
156  // Work around capstone bugs
157  if (BitOps::bits(opcode(insn), 24, 31) == 0b10011000) {
158  // LDRSW (literal) is decoded incorrectly. The second operand should be a 32-bit memory read, not an immediate value.
159  if (auto ival = isSgAsmIntegerValueExpression(insn->get_operandList()->get_operands()[1])) {
160  // Remove ival from the AST
161  insn->get_operandList()->get_operands()[1] = nullptr;
162  ival->set_parent(nullptr);
163 
164  // Create a memory reference with ival as the address
165  auto memref = new SgAsmMemoryReferenceExpression;
166  memref->set_type(SageBuilderAsm::buildTypeU32());
167  memref->set_address(ival);
168  ival->set_parent(memref);
169 
170  // Insert the memref into the AST
171  insn->get_operandList()->get_operands()[1] = memref;
172  memref->set_parent(insn->get_operandList());
173  }
174  } else if (insn->get_kind() == Aarch64InstructionKind::ARM64_INS_XTN) {
175  // The XTN instruction is disassembled incorrectly: the type of the destination argument is wrong. XTN reads the
176  // elements of the source vector, truncates each to half its width, packs the half-sizes together, and writes them to
177  // the low half of the destination register while clearing the upper half of the destination. I.e., it writes as many
178  // bits as it reads. However, the destination type is only half as wide as the source type. In contrast, the XTN2
179  // instruction which is the same except it writes the result to the upper half of the destination while clearing the
180  // lower half, has the correct type.
181  ASSERT_require(insn->nOperands() == 2);
182  ASSERT_require(insn->operand(0)->get_nBits() * 2 == insn->operand(1)->get_nBits());
183  SgAsmVectorType *dstType = isSgAsmVectorType(insn->operand(0)->get_type());
184  ASSERT_not_null(dstType);
185  insn->operand(0)->set_type(SageBuilderAsm::buildTypeVector(2 * dstType->get_nElmts(), dstType->get_elmtType()));
186  }
187  ASSERT_not_null(retval);
188 
189  // Note successors if necessary
190  if (successors) {
191  bool complete = false;
192  *successors |= architecture()->getSuccessors(retval, complete/*out*/);
193  }
194 
195  return retval;
196 }

209 SgAsmExpression*
210 Aarch64::makeOperand(const cs_insn &insn, const cs_arm64_op &op) {
211  SgAsmExpression *retval = nullptr;
212 
213  switch (op.type) {
214  case ARM64_OP_INVALID:
215  ASSERT_not_reachable("invalid operand type from Capstone");
216 
217  case ARM64_OP_REG: { // register operand
218  RegisterDescriptor reg = makeRegister(op.reg);
219  reg = subRegister(reg, op.vector_index, op.vess);
220  retval = new SgAsmDirectRegisterExpression(reg);
221  retval->set_type(registerType(reg, op.vas));
222  //retval = extractElement(retval, op.vess, op.vector_index);
223  retval = extendOperand(retval, insn, op.ext, retval->get_type(), op.shift.type, op.shift.value);
224  break;
225  }
226 
227  case ARM64_OP_IMM: // immediate operand
228  // Capstone doesn't have any types. It just stores all immediate values in a 64 bit field and assumes that the
229  // instruction semantics and unparser will only use that part of the field that is applicable for this instruction
230  // and argument type.
231  retval = new SgAsmIntegerValueExpression(op.imm, SageBuilderAsm::buildTypeU64());
232  retval = extendOperand(retval, insn, op.ext, retval->get_type(), op.shift.type, op.shift.value);
233  break;
234 
235  case ARM64_OP_MEM: { // memory operand
236  RegisterDescriptor reg = makeRegister(op.mem.base);
237  auto base = new SgAsmDirectRegisterExpression(reg);
238  base->set_type(registerType(reg, ARM64_VAS_INVALID));
239  SgAsmExpression *addr = base;
240  SgAsmType *u64 = SageBuilderAsm::buildTypeU64();
241 
242  if (op.mem.index != ARM64_REG_INVALID) {
243  RegisterDescriptor indexReg = makeRegister(op.mem.index);
244  auto index = new SgAsmDirectRegisterExpression(indexReg);
245  index->set_type(registerType(indexReg, ARM64_VAS_INVALID));
246 
247  if (op.mem.disp != 0) {
248  auto disp = new SgAsmIntegerValueExpression(op.mem.disp, u64);
249  switch (op.shift.type) {
250  case ARM64_SFT_INVALID:
251  addr = SageBuilderAsm::buildAddExpression(addr, SageBuilderAsm::buildAddExpression(index, disp, u64), u64);
252  break;
253  case ARM64_SFT_LSL:
254  addr = SageBuilderAsm::buildLslExpression(addr, disp, u64);
255  break;
256  case ARM64_SFT_MSL: // Same as LSL but filling low bits with ones
257  addr = SageBuilderAsm::buildMslExpression(addr, disp, u64);
258  break;
259  case ARM64_SFT_LSR:
260  addr = SageBuilderAsm::buildLsrExpression(addr, disp, u64);
261  break;
262  case ARM64_SFT_ASR:
263  addr = SageBuilderAsm::buildAsrExpression(addr, disp, u64);
264  break;
265  case ARM64_SFT_ROR:
266  addr = SageBuilderAsm::buildRorExpression(addr, disp, u64);
267  break;
268  }
269  } else {
270  addr = SageBuilderAsm::buildAddExpression(addr, index, u64);
271  }
272  } else if (op.mem.disp != 0) {
273  auto disp = new SgAsmIntegerValueExpression(op.mem.disp, u64);
274  addr = SageBuilderAsm::buildAddExpression(addr, disp, u64);
275  }
276  addr = extendOperand(addr, insn, op.ext, addr->get_type(), op.shift.type, op.shift.value);
277 
278  auto mre = new SgAsmMemoryReferenceExpression;
279  mre->set_address(addr);
280  mre->set_type(typeForMemoryRead(insn));
281  retval = mre;
282  break;
283  }
284 
285  case ARM64_OP_FP: // floating-point operand
286  retval = SageBuilderAsm::buildValueFloat(op.fp, SageBuilderAsm::buildIeee754Binary64());
287  break;
288 
289  case ARM64_OP_CIMM: // C-immediate
290  retval = new SgAsmAarch64CImmediateOperand(op.sys);
291  retval->set_type(SageBuilderAsm::buildTypeU(1)); // FIXME: not sure what the type should be, but probably not this
292  break;
293 
294  case ARM64_OP_REG_MRS: // MRS register operand
295  case ARM64_OP_REG_MSR: { // MSR register operand
296 #if 1
297  // Capstone appears to decode this incorrectly, so we do it according to the ARM documentation instead.
298  uint32_t reg = BitOps::bits(opcode(insn), 5, 19);
299 #else
300  uint32_t reg = op.reg;
301 #endif
302  retval = new SgAsmAarch64SysMoveOperand(reg);
303  retval->set_type(SageBuilderAsm::buildTypeU(1)); // FIXME: not sure what the type should be, but probably not this
304  break;
305  }
306 
307  case ARM64_OP_PSTATE: // PState operand
308  retval = new SgAsmAarch64PState(op.pstate);
309  retval->set_type(SageBuilderAsm::buildTypeU(1)); // FIXME: not sure what the type should be, but probably not this
310  break;
311 
312  case ARM64_OP_SYS: // SYS operand for IC/DC/AT/TLBI
313  retval = new SgAsmAarch64AtOperand((Aarch64AtOperation)op.sys);
314  retval->set_type(SageBuilderAsm::buildTypeU(1)); // FIXME: not sure what the type should be, but probably not this
315  break;
316 
317  case ARM64_OP_PREFETCH: // prefetch operand
318  retval = new SgAsmAarch64PrefetchOperand(op.prefetch);
319  retval->set_type(SageBuilderAsm::buildTypeU(1)); // FIXME: not sure what the type should be, but probably not this
320  break;
321 
322  case ARM64_OP_BARRIER: // memory barrier operand for ISB/DMB/DSB instruction
323  retval = new SgAsmAarch64BarrierOperand(op.barrier);
324  retval->set_type(SageBuilderAsm::buildTypeU(4)); // the operation is just a constant from 0 through 16
325  break;
326  }
327  ASSERT_not_null(retval);
328  ASSERT_not_null(retval->get_type());
329  return retval;
330 }

399 SgAsmExpression*
400 Aarch64::extendOperand(SgAsmExpression *expr, const cs_insn &insn, arm64_extender extender, SgAsmType *dstType,
401  arm64_shifter shifter, unsigned shiftAmount) const {
402  using namespace ::Rose::BitOps;
403  ROSE_UNUSED(insn);
404 
405  ASSERT_not_null(expr);
406  ASSERT_not_null(expr->get_type());
407 
408  if (extender != ARM64_EXT_INVALID) {
409  ASSERT_require(isSgAsmIntegerType(expr->get_type()));
410  ASSERT_require(isSgAsmIntegerType(dstType));
411 
412  // First we need to truncate the incoming value to the specified input size.
413  SgAsmIntegerType *tmpType = nullptr;
414  switch (extender) {
415  case ARM64_EXT_INVALID:
416  ASSERT_not_reachable("invalid extender function");
417  case ARM64_EXT_UXTB:
418  tmpType = SageBuilderAsm::buildTypeU8();
419  break;
420  case ARM64_EXT_SXTB:
421  tmpType = SageBuilderAsm::buildTypeI8();
422  break;
423  case ARM64_EXT_UXTH:
424  tmpType = SageBuilderAsm::buildTypeU16();
425  break;
426  case ARM64_EXT_SXTH:
427  tmpType = SageBuilderAsm::buildTypeI16();
428  break;
429  case ARM64_EXT_UXTW:
430  tmpType = SageBuilderAsm::buildTypeU32();
431  break;
432  case ARM64_EXT_SXTW:
433  tmpType = SageBuilderAsm::buildTypeI32();
434  break;
435  case ARM64_EXT_UXTX:
436  tmpType = SageBuilderAsm::buildTypeU64();
437  break;
438  case ARM64_EXT_SXTX:
439  tmpType = SageBuilderAsm::buildTypeI64();
440  break;
441  }
442  if (tmpType->get_nBits() != expr->get_type()->get_nBits())
443  expr = SageBuilderAsm::buildTruncateExpression(expr, tmpType);
444 
445  // Now that we have the intermediate type of the correct size, extend it to the final size. If the intermediate
446  // expression is already the same as the destination size then there's nothing to do.
447  if (tmpType->get_nBits() != dstType->get_nBits()) {
448  switch (extender) {
449  case ARM64_EXT_UXTB:
450  case ARM64_EXT_UXTH:
451  case ARM64_EXT_UXTW:
452  case ARM64_EXT_UXTX:
453  expr = SageBuilderAsm::buildUnsignedExtendExpression(expr, dstType);
454  break;
455  default:
456  expr = SageBuilderAsm::buildSignedExtendExpression(expr, dstType);
457  break;
458  }
459  }
460  }
461 
462  if (shifter != ARM64_SFT_INVALID && shiftAmount != 0) {
463  ASSERT_require(shiftAmount < dstType->get_nBits());
464  auto amount = SageBuilderAsm::buildValueU8(shiftAmount);
465  switch (shifter) {
466  case ARM64_SFT_LSL:
467  expr = SageBuilderAsm::buildLslExpression(expr, amount);
468  break;
469  case ARM64_SFT_MSL:
470  expr = SageBuilderAsm::buildMslExpression(expr, amount);
471  break;
472  case ARM64_SFT_LSR:
473  expr = SageBuilderAsm::buildLsrExpression(expr, amount);
474  break;
475  case ARM64_SFT_ASR:
476  expr = SageBuilderAsm::buildAsrExpression(expr, amount);
477  break;
478  case ARM64_SFT_ROR:
479  expr = SageBuilderAsm::buildRorExpression(expr, amount);
480  break;
481  case ARM64_SFT_INVALID:
482  ASSERT_not_reachable("invalid shifter");
483  }
484  }
485 
486  ASSERT_not_null(expr);
487  ASSERT_not_null(expr->get_type());
488  ASSERT_require(expr->get_type() == dstType);
489  return expr;
490 }

492 void
493 Aarch64::wrapPrePostIncrement(SgAsmOperandList *operands, const cs_arm64 &cs_detail) {
494  ASSERT_not_null(operands);
495 
496  if (cs_detail.writeback) {
497  // This AArch64 instruction must have a memory reference operand and is either pre-update or post-update. A pre-update
498  // instruction will have both the register and the increment within the memory address expression, such as:
499  //
500  // Opcode: 95 9d 87 ad
501  // ARM syntax: stp q21, q7, [x12, #0xf0]!
502  // ROSE syntax: stp q21, q7, u128 [x12 + 0xf0] (before the transformation)
503  // ROSE syntax: stp q21, q7, u128 [x12 after x12 = x12 + 0xf0] (after the transformation)
504  //
505  // Semantically, the above means that the value in register x12 is added to the constant 0xf0 resulting in the first
506  // (lowest) address where data is written to memory.
507  //
508  // The post-increment operations is slightly different in that Capstone disassembles the expression as two separate
509  // instruction operands. For instance:
510  //
511  // Opcode: 95 9d 87 ac
512  // ARM syntax: stp q21, q7, [x12], #0xf0
513  // ROSE syntax: stp q21, q7, u128 [x12], 0xf0 (before the transformation)
514  // ROSE syntax: stp q21, q7, u128 [x12 then x12 = x12 + 0xf0] (after the transofmration)
515  //
516  // Semantically, the value stored originally in x12 is used as the memory address, and then the register is incremented
517  // by 0xf0. ROSE combines these two arguments into a single expression containing a SgAsmBinaryAddPostupdate node.
518  for (size_t i = 0; i < operands->get_operands().size(); ++i) {
519  if (auto mre = isSgAsmMemoryReferenceExpression(operands->get_operands()[i])) {
520  if (i + 1 == operands->get_operands().size()) {
521  // The memory reference expression is the last operand, therefore this is a pre-increment. The address must
522  // be an addition expression whose lhs is a register reference (or just a register expression, in which
523  // case the pre-increment is a no-op). Replace the SgAsmBinaryAdd with a SgAsmBinaryAddPreupdate node.
524  // First unlink things from the tree...
525  SgAsmBinaryAdd *add = isSgAsmBinaryAdd(mre->get_address());
526  if (!add) {
527  ASSERT_require(isSgAsmDirectRegisterExpression(mre->get_address()));
528  return; // if the offset is zero then we have already removed it and the pre-increment is a no-op.
529  }
530  SgAsmType *type = add->get_type();
531  ASSERT_not_null(type);
532 
533  // Create a new register reference expression for the lhs of the pre-update operator.
534  ASSERT_require(isSgAsmDirectRegisterExpression(add->get_lhs()));
535  auto lhs = new SgAsmDirectRegisterExpression(isSgAsmDirectRegisterExpression(add->get_lhs())->get_descriptor());
536  lhs->set_type(type);
537 
538  // Unlink things from the AST and construct a new address, linking it back into the AST.
539  mre->set_address(nullptr); add->set_parent(nullptr);
540  auto preUpdate = SageBuilderAsm::buildPreupdateExpression(lhs, add);
541  mre->set_address(preUpdate); preUpdate->set_parent(mre);
542  return;
543 
544  } else {
545  // The memory reference must be a post-increment, in which case it must the the penultimate argument. The
546  // address must be a register reference. First unlink things from the tree...
547  ASSERT_require(i + 2 == operands->get_operands().size());
548  auto postLhs = isSgAsmDirectRegisterExpression(mre->get_address());
549  ASSERT_not_null(postLhs);
550  mre->set_address(nullptr); postLhs->set_parent(nullptr);
551  auto inc = operands->get_operands()[i + 1];
552  operands->get_operands().pop_back();
553  inc->set_parent(nullptr);
554 
555  // Create a new register reference expression that's the old value of the register
556  auto oldValue = new SgAsmDirectRegisterExpression(postLhs->get_descriptor());
557  oldValue->set_type(postLhs->get_type());
558 
559  // Create the post-update expression and link it into the AST
560  auto newValue = SageBuilderAsm::buildAddExpression(oldValue, inc);
561  auto postUpdate = SageBuilderAsm::buildPostupdateExpression(postLhs, newValue);
562  mre->set_address(postUpdate); postUpdate->set_parent(mre);
563  return;
564  }
565  }
566  }
567  ASSERT_not_reachable("writeback was set but there is no memory reference operand");
568  }
569 }

848  uint32_t opc = bits(code, 30, 31);
849  // Memory access is twice as wide as the registers because we're loading or storing two registers.
850  return SageBuilderAsm::buildTypeU(8 << (3 + opc));
851  } else if (bits(code, 22, 30) == 0b010100001 || // LDNP
852  bits(code, 22, 30) == 0b010100011 || // LDP Post-index
853  bits(code, 22, 30) == 0b010100111 || // LDP Pre-index
854  bits(code, 22, 30) == 0b010100101 || // LDP Signed offset
855  bits(code, 22, 30) == 0b010100000 || // STNP
856  bits(code, 22, 30) == 0b010100010 || // STP Post-index
857  bits(code, 22, 30) == 0b010100110 || // STP Pre-index
858  bits(code, 22, 30) == 0b010100100) { // STP Signed offset
859  // Memory access is twice as wide as the registers because we're loading or storing two registers.
860  return SageBuilderAsm::buildTypeU(bit(code, 31) ? 128 : 64);
861  } else {
862  ASSERT_not_reachable("invalid opcode");
863  }
864  }
865 
866  //--------------------------------------------------------------------------------------------------------
867  case Kind::ARM64_INS_LDR: // load register
868  case Kind::ARM64_INS_LDUR: // load register unscaled
869  case Kind::ARM64_INS_STR: // store register
870  case Kind::ARM64_INS_STUR: { // store register unscaled offset
871  if ((bits(code, 24, 29) == 0b111100 && bits(code, 21, 22) == 0b10 && bits(code, 10, 11) == 0b01) || // LDR (immediate, SIMD&FP) Post-index
872  (bits(code, 24, 29) == 0b111100 && bits(code, 21, 22) == 0b10 && bits(code, 10, 11) == 0b11) || // LDR (immediate, SIMD&FP) Pre-index
873  (bits(code, 24, 29) == 0b111101 && bit(code, 22)) || // LDR (immediate, SIMD&FP) Unsigned offset
874  (bits(code, 24, 29) == 0b111100 && bits(code, 21, 22) == 0b11 && bits(code, 10, 11) == 0b10) || // LDR (register, SIMD&FP)
875  (bits(code, 24, 29) == 0b111100 && bits(code, 21, 22) == 0b10 && bits(code, 10, 11) == 0b00) || // LDUR (SIMD&FP)
876  (bits(code, 24, 29) == 0b111100 && bits(code, 21, 22) == 0b00 && bits(code, 10, 11) == 0b01) || // STR (immediate, SIMD&FP) Post-index
877  (bits(code, 24, 29) == 0b111100 && bits(code, 21, 22) == 0b00 && bits(code, 10, 11) == 0b11) || // STR (immediate, SIMD&FP) Pre-index
878  (bits(code, 24, 29) == 0b111101 && !bit(code, 22)) || // STR (immediate, SIMD&FP) Unsigned offset
879  (bits(code, 24, 29) == 0b111100 && bits(code, 21, 22) == 0b01 && bits(code, 10, 11) == 0b10) || // STR (register, SIMD&FP)
880  (bits(code, 24, 29) == 0b111100 && bits(code, 21, 22) == 0b00 && bits(code, 10, 11) == 0b00)) { // STUR (SIMD&FP)
881  if (bits(code, 30, 31) == 0b00 && !bit(code, 23)) {
882  return SageBuilderAsm::buildTypeU(8);
883  } else if (bits(code, 30, 31) == 0b01 && !bit(code, 23)) {
884  return SageBuilderAsm::buildTypeU(16);
885  } else if (bits(code, 30, 31) == 0b10 && !bit(code, 23)) {
886  return SageBuilderAsm::buildTypeU(32);
887  } else if (bits(code, 30, 31) == 0b11 && !bit(code, 23)) {
888  return SageBuilderAsm::buildTypeU(64);
889  } else {
890  ASSERT_require(bits(code, 30, 31) == 0b00);
891  ASSERT_require(bit(code, 23));
892  return SageBuilderAsm::buildTypeU(128);
893  }
894  } else if (( bit(code, 31) && bits(code, 21, 29) == 0b111000010 && bits(code, 10, 11) == 0b01) || // LDR (immediate) Post-index
895  ( bit(code, 31) && bits(code, 21, 29) == 0b111000010 && bits(code, 10, 11) == 0b11) || // LDR (immediate) Pre-index
896  ( bit(code, 31) && bits(code, 22, 29) == 0b11100101) || // LDR (immediate) unsigned offset
897  (!bit(code, 31) && bits(code, 24, 29) == 0b011000) || // LDR (literal)
898  ( bit(code, 31) && bits(code, 21, 29) == 0b111000011 && bits(code, 10, 11) == 0b10) || // LDR (register)
899  ( bit(code, 31) && bits(code, 21, 29) == 0b111000010 && bits(code, 10, 11) == 0b00) || // LDUR
900  ( bit(code, 31) && bits(code, 21, 29) == 0b111000000 && bits(code, 10, 11) == 0b01) || // STR (immediate) Post-index
901  ( bit(code, 31) && bits(code, 21, 29) == 0b111000000 && bits(code, 10, 11) == 0b11) || // STR (immediate) Pre-index
902  ( bit(code, 31) && bits(code, 22, 29) == 0b11100100) || // STR (immediate) unsigned offset
903  ( bit(code, 31) && bits(code, 21, 29) == 0b111000001 && bits(code, 10, 11) == 0b10) || // STR (register)
904  ( bit(code, 31) && bits(code, 21, 29) == 0b111000000 && bits(code, 10, 11) == 0b00)) { // STUR
905  if (bit(code, 30) == 0) {
906  return SageBuilderAsm::buildTypeU(32);
907  } else {
908  return SageBuilderAsm::buildTypeU(64);
909  }
910  } else if (bits(code, 24, 29) == 0b011100) { // LDR (literal, SIMD&FP)
911  if (bits(code, 30, 31) == 0b00) {
912  return SageBuilderAsm::buildTypeU(32);
913  } else if (bits(code, 30, 31) == 0b01) {
914  return SageBuilderAsm::buildTypeU(64);
915  } else {
916  ASSERT_require(bits(code, 30, 31) == 0b10);
917  return SageBuilderAsm::buildTypeU(128);
918  }
919  } else {
920  ASSERT_not_reachable("invalid opcode");
921  }
922  break;
923  }
924 
925  //--------------------------------------------------------------------------------------------------------
926  default:
927  ASSERT_not_reachable("memory read instruction not handled");
928  }
929 }
