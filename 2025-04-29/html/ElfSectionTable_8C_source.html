103 SgAsmElfSectionTable *
104 SgAsmElfSectionTable::parse()
105 {
106  SgAsmGenericSection::parse();
107 
108  SgAsmElfFileHeader *fhdr = dynamic_cast<SgAsmElfFileHeader*>(get_header());
109  ROSE_ASSERT(fhdr!=NULL);
110  Rose::BinaryAnalysis::ByteOrder::Endianness sex = fhdr->get_sex();
111 
112  size_t ent_size, struct_size, opt_size, nentries;
113  calculateSizes(&ent_size, &struct_size, &opt_size, &nentries);
114  ROSE_ASSERT(opt_size==fhdr->get_shextrasz() && nentries==fhdr->get_e_shnum());
115 
116  /* If the current size is very small (0 or 1 byte) then we're coming straight from the constructor and the parsing should
117  * also extend this section to hold all the entries. Otherwise the caller must have assigned a specific size for a good
118  * reason and we should leave that alone, reading zeros if the entries extend beyond the defined size. */
119  if (get_size()<=1 && get_size()<nentries*ent_size)
120  extend(nentries*ent_size - get_size());
121 
122  // Read all the section headers. Section headers are not essential to the Unix loader, which uses only segments. Therefore
123  // we should be prepared to handle bad entries.
124  std::vector<SgAsmElfSectionTableEntry*> entries;
125  Rose::BinaryAnalysis::Address offset = 0;
126  try {
127  for (size_t i=0; i<nentries; i++, offset+=ent_size) {
128  SgAsmElfSectionTableEntry *shdr = NULL;
129  if (4 == fhdr->get_wordSize()) {
130  SgAsmElfSectionTableEntry::Elf32SectionTableEntry_disk disk;
131  readContentLocal(offset, &disk, struct_size);
132  shdr = new SgAsmElfSectionTableEntry(sex, &disk);
133  } else {
134  SgAsmElfSectionTableEntry::Elf64SectionTableEntry_disk disk;
135  readContentLocal(offset, &disk, struct_size);
136  shdr = new SgAsmElfSectionTableEntry(sex, &disk);
137  }
138  if (opt_size>0)
139  shdr->get_extra() = readContentLocalUcl(offset+struct_size, opt_size);
140  entries.push_back(shdr);
141  }
142  } catch (const ShortRead &error) {
143  mlog[ERROR] <<"short read for elf section header #" <<entries.size()
144  <<" at file offset " <<StringUtility::addrToString(error.offset)
145  <<" when reading " <<StringUtility::plural(error.size, "bytes") <<"\n";
146  mlog[ERROR] <<"expected " <<StringUtility::plural(nentries, "sections") <<", but bailing out early\n";
147  nentries = entries.size();
148  }
149 
150  /* This vector keeps track of which sections have already been parsed. We could get the same information by calling
151  * fhdr->get_section_by_id() and passing the entry number since entry numbers and IDs are one and the same in ELF. However,
152  * this is a bit easier. */
153  std::vector<SgAsmElfSection*> is_parsed;
154  is_parsed.resize(entries.size(), NULL);
155 
156  /* All sections implicitly depend on the section string table for their names. */
157  SgAsmElfStringSection *section_name_strings=NULL;
158  if (fhdr->get_e_shstrndx() > 0 && fhdr->get_e_shstrndx() < entries.size()) {
159  SgAsmElfSectionTableEntry *entry = entries[fhdr->get_e_shstrndx()];
160  ASSERT_not_null(entry);
161  section_name_strings = new SgAsmElfStringSection(fhdr);
162  section_name_strings->initFromSectionTable(entry, section_name_strings, fhdr->get_e_shstrndx());
163  section_name_strings->parse();
164  is_parsed[fhdr->get_e_shstrndx()] = section_name_strings;
165  }
166 
167  /* Read all the sections. Some sections depend on other sections, so we read them in such an order that all dependencies
168  * are satisfied first. */
169  while (1) {
170  bool try_again=false;
171  for (size_t i=0; i<entries.size(); i++) {
172  SgAsmElfSectionTableEntry *entry = entries[i];
173  ROSE_ASSERT(entry->get_sh_link()<entries.size());
174 
175  /* Some sections might reference another section through the sh_link member. */
176  bool need_linked = entry->get_sh_link() > 0;
177  ROSE_ASSERT(!need_linked || entry->get_sh_link()<entries.size());
178  SgAsmElfSection *linked = need_linked ? is_parsed[entry->get_sh_link()] : NULL;
179 
180  /* Relocation sections might have a second linked section stored in sh_info. */
181  bool need_info_linked = (entry->get_sh_type() == SgAsmElfSectionTableEntry::SHT_REL ||
182  entry->get_sh_type() == SgAsmElfSectionTableEntry::SHT_RELA) &&
183  entry->get_sh_info() > 0;
184  ROSE_ASSERT(!need_info_linked || entry->get_sh_info()<entries.size());
185  SgAsmElfSection *info_linked = need_info_linked ? is_parsed[entry->get_sh_info()] : NULL;
186 
187  if (is_parsed[i]) {
188  /* This section has already been parsed. */
189  } else if ((need_linked && !linked) || (need_info_linked && !info_linked)) {
190  /* Don't parse this section yet because it depends on something that's not parsed yet. */
191  try_again = true;
192  } else {
193  switch (entry->get_sh_type()) {
194  case SgAsmElfSectionTableEntry::SHT_NULL:
195  /* Null entry. We still create the section just to hold the section header. */
196  is_parsed[i] = new SgAsmElfSection(fhdr);
197  break;
198  case SgAsmElfSectionTableEntry::SHT_NOBITS:
199  /* These types of sections don't occupy any file space (e.g., BSS) */
200  is_parsed[i] = new SgAsmElfSection(fhdr);
201  break;
202  case SgAsmElfSectionTableEntry::SHT_DYNAMIC: {
203  SgAsmElfStringSection *strsec = dynamic_cast<SgAsmElfStringSection*>(linked);
204  ROSE_ASSERT(strsec);
205  is_parsed[i] = new SgAsmElfDynamicSection(fhdr, strsec);
206  break;
207  }
208  case SgAsmElfSectionTableEntry::SHT_DYNSYM: {
209  SgAsmElfStringSection *strsec = dynamic_cast<SgAsmElfStringSection*>(linked);
210  ROSE_ASSERT(strsec);
211  SgAsmElfSymbolSection *symsec = new SgAsmElfSymbolSection(fhdr, strsec);
212  symsec->set_isDynamic(true);
213  is_parsed[i] = symsec;
214  break;
215  }
216  case SgAsmElfSectionTableEntry::SHT_SYMTAB: {
217  SgAsmElfStringSection *strsec = dynamic_cast<SgAsmElfStringSection*>(linked);
218  ROSE_ASSERT(strsec);
219  SgAsmElfSymbolSection *symsec = new SgAsmElfSymbolSection(fhdr, strsec);
220  symsec->set_isDynamic(false);
221  is_parsed[i] = symsec;
222  break;
223  }
224  case SgAsmElfSectionTableEntry::SHT_STRTAB:
225  is_parsed[i] = new SgAsmElfStringSection(fhdr);
226  break;
227  case SgAsmElfSectionTableEntry::SHT_REL: {
228  SgAsmElfSymbolSection *symbols = dynamic_cast<SgAsmElfSymbolSection*>(linked);
229  SgAsmElfRelocSection *relocsec = new SgAsmElfRelocSection(fhdr, symbols, info_linked);
230  relocsec->set_usesAddend(false);
231  is_parsed[i] = relocsec;
232  break;
233  }
234  case SgAsmElfSectionTableEntry::SHT_RELA: {
235  SgAsmElfSymbolSection *symbols = dynamic_cast<SgAsmElfSymbolSection*>(linked);
236  SgAsmElfRelocSection *relocsec = new SgAsmElfRelocSection(fhdr, symbols, info_linked);
237  relocsec->set_usesAddend(true);
238  is_parsed[i] = relocsec;
239  break;
240  }
241  case SgAsmElfSectionTableEntry::SHT_PROGBITS: {
242  if (!section_name_strings) {
243  fprintf(stderr, "SgAsmElfSectionTable::parse(): no string table for section table\n");
244  is_parsed[i] = new SgAsmElfSection(fhdr);
245  } else {
246  std::string section_name = section_name_strings->readContentLocalString(entry->get_sh_name());
247  if (section_name == ".eh_frame") {
248  is_parsed[i] = new SgAsmElfEHFrameSection(fhdr);
249  } else {
250  is_parsed[i] = new SgAsmElfSection(fhdr);
251  }
252  }
253  break;
254  }
255  case SgAsmElfSectionTableEntry::SHT_GNU_versym: {
256  is_parsed[i] = new SgAsmElfSymverSection(fhdr);
257  break;
258  }
259  case SgAsmElfSectionTableEntry::SHT_GNU_verdef: {
260  SgAsmElfStringSection *strsec = dynamic_cast<SgAsmElfStringSection*>(linked);
261  ROSE_ASSERT(strsec);
262  is_parsed[i] = new SgAsmElfSymverDefinedSection(fhdr,strsec);
263  break;
264  }
265  case SgAsmElfSectionTableEntry::SHT_GNU_verneed: {
266  SgAsmElfStringSection *strsec = dynamic_cast<SgAsmElfStringSection*>(linked);
267  ROSE_ASSERT(strsec);
268  is_parsed[i] = new SgAsmElfSymverNeededSection(fhdr,strsec);
269  break;
270  }
271  default:
272  is_parsed[i] = new SgAsmElfSection(fhdr);
273  break;
274  }
275  is_parsed[i]->initFromSectionTable(entry, section_name_strings, i);
276  is_parsed[i]->parse();
277  }
278  }
279  if (!try_again)
280  break;
281  }
282 
283  /* Initialize links between sections */
284  for (size_t i = 0; i < entries.size(); i++) {
285  SgAsmElfSectionTableEntry *shdr = entries[i];
286  if (shdr->get_sh_link() > 0) {
287  SgAsmElfSection *source = isSgAsmElfSection(fhdr->get_file()->get_sectionById(i));
288  SgAsmElfSection *target = isSgAsmElfSection(fhdr->get_file()->get_sectionById(shdr->get_sh_link()));
289  assert(source); /* because we created it above */
290  source->set_linkedSection(target);
291  }
292  }
293 
294  /* Finish parsing sections now that we have basic info for all the sections. */
295  for (size_t i=0; i<is_parsed.size(); i++)
296  is_parsed[i]->finishParsing();
297 
298  return this;
299 }

307 SgAsmElfSectionTableEntry *
308 SgAsmElfSectionTable::addSection(SgAsmElfSection *section)
309 {
310  ROSE_ASSERT(section!=NULL);
311  ROSE_ASSERT(section->get_file()==get_file());
312  ROSE_ASSERT(section->get_header()==get_header());
313  ROSE_ASSERT(section->get_sectionEntry()==NULL); /* must not be in the section table yet */
314 
315  SgAsmElfFileHeader *fhdr = dynamic_cast<SgAsmElfFileHeader*>(get_header());
316  ROSE_ASSERT(fhdr!=NULL);
317 
318  /* Assign an ID if there isn't one yet */
319  if (section->get_id()<0) {
320  int id = fhdr->get_e_shnum();
321  fhdr->set_e_shnum(id+1);
322  section->set_id(id);
323  }
324 
325  /* If the supplied section is a string table and the ELF Section Table doesn't have a string table associated with it yet,
326  * then use the supplied section as the string table to hold the names of the sections. When this happens, all sections
327  * that are already defined in the ELF Section Table should have their names moved into the new string table. */
328  SgAsmElfStringSection *strsec = NULL;
329  if (fhdr->get_e_shstrndx()==0) {
330  strsec = dynamic_cast<SgAsmElfStringSection*>(section);
331  if (strsec) {
332  fhdr->set_e_shstrndx(section->get_id());
333  SgAsmGenericSectionList *all = fhdr->get_sections();
334  for (size_t i=0; i<all->get_sections().size(); i++) {
335  SgAsmElfSection *s = dynamic_cast<SgAsmElfSection*>(all->get_sections()[i]);
336  if (s && s->get_id()>=0 && s->get_sectionEntry()!=NULL) {
337  s->allocateNameToStorage(strsec);
338  }
339  }
340  }
341  } else {
342  strsec = dynamic_cast<SgAsmElfStringSection*>(fhdr->get_sectionById(fhdr->get_e_shstrndx()));
343  ROSE_ASSERT(strsec!=NULL);
344  }
345 
346  /* Make sure the name is in the correct string table */
347  if (strsec)
348  section->allocateNameToStorage(strsec);
349 
350  /* Create a new section table entry. */
351  SgAsmElfSectionTableEntry *shdr = new SgAsmElfSectionTableEntry;
352  shdr->updateFromSection(section);
353  section->set_sectionEntry(shdr);
354 
355  return shdr;
356 }

364 Rose::BinaryAnalysis::Address
365 SgAsmElfSectionTable::calculateSizes(size_t *entsize, size_t *required, size_t *optional, size_t *entcount) const
366 {
367  SgAsmElfFileHeader *fhdr = dynamic_cast<SgAsmElfFileHeader*>(get_header());
368  ROSE_ASSERT(fhdr!=NULL);
369 
370  size_t struct_size = 0;
371  size_t extra_size = fhdr->get_shextrasz();
372  size_t entry_size = 0;
373  size_t nentries = 0;
374 
375  /* Size of required part of each entry */
376  if (4==fhdr->get_wordSize()) {
377  struct_size = sizeof(SgAsmElfSectionTableEntry::Elf32SectionTableEntry_disk);
378  } else if (8==fhdr->get_wordSize()) {
379  struct_size = sizeof(SgAsmElfSectionTableEntry::Elf64SectionTableEntry_disk);
380  } else {
381  throw FormatError("bad ELF word size");
382  }
383 
384  /* Entire entry should be at least large enough for the required part. */
385  entry_size = struct_size;
386 
387  /* Size of optional parts. If we've parsed the table then use the largest optional part, otherwise assume the entry from
388  * the ELF File Header is correct. */
389  SgAsmGenericSectionPtrList sections = fhdr->get_sections()->get_sections();
390  for (size_t i=0; i<sections.size(); i++) {
391  SgAsmElfSection *elfsec = dynamic_cast<SgAsmElfSection*>(sections[i]);
392  if (elfsec && elfsec->get_sectionEntry()) {
393  ROSE_ASSERT(elfsec->get_id()>=0);
394  nentries = std::max(nentries, (size_t)elfsec->get_id()+1);
395  extra_size = std::max(extra_size, elfsec->get_sectionEntry()->get_extra().size());
396  }
397  }
398 
399  /* Total number of entries. Either we haven't parsed the section table yet (nor created the sections it defines) or we
400  * have. In the former case we use the setting from the ELF File Header. Otherwise the table has to be large enough to
401  * store the section with the largest ID (ID also serves as the index into the ELF Section Table). */
402  if (0==nentries)
403  nentries = fhdr->get_e_shnum();
404 
405  /* Return values */
406  if (entsize)
407  *entsize = entry_size;
408  if (required)
409  *required = struct_size;
410  if (optional)
411  *optional = extra_size;
412  if (entcount)
413  *entcount = nentries;
414  return entry_size * nentries;
415 }

423 void
424 SgAsmElfSectionTableEntry::updateFromSection(SgAsmElfSection *section)
425 {
426  if (section->get_name()->get_offset()==SgAsmGenericString::unallocated) {
427  p_sh_name = 0; /*not a stored string after all*/
428  } else {
429  p_sh_name = section->get_name()->get_offset();
430  }
431 
432  set_sh_offset(section->get_offset());
433  if (get_sh_type()==SHT_NOBITS && section->isMapped()) {
434  set_sh_size(section->get_mappedSize());
435  } else {
436  set_sh_size(section->get_size());
437  }
438 
439  if (section->isMapped()) {
440  set_sh_addr(section->get_mappedPreferredRva());
441  set_sh_addralign(section->get_mappedAlignment());
442  if (section->get_mappedWritePermission()) {
443  p_sh_flags |= SHF_WRITE;
444  } else {
445  p_sh_flags &= ~SHF_WRITE;
446  }
447  if (section->get_mappedExecutePermission()) {
448  p_sh_flags |= SHF_EXECINSTR;
449  } else {
450  p_sh_flags &= ~SHF_EXECINSTR;
451  }
452  } else {
453  set_sh_addr(0);
454  p_sh_flags &= ~(SHF_WRITE | SHF_EXECINSTR); /*clear write & execute bits*/
455  }
456 
457  SgAsmElfSection *linked_to = section->get_linkedSection();
458  if (linked_to) {
459  ROSE_ASSERT(linked_to->get_id()>0);
460  set_sh_link(linked_to->get_id());
461  } else {
462  set_sh_link(0);
463  }
464 }

472 std::string
473 SgAsmElfSectionTableEntry::toString(SectionType t)
474 {
475 #ifndef _MSC_VER
476  std::string retval = stringifySgAsmElfSectionTableEntrySectionType(t);
477 #else
478  ROSE_ABORT();
479  std::string retval = "";
480 #endif
481  if ('('!=retval[0])
482  return retval;
483 
484  char buf[128];
485  if(t>=SHT_LOOS && t <= SHT_HIOS) {
486  snprintf(buf,sizeof(buf),"os-specific (%" PRIuPTR ")",size_t(t)) ;
487  return buf;

589

591  size_t opt_size, nentries;
592  Rose::BinaryAnalysis::Address need = calculateSizes(NULL, NULL, &opt_size, &nentries);
593  if (need < get_size()) {

596  set_mappedSize(need);

603  reallocated = true;
604  }

607  SgAsmElfFileHeader *fhdr = dynamic_cast<SgAsmElfFileHeader*>(get_header());
608  fhdr->set_shextrasz(opt_size);
609  fhdr->set_e_shnum(nentries);

614 void
615 SgAsmElfSectionTable::unparse(std::ostream &f) const
616 {
617  SgAsmElfFileHeader *fhdr = dynamic_cast<SgAsmElfFileHeader*>(get_header());
618  ROSE_ASSERT(fhdr!=NULL);
619  Rose::BinaryAnalysis::ByteOrder::Endianness sex = fhdr->get_sex();
620  SgAsmGenericSectionPtrList sections = fhdr->get_sectionTableSections();
621 
622  /* Write the sections first */
623  for (size_t i=0; i<sections.size(); i++)
624  sections[i]->unparse(f);
625  unparseHoles(f);
626 
627  /* Calculate sizes. The ELF File Header should have been updated in reallocate() prior to unparsing. */
628  size_t ent_size, struct_size, opt_size, nentries;
629  calculateSizes(&ent_size, &struct_size, &opt_size, &nentries);
630  ROSE_ASSERT(fhdr->get_shextrasz()==opt_size);
631  ROSE_ASSERT(fhdr->get_e_shnum()==nentries);
632 
633  /* Write the section table entries */
634  for (size_t i=0; i<sections.size(); ++i) {
635  SgAsmElfSection *section = dynamic_cast<SgAsmElfSection*>(sections[i]);
636  ROSE_ASSERT(section!=NULL);
637  SgAsmElfSectionTableEntry *shdr = section->get_sectionEntry();
638  ROSE_ASSERT(shdr!=NULL);
639  ROSE_ASSERT(shdr->get_sh_offset()==section->get_offset());/*section table entry should have been updated in reallocate()*/
640 
641  int id = section->get_id();
642  ROSE_ASSERT(id>=0 && (size_t)id<nentries);
643 
644  SgAsmElfSectionTableEntry::Elf32SectionTableEntry_disk disk32;
645  SgAsmElfSectionTableEntry::Elf64SectionTableEntry_disk disk64;
646  void *disk = NULL;
647 
648  if (4==fhdr->get_wordSize()) {
649  disk = shdr->encode(sex, &disk32);
650  } else if (8==fhdr->get_wordSize()) {
651  disk = shdr->encode(sex, &disk64);
652  } else {
653  ROSE_ASSERT(!"invalid word size");
654  }
655 
656  /* The disk struct */
657  Rose::BinaryAnalysis::Address spos = write(f, id*ent_size, struct_size, disk);
658  if (shdr->get_extra().size() > 0) {
659  ROSE_ASSERT(shdr->get_extra().size()<=opt_size);
660  write(f, spos, shdr->get_extra());
661  }
662  }
