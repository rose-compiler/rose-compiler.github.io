283 POETCode* POETProgram::
284 make_Bop( POETOperatorType t, POETCode* arg1, POETCode* arg2, int lineno)
285 {
286  switch (t)
287  {
288  case POET_OP_MAP : return ASTFactory::inst()->new_MAP(arg1,arg2);
289  case POET_OP_LIST : case POET_OP_LIST1 :
290  return new POETParseList(t,arg1,arg2);
291  case POET_OP_TYPEMATCH: case POET_OP_TYPEMATCH_Q:
292  case POET_OP_ASTMATCH: case POET_OP_SPLIT: case POET_OP_POND: case POET_OP_PLUS: case POET_OP_CONCAT:
293  case POET_OP_MINUS: case POET_OP_MULT: case POET_OP_DIVIDE: case POET_OP_MOD: case POET_OP_AND:
294  case POET_OP_OR: case POET_OP_EQ: case POET_OP_LT: case POET_OP_LE: case POET_OP_GT: case POET_OP_RANGE:
295  case POET_OP_GE: case POET_OP_NE: case POET_OP_CONS: case POET_OP_DOT: case POET_OP_DEBUG:
296  case POET_OP_REPLACE: case POET_OP_PERMUTE: case POET_OP_INSERT:
297  case POET_OP_ERASE: case POET_OP_TRACE:
298  return new POETBop(t,arg1,arg2);
299  case POET_OP_CASE:
300  return (lineno < 0)? new POETBop(t, arg1, arg2) :
301  (POETCode*)new POETEvalExp(new POETBop(t, arg1, arg2), get_filename(),lineno);
302  case POET_OP_SEQ:
303  return (lineno < 0)? new POETBop(t, arg1, arg2) :
304  new POETBop(t, new POETEvalExp(arg1,get_filename(),lineno), arg2);
305  default: std::cerr << "unexpected Bop: " << OpName[t] << "\n"; assert(0);
306  }
307 }
