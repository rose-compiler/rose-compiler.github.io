309 void
310 RiscOperators::emit_prerequisites(std::ostream &o, const RegisterDescriptors &regs, const RegisterDictionary::Ptr& dictionary)
311 {
312  struct T1: SymbolicExpression::Visitor {
313  RiscOperators *ops;
314  std::ostream &o;
315  const RegisterDescriptors &regs;
316  RegisterDictionary::Ptr dictionary;
317  std::set<SymbolicExpression::Hash> seen;
318  T1(RiscOperators *ops, std::ostream &o, const RegisterDescriptors &regs, const RegisterDictionary::Ptr &dictionary)
319  : ops(ops), o(o), regs(regs), dictionary(dictionary) {}
320  virtual SymbolicExpression::VisitAction preVisit(const ExpressionPtr &node) override {
321  if (!seen.insert(node->hash()).second)
322  return SymbolicExpression::TRUNCATE; // already processed this same expression
323  size_t width = node->nBits();
324  if (InteriorPtr inode = node->isInteriorNode()) {
325  if (SymbolicExpression::OP_READ==inode->getOperator()) {
326  ASSERT_require(2==inode->nChildren());
327  ops->emit_assignment(o, ops->emit_memory_read(o, inode->child(1), width));
328  }
329  } else {
330  LeafPtr leaf = node->isLeafNode();
331  ASSERT_not_null(leaf);
332  if (leaf->isIntegerVariable()) {
333  std::string comment = leaf->comment();
334  if (comment.size()>2 && 0==comment.substr(comment.size()-2).compare("_0"))
335  ops->add_variable(leaf); // becomes a global variable if not already
336  LeafPtr t1 = ops->emit_expression(o, leaf);// handles local vars, global vars, and undefs
337  }
338  }
339  return SymbolicExpression::CONTINUE;
340  }
341  virtual SymbolicExpression::VisitAction postVisit(const ExpressionPtr&) override {
342  return SymbolicExpression::CONTINUE;
343  }
344  } t1(this, o, regs, dictionary);
345 
346  // Prerequisites for the registers
347  RegisterState::Ptr regstate = RegisterState::promote(currentState()->registerState());
348  for (size_t i=0; i<regs.size(); ++i) {
349  BaseSemantics::SValue::Ptr dflt = undefined_(regs[i].nBits());
350  SValue::Ptr value = SValue::promote(regstate->readRegister(regs[i], dflt, this));
351  value->get_expression()->depthFirstTraversal(t1);
352  }
353 
354  // Prerequisites for memory writes
355  for (TreeNodes::const_iterator mwi=mem_writes.begin(); mwi!=mem_writes.end(); ++mwi) {
356  const ExpressionPtr mem_write = *mwi;
357  mem_write->depthFirstTraversal(t1);
358  }
359 
360  // Prerequisites for the instruction pointer.
361  get_instruction_pointer()->get_expression()->depthFirstTraversal(t1);
362 }

392 void
393 RiscOperators::emit_next_eip(std::ostream &o, SgAsmInstruction *latest_insn)
394 {
395  SgAsmBlock *bb = AST::Traversal::findParentTyped<SgAsmBlock>(latest_insn);
396  SgAsmFunction *func = AST::Traversal::findParentTyped<SgAsmFunction>(bb);
397  SgAsmInterpretation *interp = AST::Traversal::findParentTyped<SgAsmInterpretation>(func);
398  ASSERT_not_null(interp); // instructions must be part of the global AST
399  const InstructionMap &insns = interp->get_instructionMap();
400  SValue::Ptr eip = get_instruction_pointer();
401  Address fallthrough_va = latest_insn->get_address() + latest_insn->get_size();
402 
403  // If EIP is a constant then it is one of the following cases:
404  // 1. It points to an instruction that's in a different function than the last executed instruction, in which
405  // case we treat it like a function call. LLVM requires us to treat all inter-function control flow as a
406  // function call even though the same restriction isn't present in the binary. FIXME[Robb P. Matzke 2014-01-09]
407  // 2. It is an unconditional intra-function branch which can be translated to an LLVM unconditional "br" instruction.
408  // 3. It is the fall-through address added by transcodeBasicBlock for an instruction for which semantics failed when
409  // quiet-errors mode is enabled, and therefore might be completely invalid.
410  if (auto eipVal = eip->toUnsigned()) {
411  SgAsmInstruction *dst_insn = insns.get_value_or(*eipVal, NULL);
412  SgAsmFunction *dst_func = AST::Traversal::findParentTyped<SgAsmFunction>(dst_insn);
413  if (!dst_func) {
414  o <<prefix() <<"unreachable\n";
415  } else if (func!=dst_func) { // func could be null
416  std::string funcname = function_label(dst_func);
417  o <<prefix() <<"call void " <<funcname <<"()\n";
418  Address ret_addr = fallthrough_va;
419  SgAsmFunction *ret_func = AST::Traversal::findParentTyped<SgAsmFunction>(insns.get_value_or(ret_addr, NULL));
420  if (ret_func!=func) {
421  // The fall through address might be invalid or in a different function if the call never returns.
422  o <<prefix() <<"unreachable\n";
423  } else {
424  o <<prefix() <<"br label %" <<addr_label(ret_addr) <<"\n";
425  }
426  } else {
427  o <<prefix() <<"br label %" <<addr_label(*eipVal) <<"\n";
428  }
429  return;
430  }
431 
432  o <<prefix() <<"; register eip = " <<*eip <<"\n";
433 
434  // If EIP is a symbolic if-then-else ("ite") and both operands are constants then the binary has a conditional branch
435  // instruction (like an x86 "je", "jne", etc.) and we can emit an LLVM conditional "br" with true and false parts. However,
436  // we must watch out for the case when the ROSE disassembler determined that the predicate is opaque and one of the target
437  // addresses isn't valid. This can happen because the ROSE disassembler might be using a more advanced analysis than we
438  // use here.
439  InteriorPtr inode = eip->get_expression()->isInteriorNode();
440  if (inode && SymbolicExpression::OP_ITE==inode->getOperator()) {
441  LeafPtr leaf1 = inode->child(1)->isLeafNode();
442  LeafPtr leaf2 = inode->child(2)->isLeafNode();
443  if (leaf1!=NULL && leaf1->isIntegerConstant() && leaf2!=NULL && leaf2->isIntegerConstant()) {
444  Address true_va = leaf1->toUnsigned().get();
445  Address false_va = leaf2->toUnsigned().get();
446  if (false_va != fallthrough_va)
447  std::swap(true_va, false_va);
448  SgAsmFunction *true_func = AST::Traversal::findParentTyped<SgAsmFunction>(insns.get_value_or(true_va, NULL));
449  SgAsmFunction *false_func = AST::Traversal::findParentTyped<SgAsmFunction>(insns.get_value_or(false_va, NULL));
450  const SgAsmIntegerValuePtrList &succs = bb->get_successors();
451  std::vector<Address> succs_va;
452  for (SgAsmIntegerValuePtrList::const_iterator si=succs.begin(); si!=succs.end(); ++si)
453  succs_va.push_back((*si)->get_absoluteValue());
454 
455  if (succs.size()==2 && true_func==func && false_func==func &&
456  std::min(succs_va[0], succs_va[1])==std::min(true_va, false_va) &&
457  std::max(succs_va[0], succs_va[1])==std::max(true_va, false_va)) {
458  // This is a normal intra-function conditional branch that can be translated directly to an LLVM "br"
459  LeafPtr t1 = emit_expression(o, inode->child(0));
460  o <<prefix() <<"br i1 " <<llvm_term(t1)
461  <<", label %" <<addr_label(true_va) <<", label %" <<addr_label(false_va) <<"\n";
462  return;
463  } else if (succs.size()==1 && (succs_va[0]==true_va || succs_va[0]==false_va) &&
464  AST::Traversal::findParentTyped<SgAsmFunction>(insns.get_value_or(succs_va[0], NULL))==func) {
465  // An intra-function conditional branch with opaque predicate.
466  o <<prefix() <<"br label %" <<addr_label(succs_va[0]) <<"\n";
467  return;
468  } else if (true_func==func && false_func==func) {
469  // CFG succs info is fishy, but both values of the "ite" are valid intra-function blocks
470  LeafPtr t1 = emit_expression(o, inode->child(0));
471  o <<prefix() <<"br i1 " <<llvm_term(t1)
472  <<", label %" <<addr_label(true_va) <<", label %" <<addr_label(false_va) <<"\n";
473  return;
474  }
475  }
476  }

479  // The stack used by LLVM is distinct from the stack used by the binary. The latter's stack is implemented via the @ebp (or
480  // similar) global pointer. FIXME[Robb P. Matzke 2014-01-09]: This is architecture dependent.
481  if (SgAsmX86Instruction *insn_x86 = isSgAsmX86Instruction(latest_insn)) {
482  if (insn_x86->get_kind() == x86_ret || insn_x86->get_kind() == x86_retf) {
483  o <<prefix() <<"ret void\n";
484  return;
485  }
486  }

488  // If this function is a thunk, then we need to treat it as an LLVM function call (because LLVM doesn't allow
489  // inter-function branches). FIXME[Robb P. Matzke 2014-01-09]: This is architecture dependent.
490  if (SgAsmX86Instruction *insn_x86 = isSgAsmX86Instruction(latest_insn)) {
491  std::vector<SgAsmInstruction*> func_insns;
492  AST::Traversal::forwardPre<SgAsmInstruction>(func, [&func_insns](SgAsmInstruction *insn) {
493  func_insns.push_back(insn);
494  });

496  if (func_insns.size()==1 && func_insns.front()==insn_x86 &&
497  (insn_x86->get_kind() == x86_jmp || insn_x86->get_kind() == x86_farjmp)) {
498  LeafPtr t1 = emit_expression(o, eip);
499  LeafPtr t2 = next_temporary(32); // pointer to the function
500  o <<prefix() <<llvm_lvalue(t2) <<" = inttoptr "
501  <<llvm_integer_type(t1->nBits()) <<" " <<llvm_term(t1) <<" to void()*\n";
502  o <<prefix() <<"call void " <<llvm_term(t2) <<"()\n";
503  o <<prefix() <<"ret void\n";
504  return;
505  }
506  }
507 
508  // If we don't know the target address and this is an indirect function call, then the successors are the entry points of
509  // all known functions. FIXME[Robb P. Matzke 2014-01-09]: Detection of a function call is architecture dependent.
510  if (SgAsmX86Instruction *insn_x86 = isSgAsmX86Instruction(latest_insn)) {
511  if (insn_x86->get_kind() == x86_call || insn_x86->get_kind() == x86_farcall) {
512  LeafPtr t1 = emit_expression(o, eip);
513  LeafPtr t2 = next_temporary(32); // pointer to the function
514  std::string ret_label = addr_label(latest_insn->get_address() + latest_insn->get_size());
515  o <<prefix() <<llvm_lvalue(t2) <<" = inttoptr "
516  <<llvm_integer_type(t1->nBits()) <<" " <<llvm_term(t1) <<" to void()*\n";
517  o <<prefix() <<"call void " <<llvm_term(t2) <<"()\n";
518  o <<prefix() <<"br label %" <<ret_label <<"\n";
519  return;
520  }
521  }
522 
523  // Catch-all: this must be an intra-function indirect branch. LLVM requires us to enumerate all possible targets, but
524  // since we don't actually know them we must enumerate all basic blocks of the function. Note that LLVM 2.5 does not have
525  // the "indirectbr" instruction, so we need to use the "switch" instruction.
526  {
527  LeafPtr t1 = emit_expression(o, eip);
528  std::string type = llvm_integer_type(t1->nBits());
529  std::string dflt_label = next_label();
530  o <<prefix() <<"switch " <<type <<" " <<llvm_term(t1) <<", label %" <<dflt_label <<" [";
531  for (InstructionMap::const_iterator ii=insns.begin(); ii!=insns.end(); ++ii) {
532  if (AST::Traversal::findParentTyped<SgAsmFunction>(ii->second)==func && ii->second->isFirstInBlock())
533  o <<" " <<type <<" " <<ii->first <<", label %" <<addr_label(ii->first);
534  }
535  o <<" ]\n";
536  {
537  Indent label_undent(this, -1);
538  o <<prefix() <<dflt_label <<":\n";
539  }
540  o <<prefix() <<"unreachable\n";
541  return;
542  }

985 ExpressionPtr
986 RiscOperators::emit_signed_multiply(std::ostream &o, const TreeNodes &operands)
987 {
988  ASSERT_require(!operands.empty());
989 
990  size_t result_width = 0;
991  for (size_t i=0; i<operands.size(); ++i)
992  result_width += operands[i]->nBits();
993 
994  ExpressionPtr result = operands[0];
995  for (size_t i=1; i<operands.size(); ++i) {
996  ExpressionPtr t1 = emit_sign_extend(o, result, result_width);
997  ExpressionPtr t2 = emit_sign_extend(o, operands[i], result_width);
998  result = emit_binary(o, "mul", t1, t2);
999  }
1000 
1001  return result;
1002 }

1026 // Rotate the bits of "value" by "amount" bits. LLVM doesn't have a rotate instruction so we use left and right shifting and
1027 // bitwise OR. If called like emit_rotate_left(o, i32 %1, i32 %2) the output will be:
1028 // %3 = shl i32 %1, %2 ; result high bits
1029 // %4 = sub i32 32, %2 ; right shift amount
1030 // %5 = lshr i32 %1, %4 ; result low bits
1031 // %6 = or i32 %3, %5 ; result
1032 ExpressionPtr
1033 RiscOperators::emit_rotate_left(std::ostream &o, const ExpressionPtr &value, const ExpressionPtr &amount)
1034 {

1036  ExpressionPtr t4 = emit_unsigned_binary(o, "sub",
1037  SymbolicExpression::makeIntegerConstant(amount->nBits(), 32),
1038  amount);

1040  ExpressionPtr t6 = emit_unsigned_binary(o, "or", t3, t5);

1164 // Emits LLVM for an expression and returns a terminal (variable or constant). If the expression is already a terminal then
1165 // this method is a no-op.
1166 LeafPtr
1167 RiscOperators::emit_expression(std::ostream &o, const ExpressionPtr &orig_expr)
1168 {
1169  ASSERT_not_null(orig_expr);
1170  ExpressionPtr cur_expr = orig_expr;
1171 
1172  // If we've seen this expression already, then use the replacement value (an LLVM variable).
1173  if (rewrites.exists(cur_expr->hash()))
1174  cur_expr = rewrites.get_one(cur_expr->hash());
1175 
1176  // Handle leaf nodes
1177  if (LeafPtr leaf = cur_expr->isLeafNode()) {
1178  if (leaf->isVariable2()) {
1179  std::string varname = get_variable(leaf);
1180  if (varname.empty()) {
1181  // This is a reference to a ROSE variable that has no corresponding LLVM variable. This can happen for things
1182  // like the x86 TEST instruction, which leaves the AF register in an undefined state--if we then try to print
1183  // the definition of AF we will encounter a ROSE variable with no LLVM variable.
1184  LeafPtr t1 = next_temporary(leaf->nBits());
1185  o <<prefix() <<llvm_lvalue(t1) <<" = add " <<llvm_integer_type(leaf->nBits()) <<" undef, undef\n";
1186  cur_expr = t1;
1187  } else if ('@'==varname[0]) {
1188  cur_expr = emit_global_read(o, varname, leaf->nBits());
1189  }
1190  }
1191  }
1192 
1193  // Emit LLVM for symbolic operators until the result is a leaf node (LLVM variable or constant). This causes recursive
1194  // calls to emit_expression().
1195  while (InteriorPtr inode = cur_expr->isInteriorNode()) {
1196  ExpressionPtr operator_result;
1197  TreeNodes operands = inode->children();
1198  switch (inode->getOperator()) {
1199  case SymbolicExpression::OP_NONE:
1200  ASSERT_not_reachable("cannot happen for an interior node");
1201  case SymbolicExpression::OP_ADD:
1202  operator_result = emit_left_associative(o, "add", operands);
1203  break;
1204  case SymbolicExpression::OP_AND:
1205  operator_result = emit_left_associative(o, "and", operands);
1206  break;
1207  case SymbolicExpression::OP_ASR:
1208  ASSERT_require(2==operands.size());
1209  operator_result = emit_arithmetic_right_shift(o, operands[1], operands[0]);
1210  break;
1211  case SymbolicExpression::OP_XOR:
1212  operator_result = emit_left_associative(o, "xor", operands);
1213  break;
1214  case SymbolicExpression::OP_CONCAT:
1215  operator_result = emit_concat(o, operands);
1216  break;
1217  case SymbolicExpression::OP_EQ:
1218  ASSERT_require(2==operands.size());
1219  operator_result = emit_compare(o, "icmp eq", operands[0], operands[1]);
1220  break;
1221  case SymbolicExpression::OP_EXTRACT:
1222  ASSERT_require(3==operands.size());
1223  operator_result = emit_extract(o, operands[2], operands[0], inode->nBits());
1224  break;
1225  case SymbolicExpression::OP_INVERT:
1226  ASSERT_require(1==operands.size());
1227  operator_result = emit_invert(o, operands[0]);
1228  break;
1229  case SymbolicExpression::OP_ITE:
1230  ASSERT_require(3==operands.size());
1231  operator_result = emit_ite(o, operands[0], operands[1], operands[2]);
1232  break;
1233  case SymbolicExpression::OP_LSSB:
1234  ASSERT_require(1==operands.size());
1235  operator_result = emit_lssb(o, operands[0]);
1236  break;
1237  case SymbolicExpression::OP_MSSB:
1238  ASSERT_require(1==operands.size());
1239  operator_result = emit_mssb(o, operands[0]);
1240  break;
1241  case SymbolicExpression::OP_NE:
1242  ASSERT_require(2==operands.size());
1243  operator_result = emit_compare(o, "icmp ne", operands[0], operands[1]);
1244  break;
1245  case SymbolicExpression::OP_NEGATE:
1246  ASSERT_require(1==operands.size());
1247  operator_result = emit_binary(o, "sub", SymbolicExpression::makeIntegerConstant(operands[0]->nBits(), 0), operands[0]);
1248  break;
1249  case SymbolicExpression::OP_OR:
1250  operator_result = emit_left_associative(o, "or", operands);
1251  break;
1252  case SymbolicExpression::OP_READ:
1253  ASSERT_require(2==operands.size());
1254  operator_result = emit_memory_read(o, operands[1], inode->nBits());
1255  break;
1256  case SymbolicExpression::OP_ROL:
1257  ASSERT_require(2==operands.size());
1258  operator_result = emit_rotate_left(o, operands[1], operands[0]);
1259  break;
1260  case SymbolicExpression::OP_ROR:
1261  ASSERT_require(2==operands.size());
1262  operator_result = emit_rotate_right(o, operands[1], operands[0]);
1263  break;
1264  case SymbolicExpression::OP_SDIV:
1265  ASSERT_require(2==operands.size());
1266  operator_result = emit_signed_divide(o, operands[0], operands[1]);
1267  break;
1268  case SymbolicExpression::OP_SEXTEND:
1269  ASSERT_require(2==operands.size());
1270  operator_result = emit_sign_extend(o, operands[1], inode->nBits());
1271  break;
1272  case SymbolicExpression::OP_SGE:
1273  ASSERT_require(2==operands.size());
1274  operator_result = emit_compare(o, "icmp sge", operands[0], operands[1]);
1275  break;
1276  case SymbolicExpression::OP_SGT:
1277  ASSERT_require(2==operands.size());
1278  operator_result = emit_compare(o, "icmp sgt", operands[0], operands[1]);
1279  break;
1280  case SymbolicExpression::OP_SHL0:
1281  ASSERT_require(2==operands.size());
1282  operator_result = emit_left_shift(o, operands[1], operands[0]);
1283  break;
1284  case SymbolicExpression::OP_SHL1:
1285  ASSERT_require(2==operands.size());
1286  operator_result = emit_left_shift_ones(o, operands[1], operands[0]);
1287  break;
1288  case SymbolicExpression::OP_SHR0:
1289  ASSERT_require(2==operands.size());
1290  operator_result = emit_logical_right_shift(o, operands[1], operands[0]);
1291  break;
1292  case SymbolicExpression::OP_SHR1:
1293  ASSERT_require(2==operands.size());
1294  operator_result = emit_logical_right_shift_ones(o, operands[1], operands[0]);
1295  break;
1296  case SymbolicExpression::OP_SLE:
1297  ASSERT_require(2==operands.size());
1298  operator_result = emit_compare(o, "icmp sle", operands[0], operands[1]);
1299  break;
1300  case SymbolicExpression::OP_SLT:
1301  ASSERT_require(2==operands.size());
1302  operator_result = emit_compare(o, "icmp slt", operands[0], operands[1]);
1303  break;
1304  case SymbolicExpression::OP_SMOD:
1305  ASSERT_require(2==operands.size());
1306  operator_result = emit_signed_modulo(o, operands[0], operands[1]);
1307  break;
1308  case SymbolicExpression::OP_SMUL:
1309  operator_result = emit_signed_multiply(o, operands);
1310  break;
1311  case SymbolicExpression::OP_UDIV:
1312  ASSERT_require(2==operands.size());

1314  break;
1315  case SymbolicExpression::OP_UEXTEND:

1320  ASSERT_require(2==operands.size());

1322  break;
1323  case SymbolicExpression::OP_UGT:
1324  ASSERT_require(2==operands.size());
1325  operator_result = emit_compare(o, "icmp ugt", operands[0], operands[1]);
1326  break;
1327  case SymbolicExpression::OP_ULE:
1328  ASSERT_require(2==operands.size());
1329  operator_result = emit_compare(o, "icmp ule", operands[0], operands[1]);
1330  break;
1331  case SymbolicExpression::OP_ULT:
1332  ASSERT_require(2==operands.size());
1333  operator_result = emit_compare(o, "icmp ult", operands[0], operands[1]);
1334  break;
1335  case SymbolicExpression::OP_UMOD:
1336  ASSERT_require(2==operands.size());
1337  operator_result = emit_unsigned_modulo(o, operands[0], operands[1]);
1338  break;
1339  case SymbolicExpression::OP_UMUL:
1340  operator_result = emit_unsigned_multiply(o, operands);
1341  break;
1342  case SymbolicExpression::OP_ZEROP:
1343  ASSERT_require(1==operands.size());
1344  operator_result = emit_compare(o, "icmp eq", operands[0],
1345  SymbolicExpression::makeIntegerConstant(operands[0]->nBits(), 0));

1350  case SymbolicExpression::OP_WRITE:
1351  case SymbolicExpression::OP_SET:
1352  case SymbolicExpression::OP_FP_ABS:
1353  case SymbolicExpression::OP_FP_NEGATE:
1354  case SymbolicExpression::OP_FP_ADD:
1355  case SymbolicExpression::OP_FP_MUL:
1356  case SymbolicExpression::OP_FP_DIV:
1357  case SymbolicExpression::OP_FP_MULADD:
1358  case SymbolicExpression::OP_FP_SQRT:
1359  case SymbolicExpression::OP_FP_MOD:
1360  case SymbolicExpression::OP_FP_ROUND:
1361  case SymbolicExpression::OP_FP_MIN:
1362  case SymbolicExpression::OP_FP_MAX:
1363  case SymbolicExpression::OP_FP_LE:
1364  case SymbolicExpression::OP_FP_LT:
1365  case SymbolicExpression::OP_FP_GE:
1366  case SymbolicExpression::OP_FP_GT:
1367  case SymbolicExpression::OP_FP_EQ:
1368  case SymbolicExpression::OP_FP_ISNORM:
1369  case SymbolicExpression::OP_FP_ISSUBNORM:
1370  case SymbolicExpression::OP_FP_ISZERO:
1371  case SymbolicExpression::OP_FP_ISINFINITE:
1372  case SymbolicExpression::OP_FP_ISNAN:
1373  case SymbolicExpression::OP_FP_ISNEG:
1374  case SymbolicExpression::OP_FP_ISPOS:

1376  case SymbolicExpression::OP_REINTERPRET:

1382  }
1383  ASSERT_not_null(operator_result); // no case was executed

1385  }
1386

1393  add_rewrite(orig_expr, retval);
