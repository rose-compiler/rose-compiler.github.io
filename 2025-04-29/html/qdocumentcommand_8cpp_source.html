360 {

362 
363  foreach ( QDocumentCursorHandle *ch, m_autoUpdated )
364  {
365  if ( ch == m_cursor || ch->document() != m_doc )
366  continue;
367 
368  //printf("[[watch:0x%x(%i, %i)]]", ch, ch->m_begLine, ch->m_begOffset);
369 
370  // TODO : better selection handling
371  if ( ch->hasSelection() )
372  {
373  int lbeg = line, cbeg = column, lend = line, cend = column;
374 
375  ch->intersectBoundaries(lbeg, cbeg, lend, cend);
376 
377  if ( lbeg == line && cbeg == column )
