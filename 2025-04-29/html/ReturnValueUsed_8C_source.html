302  ASSERT_not_null(partitioner);
303  std::vector<P2::Function::Ptr> callees;
304  for (const P2::ControlFlowGraph::Edge &edge: callSite->outEdges()) {
305  if (edge.value().type() == P2::E_FUNCTION_CALL) {
306  for (const P2::Function::Ptr &f: partitioner->functionsOwningBasicBlock(edge.target()))
307  P2::insertUnique(callees, f, P2::sortFunctionsByAddress);
308  }

324 Analysis::analyzeCallSite(const P2::Partitioner::ConstPtr &partitioner, const P2::ControlFlowGraph::ConstVertexIterator &callSite) {
325  ASSERT_not_null(partitioner);
326  ASSERT_require(partitioner->cfg().isValidVertex(callSite));
327  CallSiteResults retval;
328 
329  SAWYER_MESG(mlog[DEBUG]) <<"analyzing " <<partitioner->vertexName(callSite) <<"\n";
330 
331  // Find callers, callees, and return points in the CFG
332  std::vector<P2::Function::Ptr> callers = partitioner->functionsOwningBasicBlock(callSite);
333  retval.callees(findCallees(partitioner, callSite));
334  std::vector<P2::ControlFlowGraph::ConstVertexIterator> returnTargets = findReturnTargets(partitioner, callSite);
335 
336  // FIXME[Robb P Matzke 2017-03-03]: To simplify things for now, handle only the case where the call site is owned by a
337  // single function, a single callee is called, and it returns to a single point in the caller.
338  if (callers.size() > 1) {
339  mlog[ERROR] <<"multiple callers not implemented yet at vertex " <<partitioner->vertexName(callSite) <<"\n";
340  return retval;
341  }
342  if (retval.callees().size() > 1) {
343  mlog[ERROR] <<retval.callees().size() <<"-call not implemented yet at vertex " <<partitioner->vertexName(callSite) <<"\n";
344  return retval;
345  }
346  if (returnTargets.size() > 1) {
347  mlog[ERROR] <<returnTargets.size() <<"-return not implemented yet at vertex " <<partitioner->vertexName(callSite) <<"\n";
348  return retval;
349  }
350 
351  // Handle the no-op cases.
352  if (callers.empty()) {
353  mlog[WARN] <<"no caller at vertex " <<partitioner->vertexName(callSite) <<"\n";
354  return retval;
355  }
356  if (retval.callees().empty()) {
357  mlog[WARN] <<"no callee at vertex " <<partitioner->vertexName(callSite) <<"\n";
358  return retval;
359  }
360  if (returnTargets.empty()) {
361  mlog[WARN] <<"no return target at vertex " <<partitioner->vertexName(callSite) <<"\n";
362  return retval;
363  }
364 
365  // Simplified version assumes one caller owning the call site, one callee, and one return target.
366  P2::Function::Ptr caller = callers[0];
367  P2::Function::Ptr callee = retval.callees()[0];
368  P2::ControlFlowGraph::ConstVertexIterator returnTarget = returnTargets[0];
369 
370  // Get cached calling convention properties for the callee; avoid doing a calling convention analysis here -- the user
371  // should have already done that.
372  const CallingConvention::Analysis &calleeBehavior = callee->callingConventionAnalysis();
373  if (!calleeBehavior.hasResults() || !calleeBehavior.didConverge()) {
374  mlog[WARN] <<"no calling convention behavior for " <<callee->printableName()
375  <<" called at vertex " <<partitioner->vertexName(callSite) <<"\n";
376  }
377  const CallingConvention::Definition::Ptr calleeDefinition = callee->callingConventionDefinition();
378  if (!calleeDefinition) {
379  mlog[ERROR] <<"no calling convention definition for " <<callee->printableName()
380  <<" called at vertex " <<partitioner->vertexName(callSite) <<"\n";
381  return retval;
382  }
383 
384  // Find the intersection of the calling convention definition's return value locations and the outputs based on callee
385  // behavior. This takes care of two issues: (1) some behavior-based outputs are only scratch locations according to the
386  // definition, and (2) some return locations according to the definition might not have been outputs according to the
387  // callee behavior.
388  RegisterParts calleeReturnRegs = calleeDefinition->outputRegisterParts();
389  if (calleeBehavior.hasResults() && calleeBehavior.didConverge())
390  calleeReturnRegs &= calleeBehavior.outputRegisters();
391  calleeReturnRegs -= RegisterParts(partitioner->instructionProvider().stackPointerRegister());
392  Variables::StackVariables calleeReturnMem;
393 #if 0 // [Robb Matzke 2019-08-14]: turning off warning
394  for (const ConcreteLocation &location: calleeDefinition->outputParameters()) {
395  // FIXME[Robb P Matzke 2017-03-20]: todo
396  }
397 #endif
398 
399  // Build the instruction semantics that will look for which of the callee's return values are used by the caller. "Used"
400  // means (1) the caller reads the callee output location without first writing to it, or (2) the caller calls a second
401  // function whose input is one of the original callee outputs with no intervening write, or (3) one of the caller's own
402  // return values is one of the calle's return values with no intervening write.
403  RegisterDictionary::Ptr regdict = partitioner->instructionProvider().registerDictionary();
404  ASSERT_not_null(regdict);
405  SmtSolverPtr solver = SmtSolver::instance(Rose::CommandLine::genericSwitchArgs.smtSolver);
406  RiscOperators::Ptr ops = RiscOperators::instance(regdict, solver);
407  ops->registerDictionary(regdict);
408  ops->insertOutputs(calleeReturnRegs, calleeReturnMem);
409  if (!ops->hasOutputs())
410  return retval;

412  if (!cpu) {
413  mlog[ERROR] <<"no instruction semantics for this architecture\n";
414  return retval;
415  }
416

418  P2::DataFlow::DfCfg dfCfg = P2::DataFlow::buildDfCfg(partitioner, partitioner->cfg(), returnTarget);
419  if (mlog[DEBUG]) {
420  boost::filesystem::path debugDir = "./rose-debug/BinaryAnalysis/ReturnValueUsed";
421  boost::filesystem::create_directories(debugDir);
422  boost::filesystem::path fileName = debugDir /
423  ("B_" + StringUtility::addrToString(callSite->value().address()).substr(2) + ".dot");
424  std::ofstream f(fileName.string().c_str());

431  P2::DataFlow::TransferFunction xfer(cpu);
432  xfer.defaultCallingConvention(defaultCallingConvention_);
433

435  // traversal.
436  std::vector<State::Ptr> inputStates(dfCfg.nVertices());
437  inputStates[0] = xfer.initialState();

440  ops->callSiteResults(&retval);

442  for (Traversal t(dfCfg, dfCfg.findVertex(0)); t; ++t) {

474  <<" unused return locations: " <<locationNames(retval.returnRegistersUnused(), regdict) <<"\n"

479  for (const P2::DataFlow::DfCfg::Edge &edge: t.vertex()->outEdges()) {

486  // return value by virtue of the caller returning it. For example:
487  // callee: push ebp
488  // mov ebp, esp
489  // mov eax, 1 ; the return value

492  //

503  if (dfCfg.isValidVertex(returnVertex)) {
504  State::Ptr returnState = inputStates[returnVertex->id()];
505  ASSERT_always_not_null(returnState);
506  ops->currentState(returnState);
507  for (const P2::Function::Ptr &caller: callers) {

509  if (callerBehavior.didConverge()) {

511  <<locationNames(callerBehavior.outputRegisters(), regdict) <<"\n";
512  for (RegisterDescriptor reg: callerBehavior.outputRegisters().listAll(regdict))
513  (void) ops->readRegister(reg, ops->undefined_(reg.nBits()));
514  }
515  }
516  }
517  }
518

520  <<locationNames(retval.returnRegistersUnused(), regdict) <<"\n";
521 
522  // Assume all unresolved return locations of the callee(s) are unused return values.
523  retval.returnRegistersUnused() |= ops->unreferencedRegisterOutputs();
524 
525  retval.didConverge(true);
526  return retval;
527 }
