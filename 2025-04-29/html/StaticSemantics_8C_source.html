31 void attachInstructionSemantics(SgNode *ast, const Architecture::Base::ConstPtr &arch) {
32  ASSERT_not_null(ast);
33  ASSERT_not_null(arch);
34  RiscOperators::Ptr ops = RiscOperators::instanceFromProtoval(SValue::instance(), SmtSolver::Ptr());
35  BaseSemantics::Dispatcher::Ptr cpu = arch->newInstructionDispatcher(ops);
36  if (!cpu)
37  throw BaseSemantics::Exception("no instruction semantics for architecture", NULL);
38  attachInstructionSemantics(ast, cpu);

41 void attachInstructionSemantics(SgNode *ast, const BaseSemantics::Dispatcher::Ptr &cpu) {
42  using namespace Rose::AST::Traversal;
43  std::vector<SgAsmInstruction*> insns;
44  forwardPre<SgAsmInstruction>(ast, [&insns](SgAsmInstruction *node) {
45  insns.push_back(node);
46  });
47 
48  for (SgAsmInstruction *insn: insns) {
49  if (insn->get_semantics()==NULL) {
50  try {
51  // Side effect is that semanics are attached to the instruction.
52  cpu->processInstruction(insn);
53  } catch (const BaseSemantics::Exception&) {
54  // FIXME[Robb P. Matzke 2015-06-06]: semantics subtree should be deleted, but delete mechanism seems to be
55  // broken.
56  if (SgAsmNode *tmp = insn->get_semantics()) {
57  insn->set_semantics(nullptr);
58  tmp->set_parent(nullptr);
59  }
60  }
61  }
62  }
63 }

182 // Save the semantic effect in the current instruction.
183 void
184 RiscOperators::saveSemanticEffect(const BaseSemantics::SValue::Ptr &a_) {
185  if (SgAsmInstruction *insn = currentInstruction()) {
186  SValue::Ptr a = SValue::promote(a_);
187  ASSERT_not_null(a->ast());
188 
189  // Ensure that the instruction has a place to store semantics
190  SgAsmExprListExp *semantics = insn->get_semantics();
191  if (NULL == semantics) {
192  semantics = new SgAsmExprListExp;
193  insn->set_semantics(semantics);
194  semantics->set_parent(insn);
195  }
196 
197  // We're about to make a copy of the a->ast(), so we need to ensure all the parent pointers are correct.
198  AST::Utility::repairParentPointers(a->ast());
199  AST::Utility::checkParentPointers(a->ast());
200 
201  // Deep-copy the SValue's AST because we're about to link it into the real AST.
202  SgTreeCopy deep;
203  SgAsmExpression *copied = isSgAsmExpression(a->ast()->copy(deep));
204  ASSERT_not_null(copied);
205  semantics->get_expressions().push_back(copied);
206  copied->set_parent(semantics);
207  }
208 }

377 void
378 RiscOperators::startInstruction(SgAsmInstruction *insn) {
379  // Make sure any previous semantics are erased before we start.
380  if (SgAsmExprListExp *semantics = insn->get_semantics()) {
381  for (size_t i = 0; i < semantics->get_expressions().size(); ++i) {
382  if (SgNode *tmp = semantics->get_expressions()[i]) {
383  // FIXME[Robb P. Matzke 2015-06-06]: delete subtree mechanism is broken; this is a memory leak
384  semantics->get_expressions()[i] = nullptr;
385  tmp->set_parent(nullptr);
386  }
387  }
388  semantics->get_expressions().clear();
389  }
390  BaseSemantics::RiscOperators::startInstruction(insn);
391 }
