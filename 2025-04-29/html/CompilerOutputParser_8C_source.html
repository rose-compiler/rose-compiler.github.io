106 void CompilerOutputParser::parseIncludedFilesFromCompilerOutput(const string& includingFileName, const string& parsePrefix) {
107  ROSE_ASSERT(FileHelper::fileExists(includingFileName)); //should be an existing file
108 
109  size_t parsePrefixSize = parsePrefix.size();
110  string oneLevelDeeperParsePrefix = "." + parsePrefix;
111  string lastIncludedFile;
112  set<string> includedFiles;
113 
114  while (compilerOutputReader -> hasLine()) {
115  string currentLine = compilerOutputReader -> getLine();
116  if (currentLine.compare(0, parsePrefixSize, parsePrefix) == 0) {
117  //The current line belongs to this including file.
118  lastIncludedFile = FileHelper::makeAbsoluteNormalizedPath(currentLine.substr(parsePrefixSize), workingDirectory);
119  includedFiles.insert(lastIncludedFile);
120  compilerOutputReader -> readNextLine(); //read the next line, because the current line was processed
121  } else {
122  if (currentLine.compare(0, oneLevelDeeperParsePrefix.size(), oneLevelDeeperParsePrefix) == 0) {
123  //The current line belongs to the last processed included file.
124  parseIncludedFilesFromCompilerOutput(lastIncludedFile, oneLevelDeeperParsePrefix);
125  } else {
126  //The current line is neither of this including file nor of the last processed included file,
127  //so stop and let the caller process it (and the following lines, if any), unless this is the top level.
128  if (parsePrefix.compare(topLevelParsePrefix) != 0) {
129  break;
130  } else {
131  compilerOutputReader -> readNextLine(); //read the next line, because the current line was skipped
132  }
133  }
134  }
135  }
136  addIncludedFilesToMap(includingFileName, includedFiles);
137 }

139 void CompilerOutputParser::parseIncludedFilesSearchPathsFromCompilerOutput() {
140  const string quotedIncludesStart = "#include \"...\" search starts here:";
141  const string bracketedIncludesStart = "#include <...> search starts here:";
142  const string includesEnd = "End of search list.";
143  bool isInsideQuotedIncludes = false;
144  bool isInsideBracketedIncludes = false;
145 
146  while (compilerOutputReader -> hasLine()) {
147  string currentLine = compilerOutputReader -> getLine();
148  compilerOutputReader -> readNextLine();
149  if (currentLine.compare(quotedIncludesStart) == 0) {
150  isInsideQuotedIncludes = true;
151  continue;
152  }
153  if (currentLine.compare(bracketedIncludesStart) == 0) {
154  isInsideQuotedIncludes = false;
155  isInsideBracketedIncludes = true;
156  continue;
157  }
158  if (currentLine.compare(includesEnd) == 0) {
159  //done
160  return;
161  }
162 
163  if (isInsideQuotedIncludes || isInsideBracketedIncludes) {
164  string path = FileHelper::makeAbsoluteNormalizedPath(/*the first character is a white space, so ignore it*/currentLine.substr(1), workingDirectory);
165  if (isInsideQuotedIncludes) {
166  quotedIncludesSearchPaths.push_back(path);
167  } else {
168  bracketedIncludesSearchPaths.push_back(path);
169  }
170  }
171  }
