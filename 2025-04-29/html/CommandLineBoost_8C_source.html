27 Switch
28 createSwitch(const std::string &switchNames, const std::string &doc) {
29  // Split comma-separated list of switch names into long names and short names.
30  std::vector<std::string> longNames;
31  boost::split(longNames, switchNames, boost::is_any_of(","));
32  std::vector<std::string> shortNames = longNames;
33  longNames.erase(std::remove_if(longNames.begin(), longNames.end(), isShortName), longNames.end());
34  shortNames.erase(std::remove_if(shortNames.begin(), shortNames.end(), isLongName), shortNames.end());
35 
36  std::vector<std::string>::iterator longIter = longNames.begin();
37  std::vector<std::string>::iterator shortIter = shortNames.begin();
38 
39  // Use the first long or short name to create the switch
40  ASSERT_require(!longNames.empty() || !shortNames.empty());
41  Switch retval = longNames.empty() ? Switch("", (*shortIter++)[0]) : Switch(*longIter++);
42 
43  // Add the remaining long and short names
44  while (longIter != longNames.end())
45  retval.longName(*longIter++);
46  while (shortIter != shortNames.end())
47  retval.shortName((*shortIter++)[0]);
48 
49  retval.doc(doc);
50  return retval;
51 }
