177 SgAsmIntegerValueExpression::makeRelativeTo(SgNode *baseNode)
178 {
179  ASSERT_require(get_significantBits() <= 8*sizeof(uint64_t));
180  ASSERT_require(get_significantBits() > 0);
181  int64_t curValue = get_absoluteValue();
182  int64_t baseVa = virtualAddress(baseNode);
183  int64_t offset = curValue - baseVa;
184  set_baseNode(baseNode);
185 
186  // We don't want to change the size of the offset if we can help it. But if we do need to widen it then use a power of two
187  // bytes: 1, 2, 4, or 8
188  size_t needWidth = 0;
189  if (offset >= 0) {
190  if ((uint64_t)offset > 0xffffffff) {
191  needWidth = 64;
192  } else if ((uint64_t)offset > 0xffff) {
193  needWidth = 32;
194  } else if ((uint64_t)offset > 0xff) {
195  needWidth = 16;
196  } else {
197  needWidth = 8;
198  }
199  } else {
200  if (offset < -4294967296ll) {
201  needWidth = 64;
202  } else if (offset < -65536) {
203  needWidth = 32;
204  } else if (offset < -256) {
205  needWidth = 16;
206  } else {
207  needWidth = 8;
208  }
209  }
210 
211  size_t newWidth = std::max(get_significantBits(), needWidth);
212  ASSERT_require(newWidth>0);
213  uint64_t uoffset = (uint64_t)offset & IntegerOps::genMask<uint64_t>(newWidth);
214  set_relativeValue(uoffset, newWidth);
215 }
