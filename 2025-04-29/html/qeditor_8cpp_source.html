277 void QEditor::setDefaultFlags(int flags)
278 {
279  m_defaultFlags = flags & Accessible;
280 
281  foreach ( QEditor *e, m_editors )
282  {
283  bool ontoWrap = (m_defaultFlags & LineWrap) && !(e->m_state & LineWrap);
284  bool outOfWrap = !(m_defaultFlags & LineWrap) && (e->m_state & LineWrap);
285 
286  e->m_state &= Internal;
287  e->m_state |= m_defaultFlags;
288 
289  if ( ontoWrap )
290  {
291  e->document()->setWidthConstraint(e->wrapWidth());
292  } else if ( outOfWrap ) {
293  e->document()->clearWidthConstraint();
294  }
295 
296  QAction *a = e->m_actions.value("wrap");
297 
298  if ( a && (a->isChecked() != (bool)(e->m_state & LineWrap)) )
299  a->setChecked(e->m_state & LineWrap);
300 
301  }

793  QAction *a = m_actions.value("wrap");
794 
795  if ( a && !a->isChecked() )
796  a->setChecked(true);
797  }
798  } else {

807 
808  QAction *a = m_actions.value("wrap");
809 
810  if ( a && a->isChecked() )
811  a->setChecked(false);
812  }
813  }
814 
815  // TODO : only update cpos if cursor used to be visible?
816  if ( f == LineWrap )
817  ensureCursorVisible();

889 void QEditor::save()
890 {
891  if ( !m_doc )
892  return;
893 
894  QString oldFileName = fileName();
895 
896  if ( fileName().isEmpty() )
897  {
898  QString fn = QFileDialog::getSaveFileName();
899 
900  if ( fn.isEmpty() )
901  return;
902 
903  setFileName(fn);
904  } else if ( isInConflict() ) {
905  int ret = QMessageBox::warning(this,
906  tr("Conflict!"),
907  tr(
908  "%1\nhas been modified by another application.\n"
909  "Press \"Save\" to overwrite the file on disk\n"
910  "Press \"Reset\"to be reload the file from disk.\n"
911  "Press \"Discard\" to ignore this warning.\n"
912  ).arg(fileName()),
913  QMessageBox::Save
914  |
915  QMessageBox::Reset
916  |
917  QMessageBox::Discard
918  |
919  QMessageBox::Cancel
920  );
921  if ( ret == QMessageBox::Save )
922  {
923  m_saveState = Undefined;
924  } else if ( ret == QMessageBox::Reset ) {
925  load(fileName());
