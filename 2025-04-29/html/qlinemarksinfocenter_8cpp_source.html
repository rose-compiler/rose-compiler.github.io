335 void QLineMarksInfoCenter::loadMarkTypes(const QString& f)
336 {
337  QFile file(f);
338 
339  if ( !file.open(QFile::ReadOnly | QFile::Text) )
340  {
341  qDebug() << "Failed to load MarkTypes" << f;
342  return;
343  }
344  // TODO : prefer QXmlStreamReader when building against Qt 4.3.0
345 
346  QDomDocument doc;
347  doc.setContent(&file);
348 
349  QDomNodeList l = doc.documentElement().childNodes();
350 
351  for ( int i = 0; i < l.count(); i++ )
352  {
353  QDomElement e = l.at(i).toElement();
354 
355  if ( e.isNull() || (e.tagName() != "mark") )
356  continue;
357 
358  QLineMarkType t;
359  QDomNodeList c = e.childNodes();
360 
361  //qDebug("mark {");
362 
363  for ( int j = 0; j < c.count(); j++ )
364  {
365  QDomElement attr = c.at(j).toElement();
366 
367  if ( attr.isNull() )
368  continue;
369 
370  const QString field = attr.tagName();
371  const QString value = attr.firstChild().toText().data();
372 
373  //qDebug("\t%s = %s;", qPrintable(field), qPrintable(value));
374 
375  const bool flag = (value == "true") || value.toUInt();
376 
377  if ( field == "id" )
378  {
379  t.id = value;
380  } else if ( field == "user" ) {
381  t.user = flag;
382  } else if ( field == "focus" ) {
383  t.focus = flag;
384  } else if ( field == "icon" ) {
385  t.icon = QPixmap(value);
386  } else if ( field == "color" ) {
387  //t.color = QColor(value);
388 
389  /*
390  color value MUST be a valid value for QColor::setNamedColor()
391  with one exception though : an alpha channel indication (unsupported
392  by QColor::setNamedColor()) can be prepended using '@' followed by a
393  sequence of hex digits (preferabily two of them...)
394 
395  examples :
396  #ff0c80
397  #ff0c80@80
398  blue
399  blue@10
400  */
401 
402  if ( value.contains('@') )
403  {
404  t.color = QColor(value.section('@', 0, 0, QString::SectionSkipEmpty));
405  t.color.setAlpha(value.section('@', 1, 1, QString::SectionSkipEmpty).toUInt(0, 16));
406  } else {
407  t.color = QColor(value);
408  }
409  } else if ( field == "priority" ) {
410  t.priority = value.toUInt();
411  } else if ( field == "persistency" ) {
412  t.persistency = value.toUInt();
413  } else if ( field == "rule" ) {
414  t.rules << value;
415  }
416 
417  }
418 
419  m_lineMarkTypes << t;
420 
421  //qDebug("};");
422  }
423 }
