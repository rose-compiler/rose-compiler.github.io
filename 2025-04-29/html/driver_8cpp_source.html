108  }

131 void push_scope(POETCode* s, bool insert)
132 {
133  switch (s->get_enum()) {
134  case SRC_XVAR: {
135  LvarSymbolTable* t = static_cast<XformVar*>(s)->get_entry().get_symTable();
136  if (t == 0) {
137  std::cerr << "Error: xform routine undefined: " << s->toString() << "\nAt line " << yylineno << " of file " << curfile->get_filename() << "\n"; exit(1);
138  }
139  tmpScope.push_front(std::pair<LvarSymbolTable*,bool>(t,insert));
140  break;
141  }
142  case SRC_CVAR: {
143  LvarSymbolTable* t = static_cast<CodeVar*>(s)->get_entry().get_symTable();
144  assert(t != 0);
145  tmpScope.push_front(std::pair<LvarSymbolTable*,bool>(t,insert));
146  break;
147  }
148  default: assert(0);

182 extern "C" POETCode* make_varRef(POETCode* name, int config)
183 {
184  try {
185  ASTFactory* fac = ASTFactory::inst();
186  if (name == fac->make_any()) return name;
187  switch (config) {
188  case CODE_VAR: return curfile->make_codeRef(name, 0);
189  case XFORM_VAR: return curfile->make_xformVar(name);
190  case CODE_OR_XFORM_VAR: {
191  POETCode* res = find_code_or_xform_var(name);
192  if (res != 0) return res;
193  CODE_OR_XFORM_UNDEFINED(name->toString());
194  }
195  case GLOBAL_SCOPE: {
196  POETCode* res = find_code_or_xform_var(name);
197  if (res != 0) return res;
198  res = find_global_var(name);
199  if (res != 0) return res;
200  SYM_UNDEFINED(name->toString());
201  }
202  case GLOBAL_VAR: {
203  POETCode* res = find_code_or_xform_var(name);
204  if (res != 0) return res;
205  res = find_global_var(name);
206  if (res != 0) return res;
207  return curfile->make_evalVar(name);
208  }
209  case ID_DEFAULT: {
210  LvarSymbolTable* local = top_scope();
211  POETCode* res = find_code_or_xform_var(name);
212  if (res != 0) return res;
213  if (local == 0) {
214  res = find_global_var(name);
215  if (res != 0) return res;
216  return curfile->make_evalVar(name);
217  }
218  return POETProgram::make_localVar(local, name, LVAR_REG);
219  }
220  case ASSIGN_VAR: {
221  POETCode* res = find_code_or_xform_var(name);
222  if (res != 0) return res;
223  LvarSymbolTable* local = top_scope();
224  if (local == 0) {
225  res = find_global_var(name);
226  if (res != 0) return res;
227  return curfile->make_evalVar(name);
228  }
229  return POETProgram::make_localVar(local, name, LVAR_REG);
230  }
231  default:
232  std::cerr << "something is wrong with variable creation config: " << name->toString() << " config = " << config << "\n";
233  std::cerr << "At line " << yylineno << " of file " << curfile->get_filename() << "\n";
234  assert(0);
235  }
236  }
237  catch (Error err) { std::cerr << "At line " << yylineno << " of file " << curfile->get_filename() << "\n"; exit(1); }
238 }
