75 bool QEditConfig::hasUnsavedChanges() const
76 {
77  if ( m_direct )
78  return false;
79 
80  QFont font = ui->cbFont->currentFont();
81  //font.setPointSize(spnFontSize->value());
82 
83  const QFont& docFont = QDocument::font();
84 
85  if ( font.family() != docFont.family() || ui->spnFontSize->value() != docFont.pointSize() )
86  {
87  //qDebug("font!");
88  return true;
89  }
90 
91  if ( ui->spnTabWidth->value() != QDocument::tabStop() )
92  {
93  //qDebug("tab stop!");
94  return true;
95  }
96 
97  QDocument::LineEnding le = QDocument::defaultLineEnding();
98 
99  if ( ui->chkDetectLE->isChecked() )
100  {
101  if ( le != QDocument::Conservative )
102  {
103  //qDebug("conservative line endings! : %i", le);
104  return true;
105  }
106  } else {
107  if ( le != QDocument::LineEnding(ui->cbLineEndings->currentIndex() + 1) )
108  {
109  //qDebug("line endings!");
110  return true;
111  }
112  }
113 
114  QDocument::WhiteSpaceMode ws = QDocument::ShowNone;
115 
116  if ( ui->chkShowLeadingWhitespace->isChecked() )
117  ws |= QDocument::ShowLeading;
118 
119  if ( ui->chkShowTrailingWhitespace->isChecked() )
120  ws |= QDocument::ShowTrailing;
121 
122  if ( ui->chkShowTabsInText->isChecked() )
123  ws |= QDocument::ShowTabs;
124 
125  if ( ws != QDocument::showSpaces() )
126  {
127  //qDebug("spaces!");
128  return true;
129  }
130 
131  QTextCodec *c = QEditor::defaultCodec();
132 
133  if ( ui->cbEncoding->currentText() == "System" )
134  {
135  if ( c && c->name() != "System" )
136  {
137  //qDebug("system codec!");
138  return true;
139  }
140  } else {
141  if ( !c || c->name() != ui->cbEncoding->currentText().toLatin1() )
142  {
143  //qDebug("codec!");
144  return true;
145  }
146  }
147 
148  int flags = QEditor::defaultFlags();
149 
150  if ( ui->chkReplaceTabs->isChecked() )
151  flags |= QEditor::ReplaceTabs;
152  else
153  flags &= ~QEditor::ReplaceTabs;
154 
155  if ( ui->chkAutoRemoveTrailingWhitespace->isChecked() )
156  flags |= QEditor::RemoveTrailing;
157  else
158  flags &= ~QEditor::RemoveTrailing;
159 
160  if ( ui->chkPreserveTrailingIndent->isChecked() )
161  flags |= QEditor::PreserveTrailingIndent;
162  else
163  flags &= ~QEditor::PreserveTrailingIndent;
164 
165  if ( flags != QEditor::defaultFlags() )
166  {
167  //qDebug("flags!");
168  return true;
169  }
170 
171  return false;
172 }

203  else
204  QDocument::setDefaultLineEnding(QDocument::LineEnding(ui->cbLineEndings->currentIndex() + 1));
205 
206  QDocument::WhiteSpaceMode ws = QDocument::ShowNone;
207 
208  if ( ui->chkShowLeadingWhitespace->isChecked() )
209  ws |= QDocument::ShowLeading;
210 
211  if ( ui->chkShowTrailingWhitespace->isChecked() )
212  ws |= QDocument::ShowTrailing;
213 
214  if ( ui->chkShowTabsInText->isChecked() )
215  ws |= QDocument::ShowTabs;
216 
217  QDocument::setShowSpaces(ws);
218 
219  if ( ui->cbEncoding->currentText() == "System" )
220  QEditor::setDefaultCodec(0, QEditor::UpdateAll);
221  else
222  QEditor::setDefaultCodec(ui->cbEncoding->currentText().toLatin1(), QEditor::UpdateAll);
223 
224  int flags = QEditor::defaultFlags();
225 
226  if ( ui->chkReplaceTabs->isChecked() )
227  flags |= QEditor::ReplaceTabs;
228  else
229  flags &= ~QEditor::ReplaceTabs;
230 
231  if ( ui->chkAutoRemoveTrailingWhitespace->isChecked() )
232  flags |= QEditor::RemoveTrailing;
233  else
234  flags &= ~QEditor::RemoveTrailing;
235 
236  if ( ui->chkPreserveTrailingIndent->isChecked() )
237  flags |= QEditor::PreserveTrailingIndent;
238  else
239  flags &= ~QEditor::PreserveTrailingIndent;
240 
241  QEditor::setDefaultFlags(flags);

357 void QEditConfig::loadKeys(const QMap<QString, QVariant>& keys)
358 {
359  //qDebug("loading %i keys", keys.count());
360 
361  // load
362  QMap<QString, QVariant>::const_iterator it = keys.constBegin();
363 
364  while ( it != keys.constEnd() )
365  {
366  if ( it.key() == "font" )
367  {
368  QFont f = qvariant_cast<QFont>(*it);
369 
370  ui->cbFont->setCurrentFont(f);
371  ui->spnFontSize->setValue(f.pointSize());
372 
373  if ( m_direct )
374  QDocument::setFont(f);
375 
376  ui->lblSampleText->setFont(f);
377 
378  } else if ( it.key() == "tab_width" ) {
379  ui->spnTabWidth->setValue(it->toInt());
380 
381  if ( m_direct )
382  on_spnTabWidth_valueChanged(it->toInt());
383  } else if ( it.key() == "replace_tabs" ) {
384  ui->chkReplaceTabs->setChecked(it->toBool());
385  if ( m_direct )
386  on_chkReplaceTabs_toggled(it->toBool());
387  } else if ( it.key() == "remove_trailing" ) {
388  ui->chkAutoRemoveTrailingWhitespace->setChecked(it->toBool());
389  if ( m_direct )
390  on_chkAutoRemoveTrailingWhitespace_toggled(it->toBool());
391  } else if ( it.key() == "preserve_trailing_indent" ) {
392  ui->chkPreserveTrailingIndent->setChecked(it->toBool());
393  if ( m_direct )
394  on_chkPreserveTrailingIndent_toggled(it->toBool());
395  } else if ( it.key() == "show_tabs_in_text" ) {
396  ui->chkShowTabsInText->setChecked(it->toBool());
397  if ( m_direct )
398  on_chkShowTabsInText_toggled(it->toBool());
399  } else if ( it.key() == "show_leading_whitespace" ) {
400  ui->chkShowLeadingWhitespace->setChecked(it->toBool());
401  if ( m_direct )
402  on_chkShowLeadingWhitespace_toggled(it->toBool());
403  } else if ( it.key() == "show_trailing_whitespace" ) {
404  ui->chkShowTrailingWhitespace->setChecked(it->toBool());
405  if ( m_direct )
406  on_chkShowTrailingWhitespace_toggled(it->toBool());
407  } else if ( it.key() == "encoding" ) {
408  ui->cbEncoding->setCurrentIndex(ui->cbEncoding->findText(it->toString()));
409  if ( m_direct )
410  on_cbEncoding_currentIndexChanged(it->toString());
411  } else if ( it.key() == "line_endings" ) {
412  int le = it->toInt();
413 
414  if ( le )
415  {
416  ui->chkDetectLE->setChecked(false);
417  ui->cbLineEndings->setCurrentIndex(le - 1);
418  } else {
419  ui->cbLineEndings->setCurrentIndex(0);
420  ui->chkDetectLE->setChecked(true);
421  }
422  } else {
423  /*
424  qWarning("QEditConfig::loadKeys() fed with unsupported settings key : \"%s\" ",
425  qPrintable(it.key())
426  );
427  */
428  }
429 
430  ++it;
431  }
432 }
