31 void ControlFlowGraph::build(SgFunctionDefinition* funcDef)
32 {
33  ROSE_ASSERT(funcDef);
34  funcDef_ = funcDef;
35 
36  // The following two variables are used to record the nodes traversed.
37  nodesToVertices_.clear();
38  std::set<CFGNode> nodesProcessed;
39 
40  // Remove all nodes and edges first.
41  clear();
42  entry_ = GraphTraits::null_vertex();
43  exit_ = GraphTraits::null_vertex();
44 
45  CFGNode entry(funcDef->cfgForBeginning());
46  buildCFG(entry, nodesToVertices_, nodesProcessed);
47 
48  // Find the entry and exit of this CFG.
49  setEntryAndExit();
50 
51  ROSE_ASSERT(isSgFunctionDefinition((*this)[entry_]->getNode()));
52  ROSE_ASSERT(isSgFunctionDefinition((*this)[exit_]->getNode()));
53 }

55 void ControlFlowGraph::setEntryAndExit()
56 {
57  for (Vertex v : boost::vertices(*this))
58  {
59  CFGNodePtr node = (*this)[v];
60  if (isSgFunctionDefinition(node->getNode()))
61  {
62  if (node->getIndex() == 0)
63  entry_ = v;
64  else if (node->getIndex() == 3)
65  exit_ = v;
66  }
67  }
68 
69  //In graphs with an infinite loop, we might never get to the end vertex
70  //In those cases, we need to add it explicitly
71  if (exit_ == GraphTraits::null_vertex())
72  {
73  std::cerr << "This function may contain an infinite loop "
74  "inside so that its CFG cannot be built" << std::endl;
75  exit_ = add_vertex(*this);
76  (*this)[exit_] = newCFGNode(CFGNode(funcDef_->cfgForEnd()));
77  }
78 
79  ROSE_ASSERT(entry_ != GraphTraits::null_vertex());
80  ROSE_ASSERT(exit_ != GraphTraits::null_vertex());
81 }

83 void ControlFlowGraph::buildCFG(
84  const CFGNode& node,
85  std::map<CFGNode, Vertex>& nodesAdded,
86  std::set<CFGNode>& nodesProcessed)
87 {
88  ROSE_ASSERT(node.getNode());
89 
90  if (nodesProcessed.count(node) > 0)
91  return;
92  nodesProcessed.insert(node);
93 
94  std::map<CFGNode, Vertex>::iterator iter;
95  bool inserted;
96  Vertex from, to;
97 
98  // Add the source node.
99  const CFGNode& src = node;
100  ROSE_ASSERT(src.getNode());
101 
102  boost::tie(iter, inserted) = nodesAdded.insert(std::make_pair(src, Vertex()));
103 
104  if (inserted)
105  {
106  from = add_vertex(*this);
107  (*this)[from] = newCFGNode(src);
108  iter->second = from;
109  }
110  else
111  {
112  from = iter->second;
113  }
114 
115  CFGNode nodeCopy = node;
116 #if (BACKEND_CXX_COMPILER_MAJOR_VERSION_NUMBER >= 4)
117 #if (BACKEND_CXX_COMPILER_MINOR_VERSION_NUMBER >= 7)
118 # errror "This line does not compile using g++ versions greater than 4.7"
119 #else
120  nodeCopy.setFilter(filter_);
121  std::vector<CFGEdge> outEdges = nodeCopy.outEdges();
122 
123  for (const CFGEdge& cfgEdge : outEdges)
124  {
125  // For each out edge, add the target node.
126  CFGNode tar = cfgEdge.target();
127  ROSE_ASSERT(tar.getNode());
128 
129  boost::tie(iter, inserted) = nodesAdded.insert(std::make_pair(tar, Vertex()));
130 
131  if (inserted)
132  {
133  to = add_vertex(*this);
134  (*this)[to] = newCFGNode(tar);
135  iter->second = to;
136  }
137  else
138  {
139  to = iter->second;
140  }
141 
142  // Add the edge.
143  Edge edge = add_edge(from, to, *this).first;
144  (*this)[edge] = newCFGEdge(cfgEdge);
145 
146  // Build the CFG recursively.
147  buildCFG(tar, nodesAdded, nodesProcessed);
148  }
149 #endif
150 #endif
151 }

178 ControlFlowGraph ControlFlowGraph::makeReverseCopy() const
179 {
180  ControlFlowGraph reverseCFG;
181  // The following function makes a reverse CFG copy.
182  boost::transpose_graph(*this, reverseCFG,
183  boost::vertex_copy(VertexCopier(*this, reverseCFG)).
184  edge_copy(EdgeCopier(*this, reverseCFG)));
185 
186  // Swap entry and exit.
187  reverseCFG.entry_ = this->exit_;
188  reverseCFG.exit_ = this->entry_;
189  return reverseCFG;

220 std::set<ControlFlowGraph::Edge>
221  ControlFlowGraph::getAllBackEdges() const
222  {
223  std::set<Edge> backEdges;
224 
225  // If the dominator tree is not built yet, build it now.
226  getDominatorTree();
227 
228  for (const Edge& e : boost::edges(*this))
229  {
230  Vertex src = boost::source(e, *this);
231  Vertex tar = boost::target(e, *this);
232 
233  //Vertex v = *(dominatorTree.find(src));
234  VertexVertexMap::const_iterator iter = dominatorTree_.find(src);
235  while (iter != dominatorTree_.end())
236  {
237  if (iter->second == tar)
238  {
239  backEdges.insert(e);
240  break; // break the while loop
241  }
242  iter = dominatorTree_.find(iter->second);
243  }
244  }

246  return backEdges;

270 std::map<ControlFlowGraph::Vertex, ControlFlowGraph::Vertices>
271  ControlFlowGraph::getAllLoops() const
272  {
273  std::set<Edge> backEdges = getAllBackEdges();
274  std::map<Vertex, Vertices> loops;
275 
276  // Build a reverse CFG.
277  ControlFlowGraph rvsCFG = makeReverseCopy();
278  size_t verticesNum = boost::num_vertices(rvsCFG);
279 
280  // For each back edge, make the target (loop header) visited then
281  // do a DFS from the source of this edge on the reverse CFG.
282  for (const Edge& edge : backEdges)
283  {
284  Vertex root = boost::source(edge, rvsCFG);
285  Vertex header = boost::target(edge, rvsCFG);
286  Vertices vertices;
287 
288  // Create a DFS visitor in which we add all nodes to vertices set.
289  DFSVisitor<Vertex> dfsVisitor(vertices);
290  // Build a vector of colors and set all initial colors to white.
291  std::vector<boost::default_color_type> colors(verticesNum, boost::white_color);
292  // Set the color of the header to black.
293  colors[header] = boost::black_color;
294 
295  // Do a DFS.
296  boost::depth_first_visit(rvsCFG, root, dfsVisitor, &colors[0]);
297 
298  Vertices& verticesInLoop = loops[header];
299  verticesInLoop.insert(vertices.begin(), vertices.end());
300  //verticesInLoop.insert(header);
301  }

322 void writeCFGNode(std::ostream& out, const CFGNode& cfgNode)
323 {
324  SgNode* node = cfgNode.getNode();
325  ROSE_ASSERT(node);
326 
327  std::string nodeColor = "black";
328  if (isSgStatement(node))
329  nodeColor = "blue";
330  else if (isSgExpression(node))
331  nodeColor = "green";
332  else if (isSgInitializedName(node))
333  nodeColor = "red";
334 
335  std::string label;
336 
337  if (SgFunctionDefinition* funcDef = isSgFunctionDefinition(node))
338  {
339  std::string funcName = funcDef->get_declaration()->get_name().str();
340  if (cfgNode.getIndex() == 0)
341  label = "Entry\\n" + funcName;
342  else if (cfgNode.getIndex() == 3)
343  label = "Exit\\n" + funcName;
344  }
345 
346  if (!isSgScopeStatement(node) && !isSgCaseOptionStmt(node) && !isSgDefaultOptionStmt(node))
347  {
348  std::string content = node->unparseToString();
349  boost::replace_all(content, "\"", "\\\"");
350  boost::replace_all(content, "\\n", "\\\\n");
351  label += content;
352  }
353  else
354  label += "<" + node->class_name() + ">";
355 
356  if (label == "")
357  label += "<" + node->class_name() + ">";
358 
359  out << "[label=\"" << label << "\", color=\"" << nodeColor <<
360  "\", style=\"" << (cfgNode.isInteresting()? "solid" : "dotted") << "\"]";
361 }
