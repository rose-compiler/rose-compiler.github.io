374 // Main tasks:
375 // Convert Fortran parameters to the right C ones so XOMP_task() can handle them properly
376 // The challenge is that Fortran parameters are all passed by references.
377 // But XOMP_task() has to support both pass-by-reference and pass-by-value.
378 // Especially, the values/addresses must be copied into task data when tasks are created.
379 // Since a task's parent function stack may not exist when a task is being executed.
380 //
381 // Solution:
382 // Compiler provides all information about the nature (value vs. reference) of parameter, size, and where to get the parameter.
383 // The XOMP layer encodes all information into a single data segment, which will be copied into each task by gomp_task() etc.
384 // The XOMP layer also provides function (run_me_task_ ()) to decode the data segment, and call the outlined Fortran subroutine.
385 //
386 // Wrap function pointer to a outlined function routine and all its parameters into one single parameter gp_parameter[]
387 // The single data segment has the following layout
388 // 8 bytes for the function pointer to the outlined Fortran subroutine
389 // a triplet for each parameter. Every triplet has the following layout
390 // 1 byte for a boolean value to indicate if the parameter should be passed by value
391 // 4 bytes for the size of the parameter, size should be equal to sizeof (void*) if not pass-by-value ( pointer type instead)
392 // x bytes for value or address, x will be the size provide by the 2nd element of the triplet.
393 // For example, if we have a outlined function with two parameters:
394 // an integer index suitable for pass-by-value, This is very typical for a task's firstprivate variables.
395 // an array pass-by-reference,
396 // The data segment to be created will contain (assuming 64-bit target):
397 // 8 bytes (function pointer),
398 // 1 byte(byValue=true), 4 bytes (int value), 8 bytes(address of the index)
399 // 1 byte(byValue=false), 8 bytes (address of the array), 8 bytes(address of the array)
400 // Total 38 bytes for the data segment
401 //
402 // Finally, we can call the real xomp_task () with the bridge run_me_task_#() function

576 extern void XOMP_loop_default(int lower, int upper, int stride, long* n_lower, long* n_upper)

579  int _p_upper;
580  int _p_chunk_size;
581  int addOne ; // adjustment to input and output upper bounds, depending on if they are inclusive or non-inclusive
582 
583  int isDecremental= 0;
584  if (lower>upper)
585  isDecremental = 1;
586 
587  // calculate loop iteration count from lower, upper and stride ,
588  // adjust inclusive stride here TODO
589  // no -1/+1? if upper is already an inclusive bound
590  int _p_iter_count = 0;
591  if (isDecremental == 1)
592  {
593  addOne = 1;
594  //stride = 0 - stride; // n
595  if (stride >0)
596  {
597  printf("Error: in XOMP_loop_default() of xomp.c: stride must be negative for decremental iteration. stride = %d \n ", stride);
598  ROSE_ABORT ();
599  }
600  }
601  else // incremental
602  {
603  addOne = -1; // real bound should be non-inclusive - 1
604  if (stride <0)
605  {
606  printf("Error: in XOMP_loop_default() of xomp.c: stride must be positive for incremental iteration. stride = %d \n ", stride);
607  ROSE_ABORT ();
608  }
609  }
610  // addOne is needed here only if the input upper bound is non-inclusive
611  // we use loop normalization to ensure upper bounds are inclusive already.
612  // So we don't need addOne here anymore
613  //_p_iter_count = ( stride + addOne + upper - lower) / stride;
614  _p_iter_count = ( stride + upper - lower) / stride;
615 
616  // calculate a proper chunk size
617  // two cases: evenly divisible 20/5 =4
618  // not evenly divisible 20/3= 6
619  // Initial candidate
620 
621  int _p_num_threads = omp_get_num_threads();
622 
623  _p_chunk_size = _p_iter_count / _p_num_threads;
624  int _p_ck_temp = _p_chunk_size * _p_num_threads != _p_iter_count;
625 
626  _p_chunk_size = _p_ck_temp + _p_chunk_size;
627 
628  // decide on the lower and upper bound for the current thread
629  int _p_thread_id = omp_get_thread_num();
630 
631 // printf("inside xomp_loop_default(): _p_thread_id =%d\n", _p_thread_id);
632  _p_lower = lower + _p_chunk_size * _p_thread_id * stride;
633 
634  //addOne is needed here if the output upper bound is inclusive
635  // -1 if the output n_upper is an inclusive bound,
636  // we do use inclusive in the final normalized loop
637  _p_upper = _p_lower + _p_chunk_size * stride + addOne;
638 
639  // adjust inclusive stride here
640  // addOne is needed if the input upper bound is non-inclusive
641  // no -1/+1 since upper is already an inclusive bound
642  if (isDecremental == 1)
643  _p_upper = (_p_upper > (upper ) ?_p_upper : (upper ));
644  //_p_upper = (_p_upper > (upper + addOne) ?_p_upper : (upper + addOne));
645  else
646  _p_upper = (_p_upper < upper ?_p_upper : upper);
647  //_p_upper = (_p_upper < (upper + addOne) ?_p_upper : (upper + addOne));
648 
649  *n_lower = _p_lower;
650  *n_upper = _p_upper;
651 // printf("inside xomp_loop_default(): _p_lower=%d, _p_upper=%d\n", _p_lower,_p_upper);
652 }
