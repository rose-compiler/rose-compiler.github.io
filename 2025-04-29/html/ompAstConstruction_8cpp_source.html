367  SgExpression* replace_expression_with_macro_value(std::string define_macro, SgExpression* old_exp, bool& macro_replaced,
368  omp_construct_enum)
369  {
370  SgExpression* newExp = old_exp;
371  // Parse the macro: we are only interested in macros with the form #define MACRO_NAME MACRO_VALUE, the constant macro
372  size_t parenthesis = define_macro.find("(");
373  if(parenthesis == string::npos)
374  { // Non function macro, constant macro
375  unsigned int macroNameInitPos = (unsigned int)(define_macro.find("define")) + 6;
376  while(macroNameInitPos<define_macro.size() && define_macro[macroNameInitPos]==' ')
377  macroNameInitPos++;
378  unsigned int macroNameEndPos = define_macro.find(" ", macroNameInitPos);
379  std::string macroName = define_macro.substr(macroNameInitPos, macroNameEndPos-macroNameInitPos);
380 
381  if(macroName == isSgVarRefExp(old_exp)->get_symbol()->get_name().getString())
382  { // Clause is defined in a macro
383  size_t comma = define_macro.find(",");
384  if(comma == string::npos) // Macros like "#define MACRO_NAME VALUE1, VALUE2" are not accepted
385  { // We create here an expression with the value of the clause defined in the macro
386  unsigned int macroValueInitPos = macroNameEndPos + 1;
387  while(macroValueInitPos<define_macro.size() && define_macro[macroValueInitPos]==' ')
388  macroValueInitPos++;
389  unsigned int macroValueEndPos = macroValueInitPos;
390  while(macroValueEndPos<define_macro.size() &&
391  define_macro[macroValueEndPos]!=' ' && define_macro[macroValueEndPos]!='\n')
392  macroValueEndPos++;
393  std::string macroValue = define_macro.substr(macroValueInitPos, macroValueEndPos-macroValueInitPos);
394 
395  // Check whether the value is a valid integer
396  std::string::const_iterator it = macroValue.begin();
397  while (it != macroValue.end() && std::isdigit(*it))
398  ++it;
399  ROSE_ASSERT(!macroValue.empty() && it == macroValue.end());
400 
401  newExp = buildIntVal(atoi(macroValue.c_str()));
402  if(!isSgPragmaDeclaration(old_exp->get_parent()))
403  replaceExpression(old_exp, newExp);
404  macro_replaced = true;
405  }
406  }
407  }
408  return newExp;

411  SgExpression* checkOmpExpressionClause( SgExpression* clause_expression, SgGlobal* global, omp_construct_enum clause_type )
412  {
413  SgExpression* newExp = clause_expression;
414  // ordered (n): optional (n)
415  if (clause_expression == NULL && clause_type == e_ordered_clause)
416  return NULL;
417  ROSE_ASSERT(clause_expression != NULL);
418  bool returnNewExpression = false;
419  if( isSgTypeUnknown( clause_expression->get_type( ) ) )
420  {
421  SgVarRefExpVisitor v;
422  v.traverse(clause_expression, preorder);
423  std::vector<SgExpression*> expressions = v.get_expressions();
424  if( !expressions.empty() )
425  {
426  if( expressions.size() == 1 )
427  { // create the new expression and return it
428  // otherwise, replace the expression and return the original, which is now modified
429  returnNewExpression = true;
430  }
431 
432  bool macroReplaced;
433  SgDeclarationStatementPtrList& declarations = global->get_declarations();
434  while( !expressions.empty() )
435  {
436  macroReplaced = false;
437  SgExpression* oldExp = expressions.back();
438  for(SgDeclarationStatementPtrList::iterator declIt = declarations.begin(); declIt != declarations.end() && !macroReplaced; ++declIt)
439  {
440  SgDeclarationStatement * declaration = *declIt;
441  AttachedPreprocessingInfoType * preprocInfo = declaration->getAttachedPreprocessingInfo();
442  if( preprocInfo != NULL )
443  { // There is preprocessed info attached to the current node
444  for(AttachedPreprocessingInfoType::iterator infoIt = preprocInfo->begin();
445  infoIt != preprocInfo->end() && !macroReplaced; infoIt++)
446  {
447  if((*infoIt)->getTypeOfDirective() == PreprocessingInfo::CpreprocessorDefineDeclaration)
448  {
449  newExp = replace_expression_with_macro_value( (*infoIt)->getString(), oldExp, macroReplaced, clause_type );
450  }
451  }
452  }
453  }
454 
455  // When a macro is defined in a header without any statement, the preprocessed information is attached to the SgFile
456  if(!macroReplaced)
457  {
458  SgProject* project = SageInterface::getProject();
459  int nFiles = project->numberOfFiles();
460  for(int fileIt=0; fileIt<nFiles && !macroReplaced; fileIt++)
461  {
462  SgFile& file = project->get_file(fileIt);
463  ROSEAttributesListContainerPtr filePreprocInfo = file.get_preprocessorDirectivesAndCommentsList();
464  if( filePreprocInfo != NULL )
465  {
466  std::map<std::string, ROSEAttributesList*> preprocInfoMap = filePreprocInfo->getList();
467  for(std::map<std::string, ROSEAttributesList*>::iterator mapIt=preprocInfoMap.begin();
468  mapIt!=preprocInfoMap.end() && !macroReplaced; mapIt++)
469  {
470  std::vector<PreprocessingInfo*> preprocInfoList = mapIt->second->getList();
471  for(std::vector<PreprocessingInfo*>::iterator infoIt=preprocInfoList.begin();
472  infoIt!=preprocInfoList.end() && !macroReplaced; infoIt++)
473  {
474  if((*infoIt)->getTypeOfDirective() == PreprocessingInfo::CpreprocessorDefineDeclaration)
475  {
476  newExp = replace_expression_with_macro_value( (*infoIt)->getString(), oldExp, macroReplaced, clause_type );
477  }
478  }
479  }
480  }
481  }
482  }
483 
484  expressions.pop_back();
485  }
486  }
487  else
488  {
489  printf("error in checkOmpExpressionClause(): no expression found in an expression clause\n");
490  ROSE_ABORT();
491  }
492  }
493 
494  return (returnNewExpression ? newExp : clause_expression);
495  }
