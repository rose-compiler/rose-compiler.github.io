121 BestMapAddress&
122 BestMapAddress::analyze(const AddressInterval &restrictEntryAddresses, const AddressInterval &restrictTargetAddresses) {
123  // Build a table that contains all possible deltas.
124  const Address mask = IntegerOps::genMask<Address>(nBits_);
125  std::set<Address> deltaSet;
126  maxMatches_ = 0;
127  for (Address entryVa: entryVas_.values()) {
128  if (restrictEntryAddresses.contains(entryVa)) {
129  ++maxMatches_;
130  for (Address targetVa: targetVas_.values()) {
131  if (restrictTargetAddresses.contains(targetVa))
132  deltaSet.insert((targetVa - entryVa) & mask);
133  }
134  }
135  }
136  std::vector<Address> deltas(deltaSet.begin(), deltaSet.end());
137 
138  // Build worker thread tasks, each of which determines how many call targets would resolve to function entries if the
139  // function entries were shifted by some delta.
140  std::vector<size_t> nMatches(deltas.size(), 0);
141  Sawyer::Container::Graph<Task> tasks;
142  for (size_t i=0; i<deltas.size(); ++i)
143  tasks.insertVertex(Task(deltas[i], nMatches[i]));
144  Sawyer::ProgressBar<size_t> progressBar(tasks.nVertices(), mlog[MARCH]);
145  progressBar.suffix(" comparisons");
146  size_t nThreads = settings_.nThreads ? *settings_.nThreads : Rose::CommandLine::genericSwitchArgs.threads;
147  Sawyer::workInParallel(tasks, nThreads, Worker(this, progress_, progressBar));
148 
149  // Sort and cache the results by number of matches.
150  upToDate_ = false;
151  results_.clear();
152  for (size_t i=0; i<deltas.size(); ++i)
153  results_.insertMaybeDefault(nMatches[i]).push_back(deltas[i]);
154  upToDate_ = true;
155 
156  return *this;
157 }

177 BestMapAddress::align(const MemoryMap::Ptr &map, const P2::Engine::Settings &settings, const Progress::Ptr &progress) {
178  Sawyer::Message::Stream info(mlog[INFO]);
179  if (info) {
180  info <<"memory map before aligning segments:\n";
181  map->dump(info, " ");
182  }
183 
184  MemoryMap::Ptr nonExecutable = map->shallowCopy();
185  nonExecutable->require(MemoryMap::EXECUTABLE).prune();
186  MemoryMap::Ptr retval = nonExecutable->shallowCopy();
187 
188  // Process one executable segment (plus all non-executable) at a time.
189  BestMapAddress::AddressSet entryAddresses;
190  size_t nWork = 0, totalWork = 2 * map->nSegments();
191  for (const MemoryMap::Node &mmNode: map->nodes()) {
192  nWork += 2; // incremented early in case of "continue" statements
193 
194  // What to align
195  const AddressInterval &interval = mmNode.key();
196  const MemoryMap::Segment &segment = mmNode.value();
197  if (0 == (segment.accessibility() & MemoryMap::EXECUTABLE))
198  continue;
199  MemoryMap::Ptr tmpMap = map->shallowCopy();
200  tmpMap->require(MemoryMap::EXECUTABLE).prune();
201  tmpMap->insert(interval, segment);
202  if (info) {
203  info <<"aligning executable segment using this map:\n";
204  tmpMap->dump(info, " ");
205  }
206 
207  // Partitioning engine used by the BestMapAddress analysis.
208  P2::Engine::Ptr engine = P2::EngineBinary::instance(settings);
209  engine->memoryMap(tmpMap);
210  engine->settings().partitioner.doingPostAnalysis = false;
211  if (progress)
212  engine->progress(progress);
213 
214  // Analyze the tmpMap using the specified partitioning engine.
215  BestMapAddress mapAnalyzer;
216  mapAnalyzer.progress(progress);
217  {
218  ProgressTask t(progress, "disassemble", (double)(nWork-1) / totalWork);
219  mapAnalyzer.gatherAddresses(*engine);
220  }
221  mlog[INFO] <<"found " <<StringUtility::plural(mapAnalyzer.entryAddresses().size(), "entry addresses") <<" and "
222  <<StringUtility::plural(mapAnalyzer.targetAddresses().size(), "target addresses") <<"\n";
223  for (Address entryVa: entryAddresses.values())
224  mapAnalyzer.insertEntryAddress(entryVa);
225  mlog[INFO] <<"using " <<StringUtility::plural(mapAnalyzer.entryAddresses().size(), "total entry addresses") <<"\n";
226  info <<"performing remap analysis";
227  Sawyer::Stopwatch remapTime;
228  {
229  ProgressTask t(progress, "remap", (double)(nWork-0) / totalWork);
230  mapAnalyzer.analyze();
231  }
232  std::vector<Address> deltas = mapAnalyzer.bestDeltas();
233  info <<"; took " <<remapTime <<"\n";
234  mlog[INFO] <<"found " <<StringUtility::plural(deltas.size(), "deltas") <<" with match ratio "
235  <<(100.0*mapAnalyzer.bestDeltaRatio()) <<"%\n";
236 
237  // Find the best deltas by which to shift the executable segment. Try deltas in the order given until we find one that
238  // shifts the executable segment to an area of memory where it doesn't overlap with any non-executable segments.
239  // static const Address mask = mapAnalyzer.mask();
240  bool remapped = false;
241  size_t bestDelta = 0;
242  const Address mask = mapAnalyzer.mask();
243  for (Address delta: deltas) {
244  // Check for overflow: we don't want a delta that would split the executable segment between the highest addresses
245  // and the lowest addresses.
246  Address newLo = (interval.least() + delta) & mask;
247  Address newHi = (interval.greatest() + delta) & mask;
248  if (newHi < newLo)
249  continue; // overflow
250  const AddressInterval newInterval = AddressInterval::hull(newLo, newHi);
251  if (nonExecutable->within(newInterval).exists())
252  continue; // new location would overlap with data segments
253 
254  if (0 == delta) {
255  info <<"segment " <<StringUtility::addrToString(interval.least()) <<" is good where it is.\n";
256  } else {
257  info <<"segment " <<StringUtility::addrToString(interval.least())
258  <<" should move to " <<StringUtility::addrToString(newInterval.least()) <<"\n";
259  }
260  retval->insert(newInterval, segment);
261  remapped = true;
262  bestDelta = delta;
263  break;
264  }
265 
266  if (!remapped) {
267  mlog[WARN] <<"cannot find a valid destination address for \"" <<StringUtility::cEscape(segment.name()) <<"\""
268  <<"; leaving at " <<StringUtility::addrToString(interval.least()) <<"\n";
269  retval->insert(interval, segment);
270  }
271 
272  // Add adjusted entry addresses to the set of all entry addresses. This is optional, but sometimes helps the remap
273  // analysis by giving it more information.
274  for (Address origEntryVa: mapAnalyzer.entryAddresses().values()) {
275  Address adjustedEntryVa = (origEntryVa + bestDelta) & mask;
276  entryAddresses.insert(adjustedEntryVa);
277  }
278  }
279 
280  return retval;
281 }
