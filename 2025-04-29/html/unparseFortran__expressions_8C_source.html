129 void
130 FortranCodeGeneration_locatedNode::unparseLabelRefExp(SgExpression* expr, SgUnparse_Info&)
131  {
132  SgLabelRefExp* labelRefExp = isSgLabelRefExp(expr);
133  ASSERT_not_null(labelRefExp);
134 
135  SgLabelSymbol* labelSymbol = labelRefExp->get_symbol();
136  ASSERT_not_null(labelSymbol);
137 
138  int numericLabel = labelSymbol->get_numeric_label_value();
139  ROSE_ASSERT(numericLabel >= 0);
140 
141  string numericLabelString = StringUtility::numberToString(numericLabel);
142 
143  // DQ (2/2/2011): We can't do this since it will effect where lables are unparse in the OPEN statement (and likely other I/O statements).
144  // After some email with Scott this is required to be handled via a special case so since in all other case the SgLabelRefExp shuld
145  // have a IOStatement as a parent, we will look for where the parent is part of an expression list. This could be improved later.
146  // the best way to handle this would be to do the type checking, and I'm OK with that approach. The backup plan is to embed names
147  // into (what) that would trigger these to be treated as alternative return arguments.
148  // We could also check if the enclosing statement is an IO statement. So there are a number of options here.
149 
150  SgStatement* tmp_statement = SageInterface::getEnclosingStatement(labelRefExp);
151  ASSERT_not_null(tmp_statement);
152 
153  // Check for either a SgIOStatement or a SgReturnStatement (not the special case we are looking for)
154  if (isSgIOStatement(tmp_statement) == NULL && isSgReturnStmt(tmp_statement) == NULL)
155  {
156  // Output "*" if this is NOT a SgIOStatement (OK since I think that only functions in a function CALL statement can be used with alternative IO, is this true?
157  curprint("*");
158 
159  // Instead of the numericLabelString, we output the index into the array of arguments with type == SgLabelSymbol taken from the function declaration's parameter list.
160  curprint(numericLabelString);
161  }
162  else
163  {
164  if (isSgReturnStmt(tmp_statement) != NULL)
165  {
166  // This is a return statement, but we have to check if it is associated with a function that has SgTypeLabel parameters.
167  // bool functionHasAlternativeArgumentParameters = true;
168 
169  size_t alternativeReturnValue = 0;
170 
171  // This is always a valid value (but not be correct)... just testing for now...
172  // We have to correlate this SgLabelRefExp with the SgLabelSymbol of the correct parameter.
173 
174  // Note that this code is similar (copyied from) to R1236 c_action_return_stmt() in the ROSE Fortran support.
175  SgFunctionDefinition* functionDefinition = SageInterface::getEnclosingFunctionDefinition(tmp_statement, /* includingSelf= */ true);
176  ASSERT_not_null(functionDefinition);
177 
178  SgFunctionDeclaration* functionDeclaration = functionDefinition->get_declaration();
179  ASSERT_not_null(functionDeclaration);
180 
181  SgInitializedNamePtrList & args = functionDeclaration->get_args();
182  ROSE_ASSERT(alternativeReturnValue < args.size());
183 
184  // The Fortran world starts at one (not zero)!
185  size_t counter = 1;
186 
187  for (size_t i = 0; i < args.size(); i++)
188  {
189  SgType* argumentType = args[i]->get_type();
190  SgTypeLabel* labelType = isSgTypeLabel(argumentType);
191  if (labelType != nullptr)
192  {
193  // Search the argument list for a matching symbol.
194  SgSymbol* tmp_symbol = args[i]->get_symbol_from_symbol_table();
195  if (tmp_symbol == labelSymbol && alternativeReturnValue < 1)
196  {
197  // We have a match.
198  alternativeReturnValue = counter;
199  }
200  counter++;
201  }
202  }
203 
204  // This is using Fortran world numbering (which starts a one, not zero).
205  // curprint("1");
206  ROSE_ASSERT(alternativeReturnValue > 0);
207  ROSE_ASSERT(alternativeReturnValue <= args.size());
208  curprint(StringUtility::numberToString(alternativeReturnValue));
209  }
210  else
211  {
212  // This is the most common case.
213  curprint(numericLabelString);
214  }
215  }
216  }

232 void
233 FortranCodeGeneration_locatedNode::unparseFuncCall(SgExpression* expr, SgUnparse_Info& info)
234 {
235  // Sage node corresponds to a Fortran function/subroutine call
236  SgFunctionCallExp* func_call = isSgFunctionCallExp(expr);
237  ASSERT_not_null(func_call);
238 
239  // -----------------------------------------------------
240  // Unparse as pre-order subroutine/function call.
241  // -----------------------------------------------------
242 
243  if (isSubroutineCall(func_call)) {
244  curprint("CALL ");
245  }
246 
247  // subroutine/function name
248  unparseExpression(func_call->get_function(), info);
249 
250  // argument list
251  SgUnparse_Info ninfo(info);
252  curprint("(");
253  if (func_call->get_args()) {
254  SgExpressionPtrList& list = func_call->get_args()->get_expressions();
255  SgExpressionPtrList::iterator arg = list.begin();
256  while (arg != list.end()) {
257  unparseExpression((*arg), ninfo);
258  arg++;
259  if (arg != list.end()) {
260  curprint(",");
261  }
262  }
263  }
264  curprint(")");
265 }

319 // DQ (8/6/2010): Output the logical operator when the operands are logical (SgBoolType)
320 // the type of the expression is not enough to test, we have to test the lhs and rhs type.
321 bool
322 outputLogicalOperator(SgExpression* expr)
323  {
324  bool outputLogicalOperator = false;
325 
326  SgBinaryOp* binaryOp = isSgBinaryOp(expr);
327  if (binaryOp != NULL)
328  {
329  SgExpression* lhs = binaryOp->get_lhs_operand();
330  SgExpression* rhs = binaryOp->get_rhs_operand();
331  SgType* lhs_type = lhs->get_type();
332  SgType* rhs_type = rhs->get_type();
333  if (isSgTypeBool(lhs_type) != NULL)
334  {
335  if (isSgTypeBool(rhs_type) == NULL)
336  {
337  printf ("Error: outputLogicalOperator(). Found a boolean lhs operand paired with a non-boolean rhs operand for SgExpression:%s\n",expr->class_name().c_str());
338  ASSERT_not_null(isSgTypeBool(rhs_type));
339  }
340  outputLogicalOperator = true;
341  }
342  }
343  else
344  {
345  printf ("Error: this function only needs to handle binary operators. ");
346  ROSE_ABORT();
347  }
348 
349  return outputLogicalOperator;
350  }

569 void
570 FortranCodeGeneration_locatedNode::unparseCastOp(SgExpression* expr, SgUnparse_Info& info)
571  {
572  // DQ (8/16/2007): Allow SgCast operators to work since we wnat to test the unparser using C code
573  // and we will later want to add cast operators to the Fortran AST to explicitly mark implicit casts
574  // in fortran (marked as compiler generated).
575 
576  // Rasmussen (5/17/2023): I think this means this code can be removed, added abort to see if testing fails
577  // Perhaps the warning can be turned off as some of the switch options don't abort.
578  mlog[FATAL] << "Case operators not defined for Fortran code generation! node = " << expr->class_name() << "\n";
579  ROSE_ABORT();
580 
581  SgCastExp* cast_op = isSgCastExp(expr);
582  ASSERT_not_null(cast_op);
583 
584  SgUnparse_Info ninfo(info);
585  ninfo.unset_PrintName();
586 
587  // Never unparse the declaration from within a cast expression
588  ninfo.set_SkipDefinition();
589  ninfo.unset_SkipBaseType();
590 
591  switch(cast_op->cast_type())
592  {
593  case SgCastExp::e_unknown:
594  {
595  printf ("SgCastExp::e_unknown found \n");
596  ROSE_ABORT();
597  }
598 
599  case SgCastExp::e_default:
600  {
601  printf ("SgCastExp::e_default found \n");
602  ROSE_ABORT();
603  }
604 
605  case SgCastExp::e_dynamic_cast:
606  {
607  // dynamic_cast <P *> (expr)
608  printf ("SgCastExp::e_dynamic_cast found (Now defined in Fortran) \n");
609  ROSE_ABORT();
610  }
611  case SgCastExp::e_static_cast:
612  {
613  // static_cast <P *> (expr)
614  printf ("SgCastExp::e_static_cast found (Now defined in Fortran) \n");
615  ROSE_ABORT();
616  }
