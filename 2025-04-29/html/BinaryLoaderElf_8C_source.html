97 Address
98 BinaryLoaderElf::rebase(const MemoryMap::Ptr &map, SgAsmGenericHeader *header, const SgAsmGenericSectionPtrList &sections) {
99  static const size_t maximum_alignment = 8192;
100  AddressInterval mappableArea = AddressInterval::whole();
101 
102 #if 0 // [Robb P. Matzke 2014-10-09]
103  // If this is a library then restrict where it can be mapped. This is to try to make BinaryLoaderElf behave more like the
104  // native loader on Linux (at least when run with "i386 --addr-compat-layout --addr-no-randomize), but it's a failing
105  // proposition because the headers in ROSE are apparently not processed in the same order as in Linux.
106  SgAsmGenericFormat *fmt = header->get_executableFormat();
107  if (fmt->get_purpose() == SgAsmGenericFormat::PURPOSE_LIBRARY)
108  mappableArea = AddressInterval::hull(0x55555000, mappableArea.greatest());
109 #endif
110 
111  // Find the minimum address desired by the sections to be mapped.
112  Address min_preferred_rva = (uint64_t)(-1);
113  for (SgAsmGenericSectionPtrList::const_iterator si=sections.begin(); si!=sections.end(); ++si)
114  min_preferred_rva = std::min(min_preferred_rva, (*si)->get_mappedPreferredRva());
115  Address min_preferred_va = header->get_baseVa() + min_preferred_rva;
116 
117  // Minimum address at which to map
118  AddressInterval freeSpace = map->unmapped(mappableArea.greatest(), Sawyer::Container::MATCH_BACKWARD);
119  freeSpace = freeSpace & mappableArea;
120  if (freeSpace.isEmpty())
121  throw MemoryMap::NoFreeSpace("no free specimen memory", map, 0);
122  Address map_base_va = alignUp(freeSpace.least(), (Address)maximum_alignment);
123 
124  // If the minimum preferred virtual address is less than the floor of the page-aligned mapping area, then
125  // return a base address which moves the min_preferred_va to somewhere in the page pointed to by map_base_va.
126  if (min_preferred_va < map_base_va) {
127  size_t min_preferred_page = min_preferred_va / maximum_alignment;
128  if (map_base_va < min_preferred_page * maximum_alignment)
129  return 0;
130  return map_base_va - min_preferred_page*maximum_alignment;
131  }
132 
133  // No need to rebase
134  return header->get_baseVa();
135 }

138 BinaryLoaderElf::alignValues(SgAsmGenericSection *_section, const MemoryMap::Ptr &map,
139  Address *malign_lo_p, Address *malign_hi_p,
140  Address *va_p, Address *mem_size_p,
141  Address *offset_p, Address *file_size_p, bool *map_private_p,
142  Address *va_offset_p, bool *anon_lo_p, bool *anon_hi_p,
143  ConflictResolution *resolve_p) {
144  ASSERT_not_null(_section);
145  ASSERT_not_null(map);
146  ASSERT_not_null(malign_lo_p);
147  ASSERT_not_null(malign_hi_p);
148  ASSERT_not_null(va_p);
149  ASSERT_not_null(mem_size_p);
150  ASSERT_not_null(offset_p);
151  ASSERT_not_null(file_size_p);
152  ASSERT_not_null(map_private_p);
153  ASSERT_not_null(va_offset_p);
154  ASSERT_not_null(anon_lo_p);
155  ASSERT_not_null(anon_hi_p);
156  ASSERT_not_null(resolve_p);
157 
158  SgAsmElfSection *section = isSgAsmElfSection(_section);
159  ASSERT_not_null(section); /* This method is only for ELF files. */
160 
161  // Maximum alignment seems to be 4k regardless of what the ELF file says. You can see this by running:
162  // $ x86_64 -R /bin/cat /proc/self/maps
163  // and noticing that the /bin/cat segments are mapped at '4000, 'b000, and 'c000 even though the ELF file says the memory
164  // alignment constraint is 0x200000.
165  if (*malign_lo_p > 4096)
166  *malign_lo_p = 4096;
167  if (*malign_hi_p > 4096)
168  *malign_hi_p = 4096;
169 
170  /* ELF Segments are aligned using the superclass, but when the section has a low- or high-padding area we'll use file
171  * contents for the low area and zeros for the high area. Due to our rebase() method, there should be no conflicts between
172  * this header's sections and sections previously mapped from other headers. Any conflicts are within a single header and
173  * are resolved by moving the segment to a free area. */
174  if (section->get_segmentEntry()) {
175  MappingContribution retval = BinaryLoader::alignValues(section, map, malign_lo_p, malign_hi_p, va_p,
176  mem_size_p, offset_p, file_size_p, map_private_p, va_offset_p,
177  anon_lo_p, anon_hi_p, resolve_p);
178 
179  *anon_lo_p = false;
180  *map_private_p = true;
181  *resolve_p = RESOLVE_OVERMAP;
182 
183  if (section->get_mappedSize()==section->get_size()) {
184  *anon_hi_p = false;
185  } else {
186  *anon_hi_p = true;
187  }
188 
189  return retval;
190  }
191 
192  /* ELF Sections are often used to refine areas of a segment. A real loader ignores ELF Sections, but the refinement can
193  * be useful for analyses. Therefore, we'll allow ELF Sections to contribute to the mapping. Alignment of ELF Sections
194  * is temporarily set to none (because they live at arbitrary offsets in ELF Segments). If the section lives in a segment
195  * that has been mapped to an address other than its preferred address, then we must make sure the ELF Section is also
196  * similarly moved along with the ELF Segment. */
197  SgAsmElfSection *part_of = NULL; /*what segment is this section a part of (if any)? */
198  SgAsmGenericSectionPtrList sections = section->get_header()->get_mappedSections();
199  for (SgAsmGenericSectionPtrList::iterator si=sections.begin(); si!=sections.end(); ++si) {
200  SgAsmElfSection *segment = isSgAsmElfSection(*si);
201  if (segment && segment->get_segmentEntry() &&
202  section->get_mappedPreferredRva() >= segment->get_mappedPreferredRva() &&
203  section->get_mappedPreferredRva() < segment->get_mappedPreferredRva() + segment->get_mappedSize()) {
204  part_of = segment;
205  break;
206  }
207  }
208  Address diff = part_of ? part_of->get_mappedActualVa() - part_of->get_mappedPreferredVa() : 0;
209  *malign_hi_p = *malign_lo_p = 1; /*no alignment constraint*/
210  *va_p = section->get_header()->get_baseVa() + section->get_mappedPreferredRva() + diff;
211  *mem_size_p = section->get_mappedSize();
212  *offset_p = section->get_offset();
213  *file_size_p = section->get_size();
214  *va_offset_p = 0;
215  *anon_lo_p = *anon_hi_p = true;
216  *map_private_p = false;
217  *resolve_p = RESOLVE_OVERMAP; /*erase (part of) the previous ELF Segment's memory */
218  return CONTRIBUTE_ADD;
219 }

312  typedef std::vector<std::pair<SgAsmGenericSection*, SymbolMap*> > SymbolMapList;
313  Stream trace(mlog[TRACE]);
314  trace.facilityName("BinaryLoaderElf");
315 
316  /* Build a mapping (symbolMaps) from each ".dynsym" section (one per file header) to a SymbolMap. The SymbolMap
317  * holds the symbols defined (i.e., not local, hidden, or only referenced) by the section. */
318  SgAsmGenericHeaderPtrList& headers = interp->get_headers()->get_headers();
319  SymbolMapList symbolMaps; /*maps each .dynsym section to a SymbolMap::BaseMap object. */
320  for (size_t h=0; h < headers.size(); ++h) {
321  SgAsmGenericHeader* header = headers[h];
322  trace <<"building symbol map for " <<header->get_file()->get_name() <<"...\n";
323  SymverResolver versionResolver(header);
324 
325  /* There must be zero or one ELF Section named ".dynsym" under this header. If the section does not exist then the
326  * header is not dynamically linked and neither provides nor requires dynamic symbol resolution.
327  *
328  * Manual 1-10 under SHT_SYMTAB and SHT_DYNSYM: ...SHT_DYNSYM section holds a minimal set of dynamic linking symbols.
329  * FIXME: Technically, the dynsym may be omitted, and we should truly use the .dynamic section's DT_SYMTAB entry(ies) */
330  SgAsmElfSymbolSection *dynsym = isSgAsmElfSymbolSection(header->get_sectionByName(".dynsym"));
331  if (!dynsym)
332  continue;
333  ASSERT_not_null(dynsym->get_sectionEntry());
334  ASSERT_require(SgAsmElfSectionTableEntry::SHT_DYNSYM == dynsym->get_sectionEntry()->get_sh_type());
335 
336  /* Add each symbol definition to the SymbolMap and then add the SymbolMap to the symbolMaps */
337  SymbolMap* symbolMap = new SymbolMap;
338  for (size_t symbol_idx=0; symbol_idx<dynsym->get_symbols()->get_symbols().size(); symbol_idx++) {
339  if (0==symbol_idx) continue; /*this must be undefined, so we can skip it*/
340 
341  SgAsmElfSymbol* symbol = dynsym->get_symbols()->get_symbols()[symbol_idx];
342  trace <<" symbol [" <<symbol_idx <<"] \"" <<symbol->get_name()->get_string(true) <<"\" ";
343 
344  VersionedSymbol symver = versionResolver.getVersionedSymbol(symbol);
345 // I think we need all symbols in the symbol table for relocations. [RPM 2010-09-16]
346 // if (symver.is_local()) {
347 // /* Symbols in the dynsym should never be local (if this is isn't the case, we can ignore them). Symbol
348 // * versioning also may make symbols "local". */
349 // trace <<" local (ignoring)\n";
350 // } else if (symver.is_hidden()) {
351 // /* Symbol versioning can result in 'hidden' symbols that should be ignored. */
352 // trace <<" hidden (ignoring)\n";
353 // } else
354  if (symver.isReference()) {
355  /* Symbol versioning lets us determine which symbols are 'references' i.e. that are only used to help
356  * relocating */
357  trace <<" reference (ignoring)\n";
358  } else {
359  std::string symName = symver.getName();
360  ASSERT_require(symName==symbol->get_name()->get_string());
361  SymbolMapEntry &symbolEntry = (*symbolMap)[symName];
362  symbolEntry.addVersion(symver);
363  trace <<" added.\n";
364  }
365  }

367  }

371  symbols_.clear();

373  const SymbolMap *symbolMap = smi->second;
374  for(SymbolMap::const_iterator newSymbolIter=symbolMap->begin(); newSymbolIter!=symbolMap->end(); ++newSymbolIter) {
375  const SymbolMapEntry &newEntry = newSymbolIter->second;
376  const std::string symName = newSymbolIter->first;
377  SymbolMap::iterator oldSymbolIter=symbols_.find(symName);
378 
379  if (oldSymbolIter == symbols_.end()) {
380  /* no symbol yet, set it (this is the most common case) */

382  } else {

642 void
643 BinaryLoaderElf::SymbolMapEntry::merge(const SymbolMapEntry& newEntry) {
644  VersionedSymbol oldSymbol = versions_.front();
645  VersionedSymbol newSymbol = newEntry.versions_.front();
646  int start=0;
647 
648  /* Merge base definitions. */
649  if (oldSymbol.isBaseDefinition() && newSymbol.isBaseDefinition()) {
650  if (oldSymbol.symbol()->get_elfBinding() == SgAsmElfSymbol::STB_WEAK &&
651  newSymbol.symbol()->get_elfBinding() == SgAsmElfSymbol::STB_GLOBAL) {
652  /* the new symbol becomes the new base */
653  versions_[0] = newSymbol;
654  start=1;
655  if (!oldSymbol.getVersion().empty() && oldSymbol.getVersion() != newSymbol.getVersion()) {
656  /* The old symbol was overridden, but it still has a unique version - so we need to keep it. */
657  versions_.push_back(oldSymbol);
658  }
659  }
660  }
661 
662  /* Merge remaining versions. */
663  for (size_t i=start; i<newEntry.versions_.size(); ++i) {
664  newSymbol = newEntry.versions_[i];
665  std::string newVersion=newSymbol.getVersion();
666  if (newVersion.empty()) {
667  /* If the version has no version then it should have been the base entry, in which case it would have been handled
668  * above. Since it isn't, this is some 'weaker' non-versioned entry, and it can be dropped. */
669  continue;
670  }
671  size_t found=false;
672  for (size_t j=0; j<versions_.size(); ++j) {
673  oldSymbol = versions_[j];
674  if (oldSymbol.getVersion() == newSymbol.getVersion() && /* matching version string */
675  oldSymbol.symbol()->get_elfBinding() == SgAsmElfSymbol::STB_WEAK && /* old is weak */
676  newSymbol.symbol()->get_elfBinding() == SgAsmElfSymbol::STB_GLOBAL) {/* new is strong */
677  versions_[j] = newSymbol; /* override the old symbol */
678  found=true;
679  break;
680  }
681  }
682  if (!found) {
683  /* This version doesn't exist in this entry, so append it to the list. */
684  versions_.push_back(newSymbol);
685  }
686  }
687 }

809 void
810 BinaryLoaderElf::SymverResolver::makeVersionedSymbolMap(SgAsmElfSymbolSection* dynsym, SgAsmElfSymverSection* symver/*=NULL*/) {
811  ASSERT_require(dynsym && dynsym->get_isDynamic());
812  SgAsmElfSymbolPtrList& symbols = dynsym->get_symbols()->get_symbols();
813 
814  /* symverSection may be NULL, but if it isn't, it must have the same number of entries as dynsym */
815  ASSERT_require(NULL == symver || symbols.size() == symver->get_entries()->get_entries().size());
816 
817  for (size_t symbol_idx=0; symbol_idx<symbols.size(); symbol_idx++) {
818  SgAsmElfSymbol *symbol = symbols[symbol_idx];
819 
820  ASSERT_require(versionedSymbolMap_.end() == versionedSymbolMap_.find(symbol));
821  VersionedSymbol* versionedSymbol = new VersionedSymbol(symbol);
822  versionedSymbolMap_.insert(std::make_pair(symbol,versionedSymbol));
823 
824  if (symver) {
825  SgAsmElfSymverEntry *symverEntry = symver->get_entries()->get_entries()[symbol_idx];
826  uint16_t value = symverEntry->get_value();
827 
828  /* From the Sun Microsystem "Linker and Libraries Guide" April 2008, page 254:
829  *
830  * STV_HIDDEN: A symbol that is defined in the current component is hidden if its name is not visible in other
831  * components. Such a symbol is necessarily protected. This attribute is used to control the
832  * external interface of a component. An object named by such a symbol can still be referenced
833  * from another component if its address is passed outside. A hidden symbol contained in a
834  * relocatable object is either removed or converted to STB_LOCAL binding when the object is
835  * included in an executable file or shared object.
836  *
837  * Therefore, we will also treat this symbol as local. */
838  if (value & VERSYM_HIDDEN)
839  value = 0;
840 
841  versionedSymbol->versionEntry(symverEntry);
842 
843  /* 0 and 1 are special (local and global respectively) they DO NOT correspond to entries in the ELF Symbol
844  * Version Definition Table or the ELF Symbol Version Requirements Table. Otherwise the value should exist in
845  * exactly one of those tables. */
846  if (0 != value && 1 != value) {
847  SymbolVersionDefinitionMap::const_iterator defIter = symbolVersionDefMap_.find(value);
848  SymbolVersionNeededMap::const_iterator needIter = symbolVersionNeedMap_.find(value);
849  /* We must have a match from defs or needs, not both, not neither. */
850  ASSERT_require((symbolVersionDefMap_.end() == defIter) != (symbolVersionNeedMap_.end() == needIter));
851  if (symbolVersionDefMap_.end() != defIter) {
852  versionedSymbol->versionDef(defIter->second);
853  } else if (symbolVersionNeedMap_.end() != needIter) {
854  versionedSymbol->versionNeed(needIter->second);
855  }
856  }
857  }
858  }

1075 Address
1076 BinaryLoaderElf::fixupInfoExpr(const std::string &expression, SgAsmElfRelocEntry *reloc, const SymverResolver &resolver,
1077  const MemoryMap::Ptr &memmap, Address *target_va_p) {
1078  std::vector<Address> stack;
1079  SgAsmElfSymbol *symbol = NULL; /* Defining symbol for relocation */
1080  size_t nbytes = 0; /* Size of addend */
1081  Address target_va = fixupInfoTargetVa(reloc);
1082  if (target_va_p)
1083  *target_va_p = target_va;
1084 
1085  for (std::string::size_type i=0; i<expression.size(); i++) {
1086  switch (expression[i]) {
1087  case '0':
1088  case '4':
1089  case '8':
1090  nbytes = expression[i] - '0';
1091  break;
1092 
1093  case 'A': { /* Addend of relocation from reloc entry or specimen memory */
1094  Address addend = fixupInfoAddend(reloc, target_va, memmap, nbytes);
1095  stack.push_back(addend);
1096  break;
1097  }
1098  case 'B': { /* Adjustment for defining symbol's section */
1099  if (!symbol)
1100  symbol = fixupInfoRelocSymbol(reloc, resolver);
1101  Address symbol_adj;
1102  fixupInfoSymbolVa(symbol, NULL, &symbol_adj);
1103  stack.push_back(symbol_adj);
1104  break;
1105  }
1106  case 'S': { /* Value of defining symbol for relocation */
1107  if (!symbol)
1108  symbol = fixupInfoRelocSymbol(reloc, resolver);
1109  Address symbol_va = fixupInfoSymbolVa(symbol);
1110  stack.push_back(symbol_va);
1111  break;
1112  }
1113  case '+': { /* Addition of two top stack items */
1114  ASSERT_require(stack.size()>=2);
1115  Address opand1 = stack.back(); stack.pop_back();
1116  Address opand2 = stack.back(); stack.pop_back();
1117  stack.push_back(opand1 + opand2);
1118  break;
1119  }
1120  default:
1121  ASSERT_not_implemented(std::string("fixup info expression = '") + expression[i] + "'");
1122  }
1123  }
1124 
1125  ASSERT_require(1==stack.size());
1126  return stack[0];
1127 }
