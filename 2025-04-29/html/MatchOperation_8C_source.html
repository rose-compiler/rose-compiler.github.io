49 bool
50 MatchOpOr::performOperation(MatchStatus& status, RoseAst::iterator& i, SingleMatchResult& smr) {
51  if(status.debug) {
52  std::cout<<"performing OR-match operation."<<std::endl;
53  }
54 
55  /* match-operations are performed left to right, using
56  short-circuit. marked nodes are propagated through subexpressions
57  */
58 
59 #if 1
60  RoseAst::iterator tmp_iter_left=i;
61  RoseAst::iterator tmp_iter_right=i;
62  SingleMatchResult left_smr;
63  bool left_match=_left->performOperation(status,tmp_iter_left,left_smr);
64  if(left_match) {
65  i=tmp_iter_left;
66  smr=left_smr;
67  return true;
68  } else {
69  if(left_smr.isSMRMarkedLocation(tmp_iter_right)||status.isMarkedLocationAddress(tmp_iter_right)) {
70  i=tmp_iter_left;
71  smr=left_smr;
72  return false;
73  } else {
74  SingleMatchResult right_smr=left_smr;
75  if(_right->performOperation(status,tmp_iter_right,right_smr)) {
76  i=tmp_iter_right;
77  smr=right_smr;
78  return true;
79  } else {
80  // nothing changed
81  return false;
82  }
83  }
84  }
85 
86 #else
87  // alternate semantics (without short-circuit evaluation)
88  RoseAst::iterator tmp_iter_left=i;
89  RoseAst::iterator tmp_iter_right=i;
90  SingleMatchResult left_smr;
91  SingleMatchResult right_smr;
92  bool left_result=_left->performOperation(status,tmp_iter_left,left_smr);
93 
94  bool right_result=false;
95  // check if locations which have been marked in the lhs-match would exclude the current node
96  if(!left_smr.isSMRMarkedLocation(tmp_iter_right))
97  right_result=_right->performOperation(status,tmp_iter_right,right_smr);
98 
99  if(left_result && right_result) {
100  status.mergeSingleMatchResult(left_smr);
101  status.mergeSingleMatchResult(right_smr);
102  // both tmp_iter_left and tmp_iter_right are correct
103  assert(tmp_iter_left==tmp_iter_right);
104  i=tmp_iter_right;
105  return true;
106  } else if(left_result) {
107  i=tmp_iter_left;
108  status.mergeSingleMatchResult(left_smr);
109  return true;
110  } else if(right_result) {
111  i=tmp_iter_right;
112  status.mergeSingleMatchResult(right_smr);
113  return true;
114  }
115  return false;
116 #endif
117 }
