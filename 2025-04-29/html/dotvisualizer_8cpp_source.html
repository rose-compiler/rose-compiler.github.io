140  unsigned long long ret = 0;
141 #ifdef _WIN32
142  static LARGE_INTEGER liTimerFrequency = time_freq();

146 #else
147  struct timeval tv;

149  tz.tz_minuteswest = 0;
150  tz.tz_dsttime = 0;

153  //fprintf(stderr, " Tp s%lu us%lu \n", tv.tv_sec, tv.tv_usec );
154 #endif
155  //cout << " Tret " << ret <<endl;

439 //-----------------------------------------------------------------------------
440 // handle mouse press
441 void DotVisualizer::handleMouse(int button, int state, int x, int y)
442 {
443  float newx, newy; // map click?
444  if(!mpGraph) return;
445 
446  mMouseLastX = x;
447  mMouseLastY = y;
448  switch (button) {
449  case 1:
450  if(state == 0) {
451  mButtonCurrent = mButtonCurrent & (~VIS_BUTTONL);
452  mOnMap = false;
453  mOnNode = false;
454  } else {
455  mButtonCurrent = mButtonCurrent | VIS_BUTTONL;
456  mOnMap = mpMapC->check_click(x,y, &newx, &newy);
457  if(!mOnMap) {
458  // try to select node under cursor
459  mOnNode = mpGLC->select_click( mpGraph );
460  if(mOnNode) {
461  if(mpGLC->get_nodeposition( &mTargetX, &mTargetY, &mTargetZ, &mTargetLookX, &mTargetLookY ))
462  mTrackTarget = true;
463  } else {
464  // not clicked on map or node...
465  mTrackTarget = false;
466  }
467  }
468  }
469  break;
470  case 2:
471  if(state == 0) {
472  mButtonCurrent = mButtonCurrent & (~VIS_BUTTONM);
473  } else {
474  mButtonCurrent = mButtonCurrent | VIS_BUTTONM;
475  }
476  break;
477  case 3:
478  if(state == 0) {
479  mButtonCurrent = mButtonCurrent & (~VIS_BUTTONR);
480  } else {
481  mButtonCurrent = mButtonCurrent | VIS_BUTTONR;
482  }
483  mTrackTarget = false;
484  break;
485  case 4: // wheel up
486  if(mCamZoomPos<mCamZ) mCamZoomPos = mCamZ;
487  mCamZoomPos += wheelSpeed * transScale * (mCamZoomPos/-100.0);
488  break;
489  case 5: // wheel up
490  if(mCamZoomPos>mCamZ) mCamZoomPos = mCamZ;
491  mCamZoomPos -= wheelSpeed * transScale * (mCamZoomPos/-100.0);
492  break;
493  default:
494  //cout << "DotVisualizer::glutFuncMouse Error: unknown button ??? " << button << "\n"; // debug
495  break;
496  }
497 
498  // check if the map was clicked, and cam moved
499  if(mOnMap) {
500  float dx = newx - mCamX;
501  float dy = newy - mCamY;
502  mCamX += dx;
503  mCamY += dy;
504  mLookatX += dx;
505  mLookatY += dy;
506  mTrackTarget = false;
507 
508  redraw();
509  mpGLC->set_camera_position( mCamX, mCamY, mCamZ, mLookatX, mLookatY );
510  mpMapC->set_camera_position( mCamX, mCamY, mCamZ, mLookatX, mLookatY );
511  } else {
512  }
513  mLastAction = getTime();
514 }

517 //-----------------------------------------------------------------------------
518 void DotVisualizer::handleDragging(int x, int y)
519 {
520  if(!mpGraph) return;
521  /* translate? */
522  if( (!mOnMap) &&
523  (!mOnNode)
524  ){
525  if(mButtonCurrent==VIS_BUTTONL) {
526  mCamX += (mMouseLastX-x) * -transScale * (mCamZ/((mWindowY/(float)mWindowX)*mWindowX*-0.25));
527  mCamY += (mMouseLastY-y) * transScale * (mCamZ/(mWindowY*-0.25));
528  mLookatX += (mMouseLastX-x) * -transScale * (mCamZ/((mWindowY/(float)mWindowX)*mWindowX*-0.25));
529  mLookatY += (mMouseLastY-y) * transScale * (mCamZ/(mWindowY*-0.25));
530  }
531  if( (mButtonCurrent== (VIS_BUTTONL|VIS_BUTTONR) ) ||
532  (mButtonCurrent==VIS_BUTTONM) ) {
533  mCamZ += (mMouseLastY-y) * transScale * (mCamZ/-100.0);
534  if(mCamZ > -minZDist) mCamZ = -minZDist;
535  mCamZoomPos = mCamZ;
536  }
537  /* rotate? */
538  if(mButtonCurrent==VIS_BUTTONR) {
539  mLookatX += (mMouseLastX-x) * -transScale * (mCamZ/((mWindowY/(float)mWindowX)*mWindowX*-0.25));
540  mLookatY += (mMouseLastY-y) * transScale * (mCamZ/(mWindowY*-0.25));
541  }
542  mpGLC->set_mousepos(x,y);
543  } else {
544  if(mOnMap) {
545  float newx, newy; // map click?
546  mpMapC->check_click(x,y, &newx, &newy);
547  float dx = newx - mCamX;
548  float dy = newy - mCamY;
549  mCamX += dx;
550  mCamY += dy;
551  mLookatX += dx;
552  mLookatY += dy;
553  mpGLC->set_mousepos(-1,-1);
554  }
555  if(mOnNode) {
556  return;
557  }
558  }
559 
560  redraw();
561  mpGLC->set_camera_position( mCamX, mCamY, mCamZ, mLookatX, mLookatY );
562  mpMapC->set_camera_position( mCamX, mCamY, mCamZ, mLookatX, mLookatY );
563 
564  mMouseLastX = x;
565  mMouseLastY = y;
566  mLastAction = getTime();
567 }

586 //-----------------------------------------------------------------------------
587 // handle animations
588 void DotVisualizer::animate( long t )
589 {
590  if(!mpGraph) return;
591  bool updateCam = false;
592  // fading
593  if((mLastAction>0)&&(t-mLastAction>MAP_FADEOUT_DELAY)) {
594  if(t-mLastAction > (5000+MAP_FADEOUT_DELAY)) {
595  mMapFade = 0.2;
596  } else {
597  float fadestate = 1.0- ((t-mLastAction-MAP_FADEOUT_DELAY)/ MAP_FADEOUT_SPEED);
598  mMapFade = 0.2+(fadestate*0.8);
599  }
600  } else {
601  if(mMapFade<1.0) mMapFade += (t-mLastTime)/ MAP_FADEIN_SPEED;
602  if(mMapFade>1.0) mMapFade = 1.0;
603  if(mLastAction<0) mLastAction = getTime();
604  }
605  mpMapC->set_fade( mMapFade );
606 
607  // zoom
608  if( mCamZoomPos > -minZDist ) mCamZoomPos = -minZDist;
609  if( ABS(mCamZ-mCamZoomPos)>2.0 ) {
610  if((t-mLastTime)> MAP_ZOOMSPEED) mCamZ = mCamZoomPos;
611  else {
612  mCamZ += (-mCamZ+mCamZoomPos)*(t-mLastTime)/MAP_ZOOMSPEED;
613  //cout << mCamZ << " " << mCamZoomPos << endl;
614  }
615  updateCam = true;
616  }
617 
618  // track targets
619  if(mTrackTarget) {
620  // camera pos
621  if( ABS(mCamX-mTargetX)>2.0 ) {
622  if((t-mLastTime)> TRACK_CAMSPEED) mCamX = mTargetX;
623  else {
624  mCamX += (-mCamX+mTargetX)*(t-mLastTime)/TRACK_CAMSPEED;
625  }
626  updateCam = true;
627  }
628  if( ABS(mCamY-mTargetY)>2.0 ) {
