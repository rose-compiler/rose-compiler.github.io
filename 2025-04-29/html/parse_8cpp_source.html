140 /*QY: return the min length of tokens placed in the filter res*/
141 void EvaluatePOET::
142 compute_lookaheadInfo(POETCode* cur, std::vector<POETCode*>& res, unsigned need, POETCode* next)
143 {
144  try {
145  assert (cur != 0);
146  int size = 0;
147  if (debug_lookahead()) {
148  std::cerr << "try compute lookahead for: " << cur->toString() << "; need " << need << "\n";
149  }
150  switch (cur->get_enum()) {
151  case SRC_TYPE:
152  case SRC_STRING:
153  case SRC_ICONST:
154  if (cur != ANY && cur != EMPTY) { res.push_back(cur); size = 1; }
155  break;
156  case SRC_CVAR: {
157  CodeVar* cvar = static_cast<CodeVar*>(cur);
158  POETCode* parse=cvar->get_parseInfo();
159  unsigned lookahead = cvar->get_entry().get_lookahead();
160  if (need < lookahead) need = lookahead;
161  if (parse != ZERO && parse != EMPTY) {
162  compute_lookaheadInfo(parse, res, need, next);
163  size=need; break;
164  }
165  else if (parse == EMPTY) break;
166  res.push_back(cvar); /*QY: the code var is a token */
167  size = 1;
168  break;
169  }
170  case SRC_LVAR: {
171  LocalVar* lvar = static_cast<LocalVar*>(cur);
172  POETCode* restr = 0;
173  switch (lvar->get_entry().get_entry_type())
174  {
175  case LVAR_CODEPAR: case LVAR_ATTR:
176  restr = lvar->get_entry().get_restr(); break;
177  default: restr = lvar->get_entry().get_code();
178  break;
179  }
180  if (restr == 0) { LOOKAHEAD_AMBIGUOUS(cur); }
181  else compute_lookaheadInfo(restr,res,need, next);
182  size=need; break;
183  }
184  case SRC_LIST: {
185  ASTFactory* fac = ASTFactory::inst();
186  POETList* curlist = static_cast<POETList*>(cur);
187  if (curlist->get_rest() != 0) next = fac->new_list(curlist->get_rest(), next);
188  compute_lookaheadInfo(curlist->get_first(),res,need, next);
189  size=need; break;
190  }
191  case SRC_OP : {
192  POETOperator* op = static_cast<POETOperator*>(cur);
193  unsigned n1=0;
194  switch(op->get_op()) {
195  case TYPE_TOR: {
196  POETTypeTor* tor = static_cast<POETTypeTor*>(cur);
197  tor->compute_lookaheadInfo(res, need,next);
198  size=need; break;
199  }
200  case POET_OP_LIST1:
201  case TYPE_LIST1:
202  case POET_OP_LIST:
203  case TYPE_LIST:
204  if (op->get_arg(1) != 0) next = POETProgram::make_typeTor(fac->new_list(op->get_arg(1),op->get_arg(0)),next);
205  else next = POETProgram::make_typeTor(op->get_arg(0),next);
206  compute_lookaheadInfo(op->get_arg(0),res,need, next);
207  size=need; break;
208  case POET_OP_EXP: compute_exp_lookahead(res,next); size=need; break;
209  case POET_OP_SEQ:
210  compute_lookaheadInfo(op->get_arg(1),res,need,next); size=need; break;
211  case POET_OP_APPLY: break;
212  default:
213  LOOKAHEAD_AMBIGUOUS(cur); break;
214  }
215  }
216  break;
217  case SRC_ASSIGN : {
218  POETAssign* op = static_cast<POETAssign*>(cur);
219  compute_lookaheadInfo(op->get_rhs(), res, need, next);
220  size=need; break;
221  }
222  default:
223  LOOKAHEAD_AMBIGUOUS(cur); break;
224  }
225  if (size < need) {
226  need = need - size;
227  if (next == 0) { LOOKAHEAD_AMBIGUOUS(EMPTY); }
228  else if (res.size() == 0) {
229  compute_lookaheadInfo(next, res, need, 0);
230  }
231  else {
232  std::vector<POETCode*> tailres;
233  compute_lookaheadInfo(next, tailres, need, 0);
234  if (tailres.size() > 0) {
235  int size = res.size();
236  for (int i = 0; i < size; ++i) {
237  POETCode* first = res[i];
238  res[i] = fac->new_list(first,tailres[0]);
239  for (int j = 1; j < tailres.size(); ++j)
240  res.push_back(fac->new_list(first,tailres[j]));
241  }
242  }
243  }
244  }
245  if (debug_lookahead()) {
246  std::cerr << "lookahead for: " << cur->toString() << "; need " << need << ":";
247  for (int i = 0; i < res.size(); ++i)
248  std::cerr << res[i]->toString() << ";";
249  std::cerr << "\n";
250  }
251  }
252  catch (Error err) { std::cerr << " From computing filter information for " << cur->toString() << "\n"; throw err; }
253 }

255 POETCode* EvaluatePOET::match_lookahead(POETCode* r1, POETCode* cur_filter)
256 {
257  if (r1 == EMPTY_LIST) return r1;
258  POETCode* r1_first = get_head(r1);
259  if (r1_first == cur_filter || AST2Iconst(r1_first) == cur_filter || AST2Iconst(cur_filter) == r1_first) {
260  r1= NextToken(r1,0);
261  return r1;
262  }
263  switch (cur_filter->get_enum()) {
264  case SRC_STRING:
265  case SRC_ICONST: return r1; /* failed */
266  case SRC_CVAR: /* a code template token */
267  if (r1_first->get_enum() == SRC_CVAR &&
268  static_cast<CodeVar*>(r1_first)->get_entry() ==
269  static_cast<CodeVar*>(cur_filter)->get_entry())
270  return NextToken(r1,0); /* success */
271  else return r1; /* failure */
272  case SRC_TYPE:
273  if (match_Type(r1_first, static_cast<POETType*>(cur_filter),
274  MATCH_AST_PATTERN))
275  return NextToken(r1,0); /* success */
276  return r1; /* failure */
277  case SRC_LIST: /*QY: multiple lookahead tokens*/
278  {
279  r1_first = r1;
280  POETList* l = static_cast<POETList*>(cur_filter);
281  POETCode *res = match_lookahead(r1, l->get_first());
282  if (res == r1 || res == 0) return r1_first; /* failed */
283  if (l->get_rest() != 0) {
284  r1 = res;
285  res = match_lookahead(r1, l->get_rest());
286  if (res == r1 || res == 0) return r1_first; /* failed */
287  }
288  return res; /* success */
289  }
290  case SRC_OP: {
291  POETParseList* parse = dynamic_cast<POETParseList*>(cur_filter);
292  if (parse != 0 && parse->match_lookahead(r1)) return 0; /*succ*/
293  else return r1; /* fail*/
294  }
295  default: std::cerr << "Unexpected: " << cur_filter->toString() << "\n"; assert(0);
296  }
297  return r1;
298  }

481  if (l1 == stop) { rest = l1; return EMPTY; }
