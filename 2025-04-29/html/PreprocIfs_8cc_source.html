166 genCloseDirectives (const CPreproc::If::Case* c,
167  PreprocessingInfo::RelativePositionType pos,
168  PreprocInfoList_t& D)
169 {
170  if (c)

172  // Close 'c'.
173 
174  // Build a debugging comment so we can see to whom this new
175  // '#endif' belongs.
176  stringstream s_closer;
177  const CPreproc::If::Case* c_open = getOpenCase (c);
178  ROSE_ASSERT (c_open);
179  if (c_open != c)
180  s_closer << c_open->getDirective ()
181  << ' '
182  << c_open->getCondition ()
183  << " ... ";
184 
185  s_closer << c->getDirective ()
186  << ' '
187  << c->getCondition ();
188 
189  // Build preprocessing info object for this string.
190  string closer = "#endif /* " + makeCommentSafe (s_closer.str ()) + " */\n";
191  PreprocessingInfo* d =
192  new PreprocessingInfo (PreprocessingInfo::CpreprocessorEndifDeclaration,
193  closer,
194  string ("transformation"), 0, 0,
195  1 /* no. of lines */,
196  pos);
197  ROSE_ASSERT (d);
198 
199  // DQ (3/12/2019): We need to mark the added comments and CPP directives as a transformation so that then can be output.
200  // This is a result of a fix to support the correct handling of comments and CPP directives for shared IR nodes as happen
201  // when multiple files are used on the command line.
202  d->get_file_info()->setTransformation();
203 
204  D.push_back (d);
205 
206  // Close all parents.
207  genCloseDirectives (getParentCase (c), pos, D);

386 {

392  {

400  info->insert(info->begin(),*i);
401 
402  break;
403  }
404 
405  case e_lastInBlock:
406  copy (D.begin (), D.end (), back_inserter (*ASTtools::createInfoList (b)));
