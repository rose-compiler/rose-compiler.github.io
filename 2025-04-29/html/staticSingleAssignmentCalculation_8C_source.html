54 bool StaticSingleAssignment::isVarInScope(const VarName& var, SgNode* astNode)
55 {
56  SgScopeStatement* accessingScope = SageInterface::getScope(astNode);
57  ROSE_ASSERT(var.size() > 0 && accessingScope != NULL);
58  SgScopeStatement* varScope = SageInterface::getScope(var[0]);
59 
60  //Work around a ROSE bug that sets incorrect scopes for built-in variables.
61  if (isBuiltinVar(var))
62  {
63  return SageInterface::isAncestor(accessingScope, var[0]);
64  }
65 
66  if (varScope == accessingScope || SageInterface::isAncestor(varScope, accessingScope))
67  {
68  //FIXME: In a basic block, the definition of the variable might come AFTER the node in question
69  //We should return false in this case.
70 
71  //Special case: a variable cannot be accessed in its own assign initializer
72  //This is important for loops where a variable is redefined on every iteration
73  //E.g. while (int a = 3) {}
74  if (SageInterface::isAncestor(var[0], astNode))
75  return false;
76 
77  return true;
78  }
79 
80  if (isSgNamespaceDefinitionStatement(varScope) || isSgGlobal(varScope))
81  {
82  //Variables defined in a namespace or in global scope are always accessible if they're fully qualified
83  return true;
84  }
85 
86  if (isSgInitializedName(astNode) && isSgCtorInitializerList(astNode->get_parent()))
87  {
88  //Work around a SageInterface::getScope peculiarity
89  //SageInterface::getScope returns class scope for the initialized names in the constructor initializer list,
90  //because these are class-scoped variables. However, those initializers can actually access function parameters
91  SgFunctionDeclaration* funcDeclaration = isSgFunctionDeclaration(astNode->get_parent()->get_parent());
92  ROSE_ASSERT(funcDeclaration != NULL);
93  SgFunctionParameterList* parameters = funcDeclaration->get_parameterList();
94  const SgInitializedNamePtrList& paramList = parameters->get_args();
95  //If the variable in question in the parameter list, it can be reached by the constructor initializers
96  if (find(paramList.begin(), paramList.end(), var[0]) != paramList.end())
97  {
98  return true;
99  }
100  }
101 
102  if (SgClassDefinition * varClassScope = isSgClassDefinition(varScope))
103  {
104  //If the variable is static & public, it's accessible
105  SgVariableDeclaration* varDeclaration = isSgVariableDeclaration(var[0]->get_parent());
106  if (varDeclaration != NULL) //variable declaration is null inside constructor initializer list
107  {
108  if (varDeclaration->get_declarationModifier().get_storageModifier().isStatic() &&
109  varDeclaration->get_declarationModifier().get_accessModifier().isPublic())
110  {
111  return true;
112  }
113  }
114 
115  //If the variable is accessed by a friend function, then it is available. Check if the function
116  //looking to access the var is a friend
117  SgFunctionDeclaration* accessingFunction = SageInterface::getEnclosingFunctionDeclaration(astNode, true);
118  ROSE_ASSERT(accessingFunction != NULL);
119  SgName accessingFunctionName = accessingFunction->get_mangled_name();
120 
121  //We'll look at all functions declared inside the variables class and see if any of them is the accessing function
122  //and is declared a friend
123 
124  foreach(SgDeclarationStatement* varClassMember, varClassScope->get_members())
125  {
126  if (!isSgFunctionDeclaration(varClassMember))
127  continue;
128 
129  SgFunctionDeclaration* nestedFunction = isSgFunctionDeclaration(varClassMember);
130 
131  if (!nestedFunction->get_declarationModifier().isFriend())
132  continue;
133 
134  if (nestedFunction->get_mangled_name() != accessingFunctionName)
135  continue;
136 
137  //The accessing function is a friend, so the variable is in scope
138  return true;
139  }
140 
141  //The variable is a class member; see if the accessing function is a member function of the same class
142  SgMemberFunctionDeclaration* memFunction = isSgMemberFunctionDeclaration(accessingFunction);
143  if (memFunction == NULL)
144  return false;
145 
146  SgScopeStatement* funcScope = memFunction->get_class_scope();
147  ROSE_ASSERT(funcScope != NULL);
148 
149  //If they are members of the same class, we're done
150  if (funcScope == varClassScope)
151  {
152  return true;
153  }
154 
155  SgDeclarationStatement * decl = NULL;
156  SgClassDefinition* funcClassScope = isSgClassDefinition(funcScope);
157  SgDeclarationScope* funcNonrealScope = isSgDeclarationScope(funcScope);
158  if (funcClassScope != NULL) {
159  decl = funcClassScope->get_declaration();
160  } else if (funcNonrealScope != NULL) {
161  decl = isSgDeclarationStatement(funcNonrealScope->get_parent());
162  } else {
163  ROSE_ABORT();
164  }
165 
166  //The two are not from the same class. Let's see if there is a friend class declaration
167 
168  foreach(SgDeclarationStatement* varClassMember, varClassScope->get_members())
169  {
170  SgClassDeclaration* nestedDeclaration = isSgClassDeclaration(varClassMember);
171  if (nestedDeclaration == NULL)
172  continue;
173 
174  if (nestedDeclaration->get_declarationModifier().isFriend())
175  {
176  //The variable's class has friend class. Check if the member function in question is in that friend
177  if (nestedDeclaration->get_firstNondefiningDeclaration() ==
178  decl->get_firstNondefiningDeclaration())
179  {
180  return true;
181  }
182  }
183  }
184 
185  //The variable is not in the same class and there is no friend class declaration, but we need to check the inheritance tree
186  //We do a search of the inheritance tree; this will terminate because it's a DAG
187  set<SgBaseClass*> worklist;
188  if (funcClassScope != NULL) {
189  worklist.insert(funcClassScope->get_inheritances().begin(), funcClassScope->get_inheritances().end());
190  }
191 
192  while (!worklist.empty())
193  {
194  SgBaseClass* baseClass = *worklist.begin();
195  worklist.erase(worklist.begin());
196 
197  //Get the class definition so we can get its base classes
198  SgClassDeclaration* definingDeclaration = isSgClassDeclaration(baseClass->get_base_class()->get_definingDeclaration());
199  if (definingDeclaration == NULL)
200  continue;
201 
202  SgClassDefinition* baseClassDefinition = isSgClassDefinition(definingDeclaration->get_definition());
203  ROSE_ASSERT(baseClassDefinition != NULL);
204 
205  foreach(SgBaseClass* grandparentClass, baseClassDefinition->get_inheritances())
206  {
207  worklist.insert(grandparentClass);
208  }
209 
210  //Check if this base class matches the var scope
211  if (baseClassDefinition == varClassScope)
212  {
213  //Check that the variable is public or protected
214  ROSE_ASSERT(varDeclaration != NULL);
215  const SgAccessModifier& access = varDeclaration->get_declarationModifier().get_accessModifier();
216  return (access.isPublic() || access.isProtected());
217  }
218  }
219  }
220 
221  return false;
222 }

224 void StaticSingleAssignment::run(bool interprocedural, bool treatPointersAsStructures)
225 {
226  originalDefTable.clear();
227  expandedDefTable.clear();
228  reachingDefsTable.clear();
229  localUsesTable.clear();
230  useTable.clear();
231  ssaLocalDefTable.clear();
232 
233 #if DISPLAY_TIMINGS
234  timer time;
235 #endif
236  if (getDebug())
237  cout << "Running UniqueNameTraversal...\n";
238  UniqueNameTraversal uniqueTrav(
239  SageInterface::querySubTree<SgInitializedName > (project, V_SgInitializedName), treatPointersAsStructures);
240  uniqueTrav.traverse(project);
241  if (getDebug())
242  cout << "Finished UniqueNameTraversal." << endl;
243 #if DISPLAY_TIMINGS
244  printf("-- Timing: UniqueNameTraversal took %.2f seconds.\n", time.elapsed());
245  fflush(stdout);
246  time.restart();
247 #endif
248 
249  //Get a list of all the functions that we'll process
250  boost::unordered_set<SgFunctionDefinition*> interestingFunctions;
251  vector<SgFunctionDefinition*> funcs = SageInterface::querySubTree<SgFunctionDefinition > (project, V_SgFunctionDefinition);
252 
253  FunctionFilter functionFilter;
254 
255  foreach(SgFunctionDefinition* f, funcs)
256  {
257  if (functionFilter(f->get_declaration()))
258  interestingFunctions.insert(f);
259  }
260 #if DISPLAY_TIMINGS
261  printf("-- Timing: Creating list of functions took %.2f seconds.\n", time.elapsed());
262  fflush(stdout);
263  time.restart();
264 #endif
265 
266  DefsAndUsesTraversal defUseTrav(this, treatPointersAsStructures);
267 
268  //Generate all local information before doing interprocedural analysis. This is so we know
269  //what variables are directly modified in each function body before we do interprocedural propagation
270 
271  foreach(SgFunctionDefinition* func, interestingFunctions)
272  {
273  if (getDebug())
274  cout << "Running DefsAndUsesTraversal on function: " << SageInterface::get_name(func) << func << endl;
275 
276  defUseTrav.traverse(func->get_declaration());
277 
278  if (getDebug())
279  cout << "Finished DefsAndUsesTraversal..." << endl;
280 
281  //Expand any member variable definition to also define its parents at the same node
282  expandParentMemberDefinitions(func->get_declaration());
283 
284  //Expand any member variable uses to also use the parent variables (e.g. a.x also uses a)
285  expandParentMemberUses(func->get_declaration());
286 
287  insertDefsForChildMemberUses(func->get_declaration());
288  }
289 
290 #if DISPLAY_TIMINGS
291  printf("-- Timing: Inserting all local defs for %" PRIuPTR " functions took %.2f seconds.\n",
292  interestingFunctions.size(), time.elapsed());
293  fflush(stdout);
294  time.restart();
295 #endif
296 
297  //Interprocedural iterations. We iterate on the call graph until all interprocedural defs are propagated
298  if (interprocedural)
299  {
300  interproceduralDefPropagation(interestingFunctions);
301  }
302 
303 #if DISPLAY_TIMINGS
304  printf("-- Timing: Interprocedural propagation took %.2f seconds.\n", time.elapsed());
305  fflush(stdout);
306  time.restart();
307 #endif
308 
309  //Now we have all local information, including interprocedural defs. Propagate the defs along control-flow
310 
311  foreach(SgFunctionDefinition* func, interestingFunctions)
312  {
313  vector<FilteredCfgNode> functionCfgNodesPostorder = getCfgNodesInPostorder(func);
314 
315  //Insert definitions at the SgFunctionDefinition for external variables whose values flow inside the function
316  insertDefsForExternalVariables(func->get_declaration());
317 
318  //Create all ReachingDef objects:
319  //Create ReachingDef objects for all original definitions
320  populateLocalDefsTable(func->get_declaration());
321  //Insert phi functions at join points
322  multimap< FilteredCfgNode, pair<FilteredCfgNode, FilteredCfgEdge> > controlDependencies =
323  insertPhiFunctions(func, functionCfgNodesPostorder);
324 
325  //Renumber all instantiated ReachingDef objects
326  renumberAllDefinitions(func, functionCfgNodesPostorder);
327 
328  if (getDebug())
329  cout << "Running DefUse Data Flow on function: " << SageInterface::get_name(func) << func << endl;
330  runDefUseDataFlow(func);
331 
332  //We have all the propagated defs, now update the use table
333  buildUseTable(functionCfgNodesPostorder);
334 
335  //Annotate phi functions with dependencies
336  //annotatePhiNodeWithConditions(func, controlDependencies);
337  }

340 void StaticSingleAssignment::expandParentMemberDefinitions(SgFunctionDeclaration* function)
341 {
342 
343  class ExpandDefsTraversal : public AstSimpleProcessing
344  {
345  public:
346  StaticSingleAssignment* ssa;
347 
348  void visit(SgNode* node)
349  {
350  if (ssa->originalDefTable.count(node) == 0)
351  return;
352 
353  //We want to iterate the vars defined on this node, and expand them
354 
355  foreach(const VarName& definedVar, ssa->originalDefTable[node])
356  {
357  if (getDebugExtra())
358  {
359  cout << "Checking [" << varnameToString(definedVar) << "]" << endl;
360  }
361 
362  //Check if the variableName has multiple parts
363  if (definedVar.size() == 1)
364  {
365  continue;
366  }
367 
368  //We are dealing with a multi-part variable, loop the entry and expand it
369  //Start at one so we don't get the same defs in the original and expanded defs
370  for (unsigned int i = 1; i < definedVar.size(); i++)
371  {
372  //Create a new varName vector that goes from beginning to end - i
373  VarName newName;
374  newName.assign(definedVar.begin(), definedVar.end() - i);
375 
376  if (getDebugExtra())
377  {
378  cout << "Testing for presence of [" << varnameToString(newName) << "]" << endl;
379  }
380 
381  //Only insert the new definition if it does not already exist in the original def table
382  if (ssa->originalDefTable[node].count(newName) == 0)
383  {
384  //Insert the new name as being defined here.
385  ssa->expandedDefTable[node].insert(newName);
386 
387  if (getDebugExtra())
388  {
389  cout << "Inserted new name [" << varnameToString(newName) << "] into defs." << endl;
390  }
391  }
392  }
393  }
394  }
395  };
396 
397  ExpandDefsTraversal trav;
398  trav.ssa = this;
399  trav.traverse(function, preorder);
400 }

464 void StaticSingleAssignment::runDefUseDataFlow(SgFunctionDefinition* func)
465 {
466  if (getDebug())
467  printOriginalDefTable();
468  //Keep track of visited nodes
469  boost::unordered_set<SgNode*> visited;
470 
471  set<FilteredCfgNode> worklist;
472 
473  FilteredCfgNode current= FilteredCfgNode(func->cfgForBeginning());
474  worklist.insert(current);
475 
476  while (!worklist.empty())
477  {
478  if (getDebugExtra())
479  cout << "-------------------------------------------------------------------------" << endl;
480  //Get the node to work on
481  current = *worklist.begin();
482  worklist.erase(worklist.begin());
483 
484  //Propagate defs to the current node
485  bool changed = propagateDefs(current);
486 
487  //For every edge, add it to the worklist if it is not seen or something has changed
488 
489  reverse_foreach(const FilteredCfgEdge& edge, current.outEdges())
490  {
491  FilteredCfgNode nextNode = edge.target();
492 
493  //Insert the child in the worklist if the parent is changed or it hasn't been visited yet
494  if (changed || visited.count(nextNode.getNode()) == 0)
495  {
496  //Add the node to the worklist
497  bool insertedNew = worklist.insert(nextNode).second;
498  if (insertedNew && getDebugExtra())
499  {
500  if (changed)
501  cout << "Defs Changed: Added " << nextNode.getNode()->class_name() << nextNode.getNode() << " to the worklist." << endl;
502  else
503  cout << "Next unvisited: Added " << nextNode.getNode()->class_name() << nextNode.getNode() << " to the worklist." << endl;
504  }
505  }
506  }
507 
508  //Mark the current node as seen
509  visited.insert(current.getNode());
510  }
511 }

513 bool StaticSingleAssignment::propagateDefs(FilteredCfgNode cfgNode)
514 {
515  SgNode* node = cfgNode.getNode();
516 
517  //This updates the IN table with the reaching defs from previous nodes
518  updateIncomingPropagatedDefs(cfgNode);
519 
520  //Special Case: the OUT table at the function definition node actually denotes definitions at the function entry
521  //So, if we're propagating to the *end* of the function, we shouldn't update the OUT table
522  if (isSgFunctionDefinition(node) && cfgNode == FilteredCfgNode(node->cfgForEnd()))
523  {
524  return false;
525  }
526 
527  //Create a staging OUT table. At the end, we will check if this table
528  //Was the same as the currently available one, to decide if any changes have occurred
529  //We initialize the OUT table to the IN table
530  NodeReachingDefTable outDefsTable = reachingDefsTable[node].first;
531 
532  //Special case: the IN table of the function definition node actually denotes
533  //definitions reaching the *end* of the function. So, start with an empty table to prevent definitions
534  //from the bottom of the function from propagating to the top.
535  if (isSgFunctionDefinition(node) && cfgNode == FilteredCfgNode(node->cfgForBeginning()))
536  {
537  outDefsTable.clear();
538  }
539 
540  //Now overwrite any local definitions:
541  if (ssaLocalDefTable.count(node) > 0)
542  {
543 
544  foreach(const NodeReachingDefTable::value_type& varDefPair, ssaLocalDefTable[node])
545  {
546  const VarName& definedVar = varDefPair.first;
547  ReachingDefPtr localDef = varDefPair.second;
548 
549  outDefsTable[definedVar] = localDef;
550  }
551  }
552 
553  //Compare old to new OUT tables
554  bool changed = (reachingDefsTable[node].second != outDefsTable);
555  if (changed)
556  {
557  reachingDefsTable[node].second = outDefsTable;
558  }
559 
560  return changed;
561 }

563 void StaticSingleAssignment::updateIncomingPropagatedDefs(FilteredCfgNode cfgNode)
564 {
565  //Get the previous edges in the CFG for this node
566  vector<FilteredCfgEdge> inEdges = cfgNode.inEdges();
567  SgNode* astNode = cfgNode.getNode();
568 
569  NodeReachingDefTable& incomingDefTable = reachingDefsTable[astNode].first;
570 
571  //Iterate all of the incoming edges
572  for (unsigned int i = 0; i < inEdges.size(); i++)
573  {
574  SgNode* prev = inEdges[i].source().getNode();
575 
576  const NodeReachingDefTable& previousDefs = reachingDefsTable[prev].second;
577 
578  //Merge all the previous defs into the IN table of the current node
579 
580  foreach(const NodeReachingDefTable::value_type& varDefPair, previousDefs)
581  {
582  const VarName& var = varDefPair.first;
583  const ReachingDefPtr previousDef = varDefPair.second;
584 
585  //Here we don't propagate defs for variables that went out of scope
586  //(built-in vars are body-scoped but we inserted the def at the SgFunctionDefinition node, so we make an exception)
587  if (!isVarInScope(var, astNode) && !isBuiltinVar(var))
588  continue;
589 
590  //If this is the first time this def has propagated to this node, just copy it over
591  if (incomingDefTable.count(var) == 0)
592  {
593  incomingDefTable[var] = previousDef;
594  }
595  else
596  {
597  ReachingDefPtr existingDef = incomingDefTable[var];
598 
599  if (existingDef->isPhiFunction() && existingDef->getDefinitionNode() == astNode)
600  {
601  //There is a phi node here. We update the phi function to point to the previous reaching definition
602  existingDef->addJoinedDef(previousDef, inEdges[i]);
603  }
604  else
605  {
606  //If there is no phi node, and we get a new definition, it better be the same as the one previously
607  //propagated.
608  if (!(*previousDef == *existingDef))
609  {
610  printf("ERROR: At node %s@%d, two different definitions reach for variable %s\n",
611  astNode->class_name().c_str(), astNode->get_file_info()->get_line(), varnameToString(var).c_str());
612  ROSE_ABORT();
613  }
614  }
615  }
616  }
617  }

620 void StaticSingleAssignment::buildUseTable(const vector<FilteredCfgNode>& cfgNodes)
621 {
622 
623  foreach(const FilteredCfgNode& cfgNode, cfgNodes)
624  {
625  SgNode* node = cfgNode.getNode();
626 
627  if (localUsesTable.count(node) == 0)
628  continue;
629 
630  foreach(const VarName& usedVar, localUsesTable[node])
631  {
632  //Check the defs that are active at the current node to find the reaching definition
633  //We want to check if there is a definition entry for this use at the current node
634  if (reachingDefsTable[node].first.count(usedVar) > 0)
635  {
636  useTable[node][usedVar] = reachingDefsTable[node].first[usedVar];
637  }
638  else
639  {
640  // There are no defs for this use at this node, this shouldn't happen
641  printf("Error: Found use for the name '%s', but no reaching defs!\n", varnameToString(usedVar).c_str());
642  printf("Node is %s:%d in %s\n", node->class_name().c_str(), node->get_file_info()->get_line(),
643  node->get_file_info()->get_filename());
644  continue; // FIXME ROSE-1392
645  ROSE_ABORT();
646  }
647  }
648  }
649 
650  if (getDebug())
651  {
652  printf("Local uses table:\n");
653  printLocalDefUseTable(localUsesTable);
654  }

659 {

662  {
