82  // insert Phi functions where needed
83  CFG::NonLocalsIterator i(*cfg);

85  std::set<CFG::Node*> blks_with_phi;
86  while ((bool)i) {

90  CFG::DefBlocksIterator block(*cfg, var_name);

92  work_list.insert((CFG::Node*)block);
93  ++block;
94  }
95  std::set<CFG::Node*>::iterator b = work_list.begin();
96  while (b != work_list.end()) {
97  DomTree::DomFrontIterator d(dt.domtree_node(*b));
98  while ((bool)d) {
99  CFG::Node* blk = dynamic_cast<CFG::Node*>((DGraph::Node*)d);
100  if (blks_with_phi.find(blk) == blks_with_phi.end()) {
101  Phi* p = new Phi(var_name, cfg);
102  phi_node_sets[blk].insert(p);
103  cout << "inserted Phi node ";
104  p->dump(cout);
105  cout << " in node ";
106  blk->dump(cout);
107  cout << endl;
108  if (work_list.find(blk) == work_list.end())
109  work_list.insert(blk);
110  }
111  ++d;
112  }
113  ++b;
114  }
115  ++i;
116  }
