22  *

69  /* Bit flags for the NE header 'e_flags' member.
70  *
71  * If HF_LIBRARY (bit 15) is set then the CS:IP registers point to an initialization procedure called with the value in the AX
72  * register equal to the module handle. The initialization procedure must execute a far return to the caller. The resulting
73  * value in AX is a status indicator (non-zero for success, zero for failure). */
74  enum HeaderFlags1 {
75  HF1_RESERVED = 0x57f4, /* Reserved bits */
76  HF1_NO_DATA = 0x0000, /* (flags&0x03==0) => an exe not containing a data segment */
77  HF1_SINGLE_DATA = 0x0001, /* Executable contains one data segment; set if file is a DLL */
78  HF1_MULTIPLE_DATA = 0x0002, /* Exe with multiple data segments; set if a windows application */
79  HF1_LOADER_SEGMENT = 0x0800, /* First segment contains code that loads the application */
80  HF1_FATAL_ERRORS = 0x2000, /* Errors detected at link time; module will not load */
81  HF1_LIBRARY = 0x8000 /* Module is a library */
82  };
