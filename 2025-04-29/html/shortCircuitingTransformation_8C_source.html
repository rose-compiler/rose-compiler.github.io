203 // Split a variable declaration into two statements: the variable declaration
204 // itself and an expression statement setting the variable to the value specified in
205 // its initializer. Sets the initializer to null and does not insert the new expression
206 // statement itself, as it may be used in a different way (e.g. condition of if statement).
207 // Caveat: for constructor initializers this is not the correct behavior (as
208 // 0-argument constructor called first) but the best we
209 // can do if we want the initializer in an expression statement.
210 SgExprStatement *splitVarDecl(SgVariableDeclaration *varDecl)
211  {
212  SgInitializedName *in = varDecl->get_variables().front();
213 
214  SgInitializer *init = in->get_initializer();
215 
216  SgExpression *operand = NULL;
217  switch (init->variantT())
218  {
219  case V_SgAssignInitializer:
220  {
221  SgAssignInitializer *assignInit = isSgAssignInitializer(init);
222  ROSE_ASSERT(assignInit);
223 
224  operand = assignInit->get_operand_i();
225 
226  break;
227  }
228 
229  case V_SgConstructorInitializer:
230  {
231  SgConstructorInitializer *ctorInit = isSgConstructorInitializer(init);
232  ROSE_ASSERT(ctorInit != NULL);
233 
234  operand = ctorInit;
235  break;
236  }
237 
238  default:
239  {
240  cout << "splitVarDecl: unknown init type: " << init->class_name() << endl;
241  ROSE_ABORT();
242  }
243  }
244 
245  in->set_initializer(NULL);
246 
247  SgVariableSymbol *var = in->get_scope()->lookup_var_symbol(in->get_name());
248 
249  SgVarRefExp *varRefForAssign = new SgVarRefExp(SgNULL_FILE, var);
250  SgAssignOp *assignOp = new SgAssignOp(SgNULL_FILE, varRefForAssign, operand);
251  varRefForAssign->set_parent(assignOp);
252  operand->set_parent(assignOp);
253 
254  SgExprStatement *condExprStmt = new SgExprStatement(SgNULL_FILE, assignOp);
255  assignOp->set_parent(condExprStmt);
256 
257  return condExprStmt;
258  }
