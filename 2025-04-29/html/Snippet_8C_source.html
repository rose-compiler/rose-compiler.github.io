97 // Return the first non-empty statement from the specified source code, without the trailing semicolon. This returns the
98 // non-comment material from the beginning of the supplied string up to but not including the first semicolon that is not part
99 // of a comment or string literal. Comments are C++/Java style comments. Strings are delimited by single and double quotes
100 // (any number of characters in either) and support backslash as an escape mechanism. The source string is assumed to start
101 // outside a comment or string.
102 static std::string extractFirstStatement(const std::string &source)
103 {
104  std::string firstStmt;
105  std::string inComment; // comment state: "", "/", "//", "/*", or "/**"
106  char inString='\0'; // string state: '\0', '"' or '\''
107  bool escaped=false; // true if previous character was a backslash not in a comment
108  BOOST_FOREACH (char ch, source) {
109  if (inComment=="" && !inString && '/'==ch) {
110  assert(!escaped);
111  inComment = "/";
112  } else if (inComment=="/") {
113  assert(!escaped);
114  assert(!inString);
115  if ('/'==ch || '*'==ch) {
116  inComment += ch;
117  } else {
118  firstStmt += inComment + ch;
119  inComment = "";
120  if ('\''==ch || '"'==ch)
121  inString = ch;
122  escaped = '\\'==ch;
123  }
124  } else if (inComment=="//" && '\n'==ch) {
125  assert(!escaped);
126  assert(!inString);
127  inComment = "";
128  } else if (inComment=="/*" && '*'==ch) {
129  assert(!escaped);
130  assert(!inString);
131  inComment = "/**";
132  } else if (inComment=="/**") {
133  assert(!escaped);
134  assert(!inString);
135  if ('/'==ch) {
136  inComment = "";
137  } else {
138  inComment = "/*";
139  }
140  } else if (inComment=="//" || inComment=="/*") {
141  assert(!escaped);
142  assert(!inString);
143  } else if (inString) {
144  assert(inComment=="");
145  firstStmt += ch;
146  if (!escaped && ch==inString)
147  inString = '\0';
148  escaped = '\\'==ch;
149  } else if (';'==ch) {
150  boost::trim(firstStmt);
151  if (!firstStmt.empty())
152  break;
153  } else {
154  firstStmt += ch;
155  }
156  }
157  return boost::trim_copy(firstStmt);
158 }

392 {

394 
395  // Find statements that are calls to snippets, but don't do anything yet
396  struct FindSnippetCalls: SnippetAstTraversal {
397  SnippetFile *self;
398  SnippetCalls calls;
399 
400  FindSnippetCalls(SnippetFile *self): self(self) {}
401 
402  void operator()(SgNode *node, AstSimpleProcessing::Order when) {
403  if (preorder==when) {
404  if (SgExprStatement *stmt = isSgExprStatement(node)) {
405  if (SgFunctionCallExp *fcall = isSgFunctionCallExp(stmt->get_expression())) {
406  SgFunctionSymbol *fsym = fcall->getAssociatedFunctionSymbol();
407  SgFunctionDeclaration *fdecl = fsym ? fsym->get_declaration() : NULL;
408  std::string called_name = fdecl ? fdecl->get_qualified_name().getString() : std::string();
409  if (self->functions.exists(called_name))
410  calls.insert(std::make_pair(fcall, called_name));
411  }
412  }
413  }
414  }
415  } t1(this);
416  t1.traverse(ast);
417 
418  // Replace each of the snippet calls by expanding the snippet. The expansion occurs right before the snippet call and then
419  // the snippet call is removed.
420  for (SnippetCalls::iterator ci=t1.calls.begin(); ci!=t1.calls.end(); ++ci) {
421  SgFunctionCallExp *fcall = ci->first;
422  assert(fcall!=NULL);
423  const SgExpressionPtrList &fcall_args = fcall->get_args()->get_expressions();
424  std::vector<SgNode*> actuals(fcall_args.begin(), fcall_args.end());
425  SgStatement *toReplace = SageInterface::getEnclosingNode<SgStatement>(fcall);
426  SnippetPtr snippet;
427  BOOST_FOREACH (SnippetPtr candidate, findSnippets(ci->second)) {
428  if (candidate->numberOfArguments() == actuals.size()) {
429  if (snippet!=NULL) {
430  throw std::runtime_error("snippet call \"" + ci->second + "\" having " +
431  StringUtility::plural(actuals.size(), "arguments") +
432  " matches multiple snippet definitions");
433  }
434  snippet = candidate;
435  }
436  }
437  if (snippet==NULL) {
438  throw std::runtime_error("snippet call \"" + ci->second + "\" having " +
439  StringUtility::plural(actuals.size(), "arguments") +
440  " does not match any snippet definition");
441  }
442  snippet->insert(toReplace, actuals);
443  SageInterface::removeStatement(toReplace);
444  }

639 void

642  using namespace StringUtility;

644  assert(insertionPoint!=NULL);
645  assert(ast!=NULL);
646 
647  SgFunctionDefinition *targetFunction = SageInterface::getEnclosingNode<SgFunctionDefinition>(insertionPoint);
648  assert(targetFunction!=NULL);
649  SgScopeStatement *targetFunctionScope = targetFunction->get_body();
650  assert(targetFunctionScope!=NULL);
651 
652  // Find the first declaration statement in the insertion point's function
653  SgDeclarationStatement *targetFirstDeclaration = NULL;
654  SgStatementPtrList targetStatements = targetFunctionScope->generateStatementList();
655  for (size_t i=0; targetFirstDeclaration==NULL && i<targetStatements.size(); ++i)
656  targetFirstDeclaration = isSgDeclarationStatement(targetStatements[i]);
657  SgStatement *targetFirstStatement = targetStatements.empty() ? NULL : targetStatements.front();
658 
659  // Copy into the target file other functions, variables, imports, etc. that are above the snippet SgFunctionDefinition in
660  // the snippet's file but which the user wants copied nonetheless. Some of these things might be referenced by the
661  // snippet, and others might completely unrelated but the user wants them copied anyway.
662  insertRelatedThings(insertionPoint);
663 
664  // Insert the snippet body after the insertion point
665  SgTreeCopy deep;
666  SgScopeStatement *toInsert = isSgScopeStatement(ast->get_body()->copy(deep));
667  assert(toInsert!=NULL);
668 
669  // DQ (3/4/2014): This is a test of the structural equality of the original snippet and it's copy.
670  // If they are different then we can't support fixing up the AST. Transformations on the snippet
671  // should have been made after insertion into the AST. The complexity of this test is a traversal
672  // of the copy of the snippet to be inserted (typically very small compared to the target application).
673  // Note that we can enforce this test here, but not after calling the replaceArguments() function below.
674  bool isStructurallyEquivalent = SageInterface::isStructurallyEquivalentAST(toInsert,ast->get_body());
675  ROSE_ASSERT(isStructurallyEquivalent == true);
676 
677  // DQ (3/4/2014): I think this is untimately a fundamental problem later (e.g. for mangled name generation).
678  // So we have to attached the current scope to the scope of the insertion point. This will allow the
679  // SgStatement::get_scope() to work (which is a problem for some debugging code (at least).
680  // Note that the semantics of the AST copy mechanism is that the parent of the copy is set to NULL
681  // (which is assumed to be fixed up when the copy is inserted into the AST).
682  ROSE_ASSERT(toInsert->get_parent() == NULL);
683  SgScopeStatement* new_scope = isSgScopeStatement(insertionPoint->get_parent());
684  ROSE_ASSERT(new_scope != NULL);
685  toInsert->set_parent(new_scope);
686  ROSE_ASSERT(toInsert->get_parent() != NULL);
687 
688  renameTemporaries(toInsert);
689  causeUnparsing(toInsert, targetFunction->get_file_info());
690 
691  switch (insertMechanism) {
692  case INSERT_BODY: {
693  // Insert the body all at once. This is efficient but doesn't work well because it means that variables declared in
694  // one snippet can't be used in a later snippet injected into the same function.
695  file->addInsertionRecord(SnippetInsertion(toInsert, ast->get_body(), insertionPoint));
696  SageInterface::insertStatementBefore(insertionPoint, toInsert);
697  break;
698  }
699  case INSERT_STMTS: {
700  // Insert one statement at a time. Snippet declarations are placed at the top of the injection point's function,
701  // which means that snippet variables cannot always be inialized in their declarations because the initialization
702  // expression might be something that's only well defined at the point of insertion.
703  const SgStatementPtrList &stmts = toInsert->getStatementList();
704  const SgStatementPtrList &stmtsOrig = ast->get_body()->getStatementList();
705  assert(stmts.size()==stmtsOrig.size());
706  for (size_t i=0; i<stmts.size(); ++i) {
707  if (isSgDeclarationStatement(stmts[i])) {
708  switch (locDeclsPosition) {
709  case LOCDECLS_AT_BEGINNING:
710  if (targetFirstDeclaration!=NULL) {
711  file->addInsertionRecord(SnippetInsertion(stmts[i], stmtsOrig[i], targetFirstDeclaration));
712  SageInterface::insertStatementBefore(targetFirstDeclaration, stmts[i]);
713  } else {
714  file->addInsertionRecord(SnippetInsertion(stmts[i], stmtsOrig[i], targetFirstStatement));
715  SageInterface::insertStatementBefore(targetFirstStatement, stmts[i]);
716  }
717  break;
718  case LOCDECLS_AT_END:
719  file->addInsertionRecord(SnippetInsertion(stmts[i], stmtsOrig[i], targetFunctionScope));
720  SageInterface::insertStatementBeforeFirstNonDeclaration(stmts[i], targetFunctionScope);
721  break;
722  case LOCDECLS_AT_CURSOR:
723  file->addInsertionRecord(SnippetInsertion(stmts[i], stmtsOrig[i], insertionPoint));
724  SageInterface::insertStatementBefore(insertionPoint, stmts[i]);
725  break;
726  }
727  } else {
728  file->addInsertionRecord(SnippetInsertion(stmts[i], stmtsOrig[i], insertionPoint));
729  SageInterface::insertStatementBefore(insertionPoint, stmts[i]);
730  }
731  }
732  break;
733  }
734  }
735 
736  // Build a map binding formal argument symbols to their actual values. The goal is, in the statements we just inserted, to
737  // replace all references to snippet formal arguments with the actual arguments supplied to the Snippet::insert() call. The
738  // actual arguments are SgInitializedName or SgExpression nodes from the context in which the snippet was inserted; the
739  // references to formal arguments are the SgInitializedName nodes in the AST's that we just copied above (the
740  // *declarations* for the formal arguments were *not* copied because they are the get_parameterList() of the snippet
741  // functionin the snippet file--which we didn't insert into the target).
742  ArgumentBindings bindings; // map from formal arg references to actual arguments
743  SgFunctionDeclaration *snippet_fdecl = ast->get_declaration();

745  if (actuals.size()!=formals.size()) {

748  numberToString(actuals.size()));

750  for (size_t i=0; i<formals.size(); ++i) {
751  SgSymbol *formalSymbol = formals[i]->search_for_symbol_from_symbol_table();
752  assert(formalSymbol!=NULL);
753  if (actuals[i]==NULL)
754  throw std::runtime_error("Rose::Snippet: snippet '"+name+"' actual argument "+numberToString(i+1)+" is null");
755  if (!isSgInitializedName(actuals[i]) && !isSgExpression(actuals[i]))
756  throw std::runtime_error("Rose::Snippet: snippet '"+name+"' actual argument "+numberToString(i+1)+
757  " must be a variable declaration or expression"
758  " but has type "+actuals[i]->class_name());
759  bindings[formalSymbol] = actuals[i];
760  }
761  replaceArguments(toInsert, bindings);
762

765  if (insertRecursively)
766  file->expandSnippets(toInsert);
767

770 
771  // Build a translation map so that we can save the mapping of scopes between the target AST and the snippet AST.
772  // In the case of insertMechanism == INSERT_STMTS we also add the mapping of the scope used for the copy of the body
773  // to the parent of the insertionPoint so that symbols can be reset into the scope of the insertionPoint.
774  // This map connects snippet scopes (key) to target AST scopes (value).
775  // std::map<SgNode*,SgNode*> translationMap;
776 
777  // If we are inserting into the end of a scope then we point to the scope since there is no last statement
778  // to insert a statement before. In this case then insertionPointIsScope == true, else it is false.
779  bool insertionPointIsScope = false;
780 
781  if (insertMechanism == INSERT_BODY)
782  {
783  // Fill in the first entry to inlcude the mapping of the copy of the scope (body) to the associated scope of the
784  // insertionPoint.
785  SageBuilder::fixupCopyOfAstFromSeparateFileInNewTargetAst(insertionPoint, insertionPointIsScope, toInsert,
786  ast->get_body());
787  }
788  else
789  {
790  if (insertMechanism == INSERT_STMTS)
791  {
792  const SgStatementPtrList &stmts_copy_of_snippet_ast = toInsert->getStatementList();
793  const SgStatementPtrList &stmts_in_original_snippet_ast = ast->get_body()->getStatementList();
794  ROSE_ASSERT(stmts_copy_of_snippet_ast.size() == stmts_in_original_snippet_ast.size());

796  for (size_t i = 0; i < stmts_copy_of_snippet_ast.size(); ++i)

799  insertionPointIsScope = false;
800

802  {
803  switch (locDeclsPosition)
804  {
805  case LOCDECLS_AT_BEGINNING:
806  // Fill in the first entry to inlcude the mapping of the copy of the scope (body) to the
807  // associated scope of the insertionPoint.
808  if (targetFirstDeclaration) {
809  SageBuilder::fixupCopyOfAstFromSeparateFileInNewTargetAst(targetFirstDeclaration,
810  insertionPointIsScope,
811  stmts_copy_of_snippet_ast[i],
812  stmts_in_original_snippet_ast[i]);
813  } else {
814  SageBuilder::fixupCopyOfAstFromSeparateFileInNewTargetAst(targetFirstStatement,
815  insertionPointIsScope,
816  stmts_copy_of_snippet_ast[i],
817  stmts_in_original_snippet_ast[i]);
818  }
819  break;
820 
821  case LOCDECLS_AT_END:
822  // We are providing the scope instead of the declaration reference.
823  insertionPointIsScope = true;
824 
825  // Fill in the first entry to inlcude the mapping of the copy of the scope (body) to the
826  // associated scope of the insertionPoint.
827  SageBuilder::fixupCopyOfAstFromSeparateFileInNewTargetAst(targetFunctionScope,
828  insertionPointIsScope,
829  stmts_copy_of_snippet_ast[i],
830  stmts_in_original_snippet_ast[i]);
831  break;
832 
833  case LOCDECLS_AT_CURSOR:
834  SageBuilder::fixupCopyOfAstFromSeparateFileInNewTargetAst(insertionPoint,
835  insertionPointIsScope,
836  stmts_copy_of_snippet_ast[i],
837  stmts_in_original_snippet_ast[i]);
838  break;
839  }
840  }
841  else
842  {
843  // Fill in the first entry to inlcude the mapping of the copy of the scope (body) to the associated
844  // scope of the insertionPoint.
845  SageBuilder::fixupCopyOfAstFromSeparateFileInNewTargetAst(insertionPoint,insertionPointIsScope,
846  stmts_copy_of_snippet_ast[i],
847  stmts_in_original_snippet_ast[i]);
848  }
849  }
850  }
851  else

854  ROSE_ABORT();
855  }
856  }
