40 BinaryLoaderPe::rebase(const MemoryMap::Ptr &map, SgAsmGenericHeader *header, const SgAsmGenericSectionPtrList &sections) {
41  SgAsmPEFileHeader* pe_header = isSgAsmPEFileHeader(header);
42  ROSE_ASSERT(pe_header != NULL);
43  const size_t maximum_alignment = pe_header->get_e_section_align();
44  Address original = pe_header->get_baseVa();
45 
46  // Find the minimum address desired by the sections to be mapped.
47  Address min_preferred_rva = (uint64_t)(-1);
48  for (SgAsmGenericSectionPtrList::const_iterator si=sections.begin(); si!=sections.end(); ++si)
49  min_preferred_rva = std::min(min_preferred_rva, (*si)->get_mappedPreferredRva());
50  Address min_preferred_va = header->get_baseVa() + min_preferred_rva;
51 
52  // Minimum address at which to map
53  //AddressInterval freeSpace = map->unmapped(mappableArea.greatest(), Sawyer::Container::MATCH_BACKWARD);
54  AddressInterval valid_range = AddressInterval::hull(pe_header->get_baseVa(),0x7FFFFFFFFFFF);
55  ROSE_ASSERT(!valid_range.isEmpty());
56  Address map_base_va = map->findFreeSpace(pe_header->get_e_image_size(), maximum_alignment, valid_range).get();
57  map_base_va = alignUp(map_base_va, (Address)maximum_alignment);
58 
59  if(map_base_va != original){
60  mlog[INFO]<<"Rebasing File "<<pe_header->get_file()->get_name()<<", From:0x"<<hex<<original<<" To:0x"<<map_base_va<<dec<<endl;
61  }
62 
63  // If the minimum preferred virtual address is less than the floor of the page-aligned mapping area, then
64  // return a base address which moves the min_preferred_va to somewhere in the page pointed to by map_base_va.
65  if (min_preferred_va < map_base_va) {
66  size_t min_preferred_page = min_preferred_va / maximum_alignment;
67  if (map_base_va < min_preferred_page * maximum_alignment)
68  return 0;
69  return map_base_va;
70  }
71 
72  return header->get_baseVa();
73 }

101 BinaryLoaderPe::alignValues(SgAsmGenericSection *section, const MemoryMap::Ptr&,

103  Address *va_p, Address *mem_size_p,
104  Address *offset_p, Address *file_size_p, bool *map_private_p,
105  Address *va_offset_p, bool *anon_lo_p, bool *anon_hi_p,
106  ConflictResolution *resolve_p) {
107  SgAsmGenericHeader *header = isSgAsmPEFileHeader(section);
108  if (!header) header = section->get_header();
109  ASSERT_not_null(header);
110 
111  if (!section->isMapped())
112  return CONTRIBUTE_NONE;
113 
114  /* File and memory alignment must be between 1 and 0x200 (512), inclusive */
115  Address file_alignment = section->get_fileAlignment();
116  if (file_alignment>0x200 || 0==file_alignment)
117  file_alignment = 0x200;
118  Address mapped_alignment = section->get_mappedAlignment();
119  if (0==mapped_alignment)
120  mapped_alignment = 0x200;
121 
122  /* Align file size upward even before we align the file offset downward. */
123  Address file_size = alignUp(section->get_size(), file_alignment);
124 
125  /* Map the entire section's file content (aligned) or the requested map size, whichever is larger. */
126  Address mapped_size = std::max(section->get_mappedSize(), file_size);
127 
128  /* Align file offset downward but do not adjust file size. */
129  Address file_offset = alignDown(section->get_offset(), file_alignment);
130 
131  /* Align the preferred relative virtual address downward without regard for the base virtual address, and do not adjust
132  * mapped size. */
133  Address mapped_va = header->get_baseVa() + alignDown(section->get_mappedPreferredRva(), mapped_alignment);
134 
135  *malign_lo_p = mapped_alignment;
136  *malign_hi_p = 1;
137  *va_p = mapped_va;
138  *mem_size_p = mapped_size;
139  *offset_p = file_offset;
140  *file_size_p = file_size;
141  *map_private_p = false;
142  *va_offset_p = 0;
143  *anon_lo_p = *anon_hi_p = true;
144  *resolve_p = RESOLVE_OVERMAP;
145  return CONTRIBUTE_ADD;

296 BinaryLoaderPe::fixup(SgAsmInterpretation *interp, FixupErrors *errors) {
297  SgAsmGenericHeaderPtrList& headers = interp->get_headers()->get_headers();
298  map<pair<string,string>,SgAsmPEExportEntry*> exportEntryMap;
299  map<pair<string,unsigned>,SgAsmPEExportEntry*> exportEntryOrdinalMap;
300 
301  MemoryMap::Ptr memoryMap = interp->get_map();
302  if(memoryMap == nullptr) interp->set_map(memoryMap = MemoryMap::instance());
303 
304  //Traverse sections to ensure mapped address and sections are properly set.
305  for(auto h = headers.begin(); h != headers.end(); ++h) {
306  SgAsmGenericSectionPtrList& sections = (*h)->get_sections()->get_sections();
307  Address headerOffset = (*h)->get_mappedActualVa() - (*h)->get_baseVa();
308  for(auto s = sections.begin(); s != sections.end(); ++s){
309  SgAsmGenericSection* section = *s;
310  if((headerOffset + section->get_baseVa()) > section->get_mappedActualVa()){
311  section->set_mappedActualVa(section->get_mappedActualVa() + headerOffset);
312  }
313  }
314  }
315 
316  //Build a map with the keys DDL Name and entry name that points to the export entry.
317  //Seperate map for all loaded dll for determining when a dll was not found.
318  for(auto h = headers.begin(); h != headers.end(); ++h) {
319  SgAsmGenericSectionPtrList& sections = (*h)->get_sections()->get_sections();
320  for(auto s = sections.begin(); s != sections.end(); ++s){
321  if(SgAsmPEExportSection* exportSection = isSgAsmPEExportSection(*s)){
322  SgAsmPEExportDirectory* exportDir = exportSection->get_exportDirectory();
323  SgAsmPEExportEntryPtrList& exports = exportSection->get_exports()->get_exports();
324  string dirName = exportDir->get_name()->get_string();
325  boost::to_lower(dirName);
326 
327  for(auto e = exports.begin(); e != exports.end(); ++e){
328  string entryName = (*e)->get_name()->get_string();
329  boost::to_lower(entryName);
330  unsigned entryOrdinal = (*e)->get_ordinal();
331  exportEntryOrdinalMap[make_pair(dirName,entryOrdinal)] = (*e);
332  if(entryName != "") exportEntryMap[make_pair(dirName,entryName)] = (*e);
333  }
334  }
335  }
336  }
337 
338  //Find all the import sections and update the IAT
339  for(auto h = headers.begin(); h != headers.end(); ++h) {
340  SgAsmGenericSectionPtrList& sections = (*h)->get_sections()->get_sections();
341  for(auto s = sections.begin(); s != sections.end(); ++s){
342  if(SgAsmPEImportSection* importSection = isSgAsmPEImportSection(*s)){
343  SgAsmPEImportDirectoryPtrList& directories = (importSection)->get_importDirectories()->get_vector();
344  for(auto d = directories.begin(); d != directories.end(); ++d){
345  SgAsmPEImportItemPtrList& imports = (*d)->get_imports()->get_vector();
346  string dirName = (*d)->get_dllName()->get_string();
347  boost::to_lower(dirName);
348  for(auto i = imports.begin(); i != imports.end(); ++i){
349  SgAsmPEImportItem* importEntry = *i;
350  SgAsmPEExportEntry* exportEntry = nullptr;
351  string importName = "";
352  unsigned entryOrdinal = 0;
353  if(importEntry->get_by_ordinal()){
354  entryOrdinal = importEntry->get_ordinal();
355  auto found = exportEntryOrdinalMap.find(make_pair(dirName,entryOrdinal));
356  if (found != exportEntryOrdinalMap.end()) exportEntry = found->second;
357  }else{
358  importName = importEntry->get_name()->get_string();
359  boost::to_lower(importName);
360  auto found = exportEntryMap.find(make_pair(dirName,importName));
361  if (found != exportEntryMap.end()) exportEntry = found->second;
362  }
363  if(exportEntry != nullptr){
364  //The entry could be forwarded to another dll.
365  //Recurse down forwards till the final entry is found
366  set<SgAsmGenericString*> forwardList;
367  while(SgAsmGenericString* forward = exportEntry->get_forwarder()){
368  //This could be a circular depedency. The windows loader has special behavior
369  //for some functions that are OS functions and do not exist in dll
370  //in these cases no source code is loaded
371  if(forwardList.count(forward) == 0) forwardList.insert(forward);
372  else{
373  exportEntry = nullptr;
374  break;
375  }
376  string forwardString = forward->get_string();
377  size_t splitPos = forwardString.find(".");
378  string forwardDirName = forwardString.substr(0,splitPos+1) + "dll";
379  string forwardImportName = forwardString.substr(splitPos+1);
380  boost::to_lower(forwardDirName);
381  boost::to_lower(forwardImportName);
382  exportEntry = nullptr;
383  if(forwardImportName[0] == '#'){
384  //Forward based upon ordinal
385  string ordinalString = forwardImportName.substr(1);
386  unsigned forwardOrdinal = strtoul(ordinalString.c_str(),nullptr,0);
387  auto found = exportEntryOrdinalMap.find(make_pair(forwardDirName,forwardOrdinal));
388  if (found != exportEntryOrdinalMap.end()) exportEntry = found->second;
389  }else{
390  auto found = exportEntryMap.find(make_pair(forwardDirName,forwardImportName));
391  if (found != exportEntryMap.end()) exportEntry = found->second;
392  }
393  if(exportEntry == nullptr){
394  //Forwards to an unloaded dll
395  break;
396  }
397  }
398  }
399  //Export entry found. Set address in the IAT
400  if(exportEntry != nullptr){
401  Address exportAddr = *exportEntry->get_exportRva().va();
402  Address iatEntryAddr = importEntry->get_iatEntryVa();
403  size_t written = memoryMap->writeUnsigned(exportAddr,iatEntryAddr);
404  if(written > 0) importEntry->set_iat_written(true);
405  mlog[TRACE]<<"Setting value in IAT: "<<dirName<<"."<<importName<<"-"<<entryOrdinal<<": IAT 0x"<<hex<<iatEntryAddr<<" Export: 0x"<<exportAddr<<dec<<endl;
406  }
407  }//End import list iterator
408  }//End import directory iterator
409  }//End is import section
410  }//End section iterator
411  }//End Header iterator
