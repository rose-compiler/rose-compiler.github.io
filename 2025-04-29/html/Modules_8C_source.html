401  Address va = settings_.what.least();

403  if (avail.least() > settings_.what.greatest())
404  break;
405  const size_t nPrint = std::min(settings_.what.greatest()+1-avail.least(), avail.size());

407  const MemoryMap::Segment &segment = node.value();
408  const AddressInterval segmentInterval = node.key();
409  size_t offsetWithinSegment = avail.least() - segmentInterval.least();
410  size_t offsetWithinBuffer = segment.offset() + offsetWithinSegment;
411  debug <<" Segment \"" <<cEscape(segment.name()) <<"\" + " <<toHex(offsetWithinSegment) <<"\n";
412  ASSERT_require(offsetWithinBuffer < segment.buffer()->size());
413  ASSERT_require(nPrint <= segment.buffer()->available(offsetWithinBuffer));
414  const unsigned char *data = (const unsigned char*)segment.buffer()->data() + offsetWithinBuffer;
415  ASSERT_not_null(data);
416  debug <<fmt.prefix;
417  hexdump(debug, avail.least(), data, nPrint, fmt);
418  debug <<"\n";
419  if (avail.least() + (nPrint-1) == settings_.what.greatest())

514 bool
515 MatchThunk::match(const Partitioner::ConstPtr &partitioner, Address anchor) {
516  ASSERT_not_null(partitioner);
517 
518  // Disassemble the next few undiscovered instructions
519  static const size_t maxInsns = 2; // max length of a thunk
520  std::vector<SgAsmInstruction*> insns;
521  Address va = anchor;
522  for (size_t i=0; i<maxInsns; ++i) {
523  if (partitioner->instructionExists(va))
524  break; // look only for undiscovered instructions
525  SgAsmInstruction *insn = partitioner->discoverInstruction(va);
526  if (!insn)
527  break;
528  insns.push_back(insn);
529  va += insn->get_size();
530  }
531  if (insns.empty())
532  return false;
533 
534  functions_.clear();
535  ThunkDetection found = predicates_->isThunk(partitioner, insns);
536  if (!found)
537  return false;
538 
539  // This is a thunk
540  Function::Ptr thunk = Function::instance(anchor, SgAsmFunction::FUNC_THUNK);
541  if (!found.name.empty())
542  thunk->reasonComment("matched " + found.name);
543  functions_.push_back(thunk);
544 
545  // Do we know the successors? They would be the function(s) to which the thunk branches.
546  BasicBlock::Ptr bb = BasicBlock::instance(anchor, partitioner);
547  for (size_t i=0; i<found.nInsns; ++i)
548  bb->append(partitioner, insns[i]);
549  for (const BasicBlock::Successor &successor: partitioner->basicBlockSuccessors(bb)) {
550  if (auto targetVa = successor.expr()->toUnsigned()) {
551  if (Function::Ptr thunkTarget = partitioner->functionExists(*targetVa)) {
552  thunkTarget->insertReasons(SgAsmFunction::FUNC_THUNK_TARGET);
553  if (thunkTarget->reasonComment().empty())
554  thunkTarget->reasonComment("target of thunk " + thunk->printableName());
555  } else {
556  thunkTarget = Function::instance(*targetVa, SgAsmFunction::FUNC_THUNK_TARGET);
557  thunkTarget->reasonComment("target of thunk " + thunk->printableName());
558  insertUnique(functions_, thunkTarget, sortFunctionsByAddress);
559  }
560  }
561  }
562 
563  return true;
564 }

567 deExecuteZeros(const MemoryMap::Ptr &map /*in,out*/, size_t threshold, size_t leaveAtFront, size_t leaveAtBack) {
568  ASSERT_not_null(map);
569  AddressIntervalSet changes;
570  if (leaveAtFront + leaveAtBack >= threshold)
571  return changes;
572  Address va = map->hull().least();
573  AddressInterval zeros;
574  uint8_t buf[4096];
575  while (AddressInterval accessed = map->atOrAfter(va).limit(sizeof buf).require(MemoryMap::EXECUTABLE).read(buf)) {
576  size_t nRead = accessed.size();
577  size_t firstZero = 0;
578  while (firstZero < nRead) {
579  while (firstZero<nRead && buf[firstZero]!=0) ++firstZero;
580  if (firstZero < nRead) {
581  size_t nZeros = 1;
582  while (firstZero+nZeros < nRead && buf[firstZero+nZeros]==0) ++nZeros;
583 
584  if (zeros.isEmpty()) {
585  zeros = AddressInterval::baseSize(accessed.least()+firstZero, nZeros);
586  } else if (zeros.greatest()+1 == accessed.least()+firstZero) {
587  zeros = AddressInterval::baseSize(zeros.least(), zeros.size()+nZeros);
588  } else {
589  if (zeros.size() >= threshold && zeros.size() > leaveAtFront + leaveAtBack) {
590  AddressInterval affected = AddressInterval::hull(zeros.least()+leaveAtFront,
591  zeros.greatest()-leaveAtBack);
592  map->within(affected).changeAccess(0, MemoryMap::EXECUTABLE);
593  changes.insert(affected);
594  }
595  zeros = AddressInterval::baseSize(va+firstZero, nZeros);
596  }
597 
598  firstZero += nZeros+1;
599  }
600  }
601  if (accessed.greatest() == map->greatest())
602  break; // avoid possible overflow in next statement
603  va = accessed.greatest() + 1;
604  }
605  if (zeros.size() >= threshold && zeros.size() > leaveAtFront + leaveAtBack) {
606  AddressInterval affected = AddressInterval::hull(zeros.least()+leaveAtFront, zeros.greatest()-leaveAtBack);
607  map->within(affected).changeAccess(0, MemoryMap::EXECUTABLE);
608  changes.insert(affected);
609  }
610  return changes;
611 }

802 boost::logic::tribool
803 isStackBasedReturn(const Partitioner::ConstPtr &partitioner, const BasicBlock::Ptr &bb) {
804  ASSERT_not_null(partitioner);
805  ASSERT_not_null(bb);
806  Sawyer::Message::Stream debug(mlog[DEBUG]);
807  BasicBlockSemantics sem = bb->semantics();
808  BaseSemantics::State::Ptr state = sem.finalState();
809  if (!state)
810  return boost::logic::indeterminate;
811 
812  ASSERT_not_null(sem.dispatcher);
813  ASSERT_not_null(sem.operators);
814  SAWYER_MESG(debug) <<" block has semantic information\n";
815  const RegisterDescriptor REG_IP = partitioner->instructionProvider().instructionPointerRegister();
816  const RegisterDescriptor REG_SP = partitioner->instructionProvider().stackPointerRegister();
817  const RegisterDescriptor REG_SS = partitioner->instructionProvider().stackSegmentRegister();
818 
819  // Find the pointer to the return address. Since the return instruction (e.g., x86 RET) has been processed semantically
820  // already, the return address is beyond the end of the stack. Here we handle architecture-specific instructions that
821  // might pop more than just the return address (e.g., x86 "RET 4").
822  BaseSemantics::SValue::Ptr stackOffset; // added to stack ptr to get ptr to return address
823  if (SgAsmX86Instruction *x86insn = isSgAsmX86Instruction(bb->instructions().back())) {
824  if ((x86insn->get_kind() == x86_ret || x86insn->get_kind() == x86_retf) &&
825  x86insn->nOperands() == 1 &&
826  isSgAsmIntegerValueExpression(x86insn->operand(0))) {
827  uint64_t nbytes = isSgAsmIntegerValueExpression(x86insn->operand(0))
828  ->get_absoluteValue();
829  nbytes += REG_IP.nBits() / 8; // size of return address
830  stackOffset = sem.operators->negate(sem.operators->number_(REG_IP.nBits(), nbytes));
831  }
832  }
833  if (!stackOffset) {
834  // If no special case above, assume return address is the word beyond the top-of-stack and that the stack grows
835  // downward.
836  stackOffset = sem.operators->negate(sem.operators->number_(REG_IP.nBits(), REG_IP.nBits()/8));
837  }
838  BaseSemantics::SValue::Ptr sp = sem.operators->peekRegister(REG_SP);
839  BaseSemantics::SValue::Ptr retAddrPtr = sem.operators->add(sp, stackOffset);
840 
841  // Now that we have the ptr to the return address, read it from the stack and compare it with the new instruction
842  // pointer. If equal, then the basic block returns to the caller.
843  BaseSemantics::SValue::Ptr retAddr = sem.operators->undefined_(REG_IP.nBits());
844  retAddr = sem.operators->peekMemory(REG_SS, retAddrPtr, retAddr);
845  BaseSemantics::SValue::Ptr ip = sem.operators->peekRegister(REG_IP);
846  BaseSemantics::SValue::Ptr isEqual =
847  sem.operators->equalToZero(sem.operators->add(retAddr, sem.operators->negate(ip)));
848  bool isReturn = isEqual->isTrue();
849 
850  if (debug) {
851  debug <<" stackOffset = " <<*stackOffset <<"\n";
852  debug <<" sp = " <<*sp <<"\n";
853  debug <<" retAddrPtr = " <<*retAddrPtr <<"\n";
854  debug <<" retAddr = " <<*retAddr <<"\n";
855  debug <<" ip = " <<*ip <<"\n";
856  debug <<" retAddr==ip? = " <<*isEqual <<"\n";
857  debug <<" returning " <<(isReturn ? "true" : "false") <<"\n";
858  //debug <<" state:" <<*state; // produces lots of output!
859  }
860 
861  return isReturn;
862 }

869 SgAsmBlock*
870 buildBasicBlockAst(const Partitioner::ConstPtr &partitioner, const BasicBlock::Ptr &bb, const Function::Ptr &parentFunction,
871  const AstConstructionSettings &settings) {
872  ASSERT_not_null(partitioner);
873  ASSERT_not_null(bb);
874  ASSERT_not_null(parentFunction);
875  if (bb->isEmpty()) {
876  if (settings.allowEmptyBasicBlocks) {
877  mlog[WARN] <<"creating an empty basic block AST node at " <<StringUtility::addrToString(bb->address()) <<"\n";
878  } else {
879  return NULL;
880  }
881  }
882 
883  ControlFlowGraph::ConstVertexIterator bblockVertex = partitioner->findPlaceholder(bb->address());
884 
885  std::vector<SgAsmInstruction*> insns = bb->instructions();
886  for (size_t i=0; i<insns.size(); ++i) {
887  if (settings.copyAllInstructions) {
888  SgTreeCopy deep;
889  SgAsmInstruction *newInsn = isSgAsmInstruction(insns[i]->copy(deep));
890  ASSERT_not_null(newInsn);
891  ASSERT_require(newInsn != insns[i]);
892  ASSERT_require(newInsn->get_address() == insns[i]->get_address());
893  insns[i] = newInsn;
894  }
895  }
896 
897  SgAsmBlock *ast = SageBuilderAsm::buildBasicBlock(insns);
898  ast->set_comment(bb->comment());
899  unsigned reasons = 0;
900 
901  // Is this block an entry point for any function (not just the parent function)?
902  if (bblockVertex->value().isEntryBlock())
903  reasons |= SgAsmBlock::BLK_ENTRY_POINT;
904 
905  // Does this block have any predecessors?
906  if (bblockVertex != partitioner->cfg().vertices().end() && bblockVertex->nInEdges()==0)
907  reasons |= SgAsmBlock::BLK_CFGHEAD;
908 
909  ast->set_reason(reasons);
910  ast->set_codeLikelihood(1.0); // FIXME[Robb P. Matzke 2014-08-07]
911 
912  // Stack delta analysis results for this block in the context of the owning function. If blocks are shared they can have
913  // different stack deltas (e.g., eax might have different values in function F and G which means the block "sub esp, eax"
914  // has a different stack delta in each function. Since we're creating a SgAsmBlock for a particular function we need to
915  // use stack delta results in the context of that function.
916  const StackDelta::Analysis &sdAnalysis = parentFunction->stackDeltaAnalysis();
917  ast->set_stackDeltaOut(sdAnalysis.toInt(sdAnalysis.basicBlockOutputStackDeltaWrtFunction(bb->address()))
918  .orElse(SgAsmInstruction::INVALID_STACK_DELTA));
919 
920  // Cache the basic block successors in the AST since we've already computed them. If the basic block is in the CFG then we
921  // can use the CFG's edges to initialize the AST successors since they are canonical. Otherwise we'll use the successors
922  // from bb. In any case, we fill in the successor SgAsmIntegerValueExpression objects with only the address and not
923  // pointers to SgAsmBlock IR nodes since we don't have all the blocks yet; the SgAsmBlock pointers will be initialized
924  // higher up on the AST-building stack.
925  if (bblockVertex != partitioner->cfg().vertices().end()) {
926  bool isComplete = true;
927  for (const ControlFlowGraph::Edge &edge: bblockVertex->outEdges()) {
928  const ControlFlowGraph::Vertex &target = *edge.target();
929  if (target.value().type() == V_INDETERMINATE || target.value().type() == V_NONEXISTING) {
930  isComplete = false;
931  } else {
932  ASSERT_require(target.value().type() == V_BASIC_BLOCK);
933  SgAsmIntegerValueExpression *succ = SageBuilderAsm::buildValueU64(target.value().address());
934  succ->set_parent(ast);
935  ast->get_successors().push_back(succ);
936  }
937  }
938  ast->set_successorsComplete(isComplete);
939  } else {
940  bool isComplete = true;
941  for (Address successorVa: partitioner->basicBlockConcreteSuccessors(bb, &isComplete)) {
942  SgAsmIntegerValueExpression *succ = SageBuilderAsm::buildValueU64(successorVa);
943  succ->set_parent(ast);
944  ast->get_successors().push_back(succ);
945  }
946  ast->set_successorsComplete(isComplete);
947  }
948  return ast;
949 }

965 SgAsmFunction*
966 buildFunctionAst(const Partitioner::ConstPtr &partitioner, const Function::Ptr &function, const AstConstructionSettings &settings) {
967  ASSERT_not_null(partitioner);
968  ASSERT_not_null(function);
969 
970  // Build the child basic block IR nodes and remember all the data blocks
971  std::vector<DataBlock::Ptr> dblocks = function->dataBlocks();
972  std::vector<SgAsmBlock*> children;
973  for (Address blockVa: function->basicBlockAddresses()) {
974  ControlFlowGraph::ConstVertexIterator vertex = partitioner->findPlaceholder(blockVa);
975  if (vertex == partitioner->cfg().vertices().end()) {
976  mlog[WARN] <<function->printableName() <<" bblock "
977  <<StringUtility::addrToString(blockVa) <<" does not exist in the CFG; no AST node created\n";
978  } else if (BasicBlock::Ptr bb = vertex->value().bblock()) {
979  if (SgAsmBlock *child = buildBasicBlockAst(partitioner, bb, function, settings))
980  children.push_back(child);
981  for (const DataBlock::Ptr &dblock: bb->dataBlocks())
982  insertUnique(dblocks, dblock, sortDataBlocks);
983  } else {
984  mlog[WARN] <<function->printableName() <<" bblock "
985  <<StringUtility::addrToString(blockVa) <<" is undiscovered in the CFG; no AST node created\n";
986  }
987  }
988  if (children.empty()) {
989  if (settings.allowFunctionWithNoBasicBlocks) {
990  mlog[WARN] <<"creating an empty AST node for " <<function->printableName() <<"\n";
991  } else {
992  return NULL;
993  }
994  }
995 
996  // Build the child data block IR nodes. The data blocks attached to the SgAsmFunction node are the union of the data
997  // blocks owned by the function and the data blocks owned by each of its basic blocks.
998  for (const DataBlock::Ptr &dblock: dblocks) {
999  if (SgAsmBlock *child = buildDataBlockAst(partitioner, dblock, settings))
1000  children.push_back(child);
1001  }
1002 
1003  // Sort the children in the order usually used by ROSE
1004  std::sort(children.begin(), children.end(), sortBlocksForAst);
1005 
1006  unsigned reasons = function->reasons();
1007 
1008  // Is the function part of the CFG?
1009  ControlFlowGraph::ConstVertexIterator entryVertex = partitioner->findPlaceholder(function->address());
1010  if (entryVertex != partitioner->cfg().vertices().end())
1011  reasons |= SgAsmFunction::FUNC_GRAPH;
1012 
1013  // Is the function discontiguous?
1014  if (partitioner->aum(function).extent().nIntervals() > 1)
1015  reasons |= SgAsmFunction::FUNC_DISCONT;
1016 
1017  // Is the function the target of a function call?
1018  if (entryVertex != partitioner->cfg().vertices().end()) {
1019  for (const ControlFlowGraph::Edge &edge: entryVertex->inEdges()) {
1020  if (edge.value().type() == E_FUNCTION_CALL || edge.value().type() == E_FUNCTION_XFER) {
1021  reasons |= SgAsmFunction::FUNC_CALL_TARGET;
1022  break;
1023  }
1024  }
1025  }
1026 
1027  // Can this function return to its caller?
1028  SgAsmFunction::MayReturn mayReturn = SgAsmFunction::RET_UNKNOWN;
1029  if (entryVertex != partitioner->cfg().vertices().end()) {
1030  if (BasicBlock::Ptr bb = entryVertex->value().bblock()) {
1031  bool b = false;
1032  if (bb->mayReturn().getOptional().assignTo(b))
1033  mayReturn = b ? SgAsmFunction::RET_SOMETIMES : SgAsmFunction::RET_NEVER;
1034  }
1035  }
1036 
1037  // Function's calling convention. The AST holds the name of the best calling convention, or the empty string if no calling
1038  // convention has been assigned.
1039  std::string bestCallingConvention;
1040  if (CallingConvention::Definition::Ptr ccdef = function->callingConventionDefinition())
1041  bestCallingConvention = ccdef->name();
1042 
1043  // Build the AST
1044  SgAsmFunction *ast = SageBuilderAsm::buildFunction(function->address(), children);
1045  ast->set_reason(reasons);
1046  ast->set_name(function->name());
1047  ast->set_comment(function->comment());
1048  ast->set_mayReturn(mayReturn);
1049  ast->set_stackDelta(function->stackDeltaConcrete());
1050  ast->set_callingConvention(bestCallingConvention);
1051  return ast;
1052 }

1160 
1161  // Pass 1: Sort all calling conventions by number of occurrences within the partitioner.
1162  Sawyer::Container::Map<std::string, size_t> totals; // how many times does each calling convention name match?
1163  for (const Function::Ptr &function: partitioner->functions()) {

1168  for (const CallingConvention::Definition::Ptr &ccDef: ccDefs)
1169  ++totals.insertMaybe(ccDef->name(), 0);

1171

1173  // the first one from the function's definition list since this list is presumably sorted by how frequently the convention
1174  // is used by the whole world.

1179  continue; // don't run analysis if not run already

1181  CallingConvention::Definition::Ptr ccBest;
1182  for (const CallingConvention::Definition::Ptr &ccDef: ccDefs) {
1183  if (NULL==ccBest) {
1184  ccBest = ccDef;
1185  } else if (totals.getOrElse(ccDef->name(), 0) > totals.getOrElse(ccBest->name(), 0)) {
1186  ccBest = ccDef;
1187  }
1188  }

1191  // to be allocated on the heap, and if so, who owns them or what allocator was used.
1192  astFunction->set_callingConvention(ccBest->name());
