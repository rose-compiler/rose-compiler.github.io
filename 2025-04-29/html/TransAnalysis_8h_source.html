19 /* TransInfoOP class defines the required operations that need to be
20  implemented for the path representation of graphs during transitive
21  path analysis.
22  T is the result type of transitive path analysis and represents
23  a collection (eg. set) of paths.
24 */
25 template <class T>
26 class TransInfoOP
27 {
28  public:
29  // the set union of two collection of paths. More specifically,
30  // if info1 and info2 are sets of paths from n1 to n2,
31  // UnionWith(info1,info2) adds all paths in info2 into info1 as well.
32  virtual void UnionWith(T& info1, T info2)=0;
33 
34  // The special element of T that represents the most conservative
35  // approximation of path information. Specifically, it represents
36  // all paths from n1 to n2 into the original graph.
37  // If b = GetBottomInfo(n1,n2), then for every collection t of paths
38  // from n1 to n2, b union t = b; (t is a subset of b).
39  virtual T GetBottomInfo( const GraphAccessInterface::Node *n1, const GraphAccessInterface::Node *n2)=0 ;
40 
41  // The empty collection (set) of paths from n1 to n2.
42  // If b = GetTopInfo(n1,n2), then for every collection t of paths
43  // from n1 to n2, b union t = t; (b is a subset of t).
44  virtual T GetTopInfo(const GraphAccessInterface::Node *n1, const GraphAccessInterface::Node *n2)=0;
45 
46  // return whether a TransInfo is empty
47  virtual bool IsTop(T t) = 0;
48 
49  //If info1 is a set of paths from n1 to n2, and info2 is a set of paths
50  // from n2 to n3, Composite(info1,info2) returns the set of all paths
51  // from n1 to n3 by concatenating paths in info1 and info2 respectively.
52  virtual T Composite(T info1, T info2)=0;
53 
54  // The empty path representation from n to itself.
55  // Suppose t = GetIDTransInfo(n). If t1 is a set of paths from n to n1,
56  // then Composite(t, t1) = t1. Similarly, if t2 is a set of paths from n2
57  // to n, then Composite(t2, t) = t.
58  virtual T GetIDTransInfo( const GraphAccessInterface::Node *n) =0;
59 
60  // Composite a cycle with itself infinite times.
61  // Specifically, if info is a cycle from node n to itself,
62  // Closure(info) = info union Composite(info,info) union Composite(info,info,info)...
63  virtual T Closure(T info)=0;
64 
65  // The transitive path representation associated with edge e.
66  virtual T GetTransInfo( const GraphAccessInterface::Edge *e) =0;
67 
68  virtual ~TransInfoOP() {}
69 };
