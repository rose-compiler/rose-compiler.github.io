859 void
860 LowLevelRewrite::
861 reassociatePreprocessorDeclarations (
862  const SgStatementPtrList & newTransformationStatementList,
863  SgNode* astNode,
864  bool prependNewCode,
865  bool removeExistingNode )
866  {
867  // If a new located node is inserted before or after the input astNode, then we have to
868  // move any associated preprocessor declarations and comments to the statement before or
869  // after so that the order of preprocessor declarations and located nodes will be preserved

887  // printf ("astNode->sage_class_name() = %s \n",astNode->sage_class_name());
888  // printf ("astNode->unparseToString() = %s \n",astNode->unparseToString().c_str());
889 
890  // Comments/directives might be attached to the basic block associated with a function definition
891  // so we call reassociatePreprocessorDeclarations() recursively to handle this
892  switch (astNode->variantT())
893  {
894  case V_SgFunctionDeclaration:
895  case V_SgMemberFunctionDeclaration:
896  {
897  SgFunctionDeclaration* functionDeclaration = isSgFunctionDeclaration(astNode);
898  ROSE_ASSERT (functionDeclaration != NULL);
899  SgFunctionDefinition* functionDefinition = functionDeclaration->get_definition();
900  if (functionDefinition != NULL)
901  reassociatePreprocessorDeclarations (newTransformationStatementList,functionDefinition,prependNewCode,removeExistingNode);

904

906  {

908  ROSE_ASSERT (functionDefinition != NULL);
909  SgBasicBlock* basicBlock = functionDefinition->get_body();
910  ROSE_ASSERT (basicBlock != NULL);
911  reassociatePreprocessorDeclarations (newTransformationStatementList,basicBlock,prependNewCode,removeExistingNode);
912  break;
913  }
914 
915  default:
916  {
917  // Cases not handled yet, comments/directives may be attached to these
918  // or the SgBasicBlocks that they contain!
919  ROSE_ASSERT (astNode->variantT() != V_SgTemplateInstantiationDecl);
920  ROSE_ASSERT (astNode->variantT() != V_SgEnumDeclaration);
921  ROSE_ASSERT (astNode->variantT() != V_SgTemplateDeclaration);
922  ROSE_ASSERT (astNode->variantT() != V_SgTypedefDeclaration);

931  }
932  }

935  ROSE_ASSERT (newTransformationStatementList.size() > 0);
936 
937  SgLocatedNode* locatedNode = dynamic_cast<SgLocatedNode*>(astNode);
938 
939  // I think we can assert this for now (perhaps not later if we permit insertion of non located nodes)
940  ROSE_ASSERT (locatedNode != NULL);
941 
942  // We can't remove the elements of the list while we are traversing the list so
943  // we have to save the references to the elements to be removed and remove them
944  // after traversing the list.
945  AttachedPreprocessingInfoType preprocessorDeclarationsToRemove;
946 
947  // Get attached preprocessing info
948  AttachedPreprocessingInfoType *comments = locatedNode->getAttachedPreprocessingInfo();
949 
950  SgStatementPtrList::const_iterator firstTransformationStatementIterator;

955 #if 0

959  (*lastTransformationStatementIterator)->unparseToString().c_str());

965  // node to the previous node and comments appearing after this node to the

967 
968  // printf ("Found attached comments (at %p of type: %s): \n",locatedNode,locatedNode->sage_class_name());

970  // certain types:

985  // printf ("removeExistingNode = %s \n",removeExistingNode ? "true" : "false");
986  if (removeExistingNode == true)
987  {
988  // Replacement case must copy all comments and spearate them between remaining neighboring statements

1004  // (STL does not permit us to remove elements from a list being traversed).
1005  preprocessorDeclarationsToRemove.push_back(*forwardIterator);
1006  }
1007  else
1008  {
1009  if (!prependNewCode && (*forwardIterator)->getRelativePosition() == PreprocessingInfo::after)
1010  {

1013  // (*lastTransformationStatementIterator)->addToAttachedPreprocessingInfo(*forwardIterator,SgLocatedNode::appendPreprocessingInfoToList);
1014  (*lastTransformationStatementIterator)->addToAttachedPreprocessingInfo(*forwardIterator,PreprocessingInfo::after);
1015  preprocessorDeclarationsToRemove.push_back(*forwardIterator);
1016  }
1017  }
1018  }
1019 
1020  // Reverse iterator through comment list can only be used to processing
1021  // certain types:
1022  // 1) if removeExistingNode == true , (*i)->relativePosition == PreprocessingInfo::before
1023  // 2) if removeExistingNode == false, prependNewCode && (*i)->relativePosition == PreprocessingInfo::before
1024 
1025  AttachedPreprocessingInfoType::reverse_iterator reverseIterator;
1026  for (reverseIterator = comments->rbegin(); reverseIterator != comments->rend(); reverseIterator++)
1027  {
1028  ROSE_ASSERT ( (*reverseIterator) != NULL);
1029 #if 0
1030  printf (" Attached Comment (relativePosition=%s): %s",
1031  ((*reverseIterator)->getRelativePosition() == PreprocessingInfo::before) ? "before" : "after",
1032  (*reverseIterator)->getString());
1033 #endif
1034  // Check for replacement (removal of this statement)
1035  // printf ("removeExistingNode = %s \n",removeExistingNode ? "true" : "false");
1036  if (removeExistingNode == true)
1037  {
1038  // Replacement case must copy all comments and spearate them between remaining neighboring statements
1039  ROSE_ASSERT ( (*reverseIterator)->getRelativePosition() == PreprocessingInfo::before ||
1040  (*reverseIterator)->getRelativePosition() == PreprocessingInfo::after);
1041 
1042  // Split all the preprocessor information between the first and
1043  // last statement in the new list of statements to be added.
1044  if ( (*reverseIterator)->getRelativePosition() == PreprocessingInfo::before )
1045  {
1046  // Insert these before the first statement
1047  // printf ("Insert these before the first statement \n");
1048  // (*firstTransformationStatementIterator)->addToAttachedPreprocessingInfo(*reverseIterator,SgLocatedNode::prependPreprocessingInfoToList);
1049  (*firstTransformationStatementIterator)->addToAttachedPreprocessingInfo(*reverseIterator,PreprocessingInfo::before);

1051

1053  // (STL does not permit us to remove elements from a list being traversed).
1054  preprocessorDeclarationsToRemove.push_back(*reverseIterator);
1055  }
1056  else
1057  {
1058  if (prependNewCode && (*reverseIterator)->getRelativePosition() == PreprocessingInfo::before)
1059  {
1060  // Insert these after the previous statement
1061  // printf ("Insert these after the previous statement \n");
1062 
1063  // reattach the comment on the current node to the end of
1064  // the list of "before" elements on the new statement.
1065  // (*firstTransformationStatementIterator)->addToAttachedPreprocessingInfo(*reverseIterator,SgLocatedNode::prependPreprocessingInfoToList);

1067  preprocessorDeclarationsToRemove.push_back(*reverseIterator);
1068  }
1069  }
1070  }

1073  // We could delete the list but it does not harm to have an empty
1074  // list and there is not interface to allow us to remove the list.
1075  AttachedPreprocessingInfoType::iterator j;
1076  for (j = preprocessorDeclarationsToRemove.begin(); j != preprocessorDeclarationsToRemove.end(); j++)
1077  {
1078 #if 0
1079  printf (" Removing Comment (relativePosition=%s): %s",
1080  ((*j)->relativePosition == PreprocessingInfo::before) ? "before" : "after",(*j)->getString());
1081 #endif
1082  // Remove the element from the list of comments at the current astNode
1083  // comments->remove( (*j) );
1084  comments->erase(find(comments->begin(),comments->end(),*j));
1085  }
1086  }
1087  else
1088  {
1089  // printf ("In reassociatePreprocessorDeclarations(): No comments found! \n");
1090  }
1091  }
