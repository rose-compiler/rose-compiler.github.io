271 static void
272 parseClass(TokenStream &tokens, Scopes &scopes) {
273  ASSERT_require(tokens.matches(tokens[0], "class") || tokens.matches(tokens[0], "struct"));
274  Location loc = tokens.location(tokens[0]);
275  tokens.consume();
276 
277  // Class name is the last word after class:
278  // class DLL_EXPORT Foo
279  std::string name = "<anonymous-class>";
280  while (tokens[0].type() == TOK_WORD) {
281  name = tokens.lexeme(tokens[0]);
282  loc = tokens.location(tokens[0]);
283  tokens.consume();
284  }
285 
286  // If the class name is followed by a ":" but not "::" then we're defining a class and can scan up to the opening brace.
287  if (tokens.matches(tokens[0], ":")) {
288  tokens.consume();
289  while (tokens[0].type() != TOK_EOF && !tokens.matches(tokens[0], "{"))
290  tokens.consume();
291  }
292 
293  // If we're not at an opening brace then this isn't a class definition.
294  if (!tokens.matches(tokens[0], "{"))
295  return;
296 
297  scopes.push_back(Scope(name, loc));
298  SAWYER_MESG(mlog[DEBUG]) <<where(tokens.fileName(), loc) <<": enter class " <<scopes <<"\n";
299  tokens.consume(); // the opening brace
300 }

302 static int64_t
303 parseIntegralConstant(TokenStream &tokens) {
304  int64_t retval = 0;
305  std::string s = tokens.lexeme(tokens[0]);
306  if (boost::starts_with(s, "0b") && s.size() > 2) {
307  for (size_t i = 2; i < s.size(); ++i) {
308  if (s[i] == '0') {
309  retval = retval << 1;
310  } else if (s[i] == '1') {
311  retval = (retval << 1) | 1;
312  } else {
313  if (!quiet)
314  std::cerr <<where(tokens.fileName(), tokens.location(tokens[0])) <<": error: invalid integer literal\n";
315  break;
316  }
317  }
318  } else {
319  char *rest = NULL;
320  errno = 0;
321  retval = (int64_t) strtoul(s.c_str(), &rest, 0);
322  if ((errno != 0 || *rest) && !quiet)
323  std::cerr <<where(tokens.fileName(), tokens.location(tokens[0])) <<": error: invalid integer literal\n";
324  }
325  tokens.consume();

656 parseAssignExpr(TokenStream &tokens) {
657  int64_t retval = parseLogicalOrExpr(tokens);
658  if (tokens.matches(tokens[0], "=") || tokens.matches(tokens[0], "+=") || tokens.matches(tokens[0], "-=") ||
659  tokens.matches(tokens[0], "*=") || tokens.matches(tokens[0], "/=") || tokens.matches(tokens[0], "%=") ||
660  tokens.matches(tokens[0], "<<=") || tokens.matches(tokens[0], ">>=") || tokens.matches(tokens[0], "&=") ||
661  tokens.matches(tokens[0], "^=") || tokens.matches(tokens[0], "|=")) {
662  if (!quiet) {
663  std::cerr <<where(tokens.fileName(), tokens.location(tokens[0]))
664  <<": error: \"" <<tokens.lexeme(tokens[0]) <<"\" cannot appear in a constant-expression\n";
665  }
666  tokens.consume();
667  retval = parseAssignExpr(tokens);
668  } else if (tokens.matches(tokens[0], "?")) {
669  tokens.consume();
670  int64_t a = parseLogicalOrExpr(tokens);
671  if (tokens.matches(tokens[0], ":")) {
672  tokens.consume();
673  int64_t b = parseAssignExpr(tokens);
674  retval = retval ? a : b;
675  } else if (!quiet) {
676  std::cerr <<where(tokens.fileName(), tokens.location(tokens[0])) <<": error: \":\" expected\n";
677  }
678  }
679  return retval;
680 }

714 // Emit something like "namespace A { namespace B { ..." and also update the namespaceLattice and namespaceLatticeRoots so they
715 // have records of the namespaces.
716 static size_t
717 generateNamespaces(const std::string initial, const Scopes &scopes, size_t nOmit) {
718  size_t nns = 0;
719  NamespaceLattice::VertexIterator parent = namespaceLattice.vertices().end();
720 
721  if (!initial.empty()) {
722  std::cout <<"namespace " <<initial <<" {";
723  if (!namespaceLatticeRoots.getOptional(initial).assignTo(parent)) {
724  // Create a new namespace root in the lattice
725  parent = namespaceLattice.insertVertex(initial);
726  namespaceLatticeRoots.insert(initial, parent);
727  }
728  ++nns;
729  }
730 
731  for (size_t i = 0; i+nOmit < scopes.size(); ++i) {
732  if (!scopes[i].name.empty() && !boost::starts_with(scopes[i].name, "<")) {
733  if (parent == namespaceLattice.vertices().end()) {
734  if (!namespaceLatticeRoots.getOptional(scopes[i].name).assignTo(parent)) {
735  // Create a new namespace root in the lattice
736  parent = namespaceLattice.insertVertex(scopes[i].name);
737  namespaceLatticeRoots.insert(scopes[i].name, parent);
738  }
739  } else {
740  // Does this namespace exist in the lattice?
741  NamespaceLattice::VertexIterator me = namespaceLattice.vertices().end();
742  BOOST_FOREACH (NamespaceLattice::Edge &edge, parent->outEdges()) {
743  if (edge.target()->value() == scopes[i].name) {
744  me = edge.target();
745  break;
746  }
747  }
748  if (me == namespaceLattice.vertices().end()) {
749  // Doesn't exist yet, so create it
750  me = namespaceLattice.insertVertex(scopes[i].name);
751  namespaceLattice.insertEdge(parent, me);
752  }
753  parent = me;
754  }
755 
756  std::cout <<(nns>0?" ":"") <<"namespace " <<scopes[i].name <<" {";
757  ++nns;
758  }
759  }
760  return nns;
761 }

977 static void
978 parseEnum(TokenStream &tokens, Scopes &scopes) {
979  ASSERT_require(tokens.matches(tokens[0], "enum"));
980  Location enumLocation = tokens.location(tokens[0]);
981 
982  // Flags for the enum as a whole, which become the default flags for each member.
983  Flags enumFlags = readFlags(tokens, tokens[0]);
984  tokens.consume();
985 
986  // Advance past the opening brace while picking up the enum name. If there is no enum
987  // name then skip it. We don't support "typedef enum {...} name".
988  std::string name;
989  if (tokens.matches(tokens[0], "{")) {
990  return;
991  } else if (tokens[0].type() == TOK_WORD && tokens.matches(tokens[1], "{")) {
992  name = tokens.lexeme(tokens[0]);
993  scopes.push_back(Scope(name, tokens.location(tokens[0])));
994  mlog[WHERE] <<where(tokens.fileName(), tokens.location(tokens[0])) <<": enum " <<scopes <<"\n";
995  tokens.consume(2);
996  } else if (tokens[0].type() == TOK_WORD && tokens.matches(tokens[0], "class") &&
997  tokens[1].type() == TOK_WORD &&
998  tokens.matches(tokens[2], "{")) {
999  name = tokens.lexeme(tokens[1]);
1000  scopes.push_back(Scope(name, tokens.location(tokens[1])));
1001  mlog[WHERE] <<where(tokens.fileName(), tokens.location(tokens[1])) <<": enum " <<scopes <<"\n";
1002  tokens.consume(3);
1003  } else {
1004  return;
1005  }
1006 
1007  // Did we already emit this enum?
1008  const std::string enumName = fullyQualifiedEnumName(scopes);
1009  NameAndLocation firstLocation;
1010  if (emittedEnums.getOptional(enumName).assignTo(firstLocation)) {
1011  if (!quiet) {
1012  std::cerr <<where(tokens.fileName(), enumLocation) <<": error: enum '" <<enumName <<"' is already defined\n";
1013  std::cerr <<where(firstLocation.fileName, firstLocation.location) <<": info: previous definition\n";
1014  }
1015  return;
1016  }
1017  emittedEnums.insert(enumName, NameAndLocation(tokens.fileName(), enumLocation));
1018 
1019  // Parse the enum members
1020  currentEnumMembers.clear();
1021  EnumMembers members;
1022  int64_t memberValue = 0;
1023  while (tokens[0].type() != TOK_EOF && !tokens.matches(tokens[0], "}")) {
1024  // The enum member name
1025  if (tokens[0].type() != TOK_WORD) {
1026  if (!quiet)
1027  std::cerr <<where(tokens.fileName(), tokens.location(tokens[0])) <<": error: expected an enum constant\n";
1028  break;
1029  }
1030  std::string memberName = tokens.lexeme(tokens[0]);
1031  Flags memberFlags = enumFlags | readFlags(tokens, tokens[0]);
1032  tokens.consume();
1033 
1034  // The optional member value
1035  if (tokens.matches(tokens[0], "=")) {
1036  tokens.consume();
1037  memberValue = parseConstExpr(tokens);
1038  }
1039 
1040  // Comma separator
1041  if (tokens.matches(tokens[0], ",")) {
1042  tokens.consume();
1043  } else if (!tokens.matches(tokens[0], "}")) {
1044  if (!quiet)
1045  std::cerr <<where(tokens.fileName(), tokens.location(tokens[0])) <<": error: expected comma enum separator\n";
1046  break;
1047  }
1048 
1049  members.insert(memberName, EnumValue(memberValue, memberFlags));
1050  currentEnumMembers.insert(memberName, EnumValue(memberValue, memberFlags));
1051 
1052  ++memberValue;
1053  }
1054 
1055  // Closing brace
1056  if (!tokens.matches(tokens[0], "}") && !quiet)
1057  std::cerr <<where(tokens.fileName(), tokens.location(tokens[0])) <<": error: expected enum closing brace\n";
1058 
1059  switch (generate) {
1060  case GENERATE_DECLARATIONS:
1061  if (enumFlags.isClear(NO_STRINGIFY)) {
1062  generateDeclaration(tokens.fileName(), enumLocation, scopes);
1063  if (emitCompatible)
1064  generateCompatibleDeclaration(scopes);
1065  }
1066  break;
1067  case GENERATE_DEFINITIONS:
1068  if (enumFlags.isClear(NO_STRINGIFY)) {
1069  generateImplementation(tokens.fileName(), enumLocation, scopes, members);
1070  if (emitCompatible)
1071  generateCompatibleImplementation(scopes);
1072  }
1073  break;
1074  case GENERATE_NOTHING:
1075  break;
1076  }
1077 
1078  if (!checkCompatibility.empty())
1079  checkBackwardCompatibility(tokens.fileName(), enumLocation, scopes, members);
1080 }
