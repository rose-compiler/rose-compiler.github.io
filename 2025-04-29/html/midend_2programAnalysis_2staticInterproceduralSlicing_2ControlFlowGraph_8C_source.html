58 void ControlFlowGraph::_buildCFG() {
59 
60  //Go through each CFGNodeImpl in CFGImpl to find the "entry" node
61  DefaultCFGImpl::NodeIterator nodes;
62  CFGNodeImpl* entry = 0;
63  CFGNodeImpl* exit = 0;
64  for (nodes = _cfg->GetNodeIterator(); !nodes.ReachEnd(); ++nodes) {
65  // printf("\nNODE!\n\n");
66  CFGNodeImpl * node = *nodes;
67  if (_cfg->GetPredecessors(node).ReachEnd() &&
68  _cfg->GetSuccessors(node).ReachEnd()) {
69  // printf("Skipping broken node\n");
70  continue;
71  }
72 
73  //get incoming edges
74  DefaultCFGImpl::NodeIterator preds = _cfg->GetPredecessors(node);
75  //determine whether there are no predecessors
76  if (preds.ReachEnd()) {
77  // printf("Found entry!\n");
78  entry = node;
79  }
80  //get outgoing edges
81  DefaultCFGImpl::NodeIterator succs = _cfg->GetSuccessors(node);
82  //determine whether there are no successors
83  if (succs.ReachEnd()) {
84  // printf("Found exit!\n");
85  exit = node;
86  }
87 
88  //Add the node itself to the graph
89  createNode(node);
90  }
91 
92  //set _entry and _exit appropriately
93  if (_cfgnodemap.count(entry))
94  _entry = _cfgnodemap[entry];
95  else
96  _entry = _sgnodemap[AstNodePtrImpl(entry->GetStmts().front()).get_ptr()];
97 
98  if (_cfgnodemap.count(exit))
99  _exit = _cfgnodemap[exit];
100  else
101  _exit = _sgnodemap[AstNodePtrImpl(exit->GetStmts().back()).get_ptr()];
102 
103  //For each CFGNodeImpl, add the appropriate links in ControlFlowGraph
104  for (nodes = _cfg->GetNodeIterator(); !nodes.ReachEnd(); ++nodes) {
105  list<AstNodePtr> stmts = (*nodes)->GetStmts();
106  if(stmts.empty()) {
107  _buildBranches(_cfgnodemap[*nodes], *nodes);
108  } else {
109  list<AstNodePtr>::iterator curr = stmts.begin();
110  list<AstNodePtr>::iterator next = stmts.begin();
111  next++;
112  while (next != stmts.end()) {
113  SgNode * from = AstNodePtrImpl(*curr).get_ptr();
114  SgNode * to = AstNodePtrImpl(*next).get_ptr();
115  addLink(_sgnodemap[from], _sgnodemap[to]);
116  curr++;

118  }

121  }
122  }

149 void ControlFlowGraph::_setupIDs(ID_dir direction) {
150  //Do a BFS on the CFG, initializing IDs as we go
151  int currid = 0;
152  queue<SimpleDirectedGraphNode *> remaining;
153  set<SimpleDirectedGraphNode *> seen;
154 
155  _forIndex = new ControlNode*[_numNodes];
156  _backIndex = new ControlNode*[_numNodes];
157 
158  if (direction == FORWARD) {
159  remaining.push(_entry);
160  } else {
161  remaining.push(_exit);
162  }
163 
164  while (!remaining.empty()) {
165 
166  ControlNode * curr = dynamic_cast<ControlNode *>(remaining.front());
167  remaining.pop();
168 
169  if (seen.count(curr)) {//if we've seen it, ignore it
170  continue;
171  } else {
172  seen.insert(curr);
173  }
174 
175  //assign an id to i
176  ROSE_ASSERT(currid < _numNodes);
177  if (direction == FORWARD) {
178  _forIndex[currid] = curr;
179  curr->setID(currid, FORWARD);
180  } else {
181  _backIndex[currid] = curr;
182  curr->setID(currid, BACKWARD);
183  }
184  currid++;
185 
186  //push its neighbors onto the queue
187  set<SimpleDirectedGraphNode *> neighbors;
188  if (direction == FORWARD) {
189  neighbors = curr->getSuccessors();
190  } else {
191  neighbors = curr->getPredecessors();
192  }
193  for (set<SimpleDirectedGraphNode *>::iterator i = neighbors.begin(); i != neighbors.end(); i++) {
194  remaining.push(*i);
195  }
