283  static SgNode* leastCommonAncestor(SgNode* a, SgNode* b) {
284  // Find the closest node which is an ancestor of both a and b
285  vector<SgNode*> ancestorsOfA;
286  for (SgNode* p = a; p; p = p->get_parent()) ancestorsOfA.push_back(p);
287  while (b) {
288  vector<SgNode*>::const_iterator i = std::find(ancestorsOfA.begin(), ancestorsOfA.end(), b);

291  }

702  doEffectsForwards(children[j], data);
703  }

705  for (int j = (int)positionOfChild1 - 1; j > (int)positionOfChild2; --j) {
706  // cerr << "Walking backward in lca: " << (children[j] ? children[j]->unparseToString() : "<null>") << endl;
707  doEffectsBackwards(children[j], data);
708  }
709  }
710  }
711  // Walk from "lca" to "tgt", but not "lca" itself
712  for (unsigned int i = 1; i < ancestorsOfTo.size() - 1; ++i) {
713  SgNode* anc = ancestorsOfTo[i];
714  // cerr << "Walking down ancestor " << anc->unparseToString() << endl;
715  // cerr << " to find " << ancestorsOfTo[i + 1]->unparseToString() << endl;

721  - children.begin();
722  ROSE_ASSERT (positionOfChild != children.size());

724  for (unsigned int j = 0; j < positionOfChild; ++j) {
