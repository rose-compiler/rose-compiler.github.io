441  case JvmInstructionKind::monitorenter: // "Enter monitor for object";
442  case JvmInstructionKind::monitorexit: // "Exit monitor for object";
443  return Sawyer::Nothing();
444 
445  // A branch instruction but branch target(s) (offsets) need to be calculated
446  case JvmInstructionKind::tableswitch: // "Access jump table by index and jump";
447  case JvmInstructionKind::lookupswitch: // "Access jump table by key match and jump";
448  return Sawyer::Nothing();
449 
450  // Not a branching instruction
451  default:
452  return Sawyer::Nothing();
453  }
454 
455  if (insn->nOperands() == 1) {
456  if (SgAsmIntegerValueExpression *ival = isSgAsmIntegerValueExpression(insn->operand(0))) {
457  return insn->get_address() + ival->get_signedValue();
458  }
459  }
460  return Sawyer::Nothing();

463 static AddressSet
464 switchSuccessors(const SgAsmJvmInstruction* insn, bool &complete) {
465  SgAsmIntegerValueExpression* ival{nullptr};
466  AddressSet retval{};
467  Address va{insn->get_address()};
468  auto kind{insn->get_kind()};
469 
470  complete = false;
471 
472  if ((kind == JvmInstructionKind::lookupswitch) && (1 < insn->nOperands())) {
473  size_t nOperands{0};
474  int32_t defOff{0}, nPairs{0};
475  if ((ival = isSgAsmIntegerValueExpression(insn->operand(0))))
476  defOff = ival->get_signedValue();
477  if ((ival = isSgAsmIntegerValueExpression(insn->operand(1))))
478  nPairs = ival->get_signedValue();
479 
480  nOperands = 2 + 2*nPairs;
481  if (nOperands == insn->nOperands()) {
482  retval.insert(va + defOff);
483 #if DEBUG_PRINT
484  std::cout << "... switchSuccessors: insert " << va + defOff << std::endl;
485 #endif
486  for (size_t n{3}; n < nOperands; n+=2) {
487  if ((ival = isSgAsmIntegerValueExpression(insn->operand(n)))) {
488  retval.insert(va + ival->get_signedValue());
489 #if DEBUG_PRINT
490  std::cout << "... switchSuccessors: insert " << va + ival->get_signedValue() << std::endl;
491 #endif
492  } else {
493  return AddressSet{};
494  }
495  }
496  } else {
497  return AddressSet{};
498  }
499  } else if ((kind == JvmInstructionKind::tableswitch) && (2 < insn->nOperands())) {
500  size_t nOperands{0};
501  int32_t defOff{0}, low{0}, high{0};
502  if ((ival = isSgAsmIntegerValueExpression(insn->operand(0))))
503  defOff = ival->get_signedValue();
504  if ((ival = isSgAsmIntegerValueExpression(insn->operand(1))))
505  low = ival->get_signedValue();
506  if ((ival = isSgAsmIntegerValueExpression(insn->operand(2))))
507  high = ival->get_signedValue();
508 
509 #if DEBUG_PRINT
510  std::cout << "... switchSuccessors: " << insn->nOperands()
511  << ": " << defOff
512  << ": " << low
513  << ": " << high
514  << ": va: " << va
515  << ": fall_through: " << va + insn->get_size()
516  << std::endl;
517 #endif
518 
519  nOperands = 3 + high - low + 1;
520  if (nOperands == insn->nOperands()) {
521  retval.insert(va + defOff);
522 #if DEBUG_PRINT
523  if (kind == JvmInstructionKind::goto_)
524  std::cout << "WARNING: GOTO!!!\n";
525  std::cout << "... switchSuccessors (fall through): insert " << va + defOff << std::endl;
526 #endif
527  for (size_t n{3}; n < nOperands; n++) {
528  if ((ival = isSgAsmIntegerValueExpression(insn->operand(n)))) {
529  retval.insert(va + ival->get_signedValue());
530 #if DEBUG_PRINT
531  std::cout << "... switchSuccessors: insert " << va + ival->get_signedValue() << std::endl;
532 #endif
533  } else {
534  return AddressSet{};
535  }
536  }
537  } else {
538  return AddressSet{};
539  }
540  } else {
541  return AddressSet{};
542  }
543 
544  complete = true;
545  return retval;
546 }

548 AddressSet
549 Jvm::getSuccessors(SgAsmInstruction *insn_, bool &complete) const {
550  auto insn = isSgAsmJvmInstruction(insn_);
551  ASSERT_not_null(insn);
552 
553  auto kind{insn->get_kind()};
554  complete = false;
555 
556  switch (kind) {
557  case JvmInstructionKind::tableswitch: // "Access jump table by index and jump";
558  case JvmInstructionKind::lookupswitch: // "Access jump table by key match and jump";
559  return switchSuccessors(insn, complete);
560 
561  // A branch instruction but branch target is not immediately available
562  case JvmInstructionKind::invokevirtual: // "Invoke instance method; dispatch based on class";
563  case JvmInstructionKind::invokespecial: // "Invoke instance method; direct invocation...";
564  case JvmInstructionKind::invokestatic: // "Invoke a class (static) method";
565  case JvmInstructionKind::invokeinterface: // "Invoke interface method";
566  case JvmInstructionKind::invokedynamic: // "Invoke a dynamically-computed call site";
567  case JvmInstructionKind::monitorenter: // "Enter monitor for object";
568  case JvmInstructionKind::monitorexit: // "Exit monitor for object";
569  // Don't assume a CALL instruction returns to the fall-through address, but
570  // by default, EngineJvm reverses this with partitioner->autoAddCallReturnEdges(true)
571  return AddressSet{};
572 
573  case JvmInstructionKind::athrow: // "Throw exception or error";
574  return AddressSet{};
575 
576  default:
577  break;
578  }
579 
580  if (auto target{branchTarget(insn)}) {
581  AddressSet retval{*target};
582  // Add fall through target if not a branch always instruction
583  if ((kind != JvmInstructionKind::goto_) && (kind != JvmInstructionKind::goto_w)) {
584  retval.insert(insn->get_address() + insn->get_size());
585  }
586  complete = true;
587  return retval;
588  }
589 
590  return AddressSet{};
591 }
