108 /* Re-read instruction bytes from file if necessary in order to mark them as referenced. */
109 void
110 Base::mark_referenced_instructions(SgAsmInterpretation *interp, const MemoryMap::Ptr &map, const InstructionMap &insns)
111 {
112  unsigned char buf[32];
113  SgAsmGenericFile *file = NULL;
114  const SgAsmGenericFilePtrList &files = interp->get_files();
115  bool was_tracking = false; // only valid when file!=NULL (value here is to shut of used-before-defined warnings from GCC)
116  MemoryMap::ConstNodeIterator si = map->nodes().end();
117 
118  /* Re-read each instruction so the file has a chance to track the reference. */
119  try {
120  for (InstructionMap::const_iterator ii=insns.begin(); ii!=insns.end(); ++ii) {
121  SgAsmInstruction *insn = ii->second;
122  ASSERT_require(insn->get_size()<=sizeof buf);
123  Address va = insn->get_address();
124  size_t nbytes = insn->get_size();
125 
126  while (nbytes>0) {
127  /* Find the memory map segment and the file that goes with that segment (if any) */
128  if (si==map->nodes().end() || !si->key().contains(va)) {
129  if (file) {
130  file->set_trackingReferences(was_tracking);
131  file = NULL;
132  }
133  si = map->at(va).findNode();
134  if (si==map->nodes().end()) {
135  /* This byte of the instruction is not mapped. Perhaps the next one is. */
136  ++va;
137  --nbytes;
138  continue;
139  }
140 
141  /* Find the file that goes with this segment. */
142  for (size_t i=0; i<files.size(); i++) {
143  if (&(files[i]->get_data()[0]) == si->value().buffer()->data()) {
144  file = files[i];
145  was_tracking = file->get_trackingReferences();
146  file->set_trackingReferences(true);
147  break;
148  }
149  }
150  }
151 
152  /* Read the file for its reference tracking side effect. */
153  size_t sgmt_offset = va - si->key().least();
154  size_t n = std::min(nbytes, (size_t)si->key().size()-sgmt_offset);
155  if (file) {
156  size_t file_offset = si->value().offset() + sgmt_offset;
157  file->readContent(file_offset, buf, n, false);
158  }
159  nbytes -= n;
160  va += n;
161  }
162  }
163  if (file)
164  file->set_trackingReferences(was_tracking);
165  } catch(...) {
166  if (file)
167  file->set_trackingReferences(was_tracking);
168  throw;
169  }
170 }

174 Base::get_block_successors(const InstructionMap& insns, bool &complete)
175 {
176  std::vector<SgAsmInstruction*> block;
