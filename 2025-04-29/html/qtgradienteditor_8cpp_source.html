142 QGradient QtGradientEditorPrivate::gradient() const
143 {
144  QGradient *gradient = 0;
145  switch (m_ui.gradientWidget->gradientType()) {
146  case QGradient::LinearGradient:
147  gradient = new QLinearGradient(m_ui.gradientWidget->startLinear(),
148  m_ui.gradientWidget->endLinear());
149  break;
150  case QGradient::RadialGradient:
151  gradient = new QRadialGradient(m_ui.gradientWidget->centralRadial(),
152  m_ui.gradientWidget->radiusRadial(),
153  m_ui.gradientWidget->focalRadial());
154  break;
155  case QGradient::ConicalGradient:
156  gradient = new QConicalGradient(m_ui.gradientWidget->centralConical(),
157  m_ui.gradientWidget->angleConical());
158  break;
159  default:
160  break;
161  }
162  if (!gradient)
163  return QGradient();
164  gradient->setStops(m_ui.gradientWidget->gradientStops());
165  gradient->setSpread(m_ui.gradientWidget->gradientSpread());
166  gradient->setCoordinateMode(QGradient::StretchToDeviceMode);
167  QGradient gr = *gradient;
168  delete gradient;
169  return gr;
170 }

864 void QtGradientEditor::setGradient(const QGradient &grad)
865 {
866  if (grad == gradient())
867  return;
868 
869  QGradient::Type type = grad.type();
870  int idx = 0;
871  switch (type) {
872  case QGradient::LinearGradient: idx = 0; break;
873  case QGradient::RadialGradient: idx = 1; break;
874  case QGradient::ConicalGradient: idx = 2; break;
875  default: return;
876  }
877  d_ptr->setType(type);
878  d_ptr->m_ui.typeComboBox->setCurrentIndex(idx);
879  d_ptr->m_ui.gradientWidget->setGradientType(type);
880  d_ptr->m_typeGroup->button(idx)->setChecked(true);
881 
882  QGradient::Spread spread = grad.spread();
883  switch (spread) {
884  case QGradient::PadSpread: idx = 0; break;
885  case QGradient::RepeatSpread: idx = 1; break;
886  case QGradient::ReflectSpread: idx = 2; break;
887  default: idx = 0; break;
888  }
889  d_ptr->m_ui.spreadComboBox->setCurrentIndex(idx);
890  d_ptr->m_ui.gradientWidget->setGradientSpread(spread);
891  d_ptr->m_spreadGroup->button(idx)->setChecked(true);
892 
893  if (type == QGradient::LinearGradient) {
894  QLinearGradient *gr = (QLinearGradient *)(&grad);
895  d_ptr->setStartLinear(gr->start());
896  d_ptr->setEndLinear(gr->finalStop());
897  d_ptr->m_ui.gradientWidget->setStartLinear(gr->start());
898  d_ptr->m_ui.gradientWidget->setEndLinear(gr->finalStop());
899  } else if (type == QGradient::RadialGradient) {
900  QRadialGradient *gr = (QRadialGradient *)(&grad);
901  d_ptr->setCentralRadial(gr->center());
902  d_ptr->setFocalRadial(gr->focalPoint());
903  d_ptr->setRadiusRadial(gr->radius());
904  d_ptr->m_ui.gradientWidget->setCentralRadial(gr->center());
905  d_ptr->m_ui.gradientWidget->setFocalRadial(gr->focalPoint());
906  d_ptr->m_ui.gradientWidget->setRadiusRadial(gr->radius());
907  } else if (type == QGradient::ConicalGradient) {
908  QConicalGradient *gr = (QConicalGradient *)(&grad);
909  d_ptr->setCentralConical(gr->center());
910  d_ptr->setAngleConical(gr->angle());
911  d_ptr->m_ui.gradientWidget->setCentralConical(gr->center());
912  d_ptr->m_ui.gradientWidget->setAngleConical(gr->angle());
913  }
914 
915  d_ptr->m_gradientStopsController->setGradientStops(grad.stops());
916  d_ptr->m_ui.gradientWidget->setGradientStops(grad.stops());
917  d_ptr->updateGradient(false);
918 }
