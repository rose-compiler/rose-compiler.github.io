227  uint8_t *data; // line read from a file, allocated with operator new[]
228  Resources(): data(NULL) {}
229  ~Resources() { delete[] data; }
230  } r;
231 
232  //--------------------------------------
233  // Parse the parts of the locator string
234  //--------------------------------------
235 
236  // Leading colon
237  const char *s = locatorString.c_str();
238  if (':'!=*s++)
239  throw insertFileError(locatorString, "not a locator string");
240 
241  // Virtual address
242  Sawyer::Optional<Address> optionalVa;
243  if (isdigit(*s)) {
244  optionalVa = parseInteger<Address>(s /*in,out*/);
245  if (!optionalVa)
246  throw insertFileError(locatorString, "virtual address expected");
247  }
248 
249  // Virtual size
250  Sawyer::Optional<size_t> optionalVSize;
251  if ('+'==*s) {
252  ++s;
253  optionalVSize = parseInteger<size_t>(s /*in,out*/);
254  if (!optionalVSize)
255  throw insertFileError(locatorString, "virtual size expected");
256  }
257 
258  // Virtual accessibility
259  Sawyer::Optional<unsigned> optionalAccess;
260  if ('='==*s) {
261  ++s;
262  unsigned a = 0;
263  if ('r'==*s) {
264  ++s;
265  a |= READABLE;
266  }
267  if ('w'==*s) {
268  ++s;
269  a |= WRITABLE;
270  }
271  if ('x'==*s) {
272  ++s;
273  a |= EXECUTABLE;
274  }
275  optionalAccess = a;
276  }
277 
278  // Second colon
279  if (':'!=*s) {
280  if (*s && optionalAccess)
281  throw insertFileError(locatorString, "invalid access spec");
282  throw insertFileError(locatorString, "syntax error before second colon");
283  }
284  ++s;
285 
286  // File offset
287  Sawyer::Optional<size_t> optionalOffset;
288  if (isdigit(*s)) {
289  optionalOffset = parseInteger<size_t>(s /*in,out*/);
290  if (!optionalOffset)
291  throw insertFileError(locatorString, "file offset expected");
292  }
293 
294  // File size
295  Sawyer::Optional<size_t> optionalFSize;
296  if ('+'==*s) {
297  ++s;
298  optionalFSize = parseInteger<size_t>(s /*in,out*/);
299  if (!optionalFSize)
300  throw insertFileError(locatorString, "file size expected");
301  }
302 
303  // Third colon
304  if (':'!=*s)
305  throw insertFileError(locatorString, "syntax error before third colon");
306  ++s;
307 
308  // File name
309  if (!*s)
310  throw insertFileError(locatorString, "file name expected after third colon");
311  std::string fileName = s;
312  if (fileName.size()!=strlen(fileName.c_str()))
313  throw insertFileError(locatorString, "invalid file name");
314  std::string segmentName = FileSystem::toString(boost::filesystem::path(fileName).filename());
315 
316  //--------------------------------
317  // Open the file and read the data
318  //--------------------------------
319 
320  // Open the file and seek to the start of data
321  std::ifstream file(fileName.c_str());
322  if (!file.good())
323  throw std::runtime_error("MemoryMap::insertFile: cannot open file \""+StringUtility::cEscape(fileName)+"\"");
324  if (optionalOffset)
325  file.seekg(*optionalOffset);
326  if (!file.good())
327  throw std::runtime_error("MemoryMap::insertFile: cannot seek in file \""+StringUtility::cEscape(fileName)+"\"");
328 
329  // If no file size was specified then try to get one, or delay getting one until later. On POSIX systems we can use stat
330  // to get the file size, which is useful because infinite devices (like /dev/zero) will return zero. Otherwise we'll get
331  // the file size by trying to read from the file.
332 #if !defined(BOOST_WINDOWS) // not targeting Windows; i.e., not Microsoft C++ and not MinGW
333  if (!optionalFSize) {
334  struct stat sb;
335  if (0==stat(fileName.c_str(), &sb))
336  optionalFSize = sb.st_size;
337  }
338 #endif
339 
340  // Limit the file size according to the virtual size. We never need to read more than what would be mapped.
341  if (optionalVSize) {
342  if (optionalFSize) {
343  optionalFSize = std::min(*optionalFSize, *optionalVSize);
344  } else {
345  optionalFSize = optionalVSize;
346  }
347  }
348 
349  // Read the file data. If we know the file size then we can allocate a buffer and read it all in one shot, otherwise we'll
350  // have to read a little at a time (only happens on Windows due to stat call above).
351  size_t nRead = 0; // bytes of data actually allocated, read, and initialized in "data"
352  if (optionalFSize) {
353  // This is reasonably fast and not too bad on memory
354  if (0 != *optionalFSize) {
355  r.data = new uint8_t[*optionalFSize];
356  file.read((char*)r.data, *optionalFSize);
357  nRead = file.gcount();
358  if (nRead != *optionalFSize)
359  throw std::runtime_error("MemoryMap::insertFile: short read from \""+StringUtility::cEscape(fileName)+"\"");
360  }
361  } else {
362  while (file.good()) {
363  uint8_t page[4096];
364  file.read((char*)page, sizeof page);
365  size_t n = file.gcount();
366  uint8_t *tmp = new uint8_t[nRead + n];
367  memcpy(tmp, r.data, nRead);
368  memcpy(tmp+nRead, page, n);
369  delete[] r.data;
370  r.data = tmp;
371  nRead += n;
372  }
373  optionalFSize = nRead;
374  }
375 
376  // Choose virtual size
377  if (!optionalVSize) {
378  ASSERT_require(optionalFSize);
379  optionalVSize = optionalFSize;
380  }
381 
382  // Choose accessibility
383  if (!optionalAccess) {
384 #ifdef BOOST_WINDOWS
385  optionalAccess = READABLE | WRITABLE;
386 #else
387  unsigned a = 0;
388  if (0==::access(fileName.c_str(), R_OK))
389  a |= READABLE;
390  if (0==::access(fileName.c_str(), W_OK))
391  a |= WRITABLE;
392  if (0==::access(fileName.c_str(), X_OK))
393  a |= EXECUTABLE;
394  optionalAccess = a;
395 #endif
396  }
397

399  if (!optionalVa) {
400  ASSERT_require(optionalVSize);
401  optionalVa = findFreeSpace(*optionalVSize);
402  }
403 
404  // Adjust the memory map
405  ASSERT_require(optionalVa);
406  ASSERT_require(optionalVSize);
407  ASSERT_require(optionalAccess);
408  ASSERT_require(nRead <= *optionalVSize);
409  if (0 == *optionalVSize)
410  return AddressInterval(); // empty
411  AddressInterval interval = AddressInterval::baseSize(*optionalVa, *optionalVSize);
412  insert(interval, Segment::anonymousInstance(interval.size(), *optionalAccess, segmentName));
413  size_t nCopied = at(interval.least()).limit(nRead).write(r.data).size();

415  return interval;
416 }

673 MemoryMap::insertData(const std::string &locatorString) {
674  const char *s = locatorString.c_str();
675  if (':' != *s++)
676  throw insertDataError(locatorString, "not a locator string");
677 
678  // Virtual address
679  Sawyer::Optional<Address> optionalVa;
680  if (isdigit(*s)) {
681  optionalVa = parseInteger<Address>(s /*in,out*/);
682  if (!optionalVa)
683  throw insertDataError(locatorString, "virtual address expected");
684  }
685 
686  // Virtual size
687  Sawyer::Optional<size_t> optionalVSize;
688  if ('+' == *s) {
689  ++s;
690  optionalVSize = parseInteger<size_t>(s /*in,out*/);
691  if (!optionalVSize)
692  throw insertDataError(locatorString, "virtual size expected");
693  }
694 
695  // Virtual accessibility
696  unsigned accessFlags = READ_WRITE_EXECUTE;
697  if ('='==*s) {
698  ++s;
699  accessFlags = 0;
700  if ('r'==*s) {
701  ++s;
702  accessFlags |= READABLE;
703  }
704  if ('w'==*s) {
705  ++s;
706  accessFlags |= WRITABLE;
707  }
708  if ('x'==*s) {
709  ++s;
710  accessFlags |= EXECUTABLE;
711  }
712  }
713 
714  // Second colon
715  if (':'!=*s) {
716  if (*s && accessFlags)
717  throw insertDataError(locatorString, "invalid access specification");
718  throw insertDataError(locatorString, "syntax error before second colon");
719  }
720  ++s;
721 
722  // Third colon
723  if (':'!=*s)
724  throw insertDataError(locatorString, "syntax error before third colon");
725  ++s;
726 
727  // The data
728  std::vector<uint8_t> data;
729  while (1) {
730  while (isspace(*s)) ++s;
731  if (!*s)
732  break;
733  Address u = 0;
734  if (!parseInteger<Address>(s /*in,out*/).assignTo(u))
735  throw insertDataError(locatorString, "expected numeric value for byte " + StringUtility::numberToString(data.size()));
736  if (u > 0xff)
737  throw insertDataError(locatorString, "value " + StringUtility::numberToString(u) +
738  " is out of bounds for byte " + StringUtility::numberToString(data.size()));
739  data.push_back(u);
740  }
741  if (!optionalVSize) {
742  optionalVSize = data.size();
743  } else if (*optionalVSize > data.size()) {
744  mlog[WARN] <<"data (" <<StringUtility::plural(data.size(), "bytes") <<") will be truncated to"
745  <<" specified segment size (" <<StringUtility::plural(*optionalVSize, "bytes") <<")\n";
746  } else if (0 == *optionalVSize) {
747  mlog[WARN] <<"data is empty; nothing to map for \"" <<StringUtility::cEscape(locatorString) <<"\"\n";
748  return AddressInterval();
749  }
750 
751  // Find a place to map the file
752  ASSERT_require(optionalVSize);
753  if (!optionalVa) {
754  optionalVa = findFreeSpace(*optionalVSize);
755  if (!optionalVa) {
756  mlog[ERROR] <<"no virtual address specified and not enough space available"
757  <<" for \"" <<StringUtility::cEscape(locatorString) <<"\"\n";
758  }
759  }
760 
761  // Adjust the memory map
762  ASSERT_require(optionalVa);
763  ASSERT_require(optionalVSize);
764  ASSERT_require(*optionalVSize > 0);
765  AddressInterval interval = AddressInterval::baseSize(*optionalVa, *optionalVSize);
766  insert(interval, Segment::anonymousInstance(data.size(), accessFlags, "data"));
767  size_t nCopied = at(interval.least()).write(data).size();
768  ASSERT_always_require(data.size() == nCopied);

1128 Sawyer::Optional<Address>
1129 MemoryMap::findAny(const AddressInterval &limits, const std::vector<uint8_t> &bytesToFind,
1130  unsigned requiredPerms, unsigned prohibitedPerms) const {
1131  if (limits.isEmpty())
1132  return Sawyer::Nothing();
1133 
1134  // Start small, then increase to bufMaxSize.
1135  std::vector<uint8_t> buf(10 * bytesToFind.size());
1136  size_t bufMaxSize = std::max(buf.size(), (size_t)65536);
1137 
1138  // Search...
1139  Address va = limits.least();
1140  AddressInterval prevBuffer; // location of buffer previous time through loop
1141  while (atOrAfter(va).require(requiredPerms).prohibit(prohibitedPerms).next().assignTo(va)) {
1142  // Since the thing for which we're searching could overlap between two buffers (the previous loop iteration and this
1143  // iteration), we might need to preserve some of the previous buffer contents.
1144  size_t bufOffset = 0; // where in the buffer to put the memory about to be read
1145  if (!prevBuffer.isEmpty() && prevBuffer.greatest() + 1 == va) {
1146  bufOffset = bytesToFind.size() - 1; // number of bytes that need to be preserved from previous iteration
1147  memmove(buf.data(), buf.data() + (prevBuffer.size() - bufOffset), bufOffset); // move to front of buffer
1148  }
1149 
1150  // Read memory into the buffer and describe what the buffer contains.
1151  size_t maxRead = buf.size() - bufOffset;
1152  size_t nRead = at(va).require(requiredPerms).prohibit(prohibitedPerms).limit(maxRead).read(buf.data() + bufOffset).size();
1153  ASSERT_require(nRead > 0);
1154  AddressInterval curBuffer = AddressInterval::baseSize(va - bufOffset, bufOffset + nRead);
1155 
1156  // Search
1157  for (size_t i = 0; i + bytesToFind.size() < curBuffer.size(); ++i) {
1158  bool found = true;
1159  for (size_t j = 0; j < bytesToFind.size(); ++j) {
1160  if (buf[i+j] != bytesToFind[j]) {
1161  found = false;
1162  break;
1163  }
1164  }
1165  if (found)
1166  return curBuffer.least() + i;
1167  }
1168 
1169  // Avoid overflow
1170  if (curBuffer.greatest() == hull().greatest())
1171  break;
1172  va = curBuffer.greatest() + 1;
