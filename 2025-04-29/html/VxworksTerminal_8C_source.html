201 VxworksTerminal::parseLocatorString(const std::string &locatorString, const Settings &dflt) {
202  auto locatorError = [&locatorString](const std::string &message, const size_t locatorOffset) {
203  return Sawyer::makeError(MemoryMap::LocatorError(message, locatorString, locatorOffset));
204  };
205 
206  // The memory part between the first two colons
207  if (!boost::starts_with(locatorString, ":"))
208  return locatorError("not a valid VxWorks locator string (missing colon)", 0);
209  const size_t secondColon = locatorString.find(':', 1);
210  if (secondColon == std::string::npos)
211  return locatorError("not a valid VxWorks locator string (missing second colon)", locatorString.size());
212  const auto addrSizePerm = MemoryMap::parseAddrSizePerm(locatorString.substr(1, secondColon - 1));
213  if (!addrSizePerm)
214  return locatorError(addrSizePerm.unwrapError().first, 1 + addrSizePerm.unwrapError().second);
215  if (dflt.where && !addrSizePerm->addr)
216  return locatorError("memory address required", 1);
217  if (dflt.where && !addrSizePerm->size)
218  return locatorError("size in bytes required", 1 + addrSizePerm->sizeAt);
219 
220  // The connection part between the second and third colons
221  std::vector<std::tuple<std::string /*name*/, std::string /*value*/, size_t /*index*/>> nameValuePairs;
222  std::string remaining;
223  size_t remainingAt;
224  std::tie(nameValuePairs, remaining, remainingAt) = MemoryMap::parseNameValuePairs(locatorString.substr(secondColon + 1));
225  Sawyer::Optional<std::string> prompt;
226  Sawyer::Optional<double> delaySeconds;
227  Sawyer::Optional<double> timeoutSeconds;
228  Sawyer::Optional<size_t> swath;
229  Sawyer::Optional<uint16_t> port;
230  std::regex durationRe("([0-9]+)(s|ms)?");
231  for (const auto &tuple: nameValuePairs) {
232  const std::string &name = std::get<0>(tuple);
233  const std::string &value = std::get<1>(tuple);
234  const size_t nameAt = secondColon + 1 + std::get<2>(tuple);
235  const size_t valueAt = nameAt + name.size() + 1;
236 
237  if (name == "prompt") {
238  if (prompt) {
239  return locatorError("duplicate \"prompt\" setting", nameAt);
240  } else if (value.empty()) {
241  return locatorError("\"prompt\" value cannot be empty", valueAt);
242  } else if (value.find('\n') != std::string::npos) {
243  return locatorError("\"prompt\" value cannot contain line-feed characters", valueAt);
244  } else {
245  prompt = value;
246  }
247 
248  } else if (name == "delay") {
249  std::smatch found;
250  if (delaySeconds) {
251  return locatorError("duplicate \"delay\" setting", nameAt);
252  } else if (!std::regex_match(value, found, durationRe)) {
253  return locatorError("syntax error in duration", valueAt);
254  } else if (const auto parsed = Sawyer::parse<unsigned>(found.str(1))) {
255  delaySeconds = (double)*parsed * (found.str(2) == "ms" ? 0.001 : 1.0);
256  } else {
257  return locatorError("delay " + parsed.unwrapError(), valueAt);
258  }
259 
260  } else if (name == "timeout") {
261  std::smatch found;
262  if (timeoutSeconds) {
263  return locatorError("duplicate \"timeout\" setting", nameAt);
264  } else if (!std::regex_match(value, found, durationRe)) {
265  return locatorError("syntax error in duration", valueAt);
266  } else if (const auto parsed = Sawyer::parse<unsigned>(found.str(1))) {
267  timeoutSeconds = (double)*parsed * (found.str(2) == "ms" ? 0.001 : 1.0);
268  } else {
269  return locatorError("timeout " + parsed.unwrapError(), valueAt);
270  }
271 
272  } else if (name == "swath") {
273  if (swath) {
274  return locatorError("duplicate \"swath\" setting", nameAt);
275  } else if (const auto parsed = Sawyer::parse<size_t>(value)) {
276  swath = *parsed;
277  } else {
278  return locatorError("swath " + parsed.unwrapError(), valueAt);
279  }
280 
281  } else if (name == "port") {
282  if (port) {
283  return locatorError("duplicate \"port\" setting", nameAt);
284  } else if (const auto parsed = Sawyer::parse<uint16_t>(value)) {
285  port = *parsed;
286  } else {
287  return locatorError("port " + parsed.unwrapError(), valueAt);
288  }
289 
290  } else {
291  return locatorError("unrecognized setting \"" + name + "\"", nameAt);
292  }
293  }
294  if (boost::starts_with(remaining, ","))
295  return locatorError("invalid name=value pair", secondColon + 1 + remainingAt + 1);
296 
297  // After the third colon is a host name or IP address
298  std::regex hostIpRe("((25[0-5]|2[0-4][0-9]|1[0-9][0-9]|[1-9][0-9]|[0-9])\\.){3}"
299  "(25[0-5]|2[0-4][0-9]|1[0-9][0-9]|[1-9][0-9]|[0-9])");
300  std::regex hostNameRe("(([a-zA-Z0-9]|[a-zA-Z0-9][-a-zA-Z0-9]*[a-zA-Z0-9])\\.)*"
301  "([A-Za-z0-9]|[A-Za-z0-9][-A-Za-z0-9]*[A-Za-z0-9])");
302  const size_t thirdColon = secondColon + 1 + remainingAt; // third colon expected here
303  if (!boost::starts_with(remaining, ":")) {
304  if (remaining.empty()) {
305  return locatorError("not a valid VxWorks locator string (missing third colon)", thirdColon);
306  } else {
307  return locatorError("invalid name=value pair", thirdColon);
308  }
309  }
310  const std::string host = locatorString.substr(thirdColon + 1);
311  if (host.empty())
312  return locatorError("not a valid VxWorks locator string: no host name or address", locatorString.size());
313  if (std::regex_match(host, hostIpRe)) {
314  // okay
315  } else if (std::regex_match(host, hostNameRe)) {
316  if (host.size() > 255)
317  return locatorError("invalid RFC 1123 host name: total length must be less than 256 octets", thirdColon + 1);
318  std::vector<std::string> labels;
319  size_t labelAt = thirdColon + 1;
320  boost::split(labels, host, boost::is_any_of("."));
321  for (const std::string &label: labels) {
322  if (label.empty() || label.size() > 63)
323  return locatorError("invalid RFC 1123 host name: label must be between 1 and 63 octets", labelAt);
324  labelAt += label.size() + 1;
325  }
326  } else {
327  return locatorError("invalid RFC 1123 host name or IPv4 address", thirdColon + 1);
328  }
329 
330  // Save the results
331  Settings retval = dflt;
332  if (addrSizePerm->addr) {
333  retval.where =
334  AddressInterval::baseSize(*addrSizePerm->addr, addrSizePerm->size.orElse(dflt.where ? dflt.where.size() : 1));
335  } else if (addrSizePerm->size) {
336  retval.where = AddressInterval::baseSize(dflt.where ? dflt.where.least() : 0, *addrSizePerm->size);
337  }
338  if (addrSizePerm->perm)
339  retval.permissions = *addrSizePerm->perm;
340  if (prompt)
341  retval.prompt = *prompt;
342  if (delaySeconds)
343  retval.delaySeconds = *delaySeconds;
344  if (timeoutSeconds)
345  retval.timeoutSeconds = *timeoutSeconds;
346  if (swath)
347  retval.bytesPerRequest = *swath;
348  if (port)
349  retval.port = *port;
350  retval.host = host;
351 
352  SAWYER_MESG(mlog[DEBUG]) <<"parsed VxWorks locator string \"" + StringUtility::cEscape(locatorString) + "\"\n"
353  <<retval;
354  return Sawyer::makeOk(retval);
355 }

358 VxworksTerminal::open() {
359 #ifdef BOOST_WINDOWS
360  return "VxWorks connection is not supported on Windows";
361 #else
362 
363  // C resources that must be cleaned up on return or exceptions
364  struct Resources {
365  int server = -1; // file descriptor connected to the telnet (or other) server
366  struct addrinfo *serverAddrs = nullptr;
367 
368  ~Resources() {
369  if (server != -1)
370  close(server);
371  if (serverAddrs)
372  freeaddrinfo(serverAddrs);
373  }
374  } r;
375 
376  if (isOpen())
377  return std::string("already open");
378 
379  // Look up the server IP address
380  {
381  struct addrinfo hints;
382  memset(&hints, 0, sizeof hints);
383  hints.ai_flags = AI_ADDRCONFIG;
384  hints.ai_family = AF_INET;
385  hints.ai_socktype = SOCK_STREAM;
386  const int ecode = getaddrinfo(settings_.host.c_str(), (boost::format("%d") % settings_.port).str().c_str(), &hints,
387  &r.serverAddrs);
388  if (ecode == EAI_SYSTEM) {
389  return std::string("VxWorks connection getaddrinfo failed: ") + strerror(errno);
390  } else if (ecode != 0) {
391  return std::string("VxWorks connection getaddrinfo failed: ") + gai_strerror(ecode);
392  }
393  ASSERT_not_null(r.serverAddrs);
394  }
395 
396  // Connect to the server
397  for (struct addrinfo *addr = r.serverAddrs; addr; addr = addr->ai_next) {
398  r.server = socket(AF_INET, SOCK_STREAM, 0);
399  if (-1 == r.server) {
400  if (!addr->ai_next) {
401  return std::string("VxWorks connection socket failed: ") + strerror(errno);
402  } else {
403  continue;
404  }
405  }
406 
407  if (connect(r.server, addr->ai_addr, addr->ai_addrlen) == -1) {
408  if (!addr->ai_next) {
409  return std::string("VxWorks connection connect failed: ") + strerror(errno);
410  } else {
411  continue;
412  }
413  }
414  }
415 
416  // Make the socket non-blocking
417  {
418  const int flags = fcntl(r.server, F_GETFL, 0);
419  if (-1 == flags)
420  return std::string("VxWorks connection fntl(F_GETFL) failed: ") + strerror(errno);
421  if (-1 == fcntl(r.server, F_SETFL, flags | O_NONBLOCK))
422  return std::string("VxWorks connection fntl(F_SETFL) failed: ") + strerror(errno);
423  }
424 
425  std::swap(server_, r.server);
426  SAWYER_MESG(mlog[DEBUG]) <<"connected to server, fd=" <<server_ <<"\n";
427  return Sawyer::Nothing();
428 #endif
429 }

437 class Download: public VxworksTerminal::DataTransfer {
438  VxworksTerminal &terminal;
439  AddressInterval remaining; // addresses remaining to be read
440  Alignment lineAlignment; // alignment of data lines
441  Sawyer::Message::Stream debug, warn;
442  std::regex dataMatchRe; // regex to match a line of data
443  Sawyer::ProgressBar<size_t> progress;
444 
445 public:
446  std::vector<uint8_t> data; // data tha twas read
447 
448 public:
449  Download(VxworksTerminal &terminal, const AddressInterval &where)
450  : terminal(terminal), remaining(where), lineAlignment(16, 32), debug(terminal.mlog[DEBUG]), warn(terminal.mlog[WARN]),
451  dataMatchRe(".*?" // prompt
452  "((?:0x)?[0-9a-fA-F]{8}): " // line address
453  "( *)" // 3 spaces per skipped byte
454  "((?: [0-9a-fA-F]{2})+)" // numeric data
455  "\\s*\\*.{16}\\*"), // ASCII data
456  progress(where.size(), terminal.mlog[MARCH], "transfer")
457  {
458  if (remaining)
459  data.reserve(remaining.size());
460  progress.suffix(" bytes");
461  }

463  std::string handleInput(const std::string &line) {
464  using StringUtility::addrToString;
465  using StringUtility::plural;
466 
467  std::smatch found;
468  if (remaining && std::regex_match(line, found, dataMatchRe)) {
469  Sawyer::Message::Stream debug(VxworksTerminal::mlog[DEBUG]);
470  Sawyer::Message::Stream warn(VxworksTerminal::mlog[WARN]);
471 
472  SAWYER_MESG(debug) <<"found data line: \"" <<line <<"\"\n";
473  const Address lineAddr = *Sawyer::parse<Address>(found.str(1));
474  const Address startAddr = lineAddr + found.str(2).size() / 3;
475  SAWYER_MESG(debug) <<" remaining=" <<addrToString(remaining)
476  <<", lineAddr=" <<addrToString(lineAddr)
477  <<", startAddr=" <<addrToString(startAddr)
478  <<", nspaces=" <<found.str(2).size() <<"\n";
479 
480  if (startAddr != remaining.least()) {
481  SAWYER_MESG(warn) <<"expected data at " <<addrToString(remaining.least())
482  <<" but got data at " <<addrToString(startAddr) <<" (ignoring line)\n";
483  return "";
484  }
485  const size_t maxBytesExpected = std::min(16 - (startAddr - lineAddr), remaining.size());
486  ASSERT_require2(maxBytesExpected > 1 && maxBytesExpected <= 16, maxBytesExpected);
487 
488  // Parse the data
489  std::vector<std::string> parts;
490  boost::regex spacesRe(" +");
491  boost::split_regex(parts, boost::trim_copy(found.str(3)), spacesRe);
492  if (parts.size() > maxBytesExpected) {
493  SAWYER_MESG(warn) <<"expected up to " <<plural(maxBytesExpected, "bytes") <<" but got " <<parts.size() <<"\n";
494  return "";
495  }
496  for (const std::string &part: parts)
497  data.push_back(*Sawyer::parse<uint8_t>("0x" + part));
498  progress += parts.size();
499 
500  // Update addresses remaining
501  if (remaining.size() == parts.size()) {
502  remaining = AddressInterval();
503  } else {
504  remaining = AddressInterval::hull(startAddr + parts.size(), remaining.greatest());
505  }
506  }
507  return "";
508  }

514  } else {
515  state = State::TERMINATE;
516  return "";
517  }
518  }

552 VxworksTerminal::download(const MemoryMap::Ptr &map, const AddressInterval &where, const std::string &name) {
553  ASSERT_not_null(map);

555  ASSERT_require(data->size() == where.size());
