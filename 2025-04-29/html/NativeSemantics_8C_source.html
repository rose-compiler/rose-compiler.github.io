139 BaseSemantics::SValue::Ptr
140 MemoryState::peekMemory(const BaseSemantics::SValue::Ptr &address, const BaseSemantics::SValue::Ptr &dflt,
141  BaseSemantics::RiscOperators */*addrOps*/, BaseSemantics::RiscOperators */*valOps*/) {
142  ASSERT_not_null(process_);
143  std::vector<uint8_t> buffer((dflt->nBits() + 7) / 8);
144  ASSERT_require(buffer.size() <= 8);
145  Address va = SValue::promote(address)->toUnsigned().get();
146  size_t nRead = process_->readMemory(va, buffer.size(), &buffer[0]);
147  if (nRead != buffer.size())
148  throw Exception("short read from process at " + StringUtility::addrToString(va));
149 
150  uint64_t value = 0;
151  switch (get_byteOrder()) {
152  case ByteOrder::ORDER_LSB:
153  for (size_t i = 0; i < buffer.size(); ++i)
154  value |= (uint64_t)buffer[i] << (8*i);
155  break;
156  case ByteOrder::ORDER_MSB:
157  for (size_t i = 0; i < buffer.size(); ++i)
158  value = (value << 8) | (uint64_t)buffer[i];
159  break;
160  default:
161  ASSERT_not_reachable("invalid byte order");
162  }
163 
164  return get_val_protoval()->number_(8*buffer.size(), value);
165 }
