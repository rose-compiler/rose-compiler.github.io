142 /**********************************************************
143  * Union of two maps
144  *********************************************************/
145 void DefUseAnalysis::mapAnyUnion(tabletype* tabl, SgNode* before, SgNode* other, SgNode* sgNode) {
146 
147  bool beforeFound = true;
148  if ((*tabl).find(before)==(*tabl).end())
149  beforeFound = false;
150  bool otherFound = true;
151  if ((*tabl).find(other)==(*tabl).end())
152  otherFound = false;
153 
154  addID(sgNode);
155 
156 #if ROSE_GCC_OMP
157 #pragma omp critical (DefUseAnalysismapUse)
158 #endif
159  if (!beforeFound) {
160  if (!otherFound)
161  (*tabl)[sgNode].clear(); // both before and other nodes have empty sets
162  else // only other node has a set
163  (*tabl)[sgNode]=(*tabl)[other];
164  } else {
165  if (!otherFound) // only before node has a set
166  (*tabl)[sgNode]=(*tabl)[before];
167  else { // both has a set, perform the actual union operation : insert two sets into a single set
168  const multitype& multiA = (*tabl)[before];
169  const multitype& multiB = (*tabl)[other];
170  std::set<std::pair<SgInitializedName*, SgNode*> > s_before(multiA.begin(), multiA.end());
171  ROSE_ASSERT (s_before.size() == (*tabl)[before].size());
172 
173  s_before.insert(multiB.begin(), multiB.end());
174  multitype multiC(s_before.begin(), s_before.end());
175  (*tabl)[sgNode].swap(multiC);
176  }
177  }
178 }
