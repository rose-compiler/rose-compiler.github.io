54 bool
55 MatchStwuPrologue::match(const Partitioner::ConstPtr &partitioner, Address anchor) {
56  ASSERT_not_null(partitioner);
57  if (partitioner->instructionExists(anchor))
58  return false; // already in the CFG/AUM
59  SgAsmPowerpcInstruction *insn = isSgAsmPowerpcInstruction(partitioner->discoverInstruction(anchor));
60  if (!insn)
61  return false; // cannot decode an instruction at this address
62 
63  // Look for "stwu r1, u32 [r1 + DELTA]" where DELTA is a negative value
64  if (insn->get_kind() == powerpc_stwu && insn->nOperands() == 2) {
65  // First operand must be "r1"
66  SgAsmDirectRegisterExpression *reg = isSgAsmDirectRegisterExpression(insn->operand(0));
67  RegisterDescriptor r1 = reg ? reg->get_descriptor() : RegisterDescriptor();
68  if (r1.majorNumber() != powerpc_regclass_gpr || r1.minorNumber() != 1 ||
69  r1.offset() != 0 || (r1.nBits() != 32 && r1.nBits() != 64))
70  return false; // not the usual frame pointer register, r1
71 
72  // Second operand must be either:
73  // 'u32 [r1 + DELTA]' where DELTA is negative, or
74  // 'u64 [r1 + DELTA]' where DELTA is negative
75  SgAsmMemoryReferenceExpression *mre = isSgAsmMemoryReferenceExpression(insn->operand(1));
76  SgAsmBinaryAdd *addr = mre ? isSgAsmBinaryAdd(mre->get_address()) : NULL;
77  SgAsmDirectRegisterExpression *dre = addr ? isSgAsmDirectRegisterExpression(addr->get_lhs()) : NULL;
78  RegisterDescriptor r2 = dre ? dre->get_descriptor() : RegisterDescriptor();
79  if (r2 != r1)
80  return false;
81  int64_t delta = addr->get_rhs()->asSigned().orElse(0);
82  if (delta >= 0)
83  return false;
84 
85  function_ = Function::instance(anchor, SgAsmFunction::FUNC_PATTERN);
86  function_->reasonComment("matched STWU r1, [r1 + <delta>] s.t. delta < 0");
87  return true;
88  }
89 
90  return false;
91 }

109  return Sawyer::Nothing();

112  if (!rre || rre->get_descriptor() != REG_R11)
113  return Sawyer::Nothing();

118  ival = isSgAsmIntegerValueExpression(addis->get_operandList()->get_operands()[2]);

126  return Sawyer::Nothing();

129  return Sawyer::Nothing();

132  rre = add ? isSgAsmDirectRegisterExpression(add->get_lhs()) : NULL;
133  if (!rre || rre->get_descriptor() != REG_R11)
134  return Sawyer::Nothing();

136  if (!ival)
137  return Sawyer::Nothing();

139 
140  // Target address must be in the ".plt" section.
141  if (!pltAddresses.contains(target))
142  return Sawyer::Nothing();
143 
144  // mtspr ctr, r11
145  SgAsmPowerpcInstruction *mtspr = isSgAsmPowerpcInstruction(bb->instructions()[2]);

147  return Sawyer::Nothing();

153  if (!rre || rre->get_descriptor() != REG_R11)
154  return Sawyer::Nothing();

157  SgAsmPowerpcInstruction *bcctr = isSgAsmPowerpcInstruction(bb->instructions()[3]);
158  if (!bcctr || bcctr->get_kind() != powerpc_bcctr || bcctr->get_operandList()->get_operands().size() != 3)
159  return Sawyer::Nothing();

163  ival = isSgAsmIntegerValueExpression(bcctr->get_operandList()->get_operands()[2]);
164  if (!ival || ival->get_absoluteValue() != 0)

172  ASSERT_not_null(partitioner);

174  return;
175

179  for (SgAsmElfSection *section: pltSections) {
180  if (section->isMapped())
181  pltAddresses |= AddressInterval::baseSize(section->get_mappedActualVa(), section->get_mappedSize());
182  }
183 
184  // Look at all the functions that don't have names yet. If the function looks like a PowerPC ELF dynamic function
185  // stub that goes through a particular PLT slot, then add "@plt" to the base name. If the PLT slot has a corresponding
186  // entry in the .rela.plt section, then use that section's symbol table to get the base name of the function.
187  for (const Function::Ptr &function: partitioner->functions()) {
188  if (!function->name().empty())
189  continue;

191  Sawyer::Optional<Address> pltSlotVa = matchElfDynamicStub(partitioner, function, pltAddresses);
192  if (!pltSlotVa)
193  continue;
194

196  for (SgAsmGenericSection *genericRelocSection: relocSections) {

198  if (!relocs || !relocs->get_entries())

201  if (!symtab)

211  }
212  }
213  }
214 
215  if (!function->name().empty())
216  break;
217  }
218 
219  if (function->name().empty())
