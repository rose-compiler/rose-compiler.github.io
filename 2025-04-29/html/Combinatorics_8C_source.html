91 uint64_t
92 Hasher::make64Bits(const Digest &digest) {
93  uint64_t outHash64 = 0;
94  if(digest.size() <= 8) {
95  //If hash <= 64bits, just copy it into the 64 bit data.
96  //If the length of the hash was known statically, a cast would be sufficient
97  for (auto it = digest.begin(); it != digest.end(); ++it) {
98  outHash64 = (outHash64 << 8) ^ uint64_t{*it};
99  }
100  return outHash64;
101  }
102  //Otherwise, compress to 64 bits using XOR. This is Robb's algorithm.
103  for (uint8_t byte: digest) {
104  outHash64 = BitOps::rotateLeft(outHash64, 8) ^ uint64_t{byte};
105  }
106  return outHash64;

306 // and size to reflect the number of bytes we just consumed or saved.
307 bool
308 HasherSha256Builtin::getNextChunk(const uint8_t *&message /*in,out*/, size_t &messageSize /*in,out*/, uint32_t words[16] /*out*/) {
309  ASSERT_require(leftoverBytes_.size() < 16 * 4);
310  ASSERT_require(message != NULL || 0 == messageSize);
311 
312  if (leftoverBytes_.size() + messageSize >= 16 * 4) {
313  // Obtain the words if the previously saved data plus the current message is long enough
314  for (size_t i=0; i<16; ++i) {
315  uint32_t word = (messageByte(i*4+0, message, messageSize) << 24) |
316  (messageByte(i*4+1, message, messageSize) << 16) |
317  (messageByte(i*4+2, message, messageSize) << 8) |
318  (messageByte(i*4+3, message, messageSize) );
319  words[i] = word;
320  }
321 
322  size_t toConsume = 16 * 4 - leftoverBytes_.size();
323  message += toConsume;
324  messageSize -= toConsume;
325  leftoverBytes_.clear();
326  return true;
327 
328  } else {
329  // Still not enough data to create 16 32-bit words
330  leftoverBytes_.insert(leftoverBytes_.end(), message, message + messageSize);
331  message += messageSize;
332  messageSize = 0;
333  return false;

337 void
338 HasherSha256Builtin::accumulateChunk(const uint32_t chunk[16]) {
339  using namespace IntegerOps;
340 
341  // Initialize the message schedule array, w
342  uint32_t w[64];
343  memcpy(w, chunk, 16*4);
344  for (size_t i=16; i<64; ++i) {
345  uint32_t sigma0 = rotateRight<32>(w[i-15], 7) ^ rotateRight<32>(w[i-15], 18) ^ (w[i-15] >> 3);
346  uint32_t sigma1 = rotateRight<32>(w[i- 2], 17) ^ rotateRight<32>(w[i- 2], 19) ^ (w[i- 2] >> 10);
347  w[i] = sigma1 + w[i-7] + sigma0 + w[i-16];
348  }
349 
350  // Initialize local variables to be updated each round
351  uint32_t a = state_[0];
352  uint32_t b = state_[1];
353  uint32_t c = state_[2];
354  uint32_t d = state_[3];
355  uint32_t e = state_[4];
356  uint32_t f = state_[5];
357  uint32_t g = state_[6];
358  uint32_t h = state_[7];
359 
360  // Perform 64 rounds using the message schedule, w, initialized above
361  for (size_t i=0; i<64; ++i) {
362  const uint32_t s1 = rotateRight<32>(e, 6) ^ rotateRight<32>(e, 11) ^ rotateRight<32>(e, 25);
363  const uint32_t choice = (e & f) ^ (~e & g); // bit-wise "?:" operator
364  const uint32_t temp1 = h + s1 + choice + roundConstants_[i] + w[i];
365  const uint32_t s0 = rotateRight<32>(a, 2) ^ rotateRight<32>(a, 13) ^ rotateRight<32>(a, 22);
366  const uint32_t majority = (a & b) ^ (a & c) ^ (b & c); // bit set if at least two inputs are set
367  const uint32_t temp2 = s0 + majority;
368 
369  h = g;
370  g = f;
371  f = e;
372  e = d + temp1;
373  d = c;
374  c = b;
375  b = a;
376  a = temp1 + temp2;
377  }
378 
379  // Save the new state
380  state_[0] += a;
381  state_[1] += b;
382  state_[2] += c;
383  state_[3] += d;
384  state_[4] += e;
385  state_[5] += f;
386  state_[6] += g;
387  state_[7] += h;
388  processedBytes_ += 16 * 4;
389 }

404 const Hasher::Digest&
405 HasherSha256Builtin::digest() {
406  if (digest_.empty()) {
407  // Pad the message by appending an 0x80 followed by zero bytes, followed by the 8-byte big-endian message length so
408  // that the total length (message length plus paddng) is a multiple of 64 bytes.
409  const size_t messageSizeBytes = processedBytes_ + leftoverBytes_.size();
410  const size_t finalBlockSizeBytes = (messageSizeBytes + 1 /*0x80*/ + 8 /*length*/) % 64; // w/out zero padding [0,63]
411  const size_t nZeroPadding = 0 == finalBlockSizeBytes ? 0 : 64 - finalBlockSizeBytes; // bytes of zero needed [0,63]
412  ASSERT_require((messageSizeBytes + 1 + nZeroPadding + 8) % 64 == 0); // suffix wil padd message to multiple of 64 bytes
413 
414  // Create and hash the padding
415  uint8_t padding[64 + 9]; // big enough for the 0x80 and 8-byte length, plus zero padding
416  padding[0] = 0x80;
417  memset(padding+1, 0, nZeroPadding);
418  const uint64_t messageSizeBits = 8 * messageSizeBytes;
419  padding[1 + nZeroPadding + 0] = messageSizeBits >> 56;
420  padding[1 + nZeroPadding + 1] = messageSizeBits >> 48;
421  padding[1 + nZeroPadding + 2] = messageSizeBits >> 40;
422  padding[1 + nZeroPadding + 3] = messageSizeBits >> 32;
423  padding[1 + nZeroPadding + 4] = messageSizeBits >> 24;
424  padding[1 + nZeroPadding + 5] = messageSizeBits >> 16;
425  padding[1 + nZeroPadding + 6] = messageSizeBits >> 8;
426  padding[1 + nZeroPadding + 7] = messageSizeBits; // max index is 1 + 63 + 7 = 73
427  append(padding, 1 + nZeroPadding + 8);
428  ASSERT_require(leftoverBytes_.size() == 0);
429 
430  // The digest is the concatenation of the state words
431  for (size_t i=0; i<8; ++i) {
432  digest_.push_back(state_[i] >> 24);
433  digest_.push_back(state_[i] >> 16);
434  digest_.push_back(state_[i] >> 8);
435  digest_.push_back(state_[i]);
436  }
437  }
438  return digest_;
439 }
