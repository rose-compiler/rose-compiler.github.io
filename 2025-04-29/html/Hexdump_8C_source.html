17 hexdump(std::ostream &f, Address base_addr, const unsigned char *data, size_t n, const HexdumpFormat &fmt) {
18  // Provide default formats. This is done here so that the header file doesn't depend on <inttypes.h>
19  std::string addr_fmt = fmt.addr_fmt.empty() ? std::string("0x%|08x|: ") : fmt.addr_fmt;
20  std::string numeric_fmt = fmt.numeric_fmt.empty() ? std::string("%|02x|") : fmt.numeric_fmt;
21  std::string numeric_sep = fmt.numeric_sep.empty() ? std::string(" ") : fmt.numeric_sep;
22  std::string prefix = fmt.prefix;
23 
24  size_t numeric_width = (boost::format(numeric_fmt) % (Address)0).str().size();
25 
26  if (fmt.multiline)
27  f <<prefix;
28 
29  for (size_t i=0; i<n; i+=fmt.width) {
30  // Prefix and/or address
31  if (i>0)
32  f <<"\n" <<prefix;
33  f <<boost::format(addr_fmt) % (base_addr+i);
34 
35  /* Numeric byte values */
36  if (fmt.show_numeric) {
37  for (size_t j=0; j<fmt.width; j++) {
38  if (i+j<n) {
39  if (j>0)
40  f <<numeric_sep;
41  if (j>0 && 0 == j % fmt.colsize)
42  f <<" ";
43  std::map<unsigned char, std::string>::const_iterator special = fmt.numeric_fmt_special.find(data[i+j]);
44  if (special == fmt.numeric_fmt_special.end()) {
45  f <<boost::format(numeric_fmt) % (unsigned)data[i+j];
46  } else {
47  try {
48  f <<boost::format(special->second) % (unsigned)(special->first);
49  } catch (...) {
50  f <<special->second;
51  }
52  }
53  } else if (fmt.pad_numeric) {
54  if (j>0)
55  f <<numeric_sep;
56  if (j>0 && 0 == j % fmt.colsize)
57  f <<" ";
58  f <<std::string(numeric_width, ' ');
59  }
60  }
61  }
62 
63  if (fmt.show_numeric && fmt.show_chars)
64  f <<" |";
65 
66  /* Character byte values */
67  if (fmt.show_chars) {
68  for (size_t j=0; j<fmt.width; j++) {
69  if (i+j>=n) {
70  if (fmt.pad_chars)
71  f <<" ";
72  } else if (isprint(data[i+j])) {
73  f <<data[i+j];
74  } else {
75  f <<".";
76  }
77  }
78  f <<"|";
79  }
80  }
81 
82  if (fmt.multiline)
83  f <<"\n";
84 }
