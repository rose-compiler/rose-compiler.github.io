110 initSchema(Sawyer::Database::Connection db) {
111 
112  // SQLite has a relatively small limit for storing large `bytea` objects. This table holds a larger binary blob as a
113  // broken up sequence of parts numbered consecutively starting at zero.
114  db.run("drop table if exists blob_parts");
115  db.run("create table blob_parts ("
116  " id integer not null,"
117  " part_number integer not null,"
118  " part bytea not null)");
119 
120  db.run("drop table if exists test_suites");
121  db.run("create table test_suites ("
122  " created_ts varchar(32) not null,"
123  " id integer primary key,"
124  " name text not null unique)");
125 
126  db.run("drop table if exists specimens");
127  db.run("create table specimens ("
128  " id integer primary key,"
129  " created_ts varchar(32) not null,"
130  " name text not null,"
131  " content bytea," // maybe null
132  " rba integer," // null, or the id for a blob in the blob_parts table
133  " test_suite integer not null,"
134 
135  " constraint fk_test_suite foreign key (test_suite) references test_suites (id))");
136 
137  db.run("drop table if exists test_cases");
138  db.run("create table test_cases ("
139  " id integer primary key,"
140  " parent integer," // parent test case or null
141  " created_ts varchar(32) not null,"
142  " name text not null,"
143  " executor text not null,"
144  " specimen integer not null,"
145  " argv bytea not null,"
146  " envp bytea not null,"
147  " symbolic_state bytea," // optional serialized symbolic state
148  " assertions string," // optional serialized SMT solver assertions
149  " concrete_rank real," // null if concrete executor not run yet
150  " concrete_result bytea," // null if concrete executor not run yet
151  " concolic_result integer," // non-null if concolic executor has been run
152  " concrete_interesting integer not null default 1," // concrete results are uninteresting if present? (bool)

310 
311  if (auto parent = iter->get<size_t>(9)) {
312  obj->parent(TestCaseId(*parent));
313  } else {
314  obj->parent(TestCaseId());
315  }
316 
317  if (auto assertions = iter->get<std::string>(10)) {
318  std::istringstream ss(*assertions);
319  boost::archive::binary_iarchive archive(ss);
320  registerTypes(archive);

322  archive >>v;

324  } else {
325  obj->assertions().clear();

327

337 static void
338 updateDb(const Database::Ptr &db, TestCaseId id, const TestCase::Ptr &obj) {
339  ASSERT_not_null(db);
340  ASSERT_require(id);
341  ASSERT_not_null(obj);
342  Sawyer::Database::Statement stmt;
343  if (*db->connection().stmt("select count(*) from test_cases where id = ?id").bind("id", *id).get<size_t>()) {
344  stmt = db->connection().stmt("update test_cases set"
345  " name = ?name,"
346  " parent = ?parent,"
347  " executor = ?executor,"
348  " specimen = ?specimen,"
349  " argv = ?argv,"
350  " envp = ?envp,"
351  " assertions = ?assertions,"
352  " concrete_rank = ?concrete_rank,"
353  " concrete_interesting = ?interesting,"
354  " concolic_result = ?concolic_result"
355  " where id = ?id");
356  } else {
357  std::string ts = obj->timestamp().empty() ? Database::timestamp() : obj->timestamp();
358  obj->timestamp(ts);
359  stmt = db->connection().stmt("insert into test_cases ("
360  " id, parent, created_ts, name, executor, specimen, argv, envp,"
361  " concrete_rank, concrete_interesting, concolic_result, test_suite,"
362  " assertions"
363  ") values ("
364  " ?id, ?parent, ?ts, ?name, ?executor, ?specimen, ?argv, ?envp,"
365  " ?concrete_rank, ?interesting, ?concolic_result, ?test_suite,"
366  " ?assertions)")
367  .bind("ts", ts)
368  .bind("test_suite", *db->id(db->testSuite()));
369  }
370 
371  // Join environment variables to form a single string
372  std::string envp;
373  for (auto pair: obj->env()) {
374  if (!envp.empty())
375  envp += '\0';
376  envp += pair.first + "=" + pair.second;
377  }
378 
379  if (obj->parent()) {
380  stmt.bind("parent", *obj->parent());
381  } else {
382  stmt.bind("parent", Sawyer::Nothing());
383  }
384 
385  if (obj->assertions().empty()) {
386  stmt.bind("assertions", Sawyer::Nothing());
387  } else {
388  std::ostringstream ss;
389  {
390  boost::archive::binary_oarchive archive(ss);
391  registerTypes(archive);
392  archive <<obj->assertions();
393  }
394  stmt.bind("assertions", ss.str());
395  }
396 
397  stmt
398  .bind("id", *id)
399  .bind("name", obj->name())
400  .bind("executor", obj->executor())
401  .bind("specimen", *db->id(obj->specimen(), Update::YES))
402  .bind("argv", StringUtility::join('\0', obj->args()))
403  .bind("envp", envp)
404  .bind("concrete_rank", obj->concreteRank())
405  .bind("interesting", obj->concreteIsInteresting() ? 1 : 0)
406  .bind("concolic_result", obj->concolicResult())
407  .run();
408 }

454 template<class IdObjMap>
455 static typename IdObjMap::Target // object pointer
456 objectHelper(const Database::Ptr &db, const typename IdObjMap::Source &id, Update update, IdObjMap &objMap,
457  const std::string &objectTypeName) {
458  if (!id)
459  throw Exception("invalid " + objectTypeName + " ID");
460  auto obj = objMap.forward().getOrDefault(id);
461  if (!obj) {
462  obj = IdObjMap::Target::Pointee::instance();
463  updateObject(db, id, obj);
464  objMap.insert(id, obj);
465  } else if (Update::YES == update) {
466  updateObject(db, id, obj);
467  }
468  ASSERT_not_null(obj);
469  return obj;
470 }

472 // Helper template function for the Database::id method
473 template<class IdObjMap>
474 static typename IdObjMap::Source // ID
475 idHelper(const Database::Ptr &db, const typename IdObjMap::Target &obj, Update &update, IdObjMap &objMap) {
476  ASSERT_not_null(obj);
477  auto id = objMap.reverse().getOrDefault(obj);
478  if (id) {
479  if (Update::YES == update)
480  updateDb(db, id, obj);

482  id = generateId();
483  updateDb(db, id, obj);
484  objMap.insert(id, obj);
485  }
486  return id;

918  if (nBytes < maxPartSize)
