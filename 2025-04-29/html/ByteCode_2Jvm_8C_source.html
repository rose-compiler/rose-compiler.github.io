86 JvmMethod::decode(const Disassembler::Base::Ptr &disassembler) const {
87  Address va{classAddr_ + code_.offset()};
88  Address endVa{va + code_.size()};
89 
90  auto disassemblerJvm{as<Disassembler::Jvm>(disassembler)};
91  disassemblerJvm->codeOffset(va);
92 
93  MemoryMap::Ptr map = MemoryMap::instance();
94  MemoryMap::Buffer::Ptr buf = MemoryMap::StaticBuffer::instance(code_.bytes(), code_.size());
95  map->insert(AddressInterval::baseSize(va, buf->size()),
96  AddressSegment(buf, /*offset*/0, MemoryMap::READABLE, "JVM code segment"));
97 
98  SgAsmInstruction* insn{nullptr};
99  SgAsmInstructionList* insnList{sgMethod_->get_instruction_list()};
100 
101  while (va < endVa) {
102  insn = disassembler->disassembleOne(map, va);
103  if (insn) {
104  va += insn->get_size();
105  ASSERT_require2((666 != (int)insn->get_anyKind()), "unknown instruction");
106  insnList->get_instructions().push_back(insn);
107  insn->set_parent(insnList);
108  }
109  else {
110  ASSERT_require2(false, "disassembly failed for lack of instruction");
111  }
112  }
113  ASSERT_require(va == endVa);
114 }
