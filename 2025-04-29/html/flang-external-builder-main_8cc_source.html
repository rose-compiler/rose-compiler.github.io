9 // F18 parsing demonstration.
10 // f18-parse-demo [ -E | -fdump-parse-tree | -funparse-only ]
11 // foo.{f,F,f77,F77,f90,F90,&c.}
12 //
13 // By default, runs the supplied source files through the F18 preprocessing and
14 // parsing phases, reconstitutes a Fortran program from the parse tree, and
15 // passes that Fortran program to a Fortran compiler identified by the $F18_FC
16 // environment variable (defaulting to gfortran). The Fortran preprocessor is
17 // always run, whatever the case of the source file extension. Unrecognized
18 // options are passed through to the underlying Fortran compiler.
19 //
20 // This program is actually a stripped-down variant of f18.cpp, a temporary
21 // scaffolding compiler driver that can test some semantic passes of the
22 // F18 compiler under development.

107 void Exec(std::vector<llvm::StringRef> &argv, bool verbose = false) {
108  if (verbose) {
109  for (size_t j{0}; j < argv.size(); ++j) {
110  llvm::errs() << (j > 0 ? " " : "") << argv[j];
111  }
112  llvm::errs() << '\n';
113  }
114  std::string ErrMsg;
115  llvm::ErrorOr<std::string> Program = llvm::sys::findProgramByName(argv[0]);
116  if (!Program)
117  ErrMsg = Program.getError().message();
118  if (!Program ||
119  llvm::sys::ExecuteAndWait(
120  Program.get(), argv, std::nullopt, {}, 0, 0, &ErrMsg)) {
121  llvm::errs() << "execvp(" << argv[0] << ") failed: " << ErrMsg << '\n';
122  exit(EXIT_FAILURE);
123  }
124 }

139 std::string RelocatableName(const DriverOptions &driver, std::string path) {
140  if (driver.compileOnly && !driver.outputPath.empty()) {
141  return driver.outputPath;
142  }
143  std::string base{path};
144  auto slash{base.rfind("/")};
145  if (slash != std::string::npos) {
146  base = base.substr(slash + 1);
147  }
148  std::string relo{base};
149  auto dot{base.rfind(".")};
150  if (dot != std::string::npos) {
151  relo = base.substr(0, dot);
152  }
153  relo += ".o";
154  return relo;
155 }

159 std::string CompileFortran(
160  std::string path, Fortran::parser::Options options, DriverOptions &driver) {
161  if (!driver.forcedForm) {
162  auto dot{path.rfind(".")};
163  if (dot != std::string::npos) {
164  std::string suffix{path.substr(dot + 1)};
165  options.isFixedForm = suffix == "f" || suffix == "F" || suffix == "ff";
166  }
167  }
168  options.searchDirectories = driver.searchDirectories;
169  Fortran::parser::AllSources allSources;
170  Fortran::parser::AllCookedSources allCookedSources{allSources};
171  Fortran::parser::Parsing parsing{allCookedSources};
172 
173  auto start{CPUseconds()};
174  parsing.Prescan(path, options);
175  if (!parsing.messages().empty() &&
176  (driver.warningsAreErrors || parsing.messages().AnyFatalError())) {
177  llvm::errs() << driver.prefix << "could not scan " << path << '\n';
178  parsing.messages().Emit(llvm::errs(), parsing.allCooked());
179  exitStatus = EXIT_FAILURE;
180  return {};
181  }
182  if (driver.dumpProvenance) {
183  parsing.DumpProvenance(llvm::outs());
184  return {};
185  }
186  if (options.prescanAndReformat) {
187  parsing.messages().Emit(llvm::errs(), allCookedSources);
188  if (driver.noReformat) {
189  parsing.DumpCookedChars(llvm::outs());
190  } else {
191  parsing.EmitPreprocessedSource(llvm::outs(), driver.lineDirectives);
192  }
193  return {};
194  }
195  parsing.Parse(llvm::outs());
196  auto stop{CPUseconds()};
197  if (driver.timeParse) {
198  if (canTime) {
199  llvm::outs() << "parse time for " << path << ": " << (stop - start)
200  << " CPU milliseconds\n";
201  } else {
202  llvm::outs() << "no timing information due to lack of clock_gettime()\n";
203  }
204  }
205 
206  parsing.ClearLog();
207  parsing.messages().Emit(llvm::errs(), parsing.allCooked());
208  if (!parsing.consumedWholeFile()) {
209  parsing.EmitMessage(llvm::errs(), parsing.finalRestingPlace(),
210  "parser FAIL (final position)", "error: ", llvm::raw_ostream::RED);
211  exitStatus = EXIT_FAILURE;
212  return {};
213  }
214  if ((!parsing.messages().empty() &&
215  (driver.warningsAreErrors || parsing.messages().AnyFatalError())) ||
216  !parsing.parseTree()) {
217  llvm::errs() << driver.prefix << "could not parse " << path << '\n';
218  exitStatus = EXIT_FAILURE;
219  return {};
220  }
221  auto &parseTree{*parsing.parseTree()};
222 
223  // Test building from a move constructed copy of the parse tree
224  auto cstart {CPUseconds()};
225  Fortran::parser::Program parseTreeCopy{std::move(parseTree)};
226  auto cstop {CPUseconds()};
227  if (canTime) {
228  llvm::outs() << "parse-tree copy time: " << (cstop - cstart) << " CPU milliseconds\n";
229  }
230 
231  // Transform the parse tree using Rose::builder
232  if (driver.externalBuilder) {
233  auto start{CPUseconds()};
234  Rose::builder::Build(parseTreeCopy, allCookedSources);
235  auto stop{CPUseconds()};
236  if (canTime) {
237  llvm::outs() << "Rose::Build time for " << path << ": " << (stop - start)
238  << " CPU milliseconds\n";
239  }
240 
241  // Always dump the parse tree for now
242  Fortran::parser::DumpTree(llvm::outs(), parseTreeCopy);
243  return {};
244  }
245 
246  if (driver.dumpParseTree) {
247  Fortran::parser::DumpTree(llvm::outs(), parseTree);
248  return {};
249  }
250  if (driver.dumpUnparse) {
251  Unparse(llvm::outs(), parseTree, driver.encoding, true /*capitalize*/,
252  options.features.IsEnabled(
253  Fortran::common::LanguageFeature::BackslashEscapes));
254  return {};
255  }
256  if (driver.syntaxOnly) {
257  return {};
258  }
259 
260  std::string relo{RelocatableName(driver, path)};
261 
262  llvm::SmallString<32> tmpSourcePath;
263  {
264  int fd;
265  std::error_code EC =
266  llvm::sys::fs::createUniqueFile("f18-%%%%.f90", fd, tmpSourcePath);
267  if (EC) {
268  llvm::errs() << EC.message() << "\n";
269  std::exit(EXIT_FAILURE);
270  }
271  llvm::raw_fd_ostream tmpSource(fd, /*shouldClose*/ true);
272  Unparse(tmpSource, parseTree, driver.encoding, true /*capitalize*/,
273  options.features.IsEnabled(
274  Fortran::common::LanguageFeature::BackslashEscapes));
275  }
276 
277  RunOtherCompiler(driver, tmpSourcePath.data(), relo.data());
278  filesToDelete.emplace_back(tmpSourcePath);
279  if (!driver.compileOnly && driver.outputPath.empty()) {
280  filesToDelete.push_back(relo);
281  }
282  return relo;
283 }

285 std::string CompileOtherLanguage(std::string path, DriverOptions &driver) {
286  std::string relo{RelocatableName(driver, path)};
287  RunOtherCompiler(driver, path.data(), relo.data());

294 void Link(std::vector<std::string> &relocatables, DriverOptions &driver) {
295  std::vector<llvm::StringRef> argv;
296  for (size_t j{0}; j < driver.fcArgs.size(); ++j) {
297  argv.push_back(driver.fcArgs[j].data());
298  }
299  for (auto &relo : relocatables) {
300  argv.push_back(relo.data());
301  }
302  if (!driver.outputPath.empty()) {
303  char dashO[3] = "-o";
304  argv.push_back(dashO);
305  argv.push_back(driver.outputPath.data());
306  }
307  Exec(argv, driver.verbose);
308 }

311 int flang_external_builder_main(int argc, char *const argv[], SgSourceFile* roseSourceFile)
312 {
313  atexit(CleanUpAtExit);
314 
315  // This construction with roseSourceFile is a hack needed until Rose compiles with C++17
316  // Meanwhile, the following must be set before Rose::builder::Build functions called.
317  Rose::builder::setSgSourceFile(roseSourceFile);
318 
319  DriverOptions driver;
320  const char *fc{getenv("F18_FC")};
321  driver.fcArgs.push_back(fc ? fc : "gfortran");
322 
323  std::list<std::string> args{argList(argc, argv)};
324  std::string prefix{args.front()};
325  args.pop_front();
326  prefix += ": ";
327  driver.prefix = prefix.data();
328 
329  Fortran::parser::Options options;
330  options.predefinitions.emplace_back("__F18", "1");
331  options.predefinitions.emplace_back("__F18_MAJOR__", "1");
332  options.predefinitions.emplace_back("__F18_MINOR__", "1");
333  options.predefinitions.emplace_back("__F18_PATCHLEVEL__", "1");
334 
335  options.features.Enable(
336  Fortran::common::LanguageFeature::BackslashEscapes, true);
337 
338  // enable parsing of CUDA Fortran
339  bool enableCUDA{true};
340  if (enableCUDA) {
341  options.features.Enable(Fortran::common::LanguageFeature::CUDA);
342  }
343 
344  Fortran::common::IntrinsicTypeDefaultKinds defaultKinds;
345 
346  std::vector<std::string> fortranSources, otherSources, relocatables;
347  bool anyFiles{false};
348 
349  while (!args.empty()) {
350  std::string arg{std::move(args.front())};
351  args.pop_front();
352  if (arg.empty() || arg == "-Xflang") {
353  } else if (arg.at(0) != '-') {
354  anyFiles = true;
355  auto dot{arg.rfind(".")};
356  if (dot == std::string::npos) {
357  driver.fcArgs.push_back(arg);
358  } else {
359  std::string suffix{arg.substr(dot + 1)};
360  if (suffix == "f" || suffix == "F" || suffix == "ff" ||
361  suffix == "f90" || suffix == "F90" || suffix == "ff90" ||
362  suffix == "f95" || suffix == "F95" || suffix == "ff95" ||
363  suffix == "cuf" || suffix == "CUF" || suffix == "f18" ||
364  suffix == "F18" || suffix == "ff18") {
365  fortranSources.push_back(arg);
366  } else if (suffix == "o" || suffix == "a") {
367  relocatables.push_back(arg);
368  } else {
369  otherSources.push_back(arg);
370  }
371  }
372  } else if (arg == "-") {
373  fortranSources.push_back("-");
374  } else if (arg == "--") {
375  while (!args.empty()) {
376  fortranSources.emplace_back(std::move(args.front()));
377  args.pop_front();
378  }
379  break;
380  } else if (arg == "-Mfixed") {
381  driver.forcedForm = true;
382  options.isFixedForm = true;
383  } else if (arg == "-Mfree") {
384  driver.forcedForm = true;
385  options.isFixedForm = false;
386  } else if (arg == "-Mextend") {
387  options.fixedFormColumns = 132;
388  } else if (arg == "-Mbackslash") {
389  options.features.Enable(
390  Fortran::common::LanguageFeature::BackslashEscapes, false);
391  } else if (arg == "-Mnobackslash") {
392  options.features.Enable(
393  Fortran::common::LanguageFeature::BackslashEscapes);
394  } else if (arg == "-Mstandard") {
395  driver.warnOnNonstandardUsage = true;
396  } else if (arg == "-fopenmp") {
397  options.features.Enable(Fortran::common::LanguageFeature::OpenMP);
398  options.predefinitions.emplace_back("_OPENMP", "201511");
399  } else if (arg == "-Werror") {
400  driver.warningsAreErrors = true;
401  } else if (arg == "-ed") {
402  options.features.Enable(Fortran::common::LanguageFeature::OldDebugLines);
403  } else if (arg == "-E") {
404  options.prescanAndReformat = true;
405  } else if (arg == "-P") {
406  driver.lineDirectives = false;
407  } else if (arg == "-fno-reformat") {
408  driver.noReformat = true;
409  } else if (arg == "-fbackslash") {
410  options.features.Enable(
411  Fortran::common::LanguageFeature::BackslashEscapes);
412  } else if (arg == "-fno-backslash") {
413  options.features.Enable(
414  Fortran::common::LanguageFeature::BackslashEscapes, false);
415  } else if (arg == "-fdump-provenance") {
416  driver.dumpProvenance = true;
417  } else if (arg == "-fdump-parse-tree") {
418  driver.dumpParseTree = true;
419  } else if (arg == "-funparse") {
420  driver.dumpUnparse = true;
421  } else if (arg == "-ftime-parse") {
422  driver.timeParse = true;
423  } else if (arg == "-fparse-only" || arg == "-fsyntax-only") {
424  driver.syntaxOnly = true;
425  } else if (arg == "-fexternal-builder") {
426  driver.externalBuilder = true;
427  } else if (arg == "-c") {
428  driver.compileOnly = true;
429  } else if (arg == "-o") {
430  driver.outputPath = args.front();
431  args.pop_front();
432  } else if (arg.substr(0, 2) == "-D") {
433  auto eq{arg.find('=')};
434  if (eq == std::string::npos) {
435  options.predefinitions.emplace_back(arg.substr(2), "1");
436  } else {
437  options.predefinitions.emplace_back(
438  arg.substr(2, eq - 2), arg.substr(eq + 1));
439  }
440  } else if (arg.substr(0, 2) == "-U") {
441  options.predefinitions.emplace_back(
442  arg.substr(2), std::optional<std::string>{});
443  } else if (arg == "-r8" || arg == "-fdefault-real-8") {
444  defaultKinds.set_defaultRealKind(8);
445  } else if (arg == "-i8" || arg == "-fdefault-integer-8") {
446  defaultKinds.set_defaultIntegerKind(8);
447  } else if (arg == "-help" || arg == "--help" || arg == "-?") {
448  llvm::errs()
449  << "f18-parse-demo options:\n"
450  << " -Mfixed | -Mfree force the source form\n"
451  << " -Mextend 132-column fixed form\n"
452  << " -f[no-]backslash enable[disable] \\escapes in literals\n"
453  << " -M[no]backslash disable[enable] \\escapes in literals\n"
454  << " -Mstandard enable conformance warnings\n"
455  << " -r8 | -fdefault-real-8 | -i8 | -fdefault-integer-8 "
456  "change default kinds of intrinsic types\n"
457  << " -Werror treat warnings as errors\n"
458  << " -ed enable fixed form D lines\n"
459  << " -E prescan & preprocess only\n"
460  << " -ftime-parse measure parsing time\n"
461  << " -fsyntax-only parse only, no output except messages\n"
462  << " -funparse parse & reformat only, no code "
463  "generation\n"
464  << " -fdump-provenance dump the provenance table (no code)\n"
465  << " -fdump-parse-tree dump the parse tree (no code)\n"
466  << " -v -c -o -I -D -U have their usual meanings\n"
467  << " -help print this again\n"
468  << "Other options are passed through to the $F18_FC compiler.\n";
469  return exitStatus;
470  } else if (arg == "-V") {
471  llvm::errs() << "\nf18-parse-demo\n";
472  return exitStatus;
473  } else {
474  driver.fcArgs.push_back(arg);
475  if (arg == "-v") {
476  driver.verbose = true;
477  } else if (arg == "-I") {
478  driver.fcArgs.push_back(args.front());
479  driver.searchDirectories.push_back(args.front());
480  args.pop_front();
481  } else if (arg.substr(0, 2) == "-I") {
482  driver.searchDirectories.push_back(arg.substr(2));
483  }
484  }
485  }
486 
487  if (driver.warnOnNonstandardUsage) {
488  options.features.WarnOnAllNonstandard();
489  }
490  if (!options.features.IsEnabled(
491  Fortran::common::LanguageFeature::BackslashEscapes)) {
492  driver.fcArgs.push_back("-fno-backslash"); // PGI "-Mbackslash"
493  }
494 
495  if (!anyFiles) {
496  driver.dumpUnparse = true;
497  CompileFortran("-", options, driver);
498  return exitStatus;
499  }
500  for (const auto &path : fortranSources) {
501  std::string relo{CompileFortran(path, options, driver)};
502  if (!driver.compileOnly && !relo.empty()) {
503  relocatables.push_back(relo);
504  }
505  }
506  for (const auto &path : otherSources) {
507  std::string relo{CompileOtherLanguage(path, driver)};
508  if (!driver.compileOnly && !relo.empty()) {
509  relocatables.push_back(relo);
510  }
511  }
512  if (!relocatables.empty()) {
513  Link(relocatables, driver);
514  }
515 
516  return exitStatus;
517 }
