96 bool
97 CppStack::process(const File::Ptr &file, const Token &token, Stack &stack) {
98  ASSERT_not_null(file);
99  if (token.type() != Sawyer::Language::Clexer::TOK_CPP)
100  return false;
101 
102  const std::string lexeme = boost::trim_copy(file->lexeme(token));
103  switch (type(lexeme)) {
104  case Type::IF:
105  stack.push_back(Level{Directive{Type::IF, token, lexeme}});
106  break;
107  case Type::END:
108  if (stack.empty() || stack.back().front().type != Type::IF) {
109  stack.push_back(Level{Directive{Type::END, token, lexeme}});
110  } else {
111  stack.pop_back();
112  }
113  break;
114  case Type::ELSE:
115  if (stack.empty() || stack.back().front().type == Type::END) {
116  stack.push_back(Level{Directive{Type::ELSE, token, lexeme}});
117  } else {
118  stack.back().push_back(Directive{Type::ELSE, token, lexeme});
119  }
120  break;
121  case Type::OTHER:
122  return false;
123  }
124  return true;
125 }

244 // Read the file the hard way with extra data copying because boost::iostreams::mapped_file holds the file descriptor open even
245 // though mmap doesn't require that. On macOS, this results in the process eventually having too many open file descriptors since
246 // there's one per parsed Rosebud input file.
247 static Sawyer::Container::Buffer<size_t, char>::Ptr
248 readFile(const boost::filesystem::path &name) {
249  std::ifstream input(name.c_str());
250  if (!input)
251  throw std::runtime_error("cannot open \"" + name.string() + "\" for reading");
252  size_t nChars = boost::filesystem::file_size(name);
253  auto retval = Sawyer::Container::AllocatingBuffer<size_t, char>::instance(nChars);
254  std::vector<char> buf(4096);
255  for (size_t at = 0; at < nChars; /*void*/) {
256  input.read(buf.data(), buf.size());
257  const size_t nRead = input.gcount();
258  if (nRead > 0) {
259  const size_t nWritten = retval->write(buf.data(), at, nRead);
260  ASSERT_always_require(nWritten == nRead);
261  at += nRead;
262  } else {
263  ASSERT_always_require(at == retval->size());
264  break;
265  }
266  }
267  return retval;
268 }

408 File::trimmedContent(size_t begin, size_t end, const Token &exclude, Token &outputToken) {
409  using Interval = Sawyer::Container::Interval<size_t>;
410  using IntervalSet = Sawyer::Container::IntervalSet<Interval>;
411  IntervalSet where;
412 
413  if (begin < end)
414  where |= Interval::hull(begin, end-1);
415  if (exclude.begin() < exclude.end())
416  where -= Interval::hull(exclude.begin(), exclude.end() - 1);
417 
418  // FIXME[Robb Matzke 2023-04-03]: it would be nice if this token were a little more accurate
419  if (where.isEmpty()) {
420  outputToken = Token();
421  } else {
422  outputToken = Token(Sawyer::Language::Clexer::TOK_OTHER,
423  where.hull().least(), where.hull().least(), where.hull().greatest());
424  }
425 
426  std::string s;
427  for (const Interval &interval: where.intervals())
428  s += stream_.content().contentAsString(interval.least(), interval.greatest() + 1);

431  trimBlankLines(lines);
