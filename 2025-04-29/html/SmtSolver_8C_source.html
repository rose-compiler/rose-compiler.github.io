85 SmtSolver::Memoizer::Map::iterator
86 SmtSolver::Memoizer::searchNS(SymbolicExpression::Hash h, const ExprList &sortedNormalized) {
87  std::pair<Map::iterator, Map::iterator> range = map_.equal_range(h);
88  for (Map::iterator iter = range.first; iter != range.second; ++iter) {
89  const Record &record = iter->second;
90  bool areEqual = true;
91  if (sortedNormalized.size() != record.assertions.size()) {
92  areEqual = false;
93  } else {
94  for (size_t i = 0; i < sortedNormalized.size(); ++i) {
95  if (!sortedNormalized[i]->isEquivalentTo(record.assertions[i])) {
96  areEqual = false;
97  break;
98  }
99  }
100  }
101 
102  if (areEqual)
103  return iter;
104  }
105  return map_.end();
106 }

108 SmtSolver::Memoizer::Found
109 SmtSolver::Memoizer::find(const ExprList &assertions) {
110  ASSERT_forbid(assertions.empty()); // trivial solutions should be handled before this point
111  Found retval;
112 
113 #if 1 // full matching
114  // First rewrite each assertion individually to normalize the variables. This will allow us to sort them consistently.
115  std::vector<std::pair<SymbolicExpression::Ptr /*original*/, SymbolicExpression::Ptr /*rewritten*/>> sorted;
116  sorted.reserve(assertions.size());
117  for (const SymbolicExpression::Ptr &assertion: assertions) {
118  SymbolicExpression::ExprExprHashMap index;
119  size_t nextVariableId = 0;
120  sorted.push_back(std::make_pair(assertion, assertion->renameVariables(index, nextVariableId)));
121  }
122  std::sort(sorted.begin(), sorted.end(), [](const ExprExpr &a, const ExprExpr &b) {
123  return a.second->hash() < b.second->hash();
124  });
125 
126  // Now rewrite the variables across all the sorted assertions at once.
127  retval.sortedNormalized.reserve(assertions.size());
128  size_t nextVariableId = 0;
129  for (const ExprExpr &p: sorted)
130  retval.sortedNormalized.push_back(p.first->renameVariables(retval.rewriteMap, nextVariableId));
131  const SymbolicExpression::Hash h = SymbolicExpression::hash(retval.sortedNormalized);
132 #else // partial matching (unsorted but normalized)
133  retval.sortedNormalized.reserve(assertions.size());
134  size_t nextVariableId = 0;
135  for (const SymbolicExpression::Ptr &p: assertions)
136  retval.sortedNormalized.push_back(p->renameVariables(retval.rewriteMap, nextVariableId));
137  const SymbolicExpression::Hash h = SymbolicExpression::hash(retval.sortedNormalized);
138 #endif
139 
140  // Does our map already contain this set of assertions? Hashes can be equal without the set of assertions being equal, so
141  // we need to check.
142  {
143  SAWYER_THREAD_TRAITS::LockGuard lock(mutex_);
144  Map::iterator iter = searchNS(h, retval.sortedNormalized);
145  if (iter != map_.end()) {
146  retval.satisfiable = iter->second.satisfiable;
147  retval.evidence = iter->second.evidence;
148  }
149  }
150 
151  return retval;
152 }

167 void
168 SmtSolver::Memoizer::insert(const Found &found, Satisfiable sat, const ExprExprMap &evidence) {
169  ASSERT_forbid(found); // must have been a cache miss since we're specifying results
170  ASSERT_require(evidence.isEmpty() || SAT_YES == sat);
171 
172  ExprExprMap normalizedEvidence;
173  for (const ExprExprMap::Node &node: evidence.nodes()) {
174  const SymbolicExpression::Ptr var = node.key()->substituteMultiple(found.rewriteMap);
175  const SymbolicExpression::Ptr val = node.value()->substituteMultiple(found.rewriteMap);
176  normalizedEvidence.insert(var, val);
177  }
178 
179  const SymbolicExpression::Hash h = SymbolicExpression::hash(found.sortedNormalized);
180  {
181  // Some other thread may have beaten us here with the same set of assertions. In that case, we should not insert
182  // anything since it would result in duplicate records.
183  SAWYER_THREAD_TRAITS::LockGuard lock(mutex_);
184  if (searchNS(h, found.sortedNormalized) == map_.end())
185  map_.insert(std::make_pair(h, Record{found.sortedNormalized, sat, normalizedEvidence}));
186  }
187 }

534 SmtSolver::Satisfiable
535 SmtSolver::checkTrivial() {
536  // Empty set of assertions is YES
537  std::vector<SymbolicExpression::Ptr> exprs = assertions();
538  if (exprs.empty())
539  return SAT_YES;
540 
541  // If any assertion is a constant zero, then NO
542  // If all assertions are non-zero integer constants, then YES
543  bool allTrue = true;
544  for (const SymbolicExpression::Ptr &expr: exprs) {
545  if (expr->isIntegerConstant()) {
546  ASSERT_require(1 == expr->nBits());
547  if (expr->toUnsigned().get() == 0)
548  return SAT_NO;
549  } else {
550  allTrue = false;
551  }
552  }
553  if (allTrue)
554  return SAT_YES;
555 
556  return SAT_UNKNOWN;
557 }

580 SmtSolver::Satisfiable
581 SmtSolver::check() {
582  ++stats.ncalls;
583 
584  if (mlog[DEBUG]) {
585  mlog[DEBUG] <<"assertions:\n";
586  for (const SymbolicExpression::Ptr &expr: assertions())
587  mlog[DEBUG] <<" " <<*expr <<"\n";
588  }
589 
590  clearEvidence();
591  bool wasTrivial = false;
592  Satisfiable retval = checkTrivial();
593  if (retval != SAT_UNKNOWN) {
594  mlog[DEBUG] <<"trivial solution\n";
595  wasTrivial = true;
596  }
597 
598  // Have we seen this before?
599  Memoizer::Found memoized;
600  if (!wasTrivial && memoizer_) {
601  ProgressTask task(progress_, "smt-memoize");
602  if ((memoized = memoizer_->find(assertions()))) {
603  retval = *memoized.satisfiable;
604  ++stats.memoizationHits;
605  mlog[DEBUG] <<"using memoized result\n";
606  }
607  }
608 
609  // Do the real work
610  if (!wasTrivial && !memoized) {
611  try {
612  switch (linkage_) {
613  case LM_EXECUTABLE:
614  retval = checkExe();
615  break;
616  case LM_LIBRARY:
617  retval = checkLib();
618  break;
619  case LM_NONE:
620  throw Exception("no linkage for " + name_ + " solver");
621  default:
622  ASSERT_not_reachable("invalid solver linkage: " + boost::lexical_cast<std::string>(linkage_));
623  }
624  } catch (const Exception &e) {
625  static std::set<uint64_t> seen;
626  Rose::Combinatorics::HasherFnv hasher;
627  hasher.insert(e.what());
628  if (seen.insert(hasher.partial()).second)
629  mlog[ERROR] <<"SMT solver error: " <<e.what() <<"\n";
630  retval = SAT_UNKNOWN;
631  }
632  }
633 
634  switch (retval) {
635  case SAT_NO:
636  SAWYER_MESG(mlog[DEBUG]) <<" unsat\n";
637  ++stats.nUnsatisfied;
638  break;
639  case SAT_YES:
640  SAWYER_MESG(mlog[DEBUG]) <<" sat\n";
641  ++stats.nSatisfied;
642  break;
643  case SAT_UNKNOWN:
644  SAWYER_MESG(mlog[DEBUG]) <<" unknown\n";
645  ++stats.nUnknown;
646  break;
647  }
648 
649  // Get memoized evidence, or parse and maybe cache the evidence.
650  ProgressTask task(progress_, "smt-evidence");
651  if (memoized) {
652  ASSERT_not_null(memoizer_);
653  evidence_ = memoizer_->evidence(memoized);
654  if (mlog[DEBUG]) {
655  for (const ExprExprMap::Node &node: evidence_.nodes())
656  mlog[DEBUG] <<"evidence: " <<*node.key() <<" == " <<*node.value() <<"\n";
657  }
658 
659  } else if (!wasTrivial) {
660  if (SAT_YES == retval)

852 // The input stream of characters reinterpretted as a stream of tokens. No look-ahead is necessary during parser, which makes
853 // this class even simpler than usual. And since S-Exprs are so simple, we don't even bother with a formal token type, instead
854 // returning std::string with the following values: "" means EOF; "(" and ")"; or anything else is a symbol.
855 class TokenStream {
856 private:
857  Sawyer::Container::LineVector &input_;
858  std::string current_;
859  size_t nextTokenOffset_;
860 
861 public:
862  explicit TokenStream(Sawyer::Container::LineVector &input)
863  : input_(input), nextTokenOffset_(0) {}
864 
865  std::string current() {
866  if (current_.empty())
867  boost::tie(current_, nextTokenOffset_) = scanTokenAt(nextTokenOffset_);
868  return current_;
869  }
870 
871  void shift() {
872  current_ = "";
873  }
874 
875  std::pair<size_t /*line*/, size_t /*col*/> location() {
876  return input_.location(nextTokenOffset_);
877  }
878 
879 private:
880  // Get token starting at specified position, after skipping white space, comments, etc. Returns token lexeme and offset to
881  // first character after end of the token.
882  std::pair<std::string, size_t> scanTokenAt(size_t offset) {
883  while (true) {
884  // Skip white space
885  int ch = input_.character(offset);
886  while (isspace(ch))
887  ch = input_.character(++offset);
888  if (EOF == ch)
889  return std::make_pair(std::string(), offset);
890 
891  if ('(' == ch) {
892  return std::make_pair(std::string("("), offset + 1);
893  } else if (')' == ch) {
894  return std::make_pair(std::string(")"), offset + 1);
895  } else if (';' == ch) {
896  // comment extending to end of line
897  while (EOF != (ch = input_.character(++offset)) && '\n' != ch) /*void*/;
898  ++offset;
899  } else if ('"' == ch) {
900  // string literal
901  size_t end = offset + 1;
902  while (EOF != (ch = input_.character(end)) && '"' != ch)
903  ++end; // FIXME[Robb Matzke 2017-10-31]: what escaping is legal?
904  ASSERT_require('"'==ch); // closing quote
905  std::string s = std::string(input_.characters(offset)+1, (end-offset)-1); // w/out enclosing quotes
906  ++end; // advance past closing quote
907  return std::make_pair(s, end);
908  } else {
909  // everthing else is assumed to be a symbol continuing until the next whitespace, paren, or comment
910  size_t end = offset + 1;
911  while (EOF != (ch = input_.character(end)) && !isspace(ch) && !strchr("();", ch))
912  ++end;
913  const char *lexemeChars = input_.characters(offset);
914  ASSERT_require(lexemeChars != NULL);
915  return std::make_pair(std::string(lexemeChars, end - offset), end);
916  }
917  }
918  }
919 };
920 
921 static SmtSolver::SExpr::Ptr
922 parseSExpression(TokenStream &tokens) {
923  ASSERT_require(tokens.current() != "");
924  if (tokens.current() == "(") {
925  tokens.shift();
926  SmtSolver::SExpr::Ptr sexpr = SmtSolver::SExpr::instance();
927  while (tokens.current() != ")") {
928  if (tokens.current() == "")
929  throw SmtSolver::ParseError(tokens.location(), "EOF before end of S-Expr");
930  sexpr->children().push_back(parseSExpression(tokens));
931  }
932  ASSERT_require(tokens.current() == ")");
933  tokens.shift();
934  return sexpr;
935  } else if (tokens.current() == ")") {
936  throw SmtSolver::ParseError(tokens.location(), "an S-Expr cannot begin with ')'");
937  } else {
938  SmtSolver::SExpr::Ptr sexpr = SmtSolver::SExpr::instance(tokens.current());
939  tokens.shift();
940  return sexpr;
941  }
942 }

944 std::vector<SmtSolver::SExpr::Ptr>
945 SmtSolver::parseSExpressions(const std::string &s) {
946  std::vector<SExpr::Ptr> retval;

948  TokenStream tokens(input);
949 
950  while (tokens.current() != "")
951  retval.push_back(parseSExpression(tokens));
