63 std::string
64 SgAsmFunction::reasonString(bool do_pad, unsigned r)
65 {
66  using namespace StringUtility; // for add_to_reason_string()
67  std::string result;
68 
69  /* entry point and instruction heads are mutually exclusive, so we use the same column for both when padding. */
70  if (r & FUNC_ENTRY_POINT) {
71  add_to_reason_string(result, true, do_pad, "E", "entry point");
72  } else {
73  add_to_reason_string(result, (r & FUNC_INSNHEAD), do_pad, "H", "insn head");
74  }
75 
76  /* Function call:
77  * "C" means the function was detected because we saw a call-like instructon (such as x86 CALL or FARCALL) or instruction
78  * sequence (such as pushing the return value and then branching) in code that was determined to be reachable by
79  * analyzing the control flow graph.
80  *
81  * "c" means this function is the target of some call-like instruction (such as x86 CALL or FARCALL) but could not
82  * determine whether the instruction is actually executed.
83  */
84  if (r & FUNC_CALL_TARGET) {
85  add_to_reason_string(result, true, do_pad, "C", "function call");
86  } else {
87  add_to_reason_string(result, (r & FUNC_CALL_INSN), do_pad, "c", "call instruction");
88  }
89 
90  if (r & FUNC_EH_FRAME) {
91  add_to_reason_string(result, true, do_pad, "X", "exception frame");
92  } else {
93  add_to_reason_string(result, (r & FUNC_THUNK), do_pad, "T", "thunk");
94  }
95  add_to_reason_string(result, (r & FUNC_IMPORT), do_pad, "I", "import");
96  add_to_reason_string(result, (r & FUNC_EXPORT), do_pad, "E", "export");
97  add_to_reason_string(result, (r & FUNC_SYMBOL), do_pad, "S", "symbol");
98  add_to_reason_string(result, (r & FUNC_PATTERN), do_pad, "P", "pattern");
99  add_to_reason_string(result, (r & FUNC_GRAPH), do_pad, "G", "graph");
100  add_to_reason_string(result, (r & FUNC_USERDEF), do_pad, "U", "user defined");
101  add_to_reason_string(result, (r & FUNC_PADDING), do_pad, "N", "padding");
102  add_to_reason_string(result, (r & FUNC_DISCONT), do_pad, "D", "discontiguous");
103  add_to_reason_string(result, (r & FUNC_LEFTOVERS), do_pad, "L", "leftovers");
104  add_to_reason_string(result, (r & FUNC_INTRABLOCK), do_pad, "V", "intrablock");
105  add_to_reason_string(result, (r & FUNC_THUNK_TARGET), do_pad, "t", "thunk target");
106 
107  /* The miscellaneous marker is special. It's a single letter like the others, but is followed by a fixed width
108  * integer indicating the (user-defined) algorithm that added the function. */
109  {
110  char abbr[32], full[64];
111  int width = snprintf(abbr, sizeof abbr, "%u", FUNC_MISCMASK);
112  snprintf(abbr, sizeof abbr, "M%0*u", width, (r & FUNC_MISCMASK));
113  abbr[sizeof(abbr)-1] = '\0';
114  if (!do_pad) {
115  std::string miscname = stringifySgAsmFunctionFunctionReason((r & FUNC_MISCMASK), "FUNC_");
116  if (miscname.empty() || miscname[0]=='(') {
117  snprintf(full, sizeof full, "misc-%u", (r & FUNC_MISCMASK));
118  } else {
119  for (size_t i=0; i<miscname.size(); ++i)
120  miscname[i] = tolower(miscname[i]);
121  strncpy(full, miscname.c_str(), sizeof full);
122  }
123  full[sizeof(full)-1] = '\0';
124  } else {
125  full[0] = '\0';
126  }
127  add_to_reason_string(result, (r & FUNC_MISCMASK), do_pad, abbr, full);
128  }
129 
130  return result;
131 }

135  if (lo_addr)
136  *lo_addr = 0;
137  if (hi_addr)
138  *hi_addr = 0;
139  size_t nnodes = 0;
140 
141  AST::Traversal::forwardPre<SgNode>(this, [extents, lo_addr, hi_addr, selector, &nnodes](SgNode *node) {
142  if (selector && !(*selector)(node))
143  return;
144  SgAsmInstruction *insn = isSgAsmInstruction(node);
145  SgAsmStaticData *data = isSgAsmStaticData(node);
146  Address lo, hi;
147  if (insn) {
148  lo = insn->get_address();
149  hi = lo + insn->get_size();
150  } else if (data) {
151  lo = data->get_address();
152  hi = lo + data->get_size();
153  } else {
154  return;
155  }
156 
157  if (0==nnodes++) {
158  if (lo_addr)
159  *lo_addr = lo;
160  if (hi_addr)
161  *hi_addr = hi;
162  } else {
163  if (lo_addr)
164  *lo_addr = std::min(*lo_addr, lo);
165  if (hi_addr)
166  *hi_addr = std::max(*hi_addr, hi);
167  }
168  if (extents && hi>lo)
169  extents->insert(AddressInterval::baseSize(lo, hi-lo));
170  });
171 
172  return nnodes;
173 }

176 SgAsmFunction::get_sha1(uint8_t digest[20], NodeSelector *selector)
177 {
178 #ifdef ROSE_HAVE_LIBGCRYPT
179  gcry_md_hd_t md; // message digest
180  gcry_error_t error = gcry_md_open(&md, GCRY_MD_SHA1, 0);
181  ASSERT_always_require(GPG_ERR_NO_ERROR == error);
182  BOOST_SCOPE_EXIT(md) {
183  gcry_md_close(md);
184  } BOOST_SCOPE_EXIT_END;
185 
186  AST::Traversal::forwardPre<SgNode>(this, [selector, &md](SgNode *node) {
187  if (selector && !(*selector)(node)) {
188  return;
189  } else if (SgAsmInstruction *insn = isSgAsmInstruction(node)) {
190  SgUnsignedCharList buf = insn->get_rawBytes();
191  gcry_md_write(md, &buf[0], buf.size());
192  } else if (SgAsmStaticData *data = isSgAsmStaticData(node)) {
193  SgUnsignedCharList buf = data->get_rawBytes();
194  gcry_md_write(md, &buf[0], buf.size());
195  }
196  });
197 
198  ASSERT_require(gcry_md_get_algo_dlen(GCRY_MD_SHA1) == 20);
199  gcry_md_final(md);
200  unsigned char *d = gcry_md_read(md, GCRY_MD_SHA1);
201  ASSERT_not_null(d);
202  memcpy(digest, d, 20);
203  return true;
204 #else
205  ROSE_UNUSED(selector);
206  memset(digest, 0, 20);
207  return false;
208 #endif
209 }
