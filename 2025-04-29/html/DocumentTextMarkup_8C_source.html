25 // This text-based markup is similar to Perl POD but be careful: it's not the same:
26 //
27 // Blank lines around "=foo" directives are significant, although the linefeed at the end of "=foo" lines is eaten.
28 //
29 // Consecutive occurrences of =back/=over are removed since they would have no effect. Any white space between these
30 // two lines is removed (as well as the linefeeds at the ends of these two lines).
31 //
32 // The =item directive is an implied =over and must therefore end with a =back
33 //
34 // The =item directive is not valid for named lists. Instead, named list are created by emitting the name, a line break, and
35 // an extra level of =over/=back for the body.
36 //
37 // The =line directive means force a line break, otherwise isolated linefeeds are treated as spaces. Two or more consecutive
38 // linefeeds will generate a blank line in the output.
39 //
40 // There are no inline markups like I<text> and B<text> since we can't do Italics or bold faces. Perhaps a future version can
41 // use ANSI escapes to accomplish this.

176 SAWYER_EXPORT std::string
177 TextMarkup::finalizeDocument(const std::string &s_) {
178  std::string s = boost::trim_copy(s_);
179 
180  // Remove pairs of "=back =over" and spaces at the ends of lines
181  {
182  boost::regex backOverRe("(^=back\\s*=over[ \t]*)|[ \\t\\r\\f]+$");
183  std::ostringstream out(std::ios::out | std::ios::binary);
184  std::ostream_iterator<char, char> oi(out);
185  boost::regex_replace(oi, s.begin(), s.end(), backOverRe, "");
186  s = out.str();
187  }
188 
189  // Replace lots of blank lines with just one blank line
190  {
191  boost::regex blankLines("\\n{3,}");
192  std::ostringstream out(std::ios::out | std::ios::binary);
193  std::ostream_iterator<char, char> oi(out);
194  boost::regex_replace(oi, s.begin(), s.end(), blankLines, "\n\n");
195  s = out.str();
196  }
197 
198  std::vector<std::string> lines;
199  boost::split(lines, s, boost::is_any_of("\n"));
200 
201  // Indent and reflow the text since it tends to not have enough linefeeds.
202  Markup::Reflow reflow(80); // standard tty width of 80 columns
203  reflow.indentationString(" "); // four is good breathing room without looking stretched
204  static const size_t itemFieldWidth = 3; // one less than indentation
205  std::vector<int> itemNumbers; // for numbered lists
206  BOOST_FOREACH (const std::string &line, lines) {
207  if (boost::starts_with(line, "=line")) {
208  reflow.lineBreak();
209  } else if (boost::starts_with(line, "=over")) {
210  ++reflow;
211  itemNumbers.push_back(0);
212  } else if (boost::starts_with(line, "=back")) {
213  --reflow;
214  if (!itemNumbers.empty())
215  itemNumbers.pop_back();
216  } else if (boost::starts_with(line, "=item")) { // =item is an implied =over
217  std::string style = boost::trim_copy(line.substr(5));
218  if (style.empty()) {
219  style = "*";
220  } else if (isdigit(style[0]) && !itemNumbers.empty()) {
221  style = boost::lexical_cast<std::string>(++itemNumbers.back()) + ".";
222  }
223  reflow(BaseMarkup::leftJustify(style, itemFieldWidth) + " ");
224  ++reflow;
225  itemNumbers.push_back(0);
226  } else {
227  reflow(line + "\n");
228  }
229  }
230  s = reflow.toString();
231 
232  // Add the header and footer
233  {
234  std::string page = boost::to_upper_copy(pageName()) + "(" + chapterNumberOrDefault() + ")";
235  std::string title = chapterTitleOrDefault();
236  std::string version = versionStringOrDefault();
237  std::string date = versionDateOrDefault();
238 
239  size_t headSize = page.size() + title.size() + page.size();
240  size_t footSize = version.size() + date.size() + page.size();
241 
242  std::string headPad = reflow.pageWidth() > headSize + 2 ?
243  std::string(round(0.5 * (reflow.pageWidth() - (headSize+2))), ' ') :
244  std::string(" ");
245 
246  std::string footPad = reflow.pageWidth() > footSize + 2 ?
247  std::string(round(0.5 * (reflow.pageWidth() - (footSize+2))), ' ') :
248  std::string(" ");
249 
250  s = (doPageHeader_ ? page + headPad + title + headPad + page + "\n\n" : std::string()) +
251  s +
252  (doPageFooter_ ? "\n" + version + footPad + date + footPad + page + "\n" : std::string());
253  }
254 
255  return s;
256 }
