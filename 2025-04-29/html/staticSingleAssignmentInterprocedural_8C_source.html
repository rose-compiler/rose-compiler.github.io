59 vector<SgFunctionDefinition*> StaticSingleAssignment::calculateInterproceduralProcessingOrder(
60  const unordered_set<SgFunctionDefinition*>& interestingFunctions)
61 {
62  //First, let's build a call graph. Our goal is to find an order in which to process the functions
63  //So that callees are processed before callers. This way we would have exact information at each call site
64  CallGraphBuilder cgBuilder(project);
65  FunctionFilter functionFilter;
66 #if DISPLAY_TIMINGS
67  timer time;
68 #endif
69  cgBuilder.buildCallGraph(functionFilter);
70 #if DISPLAY_TIMINGS
71  printf("-- Timing: Building call graph took %.2f seconds.\n", time.elapsed());
72 #endif
73 
74  SgIncidenceDirectedGraph* callGraph = cgBuilder.getGraph();
75 
76  //Build a map from SgGraphNode* to the corresponding function definitions
77  boost::unordered::unordered_map<SgFunctionDefinition*, SgGraphNode*> graphNodeToFunction;
78  set<SgGraphNode*> allNodes = callGraph->computeNodeSet();
79 
80  for (SgGraphNode* graphNode : allNodes)
81  {
82  SgFunctionDeclaration* funcDecl = isSgFunctionDeclaration(graphNode->get_SgNode());
83  ROSE_ASSERT(funcDecl != NULL);
84  funcDecl = isSgFunctionDeclaration(funcDecl->get_definingDeclaration());
85  ROSE_ASSERT(funcDecl != NULL);
86 
87  SgFunctionDefinition* funcDef = funcDecl->get_definition();
88  graphNodeToFunction[funcDef] = graphNode;
89  }
90 
91  //Order the functions of interest such that callees are processed before callers whenever possible
92  vector<SgFunctionDefinition*> processingOrder;
93 
94  for (SgFunctionDefinition* interestingFunction : interestingFunctions)
95  {
96  if (find(processingOrder.begin(), processingOrder.end(), interestingFunction) != processingOrder.end())
97  continue;
98 
99  set<SgFunctionDefinition*> visited;
100  processCalleesThenFunction(interestingFunction, callGraph, graphNodeToFunction, processingOrder, visited);
101  }
102 
103  return processingOrder;
104 }

106 void StaticSingleAssignment::processCalleesThenFunction(SgFunctionDefinition* targetFunction, SgIncidenceDirectedGraph* callGraph,
107  const boost::unordered_map<SgFunctionDefinition*, SgGraphNode*>& graphNodeToFunction,
108  std::vector<SgFunctionDefinition*> &processingOrder, std::set<SgFunctionDefinition*> visited)
109 {
110  if (visited.count(targetFunction) != 0)
111  return;
112  else
113  visited.insert(targetFunction);
114 
115  boost::unordered::unordered_map<SgFunctionDefinition*, SgGraphNode*>::const_iterator functionIter = graphNodeToFunction.find(targetFunction);
116  if (functionIter == graphNodeToFunction.end())
117  {
118  printf("The function %s has no vertex in the call graph!\n", targetFunction->get_declaration()->get_name().str());
119  return; // FIXME ROSE-1392
120  ROSE_ABORT();
121  }
122 
123  SgGraphNode const * graphNode = functionIter->second;
124 
125  vector<SgGraphNode*> callees;
126  callGraph->getSuccessors(graphNode, callees);
127 
128  //Recursively process all the callees before adding this function to the list
129 
130  for (SgGraphNode* callerNode : callees)
131  {
132  SgFunctionDeclaration* callerDecl = isSgFunctionDeclaration(callerNode->get_SgNode());
133  ROSE_ASSERT(callerDecl != NULL);
134  callerDecl = isSgFunctionDeclaration(callerDecl->get_definingDeclaration());
135  ROSE_ASSERT(callerDecl != NULL);
136  SgFunctionDefinition* callee = callerDecl->get_definition();
137 
138  if (callee == NULL)
139  {
140  fprintf(stderr, "BUG IN ROSE: The function %s has a defining declaration but no definition!?!\n",
141  callerDecl->get_qualified_name().str());
142  continue;
143  }
144 
145  processCalleesThenFunction(callee, callGraph, graphNodeToFunction, processingOrder, visited);
146  }
147 
148  //If the function is already in the list of functions to be processed, don't add it again.
149  //Some of our callees might have added us due to recursion
150  if (find(processingOrder.begin(), processingOrder.end(), targetFunction) == processingOrder.end())
151  processingOrder.push_back(targetFunction);
152 }

154 bool StaticSingleAssignment::insertInterproceduralDefs(SgFunctionDefinition* funcDef,
155  const boost::unordered_set<SgFunctionDefinition*>& processed,
156  ClassHierarchyWrapper* classHierarchy)
157 {
158  ROSE_ASSERT(funcDef != NULL);
159  vector<SgExpression*> functionCalls = SageInterface::querySubTree<SgExpression > (funcDef, V_SgFunctionCallExp);
160  vector<SgExpression*> constructorCalls = SageInterface::querySubTree<SgExpression > (funcDef, V_SgConstructorInitializer);
161  functionCalls.insert(functionCalls.end(), constructorCalls.begin(), constructorCalls.end());
162 
163  bool changedDefs = false;
164 
165  for (SgExpression* callSite : functionCalls)
166  {
167  //First, see which functions this call site leads too
168  vector<SgFunctionDeclaration*> callees;
169  CallTargetSet::getDeclarationsForExpression(callSite, classHierarchy, callees);
170 
171  LocalDefUseTable::mapped_type oldDefs = originalDefTable[callSite];
172 
173  //process each callee
174 
175  for (SgFunctionDeclaration* callee : callees)
176  {
177  processOneCallSite(callSite, callee, processed, classHierarchy);
178  }
179 
180  const LocalDefUseTable::mapped_type& newDefs = originalDefTable[callSite];
181  if (oldDefs != newDefs)
182  {
183  changedDefs = true;
184  }
185  }
186

190 void StaticSingleAssignment::processOneCallSite(SgExpression* callSite, SgFunctionDeclaration* callee,
191  const unordered_set<SgFunctionDefinition*>& processed, ClassHierarchyWrapper* classHierarchy)
192 {
193  ROSE_ASSERT(isSgFunctionCallExp(callSite) || isSgConstructorInitializer(callSite));
194  SgFunctionDefinition* calleeDef = NULL;
195  if (callee->get_definingDeclaration() != NULL)
196  {
197  calleeDef = isSgFunctionDeclaration(callee->get_definingDeclaration())->get_definition();
198  if (calleeDef == NULL)
199  {
200  fprintf(stderr, "WARNING: Working around a ROSE bug. The function %s\n", callee->get_name().str());
201  fprintf(stderr, "has a defining declaration but no definition!");
202  }
203  }
204 
205  //See if we can get exact information because the function has already been processed
206  set<VarName> varsDefinedinCallee;
207  if (calleeDef != NULL && processed.count(calleeDef) > 0)
208  {
209  //Yes, use exact info!
210  varsDefinedinCallee = getOriginalVarsDefinedInSubtree(calleeDef);
211  }
212  else
213  {
214  //Nope, use an approximate bound :(
215  }
216 
217  //Filter the variables that are not accessible from the caller and insert the rest as definitions
218 
219  for (const VarName& definedVar : varsDefinedinCallee)
220  {
221  if (isVarAccessibleFromCaller(definedVar, callSite, callee))
222  originalDefTable[callSite].insert(definedVar);
223  }
224 
225  //Check if this is a member function. In this case, we should check if the "this" instance is modified
226  SgMemberFunctionDeclaration* calleeMemFunDecl = isSgMemberFunctionDeclaration(callee);
227  if (calleeMemFunDecl != NULL
228  && !calleeMemFunDecl->get_declarationModifier().get_storageModifier().isStatic()
229  && isSgFunctionCallExp(callSite)) //No need to consider constructor initializers here, because they don't have a LHS var
230  {
231  //Get the LHS variable (e.g. x in the call site x.foo())
232  SgBinaryOp* functionRefExpression = isSgBinaryOp(isSgFunctionCallExp(callSite)->get_function());
233  ROSE_ASSERT(functionRefExpression != NULL);
234  VarName lhsVar = getVarForExpression(functionRefExpression->get_lhs_operand());
235 
236  //It's possible that the member function is not operating on a variable; e.g. the function bar in foo().bar()
237  if (lhsVar != emptyName)
238  {
239  //If the callee has no definition, then we assume it modifies the object unless it is declared const
240  //This is also our loose estimate in case there is recursion
241  if (calleeDef == NULL || processed.count(calleeDef) == 0)
242  {
243  SgMemberFunctionType* calleeFuncType = isSgMemberFunctionType(calleeMemFunDecl->get_type());
244  ROSE_ASSERT(calleeFuncType != NULL);
245  if (!calleeFuncType->isConstFunc())
246  {
247  originalDefTable[callSite].insert(lhsVar);
248  }
249  }
250  //If the callee has a definition and we have already processed it we can use exact info to check if 'this' is modified
251  else
252  {
253  //TODO: We can be more precise here! Instead of defining the lhsVar, we can find exactly which
254  //elements of the lhs var were defined and only define those.
255  //For example, obj.setX(3) should only have a def for obj.x rather than for all of obj.
256 
257  //Get the scope of variables in this class
258  SgScopeStatement* calleeScope = calleeMemFunDecl->get_class_scope();
259  ROSE_ASSERT(calleeScope != NULL);
260 
261  //If any of the callee's defined variables is a member variable, then the "this" instance has been modified
262 
263  for (const VarName& definedVar : varsDefinedinCallee)
264  {
265  //Only consider defs of member variables
266  if (!varRequiresThisPointer(definedVar)) {
267  continue;
268  }
269 
270  //If the modified var is in the callee class scope, we know "this" has been modified
271  SgScopeStatement* varScope = SageInterface::getScope(definedVar[0]);
272  ROSE_ASSERT(isSgClassDefinition(varScope));
273  if (varScope == calleeScope)
274  {
275  originalDefTable[callSite].insert(lhsVar);
276  break;
277  }
278 
279  //Even if the modified var is not in the callee's class scope, it could be an inherited variable
280  SgClassDefinition* calleeClassScope = isSgClassDefinition(calleeScope);
281  if (calleeClassScope != NULL) {
282  const ClassHierarchyWrapper::ClassDefSet& superclasses = classHierarchy->getAncestorClasses(calleeClassScope);
283  if (superclasses.find(isSgClassDefinition(varScope)) != superclasses.end())
284  {
285  originalDefTable[callSite].insert(lhsVar);
286  break;
287  }
288  }
289  }
290  }
291  }
292  }
293 
294  //
295  //Handle aliasing of parameters (e.g. parameters passed by reference)
296  //
297 
298  //Get the actual arguments
299  SgExprListExp* actualArguments = NULL;
300  if (isSgFunctionCallExp(callSite))
301  actualArguments = isSgFunctionCallExp(callSite)->get_args();
302  else if (isSgConstructorInitializer(callSite))
303  actualArguments = isSgConstructorInitializer(callSite)->get_args();
304  ROSE_ASSERT(actualArguments != NULL);
305 
306  const SgExpressionPtrList& actualArgList = actualArguments->get_expressions();
307 
308  //Get the formal arguments. We use the defining declaration if possible, because that's the one that varNames are attached to
309  SgInitializedNamePtrList formalArgList;
310  if (calleeDef != NULL)
311  formalArgList = calleeDef->get_declaration()->get_args();
312  else
313  formalArgList = callee->get_args();
314  //The number of actual arguments can be less than the number of formal arguments (with implicit arguments) or greater
315  //than the number of formal arguments (with varargs)
316 
317  //First, treat the true arguments
318  for (size_t i = 0; i < actualArgList.size() && i < formalArgList.size(); i++)
319  {
320  //Check that the actual argument was a variable name
321  const VarName& callerArgVarName = getVarForExpression(actualArgList[i]);
322  if (callerArgVarName == emptyName)
323  continue;
324 
325  //Check that the argument is passed by nonconst reference or is of a pointer type
326  //Note: here we are also filtering varArg types (SgTypeEllipse)
327  if (!isArgumentNonConstReferenceOrPointer(formalArgList[i]))
328  continue;
329 
330  //See if we can use exact info here to determine if the callee modifies the argument
331  //If not, we just take the safe assumption that the argument is modified
332  bool argModified = true;
333  if (calleeDef != NULL && processed.count(calleeDef) > 0)
334  {
335  //Get the variable name in the callee associated with the argument (since we've processed this function)
336  const VarName& calleeArgVarName = getVarName(formalArgList[i]);
337 
338  ROSE_ASSERT(calleeArgVarName != emptyName);
339  argModified = (varsDefinedinCallee.count(calleeArgVarName) > 0);
340  }
341 
342  //Define the actual parameter in the caller if the callee modifies it
343  if (argModified)
344  {
345  originalDefTable[callSite].insert(callerArgVarName);
346  }
347  }
348 
349  //Now, handle the implicit arguments. We can have an implicit argument such as (int& x = globalVar)
350  //If there are more formal arguments than actual arguments there are two cases
351  //case 1: The callee is a varArg function can no varargs are passed at the call site
352  //case 2: The callee has default argument values passed in
353  //We conly handle case 2. i.e. foo(int& x = globalVar)
354  for (size_t i = actualArgList.size(); i < formalArgList.size(); i++)
355  {
356  SgInitializedName* formalArg = formalArgList[i];
357 
358  //Again, filter out parameters passed by value and const references
359  if (!isArgumentNonConstReferenceOrPointer(formalArg))
360  continue;
361 
362  //Default arguments always have an assign initializer
363  if (formalArg->get_initializer() == NULL)
364  continue;
365 
366  ROSE_ASSERT(isSgAssignInitializer(formalArg->get_initializer()));
367  SgExpression* defaultArgValue = isSgAssignInitializer(formalArg->get_initializer())->get_operand();
368 
369  //See if the default value is a variable and that variable is in the caller's scope
370  const VarName& defaultArgVar = getVarForExpression(defaultArgValue);
371  if (defaultArgVar == emptyName || !isVarAccessibleFromCaller(defaultArgVar, callSite, callee))
372  continue;
373 
374  //See if we can use exact info here to determine if the callee modifies the argument
375  //If not, we just take the safe assumption that the argument is modified
376  bool argModified = true;
377  if (calleeDef != NULL && processed.count(calleeDef) > 0)
378  {
379  //Get the variable name in the callee associated with the argument (since we've processed this function)
380  const VarName& calleeArgVarName = getVarName(formalArgList[i]);
381 
382  ROSE_ASSERT(calleeArgVarName != emptyName);
383  argModified = (varsDefinedinCallee.count(calleeArgVarName) > 0);
384  }
385 
386  //Define the default argument value in the caller if the callee modifies it
387  if (argModified)
388  {
389  originalDefTable[callSite].insert(defaultArgVar);
390  }
391  }
