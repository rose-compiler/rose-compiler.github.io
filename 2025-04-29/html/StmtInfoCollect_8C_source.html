74 bool StmtInfoCollect ::
75 ProcessTree( AstInterface &fa, const AstInterface::AstNodePtr& s,
76  AstInterface::TraversalVisitType t)
77 {
78  if (t == AstInterface::PreVisit) {
79  AstInterface::AstNodePtr lhs, rhs, body;
80  AstInterface::AstNodeList vars, args;
81  AstInterface::OperatorEnum opr;
82  bool readlhs = false;
83 
84  AstNodePtr s_ptr = AstNodePtrImpl(s).get_ptr();
85  if (fa.IsFunctionDefinition(s, 0, &vars, &args, &body)) {
86  curstmt = AstNodePtrImpl(s).get_ptr();
87  // Below is necessary for constructors that initialize member variables.
88  for (AstInterface::AstNodePtr param : args) {
89  if (fa.IsAssignment(param, &lhs, &rhs, &readlhs)) {
90  if (rhs != 0) {
91  operator()(fa, AstNodePtrImpl(rhs).get_ptr());
92  }
93  AppendModLoc(fa, AstNodePtrImpl(lhs).get_ptr(), AstNodePtrImpl(rhs).get_ptr());
94  }
95  }
96  SkipUntil(body);
97  return true;
98  }
99  else if (fa.IsStatement(s)) {
100  DebugLocalInfoCollect([&s]() { return "previsiting cur statement " + AstInterface::AstToString(s); });
101  curstmt = AstNodePtrImpl(s).get_ptr();
102  }
103  else {
104  DebugLocalInfoCollect([&s](){return "previsiting cur node " + AstInterface::AstToString(s); });
105  if (curstmt == 0) {
106  AstInterface::AstNodePtr s1 = fa.GetParent(s);
107  for ( ; s1 != 0 && !fa.IsStatement(s1); s1 = fa.GetParent(s1));
108  if (s1 != 0) {
109  curstmt = AstNodePtrImpl(s1).get_ptr();
110  } else {
111  DebugLocalInfoCollect([](){ return "curstmt = 0"; });
112  }
113  }
114  }
115 
116  if (fa.IsAssignment(s, &lhs, &rhs, &readlhs)) {
117  DebugLocalInfoCollect([](){return "Is assignment"; });
118  ModMap *mp = modstack.size()? &modstack.back().modmap : 0;
119  if (mp == 0 || mp->find(AstNodePtrImpl(lhs).get_ptr()) == mp->end()) {
120  modstack.push_back(s_ptr);
121  modstack.back().modmap[AstNodePtrImpl(lhs).get_ptr()] = ModRecord(AstNodePtrImpl(rhs).get_ptr(),readlhs);
122  }
123  }
124  else if (fa.IsUnaryOp(s, &opr, &lhs) &&
125  (opr == AstInterface::UOP_INCR1 || opr == AstInterface::UOP_DECR1)){
126  DebugLocalInfoCollect([](){ return "Is unary operator."; });
127  ModMap *mp = modstack.size()? &modstack.back().modmap : 0;
128  if (mp == 0 || mp->find(AstNodePtrImpl(lhs).get_ptr()) == mp->end()) {
129  modstack.push_back(s_ptr);
130  modstack.back().modmap[AstNodePtrImpl(lhs).get_ptr()] = ModRecord(AstNodePtrImpl(s).get_ptr(),true);
131  }
132  }
133  else if (fa.IsVariableDecl( s, &vars, &args)) {
134  DebugLocalInfoCollect([](){ return "Is variable declaration."; });
135  AstInterface::AstNodeList::const_iterator pv = vars.begin();
136  AstInterface::AstNodeList::const_iterator pa = args.begin();
137  modstack.push_back(s_ptr);
138  while (pv != vars.end()) {
139  AstNodePtr ast = AstNodePtrImpl(*pv).get_ptr();
140  AstNodePtr read_ast = AstNodePtrImpl(*pa).get_ptr();
141  if (read_ast != 0) {
142  operator()(fa, read_ast);
143  }
144  AppendVariableDecl( fa, ast, read_ast);
145  ++pv;
146  ++pa;
147  }
148  Skip(s);
149  }
150  else if (fa.IsIOInputStmt(s, &args)) {
151  DebugLocalInfoCollect([](){ return "Is IOInput stmt."; });
152  modstack.push_back(s_ptr);
153  for (AstInterface::AstNodeList::reverse_iterator p = args.rbegin();
154  p != args.rend(); ++p) {
155  AstNodePtr c = AstNodePtrImpl(*p).get_ptr();
156  modstack.back().modmap[c]= ModRecord(0,false);
157  }
158  }
159  else if (AstInterface::IsMemoryAllocation(s)) {
160  AppendMemoryAllocate(fa, s.get_ptr());
161  Skip(s);
162  }
163  else if (AstInterface::IsMemoryFree(s, 0, &lhs)) {
164  AppendMemoryFree(fa, lhs.get_ptr());
165  Skip(s);
166  }
167  else if (fa.IsFunctionCall(s)) {
168  DebugLocalInfoCollect([&s]() { return " append function call " + AstInterface::AstToString(s); });
169  AppendFuncCall(fa, AstNodePtrImpl(s).get_ptr());
170  Skip(s);
171  }
172  // Jim Leek 2023/02/07 Added IsSgAddressOfOp because I want it
173  // to behave the same as a memory access, although it isn't one exactly
174  else if ( fa.IsMemoryAccess(s) || fa.IsAddressOfOp(s)) {
175  DebugLocalInfoCollect([&s](){ return " append read set " + AstInterface::AstToString(s); });
176  if (!fa.IsSameVarRef(s, fa.GetParent(s))) { /*QY: skip s if it refers to the same thing as parent*/
177  ModMap *mp = modstack.size()? &modstack.back().modmap : 0;
178  if (mp == 0 || mp->find(s_ptr) == mp->end() || (*mp)[s_ptr].readlhs) {
179  AppendReadLoc(fa, AstNodePtr(s_ptr));
180  }
181  }
182  AstNodeList arglist;
183  if (fa.IsArrayAccess(s, 0, &arglist)) {
184  for (AstNodeList::const_iterator p = arglist.begin();
185  p != arglist.end(); ++p) {
186  AstNodePtr c = AstNodePtrImpl(*p).get_ptr();
187  operator()(fa, c);
188  }
189  }
190  Skip(s);
191  }
192  }
193  else {
194  DebugLocalInfoCollect([&s](){ return "postvisiting cur node " + AstInterface::AstToString(s); });
195  if (modstack.size() && modstack.back().root == AstNodePtrImpl(s).get_ptr()) {
196  const ModMap &modmap = modstack.back().modmap;
197  for (typename ModMap::const_iterator p = modmap.begin();
198  p != modmap.end(); ++p) {
199  std::pair<const AstNodePtr,ModRecord> c = *p;
200  AppendModLoc( fa, c.first, c.second.rhs);
201  }
202  modstack.pop_back();
203  }
204  if (s == curstmt) {
205  curstmt = 0;
206  }
207  }
208  return true;
209 }

546 bool StmtVarAliasCollect::
547 may_alias(AstInterface& fa, const AstNodePtr& r1,
548  const AstNodePtr& r2)
549 {
550  DebugLog DebugAliasAnal("-debugaliasanal");
551  if (r1 == AST_NULL || r2 == AST_NULL) {
552  return false;
553  }
554  if (r1 == AST_UNKNOWN || r2 == AST_UNKNOWN) {
555  return true;
556  }
557  AstNodeType t1, t2;
558  if (!fa.IsExpression(r1, &t1) || !fa.IsExpression(r2, &t2))
559  ROSE_ABORT();
560  if (!hasresult) {
561  DebugAliasAnal([&r1,&r2](){ return "No alias analysis performed. Has alias between " + AstInterface::AstToString(r1) + " and " + AstInterface::AstToString(r2); });
562  return true;
563  }
564  std::string varname1, varname2;
565  AstNodePtr scope1, scope2;
566  bool global1, global2;
567  if (!fa.IsVarRef(r1, 0, &varname1, &scope1, &global1)
568  || !fa.IsVarRef(r2, 0, &varname2, &scope2, &global2)) {
569  DebugAliasAnal([&r1, &r2](){ return "no alias analysis between non_variables performed. Has alias between " + AstInterface::AstToString(r1) + " and " + AstInterface::AstToString(r2); });
570  return true;
571  }
572 
573  if (global1 && global2 && hasunknown) {
574  if (global1) {
575  DebugAliasAnal([&varname1,&varname2](){ return "INFO:" + varname1 + " and " + varname2 + " are global"; });
576  DebugAliasAnal([&r1,&r2]() { return "INFO: Has alias between " + AstInterface::AstToString(r1) + " and " + AstInterface::AstToString(r2); });
577  }
578  return true;
579  }
580  if ( aliasmap.get_alias_map(varname1, scope1)->in_same_group( aliasmap.get_alias_map(varname2, scope2))) {
581  DebugAliasAnal([&r1,&r2](){ return "INFO: Alias analysis performed. Has alias between " + AstInterface::AstToString(r1) + " and " + AstInterface::AstToString(r2); });
582  return true;
583  }
584  return false;
585 }
