193 bool QPanel::forward(QMouseEvent *e)
194 {
195  QPoint pos, globalPos = e->globalPos(), ref = editor()->viewport()->pos();
196 
197  if ( editor()->viewport()->parentWidget() )
198  ref = editor()->viewport()->parentWidget()->mapToGlobal(ref);
199 
200  globalPos.setX(qBound(ref.x(), globalPos.x(), ref.x() + editor()->width()));
201  globalPos.setY(qBound(ref.y(), globalPos.y(), ref.y() + editor()->height()));
202 
203  pos = editor()->viewport()->mapFromGlobal(globalPos);
204 
205  QMouseEvent fw(
206  e->type(),
207  pos,
208  globalPos,
209  e->button(),
210  e->buttons(),
211  e->modifiers()
212  );
213 
214  bool ok = qApp->sendEvent(editor()->viewport(), &fw) && fw.isAccepted();
215 
216  //qDebug("forwarding mouse event : (%i, %i) => %i", pos.x(), pos.y(), ok);
217 
218  return ok;
219 }
