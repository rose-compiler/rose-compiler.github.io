56 class LvarSymbolTable {
57  public:
58  struct TableEntry {
59  POETCode *code, *restr;
60  LocalVar *var;
61  LocalVarType t;
62  TableEntry* next;
63  TableEntry() : code(0), restr(0), var(0), t(LVAR_REG), next(0) {}
64  TableEntry(POETCode* c, LocalVarType _t, TableEntry* n, LocalVar* v)
65  : code(c), restr(0), var(v), t(_t), next(n) { }
66  void reset() {
67  if (t == LVAR_ATTR || t == LVAR_OUTPUT) return;
68  if (t != LVAR_TUNE && t != LVAR_TUNE_GROUP)
69  code = 0;
70  }
71  void push(bool copycurrent = false) {
72  if (t == LVAR_OUTPUT) return;
73  next = new TableEntry(code,t, next,var);
74  switch (t)
75  {
76  case LVAR_REG:
77  case LVAR_CODEPAR:
78  case LVAR_XFORMPAR:
79  case LVAR_TRACE:
80  case LVAR_TRACE_OUTDATE:
81  if (!copycurrent) code = 0; break;
82  case LVAR_ATTR:
83  if (!copycurrent) code = restr;
84  default: ; /* do nothing */
85  }
86  }
87  void pop() {
88  TableEntry* d = next;
89  if (d != 0) {
90  t = d->t;
91  code = d->code;
92  next = d->next;
93  }
94  else if (t == LVAR_REG) {
95  code = 0;
96  }
97  if (d != 0) delete d;
98  }
99  };
100  private:
101  SymbolTable<TableEntry> impl;
102  public:
103  LvarSymbolTable() {}
104  class Entry {
105  SymbolTable<TableEntry>::iterator impl;
106  void set_var(LocalVar* v) { (*impl).second.var = v; }
107  public:
108  Entry(const SymbolTable<TableEntry>::iterator& p, LocalVarType _t)
109  : impl(p) { (*p).second.t = _t; }
110  Entry(const SymbolTable<TableEntry>::iterator& p) : impl(p) {}
111  POETCode* get_name() const { return (*impl).first; }
112  POETCode* get_code() const { return (*impl).second.code; }
113  void set_code(POETCode* c) { (*impl).second.code = c; }
114  POETCode* get_restr() const { return (*impl).second.restr; }
115  void set_restr(POETCode* c) { (*impl).second.restr = c; }
116  LocalVar* get_var() const { return (*impl).second.var; }
117  LocalVarType get_entry_type() const { return (*impl).second.t; }
118  void set_entry_type(LocalVarType t) { (*impl).second.t = t; }
119  Entry operator ++() { return ++impl; }
120  Entry operator ++(int) { return impl++; }
121  bool operator == (const Entry& that) const { return impl == that.impl; }
122  bool operator != (const Entry& that) const { return impl != that.impl; }
123 
124  void push(bool copycurrent = false) {
125  (*impl).second.push(copycurrent);
126  }
127  void pop() {
128  (*impl).second.pop();
129  }
130  friend class LvarSymbolTable;
131  };
132  Entry begin() { return impl.begin(); }
133  Entry end() { return impl.end(); }
134  LocalVar* find (POETCode* var) {
135  Entry r = impl.find(var);
136  if (r.impl == impl.end()) return 0;
137  return r.get_var();
138  }
139  LocalVar* insert(POETCode* var, LocalVarType _t = LVAR_UNKNOWN);
140  int size() const { return impl.size(); }
141  void push_table(bool copycurrent) {
142  for (Entry p = begin(); p != end(); ++p) {
143  TableEntry& e = (*p.impl).second;
144  e.push(copycurrent);
145  }
146  }
147  void reset_table() {
148  for (Entry p = begin(); p != end(); ++p) {
149  TableEntry& e = (*p.impl).second;
150  e.reset();
151  }
152  }
153  void pop_table() {
154  for (Entry p = begin(); p != end(); ++p) {
155  TableEntry& e = (*p.impl).second;
156  e.pop();
157  }
158  }
