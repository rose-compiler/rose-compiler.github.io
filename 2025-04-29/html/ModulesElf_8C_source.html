117 PltEntryMatcher::matchIndirectJump(const Partitioner::ConstPtr &partitioner, Address va, Address &indirectVa /*out*/,
118  size_t &indirectNBytes /*out*/) {
119  ASSERT_not_null(partitioner);
120  indirectVa = 0;
121  indirectNBytes = 0;
122  const RegisterDescriptor REG_IP = partitioner->instructionProvider().instructionPointerRegister();
123  if (SgAsmX86Instruction *insn = isSgAsmX86Instruction(partitioner->discoverInstruction(va))) {
124  if (x86InstructionIsUnconditionalBranch(insn) && insn->nOperands() == 1 &&
125  isSgAsmMemoryReferenceExpression(insn->operand(0))) {
126  SgAsmMemoryReferenceExpression *mre = isSgAsmMemoryReferenceExpression(insn->operand(0));
127 
128  // The address could be either a constant, or the IP register plus a constant (or a constant plus IP).
129  if (SgAsmIntegerValueExpression *ival = isSgAsmIntegerValueExpression(mre->get_address())) {
130  indirectVa = ival->get_absoluteValue();
131  indirectNBytes = mre->get_type()->get_nBytes();
132  return insn;
133  } else if (SgAsmBinaryAdd *add = isSgAsmBinaryAdd(mre->get_address())) {
134  SgAsmDirectRegisterExpression *dre = NULL;
135  SgAsmIntegerValueExpression *ival = NULL;
136  if (isSgAsmDirectRegisterExpression(add->get_lhs()) && isSgAsmIntegerValueExpression(add->get_rhs())) {
137  dre = isSgAsmDirectRegisterExpression(add->get_lhs());
138  ival = isSgAsmIntegerValueExpression(add->get_rhs());
139  } else if (isSgAsmIntegerValueExpression(add->get_lhs()) && isSgAsmDirectRegisterExpression(add->get_rhs())) {
140  dre = isSgAsmDirectRegisterExpression(add->get_rhs());
141  ival = isSgAsmIntegerValueExpression(add->get_lhs());
142  }
143  if (dre && ival && dre->get_descriptor() == REG_IP) {
144  indirectVa = insn->get_address() + insn->get_size() + ival->get_absoluteValue();
145  indirectNBytes = mre->get_type()->get_nBytes();
146  return insn;
147  }
148  }
149  }
150  }
151  return NULL;

285 PltEntryMatcher::matchAarch32CopyPcToIp(const Partitioner::ConstPtr &partitioner, Address va, uint32_t &result) {
286  ASSERT_not_null(partitioner);
287  // Matches: "add ip, pc, ror(X,Y)" where X and Y are constants. The resulting IP value is returned via "value" argument.
288 #ifdef ROSE_ENABLE_ASM_AARCH32
289  auto insn = isSgAsmAarch32Instruction(partitioner->discoverInstruction(va));
290  if (!insn || insn->get_kind() != Aarch32InstructionKind::ARM_INS_ADD || insn->nOperands() != 3)
291  return nullptr;
292 
293  const RegisterDescriptor REG_IP(aarch32_regclass_gpr, aarch32_gpr_ip, 0, 32);
294  auto ip = isSgAsmDirectRegisterExpression(insn->operand(0));
295  if (!ip || ip->get_descriptor() != REG_IP)
296  return nullptr;
297 
298  const RegisterDescriptor REG_PC(aarch32_regclass_gpr, aarch32_gpr_pc, 0, 32);
299  auto pc = isSgAsmDirectRegisterExpression(insn->operand(1));
300  if (!pc || pc->get_descriptor() != REG_PC)
301  return nullptr;
302 
303  if (auto ror = isSgAsmBinaryRor(insn->operand(2))) {
304  auto base = isSgAsmIntegerValueExpression(ror->get_lhs());
305  auto amount = isSgAsmIntegerValueExpression(ror->get_rhs());
306  if (!base || !amount)
307  return nullptr;
308  uint32_t b = base->get_absoluteValue();
309  uint32_t a = amount->get_absoluteValue();
310  result = va + BitOps::rotateRight(b, a);
311  return insn;
312  } else if (auto ival = isSgAsmIntegerValueExpression(insn->operand(2))) {
313  result = va + ival->get_absoluteValue();
314  return insn;
315  }
316 #else
317  ROSE_UNUSED(partitioner);
318  ROSE_UNUSED(va);
319  ROSE_UNUSED(result);
320 #endif
321  return nullptr;
322 }

393 bool
394 PltEntryMatcher::match(const Partitioner::ConstPtr &partitioner, Address anchor) {
395  ASSERT_not_null(partitioner);
396  nBytesMatched_ = 0;
397  gotEntryVa_ = 0;
398  gotEntry_ = 0;
399  gotEntryNBytes_ = 0;
400  functionNumber_ = 0;
401 
402  SgAsmInstruction *insn = partitioner->discoverInstruction(anchor);
403  if (!insn)
404  return false;
405 
406  // Look for the PLT entry.
407  if (isSgAsmX86Instruction(insn)) {
408  bool found = false;
409  Address indirectVa=0;
410  size_t indirectSize=0;
411 
412  if (!found) {
413  // i386 entries that look like this:
414  // jmp dword [CONST] ; where CONST is a GOT entry address
415  // push N ; where N is a small integer, unique for each dynamically linked function
416  // jmp X ; where X is the address of the first PLT entry
417  SgAsmInstruction *ijmp = matchIndirectJump(partitioner, anchor, indirectVa /*out*/, indirectSize /*out*/);
418  SgAsmInstruction *push = ijmp ? matchPush(partitioner, ijmp->get_address() + ijmp->get_size(), functionNumber_) : NULL;
419  SgAsmInstruction *djmp = push ? matchDirectJump(partitioner, push->get_address() + push->get_size()) : NULL;
420  if (djmp) {
421  gotEntryNBytes_ = indirectSize;
422  gotEntryVa_ = indirectVa;
423  nBytesMatched_ = djmp->get_address() + djmp->get_size() - anchor;
424  pltEntryAlignment_ = nBytesMatched_;
425  found = true;
426  }
427  }
428 
429  if (!found) {
430  // i386 compiled with -fPIC, occuring in the ".plt.sec" section. There's also a .plt section that gets called after
431  // lookup up the address in the GOT.
432  // nop ; 4 bytes
433  // jmp [ ebx + CONST ] ; where CONST is the offset into the GOT
434  // nop ; 6 bytes
435  Address offset = 0;
436  SgAsmInstruction *nop1 = matchNop(partitioner, anchor);
437  SgAsmInstruction *ijmp = nop1 ? matchIndirectJumpEbx(partitioner, nop1->get_address() + nop1->get_size(),
438  offset /*out*/, indirectSize /*out*/) : NULL;
439  SgAsmInstruction *nop2 = ijmp ? matchNop(partitioner, ijmp->get_address() + ijmp->get_size()) : NULL;
440  if (nop2) {
441  gotEntryNBytes_ = indirectSize;
442  gotEntryVa_ = gotVa_ + offset;
443  nBytesMatched_ = nop2->get_address() + nop2->get_size() - anchor;
444  pltEntryAlignment_ = nBytesMatched_;
445  found = true;
446  }
447  }
448 
449  if (!found) {
450  // i386 that look like this:
451  // jmp dword [ ebx + CONST ]; where CONST is the offset into the GOT
452  // push N ; where N is a small integer, unique for each dynamically linked function
453  // jmp X ; where X is the address of the first PLT entry
454  Address offset = 0;
455  SgAsmInstruction *ijmp = matchIndirectJumpEbx(partitioner, anchor, offset /*out*/, indirectSize /*out*/);
456  SgAsmInstruction *push = ijmp ? matchPush(partitioner, ijmp->get_address() + ijmp->get_size(), functionNumber_) : NULL;
457  SgAsmInstruction *djmp = push ? matchDirectJump(partitioner, push->get_address() + push->get_size()) : NULL;
458  if (djmp) {
459  gotEntryNBytes_ = indirectSize;
460  gotEntryVa_ = gotVa_ + offset;
461  nBytesMatched_ = djmp->get_address() + djmp->get_size() - anchor;
462  pltEntryAlignment_ = nBytesMatched_;
463  found = true;
464  }
465  }
466 
467  if (!found) {
468  // Amd64 entries that look like this:
469  // nop; 4 bytes
470  // jmp qword [rip + CONST]; address of a GOT entry
471  // nop; 5 bytes
472  SgAsmInstruction *nop1 = matchNop(partitioner, anchor);
473  SgAsmInstruction *ijmp = nop1 ? matchIndirectJump(partitioner, nop1->get_address() + nop1->get_size(),
474  indirectVa /*out*/, indirectSize /*out*/) : NULL;
475  SgAsmInstruction *nop2 = ijmp ? matchNop(partitioner, ijmp->get_address() + ijmp->get_size()) : NULL;
476  if (nop2) {
477  gotEntryNBytes_ = indirectSize;
478  gotEntryVa_ = indirectVa;
479  nBytesMatched_ = nop2->get_address() + nop2->get_size() - anchor;
480  pltEntryAlignment_ = nBytesMatched_;
481  found = true;
482  }
483  }
484 
485  if (!found) {
486  // i386 and amd64 entries that look like this when compiled with -fPIC
487  // nop ; 4 bytes
488  // push N ; where N is a small integer, unique for each dynamically linked function
489  // jmp X ; where X is the address of the first PLT entry
490  // nop ; 5 bytes
491  SgAsmInstruction *nop1 = matchNop(partitioner, anchor);
492  SgAsmInstruction *push = nop1 ? matchPush(partitioner, nop1->get_address() + nop1->get_size(), functionNumber_) : NULL;
493  SgAsmInstruction *djmp = push ? matchDirectJump(partitioner, push->get_address() + push->get_size()) : NULL;
494  SgAsmInstruction *nop2 = djmp ? matchNop(partitioner, djmp->get_address() + djmp->get_size()) : NULL;
495  if (nop2) {
496  gotEntryNBytes_ = 0; // not present in PLT entry
497  gotEntryVa_ = 0; // not present in PLT entry
498  nBytesMatched_ = nop2->get_address() + nop2->get_size() - anchor;
499  pltEntryAlignment_ = nBytesMatched_;
500  found = true;
501  }
502  }
503 
504 #ifdef ROSE_ENABLE_ASM_AARCH64
505  } else if (isSgAsmAarch64Instruction(insn)) {
506  if (0 == gotEntryNBytes_) {
507  // Aarch64 entries look like this:
508  // adrp x16, 0x00011000
509  // ldr x17, u64 [x16 + 0x0000000000000ef8<3832>]
510  // add x16, x16, 0x0000000000000ef8<3832>
511  // br x17
512  Address indirectVa = 0;
513  Address indirectNBytes = 0;
514  SgAsmInstruction *adrp = matchAarch64Adrp(partitioner, anchor, indirectVa /*out*/);
515  SgAsmInstruction *ldr = adrp ? matchAarch64Ldr(partitioner, adrp->get_address() + adrp->get_size(),
516  indirectVa /*in,out*/, indirectNBytes /*out*/) : NULL;
517  SgAsmInstruction *add = ldr ? matchAarch64Add(partitioner, ldr->get_address() + ldr->get_size()) : NULL;
518  SgAsmInstruction *br = add ? matchAarch64Br(partitioner, add->get_address() + add->get_size()) : NULL;
519  if (br) {
520  gotEntryNBytes_ = indirectNBytes;
521  gotEntryVa_ = indirectVa;
522  nBytesMatched_ = br->get_address() + br->get_size() - anchor;
523  pltEntryAlignment_ = nBytesMatched_;
524  }
525  }
526 #endif
527 
528 #ifdef ROSE_ENABLE_ASM_AARCH32
529  } else if (isSgAsmAarch32Instruction(insn)) {
530  if (0 == gotEntryNBytes_) {
531  // AArch32 entries look like this:
532  // add ip, pc, ror(12,0) // the ror(12,0) doesn't matter since it's always zero
533  // add ip, ip, 0x00011000 // higher order bits of offset into the .got section
534  // ldr pc, u32 [ip += 0x000008b4] // lower order bits of offset into the .got section
535  uint32_t addend1=0, addend2=0, addend3=0;
536 
537  SgAsmInstruction *add1 = matchAarch32CopyPcToIp(partitioner, anchor, addend1 /*out*/);
538  SgAsmInstruction *add2 = add1 ? matchAarch32AddConstToIp(partitioner, anchor+4, addend2 /*out*/) : NULL;
539  SgAsmInstruction *ldr = add2 ? matchAarch32IndirectBranch(partitioner, anchor+8, addend3 /*out*/) : NULL;
540  if (ldr) {
541  gotEntryVa_ = addend1 + addend2 + addend3 - 4;
542  gotEntryNBytes_ = 4;
543  nBytesMatched_ = 12;
544  pltEntryAlignment_ = 4; // an alignment of 12 (entry size) is apparently not needed
545  }
546  }
547 #endif
548 
549  } else if (isSgAsmM68kInstruction(insn)) {
550  // m68k entries look like this (from GDB-4.1.6's bfd/elf32-m68k.c file):
551  // 4e fb 01 71: jmp ([%pc, symbol@GOTPC])
552  // 00 00 00 00: replaced with offset to symbol's .got entry
553  // 2f 3c : move.l #offset, -(%sp)
554  // 00 00 00 00: replaced with offset into relocation table
555  // 60 ff : bra.l .plt
556  // 00 00 00 00: replaced with offset to start of .plt
557  //
558  // In practice, the 20-byte entry looks like this on disk:
559  // This example's PLT starts at 8000081c and GOT starts at 80004000.
560  // 800008a8: entry #7 for "time" function
561  // 00 00 00 00
562  // 00 00 37 7a -- 800008a8 + 377a = 80004022, or two bytes before time's .got entry of 4024
563  // 2f 3c
564  // 00 00 00 48
565  // 60 ff
566  // ff ff ff 64 -- + ffffff64 = 8000080c, or 0x10 bytes before the beginning of this PLT
567  //
568  // From the same source as above also describes the first PLT entry as being:
569  // 2f 3b 01 70: move.l (%pc,addr), -(%sp)
570  // 00 00 00 00: replaced with offset to .got+4
571  // 4e fb 01 71: jmp ([%pc, addr])
572  // 00 00 00 00: replaced with offset to .got+8
573  // 00 00 00 00: pad out to 20 bytes
574  if (partitioner->memoryMap() != NULL) {
575  uint8_t entry[20];
576  size_t nRead = partitioner->memoryMap()->at(anchor).limit(sizeof entry).read(entry).size();
577  if (nRead == sizeof entry && entry[8] == 0x2f && entry[9] == 0x3c && entry[14] == 0x60 && entry[15] == 0xff) {
578  Address gotOffset = (entry[4] << 24) | (entry[5] << 16) | (entry[6] << 8) | entry[7];
579  gotEntryVa_ = (anchor + gotOffset + 2) & 0xffffffffUL;
580  gotEntryNBytes_ = 4;
581  nBytesMatched_ = 20;
582  pltEntryAlignment_ = nBytesMatched_;
583  }
584  }
585 
586  } else if (isSgAsmNullInstruction(insn)) {
587  // Null ISA has no parsable PLT section
588 
589  } else {
590  // FIXME[Robb P. Matzke 2014-08-23]: Architecture is not supported yet
591  static bool warned = false;
592  if (!warned) {
593  mlog[WARN] <<"ModulesElf::pltEntryMatcher does not yet support this ISA (" <<insn->class_name() <<")\n";
594  warned = true;
595  }
596  }
597 
598  // Read the GOT entry if it's mapped
599  if (nBytesMatched_ > 0 && partitioner->memoryMap() != NULL) {
600  switch (gotEntryNBytes_) {
601  case 0:
602  gotEntry_ = 0;
603  break;
604  case 1:
605  gotEntry_ = partitioner->memoryMap()->readUnsigned<uint8_t>(gotEntryVa_).orElse(0);
606  break;
607  case 2:
608  gotEntry_ = partitioner->memoryMap()->readUnsigned<uint16_t>(gotEntryVa_).orElse(0);
609  break;
610  case 4:
611  gotEntry_ = partitioner->memoryMap()->readUnsigned<uint32_t>(gotEntryVa_).orElse(0);
612  break;
613  case 8:
614  gotEntry_ = partitioner->memoryMap()->readUnsigned<uint64_t>(gotEntryVa_).orElse(0);
615  break;
616  default:
617  ASSERT_not_reachable("invalid GOT entry size: " + StringUtility::numberToString(gotEntryNBytes_));
618  }
619  }
620 
621  return nBytesMatched_ > 0;
622 }

624 PltInfo
625 findPlt(const Partitioner::ConstPtr &partitioner, SgAsmGenericSection *got, SgAsmElfFileHeader *elfHeader) {
626  if (elfHeader) {
627  // The procedure lookup table can be in a variety of places.
628  bool foundSection = false;
629  for (size_t i = 0; i < 3; ++i) {
630  SgAsmGenericSection *section = NULL;
631  switch (i) {
632  case 0:
633  // A .plt.sec is either a subregion of .plt (it excludes the leading bogus entries) or it's a completely
634  // separate table. When it's a separate table, we want to parse it instead of .plt because the .plt.sec is what
635  // has the offsets into the .got, which in turn is what the relocation symbols refer to. This is used, for
636  // instance, when compiling with -fPIC on i386.
637  section = elfHeader->get_sectionByName(".plt.sec");
638  break;
639  case 1:
640  // The normal .plt is used when there is no .plt.sec.
641  section = elfHeader->get_sectionByName(".plt");
642  break;
643  case 2:
644  // Not sure what this is.
645  section = elfHeader->get_sectionByName(".plt.got");
646  break;
647  }
648 
649  // Search for the first valid entry.
650  if (section) {
651  foundSection = true;
652  SAWYER_MESG(mlog[DEBUG]) <<"searching for PLT in \""
653  <<StringUtility::cEscape(section->get_name()->get_string()) <<"\""
654  <<" at " <<StringUtility::addrToString(section->get_mappedActualVa())
655  <<" + " <<StringUtility::addrToString(section->get_mappedSize())
656  <<" = " <<StringUtility::addrToString(section->get_mappedActualVa() +
657  section->get_mappedSize())
658  <<"\n";
659  for (size_t offset = 0; offset < section->get_size(); ++offset) {
660  PltEntryMatcher matcher(got ? got->get_mappedActualVa() : 0);
661  if (matcher.match(partitioner, section->get_mappedActualVa() + offset)) {
662  SAWYER_MESG(mlog[DEBUG]) <<" matched PLT " <<matcher.nBytesMatched() <<" byte entry"
663  <<" at offset " <<offset <<"\n";
664  if (offset + matcher.nBytesMatched() > section->get_size()) {
665  SAWYER_MESG(mlog[DEBUG]) <<" discarded; extends past section end\n";
666  break;
667  } else if (offset % matcher.pltEntryAlignment() != 0) {
668  SAWYER_MESG(mlog[DEBUG]) <<" offset is not a multiple of the "
669  <<matcher.pltEntryAlignment() <<" byte entry size\n";
670  continue;
671  } else {
672  SAWYER_MESG(mlog[DEBUG]) <<" entries are each "
673  <<StringUtility::plural(matcher.nBytesMatched(), "bytes") <<"\n";
674  SAWYER_MESG(mlog[DEBUG]) <<" entries are each aligned on a "
675  <<matcher.pltEntryAlignment() <<"-byte boundary\n";
676  PltInfo plt;
677  plt.section = section;
678  plt.firstOffset = offset;
679  plt.entrySize = matcher.nBytesMatched();
680  return plt;
681  }
682  }
683  }
684  }
685  }
686  if (foundSection)
687  mlog[WARN] <<"found PLT section(s) but could not parse table or table is empty\n";
688  }
689 
690  return PltInfo();
691 }

693 size_t
694 findPltFunctions(const Partitioner::Ptr &partitioner, SgAsmElfFileHeader *elfHeader, std::vector<Function::Ptr> &functions) {
695  Sawyer::Message::Stream debug(mlog[DEBUG]);
696 
697  // Find important sections
698  SgAsmGenericSection *got = partitioner->elfGot(elfHeader);
699  PltInfo plt = findPlt(partitioner, got, elfHeader);
700  if (!plt.section || !plt.section->isMapped() || !got || !got->isMapped() || 0 == plt.entrySize)
701  return 0;
702 
703  // Find all relocation sections
704  std::set<SgAsmElfRelocSection*> relocSections;
705  for (SgAsmGenericSection *section: elfHeader->get_sections()->get_sections()) {
706  if (SgAsmElfRelocSection *relocSection = isSgAsmElfRelocSection(section))
707  relocSections.insert(relocSection);
708 
709  }
710  if (relocSections.empty())
711  return 0;
712 
713  // Look at each instruction in the .plt section. If the instruction is a computed jump to an address stored in the .got.plt
714  // then we've found the beginning of a plt trampoline.
715  size_t nInserted = 0;
716  for (Address pltOffset = plt.firstOffset;
717  pltOffset + plt.entrySize <= plt.section->get_mappedSize();
718  pltOffset += plt.entrySize) {
719  PltEntryMatcher matcher(got->get_mappedActualVa());
720  Address pltEntryVa = plt.section->get_mappedActualVa() + pltOffset;
721  if (!matcher.match(partitioner, pltEntryVa))
722  continue;
723  Address gotVa = matcher.gotEntryVa(); // address that was read by indirect branch
724  if (gotVa < elfHeader->get_baseVa() + got->get_mappedPreferredRva() ||
725  gotVa >= elfHeader->get_baseVa() + got->get_mappedPreferredRva() + got->get_mappedSize()) {
726  continue; // jump is not indirect through the .got.plt section
727  }
728  SAWYER_MESG(debug) <<"symbolizing PLT entry " <<StringUtility::addrToString(pltEntryVa)
729  <<" GOT entry " <<StringUtility::addrToString(gotVa) <<"\n";
730 
731  // Find the relocation entry whose offset is the gotVa and use that entry's symbol for the function name
732  std::string name;
733  for (SgAsmElfRelocSection *relocSection: relocSections) {
734  SgAsmElfSymbolSection *symbolSection = isSgAsmElfSymbolSection(relocSection->get_linkedSection());
735  if (SgAsmElfSymbolList *symbols = symbolSection ? symbolSection->get_symbols() : NULL) {
736  for (SgAsmElfRelocEntry *rel: relocSection->get_entries()->get_entries()) {
737  if (rel->get_r_offset() == gotVa) {
738  unsigned long symbolIdx = rel->get_sym();
739  if (symbolIdx < symbols->get_symbols().size()) {
740  SgAsmElfSymbol *symbol = symbols->get_symbols()[symbolIdx];
741  name = symbol->get_name()->get_string() + "@plt";
742  SAWYER_MESG(debug) <<" found relocation symbol " <<StringUtility::addrToString(rel->get_r_offset())
743  <<" \"" <<StringUtility::cEscape(symbol->get_name()->get_string()) <<"\"\n";
744  goto foundName;
745  }
746  }
747  }
748  }
749  }
750  foundName:
751 
752  Function::Ptr function = Function::instance(pltEntryVa, name, SgAsmFunction::FUNC_IMPORT);
753  if (insertUnique(functions, function, sortFunctionsByAddress))
754  ++nInserted;
755 
756  // FIXME[Robb P. Matzke 2014-08-23]: we can assume that some functions don't ever return, or they return to the call
757  // site: abort, execl, execlp, execv, execvp, exit, _exit, fexecve, longjmp, __longjmp, siglongjmp.
758  }
759 
760  // The first entry of the .plt section is the call to the function that resolves dynamic links
761  if (plt.firstOffset > 0) {
762  Function::Ptr linkerTrampoline = Function::instance(plt.section->get_mappedActualVa(), "DYNAMIC_LINKER_TRAMPOLINE",
763  SgAsmFunction::FUNC_IMPORT);
764  if (insertUnique(functions, linkerTrampoline, sortFunctionsByAddress))
765  ++nInserted;
766  }
767 
768  return nInserted;
769 }

828 bool
829 isObjectFile(const boost::filesystem::path &fileName) {
830  if (!boost::filesystem::exists(fileName))
831  return false; // file doesn't exist
832  MemoryMap::Ptr map = MemoryMap::instance();
833  if (0 == map->insertFile(fileName.string(), 0))
834  return false; // file cannot be mmap'd
835 
836  uint8_t magic[4];
837  if (map->at(0).limit(4).read(magic).size() != 4)
838  return false; // file is too short
839  if (magic[0] != 0x7f || magic[1] != 'E' || magic[2] != 'L' || magic[3] != 'F')
840  return false; // wrong magic number
841 
842  uint8_t encoding = 0;
843  if (map->at(5).limit(1).read(&encoding).size() != 1)
844  return false; // file is too short
845 
846  uint8_t elfTypeBuf[2];
847  if (map->at(16).limit(2).read(elfTypeBuf).size() != 2)
848  return false; // file is too short
849 
850  unsigned elfType = 0;
851  switch (encoding) {
852  case 1: // little endian
853  elfType = ((unsigned)elfTypeBuf[1] << 8) | elfTypeBuf[0];
854  break;
855  case 2: // big endian
856  elfType = ((unsigned)elfTypeBuf[0] << 8) | elfTypeBuf[1];
857  break;
858  default:
859  return false; // invalid data encoding
860  }
861 
862  return elfType == SgAsmElfFileHeader::ET_REL;
863 }

878 bool
879 tryLink(const std::string &commandTemplate, const boost::filesystem::path &outputName,
880  std::vector<boost::filesystem::path> inputNames, Sawyer::Message::Stream &errors,
881  FixUndefinedSymbols::Boolean fixUndefinedSymbols) {
882  if (commandTemplate.empty() || outputName.empty() || inputNames.empty())
883  return false;
884 
885  std::string allInputs;
886  for (const boost::filesystem::path &input: inputNames)
887  allInputs += (allInputs.empty()?"":" ") + StringUtility::bourneEscape(input.string());
888 
889  std::string cmd;
890  bool escaped = false;
891  for (size_t i=0; i<commandTemplate.size(); ++i) {
892  if ('\\' == commandTemplate[i]) {
893  escaped = !escaped;
894  cmd += commandTemplate[i];
895  } else if (escaped) {
896  cmd += commandTemplate[i];
897  } else if ('%' == commandTemplate[i] && i+1 < commandTemplate.size() && 'o' == commandTemplate[i+1]) {
898  cmd += StringUtility::bourneEscape(outputName.string());
899  ++i; // skip the "o"
900  } else if ('%' == commandTemplate[i] && i+1 < commandTemplate.size() && 'f' == commandTemplate[i+1]) {
901  cmd += allInputs;
902  ++i; // skip the "f"
903  } else {
904  cmd += commandTemplate[i];
905  }
906  }

908  struct Resources {
909  FILE *ldOutput;
910  char *line;
911 
912  Resources()
913  : ldOutput(NULL), line(NULL) {}
914  ~Resources() {
915  if (ldOutput)
916  pclose(ldOutput);
917  if (line)
918  free(line);
919  }
920  } r;
921 
922  // Run the linker the first time and parse the error messages looking for complaints of undefined symbols.
923  cmd += " 2>&1";
924  mlog[DEBUG] <<"running command: " <<cmd <<"\n";
925  r.ldOutput = popen(cmd.c_str(), "r");
926  if (!r.ldOutput)
927  return false;
928  std::set<std::string> undefinedSymbols;
929  boost::regex undefReferenceRe("undefined reference to `(.*)'");
930  size_t lineSize = 0;
931  while (rose_getline(&r.line, &lineSize, r.ldOutput) > 0) {
932  std::string line = r.line;
933  mlog[DEBUG] <<"command output: " <<line;
934  boost::smatch matched;
935  if (boost::regex_search(line, matched, undefReferenceRe))
936  undefinedSymbols.insert(matched.str(1));
937  }
938  int exitStatus = pclose(r.ldOutput);
939  r.ldOutput = NULL;
940  if (0 == exitStatus || undefinedSymbols.empty() || !fixUndefinedSymbols)
941  return 0 == exitStatus;
942 
943  // Create a .c file that defines those symbols that the linker complained about
944  Sawyer::FileSystem::TemporaryFile cFile((boost::filesystem::temp_directory_path() /
945  boost::filesystem::unique_path()).string() + ".c");
946  mlog[DEBUG] <<"defining symbols in a C file\n";
947  for (const std::string &symbol: undefinedSymbols) {
948  cFile.stream() <<"void " <<symbol <<"() {}\n";
949  mlog[DEBUG] <<" void " <<symbol <<"() {}\n";
950  }
951  cFile.stream().close();
952 
953  // Compile the .c file to get an object file
954  Sawyer::FileSystem::TemporaryFile oFile((boost::filesystem::temp_directory_path() /
955  boost::filesystem::unique_path()).string() + ".o");
956  cmd = "cc -o " + oFile.name().string() + " -c " + cFile.name().string();
957  mlog[DEBUG] <<"running command: " <<cmd <<"\n";
958  if (system(cmd.c_str()) != 0)
959  return false;
960 
961  // Try the linking again, but with the object file as well
962  inputNames.insert(inputNames.begin(), oFile.name().string());
963  return tryLink(commandTemplate, outputName, inputNames, errors, FixUndefinedSymbols::NO);
964 }
