34 // The variables used (U) but not internally declared (L), and declared within the function (not globally declared)
35 // should be passed as parameters to the outlined function
36 //
37 // It classifies variables used in s as the following categories
38 // * U: all used (referenced) variables within 's'
39 // * L: locally (internally to be accurate) declared within the code block 's' to be outlined
40 // * Q: declared within the enclosing function surrounding 's'
41 // but not globally declared beyond the function surrounding 's' (global variables should not be passed if the outlined function is put within the same file )
42 //

49  ROSE_ASSERT (outer_func_s);

51  // U = {symbols used within 's'}
52  ASTtools::VarSymSet_t U;
53  ASTtools::collectRefdVarSyms (s, U);
