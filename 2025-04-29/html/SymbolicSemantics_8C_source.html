158 Sawyer::Optional<BaseSemantics::SValue::Ptr>
159 SValue::createOptionalMerge(const BaseSemantics::SValue::Ptr &other_, const BaseSemantics::Merger::Ptr &merger_,
160  const SmtSolverPtr &solver) const {
161  SValue::Ptr other = SValue::promote(other_);
162  ASSERT_require(nBits() == other->nBits());
163  Merger::Ptr merger = as<Merger>(merger_);
164  bool changed = false;
165  unsigned mergedFlags = get_expression()->flags() | other->get_expression()->flags();
166  SValue::Ptr retval = SValue::promote(copy());
167  retval->set_expression(retval->get_expression()->newFlags(mergedFlags));
168 
169  // If one or the other is BOTTOM then the return value should be BOTTOM. Some simplifications (like "xor x x => 0") might
170  // cause the BOTTOM flag to be removed, and we can't have that, so make sure BOTTOM stays set!
171  if (isBottom())
172  return Sawyer::Nothing(); // no change
173  if (other->isBottom())
174  return bottom_(nBits());
175 
176  // If the values are different types then return bottom.
177  if (get_expression()->type() != other->get_expression()->type())
178  return bottom_(nBits());
179 
180  // Merge symbolic expressions. The merge of x and y is the set {x, y}. If the size of this set is greater than the set size
181  // limit (or 1 if merger is null) then the result is bottom. Normal set simplifcations happen first (e.g., {x, x} => {x}
182  // => x).
183  if (!get_expression()->mustEqual(other->get_expression(), solver)) {
184  ExprPtr expr = SymbolicExpression::makeSet(get_expression(), other->get_expression(), solver, "", mergedFlags);
185  size_t exprSetSize = expr->isInteriorNode() && expr->isInteriorNode()->getOperator()==SymbolicExpression::OP_SET ?
186  expr->isInteriorNode()->nChildren() : (size_t)1;
187  size_t setSizeLimit = merger ? merger->setSizeLimit() : (size_t)1;
188  if (exprSetSize > setSizeLimit) {
189  expr = SymbolicExpression::makeVariable(retval->get_expression()->type());
190  mergedFlags |= SymbolicExpression::Node::BOTTOM;
191  }
192  retval->set_expression(expr);
193  changed = true;
194  }
195 
196  // Merge flags.
197  if (get_expression()->flags() != mergedFlags) {
198  retval->set_expression(retval->get_expression()->newFlags(mergedFlags));
199  changed = true;
200  }
201 
202  // Merge definers
203  InsnSet mergedDefiners = retval->get_defining_instructions();
204  InsnSet otherDefiners = other->get_defining_instructions();
205  for (SgAsmInstruction *definer: otherDefiners) {
206  if (mergedDefiners.insert(definer).second)
207  changed = true;
208  }
209  retval->add_defining_instructions(mergedDefiners);
210 
211  return changed ? Sawyer::Optional<BaseSemantics::SValue::Ptr>(retval) : Sawyer::Nothing();
212 }

605 BaseSemantics::SValue::Ptr
606 MemoryListState::readOrPeekMemory(const BaseSemantics::SValue::Ptr &address_, const BaseSemantics::SValue::Ptr &dflt,
607  BaseSemantics::RiscOperators *addrOps, BaseSemantics::RiscOperators *valOps,
608  AllowSideEffects::Flag allowSideEffects) {
609  size_t nBits = dflt->nBits();
610  SValue::Ptr address = SValue::promote(address_);
611  ASSERT_require(8==nBits); // SymbolicSemantics::MemoryListState assumes that memory cells contain only 8-bit data
612 
613  BaseSemantics::CellList::iterator cursor = get_cells().begin();
614  BaseSemantics::CellList cells = scan(cursor /*in,out*/, address, nBits, addrOps, valOps);
615 
616  // If we fell off the end of the list then the read could be reading from a memory location for which no cell exists. If
617  // side effects are allowed, we should add a new cell to the return value.
618  if (cursor == get_cells().end()) {
619  if (AllowSideEffects::YES == allowSideEffects) {
620  BaseSemantics::MemoryCell::Ptr newCell = insertReadCell(address, dflt);
621  cells.push_back(newCell);
622  } else {
623  BaseSemantics::MemoryCell::Ptr newCell = protocell->create(address, dflt);
624  cells.push_back(newCell);
625  }
626  }
627 
628  // If we're doing an actual read (rather than just a peek), then update the returned cells to indicate that they've been
629  // read. Since the "cells" vector is pointers that haven't been deep-copied, this is a side effect on the memory
630  // state. But even if it weren't a side effect, we don't want the returned value to be marked as having been actually
631  // read when we're only peeking.
632  if (AllowSideEffects::YES == allowSideEffects)
633  updateReadProperties(cells);
634 
635  SValue::Ptr retval = cellCompressor()->operator()(address, dflt, addrOps, valOps, cells);
636  ASSERT_require(retval->nBits()==8);
637  return retval;
638 }

837 RiscOperators::substitute(const SValue::Ptr &from, const SValue::Ptr &to)
838 {
839  BaseSemantics::State::Ptr state = currentState();
840 
841  // Substitute in registers
842  struct RegSubst: RegisterState::Visitor {
843  SValue::Ptr from, to;
844  SmtSolverPtr solver;
845  RegSubst(const SValue::Ptr &from, const SValue::Ptr &to, const SmtSolverPtr &solver)
846  : from(from), to(to), solver(solver) {}
847  virtual BaseSemantics::SValue::Ptr operator()(RegisterDescriptor, const BaseSemantics::SValue::Ptr &val_) {
848  SValue::Ptr val = SValue::promote(val_);

850  }

852  RegisterState::promote(state->registerState())->traverse(regsubst);

854  // Substitute in memory
855  struct MemSubst: BaseSemantics::MemoryCell::Visitor {
856  SValue::Ptr from, to;
857  SmtSolverPtr solver;

859  : from(from), to(to), solver(solver) {}
860  virtual void operator()(BaseSemantics::MemoryCell::Ptr &cell) {
861  SValue::Ptr addr = SValue::promote(cell->address());
862  cell->address(addr->substitute(from, to, solver));
863  SValue::Ptr val = SValue::promote(cell->value());
864  cell->value(val->substitute(from, to, solver));
865  }
866  } memsubst(from, to, solver());
867  MemoryState::promote(state->memoryState())->traverse(memsubst);
868 }

995 RiscOperators::extract(const BaseSemantics::SValue::Ptr &a_, const size_t begin_bit, const size_t end_bit)
996 {
997  SValue::Ptr a = SValue::promote(a_);
998  ASSERT_require(end_bit<=a->nBits());
999  ASSERT_require(begin_bit<end_bit);
1000  SymbolicExpression::Ptr beginExpr = SymbolicExpression::makeIntegerConstant(32, begin_bit);
1001  SymbolicExpression::Ptr endExpr = SymbolicExpression::makeIntegerConstant(32, end_bit);
1002 
1003  if (a->isBottom()) {
1004  return filterResult(svalueExpr(SymbolicExpression::makeExtract(beginExpr, endExpr, a->get_expression(), solver(),
1005  "", SymbolicExpression::Node::BOTTOM)));
1006  }
1007 
1008  SValue::Ptr retval = svalueExpr(SymbolicExpression::makeExtract(beginExpr, endExpr, a->get_expression(), solver()));
1009  switch (computingDefiners_) {
1010  case TRACK_NO_DEFINERS:
1011  if (retval->nBits() == a->nBits())
1012  retval->add_defining_instructions(a); // preserve definers if this extract is a no-op
1013  break;
1014  case TRACK_ALL_DEFINERS:
1015  retval->add_defining_instructions(a); // old definers and...
1016  if (retval->nBits() != a->nBits()) // ...new definer but only if this extract is not a no-op
1017  retval->add_defining_instructions(omit_cur_insn ? NULL : currentInstruction());
1018  break;
1019  case TRACK_LATEST_DEFINER:
1020  if (retval->nBits() != a->nBits()) {
1021  retval->add_defining_instructions(omit_cur_insn ? NULL : currentInstruction());
1022  } else {
1023  retval->add_defining_instructions(a); // preserve definers if this extract is a no-op
1024  }
1025  break;
1026  }
1027  return filterResult(retval);
1028 }

1074 BaseSemantics::SValue::Ptr
1075 RiscOperators::iteWithStatus(const BaseSemantics::SValue::Ptr &sel_,
1076  const BaseSemantics::SValue::Ptr &a_, const BaseSemantics::SValue::Ptr &b_,
1077  IteStatus &status /*out*/)
1078 {
1079  SValue::Ptr sel = SValue::promote(sel_);
1080  SValue::Ptr a = SValue::promote(a_);
1081  SValue::Ptr b = SValue::promote(b_);
1082  ASSERT_require(1==sel->nBits());
1083  ASSERT_require(a->nBits()==b->nBits());
1084 
1085  // (ite bottom A B) should be A when A==B. However, SymbolicExpression would have already simplified that to A.
1086  if (sel->isBottom()) {
1087  status = IteStatus::NEITHER;
1088  return filterResult(bottom_(a->nBits()));
1089  }
1090 
1091  SValue::Ptr retval;
1092  if (auto selNum = sel->toUnsigned()) {
1093  retval = SValue::promote(*selNum ? a->copy() : b->copy());
1094  status = *selNum ? IteStatus::A : IteStatus::B;
1095  switch (computingDefiners_) {
1096  case TRACK_NO_DEFINERS:
1097  break;
1098  case TRACK_ALL_DEFINERS:
1099  retval->add_defining_instructions(sel); // fall through...
1100  case TRACK_LATEST_DEFINER:
1101  retval->add_defining_instructions(omit_cur_insn ? NULL : currentInstruction());
1102  break;
1103  }
1104  return filterResult(retval);
1105  }
1106  if (solver()) {
1107  // If the selection expression cannot be true, then return b
1108  {
1109  SmtSolver::Transaction transaction(solver());
1110  ExprPtr condition = sel->get_expression();
1111  solver()->insert(condition);
1112  bool can_be_true = SmtSolver::SAT_NO != solver()->check();
1113  if (!can_be_true) {
1114  retval = SValue::promote(b->copy());
1115  status = IteStatus::B;
1116  switch (computingDefiners_) {
1117  case TRACK_NO_DEFINERS:
1118  break;
1119  case TRACK_ALL_DEFINERS:
1120  retval->add_defining_instructions(sel); // fall through...
1121  case TRACK_LATEST_DEFINER:
1122  retval->add_defining_instructions(omit_cur_insn ? NULL : currentInstruction());
1123  break;
1124  }
1125  return filterResult(retval);
1126  }
1127  }
1128 
1129  // If the selection expression cannot be false, then return a
1130  {
1131  SmtSolver::Transaction transaction(solver());
1132  ExprPtr inverseCondition = SymbolicExpression::makeInvert(sel->get_expression(), solver());
1133  solver()->insert(inverseCondition);
1134  bool can_be_false = SmtSolver::SAT_NO != solver()->check();
1135  if (!can_be_false) {
1136  retval = SValue::promote(a->copy());
1137  status = IteStatus::A;
1138  switch (computingDefiners_) {
1139  case TRACK_NO_DEFINERS:
1140  break;
1141  case TRACK_ALL_DEFINERS:
1142  retval->add_defining_instructions(sel); // fall through...
1143  case TRACK_LATEST_DEFINER:
1144  retval->add_defining_instructions(omit_cur_insn ? NULL : currentInstruction());
1145  break;
1146  }
1147  return filterResult(retval);
1148  }
1149  }
1150  }
1151 
1152  retval = svalueExpr(SymbolicExpression::makeIte(sel->get_expression(), a->get_expression(), b->get_expression(), solver()));
1153  status = IteStatus::BOTH;
1154  switch (computingDefiners_) {
1155  case TRACK_NO_DEFINERS:
1156  break;
1157  case TRACK_ALL_DEFINERS:
1158  retval->add_defining_instructions(a);
1159  retval->add_defining_instructions(b);
1160  retval->add_defining_instructions(sel); // fall through...
1161  case TRACK_LATEST_DEFINER:
1162  retval->add_defining_instructions(omit_cur_insn ? NULL : currentInstruction());
1163  break;
1164  }
1165  return filterResult(retval);
1166 }

1561 SgAsmFloatType*
1562 RiscOperators::sgIsIeee754(SgAsmType *sgType) {
1563  ASSERT_not_null(sgType);
1564  SgAsmFloatType *fpType = isSgAsmFloatType(sgType);
1565  if (NULL == fpType)
1566  return NULL;
1567  if (fpType->significandBits().least() != 0)
1568  return NULL;
1569  if (fpType->significandBits().size() == 0)
1570  return NULL;
1571  if (fpType->exponentBits().size() < 2)
1572  return NULL;
1573  if (fpType->exponentBias() != ((uint64_t)1 << (fpType->exponentBits().size() - 1)) - 1)
1574  return NULL;
1575 
1576  // The sign, exponent, and significand regions must not overlap, although IEEE 754 allows unused regions. For instance, the
1577  // Motorola 68000 family has an "extended real" type that's 96 bits but 17 bits are unused (the format is 63-bit
1578  // significand, a bit that's always set (i.e., the explicit leading one bit for the significand), 16 bits that are always
1579  // clear, a 15-bit exponent, and a sign bit).
1580  if (fpType->significandBits().overlaps(fpType->exponentBits()))
1581  return NULL;
1582  if (fpType->significandBits().overlaps(fpType->signBit()))
1583  return NULL;
1584  if (fpType->exponentBits().overlaps(fpType->signBit()))
1585  return NULL;
1586 
1587  return fpType;
1588 }

1755  BaseSemantics::SValue::Ptr byte_addr = add(adjustedVa, number_(adjustedVa->nBits(), bytenum));

1759  // initial memory it will be added.
1760  if (initialState()) {

1762  byte_dflt = initialState()->readMemory(byte_addr, byte_dflt, this, this);
1763  } else {

1765  }
1766  }
1767

1769  SValue::Ptr byte_value;
1770  if (AllowSideEffects::YES == allowSideEffects) {
1771  byte_value = SValue::promote(currentState()->readMemory(byte_addr, byte_dflt, this, this));
1772  } else {
1773  byte_value = SValue::promote(currentState()->peekMemory(byte_addr, byte_dflt, this, this));
1774  }
1775  if (0==bytenum) {
1776  retval = byte_value;
1777  } else if (ByteOrder::ORDER_MSB==currentMem->get_byteOrder()) {
1778  retval = SValue::promote(concat(byte_value, retval));
1779  } else if (ByteOrder::ORDER_LSB==currentMem->get_byteOrder()) {
1780  retval = SValue::promote(concat(retval, byte_value));
1781  } else {
1782  // See BaseSemantics::MemoryState::set_byteOrder
1783  throw BaseSemantics::Exception("multi-byte read with memory having unspecified byte order", currentInstruction());
1784  }
1785 
1786  // Accumulating the bytes of a multibyte read is sort of like a merge operation in the way it treats definers. Since
1787  // we're returning a value that's composed of multiple bytes, the latest definers for the reutrn value are the union of
1788  // the latest definers for the individual bytes. Similarly, for TRACK_ALL_DEFINERS, the set of all definers for the
1789  // return value is the union of the all-definers sets of the individual bytes.
1790  if (computingDefiners() == TRACK_ALL_DEFINERS || computingDefiners() == TRACK_LATEST_DEFINER) {
1791  const InsnSet &byteDefiners = byte_value->get_defining_instructions();
1792  allDefiners.insert(byteDefiners.begin(), byteDefiners.end());
1793  }
1794  }
1795 
1796  ASSERT_require(retval!=NULL && retval->nBits()==nbits);
1797  switch (computingDefiners_) {
1798  case TRACK_NO_DEFINERS:
1799  break;
1800  case TRACK_ALL_DEFINERS:
1801  case TRACK_LATEST_DEFINER:
1802  retval->add_defining_instructions(omit_cur_insn ? NULL : currentInstruction());
1803  retval->add_defining_instructions(allDefiners);
1804  break;
1805  }
1806 
1807  if (reinterpretMemoryReads_)
1808  retval = SValue::promote(reinterpret(retval, SageBuilderAsm::buildTypeU(retval->nBits())));
1809 
1810  return filterResult(retval);
1811 }
