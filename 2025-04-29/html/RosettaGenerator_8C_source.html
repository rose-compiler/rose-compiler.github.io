337 // Generate property data member
338 void
339 RosettaGenerator::genPropertyDataMember(std::ostream &rosetta, std::ostream &header, const Ast::Property::Ptr &p) {
340  ASSERT_not_null(p);
341  auto c = p->findAncestor<Ast::Class>();
342  ASSERT_not_null(c);
343 
344  // Data memeber.
345  // NO_CONSTRUCTOR_PARAMETER.
346  // This is not used because we've turned off ROSETTA's automatic generation of constructors and destructors.
347  // NO_ACCESS_FUNCTIONS | BUILD_ACCESS_FUNCTIONS
348  // We disable ROSETTA's automatic generation of data member access and mutator functions because we can easily generate our
349  // own and have more control over them. For instance, ROSETTA distinguishes between passing arguments and return values by
350  // value (BUILD_ACCESS_FUNCTIONS) or reference (BUILD_LIST_ACCESS_FUNCTIONS), which is unecessary since passing by reference
351  // is more general than passing by value (therefore Rosebud always uses references). Rosebud also has more flexibility in
352  // choosing the accessor and mutator names, using either ROSETTA's "get_"/"set_" prefixes or just the property name by
353  // itself.
354  // NO_TRAVERSAL | DEF_TRAVERSAL
355  // The common case is NO_TRAVERSAL; the uncommon case is caused when the Rosebud::traverse attribute is present.
356  // NO_DELETE
357  // This controls whether ROSETTA's generated destructors delete data members. However, it's always possible to define
358  // data members in such a way that they get deleted automatically by the default destructor, so Rosetta always uses
359  // NO_DELETE, does not generate `delete` calls in the destructor, and leaves it up to the node authors. For instance,
360  // if a node points to something that should be deleted, then the type for that member should be a smart pointer that
361  // indicates that the node owns the pointed-to data. Furthermore, we've disabled ROSETTA's generation of destructors.
362  // COPY_DATA | CLONE_PTR
363  // This controls whether ROSETTA's generated copy mechanism (not copy constructors) copy the data member's value from
364  // the source node to the destination node. We assume that all data members should be copied unless an attribute such
365  // as Rosetta::cloneptr is present.
366  if (p->findAttribute("Rosebud::rosetta")) {
367  rosetta <<"\n"
368  <<THIS_LOCATION <<"#ifndef DOCUMENTATION\n";
369  rosetta <<locationDirective(p, p->startToken) <<" " <<shortName(c) <<".setDataPrototype(\n"
370  <<" \"" <<propertyDataMemberType(p) <<"\", \"" <<p->name <<"\", \"";
371  const std::string initExpr = ctorInitializerExpression(p, initialValue(p));
372  if (!initExpr.empty())
373  rosetta <<"= " <<initExpr; // rosetta wants contructor initializer expressions even though we have an "=" here
374  rosetta <<"\",\n"
375  <<" NO_CONSTRUCTOR_PARAMETER, NO_ACCESS_FUNCTIONS, "
376  <<(p->findAttribute("Rosebud::traverse") ? "DEF_TRAVERSAL" : "NO_TRAVERSAL") <<", "
377  <<"NO_DELETE, "
378  <<(p->findAttribute("Rosebud::cloneptr") ? "CLONE_PTR" : "COPY_DATA")
379  <<");\n"
380  <<THIS_LOCATION <<"#endif // !DOCUMENTATION\n";
381  } else {
382  header <<"\n"
383  <<THIS_LOCATION <<"private:\n"
384  <<" " <<propertyDataMemberType(p) <<" " <<propertyDataMemberName(p) <<";\n";
385  }
386 }

446 // * rosetta: the stuff that will be compiled into ROSETTA's CxxGrammarMetaProgram.
447 // * header: the stuff to pass through ROSETTA directly into the class definition header file.

449 void
450 RosettaGenerator::genProperty(std::ostream &rosetta, std::ostream &header, std::ostream &impl, const Ast::Property::Ptr &p) {

453  ASSERT_not_null(c);
454 
455  p->cppStack->emitOpen(rosetta);
456 
457  // Fix up the doxygen comment so it can apply to more than one class member. If the doxygen comment doesn't end with "@{" then
458  // we need to add that.
459  const auto doc = [&p]() -> std::pair<std::string, std::string> {
460  std::string begin = p->doc;
461  std::string end;
462  if (!begin.empty()) {
463  if (begin.find("@{") == std::string::npos) {
464  begin = appendToDoxygen(begin, "\n@{");
465  end = " /** @} */\n";
466  }
467  if (!boost::ends_with(begin, "\n"))
468  begin += "\n";
469  }
470  return {begin, end};
471  }();
472 
473  if (auto attribute = p->findAttribute("Rosebud::not_null")) {
474  message(WARN, attribute->findAncestor<Ast::File>(), attribute->nameTokens.front(),
475  "the \"Rosebud::not_null\" attribute is not supported by the ROSETTA code generator");
476  }
477 
478  genPropertyDataMember(rosetta, header, p);
479 
480  // Accessors and mutators
481  header <<"\n"
482  <<THIS_LOCATION <<"public:\n"
483  <<locationDirective(p, p->docToken) <<doc.first;
484  genPropertyAccessors(header, impl, p);
485  genPropertyMutators(header, impl, p);

518 // Emit call to NEW_NONTERMINAL_MACRO macro
519 void
520 RosettaGenerator::genNewNonterminalMacro(std::ostream &rosetta, const Ast::Class::Ptr &c, const Hierarchy &h) {
521  ASSERT_not_null(c);
522 
523  // We would normally use the NEW_NONTERMINAL_MACRO macro defined in ROSETTA_macros.h here, but that macro makes it difficult to
524  // handle a list of subclass names where some of the subclasses are conditionally compiled. Especially when the first one is
525  // conditionally compiled.
526  rosetta <<THIS_LOCATION <<"AstNodeClass& " <<shortName(c) <<" = nonTerminalConstructor(\n"
527  <<" \"" <<shortName(c) <<"\",\n"
528  <<" *this,\n"
529  <<" \"" <<shortName(c) <<"\",\n"
530  <<" \"" <<shortName(c) <<"Tag\",\n"
531  <<" SubclassListBuilder()\n";
532 
533  auto vertex = h.findVertexKey(c);
534  ASSERT_require(vertex != h.vertices().end());
535  for (auto subclassEdge: vertex->outEdges()) {
536  Ast::Class::Ptr subclass = subclassEdge.target()->value();
537  subclass->cppStack->emitOpen(rosetta);
538  rosetta <<" | " <<shortName(subclass) <<"\n";
539  subclass->cppStack->emitClose(rosetta);
540  }
541  if (c->findAttribute("Rosebud::abstract")) {
542  rosetta <<" , false);\n";
543  } else {
544  rosetta <<" , true);\n";
545  }
546 
547  rosetta <<THIS_LOCATION <<"assert(" <<shortName(c) <<".associatedGrammar != nullptr);\n";
548 }

620 RosettaGenerator::isBaseClass(const Ast::Class::Ptr &c, const Hierarchy &h) {

637 void
638 RosettaGenerator::genClassDefinition(std::ostream &rosetta, const Ast::Class::Ptr &c, const Hierarchy &h) {
639  ASSERT_not_null(c);
640  if (c->findAttribute("Rosebud::suppress"))
641  return;
642 
643  std::ostringstream header; // Non-ROSETTA class definition stuff
644 
645  // Open the output stream for the generated function implementations
646  const std::string implFileName = implDirectoryName.empty() ?
647  "/dev/null" :
648  (implDirectoryName / (c->name + ".C")).string();
649  ASSERT_forbid(implFileName.empty());
650  std::ofstream impl(implFileName, std::ios_base::trunc);
651  if (!impl) {
652  message(ERROR, "unable to open implementation output file \"" + std::string(implFileName) + "\"");
653  return;
654  }
655  genImplFileBegin(impl, c);
656 
657  // Title comment
658  rosetta <<"\n\n"
659  <<THIS_LOCATION <<makeTitleComment(c->name + " -- MACHINE GENERATED; DO NOT MODIFY --", "", '/', outputWidth)
660  <<"\n";
661 
662  // Emit the conditional compilation for the class as a whole
663  c->cppStack->emitOpen(rosetta);
664 
665  // Emit the ROSETTA-specific macros for the class
666  if (isBaseClass(c, h)) {
667  genNonterminalMacros(rosetta, c, h);
668  } else {
669  genLeafMacros(rosetta, c);
670  }
671 
672  // Small header classes are classes whose definition ROSETTA will emit to its own header.
673  if (smallHeaders || c->findAttribute("Rosebud::small_header")) {
674  rosetta <<"\n"
675  <<THIS_LOCATION <<"#ifndef DOCUMENTATION\n"
676  <<shortName(c) <<".useSmallHeader(true);\n"
677  <<"#endif // !DOCUMENTATION\n";
678  }
679 
680  // ROSETTA "headers" for the class. This is usually just #include statements, but it can be anything
681  // needed before the ROSETTA-generated class definition.
682  if (!c->priorText.empty()) {
683  static const std::regex implRe("#\\s*(ifdef\\s+ROSE_IMPL|if\\s+defined\\s*\\(\\s*ROSE_IMPL\\s*\\))");
684  const std::string cppImplSymbol = "ROSE_" + c->name + "_IMPL";
685  const std::string s = std::regex_replace(c->priorText, implRe, "#ifdef " + cppImplSymbol);
686  rosetta <<"\n"
687  <<THIS_LOCATION <<"DECLARE_HEADERS(" <<shortName(c) <<");\n"
688  <<"#if defined(" <<c->name <<"_HEADERS) || defined(DOCUMENTATION)\n"
689  <<locationDirective(c, c->priorTextToken) <<s <<"\n";
690  c->cppStack->emitClose(rosetta);
691  rosetta <<"#endif // " <<c->name <<"_HEADERS\n";
692  }
693 
694  genClassBegin(rosetta, c);
695 
696  size_t nPragmas = 0;
697  for (const auto &p: c->properties) {
698  nPragmas += genRosettaPragmas(rosetta, c, p());
699 
700  // Stuff in the input that's prior to the property definition should go in the "other" section of output
701  header <<locationDirective(p(), p->priorTextToken) <<p->priorText;
702 
703  // Emit the property definition, and accumulate anything that should go in the "other" section of output.
704  genProperty(rosetta, header, impl, p());
705  }
706 
707  // If anything else is in the input class definition after the last property definition, append it to the "other" section of
708  // output
709  nPragmas += genRosettaPragmas(rosetta, c);
710  header <<locationDirective(c, c->endTextToken) <<c->endText;
711 
712  // Class constructors and destructures emitted quite late in the class definition so we're sure that all the types needed by
713  // their arguments are already emitted.
714  if (!c->findAttribute("Rosebud::no_desctructor"))
715  genDestructor(header, impl, c);
716  if (!c->findAttribute("Rosebud::no_constructors"))
717  genClassConstructors(header, impl, c, h);
718  genInitProperties(header, impl, c);
719 
720  // ROSETTA "others" is everything else in the class that's not generated by making ROSETTA function calls.
721  genOtherContent(rosetta, c, h, header.str());
722  genClassEnd(rosetta, c);
723 
724  c->cppStack->emitClose(rosetta);
725  genImplFileEnd(impl, c);
726 
727  // Delayed warnings
728  if (nPragmas > 0 && settings.showingWarnings) {
729  message(WARN, c->findAncestor<Ast::File>(), c->nameToken,
730  "class " + c->name + " contains " + boost::lexical_cast<std::string>(nPragmas) + " pragma " +
731  std::string(1 == nPragmas ? "directive" : "directives") + " which should be used only as a last resort to "
732  "achieve ROSETTA compatibility. Most of the things done with these pragmas can be done more effectively and "
733  "with less generated code by using C++ features like dynamic dispatch, template metaprogramming and "
734  "introspection. The presence of ROSETTA-specific pragmas will likely hinder our ability to write other code "
735  "generators using the Rosebud framework.");
736  }
737 }

930 void
931 RosettaGenerator::generate(const Ast::Project::Ptr &project) {
932  ASSERT_not_null(project);
933 
934  Hierarchy h = classHierarchy(project->allClassesFileOrder());
935  checkClassHierarchy(h);
936  const Classes classes = bottomUp(h);
937 
938  // Check that we have all the classes we need
939  for (const Ast::Class::Ptr &c: classes) {
940  std::string sageBase;
941  for (const auto &base: c->inheritance) {
942  if (base.second.size() > 2 && boost::starts_with(base.second, "Sg")) {
943  if (!sageBase.empty()) {
944  message(ERROR, c->findAncestor<Ast::File>(), c->nameToken,
945  "class \"" + c->name + "\" has more than one Sage node as a base class: "
946  "\"" + sageBase + "\" and \"" + base.second + "\"");
947  } else {
948  sageBase = base.second;
949  }
950 
951  const auto baseVertex = h.findVertexKey(base.second);
952  if (baseVertex == h.vertices().end()) {
953  // FIXME[Robb Matzke 2023-03-20]: points to the class name, but should point to the base class
954  // name. Unfortunately we didn't save that information when parsing.
955  if (settings.showingWarnings) {
956  message(WARN, c->findAncestor<Ast::File>(), c->nameToken,
957  "class \"" + c->name + "\" derives from class \"" + base.second + "\""
958  " whose definition is not known to Rosebud (did you intend to supply the base class's header file "
959  "on the rosebud command-line?)");
960  }
961  }
962  }
963  }
964  }
965 
966  // Open the stream to which we write ROSETTA input
967  std::ofstream rosettaFile;
968  if ("-" != rosettaFileName) {
969 #ifdef _MSC_VER
970  if (rosettaFileName.empty()) {
971  message(FATAL, "ROSETTA output file must be specified on Windows");
972  exit(1);
973  } else {
974  rosettaFile.open(rosettaFileName.c_str());
975  }
976 #else
977  rosettaFile.open(rosettaFileName.empty() ? "/dev/null" : rosettaFileName.c_str());
978 #endif
979  }
980  std::ostream &rosetta = "-" == rosettaFileName ? std::cout : rosettaFile;
981  if (!rosetta) {
982  message(ERROR, "cannot open output file \"" + rosettaFileName.string() + "\"");
983  return;
984  }
985 
986  // Generate code
987  genRosettaFileBegin(rosetta); {
988  // ROSETTA files define classes from the bottom up because the base class definitions depend on the subclasses instead
989  // of the other way around.
990  genClassDeclarations(rosetta, classes);
991 
992  genRosettaFunctionBegin(rosetta); {
993  for (auto c: classes)
994  genClassDefinition(rosetta, c, h);
995  } genRosettaFunctionEnd(rosetta);
996  } genRosettaFileEnd(rosetta);
997 
998  // Generate build system files
999  const std::vector<std::string> implFileNames = implementationFileNames(classes);
1000  adjustNodeList(project);
1001  genTupFile(implFileNames);
1002  genMakeFile(implFileNames);
1003  genCmakeFile(implFileNames);
1004 }
