71 // Find padding that appears before the entry address of a function that aligns the entry address on a 4-byte boundary.
72 // For m68k, padding is either 2-byte TRAPF instructions (0x51 0xfc) or zero bytes. Patterns we've seen are 51 fc, 51 fc 00
73 // 51 fc, 00 00, 51 fc 51 fc, but we'll allow any combination.
74 Address
75 MatchFunctionPadding::match(const Partitioner::ConstPtr &partitioner, Address anchor) {
76  ASSERT_not_null(partitioner);
77  MemoryMap::Ptr m = partitioner->memoryMap();
78  if (0==anchor)
79  return anchor;
80 
81  // Read backward from the anchor, skipping over padding as we go
82  Address padMin = anchor;
83  uint8_t buf[2]; // reading two bytes at a time
84  while (AddressInterval accessed = m->at(padMin-1).limit(2).require(MemoryMap::EXECUTABLE)
85  .read(buf, Sawyer::Container::MATCH_BACKWARD)) {
86 
87  // Match zero byte or (0x51 0xfc) pair
88  AddressInterval matched;
89  if (2==accessed.size() && (0x51==buf[0] && 0xfc==buf[1])) {
90  matched = accessed;
91  } else if (2==accessed.size() && 0==buf[1]) {
92  matched = AddressInterval::baseSize(accessed.least()+1, 1);
93  } else if (1==accessed.size() && 0==buf[0]) {
94  matched = accessed;
95  } else {
96  break;
97  }
98 
99  // Make sure that what we matched is not already part of some other instruction
100  if (!partitioner->instructionsOverlapping(matched).empty())
101  break;
102 
103  // This match appears to be valid padding
104  padMin = matched.least();
105  if (0==padMin)
106  break; // avoid overflow
107  }
108 
109  return padMin;
110 }

112 // A "switch" statement looks like this:
113 // 0x1000e062: 30 7b 1a 06 |0{.. | movea.w [pc+0x00000006+d1*0x00000002], a0
114 // 0x1000e066: 4e fb 88 02 |N... | jmp pc+0x00000002+a0*0x00000001
115 // The offset table appears immediately after the JMP (thus pc+6 in the MOVEA) and the code for the various cases starts
116 // immediately after the offset table.
117 bool
118 SwitchSuccessors::operator()(bool chain, const Args &args) {
119  ASSERT_not_null(args.bblock);
120  if (!chain)
121  return false;
122  size_t nInsns = args.bblock->nInstructions();
123  if (nInsns < 2)
124  return chain;
125  SgAsmM68kInstruction *movea = isSgAsmM68kInstruction(args.bblock->instructions()[nInsns-2]);
126  SgAsmM68kInstruction *jmp = isSgAsmM68kInstruction(args.bblock->instructions().back());
127  if (!movea || movea->get_kind()!=m68k_movea || !jmp || jmp->get_kind()!=m68k_jmp)
128  return chain;
129  const SgAsmExpressionPtrList &movea_args = movea->get_operandList()->get_operands();
130  const SgAsmExpressionPtrList &jmp_args = jmp->get_operandList()->get_operands();
131  if (movea_args.size()!=2 || jmp_args.size()!=1)
132  return chain;
133  const RegisterDescriptor REG_PC = args.partitioner->instructionProvider().instructionPointerRegister();
134 
135  // MOVEA first argument is: (+ (+ PC 6) (* Dx 2))
136  // prog variables: sum1 ---^ ^ ^^ ^ ^ ^^ ^---- scale1
137  // sum2 ------' || | | ''------ reg2
138  // reg1 --------'' | '--------- prod1
139  // offset1 --------'
140  SgAsmMemoryReferenceExpression *mre = isSgAsmMemoryReferenceExpression(movea_args[0]);
141  SgAsmBinaryAdd *sum1 = mre ? isSgAsmBinaryAdd(mre->get_address()) : NULL;
142  SgAsmBinaryAdd *sum2 = sum1 ? isSgAsmBinaryAdd(sum1->get_lhs()) : NULL;
143  SgAsmDirectRegisterExpression *reg1 = sum2 ? isSgAsmDirectRegisterExpression(sum2->get_lhs()) : NULL;
144  SgAsmIntegerValueExpression *offset1 = sum2 ? isSgAsmIntegerValueExpression(sum2->get_rhs()) : NULL;
145  SgAsmBinaryMultiply *prod1 = sum1 ? isSgAsmBinaryMultiply(sum1->get_rhs()) : NULL;
146  SgAsmDirectRegisterExpression *reg2 = prod1 ? isSgAsmDirectRegisterExpression(prod1->get_lhs()) : NULL;
147  SgAsmIntegerValueExpression *scale1 = prod1 ? isSgAsmIntegerValueExpression(prod1->get_rhs()) : NULL;
148  if (!reg1 || reg1->get_descriptor()!=REG_PC || !offset1 || offset1->get_absoluteValue()!=6 ||
149  !reg2 || !scale1 || scale1->get_absoluteValue()!=2)
150  return chain;
151 
152  // MOVEA second argument is: Ax
153  // reg3 ----------^^
154  SgAsmDirectRegisterExpression *reg3 = isSgAsmDirectRegisterExpression(movea_args[1]);
155  if (!reg3)
156  return chain;
157 
158  // JMP argument is: (+ (+ PC 2) (* Ax 1))
159  // sum3 -----^ ^ ^^ ^ ^ ^^ ^------ scale2
160  // sum4 --------' || | | ''-------- reg5
161  // reg4 ----------'' | '----------- prod2
162  // offset2 ----------'
163  SgAsmBinaryAdd *sum3 = isSgAsmBinaryAdd(jmp_args[0]);
164  SgAsmBinaryAdd *sum4 = sum3 ? isSgAsmBinaryAdd(sum3->get_lhs()) : NULL;
165  SgAsmDirectRegisterExpression *reg4 = sum4 ? isSgAsmDirectRegisterExpression(sum4->get_lhs()) : NULL;
166  SgAsmIntegerValueExpression *offset2 = sum4 ? isSgAsmIntegerValueExpression(sum4->get_rhs()) : NULL;
167  SgAsmBinaryMultiply *prod2 = sum3 ? isSgAsmBinaryMultiply(sum3->get_rhs()) : NULL;
168  SgAsmDirectRegisterExpression *reg5 = prod2 ? isSgAsmDirectRegisterExpression(prod2->get_lhs()) : NULL;
169  SgAsmIntegerValueExpression *scale2 = prod2 ? isSgAsmIntegerValueExpression(prod2->get_rhs()) : NULL;
170  if (!reg4 || reg4->get_descriptor()!=REG_PC || !offset2 || offset2->get_absoluteValue()!=2 ||
171  !reg5 || !scale2 || scale2->get_absoluteValue()!=1)
172  return chain;
173 
174  // Make sure the destination of the MOVEA is the index register for the JMP
175  if (reg3->get_descriptor() != reg5->get_descriptor())
176  return chain;
177 
178  // At this point we've matched the MOVEA/JMP pair and we know that the offset table follows the JMP instruction. Read
179  // offsets from the table until the table runs into one of our successors.
180  Address startOfOffsetTable = jmp->get_address() + jmp->get_size();
181  size_t tableIdx = 0;
182  Address leastCodeVa = (Address)(-1);
183  std::set<Address> codeVas;
184  MemoryMap::Ptr map = args.partitioner->memoryMap();
185  while (1) {
186  // Where is the offset in memory? It must be between the end of the JMP instruction (watch out for overflow) and the
187  // lowest address for a switch case.
188  Address offsetVa = startOfOffsetTable + tableIdx*2;
189  if (offsetVa < startOfOffsetTable || offsetVa+2 >= leastCodeVa)
190  break;
191  AddressInterval offsetExtent = AddressInterval::baseSize(offsetVa, 2);
192 
193  // If the memory containing the offset is already used by some other basic block then it probably isn't an offset, but
194  // rather an instruction. This sometimes terminates an offset table where the first thing after the offset table isn't
195  // a case that's pointed to by the offset table. But it will only catch those situations if we already know about such
196  // a basic block. This seems to work okay for well-formed (non-obfuscated) code when the recursive basic block
197  // discovery follows higher-address blocks before lower-address blocks.
198  if (args.partitioner->placeholderExists(offsetVa) || !args.partitioner->basicBlocksOverlapping(offsetExtent).empty())
199  break;
200 
201  // Read the offset from the offset table. Something went wrong if we can't read it because we know that the code for
202  // the switch cases follows the table.
203  uint16_t offsetBE;
204  if (2!=map->at(offsetVa).limit(2).require(MemoryMap::EXECUTABLE).read((uint8_t*)&offsetBE).size()) {
205  mlog[WARN] <<"short read entry[" <<tableIdx <<"] " <<StringUtility::addrToString(offsetVa)
206  <<" in offset table " <<StringUtility::addrToString(startOfOffsetTable) <<"\n";
207  break;
208  }
209 
210  // M68k offsets are 16-bit big-endian that are sign extended to 32 bits
211  Address offset = ByteOrder::beToHost(offsetBE);
212  offset = IntegerOps::signExtend2(offset, 16, 32);
213 
214  // Case code address is computed from the JMP instruction. The case code must be after the (current) end of the offset
215  // table, which is also after the JMP instruction.
216  Address codeVa = (jmp->get_address() + 4 + offset) & IntegerOps::GenMask<Address, 32>::value;
217  if (codeVa < offsetVa+2)
218  break;
219 
220  codeVas.insert(codeVa);
221  leastCodeVa = std::min(leastCodeVa, codeVa);
222  ++tableIdx;
223  }
224  if (0==tableIdx)
225  return chain; // we failed to find any entries in the offset table
226 
227  // We now know the size of the offset table and the addresses of all the switch cases. Throw away the successors that were
228  // already computed (it was probably just a single indeterminate successor) and replace them with the ones we found.
229  args.bblock->successors().clear();
230  for (Address va: codeVas)
231  args.bblock->insertSuccessor(va, 32);
232 
233  // Create a data block for the offset table and attach it to the basic block
234  SgAsmType *tableEntryType = SageBuilderAsm::buildTypeU16();
235  SgAsmType *tableType = SageBuilderAsm::buildTypeVector(tableIdx, tableEntryType);
236  DataBlock::Ptr offsetTable = DataBlock::instance(startOfOffsetTable, tableType);
237  offsetTable->comment("m68k 'switch' statement's 'case' address table");
238  args.bblock->insertDataBlock(offsetTable);
239 
240  return chain;
241 }
