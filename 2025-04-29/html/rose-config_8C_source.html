155 readConfigFile(const boost::filesystem::path &configName) {
156  struct Resources {
157  FILE *file;
158  char *line;
159  size_t linesz;
160 
161  Resources()
162  : file(NULL), line(NULL), linesz(0) {}
163 
164  ~Resources() {
165  if (file)
166  fclose(file);
167  if (line)
168  free(line);
169  }
170  } r;
171 
172  Configuration retval;
173 
174  if (NULL == (r.file = fopen(configName.string().c_str(), "r"))) {
175  mlog[FATAL] <<strerror(errno) <<": \"" <<configName <<"\"\n";
176  exit(1);
177  }
178 
179  boost::regex keyRe("[a-zA-Z][a-zA-Z_0-9]*");
180  size_t lineNumber = 0;
181  while (ssize_t nchars = rose_getline(&r.line, &r.linesz, r.file)) {
182  ++lineNumber;
183  if (nchars < 0) {
184  if (errno) {
185  mlog[FATAL] <<configName <<":" <<lineNumber <<": " <<strerror(errno) <<"\n";
186  exit(1);
187  }
188  break; // EOF
189  }
190 
191  std::string s = r.line;
192  boost::trim(s);
193  if (s.empty() || '#' == s[0])
194  continue;
195 
196  // Parse the "key=value" line
197  size_t equal = s.find('=');
198  std::string key = s.substr(0, equal);
199  boost::trim(key);
200  std::string value = equal == std::string::npos ? std::string() : s.substr(equal+1);
201  boost::trim(value);
202  if (equal == std::string::npos || !boost::regex_match(key, keyRe)) {
203  mlog[FATAL] <<configName <<":" <<lineNumber <<": syntax error: expected KEY = VALUE\n";
204  exit(1);
205  }
206 
207  // Save the key and value for returning later
208  if (!retval.insert(std::make_pair(key, value)).second) {
209  mlog[FATAL] <<configName <<":" <<lineNumber <<": duplicate key \"" <<StringUtility::cEscape(key) <<"\"\n";
210  exit(1);
211  }
212 
213  //Add the old format key to the database as well
214  if (!retval.insert(std::make_pair(toOldKeyFormat(key), value)).second) {
215  mlog[FATAL] <<configName <<":" <<lineNumber <<": duplicate key \"" <<StringUtility::cEscape(key) <<"\"\n";
216  exit(1);
217  }
218 
219 
220  }
221 
222  bool hadError = false;
223  BOOST_FOREACH (const std::string &key, requiredKeys()) {
224  if (retval.find(key) == retval.end()) {
225  mlog[FATAL] <<configName <<":" <<lineNumber <<": required key \"" <<key <<"\" is not defined\n";
226  hadError = true;
227  }
228  }
229  if (hadError)
230  exit(1);
231 
232  return retval;
233 }

261 static std::string
262 makeLibrarySearchPaths(const std::string &str) {
263  // Split string into space-separated arguments in a rather simplistic way.
264  std::vector<std::string> args; //A list of split out strings (from str)
265  {
266  char quoted = '\0';
267  std::string arg; //One paricular string being split out
268  const char *s = str.c_str(); //pointer into the current string being split out
269  for (/*void*/; s && *s; ++s) { //While we haven't hit the end of str
270  if ('\\'==*s && s[1]) { //If we see an escaped backslash, just skip past it.
271  arg += *++s;
272  } else if ('\''==*s || '"'==*s) { //If we see a quote
273  if (!quoted) { //If we aren't in a quoted string, save the quote, but it doesn't go in arg
274  quoted = *s;
275  } else if (quoted == *s) { //If we we're in a quoted string, and the quotes match, that's the end of the quoted section. Clear the quote
276  quoted = '\0';
277  } else { //We found a different quote than the string we're in, so just save it arg like any other char
278  arg += *s;
279  }
280  } else if (isspace(*s)) { //we find a space
281  if (quoted) { //in a quoted string, save as a normal characted
282  arg += *s;
283  } else if (!arg.empty()) { //NOT in a quote, this must be an argument seperator, so save the old arg, and start a new one
284  args.push_back(arg);
285  arg = "";
286  }
287  } else { //Normal case, just copy the character to arg.
288  arg += *s;
289  }
290  }
291 
292  if(arg != "") { //Get that last argument
293  args.push_back(arg);
294  arg = "";
295  }
296  ASSERT_require2(quoted=='\0', "mismatched quotes");
297  }
298 
299  //Turn them all into a colon-seperated string
300  std::string retval;
301  for (size_t i=0; i<args.size(); ++i) {
302  retval += (retval.empty()?"":":") + args[i];
303  }
304  return retval;
305 }
