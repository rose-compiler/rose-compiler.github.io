198 Unparse_Python::unparseAsSuite(SgStatement* stmt, SgUnparse_Info& info) {
199  info.inc_nestingLevel();
200  switch (stmt->variantT()) {
201  case V_SgBasicBlock: {
202  SgBasicBlock* basic_block = isSgBasicBlock(stmt);
203  foreach (SgStatement* child, basic_block->get_statements()) {
204  curprint( ws_prefix(info.get_nestingLevel()) );
205  unparseStatement(child, info);
206  curprint("\n");
207  }
208  break;
209  }
210  case V_SgClassDefinition: {
211  SgClassDefinition* class_def = isSgClassDefinition(stmt);
212  foreach (SgDeclarationStatement* child, class_def->get_members()) {
213  curprint( ws_prefix(info.get_nestingLevel()) );
214  unparseStatement(child, info);
215  curprint("\n");
216  }
217  break;
218  }
219  case V_SgGlobal: {
220  ROSE_ASSERT(!"Cannot unparseAsSuite(SgGlobal*,info). Use unparseGlobalStmt()");
221  break;
222  }
223  default: {
224  curprint( ws_prefix(info.get_nestingLevel()) );
225  unparseStatement(stmt, info);
226  curprint("\n");
227  break;
228  }
229  }
230  info.dec_nestingLevel();
231 }

1068 bool
1069 Unparse_Python::requiresParentheses(SgExpression* expr, SgUnparse_Info &) {
1070  SgExpression* parent = isSgExpression(expr->get_parent());
1071  if (parent == nullptr) return false;
1072 
1073  PrecedenceSpecifier exprPrecedence = getPrecedence(expr);
1074  PrecedenceSpecifier parentPrecedence = getPrecedence(parent);
1075  AssociativitySpecifier assoc = getAssociativity(parent);
1076  if (exprPrecedence == ROSE_UNPARSER_NO_PRECEDENCE) return false;
1077  else if (parentPrecedence == ROSE_UNPARSER_NO_PRECEDENCE) return false;
1078  else if (assoc == e_assoc_right && exprPrecedence > parentPrecedence) return false;
1079  else if (assoc == e_assoc_left && exprPrecedence >= parentPrecedence) return false;
1080  else if (assoc == e_assoc_left && exprPrecedence < parentPrecedence) return true;
1081  else if (assoc == e_assoc_right && exprPrecedence <= parentPrecedence) return true;
1082  else {
1083  ROSE_ASSERT(!"cannot determine if parens are required in UnparseLanguageIndependentConstructs::requiresParentheses");
1084  abort(); // in case ROSE_ASSERT is disabled
1085  }
