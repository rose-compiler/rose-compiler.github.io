108 ROSE_DLL_API std::string
109 escape(const std::string &s) {
110  if (s.empty())
111  return "\"\"";
112  if (isId(s))
113  return s;
114  if ('<'==s[0]) {
115  int depth = 0;
116  for (char ch: s) {
117  if ('<'==ch) {
118  ++depth;
119  } else if ('>'==ch) {
120  if (--depth < 0)
121  break;
122  }
123  }
124  if (0==depth)
125  return "<" + htmlEscape(s) + ">";
126  }
127  return "\"" + quotedEscape(s) + "\"";

396 void
397 CfgEmitter::selectFunctionCallers(const Function::Ptr &callee) {
398  // Use an iteration rather than a traversal because we want to consider all vertices that belong to the function, including
399  // those not reachable from the entry vertex.
400  for (const ControlFlowGraph::Vertex &vertex: graph_.vertices()) {
401  if (vertexOrganization(vertex).isSelected() && vertex.value().isOwningFunction(callee)) {
402 
403  // Are there edges coming into this vertex from outside this function?
404  typedef Sawyer::Container::Map<Address /*caller*/, CallInfo> Callers;
405  Callers callers;
406  for (const ControlFlowGraph::Edge &interEdge: vertex.inEdges()) {
407  if (partitioner_->isEdgeInterProcedural(interEdge, Function::Ptr(), callee) &&
408  !edgeOrganization(interEdge).isSelected()) {
409 
410  // Is the call coming from a block that belongs to a function, or from some random non-owned block?
411  if (Function::Ptr caller = firstOwningFunction(interEdge.source())) {
412  CallInfo callInfo = callers.getOptional(caller->address()).orDefault();
413  if (interEdge.value().type() == E_FUNCTION_CALL) {
414  ++callInfo.nCalls;
415  } else if (interEdge.value().type() == E_FUNCTION_XFER) {
416  ++callInfo.nTransfers;
417  } else {
418  ++callInfo.nOthers;
419  }
420  callers.insert(caller->address(), callInfo);
421  } else {
422  Organization &src = vertexOrganization(interEdge.source());
423  if (!src.isSelected()) {
424  src.select();
425  src.label(vertexLabel(interEdge.source()));
426  src.attributes(vertexAttributes(interEdge.source()));
427  }
428  edgeOrganization(interEdge).select();
429  edgeOrganization(interEdge).label(edgeLabel(interEdge));
430  edgeOrganization(interEdge).attributes(edgeAttributes(interEdge));
431  }
432  }
433  }
434 
435  // Organize the calls to this vertex from a function-as-a-whole
436  for (const Callers::Node &callNode: callers.nodes()) {
437  Function::Ptr callerFunc = partitioner_->functionExists(callNode.key());
438  ASSERT_not_null(callerFunc);
439  ControlFlowGraph::ConstVertexIterator caller = partitioner_->findPlaceholder(callerFunc->address());
440  ASSERT_require(caller != graph_.vertices().end());
441 
442  Organization &org = vertexOrganization(caller);
443  if (!org.isSelected()) {
444  org.select();
445  org.label(functionLabel(callerFunc));
446  org.attributes(functionAttributes(callerFunc));
447  }
448 
449  const CallInfo &callInfo = callNode.value();
450  std::string label;
451  if (callInfo.nCalls)
452  label += StringUtility::plural(callInfo.nCalls, "calls");
453  if (callInfo.nTransfers)
454  label += (label.empty()?"":"\n") + StringUtility::plural(callInfo.nTransfers, "xfers");
455  if (callInfo.nOthers)
456  label += (label.empty()?"":"\n") + StringUtility::plural(callInfo.nOthers, "others");
457 
458  pseudoEdges_.push_back(PseudoEdge(caller, graph_.findVertex(vertex.id()), label));
459  }
460  }
461  }

568 CfgEmitter::isInterFunctionEdge(const ControlFlowGraph::Edge &edge) {
569  if (edge.value().type() == E_FUNCTION_CALL || edge.value().type() == E_FUNCTION_XFER)
570  return true;
571  if (edge.source() == edge.target())
572  return false;
573 
574  // When the source and/or target blocks of an edge are owned by multiple functions it can be ambiguous whether the edge is
575  // inter- or intra- function. For the purposes of the GraphViz output, each vertex is assigned to at most one function
576  // and the edge classification is based on this assignment.
577  return firstOwningFunction(edge.source()) != firstOwningFunction(edge.target());

624 std::string
625 CfgEmitter::vertexLabel(const ControlFlowGraph::ConstVertexIterator &vertex) const {
626  std::string srcLoc = sourceLocation(vertex);
627  if (!srcLoc.empty())
628  srcLoc = htmlEscape(srcLoc) + "<br align=\"left\"/>";
629  switch (vertex->value().type()) {
630  case V_BASIC_BLOCK: {
631  // If this basic block is the entry point to at least one function, show the function name as the basic block's
632  // label.
633  bool foundEntryPoint = false;
634  FunctionSet functions = owningFunctions(vertex);
635  if (!functions.isEmpty()) {
636  for (const Function::Ptr &function: functions.values()) {
637  if (function->address() == vertex->value().address()) {
638  srcLoc += htmlEscape(function->printableName()) + "<br align=\"left\"/>";
639  foundEntryPoint = true;
640  }
641  }
642  if (foundEntryPoint)
643  return "<" + srcLoc + ">";
644  }
645 
646  // If this basic block has a name (i.e., an address) then use it as the label.
647  if (BasicBlock::Ptr bb = vertex->value().bblock())
648  return "<" + srcLoc + htmlEscape(bb->printableName()) + ">";
649 
650  // Last resort, use the address stored in the CFG (not sure if this code is even reachable)
651  return "<" + srcLoc + StringUtility::addrToString(vertex->value().address()) + ">";
652  }
653  case V_NONEXISTING:
654  return "\"nonexisting\"";
655  case V_UNDISCOVERED:
656  return "\"undiscovered\"";
657  case V_INDETERMINATE:
658  return "\"indeterminate\"";
659  case V_USER_DEFINED:
660  return "\"user defined\"";
661  }
662  ASSERT_not_reachable("invalid vertex type");
663 }

671 CfgEmitter::vertexLabelDetailed(const ControlFlowGraph::ConstVertexIterator &vertex) const {
672  BasicBlock::Ptr bb;
673  if (showInstructions_ && vertex->value().type() == V_BASIC_BLOCK && (bb = vertex->value().bblock())) {
674  const std::vector<SgAsmInstruction*> insns = bb->instructions();
675

677  std::vector<bool> isPartOfNoopSequence(insns.size(), false);
678  if (strikeNoopSequences_ && !isPartOfNoopSequence.empty()) {
679  NoOperation::IndexIntervals noopSequences = noOpAnalysis_.findNoopSubsequences(insns);
680  noopSequences = NoOperation::largestEarliestNonOverlapping(noopSequences);
681  for (const NoOperation::IndexInterval &where: noopSequences) {
682  for (size_t i=where.least(); i<=where.greatest(); ++i)
683  isPartOfNoopSequence[i] = true;
684  }

688  std::string srcLoc = sourceLocation(vertex);
689  if (!srcLoc.empty())
690  srcLoc = htmlEscape(srcLoc) + "<br align=\"left\"/>";

692 
693  // Instructions for this BB.
694  for (size_t i=0; i<insns.size(); ++i) {
695  SgAsmInstruction *insn = insns[i];
696 
697  if (showInstructionAddresses_)
698  s += StringUtility::addrToString(insn->get_address()).substr(2) + " ";
699 
700  if (isPartOfNoopSequence[i]) {
701  if (versionDate_ >= 20130915) {
702  // Strike out each insn of the no-op sequence
703  s += "<s>" + htmlEscape(insn->toStringNoAddr()) + "</s><br align=\"left\"/>";
704  } else {
705  // Put the no-op in parentheses because we graphViz doesn't have strike-through capability
706  s += "no-op (" + htmlEscape(insn->toStringNoAddr()) + ")<br align=\"left\"/>";
707  }

1011  Sawyer::Container::Map<std::string /*vertex name*/, PositionGraph::VertexIterator> vertexMap;
1012

1014  char buf[8192];
1015  while (in) {
1016  ++lineNumber;
1017  in.getline(buf, sizeof buf);
1018 
1019  // Split the line into space-separated words, ignoring quotes, comments, HTML tags, etc.
1020  std::vector<std::string> words;
1021  char *s = buf; // start
1022  while (s) {
1023  while (isspace(*s))
1024  ++s;
1025  if (!*s)
1026  break;
1027  char *t = s; // terminate
1028  while (t && !isspace(*t))
1029  ++t;
1030  words.push_back(std::string(s, t));
1031  s = t;
1032  }
1033  if (words.empty())
1034  continue;
1035 
1036  // Parse the words (ignore everything after the coordinates)
1037  if (1 == lineNumber) {
1038  if (words.size() < 4 || words[0] != "graph")
1039  throw std::runtime_error("expected \"graph SF DX DY\" at start of input");
1040 
1041  } else if (words[0] == "node") {
1042  if (words.size() < 6) {
1043  throw std::runtime_error("expected \"node NAME X Y DX DY\" at line " +
1044  StringUtility::numberToString(lineNumber));
1045  }
1046  VertexPosition pos;
1047  pos.name = words[1];
1048  pos.center.x = boost::lexical_cast<double>(words[2]);
1049  pos.center.y = boost::lexical_cast<double>(words[3]);
1050  pos.width = boost::lexical_cast<double>(words[4]);
1051  pos.height = boost::lexical_cast<double>(words[5]);
1052  ASSERT_forbid2(vertexMap.exists(pos.name), "duplicate vertex name");
1053  vertexMap.insert(pos.name, retval.insertVertex(pos));
1054 
1055  } else if (words[0] == "edge") {
1056  if (words.size() < 12) {
1057  throw std::runtime_error("expected \"edge SOURCE TARGET N SPLINE...\" at line " +
1058  StringUtility::numberToString(lineNumber));
1059  }
1060  EdgePosition pos;
1061  std::string &sourceVertexName = words[1];
1062  std::string &targetVertexName = words[2];
1063  size_t nControlPoints = boost::lexical_cast<size_t>(words[3]);
1064  for (size_t i=0; i<nControlPoints; ++i) {

1067  cp.y = boost::lexical_cast<double>(words[2*i+4]);

1069  }
1070  ASSERT_require(vertexMap.exists(sourceVertexName));
1071  ASSERT_require(vertexMap.exists(targetVertexName));

1073
