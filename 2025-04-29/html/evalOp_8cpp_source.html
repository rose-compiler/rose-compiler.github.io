173 POETCode* split_prefix(POETCode* code, unsigned& prefix, POETCode*& left)
174 {
175  switch (code->get_enum()) {
176  case SRC_STRING: {
177  POETString* str = static_cast<POETString*>(code);
178  std::string content = str->get_content();
179  unsigned size = content.size();
180  if (prefix < size) {
181  unsigned size1 = prefix, size2 = size - prefix + 1;
182  prefix = 0;
183  left = ASTFactory::inst()->new_string(content.substr(size1,size2));
184  return ASTFactory::inst()->new_string(content.substr(0,size1));
185  }
186  else if (prefix == size) {
187  left = EMPTY; prefix=0; return code;
188  }
189  else {
190  left = EMPTY; prefix -= size; return code;
191  }
192  }
193  case SRC_LIST: {
194  POETList* l = static_cast<POETList*>(code);
195  POETCode* first = l->get_first(), *rest = l->get_rest();
196  POETCode* split1 = split_prefix(first, prefix, left);
197  if (split1 != 0 && prefix > 0) {
198  if (rest == 0) { left=EMPTY; return split1; }
199  POETCode* split2 = split_prefix(rest, prefix, left);
200  return ASTFactory::inst()->new_list(split1, split2);
201  }
202  if (left == EMPTY && rest != 0) left = rest;
203  else left = ASTFactory::inst()->new_list(left, rest);
204  return split1;
205  }
206  default:
207  left=code;
208  return EMPTY;
209  }

223 int apply_Bop(POETOperatorType op, POETCode* op1, POETCode* op2)
224 {
225  int res = 0;
226  switch (op) {
227  case POET_OP_MINUS:
228  res = AST2Int(op1) - AST2Int(op2); break;
229  case POET_OP_PLUS:
230  res = AST2Int(op1) + AST2Int(op2); break;
231  case POET_OP_MULT:
232  res = AST2Int(op1) * AST2Int(op2); break;
233  case POET_OP_DIVIDE:
234  res = AST2Int(op1) / AST2Int(op2); break;
235  case POET_OP_MOD:
236  res = AST2Int(op1) % AST2Int(op2); break;
237  case POET_OP_EQ:
238  res = (op1 == op2); break;
239  case POET_OP_NE:
240  res = (op1 != op2); break;
241  case POET_OP_LT:
242  {
243  POETIconst* op1Int = AST2Iconst(op1), *op2Int=AST2Iconst(op2);
244  if (op1Int != 0 && op2Int != 0)
245  { res = (op1Int->get_val() < op2Int->get_val()); }
246  else if (op1->get_enum() == SRC_STRING && op2->get_enum() == SRC_STRING)
247  { res = static_cast<POETString*>(op1)->get_content()
248  < static_cast<POETString*>(op2)->get_content(); }
249  else { res = 0; }
250  break;
251  }
252  case POET_OP_LE:
253  {
254  POETIconst* op1Int = AST2Iconst(op1), *op2Int=AST2Iconst(op2);
255  if (op1Int != 0 && op2Int != 0)
256  { res = (op1Int->get_val() <= op2Int->get_val()); }
257  else if (op1->get_enum() == SRC_STRING && op2->get_enum() == SRC_STRING)
258  { res = static_cast<POETString*>(op1)->get_content()
259  <= static_cast<POETString*>(op2)->get_content(); }
260  else { res = 0; }
261  break;
262  }
263  case POET_OP_GT:
264  {
265  POETIconst* op1Int = AST2Iconst(op1), *op2Int=AST2Iconst(op2);
266  if (op1Int != 0 && op2Int != 0)
267  { res = (op1Int->get_val() > op2Int->get_val()); }
268  else if (op1->get_enum() == SRC_STRING && op2->get_enum() == SRC_STRING)
269  {
270  std::string r1= static_cast<POETString*>(op1)->get_content() ;
271  std::string r2= static_cast<POETString*>(op2)->get_content();
272  res = static_cast<POETString*>(op1)->get_content()
273  > static_cast<POETString*>(op2)->get_content(); }
274  else { res = 0; }
275  break;
276  }
277  case POET_OP_GE:
278  {
279  POETIconst* op1Int = AST2Iconst(op1), *op2Int=AST2Iconst(op2);
280  if (op1Int != 0 && op2Int != 0)
281  { res = (op1Int->get_val() >= op2Int->get_val()); }
282  else if (op1->get_enum() == SRC_STRING && op2->get_enum() == SRC_STRING)
283  { res = static_cast<POETString*>(op1)->get_content()
284  >= static_cast<POETString*>(op2)->get_content(); }
285  else { res = 0; }
286  break;
287  }
288  default: assert(false);
289  }
290  return res;
291 }

487  public:

492  if (cur == 0) {
493  if (e.get_entry_type() == LVAR_TRACE) TRACE_RECURSIVE(v)
494  else SYM_UNDEFINED(v->toString())

497  e.set_code(0);
498  apply(cur);

515  l->get_rest()->visit(this);
516  if (res.size() > 0 || rf.size() > 0) {
517  if (rf.size() == 0)
518  for (unsigned i = 0; i < res.size(); ++i)
519  res[i] = fac->new_list(l->get_first(), res[i]);
520  else if (res.size() == 0)
521  for (unsigned i = 0; i < rf.size(); ++i)
522  res.push_back(fac->new_list(rf[i],l->get_rest()));

524  for (unsigned i = 0; i < res.size(); ++i)

526  }

529  {

531  int dsize = 0;

533  v->get_entry(i)->visit(this);
534  tmp.push_back(res);
535  if (res.size() > 0) dsize = res.size();
536  res.clear();
537  }
538  for (int j=0 ; j < dsize; ++j) {
539  POETCode* cur = 0;
540  for (unsigned i=0 ; i < v->size(); ++i) {
541  if (tmp[i].size() > 0)
542  cur = fac->append_tuple(cur, tmp[i][j]);
543  else
544  cur = fac->append_tuple(cur, v->get_entry(i));
545  }
546  res.push_back(cur);
547  }
548  }

550  {

556  }

595  public:
596  CodeDupVisitor( POETCode* f, POETCode* dup)
597  : from(f), to(dup) { assert(from != 0 && to != 0); }
598  POETCode* apply(POETCode* code)
599  {
600  POETCode* res2 = DupInfoVisitor::apply(code);
601  if (res2 == EMPTY_LIST && code != 0 && code != EMPTY_LIST) {
602  assert(to != 0);
603  POETCode* cur = to;
