35  gprof_info::gprof_info( const std::string& input )
36  // set variables to default values

47  // This function parses the string with the help of boost::spirit
48  // see http://spirit.sourceforge.net/distrib/spirit_1_8_5/libs/spirit/index.html for more information
49  // deklaration of the different rules

61  // Setting up the grammar
62  // reminder of how the line looks:
63  // time_precent cumulative_seconds self_seconds calls self_ns_calls total_ns_call name
64  // double double double size_t double double
65  //
66  // name is a compound of the following:
67  // function_name (file_name:line_number @ call_address )
68  // string string size_t
69  //
70  // in between the data fields at least one space can be found
71  grammar = +ch_p( ' ' ) >> time_percent_p >>
72  +ch_p( ' ' ) >> cumulative_seconds_p >>
73  +ch_p( ' ' ) >> self_seconds_p >>
74  !( +ch_p( ' ' ) >> calls_p ) >> // calls are optional
75  !( +ch_p( ' ' ) >> self_ns_calls_p ) >> // self_ns_calls are optional
76  !( +ch_p( ' ' ) >> total_ns_calls_p ) >> // total_ns_calls are optional
77  +ch_p( ' ' ) >> function_name_p >>
78  ch_p( ' ' ) >> ch_p( '(' ) >> file_name_p >>
79  ch_p( ':' ) >> line_number_p >> *anychar_p; // *anychar_p reads in the remaining part of the string,
80  // but dismisses it
81  // definition of the different parts we want to safe.
82  // format of the input assignment action, if found
83  time_percent_p = real_p [ assign_a( time_percent_ ) ];
84  cumulative_seconds_p = real_p [ assign_a( cumulative_seconds_ ) ];
85  self_seconds_p = real_p [ assign_a( self_seconds_ ) ];
86  calls_p = int_p [ assign_a( calls_ ) ];
87  self_ns_calls_p = real_p [ assign_a( self_ns_calls_ ) ];
88  total_ns_calls_p = real_p [ assign_a( total_ns_calls_ ) ];
89  function_name_p = lexeme_d[ ( *( print_p - blank_p ) ) [ assign_a( function_name_ ) ] ];
90  file_name_p = lexeme_d[ ( *( print_p - blank_p - ch_p( ':' ) ) ) [ assign_a( file_name_ )] ];
91  line_number_p = int_p [ assign_a( line_number_ ) ];
92 
93  // actual parsing
94  bool success = parse( input.c_str(), grammar ).full;
95 
96  if( !success ) throw "gprofInfo: Parsing not successful";
97  }
