153 void QFormatScheme::load(const QDomElement& elem, bool ignoreNewIds)
154 {
155  if ( elem.attribute("version") < QFORMAT_VERSION )
156  {
157  qWarning("Format encoding version mismatch : [found]%s != [expected]%s",
158  qPrintable(elem.attribute("version")),
159  QFORMAT_VERSION);
160 
161  return;
162  }
163 
164  QDomElement e, c;
165  QDomNodeList l, f = elem.elementsByTagName("format");
166 
167  for ( int i = 0; i < f.count(); i++ )
168  {
169  e = f.at(i).toElement();
170 
171  if ( ignoreNewIds && !m_formatKeys.contains(e.attribute("id")) )
172  continue;
173 
174  l = e.childNodes();
175 
176  QFormat fmt;
177 
178  for ( int i = 0; i < l.count(); i++ )
179  {
180  c = l.at(i).toElement();
181 
182  if ( c.isNull() )
183  continue;
184 
185  QString field = c.tagName(),
186  value = c.firstChild().toText().data();
187 
188  if ( field == "bold" )
189  fmt.weight = bool_cast(value) ? QFont::Bold : QFont::Normal;
190  else if ( field == "italic" )
191  fmt.italic = bool_cast(value);
192  else if ( field == "overline" )
193  fmt.overline = bool_cast(value);
194  else if ( field == "underline" )
195  fmt.underline = bool_cast(value);
196  else if ( field == "strikeout" )
197  fmt.strikeout = bool_cast(value);
198  else if ( field == "waveUnderline" )
199  fmt.waveUnderline = bool_cast(value);
200  else if ( field == "color" || field == "foreground" )
201  fmt.foreground = QColor(value);
202  else if ( field == "background" )
203  fmt.background = QColor(value);
204  else if ( field == "linescolor" )
205  fmt.linescolor = QColor(value);
206 
207  }
208 
209  setFormat(e.attribute("id"), fmt);
210  }
