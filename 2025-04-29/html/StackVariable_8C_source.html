108 StackVariable::interval() const {
109  // We need to watch for overflows. The return type, OffsetInterval, has int64_t least and greatest values. The stack
110  // offset is also int64_t. The maximum size in bytes however is uint64_t (i.e., Address). We may need to reduce the
111  // maximum size in order to fit it into the interval return value.
112  int64_t least = stackOffset_;
113  int64_t maxSizeSigned = maxSizeBytes() > boost::numeric_cast<uint64_t>(boost::integer_traits<int64_t>::const_max)
114  ? boost::integer_traits<int64_t>::const_max
115  : boost::numeric_cast<int64_t>(maxSizeBytes());
116  if (least >= 0) {
117  int64_t headroom = boost::integer_traits<int64_t>::const_max - maxSizeSigned;
118  if (least > headroom) {
119  // overflow would occur, so we must reduce the maxSizeS appropriately
120  return OffsetInterval::hull(least, boost::integer_traits<int64_t>::max());
121  } else {
122  return OffsetInterval::baseSize(least, maxSizeSigned);
123  }
124  } else {
