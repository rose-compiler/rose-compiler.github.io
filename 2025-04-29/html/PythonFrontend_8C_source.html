77 runPythonFrontend(SgFile* file)
78 {
79  PyObject *pModule, *pFunc;
80  PyObject *pValue, *pArgs;
81 
82  Py_InitModule("sage", SageBuilderMethods);
83 
84  // Hack to set sys.path (the list of paths that Python uses to look up modules)
85  // so that the subsequent import succeeds. The program first looks in the
86  // installation directory, then the source directory. (Automake's support for
87  // Python does not copy .py (or .pyo, .pyc) into the compile tree).
88  // TODO: find a better way to run/distribute the code in sageTranslator.py
89  stringstream cmd;
90  cmd << "import sys" << endl;
91  cmd << "sys.path.insert(0, '" << ROSE_INSTALLATION_PATH << "/lib/python')" << endl;
92  cmd << "sys.path.insert(0, '" << ROSE_AUTOMAKE_TOP_SRCDIR << "/src/frontend/PythonFrontend')" << endl;
93  PyRun_SimpleString(cmd.str().c_str());
94  pModule = PyImport_ImportModule(ROSE_PYTHON_FRONTEND_MODULE_NAME);
95 
96 
97  if (pModule != NULL) {
98  pFunc = PyObject_GetAttrString(pModule, ROSE_PYTHON_FRONTEND_TRANSLATOR_FXN_NAME);
99 
100  std::string filename = file->get_sourceFileNameWithPath();
101  pValue = PyString_FromString(filename.c_str());
102  pArgs = PyTuple_New(1);
103  PyTuple_SetItem(pArgs, 0, pValue);
104 
105  if (pFunc && PyCallable_Check(pFunc)) {
106  pValue = PyObject_CallObject(pFunc, pArgs);
107  Py_DECREF(pArgs);
108 
109  if (pValue != NULL) {
110  void* capsule_ptr = PyCapsule_GetPointer(pValue, NULL);
111  SgGlobal* sg_global = static_cast<SgGlobal*>(capsule_ptr);
112  Py_DECREF(pValue);
113  return sg_global;
114  }
115  else {
116  Py_DECREF(pFunc);
117  Py_DECREF(pModule);
118  PyErr_Print();
119  ROSE_ABORT("Python frontend failed.");
120  return NULL;
121  }
122  }
123  else {
124  if (PyErr_Occurred())
125  PyErr_Print();
126  cerr << "Cannot find function " <<
127  ROSE_PYTHON_FRONTEND_TRANSLATOR_FXN_NAME << endl;
128  }
129  Py_XDECREF(pFunc);
130  Py_DECREF(pModule);
131  }
132  else {
133  PyErr_Print();
134  cerr << "Failed to load module " << ROSE_PYTHON_FRONTEND_MODULE_NAME << endl;
135  return NULL;
136  }
137  return NULL;
