89 static bool ofs_match_substr(const char* substr, bool checkTrail = true)
90 {
91  bool result = true;

97  {
98  if ((*c_char)==substr[i])

100  c_char++;
101  }
102  else

1119 
1120  if (last_pos==string::npos)
1121  last_pos = second_pos;
1122  // we can now remove from first to last pos from the buffer
