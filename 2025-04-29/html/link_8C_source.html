498

500  // This happen when a variable is declared in two headers (extern) and each header include the other header.
501  // Then, when the AST from two files that include one of the header are merged, the sharing will make the symbol point to a iname that is not the first in the chain.
502  // FIXME if performance issue it might be faster to preserve the other iname during sharing
503  std::map<SgInitializedName *, SgInitializedName *> any_to_first;

507 
508  // build longuest list so far (else it would have been seen) (could build partial list but premature optimization)
509  std::vector<SgInitializedName *> rlist;
510  rlist.push_back(iname);
511  while (true) {
512  SgInitializedName * prev = rlist.back()->get_prev_decl_item();
513  ROSE_ASSERT(prev != rlist.back());
514  if (prev == NULL) break;
515  seens.insert(prev);
516  rlist.push_back(prev);
517  }
518  SgInitializedName * first_iname = rlist.back();
519  first_to_rlist[first_iname] = rlist;
520 
521  for (auto i: rlist) {
522  any_to_first[i] = first_iname;
523  }
524  }
525 
526  for (auto sym: syms) {
527  SgInitializedName * iname = isSgInitializedName(sym->get_symbol_basis());
528  ROSE_ASSERT(iname != NULL);
529 
530  iname = any_to_first[iname];

534  ROSE_ASSERT(rlist.size() > 0);

563  auto const & refdesc = descmap.find(refsym)->second;
564 
565  createSymbolAlias(gsaf, refsym, refdesc.first->get_scope());
566 
567  // Register symbols for deletion and substitution. Patches nondef/defn declarations.
568  decl_t * last = refdesc.last;

577  // We only concatenate the chain of inames associated with this symbol if the
578  // symbol's basis is actually the first item of that chain. Other cases are due
579  // to multiple extern declaration being seen from headers (which are read in
580  // various orders) and eliminated by sharing. The result should be a single chain with
581  // symbols pointing to various point of that chain.
