53 set<SSA_UnfilteredCfg::VarName> getAllVarsUsedOrDefined(
54  const SSA_UnfilteredCfg::CFGNodeToVarNamesMap& defs,
55  const SSA_UnfilteredCfg::ASTNodeToVarRefsMap& uses,
56  const map<CFGNode, SSA_UnfilteredCfg::NodeReachingDefTable>& localDefTable)
57 {
58  set<SSA_UnfilteredCfg::VarName> usedNames;
59 
60  foreach(const SSA_UnfilteredCfg::CFGNodeToVarNamesMap::value_type& nodeVarsPair, defs)
61  {
62  map<CFGNode, SSA_UnfilteredCfg::NodeReachingDefTable>::const_iterator allDefsIter =
63  localDefTable.find(nodeVarsPair.first);
64 
65  foreach(const SSA_UnfilteredCfg::NodeReachingDefTable::value_type& varDefPair, allDefsIter->second)
66  {
67  usedNames.insert(varDefPair.first);
68  }
69  }
70 
71  foreach(const SSA_UnfilteredCfg::ASTNodeToVarRefsMap::value_type& nodeVarRefsPair, uses)
72  {
73 
74  foreach(SgVarRefExp* varRef, nodeVarRefsPair.second)
75  {
76  const SSA_UnfilteredCfg::VarName& usedVarName = SSA_UnfilteredCfg::getVarName(varRef);
77  ROSE_ASSERT(usedVarName != SSA_UnfilteredCfg::emptyName);
78 
79  //If we have a use for p.x.y, insert uses for p and p.x as well as p.x.y
80  for (size_t i = 1; i <= usedVarName.size(); ++i)
81  {
82  SSA_UnfilteredCfg::VarName var(usedVarName.begin(), usedVarName.begin() + i);
83 
84  usedNames.insert(var);
85  }
86  }
87  }

223 void SSA_UnfilteredCfg::expandParentMemberDefinitions(const CFGNodeToVarNamesMap& defs)
224 {
225 
226  foreach(const CFGNodeToVarNamesMap::value_type& cfgNodeVarsPair, defs)
227  {
228  const CFGNode& definingNode = cfgNodeVarsPair.first;
229  const set<VarName>& definedVarsAtNode = cfgNodeVarsPair.second;
230 
231  NodeReachingDefTable& nodeLocalDefs = localDefTable[definingNode];
232 
233  foreach(const VarName& definedVar, definedVarsAtNode)
234  {
235  //We are dealing with a multi-part variable, loop the entry and expand it
236  //Start at one so we don't get the same defs in the original and expanded defs
237  for (unsigned int i = 1; i < definedVar.size(); i++)
238  {
239  //Create a new varName vector that goes from beginning to end - i
240  VarName newName;
241  newName.assign(definedVar.begin(), definedVar.end() - i);
242 
243  //Only insert the new definition if it does not already exist in the original def table
244  if (definedVarsAtNode.count(newName) == 0)
245  {
246  ROSE_ASSERT(nodeLocalDefs.count(newName) == 0);
247  ReachingDefPtr expandedDef = ReachingDefPtr(new ReachingDef(definingNode, ReachingDef::EXPANDED_DEF));
248  nodeLocalDefs.insert(make_pair(newName, expandedDef));
249  }
250  }
251  }

255 void SSA_UnfilteredCfg::runDefUseDataFlow(SgFunctionDefinition* func)
256 {
257  //Keep track of visited nodes
258  set<CFGNode> visited;
259 
260  set<CFGNode> worklist;
261 
262  CFGNode current = func->cfgForBeginning();
263  worklist.insert(current);
264 
265  while (!worklist.empty())
266  {
267  if (getDebugExtra())
268  cout << "-------------------------------------------------------------------------" << endl;
269  //Get the node to work on
270  current = *worklist.begin();
271  worklist.erase(worklist.begin());
272 
273  //Propagate defs to the current node
274  bool changed = propagateDefs(current);
275 
276  //For every edge, add it to the worklist if it is not seen or something has changed
277 
278  reverse_foreach(const CFGEdge& edge, current.outEdges())
279  {
280  CFGNode nextNode = edge.target();
281 
282  //Insert the child in the worklist if the parent is changed or it hasn't been visited yet
283  if (changed || visited.count(nextNode) == 0)
284  {
285  //Add the node to the worklist
286  bool insertedNew = worklist.insert(nextNode).second;
287  if (insertedNew && getDebugExtra())
288  {
289  if (changed)
290  cout << "Defs Changed: Added " << nextNode.getNode()->class_name() << nextNode.getNode() << " to the worklist." << endl;
291  else
292  cout << "Next unvisited: Added " << nextNode.getNode()->class_name() << nextNode.getNode() << " to the worklist." << endl;
293  }
294  }
295  }
296 
297  //Mark the current node as seen
298  visited.insert(current);
299  }

302 bool SSA_UnfilteredCfg::propagateDefs(const CFGNode& cfgNode)
303 {
304  if (getDebugExtra())
305  printf("propagateDefs(%s)\n", cfgNode.toStringForDebugging().c_str());
306 
307  //This updates the IN table with the reaching defs from previous nodes
308  updateIncomingPropagatedDefs(cfgNode);
309 
310  //Create a staging OUT table. At the end, we will check if this table
311  //Was the same as the currently available one, to decide if any changes have occurred
312  //We initialize the OUT table to the IN table
313  NodeReachingDefTable stagingOutDefsTable = reachingDefTable[cfgNode];
314 
315  if (getDebugExtra())
316  {
317  printf("Incoming defs:\n");
318  printNodeDefTable(stagingOutDefsTable);
319  }
320 
321  //Now overwrite any local definitions
322  CFGNodeToDefTableMap::const_iterator localDefIter = localDefTable.find(cfgNode);
323  if (localDefIter != localDefTable.end())
324  {
325 
326  foreach(const NodeReachingDefTable::value_type& varDefPair, localDefIter->second)
327  {
328  const VarName& definedVar = varDefPair.first;
329  ReachingDefPtr localDef = varDefPair.second;
330 
331  stagingOutDefsTable[definedVar] = localDef;
332  }
333  }
334 
335  //Compare old to new OUT tables
336  NodeReachingDefTable& existingoutDefTable = outgoingDefTable[cfgNode];
337  bool changed = (existingoutDefTable != stagingOutDefsTable);
338  if (changed)
339  {
340  existingoutDefTable = stagingOutDefsTable;
341  }
342 
343  if (getDebugExtra() && changed)
344  {
345  printf("\nOutgoing defs changed!:\n");
346  printNodeDefTable(outgoingDefTable[cfgNode]);
347  }
348 
349  return changed;
350 }

352 void SSA_UnfilteredCfg::updateIncomingPropagatedDefs(const CFGNode& cfgNode)
353 {
354  //Get the previous edges in the CFG for this node
355  vector<CFGEdge> inEdges = cfgNode.inEdges();
356 
357  NodeReachingDefTable& incomingDefTable = reachingDefTable[cfgNode];
358 
359  //Iterate all of the incoming edges
360  for (unsigned int i = 0; i < inEdges.size(); i++)
361  {
362  const CFGNode& prev = inEdges[i].source();
363 
364  const NodeReachingDefTable& previousDefs = outgoingDefTable[prev];
365 
366  //Merge all the previous defs into the IN table of the current node
367 
368  foreach(const NodeReachingDefTable::value_type& varDefPair, previousDefs)
369  {
370  const VarName& var = varDefPair.first;
371  const ReachingDefPtr previousDef = varDefPair.second;
372 
373  //Here we don't propagate defs for variables that went out of scope
374  //(built-in vars are body-scoped but we inserted the def at the SgFunctionDefinition node, so we make an exception)
375  //if (!isVarInScope(var, cfgNode.getNode()) && !isBuiltinVar(var))
376  // continue;
377 
378  //If this is the first time this def has propagated to this node, just copy it over
379  NodeReachingDefTable::const_iterator existingDefIter = incomingDefTable.find(var);
380  if (existingDefIter == incomingDefTable.end())
381  {
382  incomingDefTable.insert(make_pair(var, previousDef));
383  }
384  else
385  {
386  ReachingDefPtr existingDef = existingDefIter->second;
387 
388  if (existingDef->isPhiFunction() && existingDef->getDefinitionNode() == cfgNode)
389  {
390  //There is a phi node here. We update the phi function to point to the previous reaching definition
391  existingDef->addJoinedDef(previousDef, inEdges[i]);
392  }
393  else
394  {
395  //If there is no phi node, and we get a new definition, it better be the same as the one previously
396  //propagated.
397  if (!(*previousDef == *existingDef))
398  {
399  printf("ERROR: At node %s, two different definitions reach for variable %s\n",
400  cfgNode.toStringForDebugging().c_str(), varnameToString(var).c_str());
401  ROSE_ABORT();
402  }
403  }
404  }
405  }
406  }

409 void SSA_UnfilteredCfg::insertDefsForChildMemberUses(const CFGNodeToVarNamesMap& defs, const set<VarName>& usedNames)
410 {
411  //Map each varName to all used names for which it is a prefix
412  map<VarName, set<VarName> > nameToChildNames;
413 
414  foreach(const VarName& rootName, usedNames)
415  {
416 
417  foreach(const VarName& childName, usedNames)
418  {
419  if (childName.size() <= rootName.size())
420  continue;
421 
422  if (isPrefixOfName(childName, rootName))
423  {
424  nameToChildNames[rootName].insert(childName);
425  }
426  }
427  }
428 
429  //Go through each definition and make sure all child names are also defined at each definition point
430 
431  foreach(const CFGNodeToVarNamesMap::value_type& nodeVarsPair, defs)
432  {
433  const CFGNode& definingNode = nodeVarsPair.first;
434  NodeReachingDefTable& nodeLocalDefs = localDefTable[definingNode];
435 
436  foreach(const VarName& definedVarName, nodeVarsPair.second)
437  {
438  //Find all the children of this variable that should also be defined here
439  map<VarName, set<VarName> >::iterator childVars = nameToChildNames.find(definedVarName);
440  if (childVars == nameToChildNames.end())
441  continue;
442 
443  //Go over all the child names and define them here also
444 
445  foreach(const VarName& childName, childVars->second)
446  {
447  ROSE_ASSERT(childName.size() > definedVarName.size());
448  for (size_t i = 0; i < (childName.size() - definedVarName.size()); i++)
449  {
450  //Create a new varName vector that goes from beginning to end - i
451  VarName newName;
452  newName.assign(childName.begin(), childName.end() - i);
453 
454  if (nodeLocalDefs.count(newName) == 0)
455  {
456  ReachingDefPtr expandedDef = ReachingDefPtr(new ReachingDef(definingNode, ReachingDef::EXPANDED_DEF));
457 
458  nodeLocalDefs.insert(make_pair(newName, expandedDef));
459  }
460  }
461  }
462  }
463  }

467 void SSA_UnfilteredCfg::insertDefsForExternalVariables(SgFunctionDefinition* function, const set<VarName>& usedNames)
468 {
469  //The function definition should have 4 indices in the CFG. We insert defs for external variables
470  //At the very first one
471  ROSE_ASSERT(function->cfgIndexForEnd() == 3);
472  CFGNode functionEntryNode = function->cfgForBeginning();
473 
474  NodeReachingDefTable& functionEntryDefs = localDefTable[functionEntryNode];
475 
476  //Iterate over each used variable and check it it is declared outside of the function scope
477 
478  foreach(const VarName& usedVar, usedNames)
479  {
480  ROSE_ASSERT(!usedVar.empty());
481  VarName rootName;
482  rootName.assign(1, usedVar[0]);
483 
484  SgScopeStatement* varScope = SageInterface::getScope(rootName[0]);
485  SgScopeStatement* functionScope = function;
486 
487  //If it is a local variable, there should be a def somewhere inside the function
488  if (varScope == functionScope || SageInterface::isAncestor(functionScope, varScope))
489  {
490  //We still need to insert defs for compiler-generated variables (e.g. __func__), since they don't have defs in the AST
491  if (!isBuiltinVar(rootName))
492  continue;
493  }
494  else if (isSgGlobal(varScope))
495  {
496  //Handle the case of declaring "extern int x" inside the function
497  //Then, x has global scope but it actually has a definition inside the function so we don't need to insert one
498  if (SageInterface::isAncestor(function, rootName[0]))
499  {
500  //When else could a var be declared inside a function and be global?
501  SgVariableDeclaration* varDecl = isSgVariableDeclaration(rootName[0]->get_parent());
502  ROSE_ASSERT(varDecl != NULL);
503  ROSE_ASSERT(varDecl->get_declarationModifier().get_storageModifier().isExtern());
504  continue;
505  }
506  }
507 
508  //Are there any other types of external vars?
509  ROSE_ASSERT(isBuiltinVar(rootName) || isSgClassDefinition(varScope) || isSgNamespaceDefinitionStatement(varScope)
510  || isSgGlobal(varScope) || isSgDeclarationScope(varScope));
511 
512  //The variable is not in local scope; we need to insert a def for it at the function definition
513  for (size_t i = 0; i < usedVar.size(); i++)
514  {
515  //Create a new varName vector that goes from beginning to end - i
516  VarName newName(usedVar.begin(), usedVar.end() - i);
517 
518  if (functionEntryDefs.count(newName) > 0)
519  continue;
520 
521  ReachingDefPtr def = ReachingDefPtr(new ReachingDef(functionEntryNode, ReachingDef::EXTERNAL_DEF));
522  functionEntryDefs.insert(make_pair(newName, def));
523  }
524  }

527 void SSA_UnfilteredCfg::insertPhiFunctions(SgFunctionDefinition* function, const std::vector<CFGNode>& cfgNodesInPostOrder)
528 {
529  if (getDebug())
530  printf("Inserting phi nodes in function %s...\n", function->get_declaration()->get_name().str());
531  ROSE_ASSERT(function != NULL);
532 
533  //First, find all the places where each name is defined
534  map<VarName, vector<CFGNode> > nameToDefNodesMap;
535 
536  foreach(const CFGNode& cfgNode, cfgNodesInPostOrder)
537  {
538  CFGNodeToDefTableMap::const_iterator defTableIter = localDefTable.find(cfgNode);
539  if (defTableIter == localDefTable.end())
540  continue;
541 
542  const NodeReachingDefTable& definitionsAtNode = defTableIter->second;
543 
544  foreach(const NodeReachingDefTable::value_type& varDefPair, definitionsAtNode)
545  {
546  nameToDefNodesMap[varDefPair.first].push_back(cfgNode);
547  }
548  }
549 
550  //Build an iterated dominance frontier for this function
551  map<CFGNode, CFGNode> iPostDominatorMap;
552  map<CFGNode, set<CFGNode> > domFrontiers = ssa_private::calculateDominanceFrontiers<CFGNode, CFGEdge > (function, NULL, &iPostDominatorMap);
553 
554  //Find the phi function locations for each variable
555  map<VarName, vector<CFGNode> >::const_iterator nameToDefNodesIter = nameToDefNodesMap.begin();
556  for (; nameToDefNodesIter != nameToDefNodesMap.end(); ++nameToDefNodesIter)
557  {
558  const VarName& var = nameToDefNodesIter->first;
559  const vector<CFGNode>& definitionPoints = nameToDefNodesIter->second;
560 
561  ROSE_ASSERT(!definitionPoints.empty() && "We have a variable that is not defined anywhere!");
562 
563  //Calculate the iterated dominance frontier
564  set<CFGNode> phiNodes = ssa_private::calculateIteratedDominanceFrontier(domFrontiers, definitionPoints);
565 
566  if (getDebug())
567  printf("Variable %s has phi nodes inserted at\n", varnameToString(var).c_str());
568 
569  foreach(const CFGNode& phiNode, phiNodes)
570  {
571  //We don't want to insert phi defs for functions that have gone out of scope
572  //if (!isVarInScope(var, phiNode.getNode()))
573  // continue;
574  ReachingDefPtr phiDef = ReachingDefPtr(new ReachingDef(phiNode, ReachingDef::PHI_FUNCTION));
575  reachingDefTable[phiNode].insert(make_pair(var, phiDef));
576  }
577  }

580 void SSA_UnfilteredCfg::renumberAllDefinitions(SgFunctionDefinition* /*func*/, const vector<CFGNode>& cfgNodesInPostOrder)
581 {
582  //Map from each name to the next index. Not in map means 0
583  map<VarName, int> nameToNextIndexMap;
584 
585  //We process nodes in reverse postorder; this provides a natural numbering for definitions
586 
587  reverse_foreach(const CFGNode& cfgNode, cfgNodesInPostOrder)
588  {
589  CFGNodeToDefTableMap::const_iterator reachingDefsIter = reachingDefTable.find(cfgNode);
590  if (reachingDefsIter != reachingDefTable.end())
591  {
592  //Iterate over all the phi functions inserted at this node.
593 
594  foreach(const NodeReachingDefTable::value_type& varDefPair, reachingDefsIter->second)
595  {
596  const VarName& definedVar = varDefPair.first;
597  ReachingDefPtr reachingDef = varDefPair.second;
598 
599  //At this point, only phi functions should be in the reaching defs table. Everything else is local defs
600  ROSE_ASSERT(reachingDef->isPhiFunction());
601 
602  //Give an index to the variable
603  int index = 0;
604  if (nameToNextIndexMap.count(definedVar) > 0)
605  {
606  index = nameToNextIndexMap[definedVar];
607  }
608  nameToNextIndexMap[definedVar] = index + 1;
609 
610  reachingDef->setRenamingNumber(index);
611  }
612  }
613 
614  //Number the real (non-phi) definitions
615  CFGNodeToDefTableMap::const_iterator localDefIter = localDefTable.find(cfgNode);
616  if (localDefIter != localDefTable.end())
617  {
618 
619  foreach(const NodeReachingDefTable::value_type& varDefPair, localDefIter->second)
620  {
621  const VarName& definedVar = varDefPair.first;
622  ReachingDefPtr reachingDef = varDefPair.second;
623 
624  //At this point, only phi functions should be in the reaching defs table. Everything else is local defs
625  ROSE_ASSERT(!reachingDef->isPhiFunction());
626 
627  //Give an index to the variable
628  int index = 0;
629  if (nameToNextIndexMap.count(definedVar) > 0)
630  {
631  index = nameToNextIndexMap[definedVar];
632  }
633  nameToNextIndexMap[definedVar] = index + 1;
634 
635  reachingDef->setRenamingNumber(index);
636  }
637  }
638  }
