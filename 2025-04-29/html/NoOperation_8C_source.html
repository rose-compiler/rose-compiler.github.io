57 class CellErasurePredicate: public BaseSemantics::MemoryCell::Predicate {
58  bool ignorePoppedMemory;
59  BaseSemantics::RiscOperators::Ptr ops;
60  BaseSemantics::SValue::Ptr stackCurVa;
61  BaseSemantics::SValue::Ptr stackMinVa;
62 
63 public:
64  CellErasurePredicate(const BaseSemantics::RiscOperators::Ptr &ops, const BaseSemantics::SValue::Ptr &stackCurVa,
65  Address closeness)
66  : ignorePoppedMemory(closeness!=0), ops(ops), stackCurVa(stackCurVa) {
67  stackMinVa = ops->subtract(stackCurVa, ops->number_(stackCurVa->nBits(), closeness));
68  }
69 
70  virtual bool operator()(const BaseSemantics::MemoryCell::Ptr &cell) override {
71  if (cell->getWriters().isEmpty())
72  return true;
73 
74  // Erase memory that is above (lower address) and near the current stack pointer.
75  if (ignorePoppedMemory) {
76  BaseSemantics::SValue::Ptr isPopped = // assume downward-growing stack
77  ops->and_(ops->isUnsignedLessThan(cell->address(), stackCurVa),
78  ops->isUnsignedGreaterThanOrEqual(cell->address(), stackMinVa));
79  return isPopped->isTrue();
80  }
81 
82  return false;
83  }

86 std::string
87 NoOperation::StateNormalizer::toString(const BaseSemantics::Dispatcher::Ptr &cpu, const BaseSemantics::State::Ptr &state_) {
88  BaseSemantics::State::Ptr state = state_;
89  BaseSemantics::RiscOperators::Ptr ops = cpu->operators();
90  if (!state)
91  return "";
92  bool isCloned = false; // do we have our own copy of the state?
93 
94  // If possible and appropriate, remove the instruction pointer register
95  const RegisterDescriptor regIp = cpu->instructionPointerRegister();
96  BaseSemantics::RegisterStateGeneric::Ptr rstate = BaseSemantics::RegisterStateGeneric::promote(state->registerState());
97  if (rstate && rstate->is_partly_stored(regIp)) {
98  BaseSemantics::SValue::Ptr ip = ops->peekRegister(cpu->instructionPointerRegister());
99  if (ip->isConcrete()) {
100  state = state->clone();
101  isCloned = true;
102  rstate = BaseSemantics::RegisterStateGeneric::promote(state->registerState());
103  rstate->erase_register(regIp, ops.get());
104  }
105  }
106 
107  // Get the memory state, cloning the state if not done so above.
108  BaseSemantics::MemoryCellState::Ptr mem = as<BaseSemantics::MemoryCellState>(state->memoryState());
109  if (mem && !isCloned) {
110  state = state->clone();
111  isCloned = true;
112  mem = BaseSemantics::MemoryCellState::promote(state->memoryState());
113  }
114 
115  // Erase memory that has never been written (i.e., cells that sprang into existence by reading an address) of which appears
116  // to have been recently popped from the stack.
117  CellErasurePredicate predicate(ops, ops->peekRegister(cpu->stackPointerRegister()), ignorePoppedMemory_);
118  if (mem)
119  mem->eraseMatchingCells(predicate);
120 
121  BaseSemantics::Formatter fmt;
122  fmt.set_show_latest_writers(false);
123  fmt.set_show_properties(false);
124  std::ostringstream ss;
125  ss <<(*state+fmt);
126  return ss.str();
127 }

221 NoOperation::IndexIntervals
222 NoOperation::findNoopSubsequences(const std::vector<SgAsmInstruction*> &insns) const {
223  IndexIntervals retval;
224  Sawyer::Message::Stream debug(mlog[DEBUG]);
225 
226  if (debug) {
227  debug <<"findNoopSubsequences(\n";
228  for (SgAsmInstruction *insn: insns)
229  debug <<" " <<insn->toString() <<"\n";
230  debug <<")\n";
231  }
232 
233  // If we have no instruction semantics then assume that all instructions have an effect.
234  if (!cpu_ || insns.empty())
235  return retval;
236 
237  // Process each instruction as if insns were a basic block. Store insns[i]'s initial state in states[i] and its final state
238  // in states[i+1]. States don't generally have a way to compare them for equality, so use a simple string-based comparison
239  // for now. FIXME[Robb P. Matzke 2015-05-11]
240  std::vector<std::string> states;
241  bool hadError = false;
242  cpu_->operators()->currentState(initialState(insns.front()));
243  const RegisterDescriptor regIP = cpu_->instructionPointerRegister();
244  try {
245  for (SgAsmInstruction *insn: insns) {
246  cpu_->operators()->writeRegister(regIP, cpu_->operators()->number_(regIP.nBits(), insn->get_address()));
247  states.push_back(normalizeState(cpu_->currentState()));
248  if (debug) {
249  debug <<" normalized state #" <<states.size()-1 <<":\n" <<StringUtility::prefixLines(states.back(), " ");
250  debug <<" instruction: " <<insn->toString() <<"\n";
251  }
252  cpu_->processInstruction(insn);
253  }
254  } catch (const BaseSemantics::Exception &e) {
255  hadError = true;
256  SAWYER_MESG(debug) <<" semantic exception: " <<e <<"\n";
257  }
258  if (!hadError) {
259  states.push_back(normalizeState(cpu_->currentState()));
260  if (debug)
261  debug <<" normalized state #" <<states.size()-1 <<":\n" <<StringUtility::prefixLines(states.back(), " ");
262  }
263 
264  // Ignore terminal branches? I.e., if set, then given a block like "inc eax; jmp x" where x is not the fall-through
265  // address, the JMP is not a no-op. However, if a block consists of only a JMP then it is considered a no-op because the
266  // predecessors could have branched directly to the JMP target instead.
267  //
268  // FIXME[Robb P Matzke 2017-05-31]: We look at the terminal instruction in isolation to find its successors, but maybe a
269  // better way would be to use the instruction pointer register from the state we already computed. Doing so would be a
270  // more accurate way to handle opaque predicates.
271  //
272  // The check for states.size()+1==insns.size() is because if there was an exception above, then there won't be as
273  // many states as instructions and we're in effect already ignoring the state for the last instruction (and possibly
274  // more). There's normally one more state than instructions because we've saved the initial state plus the state
275  // after each instruction.
276  if (ignoreTerminalBranches_ && insns.size() > 1 && states.size() == insns.size() + 1) {
277  bool isComplete = true;
278  AddressSet succs = cpu_->architecture()->getSuccessors(insns.back(), isComplete/*out*/);
279  if (succs.size() > 1 || !isComplete) {
280  states.pop_back();
281  } else if (succs.size() == 1 && succs.least() != insns.back()->get_address() + insns.back()->get_size()) {
282  states.pop_back();
283  }
284  }
285 
286  // Look for pairs of states that are the same, and call that sequence of instructions a no-op
287  for (size_t i=0; i+1<states.size(); ++i) {
288  for (size_t j=i+1; j<states.size(); ++j) {
289  if (states[i]==states[j]) {
290  retval.push_back(IndexInterval::hull(i, j-1));
291  SAWYER_MESG(debug) <<" no-op: " <<i <<".." <<(j-1) <<"\n";
292  }
293  }
294  }
295 
296  return retval;
297 }
