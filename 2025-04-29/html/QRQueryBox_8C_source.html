345 void QRQueryBox::opLoadVariantQueryList() {
346  // GAB(07/04/09): This code was written back in 2005, and I am not sure how
347  // it figures out the SgNode class hierarchy. I had to hack it a bit in order
348  // to work with the current version of ROSE.
349 
350  T_VariantNode root_node(V_SgNumVariants, NULL);
351  T_VariantNode *tSgNode = 0;
352  VariantVector vvector;
353  map<VariantT, T_VariantNode *> variant_nodes;
354  for (int v = 0; v < V_SgNumVariants; v++) {
355  VariantT variant;
356  variant = (VariantT) v;
357  T_VariantNode *node;
358  map<VariantT, T_VariantNode *>::iterator iter = variant_nodes.find(variant);
359  if (iter == variant_nodes.end()) {
360  node = new T_VariantNode(variant, &root_node);
361  variant_nodes[variant] = node;
362  if (variant == V_SgNode) {
363  ROSE_ASSERT(!tSgNode);
364  tSgNode = node;
365  }
366  } else {
367  node = iter->second;
368  }
369  unsigned node_depth = node->getDepth();
370  vvector = VariantVector(variant);
371  T_VariantNode *child_node;
372  VariantT child_variant;
373 
374  for (unsigned i = 1; i < vvector.size(); i++) {
375  child_variant = vvector[i];
376  iter = variant_nodes.find(child_variant);
377  if (iter == variant_nodes.end()) {
378  child_node = new T_VariantNode(child_variant, node);
379  if (child_variant == V_SgNode) {
380  ROSE_ASSERT(!tSgNode);
381  tSgNode = child_node;
382  }
383  variant_nodes[child_variant] = child_node;
384  } else {
385  child_node = iter->second;
386  if (node_depth > child_node->m_parent->getDepth()) {
387  child_node->removeParent();
388  child_node->setParent(node);
389  }
390  }
391  }
392  }
393 
394  ROSE_ASSERT(tSgNode);
395 
396 
397  m_mapOpVariantItem.clear();
398  m_mapOpItemVariant.clear();
399 
400 
401 
402  opLoadVariantQueryListAux(*tSgNode, NULL, NULL);
403  Q3CheckListItem *root_item = m_mapOpVariantItem[tSgNode->m_variant];
404  root_item->setOpen(true);
405 }

408 Q3CheckListItem* QRQueryBox::opLoadVariantQueryListAux(T_VariantNode &node, Q3CheckListItem *ancestor, Q3CheckListItem *last_sibling) {
409  Q3CheckListItem *item = NULL;
410 
411  if (!node.isRoot()) {
412  if (ancestor) {
413  item = new Q3CheckListItem(ancestor, last_sibling, getVariantName(node.m_variant).c_str(), Q3CheckListItem::CheckBoxController);
414  } else {
415  item = new Q3CheckListItem(m_wgOpQueryList, last_sibling, getVariantName(node.m_variant).c_str(), Q3CheckListItem::CheckBoxController);
416  }
417  }
418  m_mapOpVariantItem[node.m_variant] = item;
419  m_mapOpItemVariant[item] = node.m_variant;
420  last_sibling = NULL;
421  for (set<T_VariantNode *>::iterator iter = node.m_children.begin();
422  iter != node.m_children.end(); iter++)
423  {
424  last_sibling = opLoadVariantQueryListAux(**iter, item, last_sibling);
425  }
426 
427  return item;
428

446 // cache modified: need to update DomainCBox and DomainInfo
447 void QRQueryBox::eventCacheModified() {
448  int cache_size = cacheSize();
449 
450  if (cache_size == 0) {
451  m_wgDomainInfo->setText("<empty>");
452  m_wgDomainExpModes->setEnabled(false);
453  m_wgOpPerformQuery->setEnabled(false);
454  } else {
455  m_wgDomainExpModes->setEnabled(true);
456  m_wgOpPerformQuery->setEnabled(true);
457  if (cache_size == 1) {
458  m_wgDomainInfo->setText("1 node");
459  } else {
460  m_wgDomainInfo->setText(QString("%1 nodes").arg(cacheSize()));
461  }
462  }
463 
464  m_wgDomainView->clear();
465  int count = 0;
466 
467  for (set<SgNode *>::iterator iter = m_cache.begin(); iter != m_cache.end(); iter++) {
468  QPixmap *pixmap;
469  SgNode *node = *iter;
470  pixmap = QRAstInfo::getPixmap(node);
471  if (!pixmap) pixmap = QRPixmap::getPixmap(RoseIcons::empty);
472  if (cache_size == 1) {
473  m_wgDomainView->insertItem(*pixmap, QString("%2").arg(QRAstInfo::getInfo(node).c_str()));
474  } else {
475  m_wgDomainView->insertItem(*pixmap, QString("[%1.] %2").arg(++count).arg(QRAstInfo::getInfo(node).c_str()));
476  }
477  }
478  m_domainNeedsToExpand = true;
479 }

671 void QRQueryBox::opSyncDescWithVariantQuery() {
672  if (isOpCustomQueries()) return;
673 
674  // get variant vector from the list
675  VariantVector vector = getOpVariantVectorFromList();
676 
677  bool query_match = false;
678  unsigned matchedId = 0;
679  // traverse the list of builtin queries
680  if (vector.size() > 0) {
681  TypeVariantQueries::iterator iter = m_variantQueries.begin();
682  while (iter != m_variantQueries.end() && !query_match) {
683  pair<string,VariantVector> &item = iter->second;
684  VariantVector &queryVector = item.second;
685  if (vector.size() == queryVector.size()) {
686  bool differentVectors; differentVectors = false;
687  VariantVector::iterator iter1 = vector.begin();
688  while (iter1 != vector.end() && !differentVectors) {
689  VariantVector::iterator iter2 = queryVector.begin();
690  bool foundVariant; foundVariant = false;
691  while (iter2 != queryVector.end() && !foundVariant) {
692  if (*iter1 == *iter2) foundVariant = true; else iter2++;
693  }
694  if (!foundVariant) differentVectors = true; else iter1++;
695  }
696  if (!differentVectors) {
697  query_match = true;
698  matchedId = iter->first;
699  }
700  }
701  iter++;
702  }
703  }
704 
705  if (query_match) {
706  m_wgOpVariantQueryDesc->setCurrentItem(matchedId+1);
707  } else {
708  m_wgOpVariantQueryDesc->setCurrentItem(0);
709  }
710 }
