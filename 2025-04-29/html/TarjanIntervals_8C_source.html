38 // -- TarjanIntervals
39 //
40 // This file contains code to determine the Tarjan intervals (nested
41 // strongly connected regions) of a flow graph. Tarjan's original
42 // algorithm is described in the following article:
43 // Tarjan, R.E. "Testing flow graph reducibility"
44 // J. of Computer & System Sciences 9, 355-365, 1974.
45 //
46 // Extensions to Tarjan's algorithm were made here to handle
47 // reducible SCRs surrounded by or containing irreducible ones.
48 // The extensions are described in the following article:
49 // Havlak, P. "Nesting of reducible and irreducible loops"
50 // ACM TOPLAS, July, 1997.
51 //
52 // The result is returned as a tree of SCRs, where a parent SCR
53 // contains its children. A leaf SCR is a single node, which is not
54 // really a loop unless the isCyclic flag is set.
55 //
56 // Preconditions:
57 // 1. The underlying directed graph has a unique start node,
58 // dominating all others in the graph and a unique exit node,
59 // postdominating all others.
60 //
61 // The underlying node numbering scheme is not important.
62 //
63 // History:
64 // Mar. 1991 (phh):
65 // Initial implementation for the D-System.
66 // Feb. 2002 (John Mellor-Crummey):
67 // Wrapped with C++ classes and cleaned-up.
68 // Feb. 2002 (Jason Eckhardt):
69 // Further cleaned-up the code and rewrote some portions for
70 // incorporation into the OpenAnalysis system.
71 //
72 // A clean re-write in C++ would greatly enhance the style and clarity
73 // of the code.
74 //

141  int prenum; // preorder number

144  short loopIndex; // unique id for intervals

225 //
226 // Sort the Tarjan Tree in topological ordering
227 //
228 // Feb. 2002 (jle) - Removed the call to GetTopologicalMap which computes
229 // a (breadth-first) topological ordering of the CFG. This is completely
230 // unnecessary (and wasteful) since by virtue of performing a DFS as part of
231 // interval building, we have a topological ordering available as a by-product.
232 // Further, the breadth-first version in GetTopologicalMap is clumsy (in this
233 // context) since it has to make call-backs to the tarjan code to determine
234 // if the current edge being examined is a back-edge (to avoid cycling).
235 //
236 void TarjanIntervals::Sort()
237 {
238  RIFGNodeId gId;
239  int parent; // Tarj parent (loop header)
240 
241  // Now disconnect all "next" fields for all tarj nodes
242 
243  RIFGNodeIterator* ni = g.GetNodeIterator(g, RIFG::FORWARD);
244  for ( ; (ni->Current()) != RIFG_NIL; (*ni)++) {
245  gId = ni->Current();
246  int gdfn = dfnum(gId);
247  // gdfn will be invalid if the node gId was unreachable.
248  if (gdfn != INVALID_DFNUM) {
249  TARJ_inners(gdfn) = INVALID_DFNUM;
250  TARJ_next(gdfn) = INVALID_DFNUM;
251  }
252  }
253  delete ni;
254 
255  // The list is sorted in reverse topological order since each child
256  // in the loop below is prepended to the list of children.
257  std::list<RIFGNodeId>::iterator i;
258  for (i = rev_top_list.begin(); i != rev_top_list.end(); i++) {
259  gId = *i;
260  if (gId != RIFG_NIL) {
261  // Add new kid
262  int gdfn = dfnum(gId);
263  parent = TARJ_outer(gdfn);
264  if (parent != INVALID_DFNUM) {
265  TARJ_next(gdfn) = TARJ_inners(parent);
266  TARJ_inners(parent) = gdfn;
267  }
268  }
269  }
270 
271  Renumber();
272 } // end of tarj_sort()

349 //
350 // Do depth first search on control flow graph to
351 // initialize vertex[], dfnum[], last[]
352 //
353 void TarjanIntervals::DFS(RIFGNodeId v)
354 {
355  vertex(n) = v;
356  dfnum(v) = n++;
357 
358  RIFGEdgeId succ;
359  RIFGEdgeIterator* ei = g.GetEdgeIterator(g, v, RIFG::ED_OUTGOING);
360  for (; (ei->Current()) != RIFG_NIL; (*ei)++) {
361  succ = ei->Current();
362  int son = g.GetEdgeSink(succ);
363  if (dfnum(son) == INVALID_DFNUM)
364  DFS(son);
365  }
366  delete ei;
367  //
368  // Equivalent to # of descendants -- number of last descendant
369  //
370  TLast(dfnum(v)) = n-1;
371  rev_top_list.push_back(v);
372 }

375 // Fill in the backPreds and nonBackPreds lists for each node.
376 void TarjanIntervals::FillPredLists()
377 {
378  for (int i = ROOT_DFNUM; i < n; i++) {
379  RIFGEdgeIterator* ei = g.GetEdgeIterator(g, vertex(i), RIFG::ED_INCOMING);
380  for ( ; (ei->Current()) != RIFG_NIL; (*ei)++) {
381  int prednum = dfnum(g.GetEdgeSrc(ei->Current()));
382  if (is_backedge(prednum, i)) {
383  backPreds(i).push_back(prednum);
384  } else {
385  nonBackPreds(i).push_back(prednum);
386  }
387  } // for preds
388  delete ei;
389  } // for nodes
390 }

393 //
394 // In bottom-up traversal of depth-first spanning tree,
395 // determine nested strongly connected regions and flag irreducible regions.
396 // phh, 4 Mar 91
397 //
398 void TarjanIntervals::GetTarjans()
399 {
400  int w; // DFS number of current vertex
401  int firstP, firstQ; // set and worklist
402 
403  //
404  // Following loop should skip root (prenumbered as 0)
405  //
406  for (w = n - 1; w != ROOT_DFNUM; w--) // loop c
407  //
408  // skip any nodes freed or not reachable
409  //
410  if (w != INVALID_DFNUM && g.IsValid(vertex(w))) {
411  firstP = firstQ = INVALID_DFNUM;
412  //
413  // Add sources of cycle arcs to P -- and to worklist Q
414  //
415  std::list<int>::iterator prednum;
416  for (prednum = backPreds(w).begin(); prednum != backPreds(w).end();
417  prednum++) { // loop d
418  int u,v; // vertex names
419  v = *prednum;
420  // ignore predecessors not reachable
421  if (v != INVALID_DFNUM)
422  {
423  if (v == w) {
424  //
425  // Don't add w to its own P and Q sets
426  //
427  isCyclic(w) = true;
428  } else {
429  //
430  // Add FIND(v) to P and Q
431  //
432  u = FIND(v);
433  if (!inP(u)) {
434  nextP(u) = nextQ(u) = firstP;
435  firstP = firstQ = u;
436  inP(u) = true;
437  } // if (!inP(u))
438  } // if (v == w)
439  } // if
440  } // for preds
441 
442  //
443  // P nonempty -> w is header of a loop
444  //
445  if (firstP != INVALID_DFNUM)
446  isCyclic(w) = true;
447 
448  while (firstQ != INVALID_DFNUM) {
449  int x, y, yy; // DFS nums of vertices
450 
451  x = firstQ;
452  firstQ = nextQ(x); // remove x from worklist
453 
454  //
455  // Now look at non-cycle arcs
456  //
457  std::list<int>::iterator prednum;
458  for (prednum = nonBackPreds(x).begin();
459  prednum != nonBackPreds(x).end();
460  prednum++) { // loop d
461  y = *prednum;
462 
463  //
464  // ignore predecessors not reachable
465  //
466  if (y != INVALID_DFNUM)
467  {
468  //
469  // Add FIND(y) to P and Q
470  //
471  yy = FIND(y);
472 
473  if (is_backedge(yy, w)) {
474  if ((!inP(yy)) & (yy != w)) {
475  nextP(yy) = firstP;
476  nextQ(yy) = firstQ;
477  firstP = firstQ = yy;
478  inP(yy) = true;
479  }
480  //
481  // Slight change to published alg'm...
482  // moved setting of header (HIGHPT)
483  // from here to after the union.
484  //
485  } else {
486  //
487  // Irreducible region!
488  //
489  reducible(w) = false;
490 #if 1
491  // FIXME: The DSystem version of the code did not have the
492  // following line. However, this line IS in the 1997 article,
493  // and I believe it is necessary (jle, 03-02-2002).
494  nonBackPreds(w).push_back(yy);
495 #endif
496  }
497  }
498  }
499  }
500  //
501  // now P = P(w) as in Tarjan's paper
502  //
503  while (firstP != INVALID_DFNUM) {
504  //
505  // First line is a change to published algorithm;
506  // Want sources of cycle edges to have header w
507  // and w itself not to have header w.
508  //
509  if ((header(firstP) == ROOT_DFNUM) & (firstP != w))
510  header(firstP) = w;
511  UNION(firstP, w, w);
512  inP(firstP) = false;
513  firstP = nextP(firstP);
514  } // while
515  } // if
516 }

519 //
520 // Traverse df spanning tree, building tree of Tarjan intervals
521 //
522 void TarjanIntervals::Build()
523 {
524  int w; // DFS number of current vertex
525  int outer; // DFS number header of surrounding loop
526 
527  TARJ_nodeid(ROOT_DFNUM) = g.GetRootNode();
528  //
529  // Let the root of the tree be the root of the instance...
530  // Following loop can skip the root (prenumbered 0)
531  //
532  for (w = ROOT_DFNUM + 1; w < n; w++) {
533  RIFGNodeId wnode = vertex(w);
534  //
535  // skip any nodes not in current instance g
536  //
537  if (wnode != RIFG_NIL) {
538  outer = header(w);
539  TARJ_outer(w) = outer;
540  TARJ_nodeid(w) = wnode;
541  //
542  // tarj[w].inners = INVALID_DFNUM; % done in InitArrays
543  //
544  if (isCyclic(w)) {
545  //
546  // Level is deeper than outer if this is a loop
547  //
548  if (reducible(w)) {
549  TARJ_type(w) = RI_TARJ_INTERVAL;
550  TARJ_level(w) = TARJ_level(outer) + 1;
551  } else {
552  TARJ_type(w) = RI_TARJ_IRREDUCIBLE;
553  TARJ_level(w) = TARJ_level(outer);
554  }
555  } else {
556  //
557  // tarj[w].type = RI_TARJ_ACYCLIC; % done in InitArrays
558  //
559  TARJ_level(w) = TARJ_level(outer);
560  }
561  TARJ_next(w) = TARJ_inners(outer);
562  TARJ_inners(outer) = w;
563  }
564  }
565  n = 0;
566  Prenumber(ROOT_DFNUM);
567 }

570 void TarjanIntervals::Prenumber(int v)
571 {
572  int inner;
573 
574  tarj[v].prenum = ++n;
575  last_id = TARJ_nodeid(v);
576 
577  for (inner = TARJ_inners(v); inner != INVALID_DFNUM; inner = TARJ_next(inner)) {
578  Prenumber(inner);
579  }
580 
581  /* tarj[v].last = n; // 3/18/93 RvH: switch to RIFGNodeId last_id */
582  tarj[v].last_id = last_id;
583  tarj[v].last = dfnum(last_id);
584 }

675 //
676 // Return number of loops exited in traversing g edge from src to sink
677 // (0 is normal).
678 //
679 int TarjanIntervals::tarj_exits(RIFGNodeId src, RIFGNodeId sink)
680 {
681  RIFGNodeId lca = LCA(src, sink);
682  if (lca == RIFG_NIL)
683  return 0;
684 
685  int dfn_src = dfnum(src);
686  int dfn_lca = dfnum(lca);
687  return std::max(0, TARJ_level(dfn_src) - TARJ_level(dfn_lca));
688 }

721 // RI_TARJ_ITERATE (back edge)

723 RITarjEdgeType TarjanIntervals::tarj_edge_type(RIFGNodeId src,
724  RIFGNodeId sink)
725 {
726  RIFGNodeId anc = LCA(src, sink);
727  int dfn_sink = dfnum(sink);
728  int dfn_anc = dfnum(anc);
729 
730  if (dfn_anc == dfn_sink) {
731  return RI_TARJ_ITERATE;
732  } else if (TARJ_outer(dfn_sink) != INVALID_DFNUM
733  && (TARJ_type(TARJ_outer(dfn_sink)) == RI_TARJ_IRREDUCIBLE)
734  && (dfn_anc != TARJ_outer(dfn_sink))) {
735  //
736  // Entering irreducible region not through the "header"

738  return RI_TARJ_IRRED_ENTRY;
739  } else {

743  case RI_TARJ_IRREDUCIBLE:

747  return RI_TARJ_IRRED_ENTRY;
748  case RI_TARJ_ACYCLIC:
749  case RI_TARJ_NOTHING:

752  }
