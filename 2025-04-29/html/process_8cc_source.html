96  ProgramTreeList_t trees;
97  for (FilenameList_t::const_iterator i = filenames.begin ();

99  ++i)
100  {
101  cerr << "[Reading profile: " << (*i) << "]" << endl;

167 void
168 PathTranslator::visit (TreeParamPtr_t tree)
169 {
170  File* filenode = dynamic_cast<File *> (tree->value);
171  if (filenode != NULL)
172  {
173  // Save file nodes for later use
174  RoseHPCT::profFileNodes_.insert(filenode);
175  visitFile (filenode);
176  }
177  else
178  { // for other located nodes, fix up file node pointer if necessary
179  Located * locatedNode= dynamic_cast<Located*> (tree->value);
180  if (locatedNode)
181  {
182  Statement* s = dynamic_cast<Statement*> (tree->value);
183  // Save non-scope statement nodes for later use
184  if (s)
185  {
186  RoseHPCT::profStmtNodes_.insert(s);
187  //Accumulate wall clock metrics so it can be used later on to calculate percentage
188  if (!gprof_only) // gprof provides the percentage so no need to calculate here.
189  wallclk_sum += s->getMetricValue(m_wallclock);
190  }
191  if (locatedNode->getFileNode()==NULL)
192  {
193  ROSE_ASSERT(file_ != NULL);
194  locatedNode->setFileNode(file_);
195  }
196  else
197  {// if it has a file node pointer, double check it here
198  ROSE_ASSERT(file_== locatedNode->getFileNode());
199  }
200  }
201  }
202 }

204 void
205 PathTranslator::visitFile (File* node)
206 {
207  file_ = node; // save this node
208  string filename = node->getName ();
209  // find a matching entry for the eqpaths
210 #if 1
211  // Liao, 4/28/2009 we extend this function to replace a root dir of a source tree to another
212  // for example: /root1/smg2000/file1 --> /root2/smg2000/file1
213  // The original function only deals with full path replacement, not a portion of it(root path),
214  // which is less useful in practice.
215  EquivPathMap_t::const_iterator iter = eqpaths_.begin();
216  for (;iter!=eqpaths_.end(); iter++)
217  {
218  string oldpath = iter->first;
219  string newpath = iter->second;
220  //replace the matching old root path of a file with the new root path
221  size_t pos1= filename.find(oldpath);
222  if (pos1==0) // ensure it is the root path
223  {
224  if (enable_debug)
225  cout<<"Replacing a file path from:"<<filename;
226  filename.replace(pos1,oldpath.size(), newpath);
227  if (enable_debug)
228  {
229  cout<<" to: "<<filename<<endl;
230  }
231  node->setName (filename);
232  }
233  }
234 #else
235  string dirname = getDirname (filename);
236  EquivPathMap_t::const_iterator ep = eqpaths_.find (dirname);
237  if (ep != eqpaths_.end ())
238  {
239  // replace the file path with the second path of the entry
240  string basename = getBaseFilename (filename);
241  string new_filename = ep->second + "/" + basename;
242  node->setName (new_filename);
243  }
244 #endif
245 }
