37 rgb_from_hue(Component p, Component q, Component t)
38 {
39  if (t<0)
40  t += 1.0;
41  if (t>1)
42  t -= 1.0;
43  if (t < 1/6.0)
44  return p + (q-p) * 6 * t;
45  if (t < 0.5)
46  return q;
47  if (t < 2/3.0)
48  return p + (q-p) * (2/3.0 - t) * 6;
49  return p;

67 HSV::HSV(const RGB &rgb) {
68  Component hi = std::max(std::max(rgb.r(), rgb.g()), rgb.b());
69  Component lo = std::min(std::min(rgb.r(), rgb.g()), rgb.b());
70  a_ = rgb.a();
71  v_ = (hi+lo)/2.0;
72  if (hi==lo) {
73  h_ = s_ = 0.0;
74  } else {
75  Component delta = hi - lo;
76  s_ = v_ > 0.5 ? delta / (2.0-hi-lo) : delta / (hi+lo);
77  if (hi == rgb.r()) {
78  h_ = (rgb.g()-rgb.b())/delta + (rgb.g()<rgb.b() ? 6 : 0);
79  } else if (hi == rgb.g()) {
80  h_ = (rgb.b()-rgb.r())/delta + 2;
81  } else {
82  h_ = (rgb.r()-rgb.g())/delta + 4;
83  }
84  h_ /= 6.0;
85  }
