21 /*
22  This file is a self-contained interactive indenter for C
23  derivatives, in particular C++.
24 
25  The general problem of indenting a C++ program is ill posed. On the
26  one hand, an indenter has to analyze programs written in a
27  free-form formal language that is best described in terms of
28  tokens, not characters, not lines. On the other hand, indentation
29  applies to lines and white space characters matter, and otherwise
30  the programs to indent are formally invalid in general, as they are
31  begin edited.
32 
33  The approach taken here works line by line. We receive a program
34  consisting of N lines or more, and we want to compute the
35  indentation appropriate for the Nth line. Lines beyond the Nth
36  lines are of no concern to us, so for simplicity we pretend the
37  program has exactly N lines and we call the Nth line the "bottom
38  line". Typically, we have to indent the bottom line when it's still
39  empty, so we concentrate our analysis on the N - 1 lines that
40  precede.
41 
42  By inspecting the (N - 1)-th line, the (N - 2)-th line, ...
43  backwards, we determine the kind of the bottom line and indent it
44  accordingly.
45 
46  * The bottom line is a comment line. See
47  bottomLineStartsInCComment() and
48  indentWhenBottomLineStartsInCComment().
49  * The bottom line is a continuation line. See isContinuationLine()
50  and indentForContinuationLine().
51  * The bottom line is a standalone line. See
52  indentForStandaloneLine().
53 
54  Certain tokens that influence the indentation, notably braces, are
55  looked for in the lines. This is done by simple string comparison,
56  without a real tokenizer. Confusing constructs such as comments and
57  string literals are removed beforehand.
58 */

116 /*
117  Returns the first non-space character in the string t, or
118  QChar::null if the string is made only of white space.
119 */
120 static QChar firstNonWhiteSpace( const QString& t )
121 {
122  int i = 0;
123  while ( i < (int) t.length() ) {
124  if ( !t[i].isSpace() )
125  return t[i];
126  i++;
127  }
128  return QChar::null;
129 }

131 /*
132  Returns TRUE if string t is made only of white space; otherwise
133  returns FALSE.
134 */
135 static bool isOnlyWhiteSpace( const QString& t )
136 {
137  return firstNonWhiteSpace( t ).isNull();
138 }

140 /*
141  Assuming string t is a line, returns the column number of a given
142  index. Column numbers and index are identical for strings that don't
143  contain '\t's.
144 */
145 static int columnForIndex( const QString& t, int index )
146 {
147  int col = 0;
148  if ( index > (int) t.length() )
149  index = t.length();
150 
151  for ( int i = 0; i < index; i++ ) {
152  if ( t[i].latin1() == '\t' ) {
153  col = ( (col / ppHardwareTabSize) + 1 ) * ppHardwareTabSize;
154  } else {
155  col++;
156  }
157  }
158  return col;
159 }

161 /*
162  Returns the indentation size of string t.
163 */
164 static int indentOfLine( const QString& t )
165 {
166  return columnForIndex( t, t.find(firstNonWhiteSpace(t)) );
167 }

169 /*
170  Replaces t[k] by ch, unless t[k] is '\t'. Tab characters are better
171  left alone since they break the "index equals column" rule. No
172  provisions are taken against '\n' or '\r', which shouldn't occur in
173  t anyway.
174 */
175 static inline void eraseChar( QString& t, int k, QChar ch )
176 {
177  if ( t[k] != '\t' )
178  t[k] = ch;
179 }

181 /*
182  Removes some nefast constructs from a code line and returns the
183  resulting line.
184 */
185 static QString trimmedCodeLine( const QString& t )
186 {
187  QString trimmed = t;
188  int k;
189 
190  /*
191  Replace character and string literals by X's, since they may
192  contain confusing characters (such as '{' and ';'). "Hello!" is
193  replaced by XXXXXXXX. The literals are rigourously of the same
194  length before and after; otherwise, we would break alignment of
195  continuation lines.
196  */
197  k = 0;
198  while ( (k = trimmed.find(*literal, k)) != -1 ) {
199  for ( int i = 0; i < literal->matchedLength(); i++ )
200  eraseChar( trimmed, k + i, QChar('X') );
201  k += literal->matchedLength();
202  }
203 
204  /*
205  Replace inline C-style comments by spaces. Other comments are
206  handled elsewhere.
207  */
208  k = 0;
209  while ( (k = trimmed.find(*inlineCComment, k)) != -1 ) {
210  for ( int i = 0; i < inlineCComment->matchedLength(); i++ )
211  eraseChar( trimmed, k + i, QChar(' ') );
212  k += inlineCComment->matchedLength();
213  }
214 
215  /*
216  Replace case and goto labels by spaces, to allow esoteric
217  alignments:
218 
219  foo1: foo2: bar1;
220  bar2;
221  */
222  while ( trimmed.findRev(QChar(':')) != -1 && trimmed.find(*label) != -1 ) {
223  QString cap1 = label->cap( 1 );
224  int pos1 = label->pos( 1 );
225  for ( int i = 0; i < (int) cap1.length(); i++ )
226  eraseChar( trimmed, pos1 + i, QChar(' ') );
227  }
228 
229  /*
230  Remove C++-style comments.
231  */
232  k = trimmed.find( QString("//") );
233  if ( k != -1 )
234  trimmed.truncate( k );
235 
236  return trimmed;
237 }

239 /*
240  Returns '(' if the last parenthesis is opening, ')' if it is
241  closing, and QChar::null if there are no parentheses in t.
242 */
243 static inline QChar lastParen( const QString& t )
244 {
245  int i = t.length();
246  while ( i > 0 ) {
247  i--;
248  if ( t[i] == QChar('(') || t[i] == QChar(')') )
249  return t[i];
250  }
251  return QChar::null;
252 }

264  The "linizer" is a group of functions and variables to iterate
265  through the source code of the program to indent. The program is
266  given as a list of strings, with the bottom line being the line to

268  uninteresting and not passed over to us.

272 {
273  QString line;
274  int braceDepth;

277  QStringList::ConstIterator iter;
278  bool inCComment;
279  bool pendingRightBrace;

299 /*
300  Advances to the previous line in yyProgram and update yyLine
301  accordingly. yyLine is cleaned from comments and other damageable
302  constructs. Empty lines are skipped.
303 */
304 static bool readLine()
305 {
306  int k;
307 
308  yyLinizerState->leftBraceFollows =
309  ( firstNonWhiteSpace(yyLinizerState->line) == QChar('{') );
310 
311  do {
312  if ( yyLinizerState->iter == yyProgram->begin() ) {
313  yyLinizerState->line = QString::null;
314  return FALSE;
315  }
316 
317  --yyLinizerState->iter;
318  yyLinizerState->line = *yyLinizerState->iter;
319 
320  yyLinizerState->line = trimmedCodeLine( yyLinizerState->line );
321 
322  /*
323  Remove C-style comments that span multiple lines. If the
324  bottom line starts in a C-style comment, we are not aware
325  of that and eventually yyLine will contain a slash-aster.
326 
327  Notice that both if's can be executed, since
328  yyLinizerState->inCComment is potentially set to FALSE in
329  the first if. The order of the if's is also important.
330  */
331 
332  if ( yyLinizerState->inCComment ) {
333  QString slashAster( "/*" );
334 
335  k = yyLinizerState->line.find( slashAster );
336  if ( k == -1 ) {
337  yyLinizerState->line = QString::null;
338  } else {
339  yyLinizerState->line.truncate( k );
340  yyLinizerState->inCComment = FALSE;
341  }
342  }
343 
344  if ( !yyLinizerState->inCComment ) {
345  QString asterSlash( "*/" );
346 
347  k = yyLinizerState->line.find( asterSlash );
348  if ( k != -1 ) {
349  for ( int i = 0; i < k + 2; i++ )
350  eraseChar( yyLinizerState->line, i, QChar(' ') );
351  yyLinizerState->inCComment = TRUE;
352  }
353  }
354 
355  /*
356  Remove preprocessor directives.
357  */
358  k = 0;
359  while ( k < (int) yyLinizerState->line.length() ) {
360  QChar ch = yyLinizerState->line[k];
361  if ( ch == QChar('#') ) {
362  yyLinizerState->line = QString::null;
363  } else if ( !ch.isSpace() ) {
364  break;
365  }
366  k++;
367  }
368 
369  /*
370  Remove trailing spaces.
371  */
372  k = yyLinizerState->line.length();
373  while ( k > 0 && yyLinizerState->line[k - 1].isSpace() )
374  k--;
375  yyLinizerState->line.truncate( k );
376 
377  /*
378  '}' increment the brace depth and '{' decrements it and not
379  the other way around, as we are parsing backwards.
380  */
381  yyLinizerState->braceDepth +=
382  (int) (bool) yyLinizerState->line.contains( QChar('}') ) -
383  (int) (bool) yyLinizerState->line.contains( QChar('{') );
384 
385  /*
386  We use a dirty trick for
387 
388  } else ...
389 
390  We don't count the '}' yet, so that it's more or less
391  equivalent to the friendly construct
392 
393  }
394  else ...
395  */
396  if ( yyLinizerState->pendingRightBrace )
397  yyLinizerState->braceDepth++;
398  yyLinizerState->pendingRightBrace =
399  ( yyLinizerState->line.find(*braceX) == 0 );
400  if ( yyLinizerState->pendingRightBrace )
401  yyLinizerState->braceDepth--;
402  } while ( yyLinizerState->line.isEmpty() );
403 
404  return TRUE;
405 }

407 /*
408  Resets the linizer to its initial state, with yyLine containing the
409  line above the bottom line of the program.
410 */
411 static void startLinizer()
412 {
413  yyLinizerState->braceDepth = 0;
414  yyLinizerState->inCComment = FALSE;
415  yyLinizerState->pendingRightBrace = FALSE;
416 
417  yyLine = &yyLinizerState->line;
418  yyBraceDepth = &yyLinizerState->braceDepth;
419  yyLeftBraceFollows = &yyLinizerState->leftBraceFollows;
420 
421  yyLinizerState->iter = yyProgram->end();
422  --yyLinizerState->iter;
423  yyLinizerState->line = *yyLinizerState->iter;
424  readLine();
425 }

427 /*
428  Returns TRUE if the start of the bottom line of yyProgram (and
429  potentially the whole line) is part of a C-style comment; otherwise
430  returns FALSE.
431 */
432 static bool bottomLineStartsInCComment()
433 {
434  QString slashAster( "/*" );
435  QString asterSlash( "*/" );
436 
437  /*
438  We could use the linizer here, but that would slow us down
439  terribly. We are better to trim only the code lines we need.
440  */
441  QStringList::ConstIterator p = yyProgram->end();
442  --p; // skip bottom line
443 
444  for ( int i = 0; i < BigRoof; i++ ) {
445  if ( p == yyProgram->begin() )
446  return FALSE;
447  --p;
448 
449  if ( (*p).find(slashAster) != -1 ||
450  (*p).find(asterSlash) != -1 ) {
451  QString trimmed = trimmedCodeLine( *p );
452 
453  if ( trimmed.find(slashAster) != -1 ) {
454  return TRUE;
455  } else if ( trimmed.find(asterSlash) != -1 ) {
456  return FALSE;
457  }
458  }
459  }
460  return FALSE;
461 }

463 /*
464  Returns the recommended indent for the bottom line of yyProgram
465  assuming that it starts in a C-style comment, a condition that is
466  tested elsewhere.
467 
468  Essentially, we're trying to align against some text on the previous
469  line.
470 */
471 static int indentWhenBottomLineStartsInCComment()
472 {
473  int k = yyLine->findRev( QString("/*") );
474  if ( k == -1 ) {
475  /*
476  We found a normal text line in a comment. Align the
477  bottom line with the text on this line.
478  */
479  return indentOfLine( *yyLine );
480  } else {
481  /*
482  The C-style comment starts on this line. If there is
483  text on the same line, align with it. Otherwise, align
484  with the slash-aster plus a given offset.
485  */
486  int indent = columnForIndex( *yyLine, k );
487  k += 2;
488  while ( k < (int) yyLine->length() ) {
489  if ( !(*yyLine)[k].isSpace() )
490  return columnForIndex( *yyLine, k );
491  k++;
492  }
493  return indent + ppCommentOffset;
494  }
495 }

497 /*
498  A function called match...() modifies the linizer state. If it
499  returns TRUE, yyLine is the top line of the matched construct;
500  otherwise, the linizer is left in an unknown state.
501 
502  A function called is...() keeps the linizer state intact.
503 */

505 /*
506  Returns TRUE if the current line (and upwards) forms a braceless
507  control statement; otherwise returns FALSE.
508 
509  The first line of the following example is a "braceless control
510  statement":
511 
512  if ( x )
513  y;
514 */
515 static bool matchBracelessControlStatement()
516 {
517  int delimDepth = 0;
518 
519  if ( yyLine->endsWith(QString("else")) )
520  return TRUE;
521 
522  if ( !yyLine->endsWith(QChar(')')) )
523  return FALSE;
524 
525  for ( int i = 0; i < SmallRoof; i++ ) {
526  int j = yyLine->length();
527  while ( j > 0 ) {
528  j--;
529  QChar ch = (*yyLine)[j];
530 
531  switch ( ch.unicode() ) {
532  case ')':
533  delimDepth++;
534  break;
535  case '(':
536  delimDepth--;
537  if ( delimDepth == 0 ) {
538  if ( yyLine->find(*iflikeKeyword) != -1 ) {
539  /*
540  We have
541 
542  if ( x )
543  y
544 
545  "if ( x )" is not part of the statement
546  "y".
547  */
548  return TRUE;
549  }
550  }
551  if ( delimDepth == -1 ) {
552  /*
553  We have
554 
555  if ( (1 +
556  2)
557 
558  and not
559 
560  if ( 1 +
561  2 )
562  */
563  return FALSE;
564  }
565  break;
566  case '{':
567  case '}':
568  case ';':
569  /*
570  We met a statement separator, but not where we
571  expected it. What follows is probably a weird
572  continuation line. Be careful with ';' in for,
573  though.
574  */
575  if ( ch != QChar( ';' ) || delimDepth == 0 )
576  return FALSE;
577  }
578  }
579 
580  if ( !readLine() )
581  break;
582  }
583  return FALSE;
584 }

586 /*
587  Returns TRUE if yyLine is an unfinished line; otherwise returns
588  FALSE.
589 
590  In many places, we'll use the terms "standalone line", "unfinished
591  line" and "continuation line". The meaning of these should be
592  evident:
593 
594  a = b; // standalone line
595  c = d + // unfinished line
596  e + // unfinished continuation line
597  f + // unfinished continuation line
598  g; // continuation line
599 */
600 static bool isUnfinishedLine()
601 {
602  bool unf = FALSE;
603 
604  YY_SAVE();
605 
606  if ( yyLine->isEmpty() )
607  return FALSE;
608 
609  QChar lastCh = (*yyLine)[(int) yyLine->length() - 1];
610  if ( QString("{};").find(lastCh) == -1 ) {
611  /*
612  It doesn't end with ';' or similar. If it's not "Q_OBJECT"
613  nor "if ( x )", it must be an unfinished line.
614  */
615  /* qmake ignore Q_OBJECT */
616  unf = ( yyLine->contains(QString("Q_OBJECT")) == 0 &&
617  !matchBracelessControlStatement() );
618  } else if ( lastCh == QChar(';') ) {
619  if ( lastParen(*yyLine) == QChar('(') ) {
620  /*
621  Exception:
622 
623  for ( int i = 1; i < 10;
624  */
625  unf = TRUE;
626  } else if ( readLine() && yyLine->endsWith(QChar(';')) &&
627  lastParen(*yyLine) == QChar('(') ) {
628  /*
629  Exception:
630 
631  for ( int i = 1;
632  i < 10;
633  */
634  unf = TRUE;
635  }
636  }
637 
638  YY_RESTORE();
639  return unf;
640 }

642 /*
643  Returns TRUE if yyLine is a continuation line; otherwise returns
644  FALSE.
645 */
646 static bool isContinuationLine()
647 {
648  bool cont = FALSE;
649 
650  YY_SAVE();
651  if ( readLine() )
652  cont = isUnfinishedLine();
653  YY_RESTORE();
654  return cont;
655 }

657 /*
658  Returns the recommended indent for the bottom line of yyProgram,
659  assuming it's a continuation line.
660 
661  We're trying to align the continuation line against some parenthesis
662  or other bracked left opened on a previous line, or some interesting
663  operator such as '='.
664 */
665 static int indentForContinuationLine()
666 {
667  int braceDepth = 0;
668  int delimDepth = 0;
669 
670  bool leftBraceFollowed = *yyLeftBraceFollows;
671 
672  for ( int i = 0; i < SmallRoof; i++ ) {
673  int hook = -1;
674 
675  int j = yyLine->length();
676  while ( j > 0 && hook < 0 ) {
677  j--;
678  QChar ch = (*yyLine)[j];
679 
680  switch ( ch.unicode() ) {
681  case ')':
682  case ']':
683  delimDepth++;
684  break;
685  case '}':
686  braceDepth++;
687  break;
688  case '(':
689  case '[':
690  delimDepth--;
691  /*
692  An unclosed delimiter is a good place to align at,
693  at least for some styles (including Trolltech's).
694  */
695  if ( delimDepth == -1 )
696  hook = j;
697  break;
698  case '{':
699  braceDepth--;
700  /*
701  A left brace followed by other stuff on the same
702  line is typically for an enum or an initializer.
703  Such a brace must be treated just like the other
704  delimiters.
705  */
706  if ( braceDepth == -1 ) {
707  if ( j < (int) yyLine->length() - 1 ) {
708  hook = j;
709  } else {
710  return 0; // shouldn't happen
711  }
712  }
713  break;
714  case '=':
715  /*
716  An equal sign is a very natural alignment hook
717  because it's usually the operator with the lowest
718  precedence in statements it appears in. Case in
719  point:
720 
721  int x = 1 +
722  2;
723 
724  However, we have to beware of constructs such as
725  default arguments and explicit enum constant
726  values:
727 
728  void foo( int x = 0,
729  int y = 0 );
730 
731  And not
732 
733  void foo( int x = 0,
734  int y = 0 );
735 
736  These constructs are caracterized by a ',' at the
737  end of the unfinished lines or by unbalanced
738  parentheses.
739  */
740  if ( j == 0 || QString("!=<>").find((*yyLine)[j - 1]) == -1 ) {
741  if ( braceDepth == 0 && delimDepth == 0 &&
742  j < (int) yyLine->length() - 1 &&
743  !yyLine->endsWith(QChar(',')) &&
744  (yyLine->contains(QChar('(')) ==
745  yyLine->contains(QChar(')'))) )
746  hook = j;
747  }
748  }
749  }
750 
751  if ( hook >= 0 ) {
752  /*
753  Yes, we have a delimiter or an operator to align
754  against! We don't really align against it, but rather
755  against the following token, if any. In this example,
756  the following token is "11":
757 
758  int x = ( 11 +
759  2 );
760 
761  If there is no such token, we use a continuation indent:
762 
763  static QRegExp foo( QString(
764  "foo foo foo foo foo foo foo foo foo") );
765  */
766  hook++;
767  while ( hook < (int) yyLine->length() ) {
768  if ( !(*yyLine)[hook].isSpace() )
769  return columnForIndex( *yyLine, hook );
770  hook++;
771  }
772  return indentOfLine( *yyLine ) + ppContinuationIndentSize;
773  }
774 
775  if ( braceDepth != 0 )
776  break;
777 
778  /*
779  The line's delimiters are balanced. It looks like a
780  continuation line or something.
781  */
782  if ( delimDepth == 0 ) {
783  if ( isContinuationLine() || leftBraceFollowed ) {
784  /*
785  We have
786 
787  x = 1 +
788  2 +
789  3;
790 
791  or
792 
793  int main()
794  {
795 
796  The "3;" should fall right under the "2;", and the
797  "{" under the "int".
798  */
799  return indentOfLine( *yyLine );
800  } else {
801  /*
802  We have
803 
804  stream << 1 +
805  2;
806 
807  We could, but we don't, try to analyze which
808  operator has precedence over which and so on, in
809  which case we could give the excellent result
810 
811  stream << 1 +
812  2;
813 
814  (We do have a special trick above for the
815  assignment operator above, though.)
816  */
817  return indentOfLine( *yyLine ) + ppContinuationIndentSize;
818  }
819  }
820 
821  if ( !readLine() )
822  break;
823  }
824  return 0;
825 }

827 /*
828  Returns the recommended indent for the bottom line of yyProgram if
829  that line is standalone (or should be indented likewise).
830 
831  Indenting a standalone line is tricky, mostly because of braceless
832  control statements. Grossly, we are looking backwards for a special
833  line, a "hook line", that we can use as a starting point to indent,
834  and then modify the indentation level according to the braces met
835  along the way to that hook.
836 
837  Let's consider a few examples. In all cases, we want to indent the
838  bottom line.
839 
840  Example 1:
841 
842  x = 1;
843  y = 2;
844 
845  The hook line is "x = 1;". We met 0 opening braces and 0 closing
846  braces. Therefore, "y = 2;" inherits the indent of "x = 1;".
847 
848  Example 2:
849 
850  if ( x ) {
851  y;
852 
853  The hook line is "if ( x ) {". No matter what precedes it, "y;" has
854  to be indented one level deeper than the hook line, since we met one
855  opening brace along the way.
856 
857  Example 3:
858 
859  if ( a )
860  while ( b ) {
861  c;
862  }
863  d;
864 
865  To indent "d;" correctly, we have to go as far as the "if ( a )".
866  Compare with
867 
868  if ( a ) {
869  while ( b ) {
870  c;
871  }
872  d;
873 
874  Still, we're striving to go back as little as possible to accomodate
875  people with irregular indentation schemes. A hook line near at hand
876  is much more reliable than a remote one.
877 */
878 static int indentForStandaloneLine()
879 {
880  for ( int i = 0; i < SmallRoof; i++ ) {
881  if ( !*yyLeftBraceFollows ) {
882  YY_SAVE();
883 
884  if ( matchBracelessControlStatement() ) {
885  /*
886  The situation is this, and we want to indent "z;":
887 
888  if ( x &&
889  y )
890  z;
891 
892  yyLine is "if ( x &&".
893  */
894  return indentOfLine( *yyLine ) + ppIndentSize;
895  }
896  YY_RESTORE();
897  }
898 
899  if ( yyLine->endsWith(QChar(';')) ||
900  yyLine->contains(QChar('{')) > 0 ) {
901  /*
902  The situation is possibly this, and we want to indent
903  "z;":
904 
905  while ( x )
906  y;
907  z;
908 
909  We return the indent of "while ( x )". In place of "y;",
910  any arbitrarily complex compound statement can appear.
911  */
912 
913  if ( *yyBraceDepth > 0 ) {
914  do {
915  if ( !readLine() )
916  break;
917  } while ( *yyBraceDepth > 0 );
918  }
919 
920  LinizerState hookState;
921 
922  if ( *yyBraceDepth == 0 ) {
923  while ( isContinuationLine() )
924  readLine();
925  hookState = *yyLinizerState;
926 
927  readLine();
928  if ( *yyBraceDepth == 0 ) {
929  do {
930  if ( !matchBracelessControlStatement() )
931  break;
932  hookState = *yyLinizerState;
933  } while ( readLine() );
934  }
935  } else {
936  hookState = *yyLinizerState;
937  }
938 
939  *yyLinizerState = hookState;
940 
941  while ( isContinuationLine() )
942  readLine();
943 
944  /*
945  Never trust lines containing only '{' or '}', as some
946  people (Richard Stallman) format them weirdly.
947  */
948  if ( yyLine->stripWhiteSpace().length() > 1 )
949  return indentOfLine( *yyLine ) - *yyBraceDepth * ppIndentSize;
950  }
951 
952  if ( !readLine() )
953  break;
954  }
955  return 0;
956 }

988 /*
989  Returns the recommended indent for the bottom line of program.
990  Unless null, typedIn stores the character of yyProgram that
991  triggered reindentation.
992 
993  This function works better if typedIn is set properly; it is
994  slightly more conservative if typedIn is completely wild, and
995  slighly more liberal if typedIn is always null. The user might be
996  annoyed by the liberal behavior.
997 */
998 int indentForBottomLine( const QStringList& program, QChar typedIn )
999 {
1000  if ( program.isEmpty() )
1001  return 0;
1002 
1003  initializeIndenter();
1004 
1005  yyProgram = new QStringList( program );
1006  startLinizer();
1007 
1008  const QString& bottomLine = program.last();
1009  QChar firstCh = firstNonWhiteSpace( bottomLine );
1010  int indent;
1011 
1012  if ( bottomLineStartsInCComment() ) {
1013  /*
1014  The bottom line starts in a C-style comment. Indent it
1015  smartly, unless the user has already played around with it,
1016  in which case it's better to leave her stuff alone.
1017  */
1018  if ( isOnlyWhiteSpace(bottomLine) ) {
1019  indent = indentWhenBottomLineStartsInCComment();
1020  } else {
1021  indent = indentOfLine( bottomLine );
1022  }
1023  } else if ( okay(typedIn, QChar('#')) && firstCh == QChar('#') ) {
1024  /*
1025  Preprocessor directives go flush left.
1026  */
1027  indent = 0;
1028  } else {
1029  if ( isUnfinishedLine() ) {
1030  indent = indentForContinuationLine();
1031  } else {
1032  indent = indentForStandaloneLine();
1033  }
1034 
1035  if ( okay(typedIn, QChar('}')) && firstCh == QChar('}') ) {
1036  /*
1037  A closing brace is one level more to the left than the
1038  code it follows.
1039  */
1040  indent -= ppIndentSize;
1041  } else if ( okay(typedIn, QChar(':')) ) {
1042  QRegExp caseLabel( QString(
1043  "\\s*(?:case\\b[^:]+|default\\s+):\\s*") );
1044 
1045  if ( caseLabel.exactMatch(bottomLine) ) {
1046  /*
1047  Move a case label one level to the left, but only
1048  if the user did not play around with it yet. Some
1049  users have exotic tastes in the matter, and most
1050  users probably are not patient enough to wait for
1051  the final ':' to format their code properly.
1052 
1053  We don't attempt the same for goto labels, as the
1054  user might be in the middle of "foo::bar".
1055  */
1056  if ( indentOfLine(bottomLine) <= indent )
1057  indent -= ppIndentSize;
1058  else
1059  indent = indentOfLine( bottomLine );
1060  }
1061  }
1062  }
1063  delete yyProgram;
1064  terminateIndenter();
1065  return QMAX( 0, indent );
1066 }
