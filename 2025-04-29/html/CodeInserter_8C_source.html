69  ASSERT_not_null(bb);
70  ASSERT_require(startIdx < bb->nInstructions());
71  ASSERT_require(startIdx + nDeleted <= bb->nInstructions());
72  InstructionInfoMap retval;
73 
74  // Instructions before the insertion point
75  for (size_t i = 0; i < startIdx; ++i) {

77  retval.insert(relativeIndex, InstructionInfo(bb->instructions()[i]));

89 bool
90 CodeInserter::replaceBlockInsns(const P2::BasicBlock::Ptr &bb, size_t index, size_t nInsns,
91  std::vector<uint8_t> replacement, const std::vector<Relocation> &relocations) {
92  ASSERT_not_null(bb);
93  ASSERT_require(index + nInsns <= bb->nInstructions());
94  if (0 == nInsns && replacement.empty())
95  return true;
96 
97  const size_t insertAt = index;
98  const size_t nDeletes = nInsns;
99  Sawyer::Message::Stream debug(mlog[DEBUG]);
100 
101  // Where are the instructions located that are to be removed? Be careful: the instructions of a basic block need not be in
102  // address order, and need not be contiguous in memory.
103  InstructionInfoMap insnInfoMap = computeInstructionInfoMap(bb, insertAt, nDeletes);
104  std::vector<SgAsmInstruction*> toReplace(bb->instructions().begin() + index, bb->instructions().begin() + index + nInsns);
105  AddressIntervalSet toReplaceVas = instructionLocations(toReplace);
106 
107  if (debug) {
108  debug <<"replaceBlockInsns:\n";
109  debug <<" basic block with insertion point and deletions:\n";
110  if (Unparser::Base::Ptr unparser = partitioner_->unparser()) {
111  unparser->settings().function.cg.showing = false;
112  unparser->settings().insn.stackDelta.showing = false;
113  for (size_t i=0; i<bb->nInstructions(); ++i) {
114  debug <<" " <<(i==index ? "--> " : " ") <<(i>=index && i<index+nInsns ? " X " : " ");
115  unparser->unparse(debug, partitioner_, bb->instructions()[i]);
116  debug <<"\n";
117  }
118  }
119  debug <<" replacing " <<StringUtility::plural(nInsns, "instructions") <<" with [";
120  for (uint8_t byte: replacement)
121  Diagnostics::mfprintf(debug)(" 0x%02x", byte);
122  debug <<" ], " <<StringUtility::plural(replacement.size(), "bytes") <<"\n";
123  }
124 
125  // Try to insert the replacement, enlarging the replacement set with subsequent then previous instructions.
126  size_t relocStart = 0;
127  while (true) {
128  if (!toReplace.empty()) {
129  // Find largest contiguous region of to-be-replaced instructions starting at to-be-replaced entry instruction.
130  Address toReplaceVa = toReplace[0]->get_address();
131  AddressIntervalSet::ConstIntervalIterator found = toReplaceVas.findFirstOverlap(toReplaceVa);
132  ASSERT_require(found != toReplaceVas.intervals().end());
133  AddressInterval entryInterval = *found & AddressInterval::hull(toReplaceVa, AddressInterval::whole().greatest());
134 
135  // Try to replace
136  SAWYER_MESG(debug) <<" trying replacement using overwrite and/or transfer methodologies...\n";
137  if (replaceByOverwrite(toReplaceVas, entryInterval, replacement, relocations, relocStart, insnInfoMap) ||
138  replaceByTransfer(toReplaceVas, entryInterval, toReplace, replacement, relocations, relocStart, insnInfoMap)) {
139  SAWYER_MESG(debug) <<" replaceBlockInsns succeeded\n";
140  return true;
141  }
142  }
143 
144  if ((aggregationDirection_ & AGGREGATE_SUCCESSORS) != 0 && index + nInsns < bb->nInstructions()) {
145  // Enlarge to-be-replaced by appending one of the block's instructions to the replacement
146  size_t insnIdx = index + nInsns++;
147  int relIdx = (int)insnIdx - (int)(insertAt + nDeletes); // relative to one past last deleted instruction
148  SgAsmInstruction *nextInsn = bb->instructions()[insnIdx];
149 
150  insnInfoMap[relIdx].newVaOffset = replacement.size();
151 
152  toReplace.push_back(nextInsn);
153  toReplaceVas |= AddressInterval::baseSize(nextInsn->get_address(), nextInsn->get_size());
154  std::vector<uint8_t> bytes = nextInsn->get_rawBytes();
155  replacement.insert(replacement.end(), bytes.begin(), bytes.end());
156 
157  SAWYER_MESG(debug) <<" enlarged area by redirecting a successor:\n"
158  <<" " <<nextInsn->toString() <<"\n";
159 
160  } else if ((aggregationDirection_ & AGGREGATE_PREDECESSORS) != 0 && index > 0) {
161  // Enlarge to-be-replaced by prepending one of the block's instructions to the replacement
162  size_t insnIdx = --index;
163  int relIdx = (int)insnIdx - (int)insertAt; // relative to insertion point
164  SgAsmInstruction *prevInsn = bb->instructions()[insnIdx];
165 
166  for (InstructionInfo &info: insnInfoMap.values()) {
167  if (info.newVaOffset)
168  info.newVaOffset = info.newVaOffset.get() + prevInsn->get_size();
169  }
170  insnInfoMap[relIdx].newVaOffset = 0;
171 
172  toReplace.insert(toReplace.begin(), prevInsn);
173  toReplaceVas |= AddressInterval::baseSize(prevInsn->get_address(), prevInsn->get_size());
174  std::vector<uint8_t> bytes = prevInsn->get_rawBytes();
175  replacement.insert(replacement.begin(), bytes.begin(), bytes.end());
176  relocStart += bytes.size();
177 
178  SAWYER_MESG(debug) <<" enlarged area by redirecting a predecessor:\n"
179  <<" " <<prevInsn->toString() <<"\n";
180  } else {
181  break;
182  }
183  }
184  SAWYER_MESG(debug) <<" replaceBlockInsns was unable to insert new code\n";
185  return false;
186 }

244 void
245 CodeInserter::fillWithNops(const AddressIntervalSet &where) {
246  Sawyer::Message::Stream debug(mlog[DEBUG]);
247 
248  for (const AddressInterval &interval: where.intervals()) {
249  SAWYER_MESG(debug) <<"filling " <<StringUtility::addrToString(interval) <<" with no-op instructions\n";
250 
251  // Create the vector containing the encoded instructions
252  std::vector<uint8_t> nops;
253  nops.reserve(interval.size());
254  if (as<const Architecture::X86>(partitioner_->architecture())) {
255  nops.resize(interval.size(), 0x90);
256 
257  } else if (as<const Architecture::NxpColdfire>(partitioner_->architecture())) {
258  // Although coldfire instructions are normally a multiple of 2 bytes, there is one exception: a bad instruction can
259  // be an odd number of bytes. Therefore we could be asked to fill an odd number of bytes with NOP instructions. If
260  // that happens, we fill even addresses with the first half of the NOP and odd addresses with the second half, this
261  // way preserving the usual m68k instruction alignment.
262  for (size_t i = 0; i < where.size(); ++i)
263  nops.push_back(i % 2 == 0 ? 0x4e : 0x71);
264 
265  } else {
266  TODO("no-op insertion for " + partitioner_->architecture()->name() + " is not implemented yet");
267  }
268 
269  // Write the vector to memory
270  if (partitioner_->memoryMap()->at(interval.least()).write(nops).size() != nops.size()) {
271  mlog[ERROR] <<"short write of " <<interval.size() <<"-byte NOP sequence at "
272  <<StringUtility::addrToString(interval.least()) <<"\n";
273  }
274  }

331 std::vector<uint8_t>
332 CodeInserter::applyRelocations(Address va, std::vector<uint8_t> replacement, const std::vector<Relocation> &relocations,
333  size_t relocStart, const InstructionInfoMap &insnInfoMap) {
334  Sawyer::Message::Stream debug(mlog[DEBUG]);
335  if (debug) {
336  debug <<"applying relocations at va " <<StringUtility::addrToString(va) <<"\n"
337  <<" relocStart = " <<relocStart <<"\n"
338  <<" reloc input = [";
339  for (uint8_t byte: replacement)
340  Diagnostics::mfprintf(debug)(" 0x%02x", byte);
341  debug <<" ], " <<StringUtility::plural(replacement.size(), "bytes") <<"\n";
342  }
343 
344  for (const Relocation &reloc: relocations) {
345  ASSERT_require(reloc.offset < replacement.size());
346  Address value = 0;
347  SAWYER_MESG(debug) <<" at offset " <<(relocStart + reloc.offset) <<": ";
348  switch (reloc.type) {
349  case RELOC_INDEX_ABS_LE32: {
350  // Virtual address of one of the input bytes.
351  SAWYER_MESG(debug) <<"index_abs_le32(" <<StringUtility::addrToString(reloc.value) <<")";
352 
353  value = va + relocStart + reloc.value;
354 
355  ASSERT_require(relocStart + reloc.offset + 4 <= replacement.size());
356  replacement[relocStart + reloc.offset + 0] = (value >> 0) & 0xff;
357  replacement[relocStart + reloc.offset + 1] = (value >> 8) & 0xff;
358  replacement[relocStart + reloc.offset + 2] = (value >> 16) & 0xff;
359  replacement[relocStart + reloc.offset + 3] = (value >> 24) & 0xff;
360  break;
361  }
362 
363  case RELOC_INDEX_ABS_BE32: {
364  // Virtual address of one of the input bytes.
365  SAWYER_MESG(debug) <<"index_abs_be32(" <<StringUtility::addrToString(reloc.value) <<")";
366 
367  value = va + relocStart + reloc.value;
368 
369  ASSERT_require(relocStart + reloc.offset + 4 <= replacement.size());
370  replacement[relocStart + reloc.offset + 0] = (value >> 24) & 0xff;
371  replacement[relocStart + reloc.offset + 1] = (value >> 16) & 0xff;
372  replacement[relocStart + reloc.offset + 2] = (value >> 8) & 0xff;
373  replacement[relocStart + reloc.offset + 3] = (value >> 0) & 0xff;
374  break;
375  }
376 
377  case RELOC_INDEX_ABS_LE32HI: {
378  // Virtual address of one of the input bytes (but only the high 32 bits)
379  SAWYER_MESG(debug) <<"index_abs_le32hi(" <<StringUtility::addrToString(reloc.value) <<")";
380 
381  value = (va + relocStart + reloc.value) >> 32;
382 
383  ASSERT_require(relocStart + reloc.offset + 4 <= replacement.size());
384  replacement[relocStart + reloc.offset + 0] = (value >> 0) & 0xff;
385  replacement[relocStart + reloc.offset + 1] = (value >> 8) & 0xff;
386  replacement[relocStart + reloc.offset + 2] = (value >> 16) & 0xff;
387  replacement[relocStart + reloc.offset + 3] = (value >> 24) & 0xff;
388  break;
389  }
390 
391  case RELOC_ADDR_REL_LE32: {
392  // Offset from the input byte's virtual address to a specified virtual address, adjusted by adding the value
393  // originally stored in the input.
394  SAWYER_MESG(debug) <<"index_rel_le32(" <<StringUtility::addrToString(reloc.value);
395 
396  ASSERT_require(relocStart + reloc.offset + 4 <= replacement.size());
397  Address addend = replacement[relocStart + reloc.offset + 0];
398  addend |= replacement[relocStart + reloc.offset + 1] << 8;
399  addend |= replacement[relocStart + reloc.offset + 1] << 16;
400  addend |= replacement[relocStart + reloc.offset + 1] << 24;
401  addend = IntegerOps::signExtend2(addend, 32, 64);
402 
403  SAWYER_MESG(debug) <<", addend=" <<StringUtility::toHex2(addend, 32) <<")";
404 
405  value = reloc.value - (va + relocStart + reloc.offset) + addend;
406 
407  replacement[relocStart + reloc.offset + 0] = (value >> 0) & 0xff;
408  replacement[relocStart + reloc.offset + 1] = (value >> 8) & 0xff;
409  replacement[relocStart + reloc.offset + 2] = (value >> 16) & 0xff;
410  replacement[relocStart + reloc.offset + 3] = (value >> 24) & 0xff;
411  break;
412  }
413 
414  case RELOC_ADDR_REL_BE32: {
415  // Offset from the input byte's virtual address to a specified virtual address, adjusted by adding the value
416  // originally stored in the input.
417  SAWYER_MESG(debug) <<"index_rel_be32(" <<StringUtility::addrToString(reloc.value);
418 
419  ASSERT_require(relocStart + reloc.offset + 4 <= replacement.size());
420  Address addend = replacement[relocStart + reloc.offset + 0] << 24;
421  addend |= replacement[relocStart + reloc.offset + 1] << 16;
422  addend |= replacement[relocStart + reloc.offset + 1] << 8;
423  addend |= replacement[relocStart + reloc.offset + 1] << 0;
424  addend = IntegerOps::signExtend2(addend, 32, 64);
425 
426  SAWYER_MESG(debug) <<", addend=" <<StringUtility::toHex2(addend, 32) <<")";
427 
428  value = reloc.value - (va + relocStart + reloc.offset) + addend;
429 
430  replacement[relocStart + reloc.offset + 0] = (value >> 24) & 0xff;
431  replacement[relocStart + reloc.offset + 1] = (value >> 16) & 0xff;
432  replacement[relocStart + reloc.offset + 2] = (value >> 8) & 0xff;
433  replacement[relocStart + reloc.offset + 3] = (value >> 0) & 0xff;
434  break;
435  }
436 
437  case RELOC_INSN_ABS_LE32: {
438  // Virtual address of some possibly moved instruction in the original basic block. The instruction is
439  // indicated by its index relative to the instructions that were removed by this insertion.
440  int relIdx = (int)reloc.value;
441  SAWYER_MESG(debug) <<"insn_abs_le32(" <<relIdx;
442  ASSERT_require(insnInfoMap.exists(relIdx));
443 
444  const InstructionInfo &info = insnInfoMap[relIdx];
445  Address value = info.newVaOffset ? va + info.newVaOffset.get() : info.originalVa;
446 
447  ASSERT_require(relocStart + reloc.offset + 4 <= replacement.size());
448  replacement[relocStart + reloc.offset + 0] = (value >> 0) & 0xff;
449  replacement[relocStart + reloc.offset + 1] = (value >> 8) & 0xff;
450  replacement[relocStart + reloc.offset + 2] = (value >> 16) & 0xff;
451  replacement[relocStart + reloc.offset + 3] = (value >> 24) & 0xff;
452  break;
453  }
454 
455  case RELOC_INSN_REL_LE32: {
456  // Offset from the input byte's virtual address to the virtual address of some possibly moved instruction of
457  // the original basic block, adjusted by adding the value originally stored in the input. The instruction is
458  // indicated by its index relative to the instructions that were removed by this insertion.
459  int relIdx = (int)reloc.value;
460  SAWYER_MESG(debug) <<"insn_rel_le32(" <<relIdx;
461  ASSERT_require(insnInfoMap.exists(relIdx));
462 
463  ASSERT_require(relocStart + reloc.offset + 4 <= replacement.size());
464  Address addend = replacement[relocStart + reloc.offset + 0];
465  addend |= replacement[relocStart + reloc.offset + 1] << 8;
466  addend |= replacement[relocStart + reloc.offset + 1] << 16;
467  addend |= replacement[relocStart + reloc.offset + 1] << 24;
468  addend = IntegerOps::signExtend2(addend, 32, 64);
469  SAWYER_MESG(debug) <<", addend=" <<StringUtility::toHex2(addend, 32) <<")";
470 
471  const InstructionInfo &info = insnInfoMap[relIdx];
472  Address targetVa = info.newVaOffset ? va + info.newVaOffset.get() : info.originalVa;
473  SAWYER_MESG(debug) <<" [target=" <<StringUtility::addrToString(targetVa) <<"]";
474  value = targetVa - (va + relocStart + reloc.offset) + addend;
475 
476  replacement[relocStart + reloc.offset + 0] = (value >> 0) & 0xff;
477  replacement[relocStart + reloc.offset + 1] = (value >> 8) & 0xff;
478  replacement[relocStart + reloc.offset + 2] = (value >> 16) & 0xff;
479  replacement[relocStart + reloc.offset + 3] = (value >> 24) & 0xff;
480  break;
481  }
482 
483  case RELOC_INSN_REL_BE32: {
484  // Offset from the input byte's virtual address to the virtual address of some possibly moved instruction of
485  // the original basic block, adjusted by adding the value originally stored in the input. The instruction is
486  // indicated by its index relative to the instructions that were removed by this insertion.
487  int relIdx = (int)reloc.value;
488  SAWYER_MESG(debug) <<"insn_rel_be32(" <<relIdx;
489  ASSERT_require(insnInfoMap.exists(relIdx));
490 
491  ASSERT_require(relocStart + reloc.offset + 4 <= replacement.size());
492  Address addend = replacement[relocStart + reloc.offset + 0] << 24;
493  addend |= replacement[relocStart + reloc.offset + 1] << 16;
494  addend |= replacement[relocStart + reloc.offset + 1] << 8;
495  addend |= replacement[relocStart + reloc.offset + 1] << 0;
496  addend = IntegerOps::signExtend2(addend, 32, 64);
497  SAWYER_MESG(debug) <<", addend=" <<StringUtility::toHex2(addend, 32) <<")";

499  const InstructionInfo &info = insnInfoMap[relIdx];
500  Address targetVa = info.newVaOffset ? va + info.newVaOffset.get() : info.originalVa;
501  SAWYER_MESG(debug) <<" [target=" <<StringUtility::addrToString(targetVa) <<"]";
502  value = targetVa - (va + relocStart + reloc.offset) + addend;
503 
504  replacement[relocStart + reloc.offset + 0] = (value >> 24) & 0xff;
505  replacement[relocStart + reloc.offset + 1] = (value >> 16) & 0xff;
506  replacement[relocStart + reloc.offset + 2] = (value >> 8) & 0xff;
507  replacement[relocStart + reloc.offset + 3] = (value >> 0) & 0xff;
508  break;
509  }
510  }
511  SAWYER_MESG(debug)

513  <<" written to " <<StringUtility::addrToString(va)
514  <<StringUtility::addrToString(va + relocStart + reloc.offset) <<"\n";
515  }
516 
517  if (debug) {
518  debug <<" reloc output = [";
519  for (uint8_t byte: replacement)
520  Diagnostics::mfprintf(debug)(" 0x%02x", byte);
521  debug <<" ], " <<StringUtility::plural(replacement.size(), "bytes") <<"\n";
522  }

527 AddressInterval
528 CodeInserter::allocateMemory(const size_t nBytes, const Address jmpTargetVa, const Commit::Boolean commit) {
529  Sawyer::Message::Stream debug(mlog[DEBUG]);
530  SAWYER_MESG(debug) <<" allocateMemory(nBytes=" <<nBytes <<", jmpTargetVa=" <<StringUtility::addrToString(jmpTargetVa) <<"):"
531  <<(Commit::YES == commit ? "" : " (test only)") <<"\n";
532 
533  // Find the first interval that's large enough, not considering the jump. Then add the jump (whose size might depend
534  // on the address) and see if there's still enough free space. If not, continue searching through the free space list.
535  AddressIntervalSet::ConstIntervalIterator found = mappedFreeSpace().intervals().begin();
536  while (found != mappedFreeSpace().intervals().end()) {
537  found = mappedFreeSpace().firstFit(nBytes, found);
538  if (found != mappedFreeSpace().intervals().end()) {
539  size_t totalBytes = nBytes + encodeJump(found->least() + nBytes, jmpTargetVa).size();
540  if (totalBytes <= found->size()) {
541  AddressInterval retval = AddressInterval::baseSize(found->least(), totalBytes);
542  SAWYER_MESG(debug) <<" reserved " <<StringUtility::addrToString(retval)
543  <<" from free space at " <<StringUtility::addrToString(*found) <<"\n";
544  commitAllocation(retval, commit);
545  return retval;
546  } else {
547  ++found;
548  }
549  }
550  }
551 
552  // If we didn't find enough free space in segments that are already mapped, then map big chunk and use the first part
553  // of it to satisfy the request.
554  static const size_t maxJmpEncodedSize = 16; // size of largest unconditional jump on any architecture
555  //AddressInterval chunkArea = AddressInterval::hull(0x80000000, 0xbfffffff); // restrictions on chunk locations
556  size_t chunkSize = std::max(nBytes + maxJmpEncodedSize, minChunkAllocationSize_);
557  Address chunkVa = 0;
558  if (!partitioner_->memoryMap()->findFreeSpace(chunkSize, chunkAllocationAlignment_, chunkAllocationRegion_)
559  .assignTo(chunkVa)) {
560  SAWYER_MESG(debug) <<" cannot find " <<StringUtility::plural(chunkSize, "bytes")
561  <<" of " <<chunkAllocationAlignment_ <<"-byte aligned free space"
562  <<" in partitioner memory map"
563  <<" within " <<StringUtility::addrToString(chunkAllocationRegion_) <<"\n";
564  return AddressInterval(); // no virtual address space available
565  }
566 
567  // Create the chunk and map it
568  AddressInterval chunkVas = AddressInterval::baseSize(chunkVa, chunkSize);
569  partitioner_->memoryMap()->insert(chunkVas,
570  MemoryMap::Segment(MemoryMap::AllocatingBuffer::instance(chunkSize),
571  0, MemoryMap::READ_EXECUTE, chunkAllocationName_));
572  mappedFreeSpace() |= chunkVas;
573  allocatedChunks_ |= chunkVas;
574  SAWYER_MESG(debug) <<" mapped new free-space region at " <<StringUtility::addrToString(chunkVas) <<"\n";
575 
576  // Use the first part of the new chunk
577  size_t totalBytes = nBytes + encodeJump(chunkVa + nBytes, jmpTargetVa).size();
578  ASSERT_require2(totalBytes <= chunkSize, "maxJmpEncodedSize is not large enough");
579  AddressInterval retval = AddressInterval::baseSize(chunkVa, totalBytes);
580  SAWYER_MESG(debug) <<" reserved " <<StringUtility::addrToString(retval)
581  <<" from free space at " <<StringUtility::addrToString(chunkVas) <<"\n";
582  commitAllocation(retval, commit);
583  return retval;
584 }

602 bool
603 CodeInserter::replaceByOverwrite(const AddressIntervalSet &toReplaceVas, const AddressInterval &entryInterval,
604  const std::vector<uint8_t> &replacement, const std::vector<Relocation> &relocations,
605  size_t relocStart, const InstructionInfoMap &insnInfoMap) {
606  Sawyer::Message::Stream debug(mlog[DEBUG]);
607  if (debug) {
608  debug <<"replaceByOverwrite:\n";
609  debug <<" toReplaceVas = " <<StringUtility::addrToString(toReplaceVas) <<", "
610  <<StringUtility::plural(toReplaceVas.size(), "bytes") <<"\n";
611  debug <<" entryInterval = " <<StringUtility::addrToString(entryInterval)
612  <<", " <<StringUtility::plural(entryInterval.size(), "bytes") <<"\n";
613  debug <<" replacement [";
614  for (uint8_t byte: replacement)
615  Diagnostics::mfprintf(debug)(" 0x%02x", byte);
616  debug <<" ], " <<StringUtility::plural(replacement.size(), "bytes") <<"\n";
617  }
618 
619  // Figure out where the replacement should go
620  if (replacement.size() > entryInterval.size()) {
621  SAWYER_MESG(debug) <<" replaceByOverwrite failed: replacement is larger than entryInterval\n";
622  return false;
623  }
624  Address replacementVa = entryInterval.least();
625  if (replacement.size() < entryInterval.size()) {
626  switch (nopPadding_) {
627  case PAD_NOP_FRONT:
628  replacementVa += entryInterval.size() - replacement.size();
629  break;
630  case PAD_NOP_BACK:
631  case PAD_RANDOM_BACK:
632  break;
633  }
634  }
635  SAWYER_MESG(debug) <<" trying to insert replacement at " <<StringUtility::addrToString(replacementVa) <<"\n";
636  AddressInterval replacementVas = AddressInterval::baseSize(replacementVa, replacement.size());
637 
638  // Write the replacement to memory
639  std::vector<uint8_t> patched = applyRelocations(replacementVa, replacement, relocations, relocStart, insnInfoMap);
640  if (partitioner_->memoryMap()->at(replacementVas).write(patched).size() != patched.size()) {
641  SAWYER_MESG(debug) <<" replaceByOverwrite failed: short write to memory map at "
642  <<StringUtility::addrToString(replacementVas) <<"\n";
643  return false;
644  }
645 
646  // Fill everything else with no-ops, etc.
647  switch (nopPadding_) {
648  case PAD_NOP_FRONT:
649  case PAD_NOP_BACK:
650  fillWithNops(toReplaceVas - replacementVas);
651  break;
652  case PAD_RANDOM_BACK:
653  fillWithRandom(toReplaceVas - replacementVas);
654  break;
655  }
656 
657  SAWYER_MESG(debug) <<" replaceByOverwrite succeeded\n";
658  return true;
659 }

661 bool
662 CodeInserter::replaceByTransfer(const AddressIntervalSet &toReplaceVas, const AddressInterval &entryInterval,
663  const std::vector<SgAsmInstruction*> &toReplace, const std::vector<uint8_t> &replacement,
664  const std::vector<Relocation> &relocations, size_t relocStart,
665  const InstructionInfoMap &insnInfoMap) {
666  ASSERT_forbid(toReplace.empty());
667  ASSERT_forbid(toReplaceVas.isEmpty());
668  Sawyer::Message::Stream debug(mlog[DEBUG]);
669  if (debug) {
670  debug <<"replaceByTransfer:\n";
671  debug <<" toReplaceVas = " <<StringUtility::addrToString(toReplaceVas)
672  <<", " <<StringUtility::plural(toReplaceVas.size(), "bytes") <<"\n";
673  debug <<" entryInterval = " <<StringUtility::addrToString(entryInterval)
674  <<", " <<StringUtility::plural(entryInterval.size(), "bytes") <<"\n";
675  debug <<" instructions to be moved:\n";
676  for (SgAsmInstruction *insn: toReplace)
677  debug <<" " <<partitioner_->unparse(insn) <<"\n";
678  debug <<" replacement = [";
679  for (uint8_t byte: replacement)
680  Diagnostics::mfprintf(debug)(" 0x%02x", byte);
681  debug <<" ], " <<StringUtility::plural(replacement.size(), "bytes") <<"\n";
682  }
683 
684  // Allocate memory for the replacement, plus enough to add a branch back to the first instruction after those being
685  // replaced.
686  Address toReplaceFallThroughVa = toReplace.back()->get_address() + toReplace.back()->get_size();
687  AddressInterval replacementVas = allocateMemory(replacement.size(), toReplaceFallThroughVa, Commit::NO);
688  if (replacementVas.isEmpty()) {
689  SAWYER_MESG(debug) <<" replaceByTransfer failed: cannot allocate space in memory map\n";
690  return false; // failure to allocate memory
691  }
692  SAWYER_MESG(debug) <<" memory for moved instructions and returning branch is "
693  <<StringUtility::addrToString(replacementVas) <<"\n";
694 
695  // Insert the replacement
696  std::vector<uint8_t> patched = applyRelocations(replacementVas.least(), replacement, relocations, relocStart, insnInfoMap);
697  if (partitioner_->memoryMap()->at(replacementVas).write(patched).size() != patched.size()) {
698  SAWYER_MESG(debug) <<" replaceByTransfer failed: short write to memory map at "
699  <<StringUtility::addrToString(replacementVas) <<"\n";
700  return false; // write failed
701  }
702 
703  // Insert the jump back from the replacement
704  Address replacementFallThroughVa = replacementVas.least() + replacement.size();
705  std::vector<uint8_t> jmpFrom = encodeJump(replacementFallThroughVa, toReplaceFallThroughVa);
706  if (partitioner_->memoryMap()->at(replacementFallThroughVa).write(jmpFrom).size() != jmpFrom.size()) {
707  SAWYER_MESG(debug) <<" replaceByTransfer failed: short write to memory map for returning branch at "
708  <<StringUtility::addrToString(replacementFallThroughVa) <<"\n";
709  return false; // write failed
710  }
711 
712  // Insert the jump to the replacement code. If we're nop-padding before the jump this becomes a bit complicated because
713  // the size of the jump could change based on its address. If that happens, give up.
714  Address jmpToSite = toReplace[0]->get_address();
715  std::vector<uint8_t> jmpTo = encodeJump(jmpToSite, replacementVas.least());
716  if (jmpTo.size() > entryInterval.size()) {
717  SAWYER_MESG(debug) <<" replaceByTransfer failed: not enough room for the branch to moved code at "
718  <<StringUtility::addrToString(jmpToSite) <<"\n";
719  return false; // not enough room for the branch-to-replacement instruction
720  }
721  if (jmpTo.size() < entryInterval.size() && PAD_NOP_FRONT == nopPadding_) {
722  jmpToSite += entryInterval.size() - jmpTo.size();
723  std::vector<uint8_t> newJmpTo = encodeJump(jmpToSite, replacementVas.least());
724  if (newJmpTo.size() != jmpTo.size()) {
725  SAWYER_MESG(debug) <<" replaceByTransfer failed: not enough room for re-encoded branch to moved code at "
726  <<StringUtility::addrToString(jmpToSite) <<"\n";
727  return false; // padded, re-encoded jump no longer fits
728  }
729  jmpTo = newJmpTo;
730  }
731  AddressInterval jmpToVas = AddressInterval::baseSize(jmpToSite, jmpTo.size());
732  if (partitioner_->memoryMap()->at(jmpToVas).write(jmpTo).size() != jmpTo.size()) {
733  SAWYER_MESG(debug) <<" replaceByTransfer failed: short write to memory map for branch to moved code at "
734  <<StringUtility::addrToString(jmpToVas) <<"\n";
735  return false; // write failed
736  }
737 
738  // Nop-fill
739  fillWithNops(toReplaceVas - jmpToVas);
740  commitAllocation(replacementVas);
741  SAWYER_MESG(debug) <<" replaceByTransfer succeeded\n";
742  return true;
743 }
