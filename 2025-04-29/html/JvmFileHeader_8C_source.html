86 SgAsmJvmFileHeader::parse()
87 {
88  SgAsmGenericFile* gf = get_file();
89  ASSERT_not_null(gf);
90  ROSE_ASSERT(get_size() > 0);
91 
92  /* parse base class */
93  SgAsmGenericHeader::parse();
94 
95  Rose::BinaryAnalysis::Address offset{get_offset()};
96 
97  /* Construct, but don't parse (yet), constant pool */
98  auto pool = new SgAsmJvmConstantPool(this);
99  set_constant_pool(pool);
100 
101  /* Ensure magic number in file is correct */
102  unsigned char magic[4];
103  auto count = readContent(offset, magic, sizeof magic);
104  if (4!=count || p_magic.size()!=count || p_magic[0]!=magic[0]
105  || p_magic[1]!=magic[1] || p_magic[2]!=magic[2] || p_magic[3]!=magic[3]) {
106  throw FormatError("Bad Java class file magic number");
107  }
108  offset += count;
109  set_offset(offset);
110 
111  /* Minor and major version */
112  Jvm::read_value(pool, p_minor_version);
113  Jvm::read_value(pool, p_major_version);
114 
115  ASSERT_not_null(get_executableFormat());
116  get_executableFormat()->set_version(p_major_version);
117  get_executableFormat()->set_isCurrentVersion(true);
118  get_executableFormat()->set_abiVersion(p_major_version);
119 
120  /* And finally the constant pool can be parsed */
121  pool->parse();
122 
123  Jvm::read_value(pool, p_access_flags);
124  Jvm::read_value(pool, p_this_class);
125  Jvm::read_value(pool, p_super_class);
126 
127  uint16_t interfaces_count;
128  Jvm::read_value(pool, interfaces_count);
129 
130  std::list<uint16_t>& interfaces = get_interfaces();
131  for (int i = 0; i < interfaces_count; i++) {
132  uint16_t index;
133  Jvm::read_value(pool, index);
134  interfaces.push_back(index);
135  }
136 
137  /* Fields */
138  auto fields = new SgAsmJvmFieldTable(this);
139  set_field_table(fields);
140  fields->parse();
141 
142  /* Methods */
143  auto methods = new SgAsmJvmMethodTable(this);
144  set_method_table(methods);
145  methods->parse();
146 
147  /* Attributes */
148  auto attributes = new SgAsmJvmAttributeTable(this, /*parent*/this);
149  set_attribute_table(attributes);
150  ASSERT_not_null(attributes->get_parent());
151  attributes->parse(pool);
152 
153  if (1 != (get_endOffset() - get_offset())) {
154  mlog[FATAL] << "Error reading file, end of file not reached\n";

163 SgAsmJvmFileHeader::unparse(std::ostream &f) const
164 {
165  SgAsmGenericFile* gf = get_file();
166  auto count = f.tellp();
167 
168  auto bytes = reinterpret_cast<const char*>(&(get_magic()[0]));
169  f.write(bytes, 4);
170 
171  auto minor = p_minor_version;
172  auto major = p_major_version;
173  Rose::BinaryAnalysis::ByteOrder::hostToBe(minor, &minor);
174  Rose::BinaryAnalysis::ByteOrder::hostToBe(major, &major);
175 
176  f.write(reinterpret_cast<const char*>(&minor), sizeof minor);
177  f.write(reinterpret_cast<const char*>(&major), sizeof major);
178 
179  // Unparse the constant pool
180  auto pool = get_constant_pool();
181  pool->unparse(f);
182 
183  auto access_flags = p_access_flags;
184  auto this_class = p_this_class;
185  auto super_class = p_super_class;
186 
187  hostToBe(access_flags, &access_flags);
188  hostToBe(this_class, &this_class);
189  hostToBe(super_class, &super_class);
190 
191  f.write(reinterpret_cast<const char*>(&access_flags), sizeof access_flags);
192  f.write(reinterpret_cast<const char*>(&this_class), sizeof this_class);
193  f.write(reinterpret_cast<const char*>(&super_class), sizeof super_class);
194 
195  /* Interfaces */
196  uint16_t interfaces_count = get_interfaces().size();
197  hostToBe(interfaces_count, &interfaces_count);
198  f.write(reinterpret_cast<const char*>(&interfaces_count), sizeof interfaces_count);
199 
200  for (auto interface : get_interfaces()) {
201  uint16_t index = interface;
202  hostToBe(index, &index);
203  f.write(reinterpret_cast<const char*>(&index), sizeof index);
204  }
205 
206  /* Fields */
207  SgAsmJvmFieldTable* fields = get_field_table();
208  fields->unparse(f);
209 
210  /* Methods */
211  SgAsmJvmMethodTable* methods = get_method_table();
212  methods->unparse(f);
213 
214  /* Methods */
215  SgAsmJvmAttributeTable* attributes = get_attribute_table();
216  attributes->unparse(f);
217 
218  count = f.tellp();
219  auto data = gf->get_data();
220  if (count != data.size()) {
221  mlog[WARN] << "unparse didn't complete: count==" << count << " data_size==" << data.size() << "\n";
222  }
223  bytes = reinterpret_cast<const char*>(&(data.pool()[count]));
224  f.write(bytes, data.size()-count);
225 
226  // The end-of-file marker is read during parsing, don't rewrite or write '\0' bytes padding at end
227  // However, it seems logic is wrong in GenericFile.C used to call extendToEof(), thus following required for JVM
228  gf->set_truncateZeros(true);
229 }
