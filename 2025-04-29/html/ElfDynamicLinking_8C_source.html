123 SgAsmElfDynamicSection *
124 SgAsmElfDynamicSection::parse()
125 {
126  SgAsmElfSection::parse();
127 
128  SgAsmElfFileHeader *fhdr = get_elfHeader();
129  ROSE_ASSERT(fhdr);
130  SgAsmElfSectionTableEntry *shdr = get_sectionEntry();
131  ASSERT_always_require(shdr);
132  SgAsmElfStringSection *strsec = dynamic_cast<SgAsmElfStringSection*>(get_linkedSection());
133  ROSE_ASSERT(strsec!=NULL);
134 
135  size_t entry_size, struct_size, extra_size, nentries;
136  calculateSizes(&entry_size, &struct_size, &extra_size, &nentries);
137  ROSE_ASSERT(entry_size==shdr->get_sh_entsize());
138 
139  /* Parse each entry */
140  for (size_t i=0; i<nentries; i++) {
141  SgAsmElfDynamicEntry *entry=0;
142  if (4==fhdr->get_wordSize()) {
143  entry = new SgAsmElfDynamicEntry(this);
144  SgAsmElfDynamicEntry::Elf32DynamicEntry_disk disk;
145  readContentLocal(i*entry_size, &disk, struct_size);
146  entry->parse(fhdr->get_sex(), &disk);
147  } else if (8==fhdr->get_wordSize()) {
148  entry = new SgAsmElfDynamicEntry(this);
149  SgAsmElfDynamicEntry::Elf64DynamicEntry_disk disk;
150  readContentLocal(i*entry_size, &disk, struct_size);
151  entry->parse(fhdr->get_sex(), &disk);
152  } else {
153  throw FormatError("unsupported ELF word size");
154  }
155  if (extra_size>0)
156  entry->get_extra() = readContentLocalUcl(i*entry_size+struct_size, extra_size);
157 
158  /* Set name */
159  if (entry->get_d_tag()==SgAsmElfDynamicEntry::DT_NEEDED) {
160  ROSE_ASSERT(entry->get_name()==NULL);
161  SgAsmStoredString *name = new SgAsmStoredString(strsec->get_strtab(), entry->get_d_val().rva());
162  entry->set_name(name);
163  name->set_parent(entry);
164 #if 1 /* FIXME: Do we really want this stuff duplicated in the AST? [RPM 2008-12-12] */
165  SgAsmStoredString *name2 = new SgAsmStoredString(strsec->get_strtab(), entry->get_d_val().rva());
166  fhdr->addDll(new SgAsmGenericDLL(name2));
167 #endif
168  }
169  }
170  return this;
171 }

195 void
196 SgAsmElfDynamicSection::finishParsing()
197 {
198  SgAsmElfFileHeader *fhdr = get_elfHeader();
199  ROSE_ASSERT(fhdr!=NULL);
200 
201  for (size_t i=0; i<p_entries->get_entries().size(); i++) {
202  SgAsmElfDynamicEntry *entry = p_entries->get_entries()[i];
203  switch (entry->get_d_tag()) {
204  case SgAsmElfDynamicEntry::DT_NEEDED:
205  break;
206  case SgAsmElfDynamicEntry::DT_PLTGOT:
207  case SgAsmElfDynamicEntry::DT_HASH:
208  case SgAsmElfDynamicEntry::DT_STRTAB:
209  case SgAsmElfDynamicEntry::DT_SYMTAB:
210  case SgAsmElfDynamicEntry::DT_RELA:
211  case SgAsmElfDynamicEntry::DT_INIT:
212  case SgAsmElfDynamicEntry::DT_FINI:
213  case SgAsmElfDynamicEntry::DT_REL:
214  case SgAsmElfDynamicEntry::DT_DEBUG:
215  case SgAsmElfDynamicEntry::DT_JMPREL:
216  case SgAsmElfDynamicEntry::DT_INIT_ARRAY:
217  case SgAsmElfDynamicEntry::DT_FINI_ARRAY:
218  case SgAsmElfDynamicEntry::DT_PREINIT_ARRAY:
219  case SgAsmElfDynamicEntry::DT_GNU_HASH:
220  case SgAsmElfDynamicEntry::DT_TLSDESC_PLT:
221  case SgAsmElfDynamicEntry::DT_TLSDESC_GOT:
222  case SgAsmElfDynamicEntry::DT_GNU_CONFLICT:
223  case SgAsmElfDynamicEntry::DT_GNU_LIBLIST:
224  case SgAsmElfDynamicEntry::DT_CONFIG:
225  case SgAsmElfDynamicEntry::DT_DEPAUDIT:
226  case SgAsmElfDynamicEntry::DT_AUDIT:
227  case SgAsmElfDynamicEntry::DT_PLTPAD:
228  case SgAsmElfDynamicEntry::DT_MOVETAB:
229  case SgAsmElfDynamicEntry::DT_SYMINFO:
230  case SgAsmElfDynamicEntry::DT_VERSYM:
231  case SgAsmElfDynamicEntry::DT_VERDEF:
232  case SgAsmElfDynamicEntry::DT_VERNEED:
233  case SgAsmElfDynamicEntry::DT_AUXILIARY:
234  case SgAsmElfDynamicEntry::DT_FILTER: {
235  /* d_val is relative to a section. We know that all ELF Sections (but perhaps not the ELF Segments) have been
236  * created by this time. */
237  ROSE_ASSERT(entry->get_d_val().boundSection() == NULL);
238  SgAsmGenericSectionPtrList containers = fhdr->get_sectionsByRva(entry->get_d_val().rva());
239  SgAsmGenericSection *best = NULL;
240  for (SgAsmGenericSectionPtrList::iterator i=containers.begin(); i!=containers.end(); ++i) {
241  if ((*i)->isMapped()) {
242  if ((*i)->get_mappedPreferredRva()==entry->get_d_val().rva()) {
243  best = *i;
244  break;
245  } else if (!best) {
246  best = *i;
247  } else if ((*i)->get_mappedSize() < best->get_mappedSize()) {
248  best = *i;
249  }
250  }
251  }
252  if (best)
253  entry->set_d_val(Rose::BinaryAnalysis::RelativeVirtualAddress(entry->get_d_val().rva(), best));
254  break;
255  }
256  default:
257  break;
258  }
259  }
260 }

262 bool
263 SgAsmElfDynamicSection::reallocate()
264 {
265  bool reallocated = SgAsmElfSection::reallocate();
266 
267  /* Update parts of the section and segment tables not updated by superclass */
268  SgAsmElfSectionTableEntry *secent = get_sectionEntry();
269  if (secent)
270  secent->set_sh_type(SgAsmElfSectionTableEntry::SHT_DYNAMIC);
271  SgAsmElfSegmentTableEntry *segent = get_segmentEntry();
272  if (segent)
273  segent->set_type(SgAsmElfSegmentTableEntry::PT_DYNAMIC);
274 
275  /* Update entries with name offsets. The name should point to the string table to which the dynamic section links. */
276  for (size_t i=0; i<p_entries->get_entries().size(); i++) {
277  SgAsmElfDynamicEntry *entry = p_entries->get_entries()[i];
278  if (entry->get_name()) {
279  SgAsmElfStringSection *strsec = dynamic_cast<SgAsmElfStringSection*>(get_linkedSection());
280  ROSE_ASSERT(strsec);
281  SgAsmStoredString *stored_string = dynamic_cast<SgAsmStoredString*>(entry->get_name());
282  if (!stored_string || stored_string->get_strtab()!=strsec->get_strtab()) {
283  /* Not a stored string, or stored in the wrong string table. */
284  stored_string = new SgAsmStoredString(strsec->get_strtab(), entry->get_name()->get_string());
285  entry->get_name()->set_string(""); /*free old storage*/
286  entry->set_name(stored_string);
287  }
288  entry->set_d_val(entry->get_name()->get_offset());
289  }
290  }
291 
292  return reallocated;
293 }

295 void
296 SgAsmElfDynamicSection::unparse(std::ostream &f) const
297 {
298  SgAsmElfFileHeader *fhdr = get_elfHeader();
299  ROSE_ASSERT(fhdr);
300  Rose::BinaryAnalysis::ByteOrder::Endianness sex = fhdr->get_sex();
301 
302  size_t entry_size, struct_size, extra_size, nentries;
303  calculateSizes(&entry_size, &struct_size, &extra_size, &nentries);
304 
305  /* Adjust the entry size stored in the ELF Section Table */
306  get_sectionEntry()->set_sh_entsize(entry_size);
307 
308  /* Write each entry's required part followed by the optional part */
309  for (size_t i=0; i<nentries; i++) {
310  SgAsmElfDynamicEntry::Elf32DynamicEntry_disk disk32;
311  SgAsmElfDynamicEntry::Elf64DynamicEntry_disk disk64;
312  void *disk = NULL;
313 
314  SgAsmElfDynamicEntry *entry = p_entries->get_entries()[i];
315 
316  if (4==fhdr->get_wordSize()) {
317  disk = entry->encode(sex, &disk32);
318  } else if (8==fhdr->get_wordSize()) {
319  disk = entry->encode(sex, &disk64);
320  } else {
321  ROSE_ASSERT(!"unsupported word size");
322  }
323 
324  Rose::BinaryAnalysis::Address spos = i * entry_size;
325  spos = write(f, spos, struct_size, disk);
326  if (entry->get_extra().size()>0) {
327  ROSE_ASSERT(entry->get_extra().size()<=extra_size);
328  write(f, spos, entry->get_extra());
329  }
330  }
331 
332  unparseHoles(f);
333 }
