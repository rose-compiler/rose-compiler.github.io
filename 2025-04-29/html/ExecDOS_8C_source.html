51 bool
52 SgAsmDOSFileHeader::isDos(SgAsmGenericFile *file)
53 {
54  /* Turn off byte reference tracking for the duration of this function. We don't want our testing the file contents to
55  * affect the list of bytes that we've already referenced or which we might reference later. */
56  bool was_tracking = file->get_trackingReferences();
57  file->set_trackingReferences(false);
58 
59  try {
60  unsigned char magic[2];
61  file->readContent(0, magic, sizeof magic);
62  if ('M'!=magic[0] || 'Z'!=magic[1])
63  throw 1;
64  } catch (...) {
65  file->set_trackingReferences(was_tracking);
66  return false;
67  }
68  file->set_trackingReferences(was_tracking);
69  return true;
70 }

72 SgAsmDOSFileHeader*
73 SgAsmDOSFileHeader::parse(bool define_rm_section)
74 {
75  SgAsmGenericSection::parse();
76 
77  /* Read header from file */
78  DOSFileHeader_disk disk;
79  if (sizeof(disk)>get_size())
80  extend(sizeof(disk)-get_size());
81  readContentLocal(0, &disk, sizeof disk);
82 
83  /* Check magic number early.
84  * Some old compilers were little-endian ignorant and stored "ZM", but we will ignore this [DQ]. */
85  if (disk.e_magic[0]!='M' || disk.e_magic[1]!='Z')
86  throw FormatError("Bad DOS magic number");
87 
88  /* Decode file format */
89  ROSE_ASSERT(Rose::BinaryAnalysis::ByteOrder::ORDER_LSB==get_executableFormat()->get_sex());
90  p_e_last_page_size = Rose::BinaryAnalysis::ByteOrder::leToHost(disk.e_last_page_size);
91  p_e_total_pages = Rose::BinaryAnalysis::ByteOrder::leToHost(disk.e_total_pages);
92  p_e_nrelocs = Rose::BinaryAnalysis::ByteOrder::leToHost(disk.e_nrelocs);
93  p_e_header_paragraphs = Rose::BinaryAnalysis::ByteOrder::leToHost(disk.e_header_paragraphs);
94  p_e_minalloc = Rose::BinaryAnalysis::ByteOrder::leToHost(disk.e_minalloc);
95  p_e_maxalloc = Rose::BinaryAnalysis::ByteOrder::leToHost(disk.e_maxalloc);
96  p_e_ss = Rose::BinaryAnalysis::ByteOrder::leToHost(disk.e_ss);
97  p_e_sp = Rose::BinaryAnalysis::ByteOrder::leToHost(disk.e_sp);
98  p_e_cksum = Rose::BinaryAnalysis::ByteOrder::leToHost(disk.e_cksum);
99  p_e_ip = Rose::BinaryAnalysis::ByteOrder::leToHost(disk.e_ip);
100  p_e_cs = Rose::BinaryAnalysis::ByteOrder::leToHost(disk.e_cs);
101  p_e_relocs_offset = Rose::BinaryAnalysis::ByteOrder::leToHost(disk.e_relocs_offset);
102  p_e_overlay = Rose::BinaryAnalysis::ByteOrder::leToHost(disk.e_overlay);
103  p_e_res1 = Rose::BinaryAnalysis::ByteOrder::leToHost(disk.e_res1);
104 
105  /* Magic number */
106  p_magic.clear();
107  p_magic.push_back(disk.e_magic[0]);
108  p_magic.push_back(disk.e_magic[1]);
109 
110  /* The DOS File Header is followed by optional relocation entries */
111  if (p_e_nrelocs>0) {
112  SgAsmGenericSection *relocs = new SgAsmGenericSection(get_file(), this);
113  relocs->set_offset(p_e_relocs_offset);
114  relocs->set_size(p_e_nrelocs * sizeof(DOSRelocEntry_disk));
115  relocs->parse();
116  relocs->set_name(new SgAsmBasicString("DOS relocation table"));
117  relocs->set_synthesized(true);
118  relocs->set_purpose(SP_HEADER);
119  set_relocs(relocs);
120  }
121 
122  /* DOS real-mode text/data/etc. */
123  if (define_rm_section)
124  parseRealModeSection();
125 
126  /* Entry point */
127  p_baseVa = 0;
128  addEntryRva(Rose::BinaryAnalysis::ByteOrder::leToHost(disk.e_ip));
129 
130  return this;
131 }

133 void
134 SgAsmDOSFileHeader::updateFromRealModeSection()
135 {
136  /* Find the DOS Extended Header */
137  SgAsmDOSFileHeader *dos1 = this;
138  SgAsmDOSExtendedHeader *dos2 = NULL;
139  const SgAsmGenericSectionPtrList &sections = dos1->get_sections()->get_sections();
140  for (SgAsmGenericSectionPtrList::const_iterator si=sections.begin(); !dos2 && si!=sections.end(); si++)
141  dos2 = isSgAsmDOSExtendedHeader(*si);
142 
143  /* Update DOS File Header with info about the real-mode text+data section. */
144  size_t header_size = dos1->get_size() + (dos2 ? dos2->get_size() : 0);
145  size_t total_size = header_size + (p_rm_section ? p_rm_section->get_size() : 0);
146  p_e_header_paragraphs = (header_size + 15) / 16; /* rounded up to next paragraph */
147  p_e_total_pages = (total_size + 511) / 512; /* rounded up to next page */
148  p_e_last_page_size = total_size % 512;
149 }

199 /* Write the DOS file header back to disk */
200 void
201 SgAsmDOSFileHeader::unparse(std::ostream &f) const
202 {
203  /* Unparse each section reachable from the DOS File Header (e.g., the Extended DOS Header) */
204  SgAsmDOSExtendedHeader *dos2 = NULL;
205  for (SgAsmGenericSectionPtrList::iterator i=p_sections->get_sections().begin(); i!=p_sections->get_sections().end(); ++i) {
206  if (!dos2)
207  dos2 = isSgAsmDOSExtendedHeader(*i);
208  (*i)->unparse(f);
209  }
210 
211  /* Some sanity checks:
212  * 1. DOS File Header must be at the beginning of the file.
213  * 2. DOS Extended Header, if present, must immediately follow DOS File Header
214  * 3. DOS Real-Mode Text/Data section must immediately follow headers */
215  ROSE_ASSERT(0==get_offset());
216  ROSE_ASSERT(!dos2 || dos2->get_offset()==get_size());
217 // ROSE_ASSERT(get_size()+(dos2?dos2->get_size():0) < (size_t)p_e_header_paragraphs*16);
218 // ROSE_ASSERT(!p_rm_section || p_rm_section->get_offset()==(size_t)p_e_header_paragraphs*16);
219 
220  /* Unparse the header itself */
221  DOSFileHeader_disk disk;
222  encode(&disk);
223  write(f, 0, sizeof(disk), &disk);
224 }

226 SgAsmGenericSection *
227 SgAsmDOSFileHeader::parseRealModeSection(Rose::BinaryAnalysis::Address max_offset)
228 {
229  ROSE_ASSERT(NULL == p_rm_section);
230 
231  Rose::BinaryAnalysis::Address rm_offset = p_e_header_paragraphs * 16;
232  Rose::BinaryAnalysis::Address rm_end = p_e_total_pages * 512;
233  if (p_e_total_pages>0)
234  rm_end -= 512 - (p_e_last_page_size%512);
235 
236  Rose::BinaryAnalysis::Address rm_size = rm_end>rm_offset ? rm_end-rm_offset : 0;
237  if (rm_size == 0)
238  return NULL;
239 
240  if (max_offset>0) {
241  if (max_offset < rm_offset) {
242  rm_size = 0;
243  } else if (rm_offset + rm_size > max_offset) {
244  rm_size = max_offset - rm_offset;
245  }
246  }
247 
248  try {
249  p_rm_section = new SgAsmGenericSection(get_file(), this);
250  p_rm_section->set_offset(rm_offset);
251  p_rm_section->set_size(rm_size);
252  p_rm_section->parse();
253  } catch (ShortRead &p_ex) {
254  /* If the offset or size is out of bounds for the file then assume that the real-mode section does not exist. This
255  * can indicate that the DOS header is being used for something other than a DOS header. See
256  * http://www.phreedom.org/solar/code/tinype/ for some examples of overlapping the DOS header with the PE header. */
257  return NULL;
258  }
259 
260  p_rm_section->set_name(new SgAsmBasicString("DOS real-mode text/data"));
261  p_rm_section->set_synthesized(true);
262  p_rm_section->set_purpose(SP_PROGRAM);
263  p_rm_section->set_mappedPreferredRva(0);
264  p_rm_section->set_mappedActualVa(0); /*will be assigned by Loader*/
265  p_rm_section->set_mappedSize(rm_size);
266  p_rm_section->set_mappedReadPermission(true);
267  p_rm_section->set_mappedWritePermission(true);
268  p_rm_section->set_mappedExecutePermission(true);
269  return p_rm_section;
270 }
