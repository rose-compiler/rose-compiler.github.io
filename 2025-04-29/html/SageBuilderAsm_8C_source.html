380 // M68k 96-bit "extended-precision real format"
381 // The 16 bits at [64-79] are always zero.
382 SgAsmFloatType*
383 buildTypeM68kFloat96() {
384  static SgAsmFloatType *cached = NULL;
385  static SAWYER_THREAD_TRAITS::Mutex mutex;
386  SAWYER_THREAD_TRAITS::LockGuard lock(mutex);
387 
388  if (!cached) {
389  // From the Motorola 68000 documentation, "For all three [binary floating-point] precisions, a normalized mantissa is
390  // always in the range (1.0...2.0). The extended-precision data format represents the entire mantissa, including the
391  // explicit integer part bit. Single- and double-precision data formats represent only a fractional portion of the
392  // mantissa (the fraction) and always imply the integer part as one". The documentation states that for
393  // extended-precision types, the term "matissa" is equivalent to IEEE 754 "significand" since the integer part of the
394  // significand is always stored.
395  //
396  // A table subsequently describes that the significand is denormalized when the exponent field's bits are all clear,
397  // implying that gradual underflow capability is present.
398  SgAsmFloatType *fpType = new SgAsmFloatType(ByteOrder::ORDER_LSB, 96,
399  SgAsmFloatType::BitRange::baseSize(0, 64), // significand w/explicit integer bit
400  SgAsmFloatType::BitRange::baseSize(80, 15), // exponent
401  95, // sign bit
402  0x3fff, // exponent bias
403  SgAsmFloatType::ieeeFlags()
404  .clear(SgAsmFloatType::IMPLICIT_BIT_CONVENTION));
405  cached = SgAsmType::registerOrDelete(fpType);
406  }
407  return cached;
408 }

441 
442  auto arch = Architecture::findByName("intel-pentium4").orThrow();
443  SgAsmX86Instruction *instruction = new SgAsmX86Instruction(0, *arch->registrationId(), x86_nop,
