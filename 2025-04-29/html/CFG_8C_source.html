68 CFG::CFG (IRInterface &_ir, IRStmtIterator *stmt_iterptr, SymHandle _name, bool return_statements_allowed)
69  : ir(_ir)
70 {
71  name = _name;
72 
73  // Create the unique entry node.
74  entry = new CFG::Node();
75  add(entry);
76  // Create the actual start node where the first statements wil be added.
77  CFG::Node* r = new CFG::Node();
78  add(r);
79  connect (entry, r, FALLTHROUGH_EDGE);
80 
81  CFG::NodeLabelList exit_nodes;
82  CFG::NodeLabelList return_nodes;
83  if (return_statements_allowed)
84  build_block(r, stmt_iterptr, exit_nodes, 0, &return_nodes, 0 /* continue nodes */);
85  else
86  build_block(r, stmt_iterptr, exit_nodes, 0, 0, 0 /* continue nodes */);
87  CFG::Node* final = new CFG::Node(); // final node is the exit node of the sub-program
88  add(final);
89  exit = final;
90  connect(exit_nodes, final); // connect the exit nodes to the final node
91  connect(return_nodes, final); // return nodes also lead to the exit node of the sub-program
92 
93  // finalize control flow for unstructured constructs
94 #if 1
95  HandleDelayedBranches();
96 #endif
97 }

138 IRStmtType
139 CFG::build_block (CFG::Node* prev_node, IRStmtIterator *si_ptr_, CFG::NodeLabelList& exit_nodes,
140  CFG::NodeLabelList* break_nodes, CFG::NodeLabelList* return_nodes,
141  CFG::NodeLabelList* continue_nodes)
142 {
143  NodeLabelList x_nodes;
144  IRStmtType prev_statement = SIMPLE;
145  while (si_ptr_->IsValid()) {
146  StmtHandle stmt = si_ptr_->Current();
147  if (ir.GetLabel(stmt)) {
148  // FIXME: Unstructured handling (labels), partially tested.
149  // If this statement has a label, it begins a new block (if prev_node
150  // is empty, we don't need to create another new block). The new block
151  // will have already been created if the label was referenced by a
152  // previously encountered statement.
153  StmtLabel lab = ir.GetLabel(stmt);
154  CFG::Node *new_node;
155  if (label_to_node_map.find(lab) != label_to_node_map.end()) {
156  new_node = label_to_node_map[lab];
157  }
158  else if (! prev_node->empty()) {
159  new_node = new CFG::Node();
160  add(new_node);
161  label_to_node_map[lab] = new_node;
162  }
163  else {
164  label_to_node_map[lab] = prev_node;
165  new_node = prev_node;
166  }
167 
168  if (prev_statement != SIMPLE && new_node != prev_node)
169  connect(x_nodes, new_node);
170  else if (prev_node && new_node != prev_node)
171  connect(prev_node, new_node, FALLTHROUGH_EDGE);
172  x_nodes.clear();
173  prev_node = new_node;
174  } else
175  if (prev_statement != SIMPLE) {
176  // now we need to create a new CFG node
177  prev_node = new CFG::Node();
178  add(prev_node);
179  connect(x_nodes, prev_node);
180  x_nodes.clear();
181  }
182  prev_statement = build_stmt(prev_node, stmt, x_nodes, break_nodes, return_nodes, continue_nodes);
183  IRStmtIterator &tmp = *si_ptr_;
184  ++tmp;
185 // ++(*si_ptr_);
186  }
187 
188  if (prev_statement == SIMPLE)
189  exit_nodes.push_back(NodeLabel(prev_node, FALLTHROUGH_EDGE));
190  else {
191  NodeLabelListIterator x_nodes_iter(&x_nodes);
192  while ((bool)x_nodes_iter) {
193  exit_nodes.push_back((NodeLabel)x_nodes_iter);
194  ++x_nodes_iter;
195  }
196  }
197 
198  return NONE;
199 }

211 IRStmtType
212 CFG::build_stmt (CFG::Node* prev_node, StmtHandle stmt, CFG::NodeLabelList& exit_nodes, CFG::NodeLabelList* break_nodes,
213  CFG::NodeLabelList* return_nodes, CFG::NodeLabelList* continue_nodes)
214 #if __cplusplus < 201103L
215  throw (CFG::Unexpected_Break, CFG::Unexpected_Return, CFG::Unexpected_Continue)
216 #endif
217 {
218  // handle different types of nodes.
219  switch (ir.GetStmtType(stmt)) {
220  case SIMPLE:
221  prev_node->add(stmt);
222  return SIMPLE;
223  case COMPOUND:
224  build_block(prev_node, ir.GetFirstInCompound(stmt), exit_nodes, break_nodes, return_nodes, continue_nodes);
225  return COMPOUND;
226  case BREAK:
227  prev_node->add(stmt);
228  if (break_nodes == 0)
229  throw CFG::Unexpected_Break();
230  else
231  break_nodes->push_back(NodeLabel(prev_node, BREAK_EDGE));
232  return BREAK;
233  case RETURN:
234  prev_node->add(stmt);
235  if (return_nodes == 0)
236  throw CFG::Unexpected_Return();
237  else
238  return_nodes->push_back(NodeLabel(prev_node, RETURN_EDGE));
239  return RETURN;
240  case LOOP:
241  return build_CFG_loop (prev_node, stmt, exit_nodes, return_nodes);
242  case STRUCT_TWOWAY_CONDITIONAL:
243  return build_CFG_twoway_branch (prev_node, stmt, exit_nodes, break_nodes,
244  return_nodes, continue_nodes);
245  case STRUCT_MULTIWAY_CONDITIONAL:
246  if (ir.IsBreakImplied(stmt))
247  return build_CFG_multiway_branch (prev_node, stmt, exit_nodes, break_nodes,
248  return_nodes, continue_nodes);
249  else
250  return build_CFG_multiway_branch_with_fallthrough (prev_node, stmt, exit_nodes,
251  return_nodes, continue_nodes);
252  case END_TESTED_LOOP:
253  return build_CFG_end_tested_loop (prev_node, stmt, exit_nodes, return_nodes);
254  case LOOP_CONTINUE:
255  prev_node->add(stmt);
256  if (continue_nodes == 0)
257  throw CFG::Unexpected_Continue();
258  else
259  continue_nodes->push_back(NodeLabel(prev_node, CONTINUE_EDGE));
260  return LOOP_CONTINUE;
261  //
262  // These unstructured constructs have the possibility of being
263  // delayed branches (depending on the underlying machine). Those
264  // without delay slots are handled now. Those with delay slots
265  // are just added as simple statements. A post-processing step
266  // will finalize control flow for them.
267  //
268  case UNCONDITIONAL_JUMP:
269  if (ir.NumberOfDelaySlots(stmt) == 0) {
270  return build_CFG_unconditional_jump (prev_node, stmt);
271  } else {
272  prev_node->add(stmt);
273  return SIMPLE;
274  }
275  case USTRUCT_TWOWAY_CONDITIONAL_T:
276  case USTRUCT_TWOWAY_CONDITIONAL_F:
277  if (ir.NumberOfDelaySlots(stmt) == 0) {
278  return build_CFG_ustruct_twoway_branch (prev_node, stmt, exit_nodes);
279  } else {
280  prev_node->add(stmt);
281  return SIMPLE;
282  }
283  case UNCONDITIONAL_JUMP_I:
284  if (ir.NumberOfDelaySlots(stmt) == 0) {
285  return build_CFG_unconditional_jump_i (prev_node, stmt);
286  } else {
287  prev_node->add(stmt);
288  return SIMPLE;
289  }
290  case USTRUCT_MULTIWAY_CONDITIONAL:
291  // Currently assume multiways don't have delay slots.
292  assert(ir.NumberOfDelaySlots(stmt) == 0);
293  return build_CFG_ustruct_multiway_branch (prev_node, stmt);
294 
295  // FIXME: Unimplemented.
296  case ALTERNATE_PROC_ENTRY:
297  cout << "FIXME: ALTERNATE_PROC_ENTRY not yet implemented" << endl;
298  ROSE_ABORT ();
299  default:
300  ROSE_ABORT ();
301  }
302  ROSE_ABORT (); // eraxxon: eliminate compiler warnings about missing return
303 }

333 IRStmtType
334 CFG::build_CFG_loop (CFG::Node* prev_node, StmtHandle th, CFG::NodeLabelList& exit_nodes,
335  CFG::NodeLabelList* return_nodes)
336 {
337  // If there is a header, (as in a "for" loop), it is assimilated in the previous (prev_node) CFG node. The
338  // condition needs a separate CFG node. So, unless prev_node is empty (in which case that can become the condition
339  // node) a new CFG node must be allocated for the loop condition.
340  CFG::Node* c;
341 
342  StmtHandle header = ir.LoopHeader(th);
343  if (header) {
344  prev_node->add(header);
345  }
346 
347  // To handle C and Fortran loops in the simplest way we take the following
348  // measures:
349  //
350  // If the number of loop iterations is defined
351  // at loop entry (i.e. Fortran semantics), we add the loop statement
352  // representative to the header node so that definitions from inside
353  // the loop don't reach the condition and increment specifications
354  // in the loop statement.
355  //
356  // On the other hand, if the number of iterations is not defined at
357  // entry (i.e. C semantics), we add the loop statement to a node that
358  // is inside the loop in the CFG so definitions inside the loop will
359  // reach uses in the conditional test. for C style semantics, the
360  // increment itself may be a separate statement. if so, it will appear
361  // explicitly at the bottom of the loop.
362  bool definedAtEntry = ir.LoopIterationsDefinedAtEntry(th);
363  if (definedAtEntry) {
364  prev_node->add(th);
365  }
366  if (prev_node->empty()) {
367  prev_node->add(th);
368  c = prev_node;
369  }
370  else {
371  c = new CFG::Node(th);
372  connect(prev_node, c, FALLTHROUGH_EDGE);
373  }
374  exit_nodes.push_back(NodeLabel(c, FALSE_EDGE));
375 
376  // allocate a new CFG node for the loop body and recur
377  CFG::NodeLabelList x_nodes;
378  IRStmtIterator *body = ir.LoopBody(th);
379  if (body->IsValid() && body->Current()) {
380  Node* b = new CFG::Node();
381  add(b);
382  NodeLabelList break_statements;
383  NodeLabelList continue_statements;
384  if (build_block(b, body, x_nodes, &break_statements, return_nodes, &continue_statements) == SIMPLE)
385  x_nodes.push_back(NodeLabel(b, FALLTHROUGH_EDGE));
386 
387  connect(c, b, TRUE_EDGE);
388  NodeLabelListIterator break_iter(&break_statements);
389  while ((bool)break_iter) {
390  exit_nodes.push_back((NodeLabel)break_iter);
391  ++break_iter;
392  }
393 
394  NodeLabelListIterator continue_iter(&continue_statements);
395  while ((bool)continue_iter) {
396  x_nodes.push_back((NodeLabel)continue_iter);
397  ++continue_iter;
398  }
399  }
400  else // empty body
401  x_nodes.push_back(NodeLabel(c, TRUE_EDGE));
402 
403  // allocate a new CFG node for the increment part, if needed
404  StmtHandle incr = ir.GetLoopIncrement(th);
405  if (incr) {
406  CFG::Node* li = new CFG::Node(incr);
407  connect(x_nodes, li);
408  connect(li, c, FALLTHROUGH_EDGE);
409  }
410  else
411  connect(x_nodes, c);
412 
413  return LOOP;
414 }

440 IRStmtType
441 CFG::build_CFG_end_tested_loop (CFG::Node* prev_node, StmtHandle th, CFG::NodeLabelList& exit_nodes,
442  CFG::NodeLabelList* return_nodes)
443 {
444  // FIXME: New construct, partially tested.
445  // New CFG nodes must be allocated for both the body and loop condition (we can re-use
446  // prev_node for the body if it is empty).
447  Node* b = NULL;
448  Node* c = NULL;
449 
450  // Process the loop body.
451  CFG::NodeLabelList x_nodes;
452  IRStmtIterator *body = ir.LoopBody(th);
453  if (body->IsValid() && body->Current()) {
454  if (prev_node->empty()) {
455  b = prev_node;
456  } else {
457  b = new CFG::Node();
458  add(b);
459  connect(prev_node, b, FALLTHROUGH_EDGE);
460  }
461  NodeLabelList break_statements;
462  NodeLabelList continue_statements;
463  if (build_block(b, body, x_nodes, &break_statements, return_nodes, &continue_statements) == SIMPLE)
464  x_nodes.push_back(NodeLabel(b, FALLTHROUGH_EDGE));
465 
466  NodeLabelListIterator break_iter(&break_statements);
467  while ((bool)break_iter) {
468  exit_nodes.push_back((NodeLabel)break_iter);
469  ++break_iter;
470  }
471 
472  NodeLabelListIterator continue_iter(&continue_statements);
473  while ((bool)continue_iter) {
474  x_nodes.push_back((NodeLabel)continue_iter);
475  ++continue_iter;
476  }
477  }
478  else // empty body
479  // FIXME: c is NULL at this point...
480  x_nodes.push_back(NodeLabel(c, TRUE_EDGE));
481 
482  // Allocate a node for the condition and make the proper connections.
483  c = new CFG::Node(th);
484  exit_nodes.push_back(NodeLabel(c, FALSE_EDGE));
485  connect(c, b, TRUE_EDGE);
486  connect(x_nodes, c);
487 
488  return END_TESTED_LOOP;
489 }

512 IRStmtType
513 CFG::build_CFG_twoway_branch (CFG::Node* prev_node, StmtHandle th, CFG::NodeLabelList& exit_nodes,
514  CFG::NodeLabelList* break_nodes, CFG::NodeLabelList* return_nodes,
515  CFG::NodeLabelList* continue_nodes)
516 {
517  // add the if statement itself to the previous block. it represents
518  // the conditional branch terminating the block.
519  prev_node->add(th);
520 
521  // construct the CFG for the true block
522  CFG::Node* b;
523  IRStmtIterator *true_body = ir.TrueBody(th);
524  if (true_body->IsValid() && true_body->Current()) {
525  b = new CFG::Node();
526  add(b);
527  if (build_block(b, true_body, exit_nodes, break_nodes, return_nodes, continue_nodes) == SIMPLE)
528  exit_nodes.push_back(NodeLabel(b, FALLTHROUGH_EDGE));
529 
530  connect(prev_node, b, TRUE_EDGE);
531  }
532  else {
533  // the body is empty
534  b = 0;
535  exit_nodes.push_back(NodeLabel(prev_node, TRUE_EDGE));
536  }
537 
538  // handle the false block
539  IRStmtIterator *false_body = ir.ElseBody(th);
540  if (false_body->IsValid() && false_body->Current()) {
541  CFG::Node* c_body = new CFG::Node();
542  add(c_body);
543  if (build_block(c_body, false_body, exit_nodes, break_nodes, return_nodes, continue_nodes) == SIMPLE)
544  exit_nodes.push_back(NodeLabel(c_body, FALLTHROUGH_EDGE));
545 
546  connect(prev_node, c_body, FALSE_EDGE);
547  }
548  else if (b != 0) // if b is 0 then the "condition" node has already been added to exit_nodes
549  exit_nodes.push_back(NodeLabel(prev_node, FALSE_EDGE));
550 
551  return STRUCT_TWOWAY_CONDITIONAL;
552 }

576 IRStmtType
577 CFG::build_CFG_multiway_branch (CFG::Node* prev_node, StmtHandle th, CFG::NodeLabelList& exit_nodes,
578  CFG::NodeLabelList* break_nodes, CFG::NodeLabelList* return_nodes,
579  CFG::NodeLabelList* continue_nodes)
580 {
581  // add the switch statement to the previous block. the switch statement
582  // represents the multi-way conditional branch terminating the block.
583  prev_node->add(th);
584 
585  // iterate over the multi-branches
586  bool direct_drop = false; // should there be a direct edge from the condition to exit?
587 
588  for (int bridx = 0; bridx < ir.NumMultiCases(th); bridx++) {
589  ExprHandle multiCond = ir.GetSMultiCondition(th, bridx);
590  IRStmtIterator *body = ir.MultiBody(th, bridx);
591  if (body->IsValid() && body->Current()) {
592  CFG::Node* c = new CFG::Node();
593  add(c);
594  if (build_block(c, body, exit_nodes, break_nodes, return_nodes, continue_nodes) == SIMPLE)
595  exit_nodes.push_back(NodeLabel(c, FALLTHROUGH_EDGE));
596  connect(prev_node, c, MULTIWAY_EDGE, multiCond);
597  } // if branch != 0
598  else
599  direct_drop = true;
600  } /* for */
601 
602  // check for catch all case
603  IRStmtIterator *catchall = ir.GetMultiCatchall(th);
604  if (catchall->IsValid() && catchall->Current()) {
605  CFG::Node* c = new CFG::Node();
606  add(c);
607  if (build_block(c, catchall, exit_nodes, break_nodes, return_nodes, continue_nodes) == SIMPLE)
608  exit_nodes.push_back(NodeLabel(c, FALLTHROUGH_EDGE));
609  connect(prev_node, c, MULTIWAY_EDGE); // originally "default" ???
610  }
611  else
612  direct_drop = true;
613 
614  // direct drop is true iff there is not catchall or at least one multi-branch body is null
615  if (direct_drop)
616  exit_nodes.push_back(NodeLabel(prev_node, FALLTHROUGH_EDGE));
617 
618  return STRUCT_MULTIWAY_CONDITIONAL;
619 }

633 IRStmtType
634 CFG::build_CFG_multiway_branch_with_fallthrough (CFG::Node* prev_node, StmtHandle th, CFG::NodeLabelList& exit_nodes,
635  CFG::NodeLabelList* return_nodes, CFG::NodeLabelList* continue_nodes)
636 {
637  // add the switch statement to the previous block. the switch statement
638  // represents the multi-way conditional branch terminating the block.
639  prev_node->add(th);
640 
641  bool direct_drop = false; // should there be a direct edge from the condition to exit?
642 
643  // We must create a new list of break statements since any breaks seen
644  // in subtrees of this statement will exit this switch, not some other
645  // enclosing construct.
646  NodeLabelList break_switch_statements;
647 
648  // Track the last case body processed so that fall-through connections can be made
649  // This will be non-null whenever the previous case falls through (i.e., did not end in a
650  // break statement).
651  CFG::Node* prev_case_node = 0;
652 
653  // iterate over the multi-branches
654  for (int bridx = 0; bridx < ir.NumMultiCases(th); bridx++) {
655  ExprHandle multiCond = ir.GetSMultiCondition(th, bridx);
656  IRStmtIterator *body = ir.MultiBody(th, bridx);
657  if (body->IsValid() && body->Current()) {
658  CFG::Node* c = new CFG::Node();
659  add(c);
660  break_switch_statements.clear();
661  // If a previous case falls-through, connect it to the current case node.
662  if (prev_case_node) {
663  connect(prev_case_node, c, FALLTHROUGH_EDGE);
664  prev_case_node = 0;
665  }
666 
667  // A dummy placeholder for exit nodes. We don't want the case body to be an exit from the
668  // contruct unless there is a break.
669  NodeLabelList x_nodes;
670  if (build_block(c, body, x_nodes, &break_switch_statements, return_nodes, continue_nodes) == SIMPLE
671  || break_switch_statements.size() == 0) {
672  // We didn't see a break.
673  prev_case_node = c;
674  }
675  // There should only be one break, but iterate anyway.
676  NodeLabelListIterator break_stmt_iter(&break_switch_statements);
677  while ((bool)break_stmt_iter) {
678  exit_nodes.push_back((NodeLabel)break_stmt_iter);
679  ++break_stmt_iter;
680  }
681  connect(prev_node, c, MULTIWAY_EDGE, multiCond);
682  } // if branch != 0
683  else
684  direct_drop = true;
685  } /* for */
686 
687  // check for catch all case
688  IRStmtIterator *catchall = ir.GetMultiCatchall(th);
689  if (catchall->IsValid() && catchall->Current()) {
690  CFG::Node* c = new CFG::Node();
691  add(c);
692  break_switch_statements.clear();
693  // If the previous case falls-through, connect it to the default node.
694  if (prev_case_node) {
695  connect(prev_case_node, c, FALLTHROUGH_EDGE);
696  prev_case_node = 0;
697  }
698  if (build_block(c, catchall, exit_nodes, &break_switch_statements, return_nodes, continue_nodes) == SIMPLE)
699  exit_nodes.push_back(NodeLabel(c, FALLTHROUGH_EDGE));
700  connect(prev_node, c, MULTIWAY_EDGE); // originally "default" ???
701  }
702  else
703  direct_drop = true;
704 
705  // direct drop is true iff there is not catchall or at least one multi-branch body is null
706  if (direct_drop)
707  exit_nodes.push_back(NodeLabel(prev_node, FALLTHROUGH_EDGE));
708 
709  return STRUCT_MULTIWAY_CONDITIONAL;
710 }

721 IRStmtType
722 CFG::build_CFG_unconditional_jump (CFG::Node* prev_node, StmtHandle stmt)
723 {
