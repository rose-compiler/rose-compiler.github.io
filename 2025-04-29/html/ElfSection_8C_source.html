39 {

41  ROSE_ASSERT(strsec);
42  ROSE_ASSERT(id>=0);
43 
44  /* Purpose */
45  switch (shdr->get_sh_type()) {
46  case SgAsmElfSectionTableEntry::SHT_PROGBITS:
47  set_purpose(SP_PROGRAM);
48  break;
49  case SgAsmElfSectionTableEntry::SHT_STRTAB:
50  set_purpose(SP_HEADER);
51  break;
52  case SgAsmElfSectionTableEntry::SHT_DYNSYM:
53  case SgAsmElfSectionTableEntry::SHT_SYMTAB:
54  set_purpose(SP_SYMTAB);
55  break;
56  default:
57  set_purpose(SP_OTHER);
58  break;
59  }
60 
61  /* File mapping */
62  set_offset(shdr->get_sh_offset());
63  if (SgAsmElfSectionTableEntry::SHT_NULL==shdr->get_sh_type() ||
64  SgAsmElfSectionTableEntry::SHT_NOBITS==shdr->get_sh_type()) {
65  set_size(0);
66  } else {
67  set_size(shdr->get_sh_size());
68  }
69  set_fileAlignment(shdr->get_sh_addralign());
70  grabContent();
71 
72  /* Memory mapping */
73  if (shdr->get_sh_addr() > 0) {
74  set_mappedPreferredRva(shdr->get_sh_addr());
75  set_mappedActualVa(0); /*will be assigned by Loader*/
76  set_mappedSize(shdr->get_sh_size());
77  set_mappedReadPermission(true);
78  set_mappedWritePermission((shdr->get_sh_flags() & 0x01) == 0x01);
79  set_mappedExecutePermission((shdr->get_sh_flags() & 0x04) == 0x04);
80  set_mappedAlignment(shdr->get_sh_addralign());
81  } else {
82  set_mappedPreferredRva(0);
83  set_mappedActualVa(0); /*will be assigned by Loader*/
84  set_mappedSize(0);
85  set_mappedReadPermission(false);
86  set_mappedWritePermission(false);
87  set_mappedExecutePermission(false);
88  set_mappedAlignment(shdr->get_sh_addralign());
89  }
90 
91  /* Name. This has to be near the end because possibly strsec==this, in which case we have to call set_size() to extend the
92  * section to be large enough before we can try to look up the name. */
93  set_id(id);
94  set_name(new SgAsmStoredString(strsec->get_strtab(), shdr->get_sh_name()));
95 
96  /* Add section table entry to section */
97  set_sectionEntry(shdr);
98  shdr->set_parent(this);

179 Rose::BinaryAnalysis::Address
180 SgAsmElfSection::calculateSizes(size_t r32size, size_t r64size, /*size of required parts*/
181  const std::vector<size_t> &optsizes, /*size of optional parts and number of parts parsed*/
182  size_t *entsize, size_t *required, size_t *optional, size_t *entcount) const
183 {
184  size_t struct_size = 0;
185  size_t extra_size = 0;
186  size_t entry_size = 0;
187  size_t nentries = 0;
188  SgAsmElfFileHeader *fhdr = get_elfHeader();
189 
190  /* Assume ELF Section Table Entry is correct for now for the size of each entry in the table. */
191  ROSE_ASSERT(get_sectionEntry()!=NULL);
192  entry_size = get_sectionEntry()->get_sh_entsize();
193 
194  /* Size of required part of each entry */
195  if (0==r32size && 0==r64size) {
196  /* Probably called by four-argument SgAsmElfSection::calculateSizes and we don't know the sizes of the required parts
197  * because there isn't a parser for this type of section, or the section doesn't contain a table. In the latter case
198  * the ELF Section Table has a zero sh_entsize and we'll treat the section as if it were a table with one huge entry.
199  * Otherwise we'll assume that the struct size is the same as the sh_entsize and there's no optional data. */
200  struct_size = entry_size>0 ? entry_size : get_size();
201  } else if (4==fhdr->get_wordSize()) {
202  struct_size = r32size;
203  } else if (8==fhdr->get_wordSize()) {
204  struct_size = r64size;
205  } else {
206  throw FormatError("bad ELF word size");
207  }
208 
209  /* Entire entry should be at least large enough for the required part. This also takes care of the case when the ELF
210  * Section Table Entry has a zero-valued sh_entsize */
211  entry_size = std::max(entry_size, struct_size);
212 
213  /* Size of optional parts. If we've parsed the table then use the largest optional part, otherwise assume the entry from
214  * the ELF Section Table is correct. */
215  nentries = optsizes.size();
216  if (nentries>0) {
217  for (size_t i=0; i<nentries; i++) {
218  extra_size = std::max(extra_size, optsizes[i]);
219  }
220  entry_size = std::min(entry_size, struct_size+extra_size);
221  } else {
222  extra_size = entry_size - struct_size;
223  nentries = entry_size>0 ? get_size() / entry_size : 0;
224  }
225 
226  /* Return values */
227  if (entsize)
228  *entsize = entry_size;
229  if (required)
230  *required = struct_size;
231  if (optional)
232  *optional = extra_size;
233  if (entcount)
234  *entcount = nentries;
235  return entry_size * nentries;
236 }

250 bool
251 SgAsmElfSection::reallocate()
252 {
253  bool reallocated = false;
254  SgAsmElfSectionTableEntry *sechdr = get_sectionEntry();
255  SgAsmElfSegmentTableEntry *seghdr = get_segmentEntry();
256 
257  /* Change section size if this section was defined in the ELF Section Table */
258  if (sechdr!=NULL) {
259  Rose::BinaryAnalysis::Address need = calculateSizes(NULL, NULL, NULL, NULL);
260  if (need < get_size()) {
261  if (isMapped()) {
262  ROSE_ASSERT(get_mappedSize()==get_size());
263  set_mappedSize(need);
264  }
265  set_size(need);
266  reallocated = true;
267  } else if (need > get_size()) {
268  get_file()->shiftExtend(this, 0, need-get_size(), SgAsmGenericFile::ADDRSP_ALL, SgAsmGenericFile::ELASTIC_HOLE);
269  reallocated = true;
270  }
271  }
272 
273  /* Update entry in the ELF Section Table and/or ELF Segment Table */
274  if (sechdr)
275  sechdr->updateFromSection(this);
276  if (seghdr)
277  seghdr->updateFromSection(this);
278 
279  return reallocated;
280 }
