570 void SystemDependenceGraph::doInterproceduralConnections(InterproceduralInfo * ii)
571 {
572  //connect the retunring nodes from the function with the call-site
573  SgFunctionCallExp * callSite;
574  std::vector<InterproceduralInfo*> functionTargetList;
575 
576  // get all callsites within this function
577  for (int i=0;i<ii->callSiteCount();i++)
578  {
579  // get the SgFunctionCallExp node
580  callSite=isSgFunctionCallExp(ii->getFunctionCallExpNode(i));
581 
582  // cout << "Found funcall: " << callSite->unparseToString() << endl;
583 
584  // do analysis of the function call and determine the possible call-targets
585  functionTargetList=getPossibleFuncs(callSite);
586  // for every possible callable function
587  for (unsigned int j=0;j<functionTargetList.size();j++)
588  {

590 
591  ROSE_ASSERT(calledFunction!=NULL);
592 
593

595  // insert the function_formal_in and formal_out nodes, just to be safe
596 //CMI modified 19082007 establishEdge(getNode(DependenceNode::ENTRY,calledFunction->getFunctionEntry()),getNode(DependenceNode::FORMALRETURN,calledFunction->getFormalReturn()),CONTROL);
597  // and the formal out for every parameter
598  for (int k=0;k<calledFunction->getFormalCount();k++)
599  {
600  establishEdge(getNode(DependenceNode::ENTRY,calledFunction->getFunctionEntry()),getNode(DependenceNode::FORMALIN,calledFunction->getFormal(k)),CONTROL);
601  }
