241 void
242 SgAsmGenericStrtab::free(Address offset, Address size)
243 {
244  if (offset==SgAsmGenericString::unallocated || 0==size)
245  return;
246 
247  ROSE_ASSERT(offset+size <= get_container()->get_size());
248  set_isModified(true);
249 
250  /* Make sure area is not already in free list. The freelist.insert() handles this gracefully, but if we're freeing
251  * something that's already in the list then we have a logic error somewhere. */
252  ROSE_ASSERT(!get_freeList().overlaps(AddressInterval::baseSize(offset, size)));
253 
254  /* Preserve anything that's still referenced. The caller should have assigned SgAsmStoredString::unalloced to the "offset"
255  * member of the string storage to indicate that it's memory in the string table is no longer in use. */
256  AddressIntervalSet toFree;
257  toFree.insert(AddressInterval::baseSize(offset, size));
258  for (size_t i=0; i<p_storageList.size(); i++) {
259  SgAsmStringStorage *storage = p_storageList[i];
260  if (storage->get_offset()!=SgAsmGenericString::unallocated)
261  toFree.erase(AddressInterval::baseSize(storage->get_offset(), get_storageSize(storage)));
262  }
263 
264  /* Add un-referenced extents to free list. */
265  get_freeList().insertMultiple(toFree);
266 }

275 SgAsmGenericStrtab::freeAllStrings(bool blow_away_holes)
276 {
277  SgAsmGenericSection *container = get_container();
278  SgAsmGenericFile *file = container->get_file();
279  bool is_tracking = file->get_trackingReferences();
280  set_isModified(true);
281 
282  /* Mark all storage objects as being unallocated. Never free the dont_free storage (if any). */
283  for (size_t i=0; i<p_storageList.size(); i++) {
284  if (p_storageList[i]->get_offset()!=SgAsmGenericString::unallocated && p_storageList[i]!=get_dontFree()) {
285  p_numberFreed++;
286  p_storageList[i]->set_offset(SgAsmGenericString::unallocated);
287  }
288  }
289 
290  /* Mark holes as referenced */
291  if (blow_away_holes) {
292  file->set_trackingReferences(true);
293  file->markReferencedExtent(container->get_offset(), container->get_size());
294  file->set_trackingReferences(is_tracking);
295  }
296 
297  /* The free list is everything that's been referenced in the container section. */
298  get_freeList() = container->get_referencedExtents();
299 
300  /* Remove the empty string from the free list */
301  if (get_dontFree())
302  get_freeList().erase(AddressInterval::baseSize(get_dontFree()->get_offset(), get_dontFree()->get_string().size()+1));

305 bool
306 SgAsmGenericStrtab::reallocate(bool shrink)
307 {
308  bool reallocated = false;
309  SgAsmGenericSection *container = get_container();
310  Address extend_size = 0; /* amount by which to extend string table */
311 
312  /* Get list of strings that need to be allocated and sort by descending size. */
313  std::vector<size_t> map;
314  for (size_t i=0; i<p_storageList.size(); i++) {
315  SgAsmStringStorage *storage = p_storageList[i];
316  if (storage->get_offset()==SgAsmGenericString::unallocated) {
317  map.push_back(i);
318  }
319  }
320  for (size_t i=1; i<map.size(); i++) {
321  for (size_t j=0; j<i; j++) {
322  if (p_storageList[map[j]]->get_string().size() < p_storageList[map[i]]->get_string().size()) {
323  size_t x = map[i];
324  map[i] = map[j];
325  map[j] = x;
326  }
327  }
328  }
329 
330  /* Allocate from largest to smallest so we have the best chance of finding overlaps */
331  for (size_t i=0; i<map.size(); i++) {
332  SgAsmStringStorage *storage = p_storageList[map[i]];
333  ROSE_ASSERT(storage->get_offset()==SgAsmGenericString::unallocated);
334 
335  /* We point empty strings at the dont_free storage if possible. */
336  if (storage->get_string()=="" && get_dontFree()) {
337  ROSE_ASSERT(get_dontFree()->get_string()=="");
338  storage->set_offset(0);
339  }
340 
341  /* If there's already a string with the same value then they can share space in the string table. They're still
342  * considered two separate strings, so changing one doesn't affect the other. */
343  if (storage->get_offset()==SgAsmGenericString::unallocated) {
344  for (size_t j=0; j<p_storageList.size(); j++) {
345  SgAsmStringStorage *previous = p_storageList[j];
346  if (previous->get_offset()!=SgAsmGenericString::unallocated && previous->get_string()==storage->get_string()) {
347  storage->set_offset(previous->get_offset());
348  break;
349  }
350  }
351  }
352 
353  /* Some string tables may be able to overlap strings. For instance, ELF can overlap "domain" and "main" since it
354  * encodes strings with NUL termination. */
355  if (storage->get_offset()==SgAsmGenericString::unallocated)
356  allocateOverlap(storage);
357 
358  /* If we couldn't share another string then try to allocate from free space (avoiding holes) */
359  if (storage->get_offset()==SgAsmGenericString::unallocated) {
360  try {
361  size_t need = storage->get_string().size() + 1;
362  AddressIntervalSet::ConstIntervalIterator iter = get_freeList().bestFit(need,
363  get_freeList().intervals().begin());
364  if (iter==get_freeList().intervals().end())
365  throw std::bad_alloc();
366  ASSERT_require(iter->size() >= need);
367  AddressInterval allocated = AddressInterval::baseSize(iter->least(), need);
368  get_freeList().erase(allocated);
369  Address new_offset = allocated.least();
370  storage->set_offset(new_offset);
371  } catch(std::bad_alloc &x) {
372  /* nothing large enough on the free list */
373  }
374  }
375 
376  /* If no free space area large enough then prepare to extend the section. */
377  if (storage->get_offset()==SgAsmGenericString::unallocated) {
378  extend_size += storage->get_string().size() + 1;
379  }
380  }
381 
382  /* If we were unable to allocate everything and there's still free space then it may be possible to reallocate all
383  * strings in order to repack the table and avoid internal fragmentation. */
384  //FIXME (RPM 2008-09-25)
385 
386  /* Change string table size as necessary. */
387  if (extend_size>0) {
388  /* The string table isn't large enough, so make it larger by extending the section that contains the table. The
389  * containing section's "set_size" method should add the new space to the string table's free list. If our recursion
390  * level is more than two calls deep then something went horribly wrong! */
391  reallocated = true;
392  container->get_file()->shiftExtend(container, 0, extend_size);
393  reallocate(false);
394  } else if (shrink && get_freeList().size()>0) {
395  /* See if we can release any address space and shrink the containing section. The containing section's "set_size"
396  * method will adjust the free list by removing some bytes from it. */
397  AddressIntervalSet::ConstIntervalIterator iter = get_freeList().intervals().end();
398  ASSERT_forbid(iter == get_freeList().intervals().begin());
399  --iter;
400  AddressInterval hi = *iter;
401  if (hi.least() + hi.size() == container->get_size())
402  container->set_size(hi.least());
403  }
404 
405  if (reallocated)
406  set_isModified(true);
407  return reallocated;
408 }
