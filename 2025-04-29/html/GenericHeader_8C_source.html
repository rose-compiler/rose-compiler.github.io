376 void
377 SgAsmGenericHeader::dump(FILE *f, const char *prefix, ssize_t idx) const
378 {
379  char p[4096];
380  if (idx>=0) {
381  snprintf(p, sizeof(p), "%sHeader[%zd].", prefix, idx);
382  } else {
383  snprintf(p, sizeof(p), "%sHeader.", prefix);
384  }
385  const int w = std::max(size_t{1}, DUMP_FIELD_WIDTH - strlen(p));
386 
387  SgAsmGenericSection::dump(f, p, -1);
388 
389  ROSE_ASSERT(get_executableFormat() != NULL);
390  get_executableFormat()->dump(f, p, -1);
391 
392  fprintf(f, "%s%-*s = 0x%x (%s)\n", p, w, "ins_arch",
393  p_isa, stringifySgAsmExecutableFileFormatInsSetArchitecture(p_isa).c_str());
394 
395  fprintf(f, "%s%-*s = \"", p, w, "magic");
396  for (size_t i = 0; i < p_magic.size(); i++) {
397  switch (p_magic[i]) {
398  case '\\': fputs("\\\\", f); break;
399  case '\n': fputs("\\n", f); break;
400  case '\r': fputs("\\r", f); break;
401  case '\t': fputs("\\t", f); break;
402  default:
403  if (isprint(p_magic[i])) {
404  fputc(p_magic[i], f);
405  } else {
406  fprintf(f, "\\%03o", (unsigned)p_magic[i]);
407  }
408  break;
409  }
410  }
411  fputs("\"\n", f);
412 
413  /* Base virtual address and entry addresses */
414  fprintf(f, "%s%-*s = 0x%08" PRIx64 " (%" PRIu64 ")\n", p, w, "base_va", get_baseVa(), get_baseVa());
415  fprintf(f, "%s%-*s = %" PRIuPTR " entry points\n", p, w, "entry_rva.size", get_entryRvas().size());
416  for (size_t i = 0; i < get_entryRvas().size(); i++) {
417  char label[64];
418  snprintf(label, sizeof(label), "entry_rva[%" PRIuPTR "]", i);
419  Rose::BinaryAnalysis::Address entry_rva = get_entryRvas()[i].rva();
420  fprintf(f, "%s%-*s = 0x%08" PRIx64 " (%" PRIu64 ")\n", p, w, label, entry_rva, entry_rva);
421  SgAsmGenericSectionPtrList sections = get_file()->get_sections();
422  dumpContainingSections(f, std::string(p)+label, entry_rva, sections);
423  }
424 
425  fprintf(f, "%s%-*s = %" PRIuPTR " sections\n", p, w, "section", p_sections->get_sections().size());
426  for (size_t i=0; i<p_sections->get_sections().size(); i++) {
427  SgAsmGenericSection *section = p_sections->get_sections()[i];
428  char label[1024];
429  snprintf(label, sizeof(label), "section[%" PRIuPTR "]", i);
430  fprintf(f, "%s%-*s = [%d] \"%s\"\n", p, w, label, section->get_id(), section->get_name()->get_string(true).c_str());
431  }
432 
433  fprintf(f, "%s%-*s = %" PRIuPTR " entries\n", p, w, "DLL.size", p_dlls->get_dlls().size());
434  for (size_t i = 0; i < p_dlls->get_dlls().size(); i++)
435  p_dlls->get_dlls()[i]->dump(f, p, i);
436 }
