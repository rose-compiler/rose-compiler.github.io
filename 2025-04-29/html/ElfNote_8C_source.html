49 SgAsmElfNoteEntry::parse(Rose::BinaryAnalysis::Address at)
50 {
51  /* Find the section holding this note */
52  SgAsmElfNoteSection *notes = SageInterface::getEnclosingNode<SgAsmElfNoteSection>(this);
53  ROSE_ASSERT(notes!=NULL);
54  ROSE_ASSERT(at < notes->get_size());
55  SgAsmElfFileHeader *fhdr = dynamic_cast<SgAsmElfFileHeader*>(notes->get_header());
56  ROSE_ASSERT(fhdr!=NULL);
57 
58  /* Length of note entry name, including NUL termination */
59  uint32_t u32;
60  notes->readContentLocal(at, &u32, 4);
61  size_t name_size = diskToHost(fhdr->get_sex(), u32);
62  at += 4;
63 
64  /* Length of note entry description (i.e., the payload) */
65  notes->readContentLocal(at, &u32, 4);
66  size_t payload_size = diskToHost(fhdr->get_sex(), u32);
67  at += 4;
68 
69  /* Type of note */
70  notes->readContentLocal(at, &u32, 4);
71  unsigned type = diskToHost(fhdr->get_sex(), u32);
72  at += 4;
73 
74  /* NUL-terminated name */
75  std::string note_name = notes->readContentLocalString(at);
76  if (note_name.size() + 1 > name_size && name_size > 0)
77  note_name = note_name.substr(0, name_size-1);
78  at += name_size;
79  at = (at+3) & ~0x3; /* payload is aligned on a four-byte offset */
80 
81  /* Set properties */
82  get_name()->set_string(note_name);
83  set_type(type);
84  p_payload = notes->readContentLocalUcl(at, payload_size);
85 
86  return at + payload_size;
87 }
