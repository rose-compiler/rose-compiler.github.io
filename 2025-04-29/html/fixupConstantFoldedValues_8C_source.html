30 static void deleteExpressionAndOriginalExpressionTree(SgNode * node) {
31  // Skip node that have already been deleted
32  if (node->get_freepointer() != AST_FileIO::IS_VALID_POINTER()) return;
33 
34  // If it is an expression delete the associated original expression tree
35  SgExpression * exp = isSgExpression(node);
36  if (exp != NULL && exp->get_originalExpressionTree() != NULL) {
37  deleteExpressionAndOriginalExpressionTree(exp->get_originalExpressionTree());
38  }
39 
40  // Traverse the AST and delete successors
41  std::vector<SgNode*> successors = node->get_traversalSuccessorContainer();
42  for (std::vector<SgNode*>::iterator n = successors.begin(); n != successors.end(); ++n) {
43  if (*n != NULL ) {
44  deleteExpressionAndOriginalExpressionTree(*n);
45  }
46  }
47 
48  delete node;
49 }

102  if (expr != NULL) {

110  if (expr_parent == NULL) {

116  originals.insert(expr);
117  return;

121  if (std::find(successors.begin(), successors.end(), node) != successors.end()) {
