209 void
210 RiscOperators::writeRegister(RegisterDescriptor reg, const BS::SValue::Ptr &value) {
211  if (currentInstruction() && hadSharedMemoryAccess()) {
212  Sawyer::Message::Stream debug(mlog[DEBUG]);
213 
214  RegisterDictionary::Ptr regdict = currentState()->registerState()->registerDictionary();
215 
216  // This probably writing a previously-read value from shared memory into a register. It's common on RISC
217  // architectures to have an instruction that copies a value from memory into a register, in which case the value
218  // being written here is just the variable we're using to represent what was read from shared memory. On CISC
219  // architectures the value being written might be a more complex function of the value that was read.
220  //
221  // In cany case, since we can't directly adjust the byte read from concrete memory (not even by pre-writing to that
222  // address since it might be read-only or it might not follow normal memory semantics), we do the next best thing: we
223  // adjust all the side effects of the instruction that read the byte from memory.
224  SymbolicExpression::Ptr valueExpr = Emulation::SValue::promote(value)->get_expression();
225  SAWYER_MESG(debug) <<" register update for shared memory read\n"
226  <<" register = " <<regdict->quadAndName(reg) <<"\n"
227  <<" value = " <<*valueExpr <<"\n";
228  auto event = ExecutionEvent::registerWrite(process()->testCase(), process()->nextEventLocation(When::POST),
229  currentInstruction()->get_address(), reg, SymbolicExpression::Ptr(),
230  SymbolicExpression::Ptr(), valueExpr);
231  event->name(hadSharedMemoryAccess()->name() + "_toreg");
232  database()->save(event);
233  SAWYER_MESG(debug) <<" created " <<event->printableName(database()) <<"\n";
234  }
235  Super::writeRegister(reg, value);
236 }

238 BS::SValue::Ptr
239 RiscOperators::readMemory(RegisterDescriptor segreg, const BS::SValue::Ptr &addr,
240  const BS::SValue::Ptr &dfltUnused, const BS::SValue::Ptr &cond) {
241  if (isRecursive()) {
242  // Don't do anything special because we're being called from inside a user callback.
243  return Super::readMemory(segreg, addr, dfltUnused, cond);
244 
245  } else if (auto va = addr->toUnsigned()) {
246  // Read the memory's value symbolically, and if we don't have a value then read it concretely and use the concrete value to
247  // update the symbolic state. However, we can only read it concretely if the address is a constant.
248  size_t nBytes = dfltUnused->nBits() / 8;
249  SymbolicExpression::Ptr concrete = SymbolicExpression::makeIntegerConstant(dfltUnused->nBits(),
250  process_->readMemoryUnsigned(*va, nBytes));
251 
252  // Handle shared memory by invoking user-defined callbacks
253  SharedMemoryCallbacks callbacks = process()->sharedMemory().getOrDefault(*va);
254  if (!callbacks.isEmpty()) {
255  isRecursive(true);
256  BOOST_SCOPE_EXIT(this_) {
257  this_->isRecursive(false);
258  } BOOST_SCOPE_EXIT_END;
259 
260  // FIXME[Robb Matzke 2021-09-09]: use structured bindings when ROSE requires C++17 or later
261  auto x = process()->sharedMemoryRead(callbacks, partitioner(), RiscOperators::promote(shared_from_this()),
262  *va, nBytes);
263  ExecutionEvent::Ptr sharedMemoryEvent = x.first;
264  SymbolicExpression::Ptr result = x.second;
265 
266  if (result) {
267  hadSharedMemoryAccess(sharedMemoryEvent);
268  return svalueExpr(result);
269  }
270  }
271 
272  SValue::Ptr dflt = SValue::promote(undefined_(dfltUnused->nBits()));
273  dflt->set_expression(concrete);
274  return Super::readMemory(segreg, addr, dflt, cond);
275 
276  } else {
277  // Not a concrete address, so do things symbolically
278  return Super::readMemory(segreg, addr, dfltUnused, cond);
279  }
280 }
