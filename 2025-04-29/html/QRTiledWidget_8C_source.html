172 void QRTiledWidget::setTileSize(int size0, ...) {
173  if (!m_splitter)
174  throw QRException("QRTiledWidget::setTileSize - invalid window/panel: must use splitter!");
175 
176  QRWindow *win = QROSE::getQRWindow(this);
177 
178  va_list lst;
179 
180  va_start(lst, size0);
181 
182  bool is_shown = win->isVisible();
183  win->show();
184 
185  QList<int> rel_sizes;
186  int total_rel_size = 0;
187  QList<int> cur_sizes = m_splitter->sizes();
188 
189  int n = cur_sizes.count();
190  if (n == 0) return;
191 
192  QList<int>::Iterator it1 = cur_sizes.begin();
193 
194  int total_cur_size = 0;
195  int size = size0;
196  for (int i = 1; i < n; i++) {
197  total_rel_size += size;
198  if (total_rel_size < 0) {
199  total_rel_size = size = 0;
200  } else if (total_rel_size > 100) {
201  total_rel_size -= size; size = 0;
202  }
203  rel_sizes.push_back(size);
204  size = va_arg(lst, int);
205  total_cur_size += *it1; it1++;
206  }
207 
208  rel_sizes.push_back(100 - total_rel_size);
209  total_cur_size += *it1;
210 
211  it1 = cur_sizes.begin();
212  QList<int>::Iterator it2 = rel_sizes.begin();
213  int count = 0;
214  while( it1 != cur_sizes.end() ) {
215  int &val = *it1;
216  count += val;
217  val = (int) ((float) total_cur_size * (float) (*it2 / 100.0));
218  ++it1; ++it2;
219  if (it1 == cur_sizes.end()) {
220  if (count != total_cur_size) {
221  val = total_cur_size - (count - val);
222  }
223  }
224  }
225 
226  va_end(lst);
227  m_splitter->setSizes(cur_sizes);
228  win->setShown(is_shown);
229 
230 }
