288 void
289 RegisterStateGeneric::scanAccessedLocations(RegisterDescriptor reg, RiscOperators *ops,
290  RegPairs &accessedParts /*out*/, RegPairs &preservedParts /*out*/) const {
291  BitRange accessedLocation = BitRange::baseSize(reg.offset(), reg.nBits());
292  const RegPairs &pairList = registers_.getOrDefault(reg);
293  for (const RegPair &regpair: pairList) {
294  BitRange storedLocation = regpair.location(); // the thing that's already stored in this state
295  BitRange overlap = storedLocation & accessedLocation;
296  if (overlap.isEmpty())
297  continue;
298 
299  // Low-order bits of stored part that are not needed for this access
300  if (overlap.least() > storedLocation.least()) {
301  size_t nbits = overlap.least() - storedLocation.least();
302  RegisterDescriptor subreg(reg.majorNumber(), reg.minorNumber(), storedLocation.least(), nbits);
303  preservedParts.push_back(RegPair(subreg, ops->unsignedExtend(regpair.value, nbits)));
304  }
305 
306  // Bits of the part that are needed for this access.
307  {
308  RegisterDescriptor subreg(reg.majorNumber(), reg.minorNumber(), overlap.least(), overlap.size());
309  size_t extractBegin = overlap.least() - storedLocation.least();
310  size_t extractEnd = extractBegin + overlap.size();
311  accessedParts.push_back(RegPair(subreg, ops->extract(regpair.value, extractBegin, extractEnd)));
312  }
313 
314  // High-order bits of stored part that are not needed for this access
315  if (overlap.greatest() < storedLocation.greatest()) {
316  size_t nbits = storedLocation.greatest() - overlap.greatest();
317  RegisterDescriptor subreg(reg.majorNumber(), reg.minorNumber(), overlap.greatest()+1, nbits);
318  size_t extractBegin = overlap.greatest()+1 - storedLocation.least();
319  size_t extractEnd = extractBegin + nbits;
320  preservedParts.push_back(RegPair(subreg, ops->extract(regpair.value, extractBegin, extractEnd)));
321  }
322  }
323 }

338 SValue::Ptr
339 RegisterStateGeneric::readRegister(RegisterDescriptor reg, const SValue::Ptr &dflt, RiscOperators *ops) {
340  ASSERT_forbid(reg.isEmpty());
341  ASSERT_not_null(dflt);
342  ASSERT_require2(reg.nBits() == dflt->nBits(), "value being read must be same size as register" +
343  (boost::format(": %|u| -> %|u|") % reg.nBits() % dflt->nBits()).str());
344  ASSERT_not_null(ops);
345  assertStorageConditions("at start of read", reg);
346  BitRange accessedLocation = BitRange::baseSize(reg.offset(), reg.nBits());
347 #ifdef RegisterStateGeneric_20150924
348  const bool adjustLocations = false;
349 #else
350  const bool adjustLocations = accessModifiesExistingLocations_;
351 #endif
352 
353  // Fast case: the state does not store this register or any register that might overlap with this register.
354  if (!registers_.exists(reg)) {
355  if (!accessCreatesLocations_)
356  return dflt;
357  SValue::Ptr newval = dflt->copy();
358  std::string regname = regdict->lookup(reg);
359  boost::erase_all(regname, "[");
360  boost::erase_all(regname, "]");
361  if (!regname.empty() && newval->comment().empty())
362  newval->comment(regname + "_0");
363  registers_.insertMaybeDefault(reg).push_back(RegPair(reg, newval));
364  assertStorageConditions("at end of read", reg);
365  return newval;
366  }
367 
368  // Iterate over the storage/value pairs to figure out what parts of the register are already in existing storage locations,
369  // and which parts of those overlapping storage locations are not accessed.
370  RegPairs accessedParts; // parts of existing overlapping locations we access
371  RegPairs preservedParts; // parts of existing overlapping locations we don't access
372  RegPairs &pairList = registers_.insertMaybeDefault(reg);
373  scanAccessedLocations(reg, ops, accessedParts /*out*/, preservedParts /*out*/);
374  if (adjustLocations)
375  clearOverlappingLocations(reg);
376 
377  // Figure out which part of the access does not exist in the state
378  typedef Sawyer::Container::IntervalSet<BitRange> Locations;
379  Locations newLocations;
380  newLocations.insert(accessedLocation);
381  for (const RegPair &regpair: accessedParts)
382  newLocations -= regpair.location();
383 
384  // Create values for the parts of the accessed register that weren't stored in the state, but don't store them yet.
385  RegPairs newParts;
386  for (const BitRange &newLocation: newLocations.intervals()) {
387  RegisterDescriptor subreg(reg.majorNumber(), reg.minorNumber(), newLocation.least(), newLocation.size());
388  ASSERT_require(newLocation.least() >= reg.offset());
389  SValue::Ptr newval = ops->extract(dflt,
390  newLocation.least()-reg.offset(),
391  newLocation.greatest()+1-reg.offset());
392  newParts.push_back(RegPair(subreg, newval));
393  }
394 
395  // Construct the return value by combining all the parts we found or created.
396  SValue::Ptr retval;
397  RegPairs retvalParts = accessedParts;
398  retvalParts.insert(retvalParts.end(), newParts.begin(), newParts.end());
399  std::sort(retvalParts.begin(), retvalParts.end(), sortByOffset);
400  for (const RegPair &regpair: retvalParts)
401  retval = retval ? ops->concat(retval, regpair.value) : regpair.value;
402  ASSERT_require(retval->nBits() == reg.nBits());
403 
404  // Update the register state -- write parts that didn't exist and maybe combine or split some locations.
405  if (adjustLocations) {
406  // Remove parts of the pair list we're about to replace.
407  pairList.erase(std::remove_if(pairList.begin(), pairList.end(), has_null_value), pairList.end());
408  pairList.insert(pairList.end(), preservedParts.begin(), preservedParts.end()); // re-insert preserved parts
409  if (accessCreatesLocations_) {
410  // Combine all the accessed locations (just removed above) into a single location corresponding exactly to the
411  // return value.
412  pairList.push_back(RegPair(reg, retval));
413  } else {
414  // Add back the parts that were originally present and which we accessed, but not any new parts. They're added back
415  // in as large chunks as possible.
416  std::sort(accessedParts.begin(), accessedParts.end(), sortByOffset);
417  for (size_t i = 0; i<accessedParts.size(); ++i /*also incremented in body*/) {
418  // Find largest contiguous value starting at i.
419  SValue::Ptr part = accessedParts[i].value;
420  size_t firstOffset = accessedParts[i].desc.offset();
421  size_t nextOffset = firstOffset + accessedParts[i].desc.nBits();
422  while (i+1 < accessedParts.size() && accessedParts[i+1].desc.offset() == nextOffset) {
423  ++i;
424  nextOffset += accessedParts[i].desc.nBits();
425  part = ops->concat(part, accessedParts[i].value);
426  }
427  RegisterDescriptor tmpReg(reg.majorNumber(), reg.minorNumber(), firstOffset, nextOffset-firstOffset);
428  ASSERT_require(tmpReg.nBits() == part->nBits());
429  pairList.push_back(RegPair(tmpReg, part));
430  }
431  }
432  } else if (accessCreatesLocations_) {
433  // Don't adjust existing locations, just add locations that were not existing.
434  pairList.insert(pairList.end(), newParts.begin(), newParts.end());
435  }
436 
437  assertStorageConditions("at end of read", reg);
438  return retval;
439 }

491 void
492 RegisterStateGeneric::writeRegister(RegisterDescriptor reg, const SValue::Ptr &value, RiscOperators *ops)
493 {
494  ASSERT_not_null(value);
495  ASSERT_require2(reg.nBits()==value->nBits(), "value written to register must be the same width as the register" +
496  (boost::format(": %|u| -> %|u|") % value->nBits() % reg.nBits()).str());
497  ASSERT_not_null(ops);
498  assertStorageConditions("at start of write", reg);
499  BitRange accessedLocation = BitRange::baseSize(reg.offset(), reg.nBits());
500 
501  // Fast case: the state does not store this register or any register that might overlap with this register.
502  if (!registers_.exists(reg)) {
503  if (!accessCreatesLocations_)
504  throw RegisterNotPresent(reg);
505  registers_.insertMaybeDefault(reg).push_back(RegPair(reg, value));
506  assertStorageConditions("at end of write", reg);
507  return;
508  }
509 
510  // Check that we're allowed to add storage locations if necessary.
511  if (!accessCreatesLocations_) {
512  size_t nBitsFound = 0;
513  for (const RegPair &regpair: registers_.getOrDefault(reg))
514  nBitsFound += (regpair.location() & accessedLocation).size();
515  ASSERT_require(nBitsFound <= accessedLocation.size());
516  if (nBitsFound < accessedLocation.size())
517  throw RegisterNotPresent(reg);
518  }
519 
520  // Iterate over the storage/value pairs to figure out what parts of the register are already in existing storage locations,
521  // and which parts of those overlapping storage locations are not accessed.
522  RegPairs accessedParts; // parts of existing overlapping locations we access
523  RegPairs preservedParts; // parts of existing overlapping locations we don't access
524  RegPairs &pairList = registers_.insertMaybeDefault(reg);
525  scanAccessedLocations(reg, ops, accessedParts /*out*/, preservedParts /*out*/);
526  if (accessModifiesExistingLocations_)
527  clearOverlappingLocations(reg);
528 
529  // Figure out which part of the access does not exist in the state
530  typedef Sawyer::Container::IntervalSet<BitRange> Locations;
531  Locations newLocations;
532  newLocations.insert(accessedLocation);
533  for (const RegPair &regpair: accessedParts)
534  newLocations -= regpair.location();
535 
536  // Update the register state by writing to the accessed area.
537  if (accessModifiesExistingLocations_) {
538  // Combine all the accessed locations into a single location corresponding exactly to the written value, and split
539  // those previously existing locations that partly overlap the written location.
540  pairList.erase(std::remove_if(pairList.begin(), pairList.end(), has_null_value), pairList.end());
541  pairList.insert(pairList.end(), preservedParts.begin(), preservedParts.end());
542  pairList.push_back(RegPair(reg, value));
543  } else {
544  // Don't insert/erase locations that existed already -- only change their values.
545  for (RegPair &regpair: pairList) {
546  BitRange storedLocation = regpair.location();
547  if (BitRange overlap = storedLocation & accessedLocation) {
548  SValue::Ptr valueToWrite;
549  if (overlap.least() > storedLocation.least()) {
550  size_t nbits = overlap.least() - storedLocation.least();
551  SValue::Ptr loValue = ops->unsignedExtend(regpair.value, nbits);
552  valueToWrite = loValue;
553  }
554  {
555  size_t nbits = overlap.size();
556  size_t extractBegin = overlap.least() - accessedLocation.least();
557  size_t extractEnd = extractBegin + nbits;
558  SValue::Ptr midValue = ops->extract(value, extractBegin, extractEnd);
559  valueToWrite = valueToWrite ? ops->concat(valueToWrite, midValue) : midValue;
560  }
561  if (overlap.greatest() < storedLocation.greatest()) {
562  size_t nbits = storedLocation.greatest() - overlap.greatest();
563  size_t extractBegin = overlap.greatest()+1 - storedLocation.least();
564  size_t extractEnd = extractBegin + nbits;
565  SValue::Ptr hiValue = ops->extract(regpair.value, extractBegin, extractEnd);
566  valueToWrite = valueToWrite ? ops->concat(valueToWrite, hiValue) : hiValue;
567  }
568  ASSERT_not_null(valueToWrite);
569  ASSERT_require(valueToWrite->nBits() == regpair.desc.nBits());
570  regpair.value = valueToWrite;
571  }
572  }
573 
574  // Insert the parts that didn't exist before
575  for (const BitRange &newLocation: newLocations.intervals()) {
576  size_t extractBegin = newLocation.least() - accessedLocation.least();
577  size_t extractEnd = extractBegin + newLocation.size();
578  SValue::Ptr valueToWrite = ops->extract(value, extractBegin, extractEnd);
579  ASSERT_require(valueToWrite->nBits() == newLocation.size());
580  RegisterDescriptor subreg(reg.majorNumber(), reg.minorNumber(), newLocation.least(), newLocation.size());
581  pairList.push_back(RegPair(subreg, valueToWrite));

610 void
611 RegisterStateGeneric::erase_register(RegisterDescriptor reg, RiscOperators *ops)
612 {
613  ASSERT_forbid(reg.isEmpty());
614  ASSERT_not_null(ops);
615  assertStorageConditions("at start of erase", reg);
616  BitRange accessedLocation = BitRange::baseSize(reg.offset(), reg.nBits());
617 
618  // Fast case: the state does not store this register or any register that might overlap with this register
619  if (!registers_.exists(reg))
620  return; // no part of register is stored in this state
621  RegPairs &pairList = registers_[reg];
622 
623  // Look for existing registers that overlap with this register and remove them. If the overlap was only partial, then we
624  // need to eventually add the non-overlapping part back into the list.
625  RegPairs nonoverlaps; // the non-overlapping parts of overlapping registers
626  Extent need_extent(reg.offset(), reg.nBits());
627  for (RegPair &reg_val: pairList) {
628  BitRange haveLocation = reg_val.location();
629  if (BitRange intersection = accessedLocation & haveLocation) {
630  if (haveLocation.least() < intersection.least()) {
631  size_t leftSize = intersection.least() - haveLocation.least();
632  RegisterDescriptor subreg(reg.majorNumber(), reg.minorNumber(), haveLocation.least(), leftSize);
633  nonoverlaps.push_back(RegPair(subreg, ops->unsignedExtend(reg_val.value, leftSize)));
634  }
635  if (intersection.greatest() < haveLocation.greatest()) {
636  size_t rightSize = haveLocation.greatest() - intersection.greatest();
637  size_t lobit = intersection.greatest()+1 - haveLocation.least();
638  RegisterDescriptor subreg(reg.majorNumber(), reg.minorNumber(), intersection.greatest()+1, rightSize);
639  nonoverlaps.push_back(RegPair(subreg, ops->extract(reg_val.value, lobit, lobit+rightSize)));
640  }
641  reg_val.value = SValue::Ptr();
642  }
643  }
644 
645  // Remove marked pairs, then add the non-overlapping parts.
646  pairList.erase(std::remove_if(pairList.begin(), pairList.end(), has_null_value), pairList.end());
647  pairList.insert(pairList.end(), nonoverlaps.begin(), nonoverlaps.end());
648  assertStorageConditions("at end of erase", reg);
649 }

924 bool
925 RegisterStateGeneric::merge(const BaseSemantics::AddressSpace::Ptr &other_, RiscOperators*/*addrOps*/, RiscOperators *valOps) {
926  ASSERT_not_null(valOps);
927  RegisterStateGeneric::Ptr other = as<RegisterStateGeneric>(other_);
928  ASSERT_not_null(other);
929  using Interval = Sawyer::Container::Interval<size_t>;
930  using IntervalSet = Sawyer::Container::IntervalSet<Interval>;
931  bool changed = false;
932 
933  // Merge values stored in registers.
934  const RegPairs otherRegVals = other->get_stored_registers();
935  for (const RegPair &otherRegVal: otherRegVals) {
936 
937  // Part of otherRegVal that remains to be processed.
938  IntervalSet remaining;
939  remaining |= Interval::baseSize(otherRegVal.desc.offset(), otherRegVal.desc.nBits());
940 
941  // Where this register state contains a value for part of the other register, merge the values.
942  const RegPairs selfRegVals = overlappingRegisters(otherRegVal.desc); // copy since write (below) changes state
943  for (const RegPair &selfRegVal: selfRegVals) {
944  const RegisterDescriptor reg = otherRegVal.desc & selfRegVal.desc;
945  SValue::Ptr selfValue = selfRegVal.get(reg, valOps);
946  SValue::Ptr otherValue = otherRegVal.get(reg, valOps);
947  if (SValue::Ptr merged = selfValue->createOptionalMerge(otherValue, merger(), valOps->solver()).orDefault()) {
948  writeRegister(reg, merged, valOps);
949  changed = true;
950  }
951 
952  const auto processed = Interval::baseSize(reg.offset(), reg.nBits());
953  remaining -= processed;
954  }
955 
956  // Where this register state does not contain a value for part of the other register, use the other state's value
957  // without merging.
958  for (const Interval &where: remaining.intervals()) {
959  const RegisterDescriptor reg(otherRegVal.desc.majorNumber(), otherRegVal.desc.minorNumber(),
960  where.least(), where.size());
961  SValue::Ptr otherValue = otherRegVal.get(reg, valOps);
962  writeRegister(reg, otherValue, valOps);
963  changed = true;
964  }
965  }
966 
967  // Merge writer sets.
968  for (const RegisterAddressSet::Node &wmNode: other->writers_.nodes()) {
969  const BitAddressSet &otherWriters = wmNode.value();
970  BitAddressSet &thisWriters = writers_.insertMaybeDefault(wmNode.key());
971  for (const BitAddressSet::Node &otherWritten: otherWriters.nodes()) {
972  bool inserted = thisWriters.insert(otherWritten.key(), otherWritten.value());
973  if (inserted)
974  changed = true;
975  }
976  }
977 
978  // Merge property sets.
979  for (const RegisterProperties::Node &otherRegNode: other->properties_.nodes()) {
980  const BitProperties &otherBitProps = otherRegNode.value();
981  BitProperties &thisBitProps = properties_.insertMaybeDefault(otherRegNode.key());
982  for (const BitProperties::Node &otherBitNode: otherBitProps.nodes()) {
983  bool inserted = thisBitProps.insert(otherBitNode.key(), otherBitNode.value());
984  if (inserted)
985  changed = true;
986  }
987  }
988 
989  return changed;
990 }

992 void
993 RegisterStateGeneric::hash(Combinatorics::Hasher &hasher, RiscOperators* /*addrOps*/, RiscOperators *valOps) const {
994  // This register state automatically concatenates and splits register parts as needed. For instance, it might store x86 EAX as a
995  // single 32-bit value, or two 16-bit values, or four 8-bit values, or any other combination. Some parts might even be missing,
996  // such as the high 32 bits of RAX if we ony ever wrote to the low 32 bits. This complicates hashing. We could simply hash
997  // exactly the parts that are stored, but a more useful hasher would return the same value whether a register was stored as a
998  // single piece or broken into multiple smaller pieces.
999  ASSERT_not_null(valOps);
1000  ASSERT_not_null(regdict);
1001  std::vector<RegisterDescriptor> regs = regdict->getLargestRegisters();
1002  for (RegisterDescriptor reg: regs) {
1003  ExtentMap parts = stored_parts(reg);
1004  for (const auto &node: parts) {
1005  RegisterDescriptor part(reg.majorNumber(), reg.minorNumber(), node.first.first(), node.first.size());
1006  BaseSemantics::SValue::Ptr dflt = valOps->undefined_(node.first.size());
1007  BaseSemantics::SValue::Ptr value = const_cast<RegisterStateGeneric*>(this)->peekRegister(part, dflt, valOps);
1008  hasher.insert(part.majorNumber());
1009  hasher.insert(part.minorNumber());
1010  hasher.insert(part.offset());
1011  hasher.insert(part.nBits());
1012  value->hash(hasher);
1013  }
1014  }
