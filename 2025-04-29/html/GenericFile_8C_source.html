26 SgAsmGenericFile *
27 SgAsmGenericFile::parse(std::string fileName)
28 {
29  ASSERT_require(p_fd < 0); /*can call parse() only once per object*/
30 
31  set_name(fileName);
32  p_fd = open(fileName.c_str(), O_RDONLY);
33  if (p_fd<0 || fstat(p_fd, &p_sb)<0) {
34  std::string mesg = "could not open binary file \"" + StringUtility::cEscape(fileName) + "\"";
35  throw FormatError(mesg + ": " + strerror(errno));
36  }
37  size_t nbytes = p_sb.st_size;
38 
39  /* To be more portable across operating systems, read the file into memory rather than mapping it. */
40  unsigned char *mapped = new unsigned char[nbytes];
41  if (!mapped)
42  throw FormatError("could not allocate memory for binary file \"" + StringUtility::cEscape(fileName) + "\"");
43  ssize_t nread = read(p_fd, mapped, nbytes);
44  if (nread<0 || (size_t)nread!=nbytes)
45  {
46  delete [] mapped;
47  throw FormatError("could not read entire binary file \"" + StringUtility::cEscape(fileName) + "\"");
48  }
49 
50  /* Decode the memory if necessary */
51  DataConverter *dc = get_dataConverter();
52  if (dc) {
53  unsigned char *new_mapped = dc->decode(mapped, &nbytes);
54  if (new_mapped!=mapped) {
55  delete[] mapped;
56  mapped = new_mapped;
57  }
58  }
59 
60  /* Make file contents available through an STL vector without actually reading the file */
61  p_data = SgFileContentList(mapped, nbytes);
62  return this;
63 }

174 size_t
175 SgAsmGenericFile::readContent(const MemoryMap::Ptr &map, Address start_va, void *dst_buf, Address size, bool strict)
176 {
177  /* Note: This is the same algorithm as used by MemoryMap::read() except we do it here so that we have an opportunity
178  * to track the file byte references. */
179  size_t ncopied = 0;
180  while (ncopied < size) {
181  Address va = start_va + ncopied;
182  size_t nread = map->at(va).limit(size-ncopied).singleSegment().read((uint8_t*)dst_buf+ncopied).size();
183  if (0==nread) break;
184 
185  if (get_trackingReferences()) {
186  assert(map->at(va).exists());
187  const MemoryMap::Node &me = *(map->at(va).findNode());
188  if (me.value().buffer()->data()==&(get_data()[0])) {
189  /* We are tracking file reads and this segment does, indeed, point into the file. */
190  size_t file_offset = me.value().offset() + va - me.key().least();
191  markReferencedExtent(file_offset, nread);
192  }
193  }
194 
195  ncopied += nread;
196  }
197 
198  if (ncopied<size) {
199  if (strict)
200  throw MemoryMap::NotMapped("SgAsmGenericFile::read_content() no mapping", map, start_va+ncopied);
201  memset((char*)dst_buf+ncopied, 0, size-ncopied); /*zero pad result if necessary*/
202  }
203  return ncopied;
204 }

660  for (SgAsmGenericSectionPtrList::const_iterator si=sections.begin(); si!=sections.end(); ++si) {
661  SgAsmGenericSection *section = *si;
662  if (!section->isMapped() || va<section->get_mappedActualVa() ||
663  va>=section->get_mappedActualVa()+section->get_mappedSize()) {
664  // section does not contain virtual address
665  } else if (!best) {
666  best = section;
667  file_offset = section->get_offset() + (va - section->get_mappedActualVa());
668  } else if (file_offset != section->get_offset() + (va - section->get_mappedActualVa())) {
669  return nullptr; // error
670  } else if (best->get_mappedSize() > section->get_mappedSize()) {
671  best = section;
672  } else if (best->get_name()->get_string().empty() && !section->get_name()->get_string().empty()) {
673  best = section;
674  } else {
675  // prefer section defined earlier
676  }
677  }
678  return best;
679 }

715 void
716 SgAsmGenericFile::shiftExtend(SgAsmGenericSection *s, Address sa, Address sn, AddressSpace space, Elasticity elasticity)
717 {
718  ASSERT_not_null(s);
719  ASSERT_require(s->get_file()==this);
720  ASSERT_require((space & (ADDRSP_FILE|ADDRSP_MEMORY)) != 0);
721 
722  const bool debug = false;
723  static size_t ncalls=0;
724  char p[256];
725 
726  if (debug) {
727  const char *space_s="unknown";
728  if (space & ADDRSP_FILE) {
729  space_s = "file";
730  } else if (space & ADDRSP_MEMORY) {
731  space_s = "memory";
732  }
733  snprintf(p, 256, "SgAsmGenericFile::shiftExtend[%" PRIuPTR "]: ", ncalls++);
734  fprintf(stderr, "%s -- START --\n", p);
735  fprintf(stderr, "%s S = [%d] \"%s\"\n", p, s->get_id(), s->get_name()->get_string(true).c_str());
736  fprintf(stderr, "%s %s Sa=0x%08" PRIx64 " (%" PRIu64 "), Sn=0x%08" PRIx64 " (%" PRIu64 ")\n",
737  p, space_s, sa, sa, sn, sn);
738  fprintf(stderr, "%s elasticity = %s\n", p, (ELASTIC_NONE==elasticity ? "none" :
739  ELASTIC_UNREF==elasticity ? "unref" :
740  ELASTIC_HOLE==elasticity ? "unref+holes" :
741  "unknown"));
742  }
743 
744  /* No-op case */
745  if (0==sa && 0==sn) {
746  if (debug) {
747  fprintf(stderr, "%s No change necessary.\n", p);
748  fprintf(stderr, "%s -- END --\n", p);
749  }
750  return;
751  }
752 
753  bool filespace = (space & ADDRSP_FILE)!=0;
754  bool memspace = (space & ADDRSP_MEMORY)!=0;
755  Address align=1, aligned_sa, aligned_sasn;
756  SgAsmGenericSectionPtrList neighbors, villagers;
757  ExtentMap amap; /* address mappings for all extents */
758  Extent sp;
759 
760  /* Get a list of all sections that may need to be adjusted. */
761  SgAsmGenericSectionPtrList all;
762  switch (elasticity) {
763  case ELASTIC_NONE:
764  case ELASTIC_UNREF:
765  all = filespace ? get_sections() : get_mappedSections();
766  break;
767  case ELASTIC_HOLE:
768  all = filespace ? get_sections(false) : get_mappedSections();
769  break;
770  }
771  if (debug) {
772  fprintf(stderr, "%s Following sections are in 'all' set:\n", p);
773  for (size_t i=0; i<all.size(); i++) {
774  Extent ep;
775  if (filespace) {
776  ep = all[i]->get_fileExtent();
777  } else {
778  ASSERT_require(all[i]->isMapped());
779  ep = all[i]->get_mappedPreferredExtent();
780  }
781  fprintf(stderr, "%s 0x%08" PRIx64 " 0x%08" PRIx64 " 0x%08" PRIx64 " [%d] \"%s\"\n",
782  p, ep.relaxed_first(), ep.size(), ep.relaxed_first()+ep.size(), all[i]->get_id(),
783  all[i]->get_name()->get_string(true).c_str());
784  }
785  }
786 
787  for (size_t pass=0; pass<2; pass++) {
788  if (debug) {
789  fprintf(stderr, "%s -- %s --\n",
790  p, 0==pass?"FIRST PASS":"SECOND PASS (after making a larger hole)");
791  }
792 
793  /* S offset and size in file or memory address space */
794  if (filespace) {
795  sp = s->get_fileExtent();
796  } else if (!memspace || !s->isMapped()) {
797  return; /*nothing to do*/
798  } else {
799  sp = s->get_mappedPreferredExtent();
800  }
801 
802  /* Build address map */
803  for (size_t i=0; i<all.size(); i++) {
804  if (filespace) {
805  amap.insert(all[i]->get_fileExtent());
806  } else {
807  ASSERT_require(all[i]->isMapped());
808  amap.insert(all[i]->get_mappedPreferredExtent());
809  }
810  }
811  if (debug) {
812  fprintf(stderr, "%s Address map:\n", p);
813  amap.dump_extents(stderr, (std::string(p)+" ").c_str(), "amap");
814  fprintf(stderr, "%s Extent of S:\n", p);
815  fprintf(stderr, "%s start=0x%08" PRIx64 " size=0x%08" PRIx64 " end=0x%08" PRIx64 "\n",
816  p, sp.relaxed_first(), sp.size(), sp.relaxed_first()+sp.size());
817  }
818 
819  /* Neighborhood (nhs) of S is a single extent. However, if S is zero size then nhs might be empty. The neighborhood of
820  * S is S plus all sections that overlap with S and all sections that are right-contiguous with S. */
821  ExtentMap nhs_map;
822  for (ExtentMap::iterator amapi=amap.begin(); amapi!=amap.end(); ++amapi) {
823  if (amapi->first.relaxed_first() <= sp.relaxed_first()+sp.size() &&
824  amapi->first.relaxed_first()+amapi->first.size() > sp.relaxed_first())
825  nhs_map.insert(amapi->first, amapi->second);
826  }
827  if (debug) {
828  fprintf(stderr, "%s Neighborhood of S:\n", p);
829  nhs_map.dump_extents(stderr, (std::string(p)+" ").c_str(), "nhs_map");
830  }
831  Extent nhs;
832  if (nhs_map.size()>0) {
833  assert(nhs_map.nranges()==1);
834  nhs = nhs_map.begin()->first;
835  } else {
836  nhs = sp;
837  }
838 
839  /* What sections are in the neighborhood (including S), and right of the neighborhood? */
840  neighbors.clear(); /*sections in neighborhood*/
841  neighbors.push_back(s);
842  villagers.clear(); /*sections right of neighborhood*/
843  if (debug)
844  fprintf(stderr, "%s Ignoring left (L) sections:\n", p);
845  for (size_t i=0; i<all.size(); i++) {
846  SgAsmGenericSection *a = all[i];
847  if (a==s) continue; /*already pushed onto neighbors*/
848  Extent ap;
849  if (filespace) {
850  ap = a->get_fileExtent();
851  } else if (!a->isMapped()) {
852  continue;
853  } else {
854  ap = a->get_mappedPreferredExtent();
855  }
856  switch (ExtentMap::category(ap, nhs)) {
857  case 'L':
858  if (debug)
859  fprintf(stderr, "%s L 0x%08" PRIx64 " 0x%08" PRIx64 " 0x%08" PRIx64 " [%d] \"%s\"\n",
860  p, ap.relaxed_first(), ap.size(), ap.relaxed_first()+ap.size(),
861  a->get_id(), a->get_name()->get_string(true).c_str());
862  break;
863  case 'R':
864  if (ap.relaxed_first()==nhs.relaxed_first()+nhs.size() && 0==ap.size()) {
865  /* Empty sections immediately right of the neighborhood of S should actually be considered part of the
866  * neighborhood rather than right of it. */
867  neighbors.push_back(a);
868  } else if (elasticity!=ELASTIC_NONE) {
869  /* If holes are elastic then treat things right of the hole as being part of the right village; otherwise
870  * add those sections to the neighborhood of S even though they fall outside 'nhs' (it's OK because this
871  * partitioning of sections is the only thing we use 'nhs' for anyway. */
872  villagers.push_back(a);
873  } else if ('L'==ExtentMap::category(ap, sp)) {
874  /*ignore sections left of S*/
875  } else {
876  neighbors.push_back(a);
877  }
878  break;
879  default:
880  if ('L'!=ExtentMap::category(ap, sp)) /*ignore sections left of S*/
881  neighbors.push_back(a);
882  break;
883  }
884  }
885  if (debug) {
886  fprintf(stderr, "%s Neighbors:\n", p);
887  for (size_t i=0; i<neighbors.size(); i++) {
888  SgAsmGenericSection *a = neighbors[i];
889  Extent ap = filespace ? a->get_fileExtent() : a->get_mappedPreferredExtent();
890  Address align = filespace ? a->get_fileAlignment() : a->get_mappedAlignment();
891  char cat = ExtentMap::category(ap, sp);
892  fprintf(stderr, "%s %c %c0x%08" PRIx64 " 0x%08" PRIx64 " 0x%08" PRIx64,
893  p, cat, 0==ap.relaxed_first() % (align?align:1) ? ' ' : '!',
894  ap.relaxed_first(), ap.size(), ap.relaxed_first()+ap.size());
895  if (strchr("RICE", cat)) {
896  fprintf(stderr, " align=0x%08" PRIx64, align);
897  } else {
898  fputs(" ", stderr);
899  }
900  fprintf(stderr, " [%2d] \"%s\"\n", a->get_id(), a->get_name()->get_string(true).c_str());
901  }
902  if (villagers.size()>0) fprintf(stderr, "%s Villagers:\n", p);
903  for (size_t i=0; i<villagers.size(); i++) {
904  SgAsmGenericSection *a = villagers[i];
905  Extent ap = filespace ? a->get_fileExtent() : a->get_mappedPreferredExtent();
906  Address align = filespace ? a->get_fileAlignment() : a->get_mappedAlignment();
907  fprintf(stderr, "%s %c %c0x%08" PRIx64 " 0x%08" PRIx64 " 0x%08" PRIx64,
908  p, ExtentMap::category(ap, sp), /*cat should always be R*/
909  0==ap.relaxed_first() % (align?align:1) ? ' ' : '!',
910  ap.relaxed_first(), ap.size(), ap.relaxed_first()+ap.size());
911  fputs(" ", stderr);
912  fprintf(stderr, " [%2d] \"%s\"\n", a->get_id(), a->get_name()->get_string(true).c_str());
913  }
914  }
915 
916  /* Adjust Sa to satisfy all alignment constraints in neighborhood(S) for sections that will move (cats R, I, C, and E). */
917  align = 1;
918  for (size_t i=0; i<neighbors.size(); i++) {
919  SgAsmGenericSection *a = neighbors[i];
920  Extent ap = filespace ? a->get_fileExtent() : a->get_mappedPreferredExtent();
921  if (strchr("RICE", ExtentMap::category(ap, sp))) {
922  Address x = filespace ? a->get_fileAlignment() : a->get_mappedAlignment();
923 #if BOOST_VERSION < 106700
924  align = boost::math::lcm(align, x?x:1); // deprecated in boost-1.69.0
925 #else
926  align = boost::integer::lcm(align, x?x:1); // not present before boost-1.60.0
927 #endif
928  }
929  }
930  ASSERT_require(align != 0);
931  aligned_sa = (sa/align + (sa%align?1:0))*align;
932  aligned_sasn = ((sa+sn)/align + ((sa+sn)%align?1:0))*align;
933  if (debug) {
934  fprintf(stderr, "%s Alignment LCM = 0x%08" PRIx64 " (%" PRIu64 ")\n", p, align, align);
935  fprintf(stderr, "%s Aligned Sa = 0x%08" PRIx64 " (%" PRIu64 ")\n", p, aligned_sa, aligned_sa);
936  fprintf(stderr, "%s Aligned Sa+Sn = 0x%08" PRIx64 " (%" PRIu64 ")\n", p, aligned_sasn, aligned_sasn);
937  }
938 
939  /* Are there any sections to the right of neighborhood(S)? If so, find the one with the lowest start address and use
940  * that to define the size of the hole right of neighborhood(S). */
941  if (0==villagers.size()) break;
942  SgAsmGenericSection *after_hole = nullptr;
943  Extent hp(0, 0);
944  for (size_t i=0; i<villagers.size(); i++) {
945  SgAsmGenericSection *a = villagers[i];
946  Extent ap = filespace ? a->get_fileExtent() : a->get_mappedPreferredExtent();
947  if (!after_hole || ap.relaxed_first()<hp.relaxed_first()) {
948  after_hole = a;
949  hp = ap;
950  }
951  }
952  ASSERT_not_null(after_hole);
953  ASSERT_require(hp.relaxed_first() > nhs.relaxed_first()+nhs.size());
954  Address hole_size = hp.relaxed_first() - (nhs.relaxed_first()+nhs.size());
955  if (debug) {
956  fprintf(stderr, "%s hole size = 0x%08" PRIx64 " (%" PRIu64 "); need 0x%08" PRIx64 " (%" PRIu64 "); %s\n",
957  p, hole_size, hole_size, aligned_sasn, aligned_sasn,
958  hole_size>=aligned_sasn ? "large enough" : "not large enough");
959  }
960  if (hole_size >= aligned_sasn) break;
961  Address need_more = aligned_sasn - hole_size;
962 
963  /* Hole is not large enough. We need to recursively move things that are right of our neighborhood, then recompute the
964  * all-sections address map and neighborhood(S). */
965  ASSERT_require(0==pass); /*logic problem since the recursive call should have enlarged the hole enough*/
966  if (debug) {
967  fprintf(stderr, "%s Calling recursively to increase hole size by 0x%08" PRIx64 " (%" PRIu64 ") bytes\n",
968  p, need_more, need_more);
969  }
970  shiftExtend(after_hole, need_more, 0, space, elasticity);
971  if (debug) fprintf(stderr, "%s Returned from recursive call\n", p);
972  }
973 
974  /* Consider sections that are in the same neighborhood as S */
975  if (debug) fprintf(stderr, "%s -- ADJUSTING --\n", p);
976  bool resized_mem = false;
977  for (size_t i=0; i<neighbors.size(); i++) {
978  SgAsmGenericSection *a = neighbors[i];
979  Extent ap = filespace ? a->get_fileExtent() : a->get_mappedPreferredExtent();
980  switch (ExtentMap::category(ap, sp)) {
981  case 'L':
982  break;
983  case 'R':
984  if (filespace) {
985  a->set_offset(a->get_offset()+aligned_sasn);
986  } else {
987  a->set_mappedPreferredRva(a->get_mappedPreferredRva()+aligned_sasn);
988  }
989  break;
990  case 'C': /*including S itself*/
991  case 'E':
992  if (filespace) {
993  a->set_offset(a->get_offset()+aligned_sa);
994  a->set_size(a->get_size()+sn);
995  if (memspace && !resized_mem && a->isMapped()) {
996  shiftExtend(a, 0, sn, ADDRSP_MEMORY, elasticity);
997  resized_mem = true;
998  }
999  } else {
1000  a->set_mappedPreferredRva(a->get_mappedPreferredRva()+aligned_sa);
1001  a->set_mappedSize(a->get_mappedSize()+sn);
1002  }
1003  break;
1004  case 'O':
1005  if (ap.relaxed_first()==sp.relaxed_first()) {
1006  if (filespace) {
1007  a->set_offset(a->get_offset()+aligned_sa);
1008  a->set_size(a->get_size()+sn);
1009  } else {
1010  a->set_mappedPreferredRva(a->get_mappedPreferredRva()+aligned_sa);
1011  a->set_mappedSize(a->get_mappedSize()+sn);
1012  }
1013  } else {
1014  if (filespace) {
1015  a->set_size(a->get_size()+aligned_sasn);
1016  if (memspace && !resized_mem && a->isMapped()) {
1017  shiftExtend(a, 0, aligned_sasn, ADDRSP_MEMORY, elasticity);
1018  resized_mem = true;
1019  }
1020  } else {
1021  a->set_mappedSize(a->get_mappedSize()+aligned_sasn);
1022  }
1023  }
1024  break;
1025  case 'I':
1026  if (filespace) {
1027  a->set_offset(a->get_offset()+aligned_sa);
1028  } else {
1029  a->set_mappedPreferredRva(a->get_mappedPreferredRva()+aligned_sa);
1030  }
1031  break;
1032  case 'B':
1033  if (filespace) {
1034  a->set_size(a->get_size()+sn);
1035  if (memspace && !resized_mem && a->isMapped()) {
1036  shiftExtend(a, 0, sn, ADDRSP_MEMORY, elasticity);
1037  resized_mem = true;
1038  }
1039  } else {
1040  a->set_mappedSize(a->get_size()+sn);
1041  }
1042  break;
1043  default:
1044  mlog[FATAL] << "invalid extent category";
1045  exit(1);
1046  }
1047  if (debug) {
1048  const char *space_name = filespace ? "file" : "mem";
1049  Address x = filespace ? a->get_fileAlignment() : a->get_mappedAlignment();
1050  fprintf(stderr, "%s %4s-%c %c0x%08" PRIx64 " 0x%08" PRIx64 " 0x%08" PRIx64,
1051  p, space_name, ExtentMap::category(ap, sp),
1052  0==ap.relaxed_first()%(x?x:1)?' ':'!',
1053  ap.relaxed_first(), ap.size(), ap.relaxed_first()+ap.size());
1054  Extent newap = filespace ? a->get_fileExtent() : a->get_mappedPreferredExtent();
1055  fprintf(stderr, " -> %c0x%08" PRIx64 " 0x%08" PRIx64 " 0x%08" PRIx64,
1056  0==newap.relaxed_first()%(x?x:1)?' ':'!',
1057  newap.relaxed_first(), newap.size(), newap.relaxed_first()+newap.size());
1058  fprintf(stderr, " [%2d] \"%s\"\n", a->get_id(), a->get_name()->get_string(true).c_str());
1059  }
1060  }
1061  if (debug) fprintf(stderr, "%s -- END --\n", p);
1062 }
