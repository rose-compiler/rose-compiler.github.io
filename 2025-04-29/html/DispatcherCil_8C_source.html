246 DispatcherCil::updateFpsrExcOvfl(const SValuePtr &value, SgAsmType *valueType, SgAsmType *rounding, SgAsmType *dstType) {
247  ASSERT_not_null(value);
248  ASSERT_require(value->nBits() == valueType->get_nBits());
249  ASSERT_not_null(rounding);
250  ASSERT_not_null(dstType);
251 
252  SgAsmFloatType *valueFpType = isSgAsmFloatType(valueType);
253  SgAsmFloatType *roundingFpType = isSgAsmFloatType(rounding);
254  SgAsmFloatType *dstFpType = isSgAsmFloatType(dstType);
255 
256  if (!valueFpType || !roundingFpType || !dstFpType) {
257  operators()->writeRegister(REG_EXC_OVFL, operators()->boolean_(false));
258  return;
259  }
260 
261  size_t nBits = std::max(valueFpType->exponentBits().size(), roundingFpType->exponentBits().size());
262 
263  SValuePtr valueBias = operators()->number_(nBits, valueFpType->exponentBias());
264  SValuePtr valueExp = operators()->extract(value, valueFpType->exponentBits().least(), valueFpType->exponentBits().greatest()+1);
265  valueExp = operators()->subtract(operators()->unsignedExtend(valueExp, nBits), valueBias);
266 
267  SValuePtr roundingExp =
268  operators()->number_(nBits,
269  IntegerOps::genMask<uint64_t>(roundingFpType->exponentBits().size()) - roundingFpType->exponentBias());
270 
271  SValuePtr isOverflow = operators()->isUnsignedGreaterThanOrEqual(valueExp, roundingExp);
272  operators()->writeRegister(REG_EXC_OVFL, isOverflow);
273 }
