24  {
25 
26  ASSERT_not_null(type);

28  switch (type->variantT())

33  case V_SgTypeWchar: unparseTypeWchar( isSgTypeWchar(type), info); break;
34  case V_SgTypeShort: unparseTypeShort( isSgTypeShort(type), info); break;
35  case V_SgTypeInt: unparseTypeInt( isSgTypeInt(type), info); break;
36  case V_SgTypeLong: unparseTypeLong( isSgTypeLong(type), info); break;
37  case V_SgTypeFloat: unparseTypeFloat( isSgTypeFloat(type), info); break;
38  case V_SgTypeDouble: unparseTypeDouble( isSgTypeDouble(type), info); break;
39  case V_SgTypeBool: unparseTypeBool( isSgTypeBool(type), info); break;
40 
41  case V_SgArrayType: unparseArrayType( isSgArrayType(type), info); break;
42  case V_SgTypedefType: unparseTypedefType( isSgTypedefType(type), info); break;
43  case V_SgClassType: unparseClassType( isSgClassType(type), info); break;
44  case V_SgEnumType: unparseEnumType( isSgEnumType(type), info); break;
45  case V_SgModifierType: unparseModifierType( isSgModifierType(type), info); break;
46 
47  // DQ (9/5/2011): Added support for Java generics.
48  case V_SgJavaQualifiedType: unparseJavaQualifiedType(isSgJavaQualifiedType(type), info); break;
49  case V_SgJavaParameterType: unparseClassType(isSgJavaParameterType(type), info); break;
50  case V_SgJavaParameterizedType: unparseJavaParameterizedType(isSgJavaParameterizedType(type), info); break;
51  case V_SgJavaWildcardType: unparseJavaWildcardType(isSgJavaWildcardType(type), info); break;
52  case V_SgJavaUnionType: unparseJavaUnionType(isSgJavaUnionType(type), info); break;
53 
54  default:
55  cout << "Unparse_Java::unparseType(" << type->class_name() << "*,info) is unimplemented." << endl;
56  ROSE_ABORT();
57  }
