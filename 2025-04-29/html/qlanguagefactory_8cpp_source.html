129 void QLanguageFactory::setLanguage(QEditor *e, const QString& file)

131  QString lang;
132  QFileInfo inf(file);

134  cext = inf.completeSuffix();
135 
136  //qDebug("suff:%s; compSuff:%s", qPrintable(ext), qPrintable(cext));

138  QLanguageDefinition *oldLang = e->languageDefinition();
139 
140  if ( file.count() )
141  {
142  QList<Qt::CaseSensitivity> lcs;
143  lcs << Qt::CaseSensitive << Qt::CaseInsensitive;
144 
145  foreach ( Qt::CaseSensitivity cs, lcs )
146  {
147  int n = 0, idx = -1;
148  QStringList ext_langs, cext_langs, fcext_langs;
149 
150  foreach ( QString lang, m_languages )

152  const QStringList& exts = m_data[lang].extensions;
153 
154  //qDebug("%s in (%s) ?", qPrintable(ext), qPrintable(exts.join(" ")));
155 
156  foreach ( QString x, exts )
157  {
158  if ( !x.compare(ext, cs) )
159  ext_langs << lang;
160 
161  if ( !x.compare(cext, cs) )
162  cext_langs << lang;
163 
164  if ( !x.compare(file, cs) )
165  fcext_langs << lang;
166  }
167 
168  if ( !lang.compare(file, cs) )
169  idx = n;
170 
171  ++n;
172  }
173 
174  if ( cext_langs.count() )
175  {
176  // TODO : use MIME types to resolve ambiguity
177  lang = cext_langs.first();
178  } else if ( ext_langs.count() ) {
179  // TODO : use MIME types to resolve ambiguity
180  lang = ext_langs.first();
181  } else if ( fcext_langs.count() ) {
182  // TODO : use MIME types to resolve ambiguity
183  lang = fcext_langs.first();
184  } else if ( idx != -1 ) {
185  lang = m_languages.at(idx);
186  }
187 
188  if ( lang.count() )
189  break;
190  }
191  }
192 
193  if ( lang.isEmpty() )
194  {
195  //qDebug("no lang match for %s", qPrintable(file));
196  e->setLanguageDefinition(0);
197  e->setCompletionEngine(0);
198  e->document()->setFormatScheme(m_defaultFormatScheme);
199 
200  if ( oldLang )
201  e->highlight();
202 
203  } else {
204  //qDebug("lang match for %s : %s", qPrintable(file), qPrintable(lang));
205  const LangData& data = m_data[lang];
206 
207  e->setLanguageDefinition(data.d);
208  e->setCompletionEngine(data.e ? data.e->clone() : 0);
209  e->document()->setFormatScheme(data.s ? data.s : m_defaultFormatScheme);
210 
211  if ( oldLang != data.d )
212  e->highlight();
213  }
