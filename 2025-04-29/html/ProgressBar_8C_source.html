73 SAWYER_EXPORT std::string
74 ProgressBarImpl::makeBar(double ratio, bool isBackward) {
75  std::string prefix, bar, suffix;
76  if (!prefix_.empty())
77  prefix = prefix_ + ": ";
78  if (shouldSpin_) {
79  int centerIdx = (int)(nUpdates_ % (2*width_));
80  if (centerIdx >= (int)width_)
81  centerIdx = (int)(2*width_ - centerIdx - 1);
82  assert(centerIdx>=0 && (size_t)centerIdx < width_);
83  int indicatorWidth = boost::numeric::converter<int, double>::convert(ceil(0.3 * width_));
84  int indicatorIdx = centerIdx - indicatorWidth/2;
85  bar = std::string(width_, nonBarChar_);
86  for (int i=std::max(indicatorIdx, 0); i<std::min(indicatorIdx+indicatorWidth, (int)width_); ++i)
87  bar[i] = barChar_;
88  } else {
89  if (showPercent_) {
90  char buf[16];
91 #include <Sawyer/WarningsOff.h>
92  snprintf(buf, sizeof(buf), "%3.0f%% ", 100.0*ratio);
93 #include <Sawyer/WarningsRestore.h>
94  prefix += buf;
95  }
96  // Microsoft doesn't define round(double) in <cmath>
97  size_t barLen = boost::numeric::converter<size_t, double>::convert(floor(ratio * width_ + 0.5));
98  if (isBackward) {
99  bar += std::string(width_-barLen, nonBarChar_) + std::string(barLen, barChar_);
100  } else {
101  bar += std::string(barLen, barChar_) + std::string(width_-barLen, nonBarChar_);
102  }
103  }
104  if (!suffix_.empty())
105  suffix = " " + suffix_;
106 
107  return prefix + leftEnd_ + bar + rightEnd_ + suffix;
108 }

110 SAWYER_EXPORT void
111 ProgressBarImpl::updateTextMesg(double ratio) {
112  if (shouldSpin_) {
113  if (textMesg_.isEmpty() || textMesg_.text().size() > width_) {
114  textMesg_ = Message::Mesg();
115  if (!prefix_.empty())
116  textMesg_.insert(prefix_ + ": ");
117  }
118  if (suffix_.empty()) {
119  textMesg_.insert('.');
120  } else {
121  textMesg_.insert(" " + suffix_);
122  }
123  } else {
124  int pct = boost::numeric::converter<int, double>::convert(floor(ratio * 10.0 + 0.5) * 10);
125  if (!oldPercent_ || pct != *oldPercent_) {
126  bool needElipsis = !textMesg_.isEmpty();
127  if (textMesg_.isEmpty() || textMesg_.text().size() > width_) {
128  textMesg_ = Message::Mesg();
129  if (!prefix_.empty())
130  textMesg_.insert(prefix_ + ": ");
131  needElipsis = false;
132  }
133  oldPercent_ = pct;
134  if (needElipsis)
135  textMesg_.insert("...");
136  std::ostringstream ss;
137  ss <<pct <<"%";
138  if (suffix_.empty()) {
139  textMesg_.insert(ss.str());
140  } else {
141  textMesg_.insert(suffix_ + "(" + ss.str() + ")");
142  }
143  }
144  }

147 SAWYER_EXPORT void
148 ProgressBarImpl::update(double ratio, bool isBackward) {
149  if (*stream_ && width_>0) {
150  Message::BakedDestinations baked;
151  stream_->destination()->bakeDestinations(stream_->properties(), baked);
152  if (!baked.empty()) {
153  for (size_t i=0; i<baked.size(); ++i) {
154  if (baked[i].second.useColor) {
155  Message::MesgProps props = overridesAnsi_.merge(baked[i].second);
156  Message::Mesg mesg(props);
157  mesg.insert(makeBar(ratio, isBackward));
158  baked[i].first->post(mesg, props);
159  } else {
160  updateTextMesg(ratio);
161  baked[i].first->post(textMesg_, baked[i].second);
162  }
163  }
164  ++nUpdates_;
165  }
166  }
167 }
