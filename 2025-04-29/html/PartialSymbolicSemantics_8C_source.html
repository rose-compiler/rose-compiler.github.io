161 void
162 SValue::print(std::ostream &stream, BaseSemantics::Formatter &formatter_) const
163 {
164  FormatRestorer restorer(stream); // restore format flags when we leave this scope
165  uint64_t sign_bit = (uint64_t)1 << (nBits()-1); /* e.g., 80000000 */
166  uint64_t val_mask = sign_bit - 1; /* e.g., 7fffffff */
167  /*magnitude of negative value*/
168  uint64_t negative = nBits()>1 && (offset & sign_bit) ? (~offset & val_mask) + 1 : 0;
169  Formatter *formatter = dynamic_cast<Formatter*>(&formatter_);
170 
171  if (name!=0) {
172  /* This is a named value rather than a constant. */
173  uint64_t renamed = formatter ? formatter->rename(name) : name;
174  const char *sign = negate ? "-" : "";
175  stream <<sign <<"v" <<std::dec <<renamed;
176  if (negative) {
177  stream <<"-0x" <<std::hex <<negative;
178  } else if (offset) {
179  stream <<"+0x" <<std::hex <<offset;
180  }
181  } else {
182  /* This is a constant */
183  ASSERT_require(!negate);
184  stream <<"0x" <<std::hex <<offset;
185  if (negative)
186  stream <<" (-0x" <<std::hex <<negative <<")";
187  }
188 
189  stream <<"[" <<std::dec <<nBits() <<"]";
190 }

566 BaseSemantics::SValue::Ptr
567 RiscOperators::add(const BaseSemantics::SValue::Ptr &a_, const BaseSemantics::SValue::Ptr &b_)
568 {
569  SValue::Ptr a = SValue::promote(a_);
570  SValue::Ptr b = SValue::promote(b_);
571  ASSERT_require(a->nBits()==b->nBits());
572  if (a->name==b->name && (!a->name || a->negate!=b->negate)) {
573  /* [V1+x] + [-V1+y] = [x+y] or
574  * [x] + [y] = [x+y] */
575  return number_(a->nBits(), a->offset + b->offset);
576  } else if (!a->name || !b->name) {
577  /* [V1+x] + [y] = [V1+x+y] or
578  * [x] + [V2+y] = [V2+x+y] or
579  * [-V1+x] + [y] = [-V1+x+y] or
580  * [x] + [-V2+y] = [-V2+x+y] */
581  return a->create(a->nBits(), a->name+b->name, a->offset+b->offset, a->negate || b->negate);
582  } else {
583  return undefined_(a->nBits());
584  }

588 RiscOperators::addWithCarries(const BaseSemantics::SValue::Ptr &a_, const BaseSemantics::SValue::Ptr &b_,

590 {

797 BaseSemantics::SValue::Ptr
798 RiscOperators::readOrPeekMemory(RegisterDescriptor segreg,
799  const BaseSemantics::SValue::Ptr &address,
800  const BaseSemantics::SValue::Ptr &dflt_,
801  bool allowSideEffects) {
802  BaseSemantics::SValue::Ptr dflt = dflt_;
803  size_t nbits = dflt->nBits();
804  ASSERT_require2(nbits % 8 == 0, "read from memory must be in byte units");
805 
806  // Offset the address by the value of the segment register.
807  BaseSemantics::SValue::Ptr adjustedVa;
808  if (segreg.isEmpty()) {
809  adjustedVa = address;
810  } else {
811  BaseSemantics::SValue::Ptr segregValue;
812  if (allowSideEffects) {
813  segregValue = readRegister(segreg, undefined_(segreg.nBits()));
814  } else {
815  segregValue = peekRegister(segreg, undefined_(segreg.nBits()));
816  }
817  adjustedVa = add(address, signExtend(segregValue, address->nBits()));
818  }
819 
820  // Use the initial memory state if there is one.
821  if (initialState()) {
822  if (allowSideEffects) {
823  dflt = initialState()->readMemory(adjustedVa, dflt, this, this);
824  } else {
825  dflt = initialState()->peekMemory(adjustedVa, dflt, this, this);
826  }
827  }
828 
829  // Use the concrete MemoryMap if there is one. Only those areas of the map that are readable and not writable are used.
830  if (map && adjustedVa->isConcrete()) {
831  size_t nbytes = nbits/8;
832  uint8_t *buf = new uint8_t[nbytes];
833  size_t nread = map->require(MemoryMap::READABLE).prohibit(MemoryMap::WRITABLE)
834  .at(adjustedVa->toUnsigned().get()).limit(nbytes).read(buf).size();
835  if (nread == nbytes) {
836  if (nbytes > 1 && map->byteOrder() == ByteOrder::ORDER_UNSPECIFIED)
837  throw BaseSemantics::Exception("multi-byte read with memory having unspecified byte order", currentInstruction());
838  ByteOrder::convert(buf, nbytes, map->byteOrder(), ByteOrder::ORDER_LSB);
839  uint64_t dflt_val = 0;
840  for (size_t i=0; i<nbytes; ++i)
841  dflt_val |= IntegerOps::shiftLeft2<uint64_t>(buf[i], 8*i);
842  dflt = number_(nbits, dflt_val);
843  }
844  delete [] buf;
845  }
846 
847  // PartialSymbolicSemantics assumes that its memory state is capable of storing multi-byte values.
848  SValue::Ptr retval = SValue::promote(currentState()->readMemory(adjustedVa, dflt, this, this));
849  return retval;
850 }
