49 SourceLocation::parse(const std::string &s) {
50  // Parse the line and column
51  size_t line = 0;
52  Sawyer::Optional<size_t> column;
53  std::string filePart;
54  boost::regex lineCol("(.*):([0-9]+|0x[0-9a-fA-F]+|0b[01]+):([0-9]+|0x[0-9a-fA-F]+|0b[01]+)$");
55  boost::regex lineOnly("(.*):([0-9]+|0x[0-9a-fA-F]+|0b[01]+)$");
56  boost::smatch matched;
57  if (boost::regex_search(s, matched, lineCol)) {
58  filePart = matched[1];
59  line = boost::lexical_cast<size_t>(matched[2]);
60  column = boost::lexical_cast<size_t>(matched[3]);
61  } else if (boost::regex_search(s, matched, lineOnly)) {
62  filePart = matched[1];
63  line = boost::lexical_cast<size_t>(matched[2]);
64  } else {
65  filePart = s;
66  }
67 
68  // Parse the file name
69  std::string fileName;
70  if (filePart.size() >= 2 && filePart[0] == '"' && filePart[filePart.size()-1] == '"') {
71  fileName = StringUtility::cUnescape(filePart.substr(1, filePart.size()-2));
72  } else {
73  fileName = filePart;
74  }

118 SourceLocation::compare(const SourceLocation &other) const {
