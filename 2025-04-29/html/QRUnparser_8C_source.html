72 void QRAstCoords::get_target_boxes_aux(QRCoordBox *box,vector<QRCoordBox *> *target_boxes, int line, int col) {
73  target_boxes->push_back(box);
74  const list<QRCoordBox *> *enclosed_boxes = box->children();
75  if (enclosed_boxes->size() == 0) return;
76  bool found_target_box = false;
77  list<QRCoordBox *>::const_iterator iter = enclosed_boxes->begin();
78  while (!found_target_box && iter != enclosed_boxes->end()) {
79  QRCoordBox *new_box = *iter;
80  if (new_box->isTarget(line, col)) {
81  get_target_boxes_aux(new_box, target_boxes, line, col);
82  found_target_box = true;
83  } else {
84  iter++;
85  }
86  }
