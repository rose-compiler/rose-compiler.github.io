210 
211  if (bblock_->nInstructions() == 0) {
212  auto tag = ErrorTag::instance(0, "empty basic block", "basic block has no instructions", nullptr, bblock_->address());
213  tag->importance(WARN);

215 
216  } else {

219  SAWYER_MESG(mlog[DEBUG]) <<" executing " <<insn->toString() <<"\n";
220

222  if (i+1 < bblock_->nInstructions())

224  tags.push_back(tag);
225  }
226  if (!ops->currentState())
227  break;
228 
229  // Show states
230  if (i + 1 < bblock_->nInstructions()) {
231  if (settings->showIntermediateStates)
232  SAWYER_MESG(mlog[DEBUG]) <<" intermediate state\n" <<(*ops->currentState() + fmt);
233  } else {
234  if (settings->showFinalStates)
235  SAWYER_MESG(mlog[DEBUG]) <<" final state\n" <<(*ops->currentState() + fmt);
236  }
237 
238  // In order for this to be a valid basic block, the instruction pointer must have a concrete value that points
239  // to the next instruction.
240  if (i + 1 < bblock_->nInstructions()) {
