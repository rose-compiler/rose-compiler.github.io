224 SgAsmElfSegmentTable *
225 SgAsmElfSegmentTable::parse()
226 {
227  SgAsmGenericSection::parse();
228 
229  SgAsmElfFileHeader *fhdr = dynamic_cast<SgAsmElfFileHeader*>(get_header());
230  ROSE_ASSERT(fhdr!=NULL);
231  Rose::BinaryAnalysis::ByteOrder::Endianness sex = fhdr->get_sex();
232 
233  size_t ent_size, struct_size, opt_size, nentries;
234  calculateSizes(&ent_size, &struct_size, &opt_size, &nentries);
235  ROSE_ASSERT(opt_size==fhdr->get_phextrasz() && nentries==fhdr->get_e_phnum());
236 
237  /* If the current size is very small (0 or 1 byte) then we're coming straight from the constructor and the parsing should
238  * also extend this section to hold all the entries. Otherwise the caller must have assigned a specific size for a good
239  * reason and we should leave that alone, reading zeros if the entries extend beyond the defined size. */
240  if (get_size()<=1 && get_size()<nentries*ent_size)
241  extend(nentries*ent_size - get_size());
242 
243  Rose::BinaryAnalysis::Address offset=0; /* w.r.t. the beginning of this section */
244  for (size_t i=0; i<nentries; i++, offset+=ent_size) {
245  /* Read/decode the segment header */
246  SgAsmElfSegmentTableEntry *shdr = NULL;
247  if (4==fhdr->get_wordSize()) {
248  SgAsmElfSegmentTableEntry::Elf32SegmentTableEntry_disk disk;
249  readContentLocal(offset, &disk, struct_size);
250  shdr = new SgAsmElfSegmentTableEntry(sex, &disk);
251  } else {
252  SgAsmElfSegmentTableEntry::Elf64SegmentTableEntry_disk disk;
253  readContentLocal(offset, &disk, struct_size);
254  shdr = new SgAsmElfSegmentTableEntry(sex, &disk);
255  }
256  shdr->set_index(i);
257  if (opt_size>0)
258  shdr->get_extra() = readContentLocalUcl(offset+struct_size, opt_size);
259 
260  /* Null segments are just unused slots in the table; no real section to create */
261  if (SgAsmElfSegmentTableEntry::PT_NULL == shdr->get_type())
262  continue;
263 
264  /* Create SgAsmElfSection objects for each ELF Segment. However, if the ELF Segment Table describes a segment
265  * that's the same offset and size as a section from the Elf Section Table (and the memory mappings are
266  * consistent) then use the preexisting section instead of creating a new one. */
267  SgAsmElfSection *s = NULL;
268  SgAsmGenericSectionPtrList possible = fhdr->get_file()->get_sectionsByOffset(shdr->get_offset(), shdr->get_filesz());
269  for (size_t j=0; !s && j<possible.size(); j++) {
270  if (possible[j]->get_offset()!=shdr->get_offset() || possible[j]->get_size()!=shdr->get_filesz())
271  continue; /*different file extent*/
272  if (possible[j]->isMapped()) {
273  if (possible[j]->get_mappedPreferredRva()!=shdr->get_vaddr() ||
274  possible[j]->get_mappedSize()!=shdr->get_memsz())
275  continue; /*different mapped address or size*/
276  unsigned section_perms = (possible[j]->get_mappedReadPermission() ? 0x01 : 0x00) |
277  (possible[j]->get_mappedWritePermission() ? 0x02 : 0x00) |
278  (possible[j]->get_mappedExecutePermission() ? 0x04 : 0x00);
279  unsigned segment_perms = (shdr->get_flags() & SgAsmElfSegmentTableEntry::PF_RPERM ? 0x01 : 0x00) |
280  (shdr->get_flags() & SgAsmElfSegmentTableEntry::PF_WPERM ? 0x02 : 0x00) |
281  (shdr->get_flags() & SgAsmElfSegmentTableEntry::PF_XPERM ? 0x04 : 0x00);
282  if (section_perms != segment_perms)
283  continue; /*different mapped permissions*/
284  }
285 
286  /* Found a match. Set memory mapping params only. */
287  s = dynamic_cast<SgAsmElfSection*>(possible[j]);
288  if (!s) continue; /*potential match was not from the ELF Section or Segment table*/
289  if (s->get_segmentEntry()) continue; /*potential match is assigned to some other segment table entry*/
290  s->initFromSegmentTable(shdr, true); /*true=>set memory mapping params only*/
291  }
292 
293  /* Create a new segment if no matching section was found. */
294  if (!s) {
295  if (SgAsmElfSegmentTableEntry::PT_NOTE == shdr->get_type()) {
296  s = new SgAsmElfNoteSection(fhdr);
297  } else {
298  s = new SgAsmElfSection(fhdr);
299  }
300  s->initFromSegmentTable(shdr);
301  s->parse();
302  }
303  }
304  return this;
305 }

343 Rose::BinaryAnalysis::Address
344 SgAsmElfSegmentTable::calculateSizes(size_t *entsize, size_t *required, size_t *optional, size_t *entcount) const
345 {
346  SgAsmElfFileHeader *fhdr = dynamic_cast<SgAsmElfFileHeader*>(get_header());
347  ROSE_ASSERT(fhdr!=NULL);
348 
349  size_t struct_size = 0;
350  size_t extra_size = fhdr->get_phextrasz();
351  size_t entry_size = 0;
352  size_t nentries = 0;
353 
354  /* Size of required part of each entry */
355  if (4==fhdr->get_wordSize()) {
356  struct_size = sizeof(SgAsmElfSegmentTableEntry::Elf32SegmentTableEntry_disk);
357  } else if (8==fhdr->get_wordSize()) {
358  struct_size = sizeof(SgAsmElfSegmentTableEntry::Elf64SegmentTableEntry_disk);
359  } else {
360  throw FormatError("bad ELF word size");
361  }
362 
363  /* Entire entry should be at least large enough for the required part. */
364  entry_size = struct_size;
365 
366  /* Size of optional parts. If we've parsed the table then use the largest optional part, otherwise assume the entry from
367  * the ELF File Header is correct. */
368  SgAsmGenericSectionPtrList sections = fhdr->get_sections()->get_sections();
369  for (size_t i=0; i<sections.size(); i++) {
370  SgAsmElfSection *elfsec = dynamic_cast<SgAsmElfSection*>(sections[i]);
371  if (elfsec && elfsec->get_segmentEntry()) {
372  nentries++;
373  extra_size = std::max(extra_size, elfsec->get_segmentEntry()->get_extra().size());
374  }
375  }
376 
377  /* Total number of entries. Either we haven't parsed the segment table yet (nor created the segments it defines) or we
378  * have. In the former case we use the setting from the ELF File Header, otherwise we just count the number of segments
379  * that have associated segment table entry pointers. */
380  if (0==nentries)
381  nentries = fhdr->get_e_phnum();
382 
383  /* Return values */
384  if (entsize)
385  *entsize = entry_size;
386  if (required)
387  *required = struct_size;
388  if (optional)
389  *optional = extra_size;
390  if (entcount)
391  *entcount = nentries;
392  return entry_size * nentries;
393 }

395 bool
396 SgAsmElfSegmentTable::reallocate()
397 {
398  bool reallocated = false;
399 
400  /* Resize based on word size from ELF File Header */
401  size_t opt_size, nentries;
402  Rose::BinaryAnalysis::Address need = calculateSizes(NULL, NULL, &opt_size, &nentries);
403  if (need < get_size()) {
404  if (isMapped()) {
405  ROSE_ASSERT(get_mappedSize()==get_size());
406  set_mappedSize(need);
407  }
408  set_size(need);
409  reallocated = true;
410  } else if (need > get_size()) {
411  get_file()->shiftExtend(this, 0, need-get_size(), SgAsmGenericFile::ADDRSP_ALL, SgAsmGenericFile::ELASTIC_HOLE);
412  reallocated = true;
413  }
414 
415  /* Update data members in the ELF File Header. No need to return true for these changes. */
416  SgAsmElfFileHeader *fhdr = dynamic_cast<SgAsmElfFileHeader*>(get_header());
417  fhdr->set_phextrasz(opt_size);
418  fhdr->set_e_phnum(nentries);
419 
420  return reallocated;
421 }

423 void
424 SgAsmElfSegmentTable::unparse(std::ostream &f) const
425 {
426  SgAsmElfFileHeader *fhdr = dynamic_cast<SgAsmElfFileHeader*>(get_header());
427  ROSE_ASSERT(fhdr!=NULL);
428  Rose::BinaryAnalysis::ByteOrder::Endianness sex = fhdr->get_sex();
429  SgAsmGenericSectionPtrList sections = fhdr->get_segmentTableSections();
430 
431  /* Write the segments first */
432  for (size_t i=0; i<sections.size(); i++)
433  sections[i]->unparse(f);
434  unparseHoles(f);
435 
436  /* Calculate sizes. The ELF File Header should have been updated in reallocate() prior to unparsing. */
437  size_t ent_size, struct_size, opt_size, nentries;
438  calculateSizes(&ent_size, &struct_size, &opt_size, &nentries);
439  ROSE_ASSERT(fhdr->get_phextrasz() == opt_size);
440  ROSE_ASSERT(fhdr->get_e_phnum() >= nentries);
441  ROSE_ASSERT(nentries == sections.size());
442 
443  // What entries were written by the following loop
444  std::vector<bool> wasWritten(fhdr->get_e_phnum(), false);
445 
446  /* Write the segment table entries */
447  for (size_t i=0; i < sections.size(); ++i) {
448  SgAsmElfSection *section = dynamic_cast<SgAsmElfSection*>(sections[i]);
449  ROSE_ASSERT(section!=NULL);
450  SgAsmElfSegmentTableEntry *shdr = section->get_segmentEntry();
451  ROSE_ASSERT(shdr!=NULL);
452  ROSE_ASSERT(shdr->get_offset()==section->get_offset()); /*segment table entry should have been updated in reallocate()*/
453 
454  int id = shdr->get_index();
455  ROSE_ASSERT(id >= 0 && (size_t)id < wasWritten.size());
456 
457  SgAsmElfSegmentTableEntry::Elf32SegmentTableEntry_disk disk32;
458  SgAsmElfSegmentTableEntry::Elf64SegmentTableEntry_disk disk64;
459  void *disk = NULL;
460 
461  if (4==fhdr->get_wordSize()) {
462  disk = shdr->encode(sex, &disk32);
463  } else if (8==fhdr->get_wordSize()) {
464  disk = shdr->encode(sex, &disk64);
465  } else {
466  ROSE_ASSERT(!"invalid word size");
467  }
468 
469  // Write the disk struct and extra data to the table slot
470  Rose::BinaryAnalysis::Address spos = write(f, id*ent_size, struct_size, disk);
471  if (shdr->get_extra().size() > 0)
472  write(f, spos, shdr->get_extra());
473  wasWritten[id] = true;
474  }
475 
476  // Write zeroed entries to the unused slots of the segment table
477  for (size_t i = 0; i < fhdr->get_e_phnum(); ++i) {
478  if (!wasWritten[i]) {
479  std::vector<uint8_t> empty(ent_size, 0);
480  write(f, i*ent_size, ent_size, empty.data());
481  }
482  }
483 }
