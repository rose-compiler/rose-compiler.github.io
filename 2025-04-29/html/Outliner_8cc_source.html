77 string
78 Outliner::generateFuncName (const SgStatement* stmt)
79 {
80  // Generate a prefix.
81  stringstream s;
82  if (use_dlopen)
83  {
84  // We may outline basic blocks from header files. A .cpp file may include multiple header files.
85  // In order to have consistent function name, we now use per file name (.cpp or .h) counters.
86  string file_name= stmt->get_startOfConstruct()->get_raw_filename();
87  s<< g_outlined_func_names2.next(file_name);
88  }
89  else
90  s << g_outlined_func_names.next ();
91 
92  // Use the statement's raw filename to come up with a file-specific
93  // tag.
94  const Sg_File_Info* info = stmt->get_startOfConstruct ();
95  ROSE_ASSERT (info);
96  if (use_dlopen)
97  {
98  const string file_name = info->get_raw_filename ();
99  const string file_name2 = StringUtility::stripPathFromFileName(file_name);
100  // We now keep suffix to differentiate header and source files
101  //string base_name = StringUtility::stripFileSuffixFromFileName(file_name2);
102  string base_name = file_name2;
103  // base name may contain '-', replace it with '_' to get legal identifier
104  for (size_t i=0; i<base_name.size(); i++)
105  {
106  //cout<<"file base name:"<<base_name[i]<<endl;
107  if (base_name[i]=='-' || base_name[i]=='.')
108  base_name[i]='_';
109  }
110 
111  s<<"_"<<base_name<<"_"<<info->get_line();
112  }
113  else
114  s << hashStringToULong (info->get_raw_filename ()) << "__";
115 
116  return s.str ();
117 }
