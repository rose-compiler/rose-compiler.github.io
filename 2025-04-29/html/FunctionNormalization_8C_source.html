10 /*
11  * Normalization of function calls - no statement will have more than one function call
12  * This is to be done by inserting new variables, while preserving the semantics of the code.
13  * Author: Radu Popovici
14  * Date: Sept-08-2005
15  */
16 
17 /*
18 Example:
19 Assume we have the functions:
20  bool g ( int );
21  int f1 ( bool );
22  int &f2 ( bool );
23 and the code:
24  for ( ; f1( g( 1 ) ) < 10; f2( g( 2 ) ) );
25 
26 The code will become ( note the three different types of statements generated ):
27  bool temp1 = g( 1 ); // --- type1
28  int temp2 = f1( temp1 ); // --- type1
29  bool temp3; // --- type2
30  int *temp4; // --- type2
31  for ( ; temp2 < 10; *temp4 )
32  {
33  temp3 = g( 2 ); // --- type3
34  temp4 = &f2( temp3 ); // --- type3
35  temp1 = g( 1 ); // --- type3
36  temp2 = f1( temp1 ); // --- type3
37  }
38 
39 We generated 3 types of statements: initialized variable declaration, non-initialized variable declaration and assignment.
40 For each statement we process we will generate a list structures, one structure for each function call of the statement.
41 The structure is called Declarations, and contains the three types of statements for the desired function call
42 Depending on the enclosing scope we will insert the appropriate type(s) of statements to replace each function call.
43 
44 Note: in the case of functions returning references, we need to assign those to pointers, in the case when we cannot
45 initialize the substituting variable at the point of its declaration (e.g. for the increment of for-loops).
46 
47 The Algorithm:
48 1. get statement
49 2. generate list of (direct) function calls in the statement (not considering statements associated with the current one)
50 3. for each function call, generate a structure containing the 3 statements, one of each type (some may be null in some cases)
51 4. insert the newly created statements where appropriate
52 5. replace in the original expression the function calls with their corresponding variables (at the highest level)
53 */

433 /*
434 Given a node in the AST, it replaces all function calls with the corresponding expression from the fct2Var mapping.
435 Since we traverse in order of function evaluation, all nodes that are lower in the subtree are supposed to
436 have been mapped, since they have already appeared in the list. Also, we only replace the shallowest level of
437 function calls in the surrent subtree, since we assume that replaceFunctionCallsInExpression has already been called
438 for each of the subtrees rooted at the shallowest function calls lower than the current node:
439 
440 E.g.: f(g(h(i))) + 5
441 has the AST:
442  +
443  / \
444  fc1 5
445  / \
446  f fc2
447  / \
448  g fc3
449  / \
450  h i
451 where fc? represents an SgFunctionCallExpression. Order of function evaluation is h(i), g(_), f(_).
452 
453 Calling 'replaceFunctionCallsInExpression' on '+' will generate
454  temp_var + 5
455 but calling the same function on 'fc2' will generate
456  f(temp_var) + 5
457 */
458 void
459 FunctionCallNormalization::replaceFunctionCallsInExpression( SgNode *root,
460  map<SgFunctionCallExp *, SgExpression *> fct2Var )
461  {
462  if ( !root )
463  return;
464 
465  if ( NodeQuery::querySubTree( root, V_SgFunctionCallExp ).size() > 0 )
466  {
467  list<SgNode *> toVisit;
468  toVisit.push_back( root );
469 
470  while ( !toVisit.empty() )
471  {
472  SgNode *crt = toVisit.front();
473 
474  // will visit every node above an function call exp, but not below
475  // also, we will replace function call expressions found
476  if ( !isSgFunctionCallExp( crt ) )
477  {
478  vector<SgNode *> succ = ( crt )->get_traversalSuccessorContainer();
479  for ( vector<SgNode *>::iterator succIt = succ.begin(); succIt != succ.end(); succIt++ )
480  if ( isSgNode ( *succIt ) )
481  toVisit.push_back( *succIt );
482  }
483  else
484  {
485  SgFunctionCallExp *call = isSgFunctionCallExp( crt );
486  ROSE_ASSERT ( call );
487 
488  map<SgFunctionCallExp *, SgExpression *>::iterator mapIter;
489  mapIter = fct2Var.find( call );
490  if ( mapIter == fct2Var.end() )
491  cout << "NOT FOUND " << call->unparseToString() << "\t" << call << "\n";
492 
493  // a mapping for function call exps in the subtree must already exist ( postorder traversal )
494  ROSE_ASSERT ( mapIter != fct2Var.end() && mapIter->second );
495 
496  // duplicate the variable
497  SgTreeCopy treeCopy;
498  SgExpression *scnd = mapIter->second;
499  ROSE_ASSERT ( isSgExpression( scnd ) );
500  SgExpression *newVar = isSgExpression( scnd->copy( treeCopy ) );
501  ROSE_ASSERT ( newVar );
502 
503  SgExpression *parent = isSgExpression( call->get_parent() );
504  ROSE_ASSERT ( parent );
505 
506  // replace the node in the AST
507  newVar->set_parent( parent );
508  int k = parent->replace_expression( call, newVar );
509  ROSE_ASSERT ( k == 1 );
510  delete call;
511  }
512  toVisit.pop_front();
513  }
514  }
515  }

529  if ( crt->variantT() == type )

531

561 /*
562 Creates a list of nodes in Function evaluation order (FEO) (first eval function expression, then args; for other nodes, it's BFS)
563 For the previous example, if f, g, and h were expressions (pointers) evaluating to functions, the order is:
564 f, g, h, h(i), g(_), f(_).
565 Note: This is not always equivalent to postorder (because of the BFS on nodes other than function calls).
566 */
567 list<SgNode *>
568 FunctionCallNormalization::createTraversalList( SgNode *root )
569  {
570  list<SgNode *> retList;
571 
572  if ( isSgFunctionCallExp( root ) )
573  {
574  list<SgNode *> temp1 = createTraversalList( isSgFunctionCallExp( root )->get_function() );
