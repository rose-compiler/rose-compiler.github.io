56 int QEditorIndenter::indentation( const QString &s )
57 {
58  if ( s.simplifyWhiteSpace().length() == 0 )
59  return 0;
60 
61  int tabwidth = m_editor->tabStop();
62  int i = 0;
63  int ind = 0;
64  while ( i < (int)s.length() ) {
65  QChar c = s.at( i );
66  if ( c == ' ' ){
67  ind++;
68  } else if ( c == '\t' ){
69  ind += tabwidth;
70  } else {
71  break;
72  }
73  ++i;
74  }
75  return ind;
76 }

78 void QEditorIndenter::indentLine( Q3TextParagraph *p, int &oldIndent, int &newIndent )
79 {
80  QString indentString;
81  indentString.fill( ' ', newIndent );
82  indentString.append( "a" );
83  tabify( indentString );
84  indentString.remove( indentString.length() - 1, 1 );
85  newIndent = indentString.length();
86  oldIndent = 0;
87  while ( p->length() > 0 && ( p->at( 0 )->c == ' ' || p->at( 0 )->c == '\t' ) ) {
88  ++oldIndent;
89  p->remove( 0, 1 );
90  }
91  if ( p->string()->length() == 0 )
92  p->append( " " );
93  if ( !indentString.isEmpty() )
94  p->insert( 0, indentString );
95 }

97 void QEditorIndenter::tabify( QString& s )
98 {
99  int i = 0;
100  int tabSize = m_editor->tabStop();
101  for ( ;; ) {
102  for ( int j = i; j < (int)s.length(); ++j ) {
103  if ( s[ j ] != ' ' && s[ j ] != '\t' ) {
104  if ( j > i ) {
105  QString t = s.mid( i, j - i );
106  int spaces = 0;
107  for ( int k = 0; k < (int)t.length(); ++k )
108  spaces += ( t[ k ] == ' ' ? 1 : tabSize );
109  s.remove( i, t.length() );
110  int tabs = spaces / tabSize;
111  spaces = spaces - ( tabSize * tabs );
112  QString tmp;
113  tmp.fill( ' ', spaces );
114  if ( spaces > 0 )
115  s.insert( i, tmp );
116  tmp.fill( '\t', tabs );
117  if ( tabs > 0 )
118  s.insert( i, tmp );
119  }
120  break;
121  }
122  }
123  i = s.find( '\n', i );
124  if ( i == -1 )
125  break;
126  ++i;
127  }
128 }
