110  // Insert basic block into the dfCFG if it hasn't been already. This only looks at the top-most function on the virtual
111  // function call graph to decide whether to insert the basic block.
112  DfCfg::VertexIterator findOrInsertBasicBlockVertex(const ControlFlowGraph::ConstVertexIterator &cfgVertex,
113  const Function::Ptr &parentFunction, size_t inliningId) {
114  ASSERT_require(!callStack.empty());
115  CallFrame &callFrame = callStack.back();
116  ASSERT_require(cfgVertex != cfg.vertices().end());
117  ASSERT_require(cfgVertex->value().type() == V_BASIC_BLOCK);
118  DfCfg::VertexIterator retval = dfCfg.vertices().end();
119  if (!callFrame.vmap.find(cfgVertex).assignTo(retval)) {
120  BasicBlock::Ptr bblock = cfgVertex->value().bblock();
121  ASSERT_not_null(bblock);
122  retval = insertVertex(DfCfgVertex(bblock, parentFunction, inliningId), cfgVertex);
123 
124  // All function return basic blocks will point only to the special FUNCRET vertex.
125  ASSERT_not_null(partitioner);
126  if (partitioner->basicBlockIsFunctionReturn(bblock)) {
127  if (!dfCfg.isValidVertex(callFrame.functionReturnVertex))
128  callFrame.functionReturnVertex = insertVertex(DfCfgVertex::FUNCRET, parentFunction, inliningId);
129  dfCfg.insertEdge(retval, callFrame.functionReturnVertex);
130  }
131  }
132  return retval;

136  // vertex end iterator is returned.
137  DfCfg::VertexIterator findOrInsertCallReturnVertex(const ControlFlowGraph::ConstVertexIterator &cfgCallSite,
138  const Function::Ptr &parentFunction, size_t inliningId) {

140  ASSERT_require(cfgCallSite->value().type() == V_BASIC_BLOCK);

153  // top-level build function.

164  // Add vertices to the dfCFG based on the CFG. This function is recursive, with each invocation handling the insertion of
165  // one CFG function into the dfCFG -- the function whose entry is at the top of the virtual function call stack.
166  void buildRecursively(const ControlFlowGraph::ConstVertexIterator &functionEntryVertex) {
167  ASSERT_forbid(callStack.empty());
168  ASSERT_require(cfg.isValidVertex(functionEntryVertex));
169 
170  for (CfgTraversal t(cfg, functionEntryVertex, ENTER_EVENTS|LEAVE_EDGE); t; ++t) {
171  if (t.event() == ENTER_VERTEX) {
172  // Every CFG vertex we enter needs to have a corresponding dfCFG vertex created. Due to the t.skipChildren
173  // invocations below (during edge traversals), we will only ever enter the root vertex and those vertices that
174  // are reachable from non-call, non-return edges. I.e., we're visiting vertices that belong to a single
175  // function.
176  if (t.vertex()->value().type() == V_BASIC_BLOCK) {
177  findOrInsertBasicBlockVertex(t.vertex(), callStack.back().function, callStack.back().inliningId);
178  if (partitioner->basicBlockIsFunctionReturn(t.vertex()->value().bblock()))
179  t.skipChildren(); // we're handling return successors explicitly
180  } else {
181  insertVertex(DfCfgVertex::INDET, t.vertex(), callStack.back().function, callStack.back().inliningId);
182  }
183 
184  } else if (t.edge()->value().type() == E_FUNCTION_CALL) {
185  // Function calls are either recursively inlined into the dfCFG or replaced by a special "faked call" vertex
186  // that refers to the function but does not create all its vertices in the dfCFG. In either case, the
187  // traversal does not flow into the function -- either we insert the faked vertex explicitly or we invoke
188  // buildRecurisvely to inline the called function.
189  if (t.event() != ENTER_EDGE)
190  continue;
191  t.skipChildren();
192  Function::Ptr callerFunc = callStack.back().function;
193  size_t callerInliningId = callStack.back().inliningId;
194  DfCfg::VertexIterator callFrom = findVertex(t.edge()->source());
195  ASSERT_require(dfCfg.isValidVertex(callFrom));
196 
197  // Create an optional vertex to which this inlined or faked function call will return. This will be an end
198  // iterator if the call apparently doesn't return.
199  DfCfg::VertexIterator returnTo = findOrInsertCallReturnVertex(t.edge()->source(), callerFunc, callerInliningId);
200 
201  // Function being called
202  Function::Ptr calleeFunc;
203  if (t.edge()->target()->value().type() == V_BASIC_BLOCK)
204  calleeFunc = bestSummaryFunction(t.edge()->target()->value().owningFunctions());
205 
206  // Insert either a summary vertex or recursively inline the callee's body
207  callStack.push_back(CallFrame(dfCfg, calleeFunc, nextInliningId++)); {
208  bool doInline = true;
209  if (callStack.size() > maxCallStackSize) {
210  doInline = false; // too much recursive inlining
211  } else if (t.edge()->target()->value().type() != V_BASIC_BLOCK) {
212  doInline = false; // e.g., call to indeterminate address
213  } else if (!interproceduralPredicate(cfg, t.edge(), callStack.size())) {
214  doInline = false; // user says no inlining
215  } else if (ModulesElf::isUnlinkedImport(partitioner, calleeFunc)) {
216  doInline = false; // callee is not actually present (not linked in yet)
217  }
218 
219  if (doInline) {
220  // Inline the called function into the dfCFG
221  buildRecursively(t.edge()->target());
222  } else {
223  // Insert a "faked" call, i.e. a vertex that summarizes the call by referencing the callee function.
224  // The function pointer will be null if the address is indeterminate.
225  callStack.back().functionEntryVertex =
226  insertVertex(DfCfgVertex(calleeFunc, callerFunc, callerInliningId));
227  callStack.back().functionReturnVertex = callStack.back().functionEntryVertex;
228  }
229 
230  // Create the edge from the call site to the callee's entry vertex.
231  DfCfg::VertexIterator calleeVertex = callStack.back().functionEntryVertex;
232  ASSERT_require(dfCfg.isValidVertex(calleeVertex));
233  dfCfg.insertEdge(callFrom, calleeVertex);
234 
235  // Create the edge from the callee's returning vertex to the CALL's return point.
236  if (dfCfg.isValidVertex(callStack.back().functionReturnVertex) && dfCfg.isValidVertex(returnTo))
237  dfCfg.insertEdge(callStack.back().functionReturnVertex, returnTo);
238  } callStack.pop_back();
239 
240  } else if (t.edge()->value().type() == E_CALL_RETURN) {
241  // Edges from CALL to the return point are handled in the E_FUNCTION_CALL case above, so we don't need to insert
242  // an edge here. However, we must traverse these edges in order to reach the rest of the CFG.
243 
244  } else {
245  // All other edge types in the CFG have a corresponding edge in the dfCFG provided there's a corresponding
246  // source and target vertex in the dfCFG.
247  if (t.event() != LEAVE_EDGE)
248  continue;
249  DfCfg::VertexIterator dfSource = findVertex(t.edge()->source());
250  DfCfg::VertexIterator dfTarget = findVertex(t.edge()->target());
251  if (dfCfg.isValidVertex(dfSource) && dfCfg.isValidVertex(dfTarget))
252  dfCfg.insertEdge(dfSource, dfTarget);
253  }
254  }
255  callStack.back().functionEntryVertex = findVertex(functionEntryVertex);
256  ASSERT_require(dfCfg.isValidVertex(callStack.back().functionEntryVertex));
257  }

347 // If the expression is an offset from the initial stack register then return the offset, else nothing.
348 static Sawyer::Optional<int64_t>
349 isStackAddress(const Rose::BinaryAnalysis::SymbolicExpression::Ptr &expr,
350  const BaseSemantics::SValue::Ptr &initialStackPointer, const SmtSolverPtr &solver) {
351  using namespace Rose::BinaryAnalysis::InstructionSemantics;
352 
353  if (!initialStackPointer)
354  return Sawyer::Nothing();
355  SymbolicExpression::Ptr initialStack = SymbolicSemantics::SValue::promote(initialStackPointer)->get_expression();
356 
357  // Special case where (add SP0 0) is simplified to SP0
358  SymbolicExpression::LeafPtr variable = expr->isLeafNode();
359  if (variable && variable->mustEqual(initialStack, solver))
360  return 0;
361 
362  // Otherwise the expression must be (add SP0 N) where N != 0
363  SymbolicExpression::InteriorPtr inode = expr->isInteriorNode();
364  if (!inode || inode->getOperator() != SymbolicExpression::OP_ADD || inode->nChildren()!=2)
365  return Sawyer::Nothing();
366 
367  variable = inode->child(0)->isLeafNode();
368  SymbolicExpression::LeafPtr constant = inode->child(1)->isLeafNode();
369  if (!constant || !constant->isIntegerConstant())
370  std::swap(variable, constant);
371  if (!constant || !constant->isIntegerConstant())
372  return Sawyer::Nothing();
373  if (!variable || !variable->isIntegerVariable())
374  return Sawyer::Nothing();
375 
376  if (!variable->mustEqual(initialStack, solver))
377  return Sawyer::Nothing();
378 
379  int64_t val = constant->toSigned().get();
380  return val;
381 }

396 Variables::StackVariables
397 findStackVariables(const Function::Ptr &function, const BaseSemantics::RiscOperators::Ptr &ops,
398  const BaseSemantics::SValue::Ptr &initialStackPointer) {
399  using namespace Rose::BinaryAnalysis::InstructionSemantics;
400  ASSERT_not_null(ops);
401  ASSERT_not_null(initialStackPointer);
402  BaseSemantics::State::Ptr state = ops->currentState();
403  ASSERT_not_null(state);
404  SmtSolverPtr solver = ops->solver(); // might be null
405 
406  // What is the word size for this architecture? We'll assume the word size is the same as the width of the stack pointer,
407  // whose value we have in initialStackPointer.
408  ASSERT_require2(initialStackPointer->nBits() % 8 == 0, "stack pointer width is not an integral number of bytes");
409  int64_t wordNBytes = initialStackPointer->nBits() / 8;
410  ASSERT_require2(wordNBytes>0, "overflow");
411 
412  // Find groups of consecutive addresses that were written to by the same instruction(s) and which have the same I/O
413  // properties. This is how we coalesce adjacent bytes into larger variables.
414  using CellCoalescer = Sawyer::Container::IntervalMap<Variables::OffsetInterval, StackVariableMeta>;
415  CellCoalescer cellCoalescer;
416  using OffsetAddress = Sawyer::Container::Map<int64_t, BaseSemantics::SValue::Ptr>; // full address per stack offset
417  OffsetAddress offsetAddresses;
418  BaseSemantics::MemoryCellState::Ptr mem = BaseSemantics::MemoryCellState::promote(state->memoryState());
419  auto cells = mem->allCells();
420  for (const BaseSemantics::MemoryCell::Ptr &cell: boost::adaptors::reverse(cells)) {
421  SymbolicSemantics::SValue::Ptr address = SymbolicSemantics::SValue::promote(cell->address());
422  ASSERT_require2(0 == cell->value()->nBits() % 8, "memory must be byte addressable");
423  size_t nBytes = cell->value()->nBits() / 8;
424  ASSERT_require(nBytes > 0);
425  if (Sawyer::Optional<int64_t> stackOffset = isStackAddress(address->get_expression(), initialStackPointer, solver)) {
426  Variables::OffsetInterval location = Variables::OffsetInterval::baseSize(*stackOffset, nBytes);
427  StackVariableMeta meta(cell->getWriters(), cell->ioProperties());
428  cellCoalescer.insert(location, meta);
429  for (size_t i=0; i<nBytes; ++i) {
430  BaseSemantics::SValue::Ptr byteAddr = ops->add(address, ops->number_(address->nBits(), i));
431  offsetAddresses.insert(*stackOffset+i, byteAddr);
432  }
433  }
434  }
435 
436  // The cellCoalescer has automatically organized the individual bytes into the largest possible intervals that have the
437  // same set of writers and I/O properties. We just need to pick them off in order to build the return value.
438  Variables::StackVariables retval;
439  for (const Variables::OffsetInterval &interval: cellCoalescer.intervals()) {
440  int64_t offset = interval.least();
441  int64_t nRemaining = interval.size();
442  ASSERT_require2(nRemaining > 0, "overflow");
443  while (nRemaining > 0) {
444  BaseSemantics::SValue::Ptr address = offsetAddresses[offset];
445  int64_t nBytes = nRemaining;
446 
447  // Never create a variable that spans memory below and above (or equal to) the initial stack pointer. The initial
448  // stack pointer is generally the boundary between local variables and function arguments (we're considering the
449  // call-return address to be an argument on machines that pass it on the stack).
450  if (offset < 0 && offset + nBytes > 0)
451  nBytes = -offset;
452 
453  // Never create a variable that's wider than the architecture's natural word size.
454  nBytes = std::min(nBytes, wordNBytes);
455  ASSERT_require(nBytes > 0 && nBytes <= nRemaining);
456 
457  // Create the stack variable.
458  const StackVariableMeta &meta = cellCoalescer[offset];
459  Variables::StackVariable var(function, offset, nBytes, Variables::StackVariable::Purpose::UNKNOWN);
460  for (const Address insnAddr: meta.writers.values())
461  var.insertAccess(insnAddr, Variables::Access::WRITE);
462  var.setDefaultName();
463  retval.insert(var.interval(), var);
464 
465  // Advance to next chunk of bytes within this interval
466  offset += nBytes;
467  nRemaining -= nBytes;
468  }
469  }
470  return retval;
471 }

497 std::vector<AbstractLocation>
498 findGlobalVariables(const BaseSemantics::RiscOperators::Ptr &ops, size_t wordNBytes) {
499  ASSERT_not_null(ops);
500  BaseSemantics::State::Ptr state = ops->currentState();
501  ASSERT_not_null(state);
502  ASSERT_require(wordNBytes>0);
503 
504 
505  // Find groups of consecutive addresses that were written to by the same instruction. This is how we coalesce adjacent
506  // bytes into larger variables.
507  using StackWriters = Sawyer::Container::IntervalMap<AddressInterval, Address /*writer*/>;
508  using SymbolicAddresses = Sawyer::Container::Map<Address, BaseSemantics::SValue::Ptr>;
509  StackWriters stackWriters;
510  SymbolicAddresses symbolicAddrs;
511  BaseSemantics::MemoryCellState::Ptr mem = BaseSemantics::MemoryCellState::promote(state->memoryState());
512  for (const BaseSemantics::MemoryCell::Ptr &cell: boost::adaptors::reverse(mem->allCells())) {
513  ASSERT_require2(0 == cell->value()->nBits() % 8, "memory must be byte addressable");
514  size_t nBytes = cell->value()->nBits() / 8;
515  ASSERT_require(nBytes > 0);
516  if (auto va = cell->address()->toUnsigned()) {
517  // There may have been many writers for an address. Rather than write an algorithm to find the largest sets of
518  // addresses written by the same writer, we'll just arbitrarily choose the least address.
519  const AddressSet &allWriters = cell->getWriters();
520  Address leastWriter = 0;
521  if (!allWriters.isEmpty())
522  leastWriter = allWriters.least();
523  stackWriters.insert(AddressInterval::baseSize(*va, nBytes), leastWriter);
524  symbolicAddrs.insert(*va, cell->address());
525  }
526  }
527 
528  // Organize the intervals into a list of global variables
529  std::vector<AbstractLocation> retval;
530  for (const AddressInterval &interval: stackWriters.intervals()) {
531  Address va = interval.least();
532  Address nRemaining = interval.size();
533  ASSERT_require2(nRemaining>0, "overflow");
534  while (nRemaining > 0) {
535  BaseSemantics::SValue::Ptr addr = symbolicAddrs.get(va);
536  Address nBytes = std::min(nRemaining, (Address)wordNBytes);
537  retval.push_back(AbstractLocation(addr, nBytes));
538  va += nBytes;
539  nRemaining -= nBytes;
540  }
541  }
542  return retval;
543 }

702 // Required by dataflow engine: compute new output state given a vertex and input state.
703 BaseSemantics::State::Ptr
704 TransferFunction::operator()(const DfCfg &dfCfg, size_t vertexId, const BaseSemantics::State::Ptr &incomingState) const {
705  BaseSemantics::RiscOperators::Ptr ops = cpu_->operators();
706  BaseSemantics::State::Ptr retval = incomingState->clone();
707  RegisterDictionary::Ptr regDict = cpu_->registerDictionary();
708  ops->currentState(retval);
709 
710  DfCfg::ConstVertexIterator vertex = dfCfg.findVertex(vertexId);
711  ASSERT_require(vertex != dfCfg.vertices().end());
712  if (DfCfgVertex::FAKED_CALL == vertex->value().type()) {
713  CallingConvention::Definition::Ptr ccDefn;
714  Function::Ptr callee = vertex->value().callee();
715  auto genericRegState = as<BaseSemantics::RegisterStateGeneric>(retval->registerState());
716  BaseSemantics::SValue::Ptr origStackPtr = ops->peekRegister(STACK_POINTER_REG);
717 
718  BaseSemantics::SValue::Ptr stackDelta; // non-null if a stack delta is known for the callee
719  if (callee)
720  stackDelta = callee->stackDelta();
721 
722  // Clobber registers that are modified by the callee. The extra calls to updateWriteProperties is because most
723  // RiscOperators implementation won't do that if they don't have a current instruction (which we don't).
724  if (callee && callee->callingConventionAnalysis().didConverge()) {
725  // A previous calling convention analysis knows what registers are clobbered by the call.
726  const CallingConvention::Analysis &ccAnalysis = callee->callingConventionAnalysis();
727  ccDefn = ccAnalysis.defaultCallingConvention();
728  for (RegisterDescriptor reg: ccAnalysis.outputRegisters().listAll(regDict)) {
729  ops->writeRegister(reg, ops->undefined_(reg.nBits()));
730  if (genericRegState)
731  genericRegState->insertProperties(reg, BaseSemantics::IO_WRITE);
732  }
733  if (ccAnalysis.stackDelta())
734  stackDelta = ops->number_(STACK_POINTER_REG.nBits(), *ccAnalysis.stackDelta());
735 
736  } else if ((ccDefn = defaultCallingConvention_)) {
737  // Use a default calling convention definition to decide what registers should be clobbered. Don't clobber the
738  // stack pointer because we might be able to adjust it more intelligently below.
739  for (const ConcreteLocation &loc: ccDefn->outputParameters()) {
740  if (loc.type() == ConcreteLocation::REGISTER && loc.reg() != STACK_POINTER_REG) {
741  ops->writeRegister(loc.reg(), ops->undefined_(loc.reg().nBits()));
742  if (genericRegState)
743  genericRegState->updateWriteProperties(loc.reg(), BaseSemantics::IO_WRITE);
744  }
745  }
746  for (RegisterDescriptor reg: ccDefn->scratchRegisters()) {
747  if (reg != STACK_POINTER_REG) {
748  ops->writeRegister(reg, ops->undefined_(reg.nBits()));
749  if (genericRegState)
750  genericRegState->updateWriteProperties(reg, BaseSemantics::IO_WRITE);
751  }
752  }
753 
754  // Use the stack delta from the default calling convention only if we don't already know the stack delta from a
755  // stack delta analysis, and only if the default CC is caller-cleanup.
756  if (!stackDelta || !stackDelta->toUnsigned()) {
757  if (ccDefn->stackCleanup() == CallingConvention::StackCleanup::BY_CALLER) {
758  stackDelta = ops->number_(origStackPtr->nBits(), ccDefn->nonParameterStackSize());
759  if (ccDefn->stackDirection() == CallingConvention::StackDirection::GROWS_UP)
760  stackDelta = ops->negate(stackDelta);
761  }
762  }
763 
764  } else {
765  // We have not performed a calling convention analysis and we don't have a default calling convention definition. A
766  // conservative approach would need to set all registers to bottom. We'll only adjust the stack pointer (below).
767  }
768 
769  BaseSemantics::SValue::Ptr newStack;
770  if (stackDelta && stackDelta->toUnsigned()) {
771  // stackDelta might not be the same semantic domain as what we need, so create a new delta
772  newStack = ops->add(origStackPtr, ops->number_(origStackPtr->nBits(), *stackDelta->toUnsigned()));
773  } else {
774  // We don't know the callee's delta, therefore we don't know how to adjust the delta for the callee's effect.
775  newStack = ops->undefined_(STACK_POINTER_REG.nBits());
776  }
777  ASSERT_not_null(newStack);
778  ops->writeRegister(STACK_POINTER_REG, newStack);
779  if (genericRegState)
780  genericRegState->updateWriteProperties(STACK_POINTER_REG, BaseSemantics::IO_WRITE);
781 
782  // Adjust the instruction pointer since it probably points to the entry block of the called function. We need it to
783  // now point to the return-to site.
784  if (vertex->nOutEdges() == 1) {
785  if (BasicBlock::Ptr returnToBlock = vertex->outEdges().begin()->target()->value().bblock())
786  ops->writeRegister(INSN_POINTER_REG, ops->number_(INSN_POINTER_REG.nBits(), returnToBlock->address()));
787  }
788 
789  } else if (DfCfgVertex::FUNCRET == vertex->value().type()) {
790  // Identity semantics; this vertex just merges all the various return blocks in the function.
791 
792  } else if (DfCfgVertex::INDET == vertex->value().type()) {
793  // We don't know anything about the vertex, therefore we don't know anything about its semantics
794  retval->clear();
795 
796  } else {
797  // Build a new state using the retval created above, then execute instructions to update it.
798  ASSERT_require(vertex->value().type() == DfCfgVertex::BBLOCK);
799  ASSERT_not_null(vertex->value().bblock());
800  for (SgAsmInstruction *insn: vertex->value().bblock()->instructions()) {
801  if (ignoringSemanticFailures_) {
802  try {
803  cpu_->processInstruction(insn);
804  } catch (const BaseSemantics::Exception &e) {
805  SAWYER_MESG(mlog[WHERE]) <<"semantic failure for " <<insn->toString() <<"\n";
806  }
807  } else {
808  cpu_->processInstruction(insn);
809  }
810  }
811  }
812  return retval;
813 }
