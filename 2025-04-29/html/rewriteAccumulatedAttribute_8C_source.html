130  // 1) Only record declarations found within the source file (exclude all header files
131  // since they will be seen when the same header files are included).
132  // 2) Resort the variable declarations to remove redundent entries.
133  // WRONG: variable declarations could have dependences upon class declarations!
134  // 3) Don't sort all declarations since some could have dependences.
135  // a) class declarations
136  // b) typedefs

148

152  // exclude all CPP directives (since they have already been evaluated by the front-end)
153  info.set_SkipCPPDirectives();

156  {
157  // Enum declarations should not skip their definition since
158  // this is where the constants are declared.
159  case V_SgEnumDeclaration:
160

162  case V_SgTemplateDeclaration:
163  case V_SgTypedefDeclaration:

165  // more conservative and always output the complete class definition.
166

175  // more conservative and always output the complete class definition.
176 
177  // turn off the generation of the function definitions only

186  break;

257 AST_Rewrite::AccumulatedDeclarationsAttribute::generateGlobalDeclarationString() const
258  {
259  // This function generates a string of global declarations required for in the intermediate file to
260  // make sure that all transformation code will compile (since it could depend on declarations
261  // defined within the code).
262 
263  // Since the last declaration is typically the declaration of the function that we are
264  // traversing (as part of the rewrite mechanism which uses the traversal mechanism) it is
265  // tempting to skp the last declaration. But, in fact it would be required if the
266  // transformation code were to specify a recursive function call! So we leave it is and
267  // a often redundently generated forward declaration to the function we are transforming
268  // (rewriting).
269 
270  // error checking
271  ROSE_ASSERT (this != NULL);
272 
273  // Make a copy of the stack of lists becuase extracting
274  // information from the stack distroys the stack
275  stack< list<SgDeclarationStatement*> > copy = stackOfLists;
276 
277  // printf ("In generateGlobalDeclarationString(): stackOfLists.size() = %" PRIuPTR " \n",stackOfLists.size());
278 
279  // Build the list of strings that will be built from the stack of lists
280  list<string> declarationStringList;
281 
282  // error checking
283  ROSE_ASSERT(copy.size() >= 1);
284 
285  // pop all the scopes off the stack until we get to the global scope
286  while (copy.size() > 1)
287  {
288  // pop the stack to get to the outer level scope next
289  copy.pop();
290  }
291 
292  list<SgDeclarationStatement*> & scope = copy.top();
293 
294  // Note that since we only have to iterate over a single scope this code is simpler than
295  // the related loop construction in the generateOpeningNonGlobalDeclarationString() member
296  // function below.
297  string returnString;
298  list<SgDeclarationStatement*>::const_iterator i;
299  for (i=scope.begin(); i != scope.end(); i++)
300  {
301  ROSE_ASSERT ( (*i) != NULL );
302 
303  returnString += generateDeclarationString(*i);
304  }
305 
306  return returnString;

312  // This function generates a string of declarations required for in the intermediate file to
313  // make sure that all transformation code will compile (since it could depend on declarations
314  // defined within the code).
315 
316  // error checking
317  ROSE_ASSERT (this != NULL);
318 
319  // printf ("In generateOpeningNonGlobalDeclarationString(): stackOfLists.size() = %" PRIuPTR " \n",stackOfLists.size());
320 
321  // Make a copy of the stack of lists becuase extracting
322  // information from the stack distroys the stack
323  stack< list<SgDeclarationStatement*> > copy = stackOfLists;
324 
325  // Build the list of strings that will be built from the stack of lists
326  list<string> declarationStringList;
327 
328  // Look only at non-global scopes (the last scope is the global scope by definition)
329  while (copy.size() > 1)
330  {
331  // printf ("In loop over scopes: process a new scope copy.size() = %" PRIuPTR " \n",copy.size());
332 
333  list<SgDeclarationStatement*> & scope = copy.top();
334 
335  string declarationString;
336  // Traverse in reverse order because the declarations are
337  // added to the end of the list in source sequence order.
338  if (scope.size() == 0)
339  {
340  declarationString = " { ";
341  declarationStringList.push_front(declarationString);
342  }
343  else
344  {
345  list<SgDeclarationStatement*>::reverse_iterator i;
346  for (i=scope.rbegin(); i != scope.rend(); i++)
347  {
348  ROSE_ASSERT ( (*i) != NULL );
349  // printf ("(*i)->sage_class_name() = %s \n",(*i)->sage_class_name());
350  if (i == scope.rbegin())
351  declarationString = " { " + generateDeclarationString(*i);
352  else
353  declarationString = generateDeclarationString(*i);
354 
355  // Push onto the FRONT of the list (if we pushed them to the back
356  // of the list then we would have to reverse the order of accumulation
357  // of strings into the returnString.
358  declarationStringList.push_front(declarationString);
359  }
360  }
361 
362  // pop the stack to get to the outer level scope next
363  copy.pop();
364  }
365 
366  string returnString;
367 
368  // Now assemble the list of strings into a single string
369  list<string>::iterator r;
370  for (r=declarationStringList.begin(); r != declarationStringList.end(); r++)
371  {
372  returnString += *r;
373  }
374 
375  return returnString;
