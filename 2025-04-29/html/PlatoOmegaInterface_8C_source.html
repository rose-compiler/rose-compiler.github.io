186  v2 = GetRationalExpression(fa, s2, is_input);
187  print_debug("GetRationalExpression","exiting",debug_GetRationalExpression);

191  plato::RationalExpression v1 = GetRationalExpression( fa, s1, is_input );
192  plato::RationalExpression v2 = GetRationalExpression( fa, s2, is_input );
193  print_debug("GetRationalExpression","exiting",debug_GetRationalExpression);

199  print_debug("GetRationalExpression","exiting",debug_GetRationalExpression);

203  plato::RationalExpression v = GetRationalExpression( fa, s1, is_input);
204  print_debug("GetRationalExpression","exiting",debug_GetRationalExpression);
205  return (-1) * v;
206  }

208  tempVar = GetPlatoStdVar(name,is_input);
209  print_debug("GetRationalExpression","exiting",debug_GetRationalExpression);
