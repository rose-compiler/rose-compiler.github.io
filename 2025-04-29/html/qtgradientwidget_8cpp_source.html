112 double QtGradientWidgetPrivate::correctAngle(double angle) const
113 {
114  double a = angle;
115  while (a >= 360)
116  a -= 360;
117  while (a < 0)
118  a += 360;
119  return a;
120 }

288 void QtGradientWidget::mousePressEvent(QMouseEvent *e)
289 {
290  if (e->button() != Qt::LeftButton)
291  return;
292 
293  QPoint p = e->pos();
294  if (d_ptr->m_gradientType == QGradient::LinearGradient) {
295  QPointF startPoint = d_ptr->toViewport(d_ptr->m_startLinear);
296  double x = p.x() - startPoint.x();
297  double y = p.y() - startPoint.y();
298 
299  if ((d_ptr->m_handleSize * d_ptr->m_handleSize / 4) > (x * x + y * y)) {
300  d_ptr->m_dragHandle = QtGradientWidgetPrivate::StartLinearHandle;
301  d_ptr->m_dragOffset = QPointF(x, y);
302  update();
303  return;
304  }
305 
306  QPointF endPoint = d_ptr->toViewport(d_ptr->m_endLinear);
307  x = p.x() - endPoint.x();
308  y = p.y() - endPoint.y();
309 
310  if ((d_ptr->m_handleSize * d_ptr->m_handleSize / 4) > (x * x + y * y)) {
311  d_ptr->m_dragHandle = QtGradientWidgetPrivate::EndLinearHandle;
312  d_ptr->m_dragOffset = QPointF(x, y);
313  update();
314  return;
315  }
316  } else if (d_ptr->m_gradientType == QGradient::RadialGradient) {
317  QPointF focalPoint = d_ptr->toViewport(d_ptr->m_focalRadial);
318  double x = p.x() - focalPoint.x();
319  double y = p.y() - focalPoint.y();
320 
321  if ((d_ptr->m_handleSize * d_ptr->m_handleSize / 9) > (x * x + y * y)) {
322  d_ptr->m_dragHandle = QtGradientWidgetPrivate::FocalRadialHandle;
323  d_ptr->m_dragOffset = QPointF(x, y);
324  update();
325  return;
326  }
327 
328  QPointF centralPoint = d_ptr->toViewport(d_ptr->m_centralRadial);
329  x = p.x() - centralPoint.x();
330  y = p.y() - centralPoint.y();
331 
332  if ((d_ptr->m_handleSize * d_ptr->m_handleSize / 4) > (x * x + y * y)) {
333  d_ptr->m_dragHandle = QtGradientWidgetPrivate::CentralRadialHandle;
334  d_ptr->m_dragOffset = QPointF(x, y);
335  update();
336  return;
337  }
338 
339  QPointF central = d_ptr->toViewport(d_ptr->m_centralRadial);
340  QRectF r = d_ptr->pointRect(central, 2 * d_ptr->m_handleSize / 3);
341  QRectF r1(0, r.y(), size().width(), r.height());
342  QRectF r2(r.x(), 0, r.width(), r.y());
343  QRectF r3(r.x(), r.y() + r.height(), r.width(), size().height() - r.y() - r.height());
344  QPointF pF(p.x(), p.y());
345  if (r1.contains(pF) || r2.contains(pF) || r3.contains(pF)) {
346  x = pF.x() / size().width() - d_ptr->m_centralRadial.x();
347  y = pF.y() / size().height() - d_ptr->m_centralRadial.y();
348  double clickRadius = sqrt(x * x + y * y);
349  //d_ptr->m_radiusOffset = d_ptr->m_radiusRadial - clickRadius;
350  d_ptr->m_radiusFactor = d_ptr->m_radiusRadial / clickRadius;
351  if (d_ptr->m_radiusFactor == 0)
352  d_ptr->m_radiusFactor = 1;
353  d_ptr->m_dragRadius = d_ptr->m_radiusRadial;
354  d_ptr->m_dragHandle = QtGradientWidgetPrivate::RadiusRadialHandle;
355  mouseMoveEvent(e);
356  update();
357  return;
358  }
359  } else if (d_ptr->m_gradientType == QGradient::ConicalGradient) {
360  QPointF centralPoint = d_ptr->toViewport(d_ptr->m_centralConical);
361  double x = p.x() - centralPoint.x();
362  double y = p.y() - centralPoint.y();
363 
364  if ((d_ptr->m_handleSize * d_ptr->m_handleSize / 4) > (x * x + y * y)) {
365  d_ptr->m_dragHandle = QtGradientWidgetPrivate::CentralConicalHandle;
366  d_ptr->m_dragOffset = QPointF(x, y);
367  update();
368  return;
369  }
370  double radius = size().width();
371  if (size().height() < radius)
372  radius = size().height();
373  radius /= 2;
374  double corr = d_ptr->m_handleSize / 3;
375  radius -= corr;
376  QPointF vp = d_ptr->toViewport(d_ptr->m_centralConical);
377  x = p.x() - vp.x();
378  y = p.y() - vp.y();
379  if (((radius - corr) * (radius - corr) < (x * x + y * y)) &&
380  ((radius + corr) * (radius + corr) > (x * x + y * y))) {
381  QPointF central = d_ptr->toViewport(d_ptr->m_centralConical);
382  QPointF current(e->pos().x(), e->pos().y());
383  x = current.x() - central.x();
384  y = current.y() - central.y();
385  x /= size().width() / 2;
386  y /= size().height() / 2;
387  double r = sqrt(x * x + y * y);
388 
389  double arcSin = asin(y / r);
390  double arcCos = acos(x / r);
391 
392  double angle = arcCos * 180 / M_PI;
393  if (arcSin > 0) {
394  angle = -angle;
395  }
396 
397  d_ptr->m_angleOffset = d_ptr->m_angleConical - angle;
398  d_ptr->m_dragAngle = d_ptr->m_angleConical;
399  d_ptr->m_dragHandle = QtGradientWidgetPrivate::AngleConicalHandle;
400  update();
401  return;
402  }
403  }

413 void QtGradientWidget::mouseMoveEvent(QMouseEvent *e)
414 {
415  if (d_ptr->m_dragHandle == QtGradientWidgetPrivate::NoHandle)
416  return;
417 
418  QPointF newPos = QPointF((double)e->pos().x() - d_ptr->m_dragOffset.x(),
419  (double)e->pos().y() - d_ptr->m_dragOffset.y());
420  QPointF newPoint = d_ptr->fromViewport(newPos);
421  if (newPoint.x() < 0)
422  newPoint.setX(0);
423  else if (newPoint.x() > 1)
424  newPoint.setX(1);
425  if (newPoint.y() < 0)
426  newPoint.setY(0);
427  else if (newPoint.y() > 1)
428  newPoint.setY(1);
429 
430  if (d_ptr->m_dragHandle == QtGradientWidgetPrivate::StartLinearHandle) {
431  d_ptr->m_startLinear = newPoint;
432  emit startLinearChanged(newPoint);
433  } else if (d_ptr->m_dragHandle == QtGradientWidgetPrivate::EndLinearHandle) {
434  d_ptr->m_endLinear = newPoint;
435  emit endLinearChanged(newPoint);
436  } else if (d_ptr->m_dragHandle == QtGradientWidgetPrivate::CentralRadialHandle) {
437  d_ptr->m_centralRadial = newPoint;
438  emit centralRadialChanged(newPoint);
439  } else if (d_ptr->m_dragHandle == QtGradientWidgetPrivate::FocalRadialHandle) {
440  d_ptr->m_focalRadial = newPoint;
441  emit focalRadialChanged(newPoint);
442  } else if (d_ptr->m_dragHandle == QtGradientWidgetPrivate::RadiusRadialHandle) {
443  QPointF centralPoint = d_ptr->toViewport(d_ptr->m_centralRadial);
444  QPointF pF(e->pos().x(), e->pos().y());
445  double x = pF.x() - centralPoint.x();
446  double y = pF.y() - centralPoint.y();
447 
448  if ((d_ptr->m_handleSize * d_ptr->m_handleSize / 4) > (x * x + y * y)) {
449  if (d_ptr->m_radiusRadial != d_ptr->m_dragRadius) {
450  d_ptr->m_radiusRadial = d_ptr->m_dragRadius;
451  emit radiusRadialChanged(d_ptr->m_radiusRadial);
452  }
453  } else {
454  x = pF.x() / size().width() - d_ptr->m_centralRadial.x();
455  y = pF.y() / size().height() - d_ptr->m_centralRadial.y();
456  double moveRadius = sqrt(x * x + y * y);
457  //double newRadius = moveRadius + d_ptr->m_radiusOffset;
458  double newRadius = moveRadius * d_ptr->m_radiusFactor;
459  if (newRadius > 2)
460  newRadius = 2;
461  d_ptr->m_radiusRadial = newRadius;
462  emit radiusRadialChanged(d_ptr->m_radiusRadial);
463  }
464  } else if (d_ptr->m_dragHandle == QtGradientWidgetPrivate::CentralConicalHandle) {
465  d_ptr->m_centralConical = newPoint;
466  emit centralConicalChanged(newPoint);
467  } else if (d_ptr->m_dragHandle == QtGradientWidgetPrivate::AngleConicalHandle) {
468  QPointF centralPoint = d_ptr->toViewport(d_ptr->m_centralConical);
469  QPointF pF(e->pos().x(), e->pos().y());
470  double x = pF.x() - centralPoint.x();
471  double y = pF.y() - centralPoint.y();
472 
473  if ((d_ptr->m_handleSize * d_ptr->m_handleSize / 4) > (x * x + y * y)) {
474  if (d_ptr->m_angleConical != d_ptr->m_dragAngle) {
475  d_ptr->m_angleConical = d_ptr->m_dragAngle;
476  emit angleConicalChanged(d_ptr->m_angleConical);
477  }
478  } else {
479  QPointF central = d_ptr->toViewport(d_ptr->m_centralConical);
480  QPointF current = pF;
481  x = current.x() - central.x();
482  y = current.y() - central.y();
483  x /= size().width() / 2;
484  y /= size().height() / 2;
485  double r = sqrt(x * x + y * y);
486 
487  double arcSin = asin(y / r);
488  double arcCos = acos(x / r);
489 
490  double angle = arcCos * 180 / M_PI;
491  if (arcSin > 0) {
492  angle = -angle;
493  }
494 
495  angle += d_ptr->m_angleOffset;
496 
497  d_ptr->setAngleConical(angle);
498  }
499  }
500  update();
501 }

508 void QtGradientWidget::paintEvent(QPaintEvent *e)
509 {
510  Q_UNUSED(e)
511 
512  QPainter p(this);
513 
514  if (d_ptr->m_backgroundCheckered) {
515  int pixSize = 40;
516  QPixmap pm(2 * pixSize, 2 * pixSize);
517 
518  QPainter pmp(&pm);
519  pmp.fillRect(0, 0, pixSize, pixSize, Qt::white);
520  pmp.fillRect(pixSize, pixSize, pixSize, pixSize, Qt::white);
521  pmp.fillRect(0, pixSize, pixSize, pixSize, Qt::black);
522  pmp.fillRect(pixSize, 0, pixSize, pixSize, Qt::black);
523 
524  p.setBrushOrigin((size().width() % pixSize + pixSize) / 2, (size().height() % pixSize + pixSize) / 2);
525  p.fillRect(rect(), pm);
526  p.setBrushOrigin(0, 0);
527  }
528 
529  QGradient *gradient = 0;
530  switch (d_ptr->m_gradientType) {
531  case QGradient::LinearGradient:
532  gradient = new QLinearGradient(d_ptr->m_startLinear, d_ptr->m_endLinear);
533  break;
534  case QGradient::RadialGradient:
535  gradient = new QRadialGradient(d_ptr->m_centralRadial, d_ptr->m_radiusRadial, d_ptr->m_focalRadial);
536  break;
537  case QGradient::ConicalGradient:
538  gradient = new QConicalGradient(d_ptr->m_centralConical, d_ptr->m_angleConical);
539  break;
540  default:
541  break;
542  }
543  if (!gradient)
544  return;
545 
546  gradient->setStops(d_ptr->m_gradientStops);
547  gradient->setSpread(d_ptr->m_gradientSpread);
548 
549  p.save();
550  p.scale(size().width(), size().height());
551  p.fillRect(QRect(0, 0, 1, 1), *gradient);
552  p.restore();
553 
554  p.setRenderHint(QPainter::Antialiasing);
555 
556  QColor c = QColor::fromRgbF(0.5, 0.5, 0.5, 0.5);
557  QBrush br(c);
558  p.setBrush(br);
559  QPen pen(Qt::white);
560  pen.setWidthF(1);
561  p.setPen(pen);
562  QPen dragPen = pen;
563  dragPen.setWidthF(2);
564  if (d_ptr->m_gradientType == QGradient::LinearGradient) {
565  p.save();
566  if (d_ptr->m_dragHandle == QtGradientWidgetPrivate::StartLinearHandle)
567  p.setPen(dragPen);
568  d_ptr->paintPoint(&p, d_ptr->m_startLinear, d_ptr->m_handleSize);
569  p.restore();
570 
571  p.save();
572  if (d_ptr->m_dragHandle == QtGradientWidgetPrivate::EndLinearHandle)
573  p.setPen(dragPen);
574  d_ptr->paintPoint(&p, d_ptr->m_endLinear, d_ptr->m_handleSize);
575  p.restore();
576  } else if (d_ptr->m_gradientType == QGradient::RadialGradient) {
577  QPointF central = d_ptr->toViewport(d_ptr->m_centralRadial);
578 
579  p.save();
580  QRectF r = d_ptr->pointRect(central, 2 * d_ptr->m_handleSize / 3);
581  QRectF r1(0, r.y(), size().width(), r.height());
582  QRectF r2(r.x(), 0, r.width(), r.y());
583  QRectF r3(r.x(), r.y() + r.height(), r.width(), size().height() - r.y() - r.height());
584  p.fillRect(r1, c);
585  p.fillRect(r2, c);
586  p.fillRect(r3, c);
587  p.setBrush(Qt::NoBrush);
588  p.save();
589  if (d_ptr->m_dragHandle == QtGradientWidgetPrivate::CentralRadialHandle)
590  p.setPen(dragPen);
591  d_ptr->paintPoint(&p, d_ptr->m_centralRadial, d_ptr->m_handleSize);
592  p.restore();
593 
594  QRectF rect = QRectF(central.x() - d_ptr->m_radiusRadial * size().width(),
595  central.y() - d_ptr->m_radiusRadial * size().height(),
596  2 * d_ptr->m_radiusRadial * size().width(),
597  2 * d_ptr->m_radiusRadial * size().height());
598  p.setClipRect(r1);
599  p.setClipRect(r2, Qt::UniteClip);
600  p.setClipRect(r3, Qt::UniteClip);
601  p.drawEllipse(rect);
