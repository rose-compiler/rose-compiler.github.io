178 void QRSourceBox::eventCacheNewNodes(set<SgNode *> &nodes) {
179  int i;
180  i = cacheSize() - nodes.size() + 1;
181  for (set<SgNode *>::iterator iter = nodes.begin(); iter != nodes.end(); iter++) {
182  string code;
183  SgNode *node = *iter;
184  // get source-file
185  string filename;
186  SgNode *file = node;
187 
188  while (file && !isSgFile(file)) file = file->get_parent();
189  if (file) {
190  int pos;
191  filename = ((SgFile *) file)->getFileName();
192  if ((pos = filename.find_last_of("\\/")) != string::npos) {
193  filename = filename.substr(pos + 1);
194  }
195  filename = string(" <") + filename + ">";
196  }
197 
198  if (QRAstInfo::isUnparseBlackList(node)) {
199  code = QROSE::format("\n/***** [ %d: %s%s(note: node has not been unparsed) ] ****/\n",
200  i, node->sage_class_name(), filename.c_str());
201  } else {
202  code += QROSE::format("\n/***** [ %d: %s%s] ****/\n",
203  i, node->sage_class_name(), filename.c_str()) + node->unparseToCompleteString();
204  }
205  i++;
206  m_cacheSource += code;
207  }
208  if (m_isCacheView) updateCacheView();
209  updateCacheViewButton();
210  emit setInfo(QString("%1 %2 been received [total=%3]")
211  .arg(nodes.size()).arg((nodes.size() == 1)? "node has":
212  "nodes have").arg(cacheSize()), statusTimer);
213 }

496  switch (msg->id()) {
497  case msg_node:

499  SgNode *node = ((QRoseComponent::SgNodeMsg *) msg)->getNode();

517 void QRSourceBox::lookupNode(SgNode *node) {
518  if (isCacheView()) return;
519  if (!isLookup()) return;
520  if (!m_astCoords) return;
521  map<SgNode *, QRCoordBox *>* coords = m_astCoords->nodeCoords();
522  if (!coords) return;
523  std::map<SgNode *, QRCoordBox *>::iterator iter = coords->find(node);
524  if (iter == coords->end()) return;
525  QRCoordBox *box = iter->second;
526  if (!box) return;
527 
528  m_wgEditor->setSelection(box->line0(), box->col0(), box->line1(),
529  box->col1(), QEditor::sel1);
530  m_wgEditor->gotoLine(box->line0()+1);
531  // the reason why we select code in the first place is that it better positions the text.
532  if (!m_lookupWithHighlight) {
533  m_wgEditor->removeSelection(QEditor::sel1);
534  }

538 // adapted from unparseToCompleteString
539 string QRSourceBox::unparseFileToString (SgFile *file, bool use_coords) {
540  SgUnparse_Info* inputUnparseInfoPointer = new SgUnparse_Info();
541  inputUnparseInfoPointer->unset_SkipComments(); // generate comments
542  inputUnparseInfoPointer->unset_SkipWhitespaces(); // generate all whitespaces to format the code
543 
544  ostringstream outputString;
545 
546  string fileName = file->getFileName();
547 
548  Unparser_Opt roseOptions;
549 
550  QRUnparser *roseUnparser = new QRUnparser(m_astCoords, &outputString, fileName, roseOptions);
551  switch (file->variantT())
552  {
553  case V_SgSourceFile:
554  {
555  SgSourceFile* sourceFile = isSgSourceFile(file);
556  roseUnparser->unparseFile(sourceFile, *inputUnparseInfoPointer);
557 
558  break;
559  }
560 
561  case V_SgBinaryComposite:
562  {
563  SgBinaryComposite* binary = isSgBinaryComposite(file);
564  roseUnparser->unparseFile(binary, *inputUnparseInfoPointer);
565  break;
566  }
567 
568  case V_SgUnknownFile:
569  {
570  SgUnknownFile* unknownFile = isSgUnknownFile(file);
571 
572  unknownFile->set_skipfinalCompileStep(true);
573 
574  printf ("Warning: Unclear what to unparse from a SgUnknownFile (set skipfinalCompileStep) \n");
575  break;
576  }
577 
578  default:
579  {
580  printf ("Error: default reached in unparser: file = %s \n",file->class_name().c_str());
581  ROSE_ASSERT(false);
582  }
583  }
584 
585  outputString.put(0);
586  return outputString.str();
587 }
