52 class OARIFGEdgeIterator : public RIFGEdgeIterator {
53 public:
54  OARIFGEdgeIterator (RIFG &_fg, CFG::Node *nip,
55  RIFG::EdgeDirection _dir):fg(_fg),dir(_dir),
56  oa_out_iter(nip),oa_in_iter(nip) { }
57 
58  RIFGEdgeId Current() {
59  CFG::Edge *ep = 0;
60  if (dir == RIFG::ED_INCOMING) {
61  if ((bool)oa_in_iter) {
62  ep = dynamic_cast<CFG::Edge *>((DGraph::Edge *)(oa_in_iter));
63  return ((OARIFG &)fg).edge_to_id_map[ep];
64  } else {
65  return RIFG_NIL;
66  }
67  } else {
68  if ((bool)oa_out_iter) {
69  ep = dynamic_cast<CFG::Edge *>((DGraph::Edge *)(oa_out_iter));
70  return ((OARIFG &)fg).edge_to_id_map[ep];
71  } else {
72  return RIFG_NIL;
73  }
74  }
75  }
76 
77  RIFGEdgeId operator++(int) {
78  CFG::Edge *old;
79  if (dir == RIFG::ED_INCOMING) {
80  if ((bool)oa_in_iter) {
81  old = dynamic_cast<CFG::Edge *>((DGraph::Edge *)oa_in_iter);
82  ++oa_in_iter;
83  return ((OARIFG &)fg).edge_to_id_map[old];
84  } else
85  return RIFG_NIL;
86  } else {
87  if ((bool)oa_out_iter) {
88  old = dynamic_cast<CFG::Edge *>((DGraph::Edge *)oa_out_iter);
89  ++oa_out_iter;
90  return ((OARIFG &)fg).edge_to_id_map[old];
91  } else
92  return RIFG_NIL;
93  }
94  }
95  void Reset() { ROSE_ABORT(); };
96  virtual ~OARIFGEdgeIterator() { };
97 private:
98  RIFG &fg;
99  RIFG::EdgeDirection dir;
100  CFG::OutgoingEdgesIterator oa_out_iter;
101  CFG::IncomingEdgesIterator oa_in_iter;
102 };

151 OARIFG::OARIFG(CFG &_cfg):cfg(_cfg)
152 {
153  // Initialize the nodeid to node map.
154  CFG::NodesIterator ni(cfg);
155  while ((bool)ni) {
156  CFG::Node *nnn = dynamic_cast<CFG::Node *>((DGraph::Node *)ni);
157  nodeid_to_node_map[nnn->getID()] = nnn;
158  ++ni;
159  }
160 
161  // Create edges id's (move to CFG?)
162  // and Initialize the edgeid to edge map.
163  CFG::EdgesIterator ei(cfg);
164  int num = 1;
165  while ((bool)ei) {
166  CFG::Edge *eee = dynamic_cast<CFG::Edge *>((DGraph::Edge *)ei);
167  edge_to_id_map[eee] = num;
168  edgeid_to_edge_map[num] = eee;
169  ++num;
170  ++ei;
171  }
172 }
