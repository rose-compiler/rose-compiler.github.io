81 void MarkerWidget::paintEvent( QPaintEvent* /*e*/ )
82 {
83  m_buffer.fill();
84 
85  Q3TextParagraph *p = m_editor->getDocument()->firstParagraph();
86  QPainter painter( &m_buffer );
87  int yOffset = m_editor->contentsY();
88  while ( p ) {
89  if ( !p->isVisible() ) {
90  p = p->next();
91  continue;
92  }
93  if ( p->rect().y() + p->rect().height() - yOffset < 0 ) {
94  p = p->next();
95  continue;
96  }
97  if ( p->rect().y() - yOffset > height() )
98  break;
99 
100  ParagData* paragData = (ParagData*) p->extraData();
101  unsigned int mark = paragData ? paragData->mark() : 0;
102  if (mark) {
103  unsigned int current = 0x01;
104  for (; current < mark+1; current = current << 1) {
105  if (mark & current) {
106  QMap<int,QPixmap>::const_iterator it = m_pixmapMap.find(current);
107  if (it != m_pixmapMap.end()) {
108  painter.drawPixmap( 3,
109  p->rect().y() + ( p->rect().height() - (*it).height() ) / 2 - yOffset,
110  *it );
111  }
112  }
113  }
114  }
115  p = p->next();
116  }
117 
118  painter.end();
119  bitBlt( this, 0, 0, &m_buffer );
120 }

129 void MarkerWidget::contextMenuEvent( QContextMenuEvent* e )
130 {
131  Q3PopupMenu m( 0, "editor_breakpointsmenu" );
132  Q3PopupMenu sub( 0, "editor_breakpointsmenu_sub" );
133 
134  int toggleBreakpoint = 0;
135  int toggleBookmark = 0;
136  int lmbClickChangesBPs = 0;
137  int lmbClickChangesBookmarks = 0;
138 
139  Q3TextParagraph *p = m_editor->getDocument()->firstParagraph();
140  int yOffset = m_editor->contentsY();
141  while ( p ) {
142  if ( e->y() >= p->rect().y() - yOffset && e->y() <= p->rect().y() + p->rect().height() - yOffset ) {
143  ParagData* data = (ParagData*) p->extraData();
144 
145  toggleBookmark = m.insertItem( m_bookmarkDescr );
146  m.setItemEnabled(toggleBookmark, m_changeBookmarksAllowed);
147  m.setItemChecked(toggleBookmark, data->mark() & 0x01);
148 
149  toggleBreakpoint = m.insertItem( m_breakpointDescr );
150  m.setItemEnabled(toggleBreakpoint, m_changeBreakpointsAllowed);
151  m.setItemChecked(toggleBreakpoint, data->mark() & 0x02);
152 
153  m.insertItem("Set default mark type", &sub);
154  lmbClickChangesBookmarks = sub.insertItem( m_bookmarkDescr );
155  lmbClickChangesBPs = sub.insertItem( m_breakpointDescr );
156  m.setItemChecked(lmbClickChangesBPs, m_clickChangesBPs);
157  m.setItemChecked(lmbClickChangesBookmarks, !m_clickChangesBPs);
158  break;
159  }
160  p = p->next();
161  }
162 
163  int res = m.exec( e->globalPos() );
164  if ( res == -1)
165  return;
166 
167  ParagData* data = (ParagData*) p->extraData();
168 
169  Mark mark;
170  mark.line = p->paragId();
171 
172  if ( res == toggleBookmark && m_changeBookmarksAllowed ) {
173  mark.type = 0x01;
174  if ( data->mark() & 0x01 ) {
175  data->setMark( data->mark() & ~0x01 );
176  //emit markChanged(mark, KTextEditor::MarkInterfaceExtension::MarkRemoved);
177  }
178  else {
179  data->setMark( data->mark() | 0x01 );
180  // emit markChanged(mark, KTextEditor::MarkInterfaceExtension::MarkAdded);
181  }
182  } else if ( res == toggleBreakpoint && m_changeBreakpointsAllowed ) {
183  mark.type = 0x02;
184  if ( data->mark() & 0x02 ) {
185  data->setMark( data->mark() & ~0x02 );
186  //emit markChanged(mark, KTextEditor::MarkInterfaceExtension::MarkRemoved);
187  }
188  else {
189  data->setMark( data->mark() | 0x02 );
190  // emit markChanged(mark, KTextEditor::MarkInterfaceExtension::MarkAdded);
191  }
192  } else if ( res == lmbClickChangesBPs ) {
193  m_clickChangesBPs = !m.isItemChecked(lmbClickChangesBPs);
194  } else if ( res == lmbClickChangesBookmarks ) {
195  m_clickChangesBPs = m.isItemChecked(lmbClickChangesBookmarks);
196  }
197 
198  emit marksChanged();
199  doRepaint();
200 }

203 {
204  Q3TextParagraph *p = m_editor->getDocument()->firstParagraph();
205  int yOffset = m_editor->contentsY();
206  ParagData* data = 0L;
207  while ( p ) {
208  if ( e->y() >= p->rect().y() - yOffset && e->y() <= p->rect().y() + p->rect().height() - yOffset ) {
209  data = (ParagData*) p->extraData();
210  break;
211  }
212  p = p->next();
213  }
214 
215  if (e->button() == Qt::LeftButton) {
216  if (!data) return;
217 
218  Mark mark;
219  mark.line = p->paragId();
220  if (m_clickChangesBPs && m_changeBreakpointsAllowed) {
221  mark.type = 0x02;
222  if (data->mark() & 0x02) {
223  data->setMark(data->mark() & ~0x02);
224 // emit markChanged(mark, KTextEditor::MarkInterfaceExtension::MarkRemoved);
