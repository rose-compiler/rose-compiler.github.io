1 // This file represents support in the ROSE AST for Dwarf debug information.
2 // Dwarf is generated when source code is compiled using a compiler's debug mode.
3 // the Dwarf information is represented a special sections in the file format of
4 // the binary executable. These sections are read using libdwarf (open source
5 // library used for reading and writing dwarf sections). In ROSE we read the
6 // Dwarf information and build special IR nodes into the AST. This work effects

184 std::pair<LineColumnFilePosition,LineColumnFilePosition>
185 SgAsmDwarfLineList::sourceCodeRange( int file_id ) {
186  DwarfInstructionSourceMapReturnType maps = buildInstructionAddressSourcePositionMaps();
187 
188  SgSourcePositionInstructionAddressMapPtrList & source_code_instruction_map = *(maps.second);
189 
190  SgSourcePositionInstructionAddressMapPtrList::iterator lowerBound = source_code_instruction_map.begin();
191  SgSourcePositionInstructionAddressMapPtrList::reverse_iterator upperBound = source_code_instruction_map.rbegin();
192 
193  // Find the first source position in the file specified by the input file_id
194  while ( (lowerBound != source_code_instruction_map.end()) && (lowerBound->first.first != file_id) ) {
195  lowerBound++;
196  }
197 
198  while ( (upperBound != source_code_instruction_map.rend()) && (upperBound->first.first != file_id) ) {
199  upperBound++;
200  }
201 
202  LineColumnFilePosition start(lowerBound->first.second);
203  LineColumnFilePosition end(upperBound->first.second);
204 
205  // Check if this was a case of there not being any entries for this file
206  if (lowerBound == source_code_instruction_map.end()) {
207  ASSERT_require(upperBound == source_code_instruction_map.rend());
208  // printf ("lowerBound == source_code_instruction_map.end() --- no entries for file %d \n",file_id);
209 
210  // Reset the line and column information to indicate that there were no entries.
211  start = LineColumnFilePosition(std::pair<int,int>(-1,-1));
212  end = LineColumnFilePosition(std::pair<int,int>(-1,-1));
213  }
214 
215  return std::pair<LineColumnFilePosition,LineColumnFilePosition>(start,end);
216 }
