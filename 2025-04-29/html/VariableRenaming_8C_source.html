394 void VariableRenaming::insertDefsForExternalVariables(SgFunctionDeclaration* function)
395 {
396  ROSE_ASSERT(function->get_definition() != NULL);
397 
398  set<VarName> usedNames = getVarsUsedInSubtree(function);
399 
400  TableEntry& originalVarsAtFunctionEntry = originalDefTable[function->get_definition()];
401  TableEntry& expandedVarsAtFunctionEntry = expandedDefTable[function->get_definition()];
402 
403  //Iterate over each used variable and check it it is declared outside of the function scope
404 
405  foreach(const VarName& usedVar, usedNames)
406  {
407  VarName rootName;
408  rootName.assign(1, usedVar[0]);
409 
410  SgScopeStatement* varScope = SageInterface::getScope(rootName[0]);
411  SgScopeStatement* functionScope = function->get_definition();
412 
413  //If it is a local variable, there should be a def somewhere inside the function
414  if (varScope == functionScope || SageInterface::isAncestor(functionScope, varScope))
415  {
416  //We still need to insert defs for compiler-generated variables (e.g. __func__), since they don't have defs in the AST
417  if (!isBuiltinVar(rootName))
418  continue;
419  }
420  else if (isSgGlobal(varScope))
421  {
422  //Handle the case of declaring "extern int x" inside the function
423  //Then, x has global scope but it actually has a definition inside the function so we don't need to insert one
424  if (SageInterface::isAncestor(function->get_definition(), rootName[0])) {
425  //When else could a var be declared inside a function and be global?
426  if(SgVariableDeclaration* varDecl = isSgVariableDeclaration(rootName[0]->get_parent())){
427  ASSERT_always_require(varDecl->get_declarationModifier().get_storageModifier().isExtern());
428  continue;
429  } else if (isSgFunctionParameterList(rootName[0]->get_parent())) {
430  // x can also have global scope when it appears as a formal parameter in an extern function declaration,
431  // as in: void f() { void g(int x); ... }
432 #ifndef NDEBUG
433  SgFunctionDeclaration *fdecl = isSgFunctionDeclaration(rootName[0]->get_parent()->get_parent());
434  assert(fdecl!=NULL);
435  assert(fdecl!=function);
436 #endif
437  continue;
438  }
439  }
440  }
441 
442  //Are there any other types of external vars?
443  ROSE_ASSERT(isBuiltinVar(rootName) || isSgClassDefinition(varScope) || isSgNamespaceDefinitionStatement(varScope)
444  || isSgDeclarationScope(varScope) || isSgGlobal(varScope));
445 
446  //The variable is not in local scope; we need to insert a def for it at the function definition
447  for (size_t i = 0; i < usedVar.size(); i++)
448  {
449  //Create a new varName vector that goes from beginning to end - i
450  VarName newName;
451  newName.assign(usedVar.begin(), usedVar.end() - i);
452  if (originalVarsAtFunctionEntry[newName].empty())
453  {
454  originalVarsAtFunctionEntry[newName].push_back(function->get_definition());
455  }
456  ROSE_ASSERT(expandedVarsAtFunctionEntry.count(newName) == 0);
457  }
458  }

1295 bool VariableRenaming::mergeDefs(cfgNode curNode, bool *memberRefInserted, NodeVec &changedNodes)
1296 {

1298 
1299  bool changed = false;
1300 
1301  if (DEBUG_MODE)
1302  {
1303  cout << "merging defs..." << endl;
1304  printDefs(node);
1305  }
1306 
1307  //We have the definitions stored as follows:
1308  //defTable contains the definitions as propagated by the CFG analysis
1309  //originalDefTable contains the original definitions that are part of the
1310  //given statement.
1311 
1312  //When we want to propagate the defs from the previous node(s) to this one,
1313  //We perform a few steps. This is dependent on the number of incoming edges.
1314 
1315  /*1 Edge: When we have linear control flow, we do the following:
1316  * 1. Copy the definitions from the previous node wholesale to a staging table.
1317  * 2. Copy in the original definitions from the current node, overwriting those
1318  * from the previous node.
1319  * 3. Compare the staging and current tables, and only overwrite if needed.
1320  */
1321 
1322  /*2+ Edges: When we have branched control flow, we do the following:
1323  * 1. Copy the definitions from the previous node(s) wholesale to a staging table.
1324  * Be careful to not insert duplicates.
1325  * 2. Copy in the original definitions from the current node, overwriting those
1326  * from the previous node(s).
1327  * 3. Compare the staging and current tables, and only overwrite if needed.
1328  */
1329 
1330  //Expand any member variable references at the current node.
1331  expandMemberDefinitions(curNode);
1332 
1333  TableEntry propDefs;
1334  //Retrieve the defs coming from previous cfgNodes
1335  aggregatePreviousDefs(curNode, propDefs);
1336 
1337  //Make sure variables always have a first definition
1338  if (!isSgFunctionDefinition(node) && (!isSgInitializedName(node) || isSgCtorInitializerList(node->get_parent())))
1339  {
1340

1342  {
1343  if (propDefs.find(entry.first) == propDefs.end())
1344  {
1345  *memberRefInserted = insertExpandedDefsForUse(curNode, entry.first, changedNodes) || *memberRefInserted;
1346  if (*memberRefInserted)
1347  break;
1348  }
1349  }
1350  }
1351 
1352  //Replace every entry in staging table that has definition in original defs
1353  //Also assign renaming numbers to any new definitions
1354 
1355  foreach(TableEntry::value_type& entry, originalDefTable[node])
1356  {
1357  //Replace the entry for this variable with the definitions at this node.
1358  propDefs[entry.first] = entry.second;
1359 
1360  //Now, iterate the definition vector for this node
1361  ROSE_ASSERT(entry.second.size() == 1);
1362 
1363  foreach(NodeVec::value_type& defNode, entry.second)
1364  {
1365  //Assign a number to each new definition. The function will prevent duplicates
1366  addRenameNumberForNode(entry.first, defNode);
1367  }
1368  }
1369 
1370  //For every originalDef, insert expanded defs for any propagated defs
1371  //that have an originalDef as a prefix
1372  VarName expVar;
1373 
1374  foreach(TableEntry::value_type& entry, originalDefTable[node])
1375  {
1376 
1377  foreach(TableEntry::value_type& propEntry, propDefs)
1378  {
1379  //Don't insert a def if it is already originally defined.
1380  if (originalDefTable[node].count(propEntry.first) != 0)
1381  {
1382  continue;
1383  }
