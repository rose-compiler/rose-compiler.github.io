82 int action(QDomElement c, QFormatScheme *f, QHash<QString, int>& pids, int fid = 0)
83 {
84  QString paren, spid, spt, sfid;
85 
86  sfid = c.attribute("format");
87 
88  if ( sfid.count() )
89  {
90  fid |= QNFAAction::Highlight | QNFAAction::format(f->id(sfid));
91  }
92 
93  paren = c.attribute("parenthesis");
94 
95  if ( paren.count() )
96  {
97  spid = paren.section(':', 0, -2);
98  spt = paren.section(':', -1, -1);
99 
100  if ( spt.endsWith("@nomatch") )
101  {
102  spt.chop(8);
103  } else {
104  fid |= QNFAAction::MatchParen;
105  }
106 
107  if ( spid.count() )
108  {
109  /*qDebug("paren : [%s|%s] => 0x%x",
110  qPrintable(spid), qPrintable(spt),
111  (spt == "open" ? QNFAAction::ParenOpen : QNFAAction::ParenClose) | pid(spid, pids));
112  */
113 
114  if ( spt == "open" )
115  fid |= QNFAAction::ParenOpen;
116  else if ( spt == "close" )
117  fid |= QNFAAction::ParenClose;
118  else if ( spt == "boundary" )
119  fid |= QNFAAction::ParenOpen | QNFAAction::ParenClose;
120 
121  fid |= QNFAAction::parenthesis(pid(spid, pids));
122 
123  /*
124  qDebug("paren : [%s|%s] => 0x%x",
125  qPrintable(spid), qPrintable(spt),
126  fid & QNFAAction::ParenMask);
127  */
128  }
129  }
130 
131  if ( stringToBool(c.attribute("indent"), false) )
132  fid |= QNFAAction::Indent;
133 
134  if ( stringToBool(c.attribute("fold"), false) )
135  fid |= QNFAAction::Fold;
136 
137  // TODO : determine ambiguity automatically
138  if ( stringToBool(c.attribute("ambiguous"), false) )
139  fid |= QNFAAction::Ambiguous;
140 
141  return fid;
142 }
