462 static z3::expr
463 portable_z3_bv_val(z3::context *ctx, uint64_t value, size_t nBits) {
464  // Z3 prior to 4.8 didn't have a way to check the version at compile time, however a workaround for ROSE users was to
465  // define the Z3_*_VERSION constants themselves for these older Z3 versions.
466 #if defined(Z3_MAJOR_VERSION) && defined(Z3_MINOR_VERSION) && defined(Z3_BUILD_NUMBER)
467  #if Z3_MAJOR_VERSION < 4
468  // z3 < 4.0.0
469  #error "Z3 version < 4.x.x are not supported"
470  #elif Z3_MAJOR_VERSION == 4 && Z3_MINOR_VERSION < 7
471  // z3 >= 4.0.0 && z3 < 4.7.0
472  return ctx->bv_val((unsigned long long)value, (unsigned)nBits);
473  #else
474  // z3 >= 4.7.0
475  return ctx->bv_val((uint64_t)value, (unsigned)nBits);
476  #endif
477 #else
478  // Not all ROSE users manually defined the Z3_*_VERSION constants. Therefore, if the compiler is compiling this part of the
479  // code the only thing we know is that it's a version before 4.8.0 where the version numbers were first defined in the Z3
480  // source code. See [https://github.com/Z3Prover/z3/issues/1833].
481  //
482  // The Z3 API for z3::context::bv_val changed in version 4.7 that makes it incompatible with earlier versions, but since we
483  // have no Z3_*_VERSION constants we can't tell whether this is z3 4.7 or an earlier version. We assume that it's an earlier
484  // version since that's more likely.
485  //
486  // If you get an error here it's probably because we assumed wrong and you're actually using Z3 4.7. Instead of
487  // changing the ROSE source code, define the Z3_*_VERSION values in your compiler command-line.
488  return ctx->bv_val((unsigned long long)value, (unsigned)nBits); // DO NOT CHANGE; this assumes z3 < 4.7.
489 #endif
490 }

492 Z3Solver::Z3ExprTypePair
493 Z3Solver::ctxLeaf(const SymbolicExpression::Leaf *leaf) {
494  ASSERT_not_null(leaf);
495  if (leaf->isIntegerConstant()) {
496  if (leaf->nBits() <= 64) {
497  z3::expr z3expr = portable_z3_bv_val(ctx_, *leaf->toUnsigned(), leaf->nBits());
498  return Z3ExprTypePair(z3expr, BIT_VECTOR);
499  } else {
500  // Z3-4.5.0 and 4.6.0 lack an interface for creating a bit vector with more than 64 bits from a hexadecimal
501  // string. A function taking a bool[] has been added to z3's development branch but is not yet
502  // available. Therefore, we construct a wider-than-64-bit constant by concatenating up to 64 bits at a time.
503  z3::expr z3expr(*ctx_);
504  for (size_t offset=0; offset < leaf->bits().size(); offset += 64) {
505  size_t windowSize = std::min((size_t)64, leaf->bits().size() - offset);
506  Sawyer::Container::BitVector windowBits = leaf->bits();
507  windowBits.shiftRight(offset).resize(windowSize);
508  z3::expr window = portable_z3_bv_val(ctx_, windowBits.toInteger(), windowSize);
509  if (0 == offset) {
510  z3expr = window;
511  } else {
512  z3expr = z3::concat(window, z3expr);
513  }
514  }
515  return Z3ExprTypePair(z3expr, BIT_VECTOR);
516  }
517  } else if (leaf->isIntegerVariable()) {
518  z3::func_decl decl = ctxVarDecls_.get(leaf);
519  return Z3ExprTypePair(decl(), BIT_VECTOR);
520  } else {
521  ASSERT_require(leaf->isMemoryVariable());
522  z3::func_decl decl = ctxVarDecls_.get(leaf);
523  return Z3ExprTypePair(decl(), MEM_STATE);
524  }
525 }

544 Z3Solver::Z3ExprTypePair
545 Z3Solver::ctxExpression(SymbolicExpression::Node *expr) {
546  ASSERT_not_null(expr);
547  typedef std::vector<Z3ExprTypePair> Etv;
548 
549  SymbolicExpression::Leaf *leaf = dynamic_cast<SymbolicExpression::Leaf*>(expr);
550  SymbolicExpression::Interior *inode = dynamic_cast<SymbolicExpression::Interior*>(expr);
551 
552  ASSERT_not_null(ctx_);
553  z3::expr z3expr(*ctx_);
554  if (ctxCses_.exists(expr)) {
555  return ctxCses_[expr];
556  } else if (leaf != NULL) {
557  return ctxLeaf(leaf);
558  } else {
559  ASSERT_not_null(inode);
560  switch (inode->getOperator()) {
561  case SymbolicExpression::OP_NONE:
562  ASSERT_not_reachable("not possible for an interior node");
563  case SymbolicExpression::OP_ADD: {
564  Etv children = ctxCast(ctxExpressions(inode->children()), BIT_VECTOR);
565  ASSERT_forbid(children.empty());
566  z3expr = children[0].first;
567  for (size_t i = 1; i < children.size(); ++i)
568  z3expr = z3expr + children[i].first;
569  return Z3ExprTypePair(z3expr, BIT_VECTOR);
570  }
571  case SymbolicExpression::OP_AND: {
572  Etv children = ctxExpressions(inode->children());
573  Type type = mostType(children);
574  children = ctxCast(children, type);
575  ASSERT_require(children.size() >= 2);
576  if (BOOLEAN == type) {
577  z3expr = children[0].first;
578  for (size_t i = 1; i < children.size(); ++i)
579  z3expr = z3expr && children[i].first;
580  } else {
581  ASSERT_require(BIT_VECTOR == type);
582  z3expr = children[0].first;
583  for (size_t i = 1; i < children.size(); ++i)
584  z3expr = z3expr & children[i].first;
585  }
586  return Z3ExprTypePair(z3expr, type);
587  }
588  case SymbolicExpression::OP_ASR:
589  return ctxArithmeticShiftRight(inode);
590  case SymbolicExpression::OP_XOR: {
591  // Z3 doesn't have an XOR operator for Boolean arguments, only for bit vectors. Therefore, we cast
592  // the arguments to bit vectors and do a bit-wise XOR. We can leave the result a bit vector because
593  // if a Boolean result is needed, the caller will cast the result to Boolean.
594  Etv children = ctxCast(ctxExpressions(inode->children()), BIT_VECTOR);
595  ASSERT_require(children.size() >= 2);
596  z3expr = children[0].first;
597  for (size_t i = 1; i < children.size(); ++i)
598  z3expr = z3expr ^ children[i].first;
599  return Z3ExprTypePair(z3expr, BIT_VECTOR);
600  }
601  case SymbolicExpression::OP_EQ: {
602  ASSERT_require(inode->nChildren() == 2);
603  Etv children = ctxExpressions(inode->children());
604  Type type = mostType(children);
605  children = ctxCast(children, type);
606  z3expr = children[0].first == children[1].first;
607  return Z3ExprTypePair(z3expr, BOOLEAN);
608  }
609  case SymbolicExpression::OP_CONCAT: {
610  Etv children = ctxCast(ctxExpressions(inode->children()), BIT_VECTOR);
611  ASSERT_forbid(children.empty());
612  z3expr = children[0].first;
613  for (size_t i = 1; i < children.size(); ++i)
614  z3expr = z3::concat(z3expr, children[i].first);
615  return Z3ExprTypePair(z3expr, BIT_VECTOR);
616  }
617  case SymbolicExpression::OP_CONVERT:
618  throw Exception("OP_CONVERT is not implemented yet");
619  case SymbolicExpression::OP_EXTRACT:
620  return ctxExtract(inode);
621  case SymbolicExpression::OP_INVERT: {
622  ASSERT_require(inode->nChildren() == 1);
623  Z3ExprTypePair child = ctxExpression(inode->childRaw(0));
624  if (BOOLEAN == child.second) {
625  z3expr = !child.first;
626  } else {
627  ASSERT_require(BIT_VECTOR == child.second);
628  z3expr = ~child.first;
629  }
630  return Z3ExprTypePair(z3expr, child.second);
631  }
632  case SymbolicExpression::OP_FP_ABS:
633  case SymbolicExpression::OP_FP_NEGATE:
634  case SymbolicExpression::OP_FP_ADD:
635  case SymbolicExpression::OP_FP_MUL:
636  case SymbolicExpression::OP_FP_DIV:
637  case SymbolicExpression::OP_FP_MULADD:
638  case SymbolicExpression::OP_FP_SQRT:
639  case SymbolicExpression::OP_FP_MOD:
640  case SymbolicExpression::OP_FP_ROUND:
641  case SymbolicExpression::OP_FP_MIN:
642  case SymbolicExpression::OP_FP_MAX:
643  case SymbolicExpression::OP_FP_LE:
644  case SymbolicExpression::OP_FP_LT:
645  case SymbolicExpression::OP_FP_GE:
646  case SymbolicExpression::OP_FP_GT:
647  case SymbolicExpression::OP_FP_EQ:
648  case SymbolicExpression::OP_FP_ISNORM:
649  case SymbolicExpression::OP_FP_ISSUBNORM:
650  case SymbolicExpression::OP_FP_ISZERO:
651  case SymbolicExpression::OP_FP_ISINFINITE:
652  case SymbolicExpression::OP_FP_ISNAN:
653  case SymbolicExpression::OP_FP_ISNEG:
654  case SymbolicExpression::OP_FP_ISPOS:
655  throw Exception("floating point operations are not implemented yet");
656  case SymbolicExpression::OP_ITE: {
657  ASSERT_require(inode->nChildren() == 3);
658  Etv alternatives;
659  alternatives.push_back(ctxExpression(inode->childRaw(1)));
660  alternatives.push_back(ctxExpression(inode->childRaw(2)));
661  Type type = mostType(alternatives);
662  alternatives = ctxCast(alternatives, type);
663  z3expr = z3::ite(ctxCast(ctxExpression(inode->childRaw(0)), BOOLEAN).first,
664  alternatives[0].first,
665  alternatives[1].first);
666  return Z3ExprTypePair(z3expr, type);
667  }
668  case SymbolicExpression::OP_LET:
669  throw Exception("OP_LET not implemented");
670  case SymbolicExpression::OP_LSSB:
671  throw Exception("OP_LSSB not implemented");
672  case SymbolicExpression::OP_MSSB:
673  throw Exception("OP_MSSB not implemented");
674  case SymbolicExpression::OP_NE: {
675  ASSERT_require(inode->nChildren() == 2);
676  Etv children = ctxExpressions(inode->children());
677  Type type = mostType(children);
678  children = ctxCast(children, type);
679  z3expr = children[0].first != children[1].first;
680  return Z3ExprTypePair(z3expr, BOOLEAN);
681  }
682  case SymbolicExpression::OP_NEGATE: {
683  ASSERT_require(inode->nChildren() == 1);
684  Z3ExprTypePair child = ctxCast(ctxExpression(inode->childRaw(0)), BIT_VECTOR);
685  z3expr = -child.first;
686  return Z3ExprTypePair(z3expr, BIT_VECTOR);
687  }
688  case SymbolicExpression::OP_NOOP:
689  holdingExprs_.push_back(SymbolicExpression::makeIntegerConstant(inode->nBits(), 0));
690  return ctxExpression(holdingExprs_.back().getRawPointer());
691  case SymbolicExpression::OP_OR: {
692  ASSERT_require(inode->nChildren() >= 2);
693  Etv children = ctxExpressions(inode->children());
694  Type type = mostType(children);
695  children = ctxCast(children, type);
696  if (BOOLEAN == type) {
697  z3expr = children[0].first;
698  for (size_t i = 1; i < children.size(); ++i)
699  z3expr = z3expr || children[i].first;
700  } else {
701  ASSERT_require(BIT_VECTOR == type);
702  z3expr = children[0].first;
703  for (size_t i = 1; i < children.size(); ++i)
704  z3expr = z3expr | children[i].first;
705  }
706  return Z3ExprTypePair(z3expr, type);
707  }
708  case SymbolicExpression::OP_READ:
709  return ctxRead(inode);
710  case SymbolicExpression::OP_REINTERPRET:
711  throw Exception("OP_REINTERPRET not implemented");
712  case SymbolicExpression::OP_ROL:
713  return ctxRotateLeft(inode);
714  case SymbolicExpression::OP_ROR:
715  return ctxRotateRight(inode);
716  case SymbolicExpression::OP_SDIV:
717  return ctxSignedDivide(inode);
718  case SymbolicExpression::OP_SET:
719  return ctxSet(inode);
720  case SymbolicExpression::OP_SEXTEND:
721  return ctxSignExtend(inode);
722  case SymbolicExpression::OP_SLT:
723  ASSERT_require(inode->nChildren() == 2);
724  z3expr = (ctxCast(ctxExpression(inode->childRaw(0)), BIT_VECTOR).first <
725  ctxCast(ctxExpression(inode->childRaw(1)), BIT_VECTOR).first);
726  return Z3ExprTypePair(z3expr, BOOLEAN);
727  case SymbolicExpression::OP_SLE:
728  ASSERT_require(inode->nChildren() == 2);
729  z3expr = (ctxCast(ctxExpression(inode->childRaw(0)), BIT_VECTOR).first <=
730  ctxCast(ctxExpression(inode->childRaw(1)), BIT_VECTOR).first);
731  return Z3ExprTypePair(z3expr, BOOLEAN);
732  case SymbolicExpression::OP_SHL0:
733  return ctxShiftLeft(inode);
734  case SymbolicExpression::OP_SHL1:
735  return ctxShiftLeft(inode);
736  case SymbolicExpression::OP_SHR0:
737  return ctxShiftRight(inode);
738  case SymbolicExpression::OP_SHR1:
739  return ctxShiftRight(inode);
740  case SymbolicExpression::OP_SGE:
741  ASSERT_require(inode->nChildren() == 2);
742  z3expr = (ctxCast(ctxExpression(inode->childRaw(0)), BIT_VECTOR).first >=
743  ctxCast(ctxExpression(inode->childRaw(1)), BIT_VECTOR).first);
744  return Z3ExprTypePair(z3expr, BOOLEAN);
745  case SymbolicExpression::OP_SGT:
746  ASSERT_require(inode->nChildren() == 2);
747  z3expr = (ctxCast(ctxExpression(inode->childRaw(0)), BIT_VECTOR).first >
748  ctxCast(ctxExpression(inode->childRaw(1)), BIT_VECTOR).first);
749  return Z3ExprTypePair(z3expr, BOOLEAN);
750  case SymbolicExpression::OP_SMOD:
751  return ctxSignedModulo(inode);
752  case SymbolicExpression::OP_SMUL:
753  return ctxMultiply(inode);
754  case SymbolicExpression::OP_UDIV:
755  return ctxUnsignedDivide(inode);
756  case SymbolicExpression::OP_UEXTEND:
757  return ctxUnsignedExtend(inode);
758  case SymbolicExpression::OP_UGE:
759  ASSERT_require(inode->nChildren() == 2);
760  z3expr = z3::uge(ctxCast(ctxExpression(inode->childRaw(0)), BIT_VECTOR).first,
761  ctxCast(ctxExpression(inode->childRaw(1)), BIT_VECTOR).first);
762  return Z3ExprTypePair(z3expr, BOOLEAN);
763  case SymbolicExpression::OP_UGT:
764  ASSERT_require(inode->nChildren() == 2);
765  z3expr = z3::ugt(ctxCast(ctxExpression(inode->childRaw(0)), BIT_VECTOR).first,
766  ctxCast(ctxExpression(inode->childRaw(1)), BIT_VECTOR).first);
767  return Z3ExprTypePair(z3expr, BOOLEAN);
768  case SymbolicExpression::OP_ULE:
769  ASSERT_require(inode->nChildren() == 2);
770  z3expr = z3::ule(ctxCast(ctxExpression(inode->childRaw(0)), BIT_VECTOR).first,
771  ctxCast(ctxExpression(inode->childRaw(1)), BIT_VECTOR).first);
772  return Z3ExprTypePair(z3expr, BOOLEAN);
773  case SymbolicExpression::OP_ULT:
774  ASSERT_require(inode->nChildren() == 2);
775  z3expr = z3::ult(ctxCast(ctxExpression(inode->childRaw(0)), BIT_VECTOR).first,
776  ctxCast(ctxExpression(inode->childRaw(1)), BIT_VECTOR).first);
777  return Z3ExprTypePair(z3expr, BOOLEAN);
778  case SymbolicExpression::OP_UMOD:
779  return ctxUnsignedModulo(inode);
780  case SymbolicExpression::OP_UMUL:
781  return ctxMultiply(inode);
782  case SymbolicExpression::OP_WRITE:
783  return ctxWrite(inode);
784  case SymbolicExpression::OP_ZEROP:
785  return ctxZerop(inode);
786  }
787  }
788  ASSERT_not_reachable("unhandled case " + stringify::Rose::BinaryAnalysis::SymbolicExpression::Operator(inode->getOperator(), "") +
789  " (" + boost::lexical_cast<std::string>(inode->getOperator()) + ")");
790 }

1200 Z3Solver::parseEvidence() {
1201  if (linkage() != LM_LIBRARY)
1202  return SmtlibSolver::parseEvidence();
1203 
1204 #ifdef ROSE_HAVE_Z3
1205  try {
1206  Sawyer::Stopwatch evidenceTimer;
1207 
1208  // If there are no assertions, then there is no evidence.
1209  bool hasAssertions = false;
1210  for (size_t i=0; i<z3Stack_.size() && !hasAssertions; ++i)
1211  hasAssertions = !z3Stack_[i].empty();
1212  if (!hasAssertions)
1213  return;
1214 
1215  // Get all the variables in all the assertions regardless of transaction level. This is somewhat expensive but is needed
1216  // below because the Z3 interface lacks a way to get type information from the variables returned as part of the evidence.
1217  VariableSet allVariables;
1218  std::vector<SymbolicExpression::Ptr> allAssertions = assertions();
1219  for (const SymbolicExpression::Ptr &expr: allAssertions)
1220  findVariables(expr, allVariables /*in,out*/);
1221 
1222  // Parse the evidence
1223  ASSERT_not_null(solver_);
1224  z3::model model = solver_->get_model();
1225  for (size_t i=0; i<model.size(); ++i) {
1226  z3::func_decl fdecl = model[i];
1227 
1228  if (fdecl.arity() != 0)
1229  continue;
1230 
1231  // There's got to be a better way to get information about a z3::expr, but I haven't found it yet. For bit vectors, we
1232  // need to know the number of bits and the value, even if the value is wider than 64 bits. Therefore, we obtain the list
1233  // of all variables from above (regardless of transaction level) and try to match of the z3 variable name with the ROSE
1234  // variable name and obtain the type information from the ROSE variable.
1235  SymbolicExpression::LeafPtr var;
1236  for (const SymbolicExpression::LeafPtr &v: allVariables.values()) {
1237  if (v->toString() == fdecl.name().str()) {
1238  var = v;
1239  break;
1240  }
1241  }
1242  if (NULL == var) {
1243  mlog[WARN] <<"cannot find evidence variable " <<fdecl.name() <<"\n";
1244  continue;
1245  }
1246 
1247  // Get the value
1248  SymbolicExpression::Ptr val;
1249  z3::expr interp = model.get_const_interp(fdecl);
1250  if (interp.is_bv()) {
1251  if (var->nBits() <= 64) {
1252  val = SymbolicExpression::makeIntegerConstant(var->nBits(), interp.get_numeral_uint64());
1253  } else {
1254  // Z3 doesn't have an API function to obtain the bits of a constant if the constant is wider than 64 bits. We
1255  // can't use the trick of splitting the Z3 bit vector into 64-bit chunks and converting each to a uint64_t and
1256  // concatenating the results because "bv.extract(hi,lo)" is no longer a number but rather an extraction
1257  // expression. What we can do is get a string of decimal digits and parse it to binary. This isn't
1258  // particularly efficient since the conversions to/from string require divide/multiply by 10.
1259  std::string digits = interp.get_decimal_string(0);
1260  Sawyer::Container::BitVector bits(var->nBits());
1261  bits.fromDecimal(bits.hull(), digits);
1262  val = SymbolicExpression::makeIntegerConstant(bits);
1263  ASSERT_require(val->nBits() == var->nBits());
1264  ASSERT_require(val->isLeafNodeRaw() && val->isLeafNodeRaw()->isIntegerConstant());
1265  }
1266  } else if (interp.is_bool()) {
1267  val = SymbolicExpression::makeBooleanConstant(interp.get_numeral_uint() != 0);
1268  } else {
1269  mlog[WARN] <<"cannot parse evidence expression for " <<*var <<"\n";
1270  continue;
1271  }
1272 
1273  ASSERT_not_null(var);
1274  ASSERT_not_null(val);
1275  evidence_.insert(var, val);
1276  }
1277 
1278  stats.evidenceTime += evidenceTimer.stop();
1279  stats.longestEvidenceTime = std::max(stats.longestEvidenceTime, evidenceTimer.report());
1280  } catch (const z3::exception &e) {
1281  throw Exception("Z3 failure: " + std::string(e.msg()));
1282  }
1283 #else
1284  ASSERT_not_reachable("z3 not enabled");
1285 #endif
1286 }
