847  * Taken from Glassner's book, section 5.2 (Heckberts method)

849 template<class Scalar>

852 {
853  //double nair = mpGlob->getRefIndex();
854  //double nt = myRefIndex;
855  double eta = nair / nt;
856  double n = -((*this) | normal);
857  double t = 1.0 + eta*eta* (n*n-1.0);
858  if(t<0.0) {
859  // we have total reflection!
860  refRefl = 1;
861  } else {
862  // normal reflection
863  t = eta*n - sqrt(t);

865  }

974  * Assume this vector is HSV and convert to RGB
975  */
976 template<class Scalar>
977 inline void
978 ntlVector3Dim<Scalar>::hsvToRgb( void )
979 {
980  Scalar h = value[0], s = value[1], v = value[2];
981  Scalar r=0,g=0,b=0;
982  Scalar p,q,t, fracth;
983  int floorh;
984  // ...and back to rgb
985  if(s == 0) {
986  r = g = b = v; }
987  else {
988  h /= 60.0;
989  floorh = (int)h;
990  fracth = h - floorh;
991  p = v * (1.0 - s);
992  q = v * (1.0 - (s * fracth));
993  t = v * (1.0 - (s * (1.0 - fracth)));
994  switch (floorh) {
995  case 0: r = v; g = t; b = p; break;
996  case 1: r = q; g = v; b = p; break;
997  case 2: r = p; g = v; b = t; break;
998  case 3: r = p; g = q; b = v; break;
999  case 4: r = t; g = p; b = v; break;
1000  case 5: r = v; g = p; b = q; break;
1001  }
1002  }
1003 
1004  value[0] = r;
1005  value[1] = g;
1006  value[2] = b;
1007 }
