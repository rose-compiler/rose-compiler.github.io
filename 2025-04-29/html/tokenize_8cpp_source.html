36 POETCode* EvaluatePOET:: apply_tokens(POETCode* tokens, POETList* input)
37 {
38  if (tokens != 0)
39  {
40  POETList* p_tokens=dynamic_cast<POETList*>(tokens);
41  POETCode* input1 = 0;
42  POETCode* first_token=0;
43  /* there is only a single token specifiation*/
44  if (p_tokens == 0 || p_tokens->get_first()->get_enum() == SRC_STRING)
45  first_token = apply_tokenFilter(tokens,input, input1);
46  else { /* there is a list of token specs. */
47  for ( ; first_token == 0 && p_tokens != 0; p_tokens = p_tokens->get_rest())
48  first_token = apply_tokenFilter(p_tokens->get_first(),input,input1);
49  }
50  if (first_token==0) return input;
51  if (first_token != EMPTY)
52  {
53  if (debug_lex())
54  std::cerr << "recognized token: " << first_token->toString() << "\n";
55  POETCode* res = make_inputlist(first_token,input1);
56  return res;
57  }
58  if (debug_lex())
59  std::cerr << "leftover after empty token" << ((input1==0)? "NULL" : input1->toString()) << "\n";
60  return input1;
61  }
62  return input;
63 }

65 POETCode* EvaluatePOET::
66 apply_tokenFilter(POETCode* pattern, POETCode* input,
67  POETCode*& leftOver, bool make_string)
68 {
69  if (pattern == EMPTY) { leftOver=input; return pattern; }
70  if (debug_lex())
71  std::cerr << "apply token filter: " << pattern->toString() << " to " << ((input==0)? "" : input->toString()) << "\n";
72 
73  leftOver = input;
74  switch (pattern->get_enum())
75  {
76  case SRC_STRING: {
77  if (input == 0) return 0;
78  POETCode* cur_input = get_head(input);
79  if (cur_input->get_enum() != SRC_STRING) return 0;
80  std::string str_pat = static_cast<POETString*>(pattern)->get_content();
81  std::string str_input = static_cast<POETString*>(cur_input)->get_content();
82  size_t pos = str_input.find(str_pat);
83  if (pos == 0) {
84  pos = str_pat.size();
85  char nextchar = str_input[pos];
86  if (str_input.size() == pos) {
87  leftOver=get_tail(input);
88  return pattern;
89  }
90  else {
91  POETCode* new_str = ASTFactory::inst()->new_string(str_input.substr(pos,str_input.size()-pos));
92  leftOver=make_inputlist(new_str,get_tail(input));
93  return pattern;
94  }
95  }
96  return 0;
97  }
98  case SRC_ICONST:
99  if (input != 0 && get_head(input) == pattern) {
100  leftOver = get_tail(input);
101  return pattern;
102  }
103  return 0;
104  case SRC_TYPE: {
105  if (input != 0) {
106  POETCode* cur_token = get_head(input);
107  if (match_Type(cur_token, static_cast<POETType*>(pattern), true))
108  leftOver = get_tail(input);
109  else cur_token=0;
110  return cur_token;
111  }
112  return 0;
113  }
114  case SRC_LVAR: {
115  LvarSymbolTable::Entry entry = static_cast<LocalVar*>(pattern)->get_entry();
116  POETCode* restr = entry.get_restr();
117  if (restr == 0) { SYM_UNDEFINED(pattern->toString()); }
118  POETCode* cur_token = apply_tokenFilter(restr, input, leftOver);
119  if (cur_token != 0) { entry.set_code(cur_token); }
120  return cur_token;
121  }
122  case SRC_LIST: {
123  POETCode* p_input = input;
124  POETList* p_content = static_cast<POETList*>(pattern);
125  std::string token_content;
126  std::vector<POETCode*> match_res;
127  for ( ; p_content != 0; p_content=p_content->get_rest()) {
128  POETCode* cur_token = apply_tokenFilter(p_content->get_first(), p_input, leftOver,make_string);
129  if (cur_token == 0) return 0;
130  p_input=leftOver;
131  match_res.push_back(cur_token);
132  if (cur_token->get_enum() != SRC_STRING)
133  make_string=false;
134  else token_content=token_content+static_cast<POETString*>(cur_token)->get_content();
135  }
136  if (make_string) return fac->new_string(token_content);
137  if (match_res.size()==0) return 0;
138  return Vector2List(match_res);
139  }
140  case SRC_CVAR: {
141  POETCode* first = get_head(input);
142  CodeVar* cvar = static_cast<CodeVar*>(pattern);
143  if (first != 0 && first->get_enum() == SRC_CVAR && static_cast<CodeVar*>(first)->get_entry() == cvar->get_entry()) {
144  leftOver=get_tail(input);
145  return first;
146  }
147  POETCode* pars=cvar->get_entry().get_param();
148  POETCode* code=cvar->get_entry().get_code();
149  if (code == 0) code=cvar->get_entry().get_parse();
150  if (code == 0) CODE_SYNTAX_UNDEFINED(cvar->toString());
151  try {
152  if (pars != 0) cvar->get_entry().get_symTable()->push_table(false);
153  POETCode* cur_token = apply_tokenFilter(code, input, leftOver, false);
154  if (cur_token != 0) {
155  if (debug_lex())
156 std::cerr << "found code template: " << cur_token->toString(DEBUG_OUTPUT) << "\n";
157  if (pars != 0) { pars=eval_AST(pars); }
158  cur_token= cvar->invoke_rebuild(pars);
159  if (cur_token == 0) { cur_token = fac->build_codeRef(cvar->get_entry(),pars); }
160  }
161  if (pars != 0) cvar->get_entry().get_symTable()->pop_table();
162  return cur_token;
163  }
164  catch (Error err) { std::cerr << "From recognizing token: " << cvar->toString() << "\n"; throw err; }
165  }
166  case SRC_OP: {
167  POETOperator *op = static_cast<POETOperator*>(pattern);
168  switch (op->get_op()) {
169  case POET_OP_CONCAT:
170  if (input != 0) {
171  POETCode* arg1 = op->get_arg(0), *arg2=op->get_arg(1);
172  if (arg1->get_enum() == SRC_STRING)
173  input=dynamic_cast<POETList*>(split_string(static_cast<POETString*>(arg1)->get_content(), input));
174  else if (arg2->get_enum() == SRC_STRING)
175  input=dynamic_cast<POETList*>(split_string(static_cast<POETString*>(arg2)->get_content(), input));
176  else LEX_INCORRECT(op);
177  POETCode* first=apply_tokenFilter(arg1,input,leftOver,make_string);
178  if (first == 0) { leftOver=input; return 0; }
179  POETCode* second=apply_tokenFilter(arg2,dynamic_cast<POETList*>(leftOver),leftOver,make_string);
180  if (second == 0) { leftOver=input; return 0; }
181  return fac->new_string(first->toString() + second->toString());
182  }
183  return 0;
184  case POET_OP_RANGE:
185  if (input != 0) {
186  POETCode* cur_token = get_head(input);
187  if (match_AST(cur_token, pattern, MATCH_AST_PATTERN))
188  leftOver = get_tail(input);
189  else cur_token=0;
190  return cur_token;
191  }
192  return 0;
193  case TYPE_LIST1:
194  case TYPE_LIST: {
195  POETCode* arg = op->get_arg(0);
196  std::vector<POETCode*> match_res;
197  do {
198  POETCode* r1_first=apply_tokenFilter(arg,input,leftOver,make_string);
199  if (r1_first == 0) { leftOver=input; break; }
200  match_res.push_back(r1_first);
201  input = dynamic_cast<POETList*>(leftOver);
202  } while (input != 0);
203  if (op->get_op() == TYPE_LIST1 && match_res.size()==0)
204  return 0;
205  return Vector2List(match_res);
206  }
207  case TYPE_TOR: {
208  for (unsigned i = 0; i < op->numOfArgs(); ++i)
209  {
210  POETCode* arg=op->get_arg(i);
211  POETCode* res=apply_tokenFilter(arg,input,leftOver,make_string);
212  if (res != 0) return res;
213  leftOver=input;
214  }
215  }
216  return 0;
217  default: LEX_INCORRECT(pattern);
218  }
219  }
220  default:
221  LEX_INCORRECT(pattern);
222  }
223 }

225 POETCode* EvaluatePOET::
226 eval_readInput(POETCode* inputFiles, POETCode* codeType, POETCode* inputInline)
227 {
228  POETCode* res = 0;
229  if (inputFiles != 0) {
230  std::list<POETProgram*> inputPrograms;
231  ReadFiles(eval_AST(inputFiles), inputPrograms);
232  std::list<POETProgram*>::reverse_iterator p = inputPrograms.rbegin();
233  for ( ; p != inputPrograms.rend(); ++p) {
234  POETProgram* prog = *p;
235  POETProgram::const_iterator p=prog->begin();
236  if (p != prog->end()) {
237  POETCode* eval = *p;
238  eval = parse_input( eval, codeType);
239  assert(eval != 0);
240  if (res == 0) res = eval;
241  else res= make_inputlist(eval, res);
242  }
243  }
244  }
245  if (inputInline != 0) {
246  inputInline = parse_input( inputInline, codeType);
247  if (!res) res = inputInline;
248  else res = ASTFactory::inst()->new_list(res, inputInline);
249  }
250  return res;
251 }
