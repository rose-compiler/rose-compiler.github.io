30 // Approx number of tasks to create for each worker thread. The finest granularity of work (a single comparison between two
31 // functions) is often not the most efficient way to schedule worker threads because if the comparisons are cheap then the
32 // scheduling overhead accounts for a higher percentage of the total time. The opposite exteme of one task per thread, is also
33 // not always efficient because the length of the tasks can be quite different from one another. A compromise is to create some
34 // fixed number of tasks per worker thread so the individual tasks are larger, but there's still enough of them to balance the
35 // work load.
36 static const size_t tasksPerWorker = 100; // arbitrary

70 std::vector<size_t>
71 FunctionSimilarity::findMinimumAssignment(const DistanceMatrix &matrix) {
72 #ifdef ROSE_HAVE_DLIB
73  ASSERT_forbid(matrix.size() == 0);
74  ASSERT_require(matrix.nr() == matrix.nc());
75 
76  // We can avoid the O(n^3) Kuhn-Munkres algorithm if all values of the matrix are the same.
77  double minValue, maxValue;
78  dlib::find_min_and_max(matrix.dlib(), minValue /*out*/, maxValue /*out*/);
79  if (minValue == maxValue) {
80  std::vector<size_t> ident;
81  ident.reserve(matrix.nr());
82  for (size_t i=0; i<matrix.nr(); ++i)
83  ident.push_back(i);
84  return ident;
85  }
86 
87  // Dlib's Kuhn-Munkres finds the *maximum* mapping over *integers*, so we negate everything to find the minumum, and we map
88  // the doubles onto a reasonably large interval of integers. The interval should be large enough to have some precision,
89  // but not so large that things might overflow.
90  const int iGreatest = 1000000; // arbitrary upper bound for integer interval
91  dlib::matrix<long> intMatrix(matrix.nr(), matrix.nc());
92  for (size_t i=0; i<matrix.nr(); ++i) {
93  for (size_t j=0; j<matrix.nc(); ++j)
94  intMatrix(i, j) = round(-iGreatest * (matrix(i, j) - minValue) / (maxValue - minValue));
95  }
96 
97  // Return a proper type -- indexes and sizes should never be signed types since it makes no sense for them to be negative.
98  std::vector<size_t> retval;
99  retval.reserve(matrix.nr());
100  for (double d: dlib::max_cost_assignment(intMatrix))
101  retval.push_back(d);
102  return retval;
103 #else
104  ROSE_UNUSED(matrix);
105  throw FunctionSimilarity::Exception("dlib support is necessary for FunctionSimilarity analysis"
106  "; see ROSE installation instructions");
107 #endif
108 }

123 // Combine some values into a single value
124 double
125 combine(FunctionSimilarity::Statistic s, const std::vector<double> &values) {
126  ASSERT_forbid(values.empty());
127  switch (s) {
128  case FunctionSimilarity::AVERAGE: {
129  double sum = 0.0;
130  for (double v: values)
131  sum += v;
132  return sum / values.size();
133  }
134  case FunctionSimilarity::MAXIMUM: {
135  return *std::max_element(values.begin(), values.end());
136  }
137  case FunctionSimilarity::MEDIAN: {
138  std::vector<double> tmp = values;
139  std::nth_element(tmp.begin(), tmp.begin() + tmp.size()/2, tmp.end());
140  double retval = tmp[tmp.size()/2];
141  if (0 == values.size() % 2) {
142  std::nth_element(tmp.begin(), tmp.begin() + tmp.size()/2 + 1, tmp.end());
143  retval = (retval + tmp[tmp.size()/2 + 1]) / 2.0;
144  }
145  return retval;
146  }
147  }
148  ASSERT_not_reachable("invalid statistic");
149 }

256  ASSERT_require(f1 != NULL || f2 != NULL);
257  std::vector<double> categoryDistances;
258 
259  // If only one function is supplied, then compare it against no-function (i.e., use the distance of the non-null function
260  // from the origin).
261  const FunctionInfo empty;
262  const FunctionInfo &finfo1 = f1 ? functions_[f1] : empty;
263  const FunctionInfo &finfo2 = f2 ? functions_[f2] : empty;
264 
265  for (CategoryId id=0; id<categories_.size(); ++id) {
266  double d = NAN;
267  switch (categories_[id].kind) {
268  case CARTESIAN_POINT: {
269  static const PointCloud emptyCloud;
270  const PointCloud &cd1 = id < finfo1.categories.size() ? finfo1.categories[id].pointCloud : emptyCloud;
271  const PointCloud &cd2 = id < finfo2.categories.size() ? finfo2.categories[id].pointCloud : emptyCloud;
272  d = comparePointClouds(cd1, cd2);
273  break;
274  }
275  case ORDERED_LIST: {
276  static const OrderedLists emptyLists;
277  const OrderedLists &cd1 = id < finfo1.categories.size() ? finfo1.categories[id].orderedLists : emptyLists;
278  const OrderedLists &cd2 = id < finfo2.categories.size() ? finfo2.categories[id].orderedLists : emptyLists;
279  d = compareOrderedLists(cd1, cd2);
280  break;
281  }
282  }
283  ASSERT_require(!rose_isnan(d));
284  categoryDistances.push_back(d * categories_[id].weight);
285  }
286  return combine(categoryAccumulatorType_, categoryDistances);

314  const std::vector<P2::Function::Ptr> &colFunctions; // functions for each column of the matrix
315  const size_t matrixSize; // number of rows and columns in square matrix
316  Progress::Ptr progress;

318  const double dfltCompare; // distance between functions when either has no data
319 
320  ComparisonFunctor(const FunctionSimilarity *self,
321  const std::vector<P2::Function::Ptr> &rowFunctions,
322  const std::vector<P2::Function::Ptr> &colFunctions,
323  const Progress::Ptr &progress, Sawyer::ProgressBar<size_t> &progressBar,
324  double dfltCompare)
325  : self(self), rowFunctions(rowFunctions), colFunctions(colFunctions),
326  matrixSize(std::max(rowFunctions.size(), colFunctions.size())),
327  progress(progress), progressBar(progressBar), dfltCompare(dfltCompare) {}
328 
329  void operator()(size_t /*taskId*/, const ComparisonTask &task) {
330  ASSERT_require(task.nComparisons > 0);
331  ASSERT_not_null(task.results);
332  ASSERT_require(task.startRow < matrixSize);
333  ASSERT_require(task.startCol < matrixSize);
334  ASSERT_require(task.startRow * matrixSize + task.startCol + task.nComparisons <= matrixSize * matrixSize);
335 
336  size_t i = task.startRow;
337  size_t j = task.startCol;
338  for (size_t k=0; k<task.nComparisons; ++k) {
339  P2::Function::Ptr a = i < rowFunctions.size() ? rowFunctions[i] : P2::Function::Ptr();
340  P2::Function::Ptr b = j < colFunctions.size() ? colFunctions[j] : P2::Function::Ptr();
341  task.results[k] = self->compare(a, b, dfltCompare);
342  if (++j == matrixSize) {
343  ++i;
344  j = 0;
345  }
346  }
347  progressBar.increment(task.nComparisons);
348  progress->update(progressBar.ratio());
349  }
350 };

394 // Divide a large problem into parallelizable sub-problems. The large problem is to compare all functions in rowFunctions with
395 // all functions in colFunctions and return the distance between the functions of each pair. The problem is divided among the
396 // specified number of threads, which must be positive. For load balancing purposes, the number of tasks will be larger than
397 // the number of threads. The return value is the vector into which the tasks will write their results, and can be treated as a
398 // row-major vector. In other words, the distance between rowFunction[i] and colFunction[j] will be written to the return
399 // vector at index i*colFunction.size()+j. */
400 static std::vector<double>
401 buildTasks(const std::vector<P2::Function::Ptr> &rowFunctions, const std::vector<P2::Function::Ptr> &colFunctions,
402  size_t nThreads, ComparisonTasks &tasks /*in,out*/) {
403  ASSERT_require(nThreads > 0); // caller must already know number of threads
404 
405  // Allocate the results to be populated by the tasks
406  const size_t totalComparisons = rowFunctions.size() * colFunctions.size();
407  std::vector<double> results(totalComparisons, NAN);
408 
409  // Decide how many tasks and how much work per task.
410  const size_t nTasks = nThreads > 1 ? nThreads * tasksPerWorker : (size_t)1;
411  const size_t comparisonsPerTask = (totalComparisons + nTasks - 1) / nTasks;
412 
413  // Divvy up the matrix among the tasks.
414  for (size_t i = 0; i < totalComparisons; i += comparisonsPerTask) {
415  size_t startRow = i / colFunctions.size();
416  size_t startCol = i % colFunctions.size();
417  size_t n = std::min(comparisonsPerTask, totalComparisons-i);
418  ComparisonTask task(startRow, startCol, n, &(results[i]));
419  tasks.insertVertex(task);
420  }
421 
422  return results;
423 }

522 std::vector<FunctionSimilarity::FunctionPair>
523 FunctionSimilarity::findMinimumCostMapping(const std::vector<P2::Function::Ptr> &list1,
524  const std::vector<P2::Function::Ptr> &list2) const {
525  Sawyer::Message::Stream where = mlog[WHERE], debug = mlog[DEBUG];
526  SAWYER_MESG(where) <<"minimum mapping between " <<StringUtility::plural(list1.size(), "functions")
527  <<" and " <<StringUtility::plural(list2.size(), "functions") <<"\n";
528  Sawyer::Stopwatch stopwatch;
529 
530  DistanceMatrix dm = compareManyToManyMatrix(list1, list2);
531  ASSERT_require(dm.nr() == dm.nc());
532  size_t n = dm.nr();
533 
534  // Find the minimum total cost 1:1 assignment of list1 functions (rows) to list2 functions (cols)
535  debug <<"starting Kuhn-Munkres";
536  std::vector<size_t> assignment = findMinimumAssignment(dm);
537  ASSERT_require(assignment.size() == n);
538  debug <<"; done\n";

541  // least user code compiles without having to do anything special.

543  retval.reserve(n);
544  for (size_t i=0; i<n; ++i) {
545  size_t j = assignment[i];
546  retval.push_back(FunctionPair(i < list1.size() ? list1[i] : P2::Function::Ptr(),
547  j < list2.size() ? list2[j] : P2::Function::Ptr()));
548  }

625 FunctionSimilarity::measureCfgConnectivity(CategoryId id, const P2::Partitioner::ConstPtr &partitioner,
626  const P2::Function::Ptr &function, size_t maxPoints) {
627  ASSERT_not_null(partitioner);
628  size_t nPoints = 0;
629  for (Address bbva: function->basicBlockAddresses()) {
630  CartesianPoint point;
631  P2::ControlFlowGraph::ConstVertexIterator vertex = partitioner->findPlaceholder(bbva);
632  if (partitioner->cfg().isValidVertex(vertex)) {
633  if (++nPoints > maxPoints) {
634  mlog[WARN] <<"cfg connectivity truncated at " <<StringUtility::plural(maxPoints, "vertices")
635  <<" for " <<function->printableName() <<"\n";
636  break;
637  }
638 
639  // Direct neighbors
640  point.push_back(normalizedNumberOfNeighbors(vertex->outEdges(), forward));
641  point.push_back(normalizedNumberOfNeighbors(vertex->inEdges(), reverse));
642 
643  // Second-level neighbors
644  double totalForward = 0.0, totalReverse = 0.0;
645  for (const P2::ControlFlowGraph::Edge &edge: vertex->outEdges())
646  totalForward = normalizedNumberOfNeighbors(edge.target()->outEdges(), forward);
647  for (const P2::ControlFlowGraph::Edge &edge: vertex->inEdges())
648  totalReverse = normalizedNumberOfNeighbors(edge.source()->inEdges(), reverse);
649  point.push_back(vertex->nOutEdges() ? totalForward/vertex->nOutEdges() : 0.0);
650  point.push_back(vertex->nInEdges() ? totalReverse/vertex->nInEdges() : 0.0);
651 
652  insertPoint(function, id, point);
653  }
654  }
