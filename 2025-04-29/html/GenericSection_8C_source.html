525 Address
526 SgAsmGenericSection::write(std::ostream &f, Address offset, size_t bufsize, const void *buf) const
527 {
528  size_t nwrite;
529 
530  /* Don't write past end of section */
531  if (offset>=get_size()) {
532  nwrite = 0;
533  } else if (offset+bufsize<=get_size()) {
534  nwrite = bufsize;
535  } else {
536  nwrite = get_size() - offset;
537  }
538 
539  /* Don't write past end of current EOF if we can help it. */
540  f.seekp(0, std::ios::end);
541  Address filesize = f.tellp();
542  while (nwrite>0 && 0==((const char*)buf)[nwrite-1] && get_offset()+offset+nwrite>filesize)
543  --nwrite;
544 
545  /* Write bytes to file. This is a good place to set a break point if you're trying to figure out what section is writing
546  * to a particular file address. For instance, if byte 0x7c is incorrect in the unparsed file you would set a conditional
547  * breakpoint for o<=0x7c && o+nwrite>0x7c */
548  off_t o = get_offset() + offset;
549  f.seekp(o);
550  f.write((const char*)buf, nwrite);
551 
552  /* Check that truncated data is all zero and fail if it isn't */
553  for (size_t i=nwrite; i<bufsize; i++) {
554  if (((const char*)buf)[i]) {
555  char mesg[1024];
556  snprintf(mesg, 1024, "non-zero value truncated: buf[0x%zx]=0x%02x", i, ((const unsigned char*)buf)[i]);
557  mlog[ERROR] <<"SgAsmGenericSection::write: error: " <<mesg
558  <<" in [" <<get_id() <<"] \"" <<get_name()->get_string(true) <<"\"\n"
559  <<" section is at file offset " <<StringUtility::addrToString(get_offset())
560  <<" size " <<StringUtility::plural(get_size(), "bytes") <<"\n"
561  <<" write " <<StringUtility::plural(bufsize, "bytes")
562  <<" at section offset " <<StringUtility::addrToString(offset) <<"\n";
563  HexdumpFormat hf;
564  hf.prefix = " ";
565  hexdump(mlog[ERROR], get_offset()+offset, (const unsigned char*)buf, bufsize, hf);
566  mlog[ERROR] <<"\n";
567  throw SgAsmGenericFile::ShortWrite(this, offset, bufsize, mesg);
568  }
569  }
570 
571  return offset+bufsize;
572 }

634 SgAsmGenericSection::writeSleb128(unsigned char *buf, Address offset, int64_t val) const
635 {
636  if (val==0) {
637  buf[offset++] = 0;
638  } else if (val==-1) {
639  buf[offset++] = 0x7f;
640  } else {
641  while (val!=0 && val!=-1) {
642  unsigned char byte = (uint64_t)val & 0x7f;
643  val >>= 7; /*sign extending*/
644  if (val!=0 && val!=-1)
645  byte |= 0x80;
646  buf[offset++] = byte;
647  }

658 AddressIntervalSet
659 SgAsmGenericSection::get_referencedExtents() const
660 {
661  if (0==get_size())
662  return AddressIntervalSet();
663 
664  AddressIntervalSet retval;
665  AddressInterval segment = AddressInterval::baseSize(get_offset(), get_size());
666  const AddressIntervalSet &fileExtents = get_file()->get_referencedExtents();
667  for (const AddressInterval &interval : fileExtents.intervals()) {
668  if (segment.contains(interval)) {
669  retval.insert(AddressInterval::baseSize(interval.least()-get_offset(), interval.size()));
670  } else if (interval.isLeftOf(segment) || interval.isRightOf(segment)) {
671  // no overlap
672  } else if (interval.contains(segment)) {
673  retval.insert(AddressInterval::baseSize(0, get_size()));
674  break; // no point in continuing since we've referenced whole segment now
675  } else if (interval.least() < segment.least()) {
676  retval.insert(AddressInterval::baseSize(0, interval.least()+interval.size()-get_offset()));
677  } else if (interval.greatest() > segment.greatest()) {
678  retval.insert(AddressInterval::baseSize(interval.least()-get_offset(), get_offset()+get_size()-interval.least()));
679  } else {
680  ASSERT_not_reachable("invalid extent overlap category");
681  }
682  }
683  return retval;
684 }
