220 void
221 UnparseJovial::unparseJovialType(SgModifierType* modifier_type, SgUnparse_Info& info)
222  {
223  // An SgModifierType is used when an ItemTypeDescription has an initializer. The modifier type appears to
224  // be inserted after the parsing phase and I assume it is because the presence of an initializer causes the
225  // type to be wrapped with an SgModifierType with isConst and isStatic set. This required the creation of
226  // isJovialStatic to correctly unparse when the Jovial source actually has the STATIC keyword.
227 
228  ROSE_ASSERT(modifier_type);
229  SgType* base_type = modifier_type->get_base_type();
230  ROSE_ASSERT(base_type);
231 
232  // SgModifierType is also used to mark R,T,Z (round, truncate, truncate towards zero).
233  // If not used for (R,T,Z), unwrap and then unparse the base type (this should fix recurring problems).
234  if (modifier_type->get_typeModifier().isRound() ||
235  modifier_type->get_typeModifier().isTruncate() ||
236  modifier_type->get_typeModifier().isTruncateTowardsZero())
237  {
238  unparseTypeDesc(base_type, info);
239  if (modifier_type->get_typeModifier().isRound()) curprint(",R");
240  else if (modifier_type->get_typeModifier().isTruncate()) curprint(",T");
241  else if (modifier_type->get_typeModifier().isTruncateTowardsZero()) curprint(",Z");
242  unparseTypeSize(base_type, info);
243  }
244  else
245  {
246  unparseType(base_type, info);
247  }
248  }
