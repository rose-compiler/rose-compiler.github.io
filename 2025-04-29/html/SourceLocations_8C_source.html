158 SourceLocations::fillHoles(size_t maxHoleSize) {
159  SAWYER_THREAD_TRAITS::LockGuard lock(mutex_);
160  AddressToSource fill; // new entries that fill in holes
161  typedef AddressToSource::ConstNodeIterator Iter;
162  for (Iter left = addrToSrc_.nodes().begin(); left != addrToSrc_.nodes().end(); ++left) {
163  Iter right = left; ++right;
164 
165  // Find the hole
166  AddressInterval hole;
167  if (left->key().greatest() == AddressInterval::whole().greatest()) {
168  break;
169  } else if (right == addrToSrc_.nodes().end()) {
170  hole = AddressInterval::hull(left->key().greatest() + 1, AddressInterval::whole().greatest());
171  } else if (left->key().greatest() < right->key().least()) {
172  hole = AddressInterval::hull(left->key().greatest() + 1, right->key().least() - 1);
173  }
174 
175  // Fill the hole (delayed until we're done iterating
176  if (!hole.isEmpty() && hole.size() <= maxHoleSize)
177  fill.insert(hole, left->value());
178  }
179 
180  // Delayed filling of holes
181  addrToSrc_.insertMultiple(fill);
182  for (const AddressToSource::Node &node: fill.nodes())
183  srcToAddr_[node.value()] |= node.key();
