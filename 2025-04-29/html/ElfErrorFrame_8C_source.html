28 std::string
29 SgAsmElfEHFrameEntryCI::unparse(const SgAsmElfEHFrameSection *ehframe) const
30 {
31  SgAsmElfFileHeader *fhdr = ehframe->get_elfHeader();
32  ROSE_ASSERT(fhdr!=NULL);
33 
34  /* Allocate worst-case size for results */
35  size_t worst_size = (4+
36  1+
37  get_augmentation_string().size()+1+
38  10+
39  10+
40  10+
41  get_augmentation_data_length()+
42  get_instructions().size()+
43  fhdr->get_wordSize());
44  unsigned char *buf = new unsigned char[worst_size];
45 
46  Rose::BinaryAnalysis::Address at = 0;
47  uint32_t u32_disk;
48  unsigned char u8_disk;
49 
50  /* CIE back offset (always zero) */
51  u32_disk=0;
52  memcpy(buf+at, &u32_disk, 4); at+=4;
53 
54  /* Version */
55  u8_disk = get_version();
56  memcpy(buf+at, &u8_disk, 1); at+=1;
57 
58  /* NUL-terminated Augmentation String */
59  size_t sz = get_augmentation_string().size()+1;
60  memcpy(buf+at, get_augmentation_string().c_str(), sz); at+=sz;
61 
62  /* Alignment factors */
63  at = ehframe->writeUleb128(buf, at, get_code_alignment_factor());
64  at = ehframe->writeSleb128(buf, at, get_data_alignment_factor());
65 
66  /* Augmentation data */
67  at = ehframe->writeUleb128(buf, at, get_augmentation_data_length());
68  std::string astr = get_augmentation_string();
69  if (!astr.empty() && astr[0]=='z') {
70  for (size_t i=1; i<astr.size(); i++) {
71  if ('L'==astr[i]) {
72  u8_disk = get_lsda_encoding();
73  buf[at++] = u8_disk;
74  } else if ('P'==astr[i]) {
75  u8_disk = get_prh_encoding();
76  buf[at++] = u8_disk;
77  switch (get_prh_encoding()) {
78  case 0x05:
79  case 0x06:
80  case 0x07:
81  buf[at++] = get_prh_arg();
82  Rose::BinaryAnalysis::ByteOrder::hostToLe(get_prh_addr(), &u32_disk);
83  memcpy(buf+at, &u32_disk, 4); at+=4;
84  break;
85  default:
86  /* See parser */
87  if (++nwarnings<=WARNING_LIMIT) {
88  mlog[WARN] <<"unknown PRH encoding (" <<StringUtility::toHex2(get_prh_encoding(), 8) <<")\n";
89  if (WARNING_LIMIT==nwarnings)
90  mlog[WARN] <<"additional frame warnings will be suppressed\n";
91  }
92  break;
93  }
94  } else if ('R'==astr[i]) {
95  u8_disk = get_addr_encoding();
96  buf[at++] = u8_disk;
97  } else if ('S'==astr[i]) {
98  /* Signal frame; no auxilliary data */
99  } else {
100  ROSE_ASSERT(!"invalid .eh_frame augmentation string");
101  abort();
102  }
103  }
104  }
105 
106  /* Initial instructions */
107  sz = get_instructions().size();
108  if (sz>0) {
109  memcpy(buf+at, &(get_instructions()[0]), sz);
110  at += sz;
111  }
112 
113  std::string retval((char*)buf, at);
114  delete[] buf;
115  return retval;
116 }

167 std::string
168 SgAsmElfEHFrameEntryFD::unparse(const SgAsmElfEHFrameSection *ehframe, SgAsmElfEHFrameEntryCI *cie) const
169 {
170  SgAsmElfFileHeader *fhdr = ehframe->get_elfHeader();
171  ROSE_ASSERT(fhdr!=NULL);
172 
173  /* Allocate worst-case size for results */
174  size_t worst_size = 8 + get_augmentation_data().size() + get_instructions().size() + fhdr->get_wordSize();
175  unsigned char *buf = new unsigned char[worst_size];
176 
177  size_t sz;
178  Rose::BinaryAnalysis::Address at = 0;
179  uint32_t u32_disk;
180 
181  /* PC Begin (begin_rva) and size */
182  switch (cie->get_addr_encoding()) {
183  case -1: /* No address encoding specified */
184  case 0x01:
185  case 0x03:
186  case 0x1b: {
187  Rose::BinaryAnalysis::ByteOrder::hostToLe(get_begin_rva().rva(), &u32_disk);
188  memcpy(buf+at, &u32_disk, 4); at+=4;
189  Rose::BinaryAnalysis::ByteOrder::hostToLe(get_size(), &u32_disk);
190  memcpy(buf+at, &u32_disk, 4); at+=4;
191  break;
192  }
193  default:
194  /* See parser */
195  if (++nwarnings<=WARNING_LIMIT) {
196  mlog[WARN] <<"unknown FDE address encoding (" <<StringUtility::toHex2(cie->get_addr_encoding(), 8) <<")\n";
197  if (WARNING_LIMIT==nwarnings)
198  mlog[WARN] <<"additional frame warnings will be suppressed\n";
199  }
200  break;
201  }
202 
203  /* Augmentation Data */
204  std::string astr = cie->get_augmentation_string();
205  if (astr.size()>0 && astr[0]=='z') {
206  at = ehframe->writeUleb128(buf, at, get_augmentation_data().size());
207  sz = get_augmentation_data().size();
208  if (sz>0) {
209  memcpy(buf+at, &(get_augmentation_data()[0]), sz);
210  at += sz;
211  }
212  }
213 
214  /* Call frame instructions */
215  sz = get_instructions().size();
216  if (sz>0) {
217  memcpy(buf+at, &(get_instructions()[0]), sz);
218  at += sz;
219  }
220 
221  std::string retval((char*)buf, at);
222  delete[] buf;
223  return retval;
224 }

253 SgAsmElfEHFrameSection *
254 SgAsmElfEHFrameSection::parse()
255 {
256  SgAsmElfSection::parse();
257  SgAsmElfFileHeader *fhdr = get_elfHeader();
258  ROSE_ASSERT(fhdr!=NULL);
259 
260  Rose::BinaryAnalysis::Address record_offset=0;
261  std::map<Rose::BinaryAnalysis::Address, SgAsmElfEHFrameEntryCI*> cies;
262 
263  // This section consists of a Common Information Entry (CIE) followed by one or more Frame Description Entry (FDE) and this
264  // pattern is repeated for the entire section or until a termination record is reached. There is typically one CIE per
265  // object file and one FDE per function. The CIE and FDE together describe how to unwind the caller during an exception if
266  // the current instruction pointer is in the range covered by the FDE.
267 
268  Sawyer::Optional<Rose::BinaryAnalysis::Address> prevCieOffset;
269  while (record_offset<get_size()) {
270  Rose::BinaryAnalysis::Address at = record_offset;
271  unsigned char u8_disk;
272  uint32_t u32_disk;
273  uint64_t u64_disk;
274 
275  /* Length or extended length */
276  Rose::BinaryAnalysis::Address length_field_size = 4; /*number of bytes not counted in length*/
277  readContentLocal(at, &u32_disk, 4); at += 4;
278  Rose::BinaryAnalysis::Address record_size = diskToHost(fhdr->get_sex(), u32_disk);
279  if (record_size==0xffffffff) {
280  readContentLocal(at, &u64_disk, 8); at += 8;
281  record_size = diskToHost(fhdr->get_sex(), u64_disk);
282  length_field_size += 8; // length field size is both "length" and "extended length"
283  }
284  if (0==record_size) {
285  record_offset += length_field_size;
286  break;
287  }
288 
289  /* Backward offset to CIE record, or zero if this is a CIE record. */
290  readContentLocal(at, &u32_disk, 4); at += 4;
291  Rose::BinaryAnalysis::Address cie_back_offset = diskToHost(fhdr->get_sex(), u32_disk);
292  if (0==cie_back_offset) {
293  /* This is a CIE record */
294  SgAsmElfEHFrameEntryCI *cie = new SgAsmElfEHFrameEntryCI(this);
295  cies[record_offset] = cie;
296  prevCieOffset = record_offset;
297 
298  /* Version. This parser was written based on version 1 documentation. */
299  uint8_t cie_version;
300  readContentLocal(at++, &cie_version, 1);
301  cie->set_version(cie_version);
302 
303  /* Augmentation String */
304  std::string astr = readContentLocalString(at);
305  at += astr.size() + 1;
306  cie->set_augmentation_string(astr);
307 
308  /* EH data: 4 or 8 bytes depending on word size, only present "if the augmentation string contains 'eh'". The word
309  * "contains" in the documentation apparently means the string is _equal_ to "eh". */
310  if (astr == "eh") {
311  uint64_t eh_data;
312  if (4==fhdr->get_executableFormat()->get_wordSize()) {
313  readContentLocal(at, &u32_disk, 4); at += 4;
314  eh_data = diskToHost(fhdr->get_sex(), u32_disk);
315  } else {
316  readContentLocal(at, &u64_disk, 8); at += 8;
317  eh_data = diskToHost(fhdr->get_sex(), u64_disk);
318  }
319  cie->set_eh_data(eh_data);
320  }
321 
322  /* Alignment factors:
323  * + code_alignment_factor: An unsigned LEB128 encoded value that is factored out of all advance location
324  * instructions that are associated with this CIE or its FDEs. This value shall be multiplied by the delta
325  * argument of an adavance location instruction to obtain the new location value.
326  * + data_alignment_factor: A signed LEB128 encoded value that is factored out of all offset instructions that are
327  * associated with this CIE or its FDEs. This value shall be multiplied by the register offset argument of an
328  * offset instruction to obtain the new offset value. */
329  cie->set_code_alignment_factor(readContentLocalUleb128(&at));
330  cie->set_data_alignment_factor(readContentLocalSleb128(&at));
331 
332  /* Augmentation data length. This is apparently the length of the data described by the Augmentation String plus
333  * the Initial Instructions plus any padding [RPM 2009-01-15]. This field is present only if the augmentation
334  * string starts with the character "z" [Robb P. Matzke 2014-12-15]. */
335  if (!astr.empty() && astr[0]=='z')
336  cie->set_augmentation_data_length(readContentLocalUleb128(&at));
337 
338  /* Augmentation data. The format of the augmentation data in the CIE record is determined by reading the
339  * characters of the augmentation string. */
340  if (!astr.empty() && astr[0]=='z') {
341  for (size_t i=1; i<astr.size(); i++) {
342  if ('L'==astr[i]) {
343  readContentLocal(at++, &u8_disk, 1);
344  cie->set_lsda_encoding(u8_disk);
345  } else if ('P'==astr[i]) {
346  /* The first byte is an encoding method which describes the following bytes, which are the address of
347  * a Personality Routine Handler. There appears to be very little documentation about these fields. */
348  readContentLocal(at++, &u8_disk, 1);
349  cie->set_prh_encoding(u8_disk);
350  switch (cie->get_prh_encoding()) {
351  case 0x05: /* See Ubuntu 32bit /usr/bin/aptitude */
352  case 0x06: /* See second CIE record for Gentoo-Amd64 /usr/bin/addftinfo */
353  case 0x07: /* See first CIE record for Gentoo-Amd64 /usr/bin/addftinfo */
354  readContentLocal(at++, &u8_disk, 1); /* not sure what this is; arg for __gxx_personality_v0? */
355  cie->set_prh_arg(u8_disk);
356  readContentLocal(at, &u32_disk, 4); at+=4; /* address of <__gxx_personality_v0@plt> */
357  cie->set_prh_addr(Rose::BinaryAnalysis::ByteOrder::leToHost(u32_disk));
358  break;
359  case 0x09: /* *.o file generated by gcc-4.0.x */
360  /* FIXME: Cannot find any info about this entry. Fix SgAsmElfEHFrameSection::parse() if we
361  * ever figure this out. [RPM 2009-09-29] */
362  /*fallthrough*/
363  case 0x10: /* See 32-bit version of skype library */
364  /* FIXME[Robb P. Matzke 2014-12-15]: cannot find any info for this entry. */
365  /*fallthrough*/
366  default: {
367  if (++nwarnings<=WARNING_LIMIT) {
368  mlog[WARN] <<"ELF CIE " <<StringUtility::addrToString(get_offset()+record_offset)
369  <<" has unknown PRH encoding " <<StringUtility::toHex2(cie->get_prh_encoding(), 8)
370  <<"\n";
371  if (WARNING_LIMIT==nwarnings)
372  mlog[WARN] <<"additional frame warnings will be suppressed\n";
373  }
374  break;
375  }
376  }
377  } else if ('R'==astr[i]) {
378  readContentLocal(at++, &u8_disk, 1);
379  cie->set_addr_encoding(u8_disk);
380  } else if ('S'==astr[i]) {
381  /* See http://lkml.indiana.edu/hypermail/linux/kernel/0602.3/1144.html and GCC PR #26208*/
382  cie->set_sig_frame(true);
383  } else {
384  /* Some stuff we don't handle yet. Warn about it and don't read anything. */
385  if (++nwarnings<=WARNING_LIMIT) {
386  mlog[WARN] <<"ELF CIE " <<StringUtility::addrToString(get_offset() + record_offset)
387  <<" has invalid augmentation string \"" <<escapeString(astr) <<"\"\n";
388  if (WARNING_LIMIT==nwarnings)
389  mlog[WARN] <<"additional frame warnings will be suppressed\n";
390  }
391  }
392  }
393  }
394 
395  /* Initial instructions. These are apparently included in the augmentation_data_length. The final instructions can
396  * be zero padding (no-op instructions) to bring the record up to a multiple of the word size. */
397  Rose::BinaryAnalysis::Address init_insn_size = (length_field_size + record_size) - (at - record_offset);
398  cie->get_instructions() = readContentLocalUcl(at, init_insn_size);
399  ROSE_ASSERT(cie->get_instructions().size()==init_insn_size);
400 
401  } else {
402  /* This is a FDE record */
403  bool fde_parse_error = false;
404  Rose::BinaryAnalysis::Address cie_offset = record_offset + length_field_size - cie_back_offset;
405  if (cies.find(cie_offset) == cies.end()) {
406  mlog[ERROR] <<"bad CIE offset " <<cie_offset
407  <<" in section \"" <<StringUtility::cEscape(get_name()->get_string()) <<"\"\n";
408  mlog[ERROR] <<" referenced by FDE at offset " <<record_offset
409  <<" having CIE back offset " <<cie_back_offset <<"\n";
410  if (prevCieOffset) {
411  mlog[ERROR] <<" previous CIE was at offset " <<*prevCieOffset <<"\n";
412  } else {
413  mlog[ERROR] <<" there was no previous CIE in this section\n";
414  }
415  mlog[ERROR] <<" bailing out early\n";
416  break;
417  }
418 
419  assert(cies.find(cie_offset)!=cies.end());
420  SgAsmElfEHFrameEntryCI *cie = cies[cie_offset];
421  SgAsmElfEHFrameEntryFD *fde = new SgAsmElfEHFrameEntryFD(cie);
422 
423  /* PC Begin (begin_rva) and size */
424  switch (cie->get_addr_encoding()) {
425  case -1: /* No address encoding specified */
426  case 0x01:
427  case 0x03:
428  case 0x1b: /* Address doesn't look valid (e.g., 0xfffd74e8) but still four bytes [RPM 2008-01-16]*/
429  {
430  readContentLocal(at, &u32_disk, 4); at+=4;
431  fde->set_begin_rva(Rose::BinaryAnalysis::ByteOrder::leToHost(u32_disk));
432  readContentLocal(at, &u32_disk, 4); at+=4;
433  fde->set_size(Rose::BinaryAnalysis::ByteOrder::leToHost(u32_disk));
434  break;
435  }
436  default:
437  mlog[WARN] <<"ELF CIE " <<StringUtility::addrToString(get_offset()+cie_offset)
438  <<", FDE " <<StringUtility::addrToString(get_offset()+record_offset)
439  <<": unknown address encoding: " <<StringUtility::toHex2(cie->get_addr_encoding(), 8) <<"\n";
440  fde_parse_error = true;
441  break;
442  }
443 
444  // Location of following fields depends on having correctly unparsed the previous stuff which is RLE.
445  if (!fde_parse_error) {
446  /* Augmentation Data */
447  std::string astring = cie->get_augmentation_string();
448  if (astring.size()>0 && astring[0]=='z') {
449  Rose::BinaryAnalysis::Address aug_length = readContentLocalUleb128(&at);
450  fde->get_augmentation_data() = readContentLocalUcl(at, aug_length);
451  at += aug_length;
452  ROSE_ASSERT(fde->get_augmentation_data().size()==aug_length);
453  }
454 
455  /* Call frame instructions */
456  if (length_field_size + record_size < at - record_offset) {
457  mlog[ERROR] <<"ELF CIE " <<StringUtility::addrToString(get_offset()+cie_offset)
458  <<", FED " <<StringUtility::addrToString(get_offset()+record_offset)
459  <<": invalid call frame instructions size (skipping)\n";
460  } else {
461  Rose::BinaryAnalysis::Address cf_insn_size = (length_field_size + record_size) - (at - record_offset);
462  fde->get_instructions() = readContentLocalUcl(at, cf_insn_size);
463  ROSE_ASSERT(fde->get_instructions().size()==cf_insn_size);
464  }
465  }
466  }
467 
468  record_offset += length_field_size + record_size;
469  }
470 
471  if (record_offset < get_size()) {
472  mlog[WARN] <<"ELF error handling section \"" <<StringUtility::cEscape(get_name()->get_string()) <<"\""
473  <<" contains " <<StringUtility::plural(get_size()-record_offset, "more bytes")
474  <<" that were not parsed\n";
475  }
476 
477  return this;
478 }

510  ROSE_ASSERT(fhdr!=NULL);
511 
512  Rose::BinaryAnalysis::Address at=0;
513  uint32_t u32_disk;
514  uint64_t u64_disk;
515 
516  for (size_t i=0; i<get_ci_entries()->get_entries().size(); i++) {
517  Rose::BinaryAnalysis::Address last_cie_offset = at;
518  SgAsmElfEHFrameEntryCI *cie = get_ci_entries()->get_entries()[i];
519  std::string s = cie->unparse(this);
520  if (s.size()<0xffffffff) {
521  hostToDisk(fhdr->get_sex(), s.size(), &u32_disk);
522  if (fp)
523  write(*fp, at, 4, &u32_disk);
524  at += 4;
525  } else {
526  u32_disk = 0xffffffff;
527  if (fp)
528  write(*fp, at, 4, &u32_disk);
529  at += 4;
530  hostToDisk(fhdr->get_sex(), s.size(), &u64_disk);
531  if (fp)
532  write(*fp, at, 8, &u64_disk);
533  at += 8;
534  }
535  if (fp)
536  write(*fp, at, s);
537  at += s.size();
538 
539  for (size_t j=0; j<cie->get_fd_entries()->get_entries().size(); j++) {
540  SgAsmElfEHFrameEntryFD *fde = cie->get_fd_entries()->get_entries()[j];
541  std::string s = fde->unparse(this, cie);
542 
543  /* Record size, not counting run-length coded size field, but counting CIE back offset. */
544  Rose::BinaryAnalysis::Address record_size = 4 + s.size();
545  if (record_size<0xffffffff) {
546  hostToDisk(fhdr->get_sex(), record_size, &u32_disk);
547  if (fp)
548  write(*fp, at, 4, &u32_disk);
549  at += 4;
550  } else {
551  u32_disk = 0xffffffff;
552  if (fp)
553  write(*fp, at, 4, &u32_disk);
554  at += 4;
555  hostToDisk(fhdr->get_sex(), record_size, &u64_disk);
556  if (fp)
557  write(*fp, at, 8, &u64_disk);
558  at += 8;
559  }
560 
561  /* CIE back offset. Number of bytes from the beginning of the current CIE record (including the Size fields) to
562  * the beginning of the FDE record (excluding the Size fields but including the CIE back offset). */
563  Rose::BinaryAnalysis::Address cie_back_offset = at - last_cie_offset;
564  hostToDisk(fhdr->get_sex(), cie_back_offset, &u32_disk);
565  if (fp)
566  write(*fp, at, 4, &u32_disk);
567  at += 4;
568 
569  /* The FDE record itself */
570  if (fp)
571  write(*fp, at, s);
572  at += s.size();
573  }
574  }
575 
576  /* Write a zero length to indicate the end of the CIE list */
577  u32_disk = 0;
578  if (fp)
579  write(*fp, at, 4, &u32_disk);
580  at += 4;
581
