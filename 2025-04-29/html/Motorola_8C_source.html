35 const CallingConvention::Dictionary&
36 Motorola::callingConventions() const {
37  static SAWYER_THREAD_TRAITS::Mutex mutex;
38  SAWYER_THREAD_TRAITS::LockGuard lock(mutex);
39 
40  if (!callingConventions_.isCached()) {
41  CC::Dictionary dict;
42 
43  // https://m680x0.github.io/doc/abi.html
44  const RegisterDictionary::Ptr regdict = registerDictionary();
45  const RegisterDescriptor SP = regdict->stackPointerRegister();
46 
47  auto cc = CC::Definition::instance("sysv", "m68k-sysv", constPtr());
48 
49  //==== Address locations ====
50  cc->instructionPointerRegister(regdict->instructionPointerRegister());
51  cc->returnAddressLocation(ConcreteLocation(SP, 0, regdict));
52 
53  //==== Stack characteristics ====
54  cc->stackPointerRegister(SP);
55  cc->stackDirection(CC::StackDirection::GROWS_DOWN);
56  cc->nonParameterStackSize(bytesPerWord()); // return address
57 
58  // Arguments are all stored on the stack
59  {
60  auto pool = CC::StoragePoolStack::instance("args-on-stack", CC::isAnyType(), constPtr());
61  pool->initialOffset(bytesPerWord()); // return address
62  cc->argumentValueAllocator()->append(pool);
63  }
64 
65  //==== Function parameters ====
66  // All parameters are passed on the stack.
67  cc->stackParameterOrder(CC::StackParameterOrder::RIGHT_TO_LEFT);
68  cc->stackCleanup(CC::StackCleanup::BY_CALLER);
69 
70  //==== Other inputs ====
71 
72  //==== Return values ====
73 
74  // Integral return values
75  if (const RegisterDescriptor d0 = regdict->findOrThrow("d0")) {
76  cc->appendOutputParameter(d0);
77  auto pool = CC::StoragePoolEnumerated::instance("integral return value", CC::isIntegerNotWiderThan(bitsPerWord()));
78  pool->append(ConcreteLocation(d0, regdict));
79  cc->returnValueAllocator()->append(pool);
80  }
81 
82  // Address return values
83  if (const RegisterDescriptor a0 = regdict->findOrThrow("a0")) {
84  cc->appendOutputParameter(a0);
85  auto pool = CC::StoragePoolEnumerated::instance("addr return value", CC::isPointerNotWiderThan(bitsPerWord()));
86  pool->append(ConcreteLocation(a0, regdict));
87  cc->returnValueAllocator()->append(pool);
88  }
89 
90  // Floating-point return values
91  if (const RegisterDescriptor fp0 = regdict->findOrThrow("fp0")) {
92  cc->appendOutputParameter(fp0);
93  auto pool = CC::StoragePoolEnumerated::instance("fp return value", CC::isFpNotWiderThan(bitsPerWord()));
94  pool->append(ConcreteLocation(fp0, regdict));
95  cc->returnValueAllocator()->append(pool);
96  }
97 
98  // The stack pointer is an output because it is changed by the callee (incremented by four when the function returns). It
99  // does not hold a return value in the normal sense of the word, and thus isn't added to the return value allocator
100  // functions.
101  cc->appendOutputParameter(SP); // final value is usually one word greater than initial value
102 
103  //==== Scratch registers ====
104  // (i.e., modified, not callee-saved, not return registers
105  cc->scratchRegisters().insert(regdict->findOrThrow("d0"));
106  cc->scratchRegisters().insert(regdict->findOrThrow("d1"));
107  cc->scratchRegisters().insert(regdict->findOrThrow("a0"));
108  cc->scratchRegisters().insert(regdict->findOrThrow("a1"));
109  cc->scratchRegisters().insert(regdict->findOrThrow("pc"));
110  cc->scratchRegisters().insert(regdict->findOrThrow("ccr"));
111  cc->scratchRegisters().insert(regdict->findOrThrow("fp0"));
112  cc->scratchRegisters().insert(regdict->findOrThrow("fp1"));
113 
114 
115  //==== Callee-saved registers
116  // Everything else
117  RegisterParts regParts = regdict->getAllParts() - cc->getUsedRegisterParts();
118  std::vector<RegisterDescriptor> registers = regParts.extract(regdict);
119  cc->calleeSavedRegisters().insert(registers.begin(), registers.end());
120 
121  dict.push_back(cc);
122 
123  callingConventions_ = dict;
124  }
125 
126  return callingConventions_.get();
127 }

139 std::string
140 Motorola::instructionMnemonic(const SgAsmInstruction *insn_) const {
141  if (isUnknown(insn_))
142  return "unknown";
143 
144  auto insn = isSgAsmM68kInstruction(insn_);
145  ASSERT_not_null(insn);
146 
147  const std::string base = [insn]() -> std::string {
148  switch (insn->get_kind()) {
149  case m68k_move_acc:
150  case m68k_move_accext:
151  case m68k_move_ccr:
152  case m68k_move_macsr:
153  case m68k_move_mask:
154  case m68k_move_sr:
155  return "move";
156  default:
157  return stringify::Rose::BinaryAnalysis::M68kInstructionKind(insn->get_anyKind(), "m68k_");
158  }
159  }();
160 
161 
162  const std::string ext = [insn]() -> std::string {
163  switch (insn->get_dataFormat()) {
164  case m68k_fmt_i32: return ".l";
165  case m68k_fmt_f32: return ".s";
166  case m68k_fmt_f96: return ".x";
167  case m68k_fmt_p96: return ".p";
168  case m68k_fmt_i16: return ".w";
169  case m68k_fmt_f64: return ".d";
170  case m68k_fmt_i8: return ".b";
171  case m68k_fmt_unknown: return "";
172  }
173  ASSERT_not_reachable("invalid m68k data format: " + stringifyBinaryAnalysisM68kDataFormat(insn->get_dataFormat()));
174  }();
175 
176  return base + ext;
177 }

564 bool
565 Motorola::isFunctionCallSlow(const std::vector<SgAsmInstruction*>& insns, Address *target_va, Address *return_va) const {
566  if (isFunctionCallFast(insns, target_va, return_va))
567  return true;
568 
569  static const size_t EXECUTION_LIMIT = 25; // max size of basic blocks for expensive analyses
570  if (insns.empty())
571  return false;
572  auto last = isSgAsmM68kInstruction(insns.back());
573  ASSERT_not_null(last);
574  SgAsmFunction *func = SageInterface::getEnclosingNode<SgAsmFunction>(last);
575  SgAsmInterpretation *interp = SageInterface::getEnclosingNode<SgAsmInterpretation>(func);
576 
577  // Slow method: Emulate the instructions and then look at the program counter (PC) and stack (A7). If the PC points outside the
578  // current function and the top of the stack holds an address of an instruction within the current function, then this must be a
579  // function call.
580  if (interp && insns.size() <= EXECUTION_LIMIT) {
581  using namespace Rose::BinaryAnalysis;
582  using namespace Rose::BinaryAnalysis::InstructionSemantics;
583  using namespace Rose::BinaryAnalysis::InstructionSemantics::SymbolicSemantics;
584  const InstructionMap &imap = interp->get_instructionMap();
585  RegisterDictionary::Ptr regdict = registerDictionary();
586  SmtSolverPtr solver = SmtSolver::instance(Rose::CommandLine::genericSwitchArgs.smtSolver);
587  BaseSemantics::RiscOperators::Ptr ops = RiscOperators::instanceFromRegisters(regdict, solver);
588  ASSERT_not_null(ops);
589  DispatcherM68kPtr dispatcher = DispatcherM68k::promote(newInstructionDispatcher(ops));
590  SValue::Ptr orig_sp = SValue::promote(ops->peekRegister(dispatcher->REG_A[7]));
591  try {
592  for (size_t i = 0; i < insns.size(); ++i)
593  dispatcher->processInstruction(insns[i]);
594  } catch (const BaseSemantics::Exception &e) {
595  return false;
596  }
597 
598  // If the next instruction address is concrete but does not point to a function entry point, then this is not a call.
599  SValue::Ptr ip = SValue::promote(ops->peekRegister(regdict->instructionPointerRegister()));
600  if (auto target_va = ip->toUnsigned()) {
601  SgAsmFunction *target_func = SageInterface::getEnclosingNode<SgAsmFunction>(imap.get_value_or(*target_va, NULL));
602  if (!target_func || *target_va != target_func->get_entryVa())
603  return false;
604  }
605 
606  // If nothing was pushed onto the stack, then this isn't a function call.
607  SValue::Ptr sp = SValue::promote(ops->peekRegister(regdict->stackPointerRegister()));
608  SValue::Ptr stack_delta = SValue::promote(ops->add(sp, ops->negate(orig_sp)));
609  SValue::Ptr stack_delta_sign = SValue::promote(ops->extract(stack_delta, 31, 32));
610  if (stack_delta_sign->isFalse())
611  return false;
612 
613  // If the top of the stack does not contain a concrete value or the top of the stack does not point to an instruction in
614  // this basic block's function, then this is not a function call.
615  SValue::Ptr top = SValue::promote(ops->peekMemory(RegisterDescriptor(), sp, sp->undefined_(32)));
616  if (auto va = top->toUnsigned()) {
617  SgAsmFunction *return_func = SageInterface::getEnclosingNode<SgAsmFunction>(imap.get_value_or(*va, NULL));
618  if (!return_func || return_func!=func) {
619  return false;
620  }
621  } else {
622  return false;
623  }
624 
625  // Since the instruction pointer might point to a function entry address and since the top of the stack contains a pointer
626  // to an instruction in this function, we assume that this is a function call.
627  if (target_va)
628  ip->toUnsigned().assignTo(*target_va);
629  if (return_va)
630  top->toUnsigned().assignTo(*return_va);
631  return true;
632  }
633 
634  // Similar to the above method, but works when all we have is the basic block (e.g., this case gets hit quite a bit from the
635  // Partitioner). Returns true if, after executing the basic block, the top of the stack contains the fall-through address of
636  // the basic block. We depend on our caller to figure out if the instruction pointer is reasonably a function entry address.
637  if (!interp && insns.size() <= EXECUTION_LIMIT) {
638  using namespace Rose::BinaryAnalysis;
639  using namespace Rose::BinaryAnalysis::InstructionSemantics;
640  using namespace Rose::BinaryAnalysis::InstructionSemantics::SymbolicSemantics;
641  RegisterDictionary::Ptr regdict = registerDictionary();
642  SmtSolverPtr solver = SmtSolver::instance(Rose::CommandLine::genericSwitchArgs.smtSolver);
643  BaseSemantics::RiscOperators::Ptr ops = RiscOperators::instanceFromRegisters(regdict, solver);
644  DispatcherM68kPtr dispatcher = DispatcherM68k::promote(newInstructionDispatcher(ops));
645  try {
646  for (size_t i = 0; i < insns.size(); ++i)
647  dispatcher->processInstruction(insns[i]);
648  } catch (const BaseSemantics::Exception &e) {
649  return false;
650  }
651 
652  // Look at the top of the stack
653  const RegisterDescriptor SP = regdict->stackPointerRegister();
654  const RegisterDescriptor IP = regdict->instructionPointerRegister();
655  SValue::Ptr top = SValue::promote(ops->peekMemory(RegisterDescriptor(), ops->peekRegister(SP),
656  ops->protoval()->undefined_(32)));
657  if (top->toUnsigned().orElse(0) == last->get_address() + last->get_size()) {
658  if (target_va) {
659  SValue::Ptr ip = SValue::promote(ops->peekRegister(IP));
660  ip->toUnsigned().assignTo(*target_va);
661  }
662  if (return_va)
663  top->toUnsigned().assignTo(*return_va);
664  return true;
665  }
666  }
667 
668  return false;
669 }

963  const BinaryAnalysis::MemoryMap::Ptr &initial_memory) const {

975  // do a more thorough analysis now. In the case where the cursory analysis returned a complete set containing two successors, a
976  // thorough analysis might be able to narrow it down to a single successor. We should not make special assumptions about
977  // function call instructions -- their only successor is the specified address operand.
