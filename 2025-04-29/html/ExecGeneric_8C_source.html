45 SgAsmExecutableFileFormat::parseBinaryFormat(const char *name)
46 {

48  std::vector<DataConverter*> converters;
49  converters.push_back(nullptr); /*no conversion*/
50  converters.push_back(new Rot13);

52  struct Cleanup {
53  std::vector<DataConverter*> &converters;
54  SgAsmGenericFile *&ef;
55  bool canceled;
56  Cleanup(std::vector<DataConverter*> &converters, SgAsmGenericFile *&ef)
57  : converters(converters), ef(ef), canceled(false) {}
58  ~Cleanup() {
59  if (!canceled) {
60  BOOST_FOREACH (DataConverter *converter, converters)
61  delete converter;
62  if (ef)
63  delete ef->get_dataConverter();
64  SageInterface::deleteAST(ef);
65  }
66  }
67  } cleanup(converters, ef);
68 
69  for (size_t ci=0; !ef && ci<converters.size(); ci++) {
70  ef = new SgAsmGenericFile();
71  ef->set_dataConverter(converters[ci]);
72  converters[ci] = nullptr;
73  ef->parse(name);
74 
75  if (SgAsmElfFileHeader::isElf(ef)) {
76  (new SgAsmElfFileHeader(ef))->parse();
77  } else if (SgAsmDOSFileHeader::isDos(ef)) {
78  SgAsmDOSFileHeader *dos_hdr = new SgAsmDOSFileHeader(ef);
79  dos_hdr->parse(false); /*delay parsing the DOS Real Mode Section*/
80 
81  /* DOS Files can be overloaded to also be PE, NE, LE, or LX. Such files have an Extended DOS Header
82  * immediately after the DOS File Header (various forms of Extended DOS Header exist). The Extended DOS Header
83  * contains a file offset to a PE, NE, LE, or LX File Header, the first bytes of which are a magic number. The
84  * is_* methods check for this magic number. */
85  if (SgAsmPEFileHeader::isPe(ef)) {
86  SgAsmDOSExtendedHeader *dos2_hdr = new SgAsmDOSExtendedHeader(dos_hdr);
87  dos2_hdr->parse();
88  SgAsmPEFileHeader *pe_hdr = new SgAsmPEFileHeader(ef);
89  pe_hdr->set_offset(dos2_hdr->get_e_lfanew());
90  pe_hdr->parse();
91  } else if (SgAsmNEFileHeader::isNe(ef)) {
92  SgAsmNEFileHeader::parse(dos_hdr);
93  } else if (SgAsmLEFileHeader::isLe(ef)) { /*or LX*/
94  SgAsmLEFileHeader::parse(dos_hdr);
95  }
96  dos_hdr->parseRealModeSection(0);
97  } else {
98  if (ef) delete ef->get_dataConverter();
99  SageInterface::deleteAST(ef); /* ~SgAsmGenericFile() closes ef->p_fd if it was opened. */
100  ef = nullptr;
101  }
102  }
103 
104  /* If no executable file could be parsed then try to use system tools to get the file type. On Unix-based systems, the
105  * "file" command does a thorough job of trying to figure out the type of file. If we can't figure out the file name then
106  * just throw some generic error. */
107  if (!ef) {
108  /* Use file(1) to try to figure out the file type to report in the exception */
109  int child_stdout[2];
110 #ifdef _MSC_VER
111 #pragma message ("WARNING: Commented out use of functions from sys/wait.h")
112  printf ("ERROR: Commented out use of functions from sys/wait.h \n");
113  ROSE_ABORT();
114 #else
115  if (pipe(child_stdout) == -1)
116  throw FormatError("unrecognized file format for \"" + StringUtility::cEscape(name) + "\"");
117 
118  pid_t pid = fork();
119  if (0==pid) {
120  close(0);
121  dup2(child_stdout[1], 1);
122  close(child_stdout[0]);
123  close(child_stdout[1]);
124  execlp("file", "file", "-b", name, nullptr);
125  exit(1);
126  } else if (pid>0) {
127  char buf[4096];
128  ssize_t nread = read(child_stdout[0], buf, sizeof buf);
129  if (nread >= 0) {
130  memset(buf+nread, 0, sizeof(buf)-nread);
131  } else {
132  memset(buf, 0, sizeof(buf));
133  }
134  buf[sizeof(buf)-1] = '\0';
135  if (char *nl = strchr(buf, '\n')) *nl = '\0'; /*keep only first line w/o LF*/
136  waitpid(pid, nullptr, 0);
137  char mesg[64+sizeof buf];
138  snprintf(mesg, sizeof(mesg), "unrecognized file format for \"%s\": %s", StringUtility::cEscape(name).c_str(), buf);
139  throw FormatError(mesg);
140  } else {
141  throw FormatError("unrecognized file format for \"" + StringUtility::cEscape(name) + "\"");
142  }
143 #endif
144  }
145 
146  ef->set_trackingReferences(false); /*all done parsing*/
147 
148  /* Is the file large enough to hold all sections? If any section extends past the EOF then set truncateZeros, which will
149  * cause the unparser to not write zero bytes to the end of the file. */
150  ef->set_truncateZeros(ef->get_currentSize()>ef->get_originalSize());
151 
152  /* If any section is the target of a function symbol then mark that section as containing code even if that section is not
153  * memory mapped with execute permission. */
154 
155 #ifndef USE_ROSE
156  // DQ (1/27/2010): This is a problem for ROSE compiling this file.
157  struct: public AstSimpleProcessing {

159  SgAsmGenericSymbol *symbol = isSgAsmGenericSymbol(node);
160  if (symbol && symbol->get_type()==SgAsmGenericSymbol::SYM_FUNC) {
161  SgAsmGenericSection *section = symbol->get_bound();
162  if (section)
163  section->set_containsCode(true);
164  }
165  }
166  } t1;
167  t1.traverse(ef, preorder);

169 
170  cleanup.canceled = true;
171  return ef;
