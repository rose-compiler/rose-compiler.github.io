396 void VariableIdMapping::computeVariableSymbolMapping(SgProject* project, int maxWarningsCount) {
397  int numWarningsCount=0;
398  set<SgSymbol*> symbolSet;
399  list<SgGlobal*> globList=SgNodeHelper::listOfSgGlobal(project);
400  for(list<SgGlobal*>::iterator k=globList.begin();k!=globList.end();++k) {
401  RoseAst ast(*k);
402  ast.setWithTemplates(true);
403  for(RoseAst::iterator i=ast.begin();i!=ast.end();++i) {
404  SgSymbol* sym = 0;
405  SgInitializedName* initName = NULL;
406  // schroder3 (2016-08-18): Added variables that are "declared" in a lambda capture. There is currently no SgInitializedName for
407  // these closure variables.
408  if(SgLambdaCapture* lambdaCapture = isSgLambdaCapture(*i)) {
409  // Add the closure variable (which is a member of enclosing lambda's anonymous class) to the mapping. (Even in case of a
410  // capture by reference there is a new reference member that references to the same variable as the captured reference.)
411  SgVarRefExp* closureVar = isSgVarRefExp(lambdaCapture->get_closure_variable());
412  ROSE_ASSERT(closureVar);
413  sym = closureVar->get_symbol();
414  ROSE_ASSERT(sym);
415  //type = closureVar->get_type();
416  } else if((initName = isSgInitializedName(*i))) {
417  // Variable/ parameter found: Try to get its symbol:
418  sym = initName->search_for_symbol_from_symbol_table();
419  if(sym) {
420  // determine the declaration to check for bitfields
421  if(isAnonymousBitfield(initName)) {
422  // MS (2018-12-4/2019-03-20): workaround: ROSE BUG: if a struct contains more than one anonymous bitfield
423  // they are assigned the same symbol.
424  // ROSE AST BUG WORKAROUND (ROSE-1867)
425  continue;
426  }
427  }
428  }
429  if(sym) {
430  // Check if the symbol is already registered:
431  if(symbolSet.find(sym) == symbolSet.end()) {
432  // Register new symbol as normal variable symbol:
433  // ensure it's a "valid" symbol
434  if(sym->get_symbol_basis()!=0) {
435  registerNewSymbol(sym);
436  // Remember that this symbol was already registered:
437  symbolSet.insert(sym);
438  }
439  }
440  } else {
441  numWarningsCount++;
442  std::string name("UNKNOWN");
443  if(initName) {
444  name = initName->get_qualified_name().getString();
445  }
446  if(numWarningsCount<maxWarningsCount || -1 == maxWarningsCount) {
447  if(initName) {
448  Rose::Diagnostics::mlog[WARN]<<"VariableIdMapping: No symbol available for SgInitializedName " << name << " at "<<SgNodeHelper::sourceFilenameLineColumnToString(*i)<<endl;
449  }
450  } else if(numWarningsCount==maxWarningsCount) {
451  Rose::Diagnostics::mlog[WARN]<<"VariableIdMapping: No symbol available for SgInitializedName " << name << " at "<<SgNodeHelper::sourceFilenameLineColumnToString(*i)<<endl;
452  Rose::Diagnostics::mlog[WARN]<<"VariableIdMapping: No symbol available for SgInitializedName: maximum warning count of "<<maxWarningsCount<<" reached."<<endl;
453  }
454  }
455  }
456  }
457  // creates variableid for each string literal in the entire program
458  registerStringLiterals(project);
459 
460  return;
461 }

636 void VariableIdMapping::registerNewSymbol(SgSymbol* sym) {
637  ROSE_ASSERT(sym);
638  // ensure every symbol is only registered once (symbols of linked global variables can be processed multiple times)
639  if(mappingSymToVarId.find(sym)==mappingSymToVarId.end()) {
640  // If one of the sizes is zero, the other size have to be zero too:
641  ROSE_ASSERT(mappingSymToVarId.size() == 0 ? mappingVarIdToInfo.size() == 0 : true);
642  ROSE_ASSERT(mappingVarIdToInfo.size() == 0 ? mappingSymToVarId.size() == 0 : true);
643 
644  // Create new mapping entry:
645  size_t newIdCode = mappingVarIdToInfo.size();
646  VariableId newVarId=variableIdFromCode(newIdCode);
647  mappingSymToVarId[sym] = newVarId;
648  mappingVarIdToInfo[newVarId].sym=sym;
649 
650  ROSE_ASSERT(sym);
651  // exclude temporary variables from link analysis
652  if(!isTemporaryVariableIdSymbol(sym)) {
653  // determine if sym is declared in global scope and store information required for link analysis
654  // this check only succeeds for global variables (it filters member variables)
655  SgScopeStatement* scope=sym->get_scope();
656  if(isSgGlobal(scope)) {
657  SgName name=sym->get_mangled_name();
658  mappingGlobalVarNameToSymSet[name].insert(sym); // set of symbols that map to a variable with the same name
659  }
660  }
661  setNumberOfElements(newVarId,unknownSizeValue()); // unknown number of elements
662  // Mapping in both directions must be possible:
663  ROSE_ASSERT(mappingSymToVarId.at(mappingVarIdToInfo[newVarId].sym) == newVarId);
664  ROSE_ASSERT(mappingVarIdToInfo[mappingSymToVarId.at(sym)].sym == sym);
665  }
