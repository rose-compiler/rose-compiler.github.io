130 void
131 jserver_start(JvmT* je)
132 {
133  JavaVMInitArgs jvm_args; /* VM initialization args. */
134  // Rasmussen (6/28/2017): Bumped to latest(?), unsure why 1_4 was used
135  // Rasmussen (2/13/2019): Increased JNI version to 1.8. This should
136  // remove need to install legacy jdk version 1.6 from Apple.
137  jvm_args.version = JNI_VERSION_1_8;
138  jvm_args.ignoreUnrecognized = JNI_FALSE;
139 
140  //----------------------------------------------------------------------------
141  // Add all our JVM options
142  //----------------------------------------------------------------------------
143  // TOO1 (2/11/2014): JVM options now stored in the Cmdline::Fortran::Ofp namespace.
144  std::list<std::string> jvm_options =
145  Rose::Cmdline::Fortran::Ofp::jvm_options;
146 
147  std::string classpath =
148  Rose::Cmdline::Fortran::Ofp::GetRoseClasspath();
149  jvm_options.push_back(classpath);
150 
151  // Rasmussen (6/28/2017): Increasing stacksize fixes crashes on some rhel7 systems
152  std::string stack_option = "-Xss2m";
153  jvm_options.push_back(stack_option);
154 
155  jvm_args.nOptions = jvm_options.size();
156  jvm_args.options = new JavaVMOption[jvm_args.nOptions];
157  for(int i=0; i < jvm_args.nOptions; ++i)
158  {
159  std::string jvm_option = jvm_options.front();
160 
161  jvm_args.options[i].optionString =
162  strdup(jvm_option.c_str());
163 
164  jvm_options.pop_front();
165  }
166 
167  //----------------------------------------------------------------------------
168  // Create and load the Java VM.
169  //----------------------------------------------------------------------------
170  jint res = JNI_CreateJavaVM(&(je->jvm), (void **)&(je->env), &jvm_args);
171 
172  if (res != JNI_OK) {
173  printf("Failed to create Java VM\n");
174  }
175 
176 //sanity check
177  JNIEnv* env = get_env();
178  ROSE_ASSERT(env->GetVersion() >= JNI_VERSION_1_8);
179 
180  if (res<0 || je->jvm==NULL || je->env==NULL)
181  {
182  exit(1);
183  }
184 
185  return ;
186 }

225 jobjectArray
226 jserver_getJavaStringArray(int argc, char **argv)
227 {
228  jobjectArray argsStringArray = NULL;
229  jclass stringClass;
230  int i;
231  JNIEnv * env = get_env();
232 
233  /* We need the String class because that is the underlying type of
234  the array. */
235  stringClass = jserver_getJavaStringClass();
236  if(stringClass == NULL)
237  return NULL;
238 
239  /* Build a new object array. Params are: env, length, class type of the
240  array, initial object(?). */
241  argsStringArray = env->NewObjectArray((argc-1), stringClass, NULL);
242 
243  if(argsStringArray == NULL)
244  return NULL;
245 
246  /* Put all args from argv, after the first (which is this program's
247  name) into the array of Strings for FortranMain. The args array
248  for Java does not include the program name. */
249  for(i = 1; i < argc; i++)
250  env->SetObjectArrayElement(argsStringArray, (jsize)i-1,
251  jserver_getJavaString(argv[i]));
252 
253  return argsStringArray;
254 }
