37 SAWYER_EXPORT std::string
38 ErrorLocation::toString() const {
39  std::string retval;
40  BOOST_FOREACH (const Frame &frame, frames_) {
41  retval += (retval.empty() ? "" : "\n") + frame.name;
42  std::string s = boost::trim_right_copy(frame.input);
43  size_t offset = std::min(frame.offset, s.size());
44 
45  // If there's lots of stuff after the curor, replace it with "..."
46  static const size_t MAX_BEFORE_CURSOR = 50; // arbitrary amount context to show before cursor
47  static const size_t MAX_AFTER_CURSOR = 20; // arbitrary amount to show after the cursor
48  static const double ELLIPSIS_PLACEMENT = 0.5; // where to position ellipsis before the cursor
49 
50  if (offset < s.size() && s.size() - offset > MAX_AFTER_CURSOR)
51  s = s.substr(0, offset + MAX_AFTER_CURSOR) + "...";
52 
53  if (offset > MAX_BEFORE_CURSOR) {
54  size_t n1 = round(MAX_BEFORE_CURSOR * ELLIPSIS_PLACEMENT);
55  size_t n2 = MAX_BEFORE_CURSOR - n1;
56  s = s.substr(0, n1) + "..." + s.substr(offset-n2);
57  offset = n1 + 3 + n2;
58  }
59 
60  retval += "\n |" + s + "\n |" + std::string(offset, '-') + "^\n";
61  }
62 
63  return retval;
64 }

75 SAWYER_EXPORT Token
76 TokenStream::scanNextToken(const Container::LineVector &content, size_t &at /*in,out*/) {
77  size_t begin = at;
78 
79  while (1) {
80  int c = content.character(at);
81  if (EOF == c) {
82  if (at > begin)
83  return Token(TOK_DATA, begin, at);
84  return Token();
85 
86  } else if (CHAR_AT == c && isalpha(content.character(at+1))) {
87  if (at > begin)
88  return Token(TOK_DATA, begin, at);
89  ++at;
90  while (isValidNameCharacter(content.character(at)))
91  ++at;
92  return Token(TOK_FUNCTION, begin, at);
93 
94  } else if (CHAR_AT == c) {
95  c = content.character(++at);
96  if (EOF == c)
97  return Token(TOK_DATA, begin, at);
98  ++at;
99 
100  } else if (CHAR_LEFT == c) {
101  if (at > begin)
102  return Token(TOK_DATA, begin, at);
103  return Token(TOK_LEFT, begin, ++at);
104 
105  } else if (CHAR_RIGHT == c) {
106  if (at > begin)
107  return Token(TOK_DATA, begin, at);
108  return Token(TOK_RIGHT, begin, ++at);
109 
110  } else if ('\n' == c) {
111  size_t nLineFeeds = 1, i = at + 1;
112  while (isspace(c = content.character(i))) {
113  if ('\n' == c)
114  ++nLineFeeds;
115  ++i;
116  }
117  if (nLineFeeds > 1) {
118  if (at > begin)
119  return Token(TOK_DATA, begin, at);
120  at = i;
121  return Token(TOK_BLANK_LINE, begin, i);
122  } else {
123  at = i;
124  }
125 
126  } else {
127  ++at;
128  }
129  }
130 }

203  " but only " + boost::lexical_cast<std::string>(maxArgs()) + " allowed");
204  }

207  actuals.push_back(*formals_[i].dflt);

355 SAWYER_EXPORT std::string
356 Grammar::operator()(const std::string &s) {
357  static size_t callLevel = 0;
358 
359  struct CallLevel {
360  size_t &callLevel;
361  CallLevel(size_t &n): callLevel(n) { ++callLevel; }
362  ~CallLevel() { ASSERT_require(callLevel>0); --callLevel; }
363  } called(callLevel);
364 
365  ErrorLocation eloc;
366  try {
367  TokenStream tokens(s);
368  ErrorLocation::Trap t(eloc, tokens, 1==callLevel ? "top level" : "recursive eval");
369  std::string retval = unescape(eval(tokens, eloc));
370  t.passed();
371  ASSERT_require(callLevel > 0);
372  return retval;
373  } catch (const SyntaxError &e) {
374  throw SyntaxError(e.what() + std::string("\n") + eloc.toString());
375  }

414 SAWYER_EXPORT std::string
415 Grammar::readArgument(TokenStream &tokens, ErrorLocation &/*eloc*/, bool requireRight) const {
416  std::string retval;
417  size_t depth = 0;
418  for (/*void*/; !tokens.atEof(); tokens.consume()) {
419  if (tokens.isa(TOK_LEFT)) {
420  ++depth;
421  } else if (tokens.isa(TOK_RIGHT)) {
422  if (0 == depth)
423  break;
424  --depth;
425  }
426  retval += tokens.lexeme();
427  }
428  if (requireRight) {

430  throw SyntaxError("end-of-argument expected");
431  tokens.consume();
432  }
433  if (depth > 0)
434  throw SyntaxError("expected end-of-argument marker");
435  return retval;

439 Grammar::evalArgument(TokenStream &tokens, ErrorLocation &eloc, bool requireRight) {
440  std::string retval, data;
441  size_t depth = 0;
442  while (!tokens.atEof()) {
443  if (tokens.isa(TOK_LEFT)) {
444  ++depth;
445  retval += tokens.lexeme();
446  tokens.consume();
447  } else if (tokens.isa(TOK_RIGHT)) {
448  if (0 == depth)

456  retval += tokens.lexeme();

462  throw SyntaxError("end-of-argument expected");
463  tokens.consume();
464  }
465  if (depth > 0)
466  throw SyntaxError("expected end-of-argument marker");
467  return retval;

470 SAWYER_EXPORT std::string
471 Grammar::evalFunction(TokenStream &tokens, ErrorLocation &eloc) {
472  ASSERT_require(tokens.isa(TOK_FUNCTION));
473  std::string funcName = tokens.lexeme();
474  ASSERT_require(funcName.size() >= 2 && '@' == funcName[0]);
475  funcName = funcName.substr(1);
476  tokens.consume();
477 
478  // Get the function declaration
479  const Function::Ptr func = functions_.getOrDefault(funcName);
480  if (!func)
481  throw SyntaxError("function \"" + funcName + "\" is not declared");
482 
483  // Parse the actual arguments
484  std::vector<std::string> actuals;
485  while (tokens.isa(TOK_LEFT)) {
486  tokens.consume();
487  if (func->isMacro()) {
488  actuals.push_back(readArgument(tokens, eloc, CONSUME));
489  } else {
490  actuals.push_back(evalArgument(tokens, eloc, CONSUME));
491  }
492  }
493  func->validateArgs(actuals, tokens);
494 
495  ErrorLocation::Trap t(eloc, tokens, "in function \"" + funcName + "\"");
496  std::string retval = func->eval(*this, actuals);
497  t.passed();
498  return retval;
499 }
