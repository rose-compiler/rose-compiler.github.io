59 SyscallSuccessors::operator()(bool chain, const Args &args) {
60  ASSERT_not_null(args.bblock);
61  if (!chain)
62  return false;
63  if (args.bblock->nInstructions() == 0)
64  return chain;
65 
66  SgAsmInstruction *syscallInsn = analyzer_.hasSystemCall(args.bblock);
67  if (syscallInsn != args.bblock->instructions().back())
68  return chain;
69 
70  try {
71  SystemCall::Declaration decl;
72  if (analyzer_.analyze(args.partitioner, args.bblock, syscallInsn, args.partitioner->smtSolver()).assignTo(decl)) {
73  if (decl.name == "exit") {
74  // This system call doesn't return
75  args.bblock->successors(BasicBlock::Successors()); // defined, but empty
76  } else {
77  // This system call is like a function call in that it calls some undetermined location and then probably
78  // returns to the fall through address. Therefore, create two edges: one for the function call and the other
79  // for the call return.
80  args.bblock->successors(BasicBlock::Successors()); // remove existing successors
81  size_t wordsize = args.partitioner->instructionProvider().instructionPointerRegister().nBits();
82  BaseSemantics::SValue::Ptr indeterminateVa = args.partitioner->newOperators()->undefined_(wordsize);
83  args.bblock->insertSuccessor(indeterminateVa, E_FUNCTION_CALL);
84  args.bblock->insertSuccessor(args.bblock->fallthroughVa(), wordsize, E_CALL_RETURN);
85  }
86  }
87  } catch (...) {
88  // Not an error if we can't figure out the system call name.
89  }
90 
91  return chain;
92 }

112 LibcStartMain::readStack(const Partitioner::ConstPtr &partitioner, const BaseSemantics::Dispatcher::Ptr &cpu, int byteOffset,
113  size_t nBits, RegisterDescriptor segmentRegister) {
114  ASSERT_not_null(partitioner);
115  const RegisterDescriptor SP = partitioner->instructionProvider().stackPointerRegister();
116  if (SP.isEmpty())
117  return BaseSemantics::SValue::Ptr();
118  BaseSemantics::SValue::Ptr sp = cpu->operators()->peekRegister(SP, cpu->undefined_(SP.nBits()));
119  BaseSemantics::SValue::Ptr stackOffset = cpu->number_(SP.nBits(),
120  BitOps::signExtend((uint64_t)byteOffset, 8*sizeof(byteOffset)));
121  BaseSemantics::SValue::Ptr addr = cpu->operators()->add(sp, stackOffset);
122  BaseSemantics::SValue::Ptr value = cpu->operators()->peekMemory(segmentRegister, addr, cpu->undefined_(nBits));
123  return value;

126 bool
127 LibcStartMain::operator()(bool chain, const Args &args) {
128  Sawyer::Message::Stream debug(mlog[DEBUG]);
129 
130  // Look at this block only if it's a function call
131  if (!chain || !args.bblock || args.bblock->nInstructions() == 0)
132  return chain;
133  if (!args.partitioner->basicBlockIsFunctionCall(args.bblock))
134  return chain;
135 
136  // It must call a function named "__libc_start_main@plt"
137  bool foundCallToLibcStartMain = false;
138  for (const Address &succVa: args.partitioner->basicBlockConcreteSuccessors(args.bblock)) {
139  Function::Ptr func = args.partitioner->functionExists(succVa);
140  if (func && (func->name() == "__libc_start_main@plt" || func->name() == "__libc_start_main")) {
141  foundCallToLibcStartMain = true;
142  break;
143  }
144  }
145  if (!foundCallToLibcStartMain)
146  return chain;
147  SAWYER_MESG(debug) <<"LibcStartMain analysis: found call to __libc_start_main\n";
148 
149  // One of the arguments to the function call is the address of "main". We need instruction semantics to get its value.
150  // Map-based memory seems to work best for this because it's faster, simpler to understand, and avoids inconsistencies when
151  // the basic block has aliasing near the top of the stack. Since __libc_start_main is called by _start, which is also
152  // responsible for aligning the stack pointer, the stack pointer ends up often being an somewhat complicated symbolic
153  // expression, which in turn causes aliasing-aware memory states to return unknown values when reading the stack (using
154  // similar complicated expression) due to possible aliasing.
155  BaseSemantics::Dispatcher::Ptr dispatcher;
156  BaseSemantics::RiscOperators::Ptr ops;
157  BaseSemantics::State::Ptr state;
158  try {
159  ops = args.partitioner->newOperators(MAP_BASED_MEMORY);
160  if (ops) {
161  dispatcher = args.partitioner->newDispatcher(ops);
162  if (dispatcher) {
163  for (SgAsmInstruction *insn: args.bblock->instructions())
164  dispatcher->processInstruction(insn);
165  state = dispatcher->currentState();
166  }
167  }
168  } catch (...) {
169  }
170  if (!state) {
171  SAWYER_MESG(debug) <<"LibcStartMain analysis: failed to obtain basic block semantic state\n";
172  return chain;
173  }
174 
175  // Some of the arguments are going to be function pointers. Give them names if we can.
176  std::vector<BaseSemantics::SValue::Ptr> functionPtrs;
177 
178  // Location and size of argument varies by architecture
179  RegisterDictionary::Ptr regs = args.partitioner->instructionProvider().registerDictionary();
180  if (isSgAsmX86Instruction(args.bblock->instructions().back())) {
181  if (dispatcher->addressWidth() == 64) {
182  // Amd64 integer arguments are passed in registers: rdi, rsi, rdx, rcx, r8, and r9
183  const RegisterDescriptor FIRST_ARG = regs->findOrThrow("rdi");
184  const RegisterDescriptor FOURTH_ARG = regs->findOrThrow("rcx");
185  const RegisterDescriptor FIFTH_ARG = regs->findOrThrow("r8");
186  BaseSemantics::SValue::Ptr firstArg = state->peekRegister(FIRST_ARG, dispatcher->undefined_(64), ops.get());
187  BaseSemantics::SValue::Ptr fourthArg = state->peekRegister(FOURTH_ARG, dispatcher->undefined_(64), ops.get());
188  BaseSemantics::SValue::Ptr fifthArg = state->peekRegister(FIFTH_ARG, dispatcher->undefined_(64), ops.get());
189 
190  if (firstArg->isConcrete() && fourthArg->isConcrete() && fifthArg->isConcrete() &&
191  args.partitioner->memoryMap()->at(firstArg->toUnsigned().get()).require(MemoryMap::EXECUTABLE).exists() &&
192  args.partitioner->memoryMap()->at(fourthArg->toUnsigned().get()).require(MemoryMap::EXECUTABLE).exists() &&
193  args.partitioner->memoryMap()->at(fifthArg->toUnsigned().get()).require(MemoryMap::EXECUTABLE).exists()) {
194  // Sometimes the address of main is passed as the first argument (rdi) with __libc_csu_init and __libc_csu_fini
195  // passed as the fourth (rcx) and fifth (r8) arguments. By this point in the disassembly process, would not
196  // have discovered PLT function yet. So if the first, fourth, and fifth arguments seem to point at executable
197  // memory then assume they are "main", "__libc_csu_fini@plt", and "libc_csu_init@plt". Don't bother naming the
198  // two PLT functions though since we'll get their names later by processing the PLT.
199  SAWYER_MESG(debug) <<"LibcStartMain analysis: amd64 with main as the fisrt argument\n";
200  mainVa_ = firstArg->toUnsigned().get();
201  functionPtrs.push_back(firstArg);
202  functionPtrs.push_back(fourthArg);
203  functionPtrs.push_back(fifthArg);
204 
205  } else if (fourthArg->toUnsigned() &&
206  args.partitioner->memoryMap()->at(fourthArg->toUnsigned().get()).require(MemoryMap::EXECUTABLE).exists()) {
207  // Sometimes then address of main is passed as the fourth argument (in rcx).
208  SAWYER_MESG(debug) <<"LibcStartMain analysis: amd64 with main as the fourth argument\n";
209  mainVa_ = fourthArg->toUnsigned().get();
210  functionPtrs.push_back(fourthArg);
211  }
212 
213  } else if (dispatcher->addressWidth() == 32) {
214  // x86 integer arguments are passed on the stack. The address of main is the first argument, which starts four bytes
215  // into the stack because the return address has also been pushed onto the stack.
216  BaseSemantics::SValue::Ptr arg0 = readStack(args.partitioner, dispatcher, 4, 32, RegisterDescriptor());
217  if (arg0->toUnsigned().assignTo(mainVa_)) {
218  SAWYER_MESG(debug) <<"LibcStartMain analysis: x86 with main as the first argument\n";
219  functionPtrs.push_back(arg0);
220  }
221  }
222  } else if (isSgAsmM68kInstruction(args.bblock->instructions().back())) {
223  // M68k integer arguments are passed on the stack. The address of main is the first argument, which starts four bytes
224  // into the stack because the return address has also been pushed onto the stack.
225  BaseSemantics::SValue::Ptr arg0 = readStack(args.partitioner, dispatcher, 4, 32, RegisterDescriptor());
226  if (arg0->toUnsigned().assignTo(mainVa_)) {
227  SAWYER_MESG(debug) <<"LibcStartMain analysis: m68k with main as the first argument\n";
228  functionPtrs.push_back(arg0);
229  }
230 #ifdef ROSE_ENABLE_ASM_AARCH32
231  } else if (isSgAsmAarch32Instruction(args.bblock->instructions().back())) {
232  // The "main" pointer is passed to __libc_start_main@plt in the r0 register.
233  const RegisterDescriptor REG_R0 = regs->findOrThrow("r0");
234  BaseSemantics::SValue::Ptr r0 = state->peekRegister(REG_R0, dispatcher->undefined_(32), ops.get());
235  if (r0->toUnsigned().assignTo(mainVa_)) {
236  SAWYER_MESG(debug) <<"LibcStartMain analysis: AArch32 with main in r0\n";
237  functionPtrs.push_back(r0);
238  }
239 #endif
240  }
241 
242  if (!functionPtrs.empty()) {
243  ASSERT_require(args.bblock->successors().isCached());
244  BasicBlock::Successors succs = args.bblock->successors().get();
245  for (const BaseSemantics::SValue::Ptr &calleeVa: functionPtrs) {
246  succs.push_back(BasicBlock::Successor(Semantics::SValue::promote(calleeVa), E_FUNCTION_CALL));
247  SAWYER_MESG(debug) <<"LibcStartMain analysis: fcall to " <<*calleeVa
248  <<(mainVa_ && calleeVa->toUnsigned().get() == *mainVa_ ? ", assumed to be \"main\"" : "") <<"\n";
249  }
250  args.bblock->successors() = succs;
251  }
252 
253  return true;
254 }
