69 void QtColorButtonPrivate::slotEditColor()
70 {
71  bool ok;
72  const QRgb rgba = QColorDialog::getRgba(m_color.rgba(), &ok, q_ptr);
73  if (!ok)
74  return;
75  const QColor c = QColor::fromRgba(rgba);
76  if (c == q_ptr->color())
77  return;
78  q_ptr->setColor(c);
79  emit q_ptr->colorChanged(m_color);
80 }

91 QPixmap QtColorButtonPrivate::generatePixmap() const
92 {
93  QPixmap pix(24, 24);
94 
95  int pixSize = 20;
96  QBrush br(shownColor());
97 
98  QPixmap pm(2 * pixSize, 2 * pixSize);
99  QPainter pmp(&pm);
100  pmp.fillRect(0, 0, pixSize, pixSize, Qt::lightGray);
101  pmp.fillRect(pixSize, pixSize, pixSize, pixSize, Qt::lightGray);
102  pmp.fillRect(0, pixSize, pixSize, pixSize, Qt::darkGray);
103  pmp.fillRect(pixSize, 0, pixSize, pixSize, Qt::darkGray);
104  pmp.fillRect(0, 0, 2 * pixSize, 2 * pixSize, shownColor());
105  br = QBrush(pm);
106 
107  QPainter p(&pix);
108  int corr = 1;
109  QRect r = pix.rect().adjusted(corr, corr, -corr, -corr);
110  p.setBrushOrigin((r.width() % pixSize + pixSize) / 2 + corr, (r.height() % pixSize + pixSize) / 2 + corr);
111  p.fillRect(r, br);
112 
113  p.fillRect(r.width() / 4 + corr, r.height() / 4 + corr,
114  r.width() / 2, r.height() / 2,
115  QColor(shownColor().rgb()));
116  p.drawRect(pix.rect().adjusted(0, 0, -1, -1));
117 
118  return pix;
119 }

226 void QtColorButton::mouseMoveEvent(QMouseEvent *event)
227 {
228 #ifndef QT_NO_DRAGANDDROP
229  if (event->buttons() & Qt::LeftButton &&
230  (d_ptr->m_dragStart - event->pos()).manhattanLength() > QApplication::startDragDistance()) {
231  QMimeData *mime = new QMimeData;
232  mime->setColorData(color());
233  QDrag *drg = new QDrag(this);
234  drg->setMimeData(mime);
235  drg->setPixmap(d_ptr->generatePixmap());
236  setDown(false);
237  event->accept();
238  drg->start();
239  return;
240  }
241 #endif
242  QToolButton::mouseMoveEvent(event);
243 }
