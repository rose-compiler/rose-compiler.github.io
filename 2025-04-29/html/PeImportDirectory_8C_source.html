106  memset(&disk, 0, sizeof disk);
107  }

110  if (!memcmp(&disk, &zero, sizeof zero))

166 /* Parse an Import Lookup Table or Import Address Table. The table begins at the specified address and consists of 32- or 64-
167  * bit vectors which are (1) an ordinal number with high order bit set, (2) a relative virtual address of a hint/name pair, or
168  * (3) a virtual address filled in by the dynamic linker when the shared object is bound. The last case is assumed if @p
169  * @p assume_bound is set.
170  *
171  * Normally, the ILT and IAT of an executable file have identical structure and content and the IAT is changed only after the
172  * shared objects are dynamically linked. However, we might encounter cases where we discover that the IAT has values that
173  * are different than the ILT even when @p assume_bound is false. When this happens, we emit a warning message and treat the
174  * conflicting IAT entry as a bound address. */
175 void
176 SgAsmPEImportDirectory::parse_ilt_iat(const RelativeVirtualAddress &table_start, bool assume_bound)
177 {
178  SgAsmPEImportSection *isec = SageInterface::getEnclosingNode<SgAsmPEImportSection>(this);
179  assert(isec!=nullptr);
180  SgAsmPEFileHeader *fhdr = isSgAsmPEFileHeader(isec->get_header());
181  assert(fhdr!=nullptr);
182  assert(fhdr->get_wordSize() <= sizeof(uint64_t));
183  assert(get_imports()!=nullptr);
184  SgAsmPEImportItemPtrList &imports = get_imports()->get_vector();
185  bool processing_iat = !imports.empty(); // we always process the ILT first (but it might be empty)
186 
187  if (0==table_start.rva())
188  return; // no ILT/IAT present
189 
190 
191  Address entry_va=*table_start.va(), entry_size=fhdr->get_wordSize();
192  uint64_t by_ordinal_bit = 1ull << (8*entry_size-1);
193 
194  for (size_t idx=0; 1; ++idx, entry_va+=entry_size) {
195  /* Read the 32- or 64-bit ILT/IAT entry from proces mapped memory. */
196  uint64_t entry_word = 0;
197  try {
198  isec->readContent(fhdr->get_loaderMap(), entry_va, &entry_word, entry_size);
199  } catch (const MemoryMap::NotMapped &e) {
200  if (SgAsmPEImportSection::showImportMessage()) {
201  mlog[WARN] <<"SgAsmPEImportDirectory: ILT/IAT entry #" <<idx
202  <<" at va " <<StringUtility::addrToString(entry_va)
203  <<": table entry is outside mapped memory\n";
204  if (e.map) {
205  mlog[WARN] <<" Memory map in effect at time of error:\n";
206  e.map->dump(mlog[WARN], " ");
207  }
208  }
209  }
210  entry_word = ByteOrder::leToHost(entry_word);
211 
212  /* ILT/IAT are to be terminated by a zero word. However, the length of the IAT is required to be the same as the
213  * length of the ILT. Handle three cases here:
214  * (1) Termination of the ILT by a zero entry; stop now
215  * (2) Premature termination of the IAT; continue processing
216  * (3) Missing (or late) termination of the IAT; stop now */
217  if (0==entry_word) {
218  if (idx<imports.size()) {
219  if (SgAsmPEImportSection::showImportMessage()) {
220  mlog[ERROR] <<"SgAsmPEImportDirectory: IAT entry #" <<idx
221  <<" at va " <<StringUtility::addrToString(entry_va)
222  <<": IAT zero termination occurs before end of corresponding ILT;"
223  <<(assume_bound?"":"assuming this is a bound null address and ")
224  <<" continuing to read IAT entries.\n";
225  }
226  assert(idx<imports.size());
227  imports[idx]->set_bound_rva(RelativeVirtualAddress(entry_word).bindBestSection(fhdr));
228  continue;
229  } else {
230  break;
231  }
232  } else if (processing_iat && idx>=imports.size()) {
233  if (SgAsmPEImportSection::showImportMessage()) {
234  mlog[WARN] <<"SgAsmPEImportDirectory: IAT entry #" <<idx
235  <<" at va " <<StringUtility::addrToString(entry_va)
236  <<": IAT extends beyond end of corresponding ILT (it is not zero terminated)"
237  <<"; forcibly terminating here\n";
238  }
239  break;
240  }
241 
242  /* Create the new table entry if necessary. */
243  bool entry_existed = true;
244  if (idx>=imports.size()) {
245  assert(idx==imports.size());
246  new SgAsmPEImportItem(get_imports()); // adds new item to imports list
247  entry_existed = false;
248  }
249  SgAsmPEImportItem *import_item = imports[idx];
250 
251  if (assume_bound) {
252  /* The entry word is a virtual address. */
253  if (entry_existed && import_item->get_bound_rva().rva()>0 &&
254  import_item->get_bound_rva().rva()!=entry_word) {
255  if (SgAsmPEImportSection::showImportMessage()) {
256  mlog[WARN] <<"SgAsmPEImportDirectory: ILT/IAT entry #" <<idx
257  <<" at va " <<StringUtility::addrToString(entry_va)
258  <<": bound address " <<StringUtility::addrToString(entry_word)
259  <<" conflicts with bound address already discovered "
260  <<import_item->get_bound_rva().rva() <<" (using new value)\n";
261  }
262  }
263  import_item->set_bound_rva(RelativeVirtualAddress(entry_word).bindBestSection(fhdr));
264  } else if (0!=(entry_word & by_ordinal_bit)) {
265  /* The entry is an ordinal number. */
266  if (entry_existed && import_item->get_ordinal()!=0 && import_item->get_ordinal()!=(entry_word & 0xffff)) {
267  if (SgAsmPEImportSection::showImportMessage()) {
268  mlog[WARN] <<"SgAsmPEImportDirectory: ILT/IAT entry #" <<idx
269  <<" at va " <<StringUtility::addrToString(entry_va)
270  <<": ordinal " <<(entry_word & 0xffff) <<" conflicts with ordinal already discovered "
271  <<import_item->get_ordinal()
272  <<" (assuming this is a bound address)\n";
273  }
274  import_item->set_bound_rva(RelativeVirtualAddress(entry_word).bindBestSection(fhdr));
275  } else {
276  import_item->set_by_ordinal(true);
277  import_item->set_ordinal(entry_word & 0xffff);
278  if (0!=(entry_word & ~(by_ordinal_bit | 0xffff))) {
279  if (SgAsmPEImportSection::showImportMessage()) {
280  mlog[WARN] <<"SgAsmPEImportDirectory: ILT/IAT entry #" <<idx
281  <<" at va " <<StringUtility::addrToString(entry_va)
282  <<": contains extra bits in violation of PE specification"
283  <<" (extra bits removed)\n";
284  }
285  }
286  }
287  } else {
288  /* The entry word points to a Hint/Name pair: a 2-byte, little endian hint, followed by a NUL-terminated string,
289  * followed by an optional extra byte to make the total length a multiple of two. */
290  if (entry_existed && import_item->get_by_ordinal()) {
291  if (SgAsmPEImportSection::showImportMessage()) {
292  mlog[WARN] <<"SgAsmPEImportDirectory: ILT/IAT entry #" <<idx
293  <<" at va " <<StringUtility::addrToString(entry_va)
294  <<": entry type \"hint/name\" conflicts with entry type already discovered"
295  <<" \"ordinal\" (assuming this is a bound address)\n";
296  }
297  import_item->set_bound_rva(RelativeVirtualAddress(entry_word).bindBestSection(fhdr));
298  } else if (entry_existed && import_item->get_hintname_rva().rva()>0 &&
299  import_item->get_hintname_rva().rva()!=entry_word) {
300  if (SgAsmPEImportSection::showImportMessage()) {
301  mlog[WARN] <<"SgAsmPEImportDirectory: ILT/IAT entry #" <<idx
302  <<" at va " <<StringUtility::addrToString(entry_va)
303  <<": hint/name rva " <<entry_word
304  <<" conflicts with hint/name rva already discovered "
305  <<import_item->get_hintname_rva().rva()
306  <<" (assuming this is a bound address)\n";
307  }
308  import_item->set_bound_rva(RelativeVirtualAddress(entry_word).bindBestSection(fhdr));
309  } else {
310  import_item->set_by_ordinal(false);
311  import_item->set_hintname_rva(RelativeVirtualAddress(entry_word).bindBestSection(fhdr));
312  import_item->set_hintname_nalloc(0); // for now, will adjust after we read it
313  Address entry_word_va = entry_word + fhdr->get_baseVa();
314  uint16_t hint;
315  try {
316  isec->readContent(fhdr->get_loaderMap(), entry_word_va, &hint, sizeof hint);
317  } catch (const MemoryMap::NotMapped &e) {
318  import_item->set_hint(0);
319  import_item->get_name()->set_string("");
320  if (SgAsmPEImportSection::showImportMessage()) {
321  mlog[WARN] <<"SgAsmPEImportDirectory: ILT/IAT entry #" <<idx
322  <<" at va " <<StringUtility::addrToString(entry_va)
323  <<": points to unmapped Hint/Name pair at va "
324  <<StringUtility::addrToString(entry_word_va) <<"\n";
325  if (e.map) {
326  mlog[WARN] <<" Memory map in effect at time of error:\n";
327  e.map->dump(mlog[WARN], " ");
328  }
329  }
330  continue; // to next ILT/IAT entry
331  }
332  hint = ByteOrder::leToHost(hint);
333  import_item->set_hint(hint);
334 
335  std::string s;
336  try {
337  s = isec->readContentString(fhdr->get_loaderMap(), entry_word_va+2);
338  } catch (const MemoryMap::NotMapped &e) {
339  import_item->get_name()->set_string("");
340  if (SgAsmPEImportSection::showImportMessage()) {
341  mlog[WARN] <<"SgAsmPEImportDirectory: ILT/IAT entry #" <<idx
342  <<" at va " <<StringUtility::addrToString(entry_va)
343  <<": points to an unmapped Hint/Name pair at va "
344  <<StringUtility::addrToString(entry_word_va) <<"\n";
345  if (e.map) {
346  mlog[WARN] <<" Memory map in effect at time of error:\n";
347  e.map->dump(mlog[WARN], " ");
348  }
349  }
350  continue; // to next ILT/IAT entry
351  }
352  import_item->get_name()->set_string(s);
353 
354  if ((s.size()+1) % 2) {
355  uint8_t byte = 0;
356  try {
357  isec->readContent(fhdr->get_loaderMap(), entry_word_va+2+s.size()+1, &byte, 1);
358  } catch (const MemoryMap::NotMapped &e) {
359  if (SgAsmPEImportSection::showImportMessage()) {
360  mlog[WARN] <<"SgAsmPEImportDirectory: ILT/IAT entry #" <<idx
361  <<" at va " <<StringUtility::addrToString(entry_va)
362  <<": points to an unmapped Hint/Name pair at va "
363  <<StringUtility::addrToString(entry_word_va)
364  <<" (when reading pad byte)\n";
365  if (e.map) {
366  mlog[WARN] <<" Memory map in effect at time of error:\n";
367  e.map->dump(mlog[WARN], " ");
368  }
369  }
370  continue; // to next ILT/IAT entry
371  }
372  if (byte) {
373  if (SgAsmPEImportSection::showImportMessage()) {
374  mlog[WARN] <<"SgAsmPEImportDirctory: ILT/IAT entry #" <<idx
375  <<" at va " <<StringUtility::addrToString(entry_va)
376  <<": has a non-zero padding byte: "
377  <<StringUtility::toHex2(byte, 8) <<"\n";
378  }
379  }
380  }
381  import_item->set_hintname_nalloc(import_item->hintNameRequiredSize());
382  }
383  }
384  }

387 /* Unparse a PE Import Lookup Table or Import Address Table. The @p table_start address is the location where the ILT or IAT
388  * begins and should be bound to the section to which the table is being written. The table is filled with ordinals and/or
389  * hint/name pairs unless @p assume_bound is set, in which case the table is filled with the bound addresses. The PE spec
390  * requires that the IAT in the executable file has the exact same structure and content as the ILT, so this method is
391  * normally called wtih @p assume_bound set to false.
392  *
393  * The @p tablesize argument is the number of bytes allocated for the table, and may be less than the number of bytes required
394  * to write the entire table. This can happen, for instance, when the AST was modified by adding entries to the ILT/IAT but
395  * the size (and probably location) of the table could not be changed. This is typical of IATs since code in the .text
396  * section often references IAT entries via indirect jumps/calls and it is infeasible to move the IAT to a new location. If
397  * unparsing is unable to write all table entries due to the @p tablesize limit, an error message is printed. */
398 void
399 SgAsmPEImportDirectory::unparse_ilt_iat(std::ostream &f, const RelativeVirtualAddress &table_start, bool assume_bound,
400  size_t tablesize) const
401 {
402  SgAsmPEImportSection *isec = SageInterface::getEnclosingNode<SgAsmPEImportSection>(this);
403  assert(isec!=nullptr);
404  SgAsmPEFileHeader *fhdr = isSgAsmPEFileHeader(isec->get_header());
405  assert(fhdr!=nullptr);
406  assert(fhdr->get_wordSize() <= sizeof(uint64_t));
407  const SgAsmPEImportItemPtrList &imports = get_imports()->get_vector();
408  size_t entry_size = fhdr->get_wordSize();
409  uint64_t by_ordinal_bit = 1ull << (8*entry_size-1);
410 
411  if (0==table_start.rva() || imports.empty())
412  return;
413  if (!table_start.boundSection()) {
414  if (SgAsmPEImportSection::showImportMessage()) {
415  mlog[WARN] <<"SgAsmPEImportDirectory: ILT/IAT: table is not associated with a section: "
416  <<table_start.toString() <<"\n";
417  }
418  return;
419  }
420 
421  /* Must we limit how many entries are written? Don't forget the zero terminator. */
422  size_t nelmts = std::min(tablesize/entry_size, imports.size()+1);
423  if (nelmts<imports.size()+1) {
424  if (SgAsmPEImportSection::showImportMessage()) {
425  mlog[WARN] <<"SgAsmPEImportDirectory: ILT/IAT: table at " <<table_start.toString()
426  <<" is truncated from " <<StringUtility::plural(imports.size()+1, "entries", "entry")
427  <<" to " <<StringUtility::plural(nelmts, "entries", "entry")
428  <<" (inc. zero terminator) due to allocation constraints.\n";
429  }
430  }
431 
432  RelativeVirtualAddress entry_rva = table_start;
433  for (size_t idx=0; idx<nelmts/*including zero terminator*/; ++idx, entry_rva.increment(entry_size)) {
434  uint64_t entry_word = 0;
435 
436  /* Write the zero terminator? */
437  if (idx+1==nelmts) {
438  uint64_t zero = 0;
439  entry_rva.boundSection()->write(f, *entry_rva.boundOffset(), entry_size, &zero);
440  break;
441  }
442 
443  RelativeVirtualAddress hn_rva = imports[idx]->get_hintname_rva();
444 
445  /* Build the IAT/ILT entry */
446  if (assume_bound) {
447  entry_word = imports[idx]->get_bound_rva().rva();
448  } else if (imports[idx]->get_by_ordinal()) {
449  if (0!=(imports[idx]->get_ordinal() & ~0xffff)) {
450  if (SgAsmPEImportSection::showImportMessage()) {
451  mlog[WARN] <<"SgAsmPEImportDirectory: ILT/IAT entry #" <<idx
452  <<" at rva " <<entry_rva.rva()
453  <<": ordinal is out of bounds: " <<imports[idx]->get_ordinal() <<" (truncated to 16 bits)\n";
454  }
455  }
456  entry_word |= by_ordinal_bit | (imports[idx]->get_ordinal() & 0xffff);
457  } else if (0==hn_rva.rva() || nullptr==hn_rva.boundSection()) {
458  if (SgAsmPEImportSection::showImportMessage()) {
459  mlog[WARN] <<"SgAsmPEImportDirectory: ILT/IAT entry #" <<idx
460  <<" at rva " <<entry_rva.rva()
461  <<": non-ordinal entry has invalid hint/name pair address " <<hn_rva
462  <<" or is not associated with any section.\n";
463  }
464  entry_word = hn_rva.rva(); // best we can do
465  } else {
466  size_t bufsz = std::min(imports[idx]->get_hintname_nalloc(), imports[idx]->hintNameRequiredSize());
467  if (bufsz < imports[idx]->hintNameRequiredSize()) {
468  if (SgAsmPEImportSection::showImportMessage()) {
469  mlog[WARN] <<"SgAsmPEImportDirectory: ILT/IAT entry #" <<idx
470  <<" at rva " <<entry_rva.rva()
471  <<": insufficient space (" <<StringUtility::plural(bufsz, "bytes") <<")"
472  <<" allocated for hint/name pair at rva " <<hn_rva
473  <<" (need " <<StringUtility::plural(imports[idx]->hintNameRequiredSize(), "bytes") <<")\n";
474  }
475  }
476  if (bufsz>=2) {
477  std::vector<uint8_t> buf(bufsz, 0);
478  unsigned hint = imports[idx]->get_hint();
479  std::string name = imports[idx]->get_name()->get_string();
480  if (0!=(hint & ~0xffff)) {
481  if (SgAsmPEImportSection::showImportMessage()) {
482  mlog[WARN] <<"SgAsmPEImportDirectory: ILT/IAT entry #" <<idx
483  <<" at rva " <<entry_rva.rva()
484  <<": hint is out of bounds: " <<hint <<" (truncated to 16 bits)\n";
485  }
486  }
487  ByteOrder::hostToLe(hint, &hint); // nudge, nudge. Know what I mean?
488  memcpy(&buf[0], &hint, 2);
489  memcpy(&buf[2], name.c_str(), std::min(name.size()+1, bufsz-2));
490  if (bufsz>2)
491  buf[bufsz-1] = '\0';
492  hn_rva.boundSection()->write(f, *hn_rva.boundOffset(), bufsz, &buf[0]);
493  if (0!=(hn_rva.rva() & by_ordinal_bit)) {
494  if (SgAsmPEImportSection::showImportMessage()) {
495  mlog[WARN] <<"SgAsmPEImportDirectory: ILT/IAT entry #" <<idx
496  <<" at rva " <<entry_rva.rva()
497  <<": hint/name pair rva " <<hn_rva.rva()
498  <<" has by_ordinal bit set\n";
499  }
500  }
501  }
502  entry_word = hn_rva.rva() & ~by_ordinal_bit;
503  }
504 
505  /* Write the IAT/ILT entry */
506  if (0==entry_word) {
507  if (SgAsmPEImportSection::showImportMessage()) {
508  mlog[WARN] <<"SgAsmPEImportDirectory: ILT/IAT entry #" <<idx
509  <<" at rva " <<entry_rva.rva()
510  <<": zero table entry will cause table to be truncated when read\n";
511  }
512  }
513  uint64_t disk = 0; // we might write only the first few bytes
514  ByteOrder::hostToLe(entry_word, &disk);
515  entry_rva.boundSection()->write(f, *entry_rva.boundOffset(), entry_size, &disk);
516  }

530 size_t
531 SgAsmPEImportDirectory::reallocate(RelativeVirtualAddress start_rva)
532 {
533  RelativeVirtualAddress end_rva = start_rva;
534  const SgAsmPEImportItemPtrList &imports = get_imports()->get_vector();
535 
536  /* Allocate space for the name if it hasn't been allocated already; reallocate space if its allocated in the import
537  * section. Allocate space even if the name is the empty string. */
538  if (0==get_dllNameRva().rva() || get_dllNameRva().boundSection()==end_rva.boundSection()) {
539  p_dll_name_nalloc = get_dllName()->get_string().size() + 1;
540  p_dllNameRva = end_rva;
541  end_rva.increment(p_dll_name_nalloc);
542  }
543 
544  /* Allocate space for the import lookup table if it hasn't been allocated yet. The table is terminated with a zero
545  * entry (as is the IAT according to the spec even though the IAT size is implied by the ILT size). */
546  if (0==p_ilt_rva.rva() || p_ilt_rva.boundSection()==end_rva.boundSection()) {
547  p_ilt_nalloc = iatRequiredSize(); // ILT and IAT are always the same size
548  p_ilt_rva = end_rva;
549  end_rva.increment(p_ilt_nalloc);
550  }
551 
552  /* Allocate space for the import address table if it hasn't been allocated yet. Note, the import address table should
553  * usually be allocated explicitly because it is referenced by the program's instructions. */
554  if (0==p_iat_rva.rva() || p_iat_rva.boundSection()==end_rva.boundSection()) {
555  p_iat_nalloc = iatRequiredSize();
556  p_iat_rva = end_rva;
557  end_rva.increment(p_iat_nalloc);
558  }
559 
560  /* Allocate space for the Hint/Name pairs that haven't been allocated yet. The ILT and IAT will both point to the same
561  * hint/name pair when the IAT is a copy of the ILT. When the IAT is unparsed as bound addresses, then space for the
562  * hint/name pair is still needed because the ILT still points to it. We allocate space even for hint/name pairs that are
563  * zero and the empty string. We don't reallocate hint/name pairs that are stored in a different file section (so be
564  * careful that you don't increase the length of the name). */
565  for (size_t i=0; i<imports.size(); ++i) {
566  if (!imports[i]->get_by_ordinal() &&
567  (0==imports[i]->get_hintname_rva() || imports[i]->get_hintname_rva().boundSection()==end_rva.boundSection())) {
568  size_t sz = imports[i]->hintNameRequiredSize();
569  imports[i]->set_hintname_nalloc(sz);
570  imports[i]->set_hintname_rva(end_rva);
571  end_rva.increment(sz);
572  }
573  }
574 
575  return end_rva.rva() - start_rva.rva();
576 }

578 void
579 SgAsmPEImportDirectory::unparse(std::ostream &f, const SgAsmPEImportSection *section, size_t idx) const
580 {
581  /* The DLL name */
582  if (0==get_dllNameRva().rva() || nullptr==get_dllNameRva().boundSection()) {
583  if (SgAsmPEImportSection::showImportMessage()) {
584  mlog[WARN] <<"SgAsmPEImportDirectory: DLL name address " <<get_dllNameRva()
585  <<" is invalid or not bound to any file section\n";
586  }
587  } else if (get_dllName()->get_string().size()+1 > p_dll_name_nalloc) {
588  if (SgAsmPEImportSection::showImportMessage()) {
589  mlog[WARN] <<"SgAsmPEImportDirectory: insufficient space allocated ("
590  <<StringUtility::plural(p_dll_name_nalloc, "bytes") <<")"
591  <<" for DLL name (need " <<StringUtility::plural(get_dllName()->get_string().size()+1, "bytes") <<")\n";
592  }
593  } else if (p_dll_name_nalloc > 0) {
594  std::vector<uint8_t> buf(p_dll_name_nalloc, 0);
595  memcpy(&buf[0], get_dllName()->get_string().c_str(), get_dllName()->get_string().size());
596  get_dllNameRva().boundSection()->write(f, *get_dllNameRva().boundOffset(), p_dll_name_nalloc, &buf[0]);
597  }
598 
599  /* Import Lookup Table and Import Address Table (and indirectly, Hint/Name entries). According to the PE secification: "The
600  * structure and content of the import address table are identical to those of the import lookup table, until the file is
601  * bound." */
602  if (p_ilt_rva>0)
603  unparse_ilt_iat(f, p_ilt_rva, false, p_ilt_nalloc);
604  if (p_iat_rva>0)
605  unparse_ilt_iat(f, p_iat_rva, false, p_iat_nalloc);
606 
607  PEImportDirectory_disk disk;
608  encode(&disk);
609  section->write(f, idx*sizeof disk, sizeof disk, &disk);
610 }
