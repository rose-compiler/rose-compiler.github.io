26 // Adjust the behavior for failed assertions based on the command-line
27 class FailedAssertionBehaviorAdjuster: public Sawyer::CommandLine::SwitchAction {
28 protected:
29  FailedAssertionBehaviorAdjuster() {}
30 public:
31  typedef Sawyer::SharedPointer<FailedAssertionBehaviorAdjuster> Ptr;
32  enum Behavior { ABORT_ON_FAILURE, EXIT_ON_FAILURE, THROW_ON_FAILURE };
33  static Ptr instance() {
34  return Ptr(new FailedAssertionBehaviorAdjuster);
35  }
36 protected:
37  void operator()(const Sawyer::CommandLine::ParserResult &cmdline) {
38  ASSERT_require(cmdline.have("assert"));
39  Sawyer::Assert::AssertFailureHandler handler = NULL;
40  switch (cmdline.parsed("assert", 0).as<Behavior>()) {
41  case ABORT_ON_FAILURE: handler = abortOnFailedAssertion; break;
42  case EXIT_ON_FAILURE: handler = exitOnFailedAssertion; break;
43  case THROW_ON_FAILURE: handler = throwOnFailedAssertion; break;
44  }
45  failedAssertionBehavior(handler);
46  }
47 };
