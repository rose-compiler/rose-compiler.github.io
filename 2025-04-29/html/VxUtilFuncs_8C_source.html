144  const SgDeclarationModifier& declMod = sgDeclaration->get_declarationModifier();
145  const SgStorageModifier& storageMod = declMod.get_storageModifier();
146  if(isSgGlobal(sgDeclaration->get_scope()) && storageMod.isStatic()) {
147  return true;

149 
150  //I think it's only file scope if the namespace right before global is anonymous

154  SgStatement* scope = sgDeclaration->get_scope();

157  prevScope = scope;

165  if(nspaceDecl) {

172
