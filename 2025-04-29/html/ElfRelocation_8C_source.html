117 SgAsmElfRelocEntry::dump(FILE *f, const char *prefix, ssize_t idx, SgAsmElfSymbolSection *symtab) const
118 {
119  char p[4096];
120  if (idx>=0) {
121  snprintf(p, sizeof(p), "%sElfRelocEntry[%zd]", prefix, idx);
122  } else {
123  snprintf(p, sizeof(p), "%sElfRelocEntry", prefix);
124  }
125  const int w = std::max(size_t{1}, DUMP_FIELD_WIDTH - strlen(p));
126 
127  /* compact one-line-per-reloc format */
128  if (0==idx) {
129  fprintf(f, "%s%-*s %-10s %-20s %-10s %4s %-10s %s\n",
130  p, w, "", "Offset", "Type", "Addend", "Sym", "Value", "Name + Addend");
131  }
132 
133  /* Offset */
134  fprintf(f, "%s%-*s = 0x%08" PRIx64, p, w, "", p_r_offset);
135 
136  /* Type */
137  SgAsmGenericHeader* header = SageInterface::getEnclosingNode<SgAsmGenericHeader>(this);
138  if (header) {
139  fprintf(f, " %-20s", toString().c_str());
140  } else {
141  fprintf(f, " 0x%02zx ", (size_t)p_type);
142  }
143 
144  /* Addend */
145  if (p_r_addend) {
146  fprintf(f, " 0x%08" PRIx64, p_r_addend);
147  } else {
148  fprintf(f, " %10s", "");
149  }
150 
151  /* Symbol index */
152  fprintf(f, " %4lu", p_sym);
153 
154  /* Symbol value and name */
155  if (!symtab) {
156  fprintf(f, " %10s <no symtab>", "");
157  } else if (p_sym>=symtab->get_symbols()->get_symbols().size()) {
158  fprintf(f, " %10s <out of range>", "");
159  } else {
160  SgAsmGenericSymbol *sym = symtab->get_symbols()->get_symbols()[p_sym];
161  fprintf(f, " 0x%08" PRIx64 " %s", sym->get_value(), sym->get_name()->get_string(true).c_str());
162  }
163 
164  /* Addend in decimal */
165  if (p_r_addend)
166  fprintf(f, " + %" PRIu64, p_r_addend);
167  fputc('\n', f);
168 
169  /* Auxiliary data */
170  if (p_extra.size()>0) {
171  fprintf(f, "%s%-*s = %" PRIuPTR " bytes\n", p, w, ".extra", p_extra.size());
172  Rose::BinaryAnalysis::hexdump(f, 0, std::string(p)+"extra at ", p_extra);
173  }

184 SgAsmElfRelocSection::parse()
185 {
186  SgAsmElfSection::parse();
187 
188  SgAsmElfFileHeader *fhdr = get_elfHeader();
189  ROSE_ASSERT(fhdr);
190 
191  size_t entry_size, struct_size, extra_size, nentries;
192  calculateSizes(&entry_size, &struct_size, &extra_size, &nentries);
193  ROSE_ASSERT(extra_size==0);
194 
195  /* Parse each entry */
196  for (size_t i=0; i<nentries; i++) {
197  SgAsmElfRelocEntry *entry = 0;
198  if (4==fhdr->get_wordSize()) {
199  if (get_usesAddend()) {
200  SgAsmElfRelocEntry::Elf32RelaEntry_disk disk;
201  readContentLocal(i*entry_size, &disk, struct_size);
202  entry = new SgAsmElfRelocEntry(this);
203  entry->parse(fhdr->get_sex(), &disk);
204  } else {
205  SgAsmElfRelocEntry::Elf32RelEntry_disk disk;
206  readContentLocal(i*entry_size, &disk, struct_size);
207  entry = new SgAsmElfRelocEntry(this);
208  entry->parse(fhdr->get_sex(), &disk);
209  }
210  } else if (8==fhdr->get_wordSize()) {
211  if (get_usesAddend()) {
212  SgAsmElfRelocEntry::Elf64RelaEntry_disk disk;
213  readContentLocal(i*entry_size, &disk, struct_size);
214  entry = new SgAsmElfRelocEntry(this);
215  entry->parse(fhdr->get_sex(), &disk);
216  } else {
217  SgAsmElfRelocEntry::Elf64RelEntry_disk disk;
218  readContentLocal(i*entry_size, &disk, struct_size);
219  entry = new SgAsmElfRelocEntry(this);
220  entry->parse(fhdr->get_sex(), &disk);
221  }
222  } else {
223  throw FormatError("unsupported ELF word size");
224  }
225  if (extra_size>0)
226  entry->get_extra() = readContentLocalUcl(i*entry_size+struct_size, extra_size);
227  }
228  return this;

270 void
271 SgAsmElfRelocSection::unparse(std::ostream &f) const
272 {
273  SgAsmElfFileHeader *fhdr = get_elfHeader();
274  ROSE_ASSERT(fhdr);
275  Rose::BinaryAnalysis::ByteOrder::Endianness sex = fhdr->get_sex();
276 
277  size_t entry_size, struct_size, extra_size, nentries;
278  calculateSizes(&entry_size, &struct_size, &extra_size, &nentries);
279 
280  /* Adjust the entry size stored in the ELF Section Table */
281  get_sectionEntry()->set_sh_entsize(entry_size);
282 
283  /* Write each entry's required part followed by the optional part */
284  for (size_t i=0; i<nentries; i++) {
285  SgAsmElfRelocEntry::Elf32RelaEntry_disk diska32;
286  SgAsmElfRelocEntry::Elf64RelaEntry_disk diska64;
287  SgAsmElfRelocEntry::Elf32RelEntry_disk disk32;
288  SgAsmElfRelocEntry::Elf64RelEntry_disk disk64;
289  void *disk = NULL;
290 
291  SgAsmElfRelocEntry *entry = p_entries->get_entries()[i];
292 
293  if (4==fhdr->get_wordSize()) {
294  if (get_usesAddend()) {
295  disk = entry->encode(sex, &diska32);
296  } else {
297  disk = entry->encode(sex, &disk32);
298  }
299  } else if (8==fhdr->get_wordSize()) {
300  if (get_usesAddend()) {
301  disk = entry->encode(sex, &diska64);
302  } else {
303  disk = entry->encode(sex, &disk64);
304  }
305  } else {
306  ROSE_ASSERT(!"unsupported word size");
307  }
308 
309  Rose::BinaryAnalysis::Address spos = i * entry_size;
310  spos = write(f, spos, struct_size, disk);
311 #if 0 /*FIXME: padding not supported here yet (RPM 2008-10-13)*/
312  if (entry->get_extra().size()>0) {
313  ROSE_ASSERT(entry->get_extra().size()<=extra_size);
314  write(f, spos, entry->get_extra());
315  }
316 #endif
317  }
318 
319  unparseHoles(f);
320 }
