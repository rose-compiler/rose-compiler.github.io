255 inline int Q3TextString::nextCursorPosition(int next)
256 {
257  if (bidiDirty)
258  checkBidi();
259 
260  const Q3TextStringChar *c = data.data();
261  int len = length();
262 
263  if (next < len - 1) {
264  next++;
265  while (next < len - 1 && !c[next].charStop)
266  next++;
267  }
268  return next;
269 }
270 
271 inline int Q3TextString::previousCursorPosition(int prev)
272 {
273  if (bidiDirty)
274  checkBidi();
275 
276  const Q3TextStringChar *c = data.data();
277 
278  if (prev) {
279  prev--;
280  while (prev && !c[prev].charStop)
281  prev--;
282  }
283  return prev;
284 }
