146 ConcolicExecutor::startProcess() {
147  ASSERT_require(testCase());
148  ASSERT_require(process());
149 
150  process()->load(tmpDir_.name());
151 
152  if (testCase()->parent()) {
153  // This test case was created by cloning another test case, which usually happens when the parent test case encounters
154  // a conditional branch that depends on some input. Therefore, we need to fast forward this process so that it appears
155  // to have just executed that branch.
156  process()->playAllEvents(partitioner());
157 
158  } else {

160  // Therefore delete all existing execution events (left over from a prior concolic execution) and create the events that
161  // would initialize memory and registers to their current (initial) state.
162  process()->saveEvents(process()->createMemoryRestoreEvents(), When::PRE);
163  process()->saveEvents(process()->createRegisterRestoreEvents(), When::PRE);
164  }

318 bool
319 ConcolicExecutor::updateCallStack(SgAsmInstruction *insn) {
320  ASSERT_not_null(insn); // the instruction that was just executed
321 
322  Sawyer::Message::Stream debug(mlog[DEBUG]);
323  bool wasChanged = false;
324 
325  Emulation::RiscOperators::Ptr ops = cpu()->emulationOperators();
326  P2::BasicBlock::Ptr bb = partitioner()->basicBlockContainingInstruction(insn->get_address());
327  if (!bb || bb->nInstructions() == 0) {
328  SAWYER_MESG(debug) <<"no basic block at " <<StringUtility::addrToString(insn->get_address()) <<"\n";
329  return wasChanged; // we're executing something that ROSE didn't disassemble
330  }
331 
332  // Get the current stack pointer.
333  const RegisterDescriptor SP = partitioner()->instructionProvider().stackPointerRegister();
334  if (SP.isEmpty())
335  return wasChanged; // no stack pointer for this architecture?!
336  SymbolicExpression::Ptr sp =
337  Emulation::SValue::promote(ops->peekRegister(SP, ops->undefined_(SP.nBits())))->get_expression();
338  Sawyer::Optional<uint64_t> stackVa = sp->toUnsigned();
339  if (!stackVa) {
340  SAWYER_MESG(debug) <<"no concrete SP after function call at " <<StringUtility::addrToString(insn->get_address());
341  return wasChanged; // no concrete stack pointer. This should be unusual
342  }
343 
344  // If the stack pointer is larger than the saved stack pointer for the most recent function on the stack (assuming
345  // stack-grows-down), then pop that function from the call stack under the assumption that we've returned from that
346  // function.
347  while (!functionCallStack_.empty() && functionCallStack_.back().stackVa < *stackVa) {
348  SAWYER_MESG(debug) <<"assume returned from " <<functionCallStack_.back().printableName <<"\n";
349  functionCallStack_.pop_back();
350  wasChanged = true;
351  }
352 
353  // If the last instruction we just executed looks like a function call then push a new record onto our function call stack.
354  if (partitioner()->basicBlockIsFunctionCall(bb) && insn->get_address() == bb->instructions().back()->get_address()) {
355  // Get a name for the function we just called, if possible.
356  const RegisterDescriptor IP = partitioner()->instructionProvider().instructionPointerRegister();
357  SymbolicExpression::Ptr ip =
358  Emulation::SValue::promote(ops->peekRegister(IP, ops->undefined_(IP.nBits())))->get_expression();
359  Sawyer::Optional<uint64_t> calleeVa = ip->toUnsigned();
360  std::string calleeName;
361  if (calleeVa) {
362  if (P2::Function::Ptr callee = partitioner()->functionExists(*calleeVa)) {
363  calleeName = callee->printableName();
364  } else {
365  calleeName = "function " + StringUtility::addrToString(*calleeVa);
366  }
367  } else {
368  SAWYER_MESG(debug) <<"no concrete IP after function call at " <<StringUtility::addrToString(insn->get_address());
369  return wasChanged;
370  }
371 
372  // Push this function call onto the stack.
373  SAWYER_MESG(debug) <<"assume call to " <<calleeName <<" at " <<StringUtility::addrToString(insn->get_address()) <<"\n";
374  functionCallStack_.push_back(FunctionCall(calleeName, insn->get_address(), calleeVa.orElse(0), *stackVa));
375  wasChanged = true;
376  }
377 
378  return wasChanged;
379 }

387 void
388 ConcolicExecutor::handleBranch(SgAsmInstruction *insn) {
389  ASSERT_not_null(insn);
390  ASSERT_not_null(solver());
391 
392  Sawyer::Message::Stream debug(mlog[DEBUG]);
393  Sawyer::Message::Stream error(mlog[ERROR]);
394 
395  Emulation::RiscOperators::Ptr ops = cpu()->emulationOperators();
396  const RegisterDescriptor IP = partitioner()->instructionProvider().instructionPointerRegister();
397 
398  // If we processed a branch instruction whose condition depended on input variables, then the instruction pointer register
399  // in the symbolic state will be non-constant. It should be an if-then-else expression that evaluates to two constants, the
400  // true and false execution addresses, one of which should match the concrete execution address.
401  SymbolicExpression::Ptr ip = Emulation::SValue::promote(ops->peekRegister(IP, ops->undefined_(IP.nBits())))->get_expression();
402  SymbolicExpression::InteriorPtr inode = ip->isInteriorNode();
403  if (inode && inode->getOperator() == SymbolicExpression::OP_ITE) {
404  SymbolicExpression::Ptr actualTarget = SymbolicExpression::makeIntegerConstant(IP.nBits(),
405  cpu()->concreteInstructionPointer());
406  SymbolicExpression::Ptr trueTarget = inode->child(1); // true branch next va
407  SymbolicExpression::Ptr falseTarget = inode->child(2); // false branch next va
408  SymbolicExpression::Ptr followedCond; // condition for the branch that is followed
409  SymbolicExpression::Ptr notFollowedTarget; // address that wasn't branched to
410  if (!trueTarget->isIntegerConstant()) {
411  error <<"expected constant value for true branch target at " <<partitioner()->unparse(insn) <<"\n";
412  } else if (!falseTarget->isIntegerConstant()) {
413  error <<"expected constant value for false branch target at " <<partitioner()->unparse(insn) <<"\n";
414  } else if (actualTarget->mustEqual(trueTarget)) {
415  followedCond = inode->child(0); // taking the true branch
416  notFollowedTarget = inode->child(2); // the false target address
417  } else if (actualTarget->mustEqual(falseTarget)) {
418  followedCond = SymbolicExpression::makeInvert(inode->child(0)); // taking false branch
419  notFollowedTarget = inode->child(1); // the true target address
420  } else {
421  error <<"unrecognized symbolic execution address after " <<partitioner()->unparse(insn) <<"\n"
422  <<" concrete = " <<*actualTarget <<"\n"
423  <<" symbolic = " <<*ip <<"\n";
424  }
425  SymbolicExpression::Ptr otherCond = SymbolicExpression::makeInvert(followedCond); // condition for branch not taken
426 
427  // Solve for branch not taken in terms of input values.
428  if (otherCond) {
429  SAWYER_MESG(debug) <<"condition for other path is " <<*otherCond <<"\n";
430  SmtSolver::Transaction transaction(solver()); // because we'll need to cancel in order to follow the correct branch
431  solver()->insert(otherCond);
432  switch (solver()->check()) {
433  case SmtSolver::SAT_YES:
434  if (debug) {
435  debug <<"conditions are satisfied when:\n";
436  for (const std::string &varName: solver()->evidenceNames()) {
437  ExecutionEvent::Ptr inputEvent = ops->inputVariables()->event(varName);
438  if (inputEvent) {
439  debug <<" " <<inputEvent->name() <<" (" <<varName <<") = ";
440  } else {
441  debug <<" " <<varName <<" (no input event) = ";
442  }
443  if (SymbolicExpression::Ptr val = solver()->evidenceForName(varName)) {
444  debug <<*val <<"\n";
445  } else {
446  debug <<" = ???\n";
447  }
448  }
449  }
450  generateTestCase(ops, notFollowedTarget);
451  break;
452 
453  case SmtSolver::SAT_UNKNOWN:
454  SAWYER_MESG(debug) <<"satisfiability is unknown (possible time out); assuming unsat\n";
455  // fall through
456  case SmtSolver::SAT_NO:
457  SAWYER_MESG(debug) <<"conditions cannot be satisified; path is not feasible\n";
458  break;
459  }
460  }
461 
462  // Add the branch taken condition to the solver since all future assertions will also depend on having taken this branch.
463  solver()->insert(followedCond);
464  } else if (inode) {
465  SAWYER_MESG(mlog[ERROR]) <<"instruction pointer expression no handled (not a constant or ITE): " <<*inode <<"\n";
466  }
467 }

469 void
470 ConcolicExecutor::run() {
471  Sawyer::Message::Stream debug(mlog[DEBUG]);
472  Sawyer::Message::Stream trace(mlog[TRACE]);
473  Sawyer::Message::Stream where(mlog[WHERE]);
474  Sawyer::Message::Stream error(mlog[ERROR]);
475 
476  Emulation::RiscOperators::Ptr ops = cpu()->emulationOperators();
477  ops->printInputVariables(debug);
478  ops->printAssertions(debug);
479  if (debug) {
480  BS::Formatter fmt;
481  fmt.set_line_prefix(" ");
482  debug <<"initial state:\n" <<(*ops->currentState() + fmt);
483  }
484 
485  if (mlog[DEBUG]) {
486  mlog[DEBUG] <<"partitioner memory map:\n";
487  partitioner_->memoryMap()->dump(mlog[DEBUG], " ");
488  }
489 
490  // Set up periodic state reporting
491  Sawyer::ProgressBar<size_t> nInstructions(mlog[MARCH], "concolic execution");
492  nInstructions.suffix(" instructions");
493  Sawyer::Stopwatch timeSinceLastReport;
494  IS::SymbolicSemantics::Formatter formatter;
495  formatter.registerDictionary(cpu()->registerDictionary());
496  formatter.set_line_prefix(" ");
497  formatter.expr_formatter.max_depth = 5; // arbitrary
498  size_t reportInsns = 0; // number of instructions remaining to be reported
499 
500  // Process instructions in execution order
501  Address executionVa = cpu()->concreteInstructionPointer();
502  while (!cpu()->isTerminated()) {
503  BOOST_SCOPE_EXIT(ops) {
504  ops->process()->nextInstructionLocation();
505  } BOOST_SCOPE_EXIT_END;
506 
507  SgAsmInstruction *insn = partitioner()->instructionProvider()[executionVa];
508 
509  // FIXME[Robb Matzke 2020-07-13]: I'm not sure how this ends up happening yet. Perhaps one way is because we don't
510  // handle certain system calls like mmap, brk, etc.
511  //
512  // Update[Robb Matzke 2021-12-15]: It was because ROSE disassembled the "run::a.out" specimen with one memory mapping
513  // when the partitioner result was computed, but because of address space layout randomization (ASLR), the subsequent
514  // runs each had different mappings. The fix was to change the "run::" loading method so it turns off ASLR by
515  // default. The comment about mmap, brk, etc. still stands, however.
516  if (!insn) {
517  std::vector<uint8_t> buf = ops->process()->readMemory(executionVa, 64);
518  if (buf.empty()) {
519  mlog[FATAL] <<"no memory mapped at " <<StringUtility::addrToString(executionVa) <<"\n";
520  } else {
521  mlog[FATAL] <<"no executable memory mapped at " <<StringUtility::addrToString(executionVa) <<"\n";
522  mlog[FATAL] <<StringUtility::plural(buf.size(), "bytes") <<" starting at this address:\n";
523  hexdump(mlog[FATAL], executionVa, " ", buf, true);
524  }
525  ASSERT_not_reachable("terminating due to prior error");
526  }
527 
528  // Process the instruction concretely and symbolically
529  try {
530  if (where) {
531  if (timeSinceLastReport.report() > 60 /*seconds, arbitrary*/) {
532  where <<"periodic concolic execution state:\n"
533  <<" before instruction #" <<ops->process()->currentLocation().primary()
534  <<" " <<partitioner()->unparse(insn) <<"\n";
535  ops->currentState()->registerState()->print(where, formatter);
536  timeSinceLastReport.restart();
537  reportInsns = 10; // arbitrary number of subsequent instructions to report
538  }
539  if (reportInsns > 0) {
540  if (!trace && !debug) { // Emulation::Dispatcher::processInstruction also prints things
541  where <<"executing insn #" <<ops->process()->currentLocation().primary()
542  <<" " <<partitioner()->unparse(insn) <<"\n";
543  }
544  --reportInsns;
545  }
546  }
547  ++nInstructions;
548  cpu()->processInstruction(insn);
549  } catch (const BS::Exception &e) {
550  if (error) {
551  error <<e.what() <<", occurred at:\n";
552  printCallStack(error);
553  error <<" insn " <<partitioner()->unparse(insn) <<"\n";
554  error <<"register state at time of error:\n" <<(*ops->currentState()->registerState()+" ");
555  }
556 // TEMPORARILY COMMENTED OUT FOR DEBUGGING [Robb Matzke 2020-07-13]. This exception is thrown when we get an address wrong,
557 // like for "mov eax, gs:[16]", and the debugger is reporting that the memory cannot be read. We should probably be using
558 // debugger-specific exceptions for this kind of thing.
559 // } catch (std::runtime_error &e) {
560 // if (error) {
561 // error <<e.what() <<", occurred at:\n";
562 // printCallStack(error);
563 // error <<" insn " <<partitioner()->unparse(insn) <<"\n";
564 // error <<"machine state at time of error:\n" <<(*ops->currentState()+" ");
565 // }
566  }
567 
568 #if 0 // DEBUGGING [Robb Matzke 2021-12-10]
569  SAWYER_MESG(debug) <<"symbolic state after instruction:\n" <<*ops;
570 #endif
571 
572  if (cpu()->isTerminated()) {
573  SAWYER_MESG_OR(trace, debug) <<"subordinate has terminated\n";
574  break;
575  }
576 
577  if (ops->hadSystemCall())
578  ops->process()->systemCall(partitioner(), ops);
579 
580  if (ops->hadSharedMemoryAccess())
581  ops->process()->sharedMemoryAccessPost(partitioner(), ops);
582 
583  if (settings_.showingStates.exists(executionVa))
584  SAWYER_MESG(debug) <<"state after instruction:\n" <<(*ops->currentState()+" ");
585 
586  executionVa = cpu()->concreteInstructionPointer();
587  if (updateCallStack(insn) && where) {
588  where <<"function call stack:\n";
589  printCallStack(where);
590  }
591

596 void
597 ConcolicExecutor::generateTestCase(const BS::RiscOperators::Ptr &ops_, const SymbolicExpression::Ptr &childIp) {
598  ASSERT_not_null(ops_);
599 
600  TestCase::Ptr oldTestCase = testCase();
601  TestCaseId oldTestCaseId = testCaseId();
602  ASSERT_not_null(oldTestCase);
603  ASSERT_require(oldTestCaseId);
604  ASSERT_not_null(solver()); // assertions must have been checked and satisfiable
605 
606  Sawyer::Message::Stream debug(mlog[DEBUG]);
607  Sawyer::Message::Stream info(mlog[INFO]);
608  Sawyer::Message::Stream error(mlog[ERROR]);
609  SAWYER_MESG(debug) <<"generating new test case...\n";
610 
611  // During this function, set the RiscOperators state to be that of the child. This will allow us to easily update the child
612  // state before saving it, without affecting the parent.
613  auto ops = Emulation::RiscOperators::promote(ops_);
614  BS::State::Ptr parentState = ops->currentState();
615  BS::State::Ptr childState = parentState->clone();
616  ops->currentState(childState);
617  BOOST_SCOPE_EXIT(ops, parentState) {
618  ops->currentState(parentState);
619  } BOOST_SCOPE_EXIT_END;
620 
621  // The instruction pointer in the child is the childIp, which must be concrete, not the (probably) symbolic value
622  // resulting from a conditional branch in the oldTestCase.
623  ASSERT_require(childIp->isIntegerConstant());
624  const RegisterDescriptor IP = partitioner()->instructionProvider().instructionPointerRegister();
625  ops->writeRegister(IP, Emulation::SValue::instance_symbolic(childIp));
626 
627  // Create execution events for the child test case based on the execution events from the parent test case, but do not
628  // commit the child events to the database yet because we don't know for sure that we're going to create the child test
629  // case. We'll abandon all these child execution events if we decide not to create the child test case.
630  Sawyer::Container::Map<size_t, ExecutionEvent::Ptr> eventMap;
631  for (ExecutionEventId parentEventId: database()->executionEvents(oldTestCaseId)) {
632  ExecutionEvent::Ptr parentEvent = database()->object(parentEventId);
633  ExecutionEvent::Ptr childEvent = parentEvent->copy();
634  eventMap.insert(*parentEventId, childEvent);
635  }
636 
637  // Get the parent's argc, argument list, and environment list which we'll use as the child's, with possibly some
638  // modifications. These values are mostly redundant for concolic execution since the execution events are what really
639  // control what inputs the test case gets, but they may be necessary for the concrete execution used to rank test cases,
640  // and when running a test case manually. We need to add the NUL terminator to the ends of the arguments while we're
641  // making adjustments (because it's technically part of the argument), but then we'll remove it at the very end since all
642  // C strings (and Linux arguments) are implicitly NUL terminated.
643  std::vector<std::string> argv = oldTestCase->args();
644  for (std::string &arg: argv)
645  arg += '\0';
646  SymbolicExpression::Ptr argc = SymbolicExpression::makeIntegerConstant(32, argv.size());
647  std::vector<std::string> env;
648  for (const EnvValue &pair: oldTestCase->env())
649  env.push_back(pair.first + "=" + pair.second);
650 
651  // Process each piece of the solver evidence. These are the symbolic variables and their values that would cause test case
652  // execution to have gone some other way at the current control flow branch. Each of the variables in the evidence will
653  // correspond to a known symbolic input, which in turn corresponds to a concrete execution event.
654  std::vector<ExecutionEvent::Ptr> modifiedEvents;
655  bool hadError = false;
656  for (const std::string &solverVar: solver()->evidenceNames()) {
657  SymbolicExpression::Ptr newValue = solver()->evidenceForName(solverVar);
658  ASSERT_not_null(newValue);
659  ExecutionEvent::Ptr parentEvent = ops->inputVariables()->event(solverVar);
660  if (!parentEvent) {
661  error <<"solver variable \"" <<solverVar <<"\" doesn't correspond to any input event\n";
662  hadError = true;
663  continue;
664  }
665  if (!newValue->isIntegerConstant()) {
666  error <<"solver variable \"" <<solverVar <<"\" corresponding to \"" <<parentEvent->name() <<"\" is not an integer\n";
667  hadError = true;
668  continue;
669  }
670 
671  ExecutionEventId parentEventId = database()->id(parentEvent, Update::NO);
672  ASSERT_require(parentEventId);
673  ExecutionEvent::Ptr childEvent = eventMap[*parentEventId];
674 
675  switch (childEvent->inputType()) {
676  case InputType::NONE:
677  ASSERT_not_reachable("handled above");
678 
679  case InputType::ARGC: {
680  ASSERT_require(32 == newValue->nBits());
681  ASSERT_require(ops->currentState()->memoryState()->get_byteOrder() == ByteOrder::ORDER_LSB);
682  ASSERT_require(childEvent->action() == ExecutionEvent::Action::MEMORY_WRITE);
683  childEvent->value(newValue);
684  if (!newValue->toUnsigned().isEqual(parentEvent->value()->toUnsigned())) {
685  modifiedEvents.push_back(childEvent);
686  SAWYER_MESG(debug) <<" adjusting " <<childEvent->name() <<" from " <<*argc <<" to " <<*newValue <<"\n";
687  argc = newValue;
688  } else {
689  SAWYER_MESG(debug) <<" no adjustment necessary for " <<childEvent->name() <<"\n";
690  }
691  break;
692  }
693 
694  case InputType::ARGV: {
695  ASSERT_require(8 == newValue->nBits());
696  ASSERT_require(childEvent->action() == ExecutionEvent::Action::MEMORY_WRITE);
697  uint8_t byte = newValue->toUnsigned().get();
698  const std::pair<size_t, size_t> indices = childEvent->inputIndices();
699  ASSERT_require(indices.first < argv.size());
700  ASSERT_require(indices.second < argv[indices.first].size());
701  if (argv[indices.first][indices.second] != byte) {
702  childEvent->value(newValue);
703  modifiedEvents.push_back(childEvent);
704  SAWYER_MESG(debug) <<" adjusting " <<childEvent->name() <<" from '"
705  <<StringUtility::cEscape(argv[indices.first][indices.second]) <<"'"
706  <<" to '" <<StringUtility::cEscape(byte) <<"'\n";
707  argv[indices.first][indices.second] = byte;
708  } else {
709  SAWYER_MESG(debug) <<" no adjustment necessary for " <<childEvent->name() <<"\n";
710  }
711  break;
712  }
713 
714  case InputType::ENVP: {
715  ASSERT_require(8 == newValue->nBits());
716  ASSERT_require(childEvent->action() == ExecutionEvent::Action::MEMORY_WRITE);
717  uint8_t byte = newValue->toUnsigned().get();
718  const std::pair<size_t, size_t> indices = childEvent->inputIndices();
719  ASSERT_require(indices.first < env.size());
720  ASSERT_require(indices.second < env[indices.first].size());
721  if (env[indices.first][indices.second] != byte) {
722  childEvent->value(newValue);
723  modifiedEvents.push_back(childEvent);
724  SAWYER_MESG(debug) <<" adjusting " <<childEvent->name() <<" from '"
725  <<StringUtility::cEscape(env[indices.first][indices.second]) <<"'"
726  <<" to '" <<StringUtility::cEscape(byte) <<"'\n";
727  env[indices.first][indices.second] = byte;
728  } else {
729  SAWYER_MESG(debug) <<" no adjustment necessary for " <<childEvent->name() <<"\n";
730  }
731  break;
732  }
733 
734  case InputType::SYSCALL_RET: {
735  ASSERT_require(childEvent->action() == ExecutionEvent::Action::REGISTER_WRITE);
736  SymbolicExpression::Ptr oldValue = childEvent->value();
737  if (!newValue->isEquivalentTo(oldValue)) {
738  childEvent->value(newValue);
739  modifiedEvents.push_back(childEvent);
740  SAWYER_MESG(debug) <<" adjusting " <<childEvent->name() <<" from " <<*oldValue <<" to " <<*newValue <<"\n";
741  } else {
742  SAWYER_MESG(debug) <<" no adjustment necessary for " <<childEvent->name() <<"\n";
743  }
744  break;
745  }
746 
747  case InputType::SHMEM_READ: {
748  ASSERT_require(childEvent->action() == ExecutionEvent::Action::OS_SHARED_MEMORY);
749  SymbolicExpression::Ptr oldValue = childEvent->value();
750  if (!oldValue) {
751  childEvent->value(newValue);
752  modifiedEvents.push_back(childEvent);
753  SAWYER_MESG(debug) <<" adjusting " <<childEvent->name() << " from nothing to " <<*newValue <<"\n";
754  } else if (!newValue->isEquivalentTo(oldValue)) {
755  childEvent->value(newValue);
756  modifiedEvents.push_back(childEvent);
757  SAWYER_MESG(debug) <<" adjusting " <<childEvent->name() << " from " <<*oldValue <<" to " <<*newValue <<"\n";
758  } else {
759  SAWYER_MESG(debug) <<" no adjustment necessary for " <<childEvent->name() <<"\n";
760  }
761  break;
762  }
763  }
764  }
765 
766  // Check that program arguments are reasonable. These should never fail if we've set up the SMT solver constraints correctly.
767  if (argc->toUnsigned().get() < argv.size()) {
768  ASSERT_require(argc->toUnsigned().get() >= 0);
769  argv.resize(argc->toUnsigned().get());
770  } else if (argc->toUnsigned().get() > argv.size()) {
771  argv.resize(argc->toUnsigned().get());
772  }
773  for (size_t i = 0; i < argv.size(); ++i) {
774  // Remove trailing NULs.
775  while (!argv[i].empty() && '\0' == argv[i][argv[i].size()-1])
776  argv[i].resize(argv[i].size()-1);
777 
778  // Check for embedded NULs.
779  for (size_t j = 0; j < argv[i].size(); ++j) {
780  if ('\0' == argv[i][j]) {
781  error <<"argv[" <<i <<"][" <<j <<"] is NUL\n";
782  hadError = true;
783  }
784  }
785  }
786  for (size_t i = 0; i < env.size(); ++i) {
787  // Remove trailing NULs.
788  while (!env[i].empty() && '\0' == env[i][env[i].size()-1])
789  env[i].resize(env[i].size()-1);
790 
791  // Check for embedded NULs.
792  for (size_t j = 0; j < env[i].size(); ++j) {
793  if ('\0' == env[i][j]) {
794  error <<"env[" <<i <<"][" <<j <<"] is NUL\n";
795  hadError = true;
796  }
797  }
798  }
799 
800  // Create the new test case.
801  TestCase::Ptr newTestCase;
802  if (hadError) {
803  SAWYER_MESG(debug) <<"test case not created due to prior errors\n";
804  } else {
805  newTestCase = TestCase::instance(oldTestCase->specimen());
806  newTestCase->parent(oldTestCaseId);
807  newTestCase->assertions(solver()->assertions());
808  ASSERT_forbid(argv.empty());
809  newTestCase->args(argv);
810 
811  std::vector<EnvValue> envVars;
812  for (const std::string &s: env) {
813  size_t equalSign = s.find('=');
814  ASSERT_require(equalSign != std::string::npos);
815  envVars.push_back(EnvValue(s.substr(0, equalSign), s.substr(equalSign+1)));
816  }
817  newTestCase->env(envVars);
818 
819  TestCaseId newTestCaseId = database()->id(newTestCase);
820  for (ExecutionEvent::Ptr childEvent: eventMap.values()) {
821  childEvent->testCase(newTestCase);
822  database()->save(childEvent);
823  }
824 
825  saveSymbolicState(ops, newTestCaseId);
826 
827  auto currentIpEvent = ExecutionEvent::noAction(newTestCase,
828  ops->process()->nextEventLocation(When::POST),
829  childIp->toUnsigned().get());
830  currentIpEvent->name("start of test case " + boost::lexical_cast<std::string>(*newTestCaseId));
831  database()->save(currentIpEvent);
832 
833  if (debug) {
834  SAWYER_MESG_OR(info, debug) <<"created new " <<newTestCase->printableName(database()) <<"\n";
835  debug <<" command-line:\n";
836  for (const std::string &arg: argv)
837  debug <<" - " <<StringUtility::yamlEscape(arg) <<"\n";
838  debug <<" environment:\n";
839  for (const std::string &s: env)
840  debug <<" - " <<StringUtility::yamlEscape(s) <<"\n";
841  debug <<" execution-events: # Showing only those that differ from the parent test case\n";
842  for (const ExecutionEvent::Ptr &event: modifiedEvents)
843  event->toYaml(debug, database(), " - ");
844  }
845  }
