221 // Saturating addition.
222 //
223 // Add the signed value to the unsigned base, returning an unsigned result. If the result would overflow then return either
224 // zero or the maximum unsigned value, depending on the direction of overflow. The return value is a pair that contains the
225 // saturated result and a Boolean indicating whether saturation occured. We have to be careful here because signed integer
226 // overflows are undefined behavior in C++11 and earlier -- the compiler can optimize them however it likes, which might not be
227 // the 2's complement we're expecting.
228 static std::pair<uint64_t /*sum*/, bool /*saturated?*/>
229 saturatedAdd(uint64_t base, int64_t delta) {
230  if (0 == delta) {
231  return {base, false};
232  } else if (delta > 0) {
233  uint64_t addend = (uint64_t)delta;
234  if (base + addend < base) {
235  return {~(uint64_t)0, true};
236  } else {
237  return {base + addend, false};
238  }
239  } else {
240  uint64_t subtrahend = (uint64_t)(-delta);
241  if (subtrahend > base) {
242  return {0, true};
243  } else {
244  return {base - subtrahend, false};
245  }
246  }
247 }

249 // Offset an interval by a signed amount.
250 //
251 // The unsigned address interval is shifted lower or higher according to the signed @p delta value. The endpoints of the
252 // returned interval saturate to zero or maximum address if the @p delta causes the interval to partially overflow. If the
253 // interval completely overflows (both its least and greatest values overflow) then the empty interval is
254 // returned. Shifting an empty address interval any amount also returns the emtpy interval. The return value is clipped
255 // by intersecting it with the specified @p limit.
256 static AddressInterval shiftAddresses(uint64_t base, const Variables::OffsetInterval &delta, const AddressInterval &limit) {
257  if (delta.isEmpty()) {
258  return AddressInterval();
259  } else {
260  auto loSat = saturatedAdd(base, delta.least());
261  auto hiSat = saturatedAdd(base, delta.greatest());
262  if (loSat.second && hiSat.second) {
263  return AddressInterval();
264  } else {
265  return AddressInterval::hull(loSat.first, hiSat.first) & limit;
266  }
267  }
268 }

270 // Scan through the call stack and try to find a local variable that overlaps all the specified addresses. If more than one
271 // such variable exists, return the smallest one, arbitrarily breaking ties. Stack variables are described in terms of their
272 // offsets in a call frame w.r.t. a frame pointer, but we search for them using concrete addresses. We also return the concrete
273 // addresses where the variable lives on the stack.
274 static std::pair<AddressInterval, Variables::StackVariable>
275 findStackVariable(const FunctionCallStack &callStack, const AddressInterval &location, const AddressInterval &stackLimits) {
276  AddressInterval foundInterval;
277  Variables::StackVariable foundVariable;
278 
279  for (size_t i = 0; i < callStack.size(); ++i) {
280  const FunctionCall &fcall = callStack[i];
281  for (const Variables::StackVariable &var: fcall.stackVariables().values()) {
282  if (AddressInterval varAddrs = shiftAddresses(fcall.initialStackPointer() + fcall.framePointerDelta(),
283  var.interval(), stackLimits)) {
284  if (varAddrs.contains(location)) {
285  if (!foundInterval || varAddrs.size() < foundInterval.size()) {
286  foundInterval = varAddrs;
287  foundVariable = var;
288  }
289  }
290  }
291  }
292  }
293  return {foundInterval, foundVariable};
294 }

629 RiscOperators::checkNullAccess(const BS::SValue::Ptr &addrSVal, TestMode testMode, IoMode ioMode) {
630  // Null-dereferences are only tested when we're actually executing an instruciton. Other incidental operations such as
631  // initializing the first state are not checked.
632  if (!currentInstruction())
633  return;
634  if (TestMode::OFF == testMode)
635  return;
636 
637  ASSERT_not_null(addrSVal);
638  ASSERT_not_null(modelCheckerSolver_); // should have all the path assertions already
639  SymbolicExpression::Ptr addr = IS::SymbolicSemantics::SValue::promote(addrSVal)->get_expression();
640  const char *direction = IoMode::READ == ioMode ? "read" : "write";
641  ProgressTask task(modelCheckerSolver_->progress(), "nullptr");
642 
643  bool isNull = false;
644  switch (testMode) {
645  case TestMode::OFF:
646  break;
647 
648  case TestMode::MAY: {
649  // May be null if addr <= max is satisfied
650  SymbolicExpression::Ptr maxNullExpr = SymbolicExpression::makeIntegerConstant(addr->nBits(), settings_.maxNullAddress);
651  SymbolicExpression::Ptr isNullExpr = SymbolicExpression::makeLe(addr, maxNullExpr);
652  SmtSolver::Transaction tx(modelCheckerSolver_);
653  modelCheckerSolver_->insert(isNullExpr);
654  isNull = modelCheckerSolver_->check() == SmtSolver::SAT_YES;
655  (settings_.debugNull || isNull) && SAWYER_MESG(mlog[DEBUG])
656  <<" " <<direction <<" address may be in [0," <<settings_.maxNullAddress <<"]? " <<(isNull?"yes":"no")
657  <<"; address = " <<*addr <<"\n";
658  break;
659  }
660 
661  case TestMode::MUST: {
662  // Must be null if addr > max cannot be satisfied
663  SymbolicExpression::Ptr maxNullExpr = SymbolicExpression::makeIntegerConstant(addr->nBits(), settings_.maxNullAddress);
664  SymbolicExpression::Ptr isNullExpr = SymbolicExpression::makeGt(addr, maxNullExpr);
665  SmtSolver::Transaction tx(modelCheckerSolver_);
666  modelCheckerSolver_->insert(isNullExpr);
667  isNull = modelCheckerSolver_->check() == SmtSolver::SAT_NO;
668  (settings_.debugNull || isNull) && SAWYER_MESG(mlog[DEBUG])
669  <<" " <<direction <<" address must be in [0," <<settings_.maxNullAddress <<"]? " <<(isNull?"yes":"no")
670  <<"; address = " <<*addr <<"\n";
671  break;
672  }
673  }
674 
675  if (isNull && !semantics_->filterNullDeref(addrSVal, currentInstruction(), testMode, ioMode)) {
676  SAWYER_MESG(mlog[DEBUG]) <<" nullptr dereference rejected by user; this one is ignored\n";
677  isNull = false;
678  }
679 
680  if (isNull) {
681  currentState(BS::State::Ptr()); // indicates that execution failed
682  throw ThrownTag{NullDereferenceTag::instance(nInstructions(), testMode, ioMode, currentInstruction(), addrSVal)};

686 void
687 RiscOperators::checkOobAccess(const BS::SValue::Ptr &addrSVal_, TestMode testMode, IoMode ioMode, size_t nBytes) {
688  if (computeMemoryRegions_) {
689  auto addrSVal = SValue::promote(addrSVal_);
690 
691  // Out of bounds references are only tested when we're actually executing an instruction. Other incidental operations such
692  // as initializing the first state are not checked.
693  if (!currentInstruction())
694  return;
695  if (TestMode::OFF == testMode)
696  return;
697 
698  // The address must be concrete and it must have an attached region that does not fully contain the memory bytes being
699  // accessed they the address (based on the address itself and the `nBytes` parameter.
700  const auto va = addrSVal->toUnsigned();
701  if (!va)
702  return;
703  const AddressInterval referencedRegion = addrSVal->region();
704  if (!referencedRegion)
705  return;
706  const AddressInterval accessedRegion = AddressInterval::baseSizeSat(*va, nBytes);
707  ProgressTask task(modelCheckerSolver_->progress(), "oob");
708  if (referencedRegion.contains(accessedRegion))
709  return; // in bounds
710 
711  // The region attached to the address expression is the region for the variable that was intended to be accessed. This
712  // variable must exist and have been detected earlier by the variable analysis.
713  FunctionCallStack &callStack = State::promote(currentState())->callStack();
714  const FoundVariable intendedVariable = findVariable(referencedRegion, gvars_, callStack, stackLimits_);
715  if (!intendedVariable)
716  return;
717 
718  // The region actually accessed might be the address for a known source code variable that was detected earlier by the
719  // variable analysis. If we can't find a whole variable, try the individual addresses. It's okay if we don't find
720  // anything, which just means that we had an OOB access for the intended variable that doesn't land in any particular
721  // actual variable.
722  const FoundVariable accessedVariable = [this, &accessedRegion, &callStack]() {
723  if (auto v = findVariable(accessedRegion, gvars_, callStack, stackLimits_))
724  return v;
725  for (Address accessedByteVa: accessedRegion) {
726  if (auto v = findVariable(accessedByteVa, gvars_, callStack, stackLimits_))
727  return v;
728  }
729  return FoundVariable();
730  }();
731 
732  // We've found an out-of-bounds access, but is it significant from the user's point of view?
733  if (!semantics_->filterOobAccess(addrSVal, referencedRegion, accessedRegion, currentInstruction(), testMode, ioMode,
734  intendedVariable, accessedVariable)) {
735  SAWYER_MESG(mlog[DEBUG]) <<" buffer overflow rejected by user; this one is ignored\n";
736  return;
737  }
738 
739  // We've found a significant out-of-bounds access. Report it.
740  currentState(BS::State::Ptr()); // indicates that execution failed
741  throw ThrownTag{OutOfBoundsTag::instance(nInstructions(), testMode, ioMode, currentInstruction(), addrSVal,
742  intendedVariable, accessedVariable)};
743  }

874 void
875 RiscOperators::popCallStack() {
876  FunctionCallStack &callStack = State::promote(currentState())->callStack();
877  ASSERT_forbid(callStack.isEmpty());
878  if (mlog[DEBUG]) {
879  const FunctionCall &fcall = callStack.top();
880  const size_t nBits = partitioner_->instructionProvider().wordSize();
881  mlog[DEBUG] <<" returned from " <<fcall.printableName(nBits) <<"\n";
882  }
883 
884  Sawyer::Optional<Address> poppedInitialSp = callStack.top().initialStackPointer();
885  callStack.pop();
886 
887  // Predicate to determine whether a memory cell should be discarded. When popping a call from the call stack, we need to
888  // discard the stack frame for the function being popped. Assume that the stack grows down, therefore we need to discard
889  // any memory address less than the boundary address.
890  class IsPopped: public BS::MemoryCell::Predicate {
891  SymbolicExpression::Ptr lowBoundaryInclusive;
892  SymbolicExpression::Ptr highBoundaryExclusive;
893  public:
894  size_t nErased = 0;
895 
896  public:
897  explicit IsPopped(const SymbolicExpression::Ptr &lowBoundaryInclusive, const SymbolicExpression::Ptr &highBoundaryExclusive)
898  : lowBoundaryInclusive(lowBoundaryInclusive), highBoundaryExclusive(highBoundaryExclusive) {}
899 
900  bool operator()(const BS::MemoryCell::Ptr &cell) {
901  SymbolicExpression::Ptr va = SValue::promote(cell->address())->get_expression();
902  SymbolicExpression::Ptr stackGe = SymbolicExpression::makeGe(va, lowBoundaryInclusive);
903  SymbolicExpression::Ptr stackLt = SymbolicExpression::makeLt(va, highBoundaryExclusive);
904  SymbolicExpression::Ptr isDiscardable = SymbolicExpression::makeAnd(stackGe, stackLt);
905  bool retval = isDiscardable->mustEqual(SymbolicExpression::makeBooleanConstant(true));
906  if (retval) {
907  ++nErased;
908  SAWYER_MESG(mlog[DEBUG]) <<" erasing memory cell " <<*cell <<"\n";
909  }
910  return retval;
911  }
912  };
913 
914  // Figure out what memory addresses need to be erased. For instance, on x86 the caller's CALL instruction pushed a 4-byte
915  // return address onto the stack but this address is removed by the callee's RET instruction. Therefore the initial stack
916  // pointer for the callee will be off by four -- we need to remove all the stuff pushed by the callee, plus the four byte
917  // return value pushed by the caller.
918  if (poppedInitialSp) {
919  Address stackBoundary = *poppedInitialSp;
920  const Architecture::Base::ConstPtr arch = partitioner_->architecture();
921  if (as<const Architecture::Amd64>(arch)) {
922  // x86-64 "call" pushes an 8-byte return address that's popped when the function returns. Stack grows down.
923  stackBoundary += 8;
924  } else if (as<const Architecture::X86>(arch)) {
925  // x86 "call" pushes a 4-byte return address that's popped when the function returns. The stack grows down.
926  stackBoundary += 4;
927  } else if (as<const Architecture::Powerpc>(arch)) {
928  // PowerPC function calls don't push a return value.
929 #ifdef ROSE_ENABLE_ASM_AARCH32
930  } else if (as<const Architecture::ArmAarch32>(arch)) {
931  // ARM AArch32 function calls don't push a return value
932 #endif
933 #ifdef ROSE_ENABLE_ASM_AARCH64
934  } else if (as<const Architecture::ArmAarch64>(arch)) {
935  // ARM AArch64 function calls don't push a return value
936 #endif
937  } else if (as<const Architecture::Motorola>(arch)) {
938  // m68k pops the return value from the stack
939  stackBoundary += 4;
940  } else {
941  ASSERT_not_implemented("architecture = " + arch->name());
942  }
943 
944  const size_t wordSize = partitioner_->instructionProvider().stackPointerRegister().nBits();
945  ASSERT_require(wordSize > 8 && wordSize <= 64);
946  SymbolicExpression::Ptr highBoundaryExclusive = SymbolicExpression::makeIntegerConstant(wordSize, stackBoundary);
947  SymbolicExpression::Ptr lowBoundaryInclusive = SymbolicExpression::makeIntegerConstant(wordSize, stackLimits_.least());
948  SAWYER_MESG(mlog[DEBUG]) <<" initial stack pointer = " <<StringUtility::addrToString(*poppedInitialSp) <<"\n"
949  <<" erasing memory between " <<*lowBoundaryInclusive <<" (inclusive) and "
950  <<*highBoundaryExclusive <<" (exclusive)\n";
951  auto mem = BS::MemoryCellState::promote(currentState()->memoryState());
952  IsPopped predicate(lowBoundaryInclusive, highBoundaryExclusive);
953  mem->eraseMatchingCells(predicate);
954  if (mlog[DEBUG] && predicate.nErased > 0) {
955  BS::Formatter fmt;
956  fmt.set_line_prefix(" ");
957  mlog[DEBUG] <<" memory state after erasures:\n" <<(*mem + fmt);
958  }
959  }
960 }

1174  }

1176  }

1311 BS::SValue::Ptr
1312 RiscOperators::readMemory(RegisterDescriptor segreg, const BS::SValue::Ptr &addr, const BS::SValue::Ptr &dflt_,
1313  const BS::SValue::Ptr &cond) {
1314  ASSERT_not_null(addr);
1315  ASSERT_not_null(dflt_);
1316  ASSERT_not_null(cond);
1317 
1318  BS::SValue::Ptr dflt = dflt_;
1319  if (cond->isFalse())
1320  return dflt;
1321 
1322  // Offset the address by the value of the segment register
1323  BS::SValue::Ptr adjustedVa;
1324  if (segreg.isEmpty()) {
1325  adjustedVa = addr;
1326  } else {
1327  BS::SValue::Ptr segregValue = readRegister(segreg, undefined_(segreg.nBits()));
1328  adjustedVa = add(addr, signExtend(segregValue, addr->nBits()));
1329  }
1330 
1331  // Check the model and throw exception if violated.
1332  if (!isNoopRead()) {
1333  checkNullAccess(adjustedVa, settings_.nullRead, IoMode::READ);
1334  checkOobAccess(adjustedVa, settings_.oobRead, IoMode::READ, dflt->nBits() / 8);
1335  checkUninitVar(adjustedVa, settings_.uninitVar, dflt->nBits() / 8);
1336  }
1337 
1338  // Read from memory map. If we know the address and that memory exists, then read the memory to obtain the default value
1339  // which we'll use to update the symbolic state in a moment.
1340  ASSERT_require(0 == dflt->nBits() % 8);
1341  const size_t nBytes = dflt->nBits() / 8;
1342  uint8_t buf[8];
1343  if (adjustedVa->toUnsigned() && nBytes <= sizeof(buf) &&
1344  nBytes == (partitioner_->memoryMap()->at(adjustedVa->toUnsigned().get()).limit(nBytes)
1345  .require(MemoryMap::READABLE).prohibit(MemoryMap::WRITABLE).read(buf).size())) {
1346  switch (partitioner_->memoryMap()->byteOrder()) {
1347  case ByteOrder::ORDER_UNSPECIFIED:
1348  case ByteOrder::ORDER_LSB: {
1349  uint64_t value = 0;
1350  for (size_t i=0; i<nBytes; ++i)
1351  value |= (uint64_t)buf[i] << (8*i);
1352  dflt = number_(dflt->nBits(), value);
1353  break;
1354  }
1355 
1356  case ByteOrder::ORDER_MSB: {
1357  uint64_t value = 0;
1358  for (size_t i=0; i<nBytes; ++i)
1359  value = (value << 8) | (uint64_t)buf[i];
1360  dflt = number_(dflt->nBits(), value);
1361  break;
1362  }
1363  }
1364  }
1365 
1366  // Read from the symbolic state, and update the state with the default from real memory if known.
1367  return assignRegion(Super::readMemory(segreg, addr, dflt, cond));
1368 }

1395 
1396 SemanticCallbacks::SemanticCallbacks(const ModelChecker::Settings::Ptr &mcSettings, const Settings &settings,
1397  const P2::Partitioner::ConstPtr &partitioner)
1398  : ModelChecker::SemanticCallbacks(mcSettings), settings_(settings), partitioner_(partitioner) {
1399 
1400  ASSERT_not_null(partitioner);
1401 
1402  // Find global variables needed for uninitialized variable and buffer overflow model checkers. Although the variable
1403  // finder is not thread safe, it's okay to use it here without obtaining a lock because no other thread could possibly
1404  // be using it already since we just created it.
1405  variableFinder_ = Variables::VariableFinder::instance();
1406  if (TestMode::OFF != settings_.oobRead || TestMode::OFF != settings_.oobWrite || TestMode::OFF != settings_.uninitVar) {
1407  Sawyer::Message::Stream info(mlog[INFO]);
1408  Sawyer::Message::Stream debug(mlog[DEBUG]);
1409  SAWYER_MESG_FIRST(info, debug) <<"searching for global variables";
1410  gvars_ = variableFinder_->findGlobalVariables(partitioner_);
1411 
1412  // Don't allow variables at very low memory addresses because lots of pointer arithmetic uses these constants and
1413  // might get assigned global variable memory regions in this area. For instance,
1414  // mov eax, 0x1000 // not a global variable address
1415  // mov ecx, 4 // size, but happens to be a (miscalculated) global variable address
1416  // mul ecx, [esp+4] // ecx is now outside the the (miscalculated) global variable area
1417  // mov eax, [eax + ecx] // buffer overflow detected
1418  Variables::erase(gvars_, AddressInterval::baseSize(0, 256));
1419 
1420  // Don't allow global variables at the first byte of any read+write segment of memory. This is because the segment address
1421  // is sometimes an intermediate calculation before adding an offset to get to the global variable. If a variable exists at
1422  // the beginning of the segment, then the calculation would become locked to that variable's region and any subsequent
1423  // offset to get to the intended variable would look like an out-of-bounds access.
1424  for (const auto &node: partitioner->memoryMap()->require(MemoryMap::READ_WRITE).nodes())
1425  Variables::erase(gvars_, node.key().least());
1426 
1427  SAWYER_MESG_FIRST(info, debug) <<"; found " <<StringUtility::plural(gvars_.nIntervals(), "global variables") <<"\n";
1428  for (const Variables::GlobalVariable &gvar: gvars_.values())
1429  SAWYER_MESG(debug) <<" found " <<gvar <<"\n";
1430  }
1431 
1432  // Allocate the stack in memory
1433  static const size_t RESERVE_BELOW = 15*1024*1024; // memory to reserve below the initial stack inter
1434  static const size_t RESERVE_ABOVE = 1*1024*1024; // memory reserved at and above the initial stack pointer
1435  static const size_t STACK_ALIGNMENT = 16; // alignment in bytes
1436  if (settings_.initialStackVa) {
1437  // Use an existing stack address
1438  Alignment stackAlignment(STACK_ALIGNMENT, partitioner->architecture()->bitsPerWord());
1439  const Address maxAddr = BitOps::lowMask<Address>(partitioner->architecture()->bitsPerWord());
1440  const Address loAddr = stackAlignment.alignDown(*settings_.initialStackVa >= RESERVE_BELOW ?
1441  *settings_.initialStackVa - RESERVE_BELOW :
1442  0);
1443  const Address hiAddr = stackAlignment.alignUp(*settings_.initialStackVa <= maxAddr - (RESERVE_ABOVE + 1) ?
1444  *settings_.initialStackVa + RESERVE_ABOVE - 1 :
1445  maxAddr)
1446  .orElse(maxAddr + 1) - 1;
1447  ASSERT_require(loAddr <= hiAddr);
1448  stackRegion_ = AddressInterval::hull(loAddr, hiAddr);
1449  Sawyer::Optional<Address> va = partitioner->memoryMap()->findFreeSpace(stackRegion_.size(), 1, stackRegion_);
1450  if (!va) {
1451  mlog[ERROR] <<"cannot allocate stack at " <<StringUtility::addrToString(stackRegion_) <<"\n";
1452  } else {
1453  SAWYER_MESG(mlog[INFO]) <<"initial stack:\n"
1454  <<" pointer = " <<StringUtility::addrToString(*settings_.initialStackVa) <<"\n"
1455  <<" region = " <<StringUtility::addrToString(stackRegion_) <<"\n";
1456  }
1457  } else {
1458  // Choose an initial stack pointer that's unlikely to interfere with instructions or data. This model requires a concrete
1459  // stack pointer.
1460  auto where = AddressInterval::hull(0x80000000, 0xffffffff); // where to look in the address space
1461  Sawyer::Optional<Address> va = partitioner->memoryMap()->findFreeSpace(RESERVE_BELOW + RESERVE_ABOVE, STACK_ALIGNMENT,
1462  where, Sawyer::Container::MATCH_BACKWARD);
1463  if (!va) {
1464  mlog[ERROR] <<"no room for a stack anywhere in " <<StringUtility::addrToString(where) <<"\n";
1465  } else {
1466  settings_.initialStackVa = *va + RESERVE_BELOW;
1467  stackRegion_ = AddressInterval::hull(*va, *va + RESERVE_BELOW + RESERVE_ABOVE - 1);
1468  SAWYER_MESG(mlog[INFO]) <<"initial stack:\n"
1469  <<" pointer = " <<StringUtility::addrToString(*settings_.initialStackVa) <<"\n"
1470  <<" region = " <<StringUtility::addrToString(stackRegion_) <<"\n";
1471  }
1472  }
1473  const auto stackSegment = MemoryMap::Segment::anonymousInstance(stackRegion_.size(), MemoryMap::READ_WRITE, "stack");
1474  partitioner_->memoryMap()->insert(stackRegion_, stackSegment);
1475 }

1754 ExecutionUnit::Ptr
1755 SemanticCallbacks::findUnit(Address va, const Progress::Ptr &progress) {
1756  ExecutionUnit::Ptr unit;
1757 
1758  // If we're following one path, then the execution unit is always the next one on the path.
1759  {
1760  SAWYER_THREAD_TRAITS::LockGuard lock(mutex_);
1761  if (followingOnePath_) {
1762  if (!onePath_.empty()) {
1763  unit = onePath_.front();
1764  onePath_.pop_front();
1765  }
1766  }
1767  }
1768 
1769  if (unit) {
1770  // We took it from the one path we're following (see above)
1771  if (*unit->address() != va) {
1772  SAWYER_MESG(mlog[DEBUG]) <<"next follow-one-path execution unit address mismatch:"
1773  <<" expected " <<StringUtility::addrToString(va)
1774  <<" but has " <<StringUtility::addrToString(*unit->address()) <<"\n";
1775  return ExecutionUnit::Ptr();
1776  }
1777  return unit;
1778  }
1779 
1780 
1781  {
1782  // We use a separate mutex for the unit_ cache so that only one thread computes this at a time without blocking
1783  // threads trying to make progress on other things. An alternative would be to lock the main mutex, but only when
1784  // accessing the cache, and allow the computations to be duplicated with only the first thread saving the result.
1785  SAWYER_THREAD_TRAITS::LockGuard lock(unitsMutex_);
1786  unit = units_.getOrDefault(va);
1787  }
1788 
1789  if (unit)
1790  return unit; // preexisting
1791 
1792  // Compute the next unit
1793  P2::Function::Ptr func = partitioner_->functionExists(va);
1794  if (func && boost::ends_with(func->name(), "@plt")) {
1795  unit = ExternalFunctionUnit::instance(func, partitioner_->sourceLocations().get(va));
1796  } else if (P2::BasicBlock::Ptr bb = partitioner_->basicBlockExists(va)) {
1797  // Depending on partitioner settings, sometimes a basic block could have an internal loop. For instance, some x86
1798  // instructions have an optional repeat prefix. Since execution units need to know how many steps they are (so we can
1799  // inforce the K limit), and since the content of the execution unit needs to be immutable, and since we can't really
1800  // re-compute path successors with new data (we would have already thrown away the outgoing state), we have to make a
1801  // decision here and now.
1802  //
1803  // So we evaluate each instruction semantically on a clean state and check that the instruction pointer follows the
1804  // instrucitons in the basic block. If not, we switch to one-instruction-at-a-time mode for this basic block, which is
1805  // about half as fast and takes more memory. By caching our results, we only do this expensive calcultion once per
1806  // basic block, not each time a path reaches this block.
1807  if (bb->nInstructions() > 0) {
1808  ProgressTask task(progress, "findNext");
1809 
1810  // We intentionally don't use an SMT solver here because it's not usually needed--we're only processing a single
1811  // basic block. If the Z3 solver is used here, we find that Z3 eventually hangs during the z3check call on some
1812  // threads even when we tell it to time out after a few seconds.
1813  auto ops = partitioner_->newOperators(P2::MAP_BASED_MEMORY);
1814  IS::SymbolicSemantics::RiscOperators::promote(ops)->trimThreshold(mcSettings()->maxSymbolicSize);
1815  auto cpu = partitioner_->newDispatcher(ops);
1816  const RegisterDescriptor IP = partitioner_->instructionProvider().instructionPointerRegister();
1817  for (size_t i = 0; i < bb->nInstructions() - 1; ++i) {
1818  SgAsmInstruction *insn = bb->instructions()[i];
1819  try {
1820  cpu->processInstruction(insn);
1821  } catch (...) {
1822  SAWYER_MESG(mlog[DEBUG]) <<" " <<bb->printableName() <<" at " <<insn->toString() <<"; switched to insn\n";
1823  unit = BasicBlockUnit::instance(partitioner_, bb);
1824  break;
1825  }
1826  BS::SValue::Ptr actualIp = ops->peekRegister(IP);
1827  Address expectedIp = bb->instructions()[i+1]->get_address();
1828  if (actualIp->toUnsigned().orElse(expectedIp+1) != expectedIp) {
1829  SAWYER_MESG(mlog[DEBUG]) <<" " <<bb->printableName()
1830  <<" sequence error after " <<insn->toString() <<"; switched to insn\n";
1831  unit = InstructionUnit::instance(bb->instructions()[0], partitioner_->sourceLocations().get(va));
1832  break;
1833  }
1834  }
1835  }
1836  if (!unit && bb->nInstructions() > 0)
1837  unit = BasicBlockUnit::instance(partitioner_, bb);
1838  } else if (SgAsmInstruction *insn = partitioner_->instructionProvider()[va]) {
1839  SAWYER_MESG(mlog[DEBUG]) <<" no basic block at " <<StringUtility::addrToString(va) <<"; switched to insn\n";
1840  unit = InstructionUnit::instance(insn, partitioner_->sourceLocations().get(va));
1841  }
1842 
1843  // Save the result, but if some other thread beat us to this point, use their result instead.
1844  if (unit) {
1845  SAWYER_THREAD_TRAITS::LockGuard lock(unitsMutex_);
1846  unit = units_.insertMaybe(va, unit);
1847  }
1848 
1849  return unit;
1850 }

1879 std::vector<SemanticCallbacks::NextUnit>
1880 SemanticCallbacks::nextUnits(const Path::Ptr&, const BS::RiscOperators::Ptr &ops, const SmtSolver::Ptr &solver) {
1881  std::vector<SemanticCallbacks::NextUnit> units;
1882  ProgressTask task(solver->progress(), "nextUnits");
1883 
1884  // If we've seen this state before, then there's nothing new for us to do.
1885  if (seenState(ops)) {
1886  SAWYER_MESG(mlog[DEBUG]) <<" not extending path since state was previously seen\n";
1887  return {};
1888  }
1889 
1890  // Find next concrete addresses from instruction pointer symbolic expression.
1891  CodeAddresses next = nextCodeAddresses(ops);
1892  if (!next.ip)
1893  return {};
1894  auto ip = IS::SymbolicSemantics::SValue::promote(next.ip)->get_expression();
1895  if (!next.isComplete) {
1896  auto tag = ErrorTag::instance(0, "abstract jump", "symbolic address not handled yet", nullptr, ip);
1897  tag->importance(WARN);
1898  auto fail = FailureUnit::instance(Sawyer::Nothing(), SourceLocation(), "no concrete instruction pointer", tag);
1899  units.push_back({fail, SymbolicExpression::makeBooleanConstant(true), SmtSolver::Evidence()});
1900  return units;
1901  }
1902 
1903  // Create execution units for the next concrete addresses
1904  for (Address va: next.addresses) {
1905  // Test whether next address is feasible with the given previous path assertions
1906  SmtSolver::Transaction tx(solver);
1907  auto assertion = SymbolicExpression::makeEq(ip, SymbolicExpression::makeIntegerConstant(ip->nBits(), va));
1908  solver->insert(assertion);
1909 
1910  switch (solver->check()) {
1911  case SmtSolver::SAT_YES:
1912  // Create the next execution unit
1913  if (ExecutionUnit::Ptr unit = findUnit(va, solver->progress())) {
1914  units.push_back({unit, assertion, solver->evidenceByName()});
1915  } else if (settings_.nullRead != TestMode::OFF && va <= settings_.maxNullAddress) {
1916  SourceLocation sloc = partitioner_->sourceLocations().get(va);
1917  BS::SValue::Ptr addr = ops->number_(partitioner_->instructionProvider().wordSize(), va);
1918  auto tag = NullDereferenceTag::instance(0, TestMode::MUST, IoMode::READ, nullptr, addr);
1919  auto fail = FailureUnit::instance(va, sloc, "invalid instruction address", tag);
1920  units.push_back({fail, assertion, SmtSolver::Evidence()});
1921  } else {
1922  SourceLocation sloc = partitioner_->sourceLocations().get(va);
1923  auto tag = ErrorTag::instance(0, "invalid instruction pointer", "no instruction at address", nullptr, va);
1924  units.push_back({FailureUnit::instance(va, sloc, "invalid instruction address", tag), assertion,
1925  SmtSolver::Evidence()});
1926  }
1927  break;
1928  case SmtSolver::SAT_NO:
1929  // Do not extend the current path in this direction since it is infeasible.
1930  break;
1931  case SmtSolver::SAT_UNKNOWN: {
1932  // The SMT solver failed. This is could be a timeout or some other failure.
1933  SAWYER_THREAD_TRAITS::LockGuard lock(mutex_);
1934  ++nSolverFailures_;
1935  break;
1936  }
1937  }
1938  }
1939  return units;
1940 }
