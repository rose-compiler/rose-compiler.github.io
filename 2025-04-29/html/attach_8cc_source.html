63 // have any kind of overlap
64 // Note: Profiling results only contain a single line source information.
65 // ROSE AST has both start and end line information.
66 static
67 bool doLinesOverlap(const SgLocatedNode* node, const RoseHPCT::Located * plnode)
68 {
69  ROSE_ASSERT(plnode);
70  if (node == NULL
71  )
72  return false;
73  size_t b_start = plnode->getFirstLine();
74  size_t b_end = plnode->getLastLine();
75 
76  // Liao 1/25/2013, Adjust SgFunctionDefinition to be its enclosing defining function declaration, which will
77  // have the right start end end file locations to match profiling data
78  // SgFunctionDefinition starts from the "{" in ROSE using EDG 4.4, instead of starting from its function declaration line.
79  const SgLocatedNode* node_copy = node; // we don't want to accidentally modify node!! const A* node means what is pointed is const, not guarantee that the pointer itself is const. very tricky !!
80  if (const SgFunctionDefinition* f_def = isSgFunctionDefinition(node))
81  {
82  node_copy = f_def->get_declaration();
83  const SgFunctionDeclaration* fdecl = isSgFunctionDeclaration(node_copy);
84  ROSE_ASSERT (fdecl != NULL);
85  ROSE_ASSERT (fdecl->get_definingDeclaration() == fdecl);
86  }
87  Sg_File_Info* info_start = node_copy->get_startOfConstruct();
88  ROSE_ASSERT (info_start != NULL);
89  // check file name first if possible
90  // DXN: see DXN's comment in doFilenamesMatch.
91  if (plnode->getFileNode())
92  {
93  std::string plnodeFileName = plnode->getFileNode()->getName();
94  std::string info_startFileName = info_start->get_filenameString();
95  int suffixLength = info_startFileName.length() - plnodeFileName.length();
96  std::string suffix = suffixLength >= 0? info_startFileName.substr(suffixLength): "";
97  if (plnodeFileName != suffix)
98  return false;
99  }
100  // Then check for line numbers
101  size_t a_start = (size_t) info_start->get_line();
102  Sg_File_Info* info_end = node_copy->get_endOfConstruct();
103  size_t a_end = (info_end == NULL) ? a_start : info_end->get_line();
104  // adjust for wrong line information, See Bugs-Internal 311
105  // some non-scope statement's end line numbers are not correct from ROSE
106  // especially for SgForInitStatement and loop condition statements
107  // We adjust their end line info. to be the same as the beginning line numbers
108  // This is generally risky for ROSE, but should work for ROSE-HPCT
109  // since the HPCToolkit generates metrics specified with beginning line only
110  // Liao, 2/2/2009
111 
112  if (a_end < a_start)
113  {
114  a_end = a_start;
115  ROSE_ASSERT(false); // should not happen for ROSE using EDG 4.4
116  }
117  if (b_end < b_start)
118  {
119  b_end = b_start;
120  }
121 
122  return (b_start <= a_start && a_end <= b_end) // SgNode's file portion is a subset of Profile's information
123  || (a_start <= b_start && b_end <= a_end); // Profile's info is a subset of SgNode's portion
124 }

350 // SgNode to the current RoseHPCT::IRNode (profNode) node.
351 //
352 // This is used to help avoid redundant metric attachment to
353 // both parent and child nodes expanded from a single line of macro.
354 // It takes advantage of the preorder traversal to attach a metric only once
355 // on the highest level of AST node matching the source line,
356 // all other nodes below the highest level of node are called shadow nodes
357 // and are ignored. If we have multiple nodes at the same highest level,
358 // we still attach them all and let the Normalization phase to adjust metrics
359 // later.
360 //
361 // The original code naively attached performance metrics to all AST nodes
362 // with matching source line numbers,
363 // which caused problems when an AST subtree is expanded from a single line macro call.
364 // In this case, all nodes in the expanded AST tree will have the same source line number as the macro
365 // and they all got the same metrics attached. A later metric propagation phase would wrongfully
366 // accumulate the redundant metrics within the subtree and got very off metrics in the end.
367 //
368 // Limitations: Some crazy macros used in SMG2000 are expanded to both loop control headers
369 // and part of the loop body. This shadow node concept only helps get the correct final accumulated metrics
370 // and cannot attribute the metrics to headers and partial loop body (the performance tool does
371 // not provide sufficient information neither).
372 // In fact, the metrics of the crazy macros are attached to loop headers only since loop headers
373 // are visited before the loop body.
374 //
375 // TODO A final top-down propagation phase may be needed to attribute the top level metric further
376 // down the children and grandchildren nodes. But We don't see the need yet right now.
377 // Liao, 2/3/2009
378 bool MetricFinder::isShadowNode(RoseHPCT::IRNode* profNode) const
379 {
380  ROSE_ASSERT(historyRecord_!=NULL);
381  ROSE_ASSERT(profNode!=NULL);
382 
383  bool result = false;
384  if ((*historyRecord_)[profNode].size() == 0)
385  result = false; // not record at all, not shadowed
386  else
387  {
388  SgNode* existing_node = *((*historyRecord_)[profNode].begin());
389  if (existing_node->get_parent() == target_->get_parent())
390  result = false; // same level in tree path, not shadowed
391  else
392  result = true; // all other cases mean shadowed,
393  //either a child-parent relationships or different tree paths
394  }
395  return result;
396 }

479 template<class SgType>
480 static
481 bool attachOne(SgType* node, const string& attr_name, MetricAttr* attr, bool verbose = false)
482 {
483  if (node == NULL
484  ) return false;
485  if (node->attributeExists(attr_name))
486  {
487  AstAttribute* old_attr_raw = node->getAttribute(attr_name);
488  MetricAttr* old_attr = dynamic_cast<MetricAttr *>(old_attr_raw);
489  ROSE_ASSERT (old_attr);
490  attr->setValue(attr->getValue() + old_attr->getValue());
491  node->updateAttribute(attr_name, attr);
492  }
493  else
494  // node does not already have an attribute named attr_name
495  node->setAttribute(attr_name, attr);
496  if (verbose)
497  notifyAttached(node, attr_name);
498  return true;
499 }

617 void MetricAttachTraversal::visit(SgNode* n)
618 {
619  SgLocatedNode* n_loc = isSgLocatedNode(n);
620  if (n_loc)
621  {
622  MetricFinder finder(n_loc, &attached_);
623  finder.setVerbose(verbose_);
624  // for current SgNode, find matching prof ir nodes
625  finder.traverse(hpc_root_);
626  // for collecting File nodes of profile IR
627  if (finder.found())
628  {
629  if (enable_debug)
630  {
631  cout << "MetricAttachTraversal::visit() found metrics match for " << n->class_name() << ":\n"
632  << n_loc->unparseToString() << "\n at line:" << n_loc->get_file_info()->get_line() << endl;
633  }
634  MetricFinder::MatchSet_t matches = finder.getMatches();
635  for (MetricFinder::MatchSet_t::const_iterator hpc_match = matches.begin(); hpc_match != matches.end(); ++hpc_match)
636  {
637  ROSE_ASSERT (*hpc_match);
638  attached_[*hpc_match].insert(n_loc);
639  // Also save to a namescope copy for all profile IR trees
640  profSageMap_[*hpc_match].insert(n_loc);
641 
642  if (enable_debug)
643  {
644  cout << " metric:" << (*hpc_match)->toString() << endl;
645  }
646  for (Observable::ConstMetricIterator m = (*hpc_match)->beginMetric(); m != (*hpc_match)->endMetric(); ++m)
647  {
648  string attr_name = m->getName();
649  double attr_value = m->getValue();
650  MetricAttr* attr_sage = new MetricAttr(attr_name, attr_value);
651  attachOne(n_loc, attr_name, attr_sage, verbose_);
652 
653  if (dynamic_cast<const Loop *>(*hpc_match))
654  attachPostScopeStatement(isSgScopeStatement(n_loc), attr_sage, verbose_);
655  if (dynamic_cast<const Statement *>(*hpc_match))
656  attachPostFunctionDefinition(isSgFunctionDefinition(n_loc), attr_sage, verbose_);
657  }
658  }
659  }
660  }

704 static
705 void normalizeMetrics(SgProject* sage_root, MetricAttachTraversal::AttachedNodes_t& attached, bool verbose = false)
706 {
707  if (verbose)
708  cerr << "[Normalizing metrics...]" << endl;
709 
710  // Compute the depth of each node
711  TreeDepthCalculator dcalc(sage_root);
712 
713  // Normalize the metrics : for each profile IR node which has matched AST nodes
714  MetricAttachTraversal::AttachedNodes_t::iterator ir_node;
715  for (ir_node = attached.begin(); ir_node != attached.end(); ++ir_node)
716  {
717  // Extract key-value pair, <i, sg_nodes>
718  const IRNode* i = ir_node->first;
719  SgLocNodeSet_t& sg_nodes = ir_node->second;
720 
721  // Build a map, depth_map[k] == # of nodes at depth k
722  using std::map;
723  map<size_t, size_t> depth_map; //typedef std::set<SgLocatedNode *> SgLocNodeSet_t;
724  for (SgLocNodeSet_t::iterator np = sg_nodes.begin(); np != sg_nodes.end(); ++np)
725  {
726  size_t depth = getTreeDepthAttr(*np);
727  depth_map[depth]++;
728  }
729 
730  // Normalize attributes stored in the profiling IR node: for each metric attached to the profiling IR
731  for (Observable::ConstMetricIterator m = i->beginMetric(); m != i->endMetric(); ++m)
732  {
733  string attr_name = m->getName();
734  // average the metric among nodes at the same depth level
735  for (SgLocNodeSet_t::iterator np = sg_nodes.begin(); np != sg_nodes.end(); ++np)
736  {
737  MetricAttr* attr_sage = getMetric(attr_name, *np);
738  ROSE_ASSERT (attr_sage);
739 
740  size_t depth = getTreeDepthAttr(*np);
741  size_t n_matches = depth_map[depth];
742  if (n_matches > 1) // if more than one AST nodes are at the same depth
743  {
744  double old_val = attr_sage->getValue();
745  double new_val = old_val / n_matches;
746  attr_sage->setValue(new_val);
