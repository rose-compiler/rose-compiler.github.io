21 // Escape characters that are special to pod
22 static std::string
23 podEscape(const std::string &s) {
24  std::string retval;
25  BOOST_FOREACH (char ch, s) {
26  if ('<' == ch) {
27  retval += "E<lt>";
28  } else if ('>' == ch) {
29  retval += "E<gt>";
30  } else if ('|' == ch) {
31  retval += "E<verbar>";
32  } else if ('/' == ch) {
33  retval += "E<sol>";
34  } else {
35  retval += ch;
36  }
37  }
38  return retval;
39 }

176 SAWYER_EXPORT bool
177 PodMarkup::emit(const std::string &doc) {
178  // Generate POD documentation into a temporary file. Since perldc doesn't support the "name" property, but rather uses the
179  // file name, we create a temporary directory and place a POD file inside with the name we want.
180  FileSystem::TemporaryDirectory tmpDir;
181  FileSystem::TemporaryFile tmpFile(tmpDir.name() / (pageName() + ".pod"));
182  tmpFile.stream() <<(*this)(doc);
183  tmpFile.stream().close();
184 
185  std::string lessOptions = []() {
186  if (const char *less = ::getenv("LESS")) {
187  return less;
188  } else {
189  return "";
190  }
191  }();
192 
193  std::string cmd = "env LESSCHARSET=utf-8"
194  " LESS=" + escapeSingleQuoted(lessOptions + "R") +
195  " perldoc"
196  " -o man"
197  " -w 'center:" + escapeSingleQuoted(chapterTitleOrDefault()) + "'"
198  " -w 'date:" + escapeSingleQuoted(versionDateOrDefault()) + "'"
199  " -w 'release:" + escapeSingleQuoted(versionStringOrDefault()) + "'"
200  " -w 'section:" + escapeSingleQuoted(chapterNumberOrDefault()) + "'"
201  " '" + escapeSingleQuoted(tmpFile.name().string()) + "'";
202 
203  return system(cmd.c_str()) == 0;
204 };

206 SAWYER_EXPORT std::string
207 PodMarkup::finalizeDocument(const std::string &s_) {
208  std::string s = "=pod\n\n" + s_ + "\n\n=cut\n";
209 
210  // Remove pairs of "=back =over" and spaces at the ends of lines
211  {
212  boost::regex backOverRe("(^=back\\s*=over)|[ \\t\\r\\f]+$");
213  std::ostringstream out(std::ios::out | std::ios::binary);
214  std::ostream_iterator<char, char> oi(out);
215  boost::regex_replace(oi, s.begin(), s.end(), backOverRe, "");
216  s = out.str();
217  }
218 
219  // Change "=section" to "=headN" where N is 1, 2, 3, or 4, and remove =endsection
220  {
221  std::string result;
222  int sectionDepth = 0;
223  std::vector<std::string> lines;
224  boost::split(lines, s, boost::is_any_of("\n"));
225  BOOST_FOREACH (const std::string &line, lines) {
226  if (boost::starts_with(line, "=section")) {
227  size_t headLevel = std::max(1, std::min(++sectionDepth, 4));
228  result += "=head" + boost::lexical_cast<std::string>(headLevel) + line.substr(8) + "\n";
229  } else if (boost::starts_with(line, "=endsection")) {
230  --sectionDepth;
231  } else {
232  result += line + "\n";
233  }
234  }
235  std::swap(s, result);
236  }
237 
238  // Replace lots of blank lines with just one blank line
239  {
240  boost::regex blankLines("\\n{3,}");
241  std::ostringstream out(std::ios::out | std::ios::binary);
242  std::ostream_iterator<char, char> oi(out);
243  boost::regex_replace(oi, s.begin(), s.end(), blankLines, "\n\n");
244  s = out.str();
245  }
246 
247  return s;
248 }
