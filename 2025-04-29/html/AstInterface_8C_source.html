420 LookupNestedDeclaration(const std::string& name, SgNode* loc) {
421  int sz=name.size();
422  assert (sz!=0);
423  assert (loc);
424 
425  int pos=0;
426  // skip leading :: if they are present.
427  if (sz>=2 && name[0]==':' && name[1]==':')
428  pos=2;
429 
430  assert (sz-2!=0);
431 
432  // Save the current matching declarations in cur_results.
433  // Save the declarations to sewarch in to_search_next.
434  AstInterface::AstNodeList cur_results, to_search_next;
435  to_search_next.push_back(SageInterface::getGlobalScope(loc));
436 
437  // split the name into segments
438  std::string currentname;
439  DebugVariable([&name](){return "Looking for variable:" + name; });
440  // Search for each scope name and advance the search accordingly.
441  while (pos<= sz) { // we reach the last + 1 pos, very tricky here!!
442  if (name[pos] !=':' && pos< sz) {
443  // characters other than :, accumulate to current name
444  // We have not yet reached the end of a scope name.
445  currentname.push_back(name[pos++]);
446  continue;
447  }
448  // We have a complete scope name. Double checking.
449  assert (currentname.size()!=0);
450  // First, advance the given name to the next scope if needed.
451  if (pos < sz && name[pos]==':') {
452  if (name[pos-1]!=':') { // this is the first : of ::
453  assert(pos+1< sz && name[pos+1]==':');
454  pos+=2; // skip two chars
455  }
456  }
457  else {
458  // last char. Advance pos to exit the surrounding while loop.
459  pos++;
460  }
461 
462  // Use decl_ptr_list as a work list to store all declarations to check.
463  // Use matched_decls to save all declarations that match the current name.
464  // Use new_decls to save new declarations to search for current name.
465  AstInterface::AstNodeList decl_ptr_list, matched_decls;
466  // Start from all matching declarations before reaching current name.
467  decl_ptr_list = to_search_next;
468  cur_results.clear(); to_search_next.clear();
469  // Now search for the scope that have the given scope name..
470  DebugVariable([&currentname]() { return "Looking for name:" + currentname; });
471  // Iterate until the list is empty.
472  while (!decl_ptr_list.empty()) {
473  // Pop out the current declaration.
474  auto cur_decl= decl_ptr_list.back();
475  decl_ptr_list.pop_back();
476 
477  DebugVariable([&cur_decl]() { return "processing decl:" + ((cur_decl==0)?"NULL":AstInterface::AstToString(cur_decl)); });
478  std::string tmp_name;
479  AstInterface::AstNodeList new_decls;
480  if (AstInterface::IsBlock(cur_decl, &tmp_name, &new_decls)) {
481  size_t i = tmp_name.rfind("::");
482  if (i < tmp_name.size()) { // strip qualified names.
483  tmp_name = tmp_name.substr(i+2, tmp_name.size()-i+2);
484  }
485  DebugVariable([&tmp_name]() { return "Is Block " + tmp_name;});
486  if (tmp_name == currentname) {
487  cur_results.push_back(cur_decl);
488  DebugVariable([&currentname]() { return "Found name : " + currentname; });
489  to_search_next.insert(to_search_next.end(), new_decls.begin(), new_decls.end());
490  } else {
491  decl_ptr_list.insert(decl_ptr_list.end(), new_decls.begin(), new_decls.end());
492  }
493  }
494  else if (SgVariableDeclaration* var_decl = isSgVariableDeclaration(cur_decl.get_ptr())) {
495  DebugVariable([]() { return "Is variable declaration."; });
496  auto vars = var_decl->get_variables();
497  decl_ptr_list.insert(decl_ptr_list.end(), vars.begin(), vars.end());
498  } else if (SgInitializedName* initname = isSgInitializedName(cur_decl.get_ptr())) {
499  DebugVariable([&initname]() { return "name:" + initname->get_name().getString(); });
500  if (initname->get_name().getString() == currentname) {
501  cur_results.push_back(cur_decl);
502  DebugVariable([&currentname]() { return "Found name : " + currentname; });
503  }
504  } else if (auto* typedef_decl = isSgTypedefType(cur_decl.get_ptr())) {
505  if (typedef_decl->get_name().str() == currentname) {
506  cur_results.push_back(cur_decl);
507  DebugVariable([&currentname]() { return "Found name : " + currentname; });
508  auto* decl1 = typedef_decl->get_base_type();
509  to_search_next.push_back(decl1);
510  }
511  } else {
512  DebugVariable([&cur_decl]() { return "Not looking at:" + AstInterface::AstToString(cur_decl); });
513  }
514  }
515  if (cur_results.empty()) {
516  // The search fails. Output a warning and return NULL.
517  std::cerr<<"Warning: cannot find qualified name for "<< currentname << "\n";
518  return NULL; // cannot find the declaration
519  } else {
520  // reset scope name to accept next name
521  currentname = "";
522  }
523  }
524  assert(!cur_results.empty());
525  return cur_results[0].get_ptr();
526 }
