945 backendCompilesUsingOriginalInputFile ( SgProject* project, bool compile_with_USE_ROSE_macro )
946  {
947  // DQ (8/24/2009):
948  // To work with existing makefile systems, we want to force an object file to be generated.
949  // So we want to call the backend compiler on the original input file (instead of generating
950  // an new file from the AST and running it through the backend). The whole point is to
951  // gnerate a object file. The effect is that this does a less agressive test of ROSE
952  // but test only the parts required for analysis tools instead of transformation tools.
953  // This avoids some programs in the name qualification support that is called by the backend
954  // and permits testing of the parts of ROSE relevant for analysis tools (e.g. Compass).
955  // Of course we eventually want everything to work, but I want to test the compilation of
956  // ROSE using ROSE as part of test to get Compass running regularly on ROSE.
957 
958  // Note that the command generated may have to be fixup later to include more subtle details
959  // required to link libraries, etc. At present this function only handles the support required
960  // to build an object file.
961  SgStringList commandLineToGenerateObjectFile;
962 
963  enum language_enum
964  {
965  e_none = 0,
966  e_c = 1,
967  e_cxx = 2,
968  e_fortran = 3,
969  e_last_language
970  };
971 
972  language_enum language = e_none;
973  language = project->get_C_only() ? e_c : language;
974  language = project->get_Cxx_only() ? e_cxx : language;
975  language = project->get_Fortran_only() ? e_fortran : language;
976 
977  if (language == e_none)
978  {
979  // DQ (4/7/2010): Set the default language for ROSE to be C++
980  // This will mean that linking fortran object files will not be possible with ROSE.
981  // but at least configure will work propoerly since it will have a valid default.
982  // If we add state to SgProject, then we could set the default, but also allow it
983  // to be overriden using -rose:C or -rose:Cxx or -rose:Fortran options.
984  language = e_cxx;
985  }
986 
987  switch (language)
988  {
989  case e_c : commandLineToGenerateObjectFile.push_back(BACKEND_C_COMPILER_NAME_WITH_PATH); break;
990  case e_cxx : commandLineToGenerateObjectFile.push_back(BACKEND_CXX_COMPILER_NAME_WITH_PATH); break;
991  case e_fortran : commandLineToGenerateObjectFile.push_back(BACKEND_FORTRAN_COMPILER_NAME_WITH_PATH); break;
992 
993  default:
994  {
995  // Note that the default is C++, and that if there are no SgFile objects then there is no place to hold the default
996  // since the SgProject does not have any such state to hold this information. A better idea might be to give the
997  // SgProject state so that it can hold if it is used with -rose:C or -rose:Cxx or -rose:Fortran on the command line.
998 
999  printf ("Default reached in switch in backendCompilesUsingOriginalInputFile() \n");
1000  printf (" Note use options: -rose:C or -rose:Cxx or -rose:Fortran to specify which language backend compiler to link object files. \n");
1001  ROSE_ABORT();
1002  }
1003  }
1004 
1005  if (compile_with_USE_ROSE_macro == true)
1006  {
1007  // DQ (11/3/2011): Mark this as being called from ROSE (even though the backend compiler is being used).
1008  // This will help us detect where strings handed in using -D options may have lost some outer quotes.
1009  // There may also be a better fix to detect quoted strings and requote them, so this should be considered also.
1010  commandLineToGenerateObjectFile.push_back("-DUSE_ROSE");
1011  }
1012 
1013  int finalCombinedExitStatus = 0;
1014  if (project->numberOfFiles() > 0)
1015  {
1016  SgStringList originalCommandLineArgumentList = project->get_originalCommandLineArgumentList();
1017 
1018  // DQ (2/20/2010): Added filtering of options that should not be passed to the vendor compiler.
1019  SgFile::stripRoseCommandLineOptions(originalCommandLineArgumentList);
1020  SgFile::stripEdgCommandLineOptions(originalCommandLineArgumentList);
1021 
1022  SgStringList::iterator it = originalCommandLineArgumentList.begin();
1023 
1024  // Iterate past the name of the compiler being called (arg[0]).
1025  if (it != originalCommandLineArgumentList.end())
1026  it++;
1027 
1028  // JL (03/15/2018) Changed system to systemFromVector so that
1029  // command line arguments will be handled correctly ROSE-813
1030  for (SgStringList::iterator i = it; i != originalCommandLineArgumentList.end(); i++)
1031  {
1032  commandLineToGenerateObjectFile.push_back(*i);
1033  }
1034 
1035  if ( SgProject::get_verbose() >= 1 )
1036  {
1037  printf("Compile Line: ");
1038  for (SgStringList::iterator i = it; i != commandLineToGenerateObjectFile.end(); i++)
1039  {
1040  printf("%s ", (*i).c_str());
1041  }
1042  printf("\n");
1043  }
1044 
1045 
1046  // DQ (12/28/2010): If we specified to NOT compile the input code then don't do so even when it is the
1047  // original code. This is important for Fortran 2003 test codes that will not compile with gfortran and
1048  // for which the tests/nonsmoke/functional/testTokenGeneration.C translator uses this function to generate object files.
1049  if (project->get_skipfinalCompileStep() == false)
1050  {
1051  finalCombinedExitStatus = systemFromVector (commandLineToGenerateObjectFile);
1052  }
1053  }
