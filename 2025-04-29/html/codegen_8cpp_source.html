46 class PrintASTInfo : public POETCodeVisitor
47 {
48 public:
49  PrintASTInfo(std::ostream& _out) : out(_out), col(0),line(0) {}
50  POETCode* apply (POETCode* output)
51  {
52  while (output != 0) {
53  POETList* outputList = dynamic_cast<POETList*>(output);
54  if (outputList != 0) {
55  apply(outputList->get_first());
56  output = outputList->get_rest();
57  }
58  else {
59  if (output->get_enum() == SRC_STRING) {
60  out << output->toString(OUTPUT_NO_DEBUG);
61  }
62  else {
63  POETCode* res = eval_AST(output);
64  res->visit(this);
65  }
66  output = 0;
67  }
68  }
69  return 0;
70  }
71 
72 private:
73  std::ostream& out;
74  int col, line;
75 
76  void print_linebreak() { out << "\n"; ++line;
77  for (int i = 0; i < col; ++i) out << " "; }
78 
79  virtual void defaultVisit(POETCode* c) { out << c->toString(); }
80  virtual void visitUnknown(POETCode_ext* c) { out << c->toString(); }
81  virtual void visitMap(POETMap* m)
82  {
83  out << "MAP:{\n";
84  col += 2;
85  for (POETMap::const_iterator p=m->begin(); p != m->end(); ++p)
86  {
87  (*p).first->visit(this);
88  out << "->" ;
89  (*p).second->visit(this);
90  out << "\n";
91  }
92  out << "}";
93  col -= 2;
94  }
95  virtual void visitLocalVar(LocalVar* v)
96  {
97  POETCode* res = v->get_entry().get_code();
98  if (res != 0 && res != v) {
99  v->get_entry().set_code(0);
100  out << v->get_entry().get_name()->toString(OUTPUT_NO_DEBUG) << "={";
101  res->visit(this);
102  out << "}";
103  v->get_entry().set_code(res);
104  }
105  }
106  virtual void visitCodeVar(CodeVar* v)
107  {
108  out << v->get_entry().get_name()->toString(OUTPUT_NO_DEBUG);
109  if (v->get_args() != 0)
110  {
111  out << "#";
112  col += 2;
113  v->get_args()->visit(this);
114  col-=2;
115  }
116 
117  if (v->get_attr() != 0)
118  { out << "#"; col += 2; v->get_attr()->visit(this); col -= 2; }
119  }
120  virtual void visitTuple(POETTuple* l)
121  {
122  out << "("; print_linebreak();
123  int size=l->size();
124  for (int i = 0; i < size; ++i) {
125  POETCode* cur = l->get_entry(i);
126  if (cur == 0) out << "NULL,";
127  else if (cur->get_enum() == SRC_LIST)
128  { col+=2; out << "("; cur->visit(this);
129  out << ")"; col-=2; }
130  else cur->visit(this);
131  if (i < size-1) { out << ","; print_linebreak(); }
132  }
133  out << ")";
134  }
135  virtual void visitList(POETList* l)
136  {
137  POETCode* first = l->get_first();
138  int prevline = line;
139  if (first->get_enum() == SRC_LIST)
140  {out << "("; first->visit(this); out << ")"; }
141  else first->visit(this);
142  if (line > prevline) print_linebreak();
143  POETCode* rest = l->get_rest();
144  if (rest != 0) {
145  rest->visit(this);
146  }
147  else out << " NULL";
148  }

178  void output_content(const std::string& r, const std::string& orig)
179  {
180  for (unsigned j = 0; j < orig.size(); ++j)
181  {
182  if (orig[j] == '\n')
183  { start_pos=0; all_space_sofar=1; }
184  else if (orig[j] != ' ') all_space_sofar = 0;
185  else if (all_space_sofar) ++start_pos;
186  }
187  for (unsigned i = 0; i < r.size(); ++i)
188  {
189  out << r[i];
190  if (r[i] == '\n')
191  cur_pos=0;
192  else
193  ++ cur_pos;
194  }
195  }
196  void output(POETString* s) {
197  ASTFactory* fac = ASTFactory::inst();
198  std::string r = s->get_content();
199  if (r == "\n") {
200  for (int i = 0; i < align; ++i) r = r + " ";
201  s = fac->new_string(r);
202  }
203  if (output_xform == 0)
204  output_content(r,r);
205  else {
206  POETCode* args = fac->new_pair(s, fac->new_iconst(cur_pos));
207  if (!match_parameters(output_xform->get_entry().get_param(), args,
208  MATCH_PAR_MOD_CODE))
209  INCORRECT_XFORM_DEF("output(string,cur_pos)", output_xform->toString());
210  XformVarInvoke invoke(output_xform->get_entry(), output_xform->get_config(), args);
211  POETCode* res = invoke.eval(0);
212  output_content(res->toString(OUTPUT_NO_DEBUG),r);
213  }
214  }
215  virtual void visitUnknown(POETCode_ext* e)
216  {
217  POETAstInterface::unparse(e->get_content(), out, start_pos);
218  }

237  virtual void visitTuple(POETTuple* v)
238  {
239  POETCode* _listsep = listsep;
240  CodeVar* _listelem = listelem;
241  listsep = 0; listelem=0;
242  int size = v->size();
243  if (size == 0) return;
244  v->get_entry(0)->visit(this);
245  for (int i = 1; i < size; ++i) {
246  output(comma);
247  v->get_entry(i)->visit(this);
248  }
249  listsep = _listsep; listelem=_listelem;
250  }
251  virtual void visitCodeVar(CodeVar* v)
252  {
253  try {
254  POETCode* _listsep = listsep;
255  CodeVar* _listelem = listelem;
256  POETCode* result = v->invoke_output(v->get_args());
257  if (result != 0 && result != v) {
258  listsep = 0; listelem=0;
259  result->visit(this);
260  listsep = _listsep; listelem=_listelem;
261  return;
262  }
263  CvarSymbolTable::Entry e = v->get_entry();
264  LvarSymbolTable *local = 0;
265  POETCode* f = 0;
266  POETCode *parse = e.get_parse();
267  if (parse != 0 && parse->get_enum()==SRC_OP) {
268  POETOperator* op = static_cast<POETOperator*>(parse);
269  if (op->get_op() == POET_OP_LIST || op->get_op()==POET_OP_LIST1) {
270  listsep = op->get_arg(1);
271  listelem=dynamic_cast<CodeVar*>(op->get_arg(0));
272  f = v->get_args();
273  }
274  }
275  if (f == 0) {
276  listsep = 0; listelem=0;
277  f = e.get_code();
278  if (f == 0) {
279  if (parse==0) { CODE_SYNTAX_UNDEFINED(e.get_name()->toString()); }
280  else f = v->get_args();
281  }
282  else {
283  local = e.get_symTable();
284  if (local != 0)
285  local->push_table(false);
286  POETCode* pars = e.get_param();
287  if (pars != 0 && !match_parameters(pars, v->get_args(),MATCH_PAR_MOD_CODE))
288  CVAR_MISMATCH(v,pars, v->get_args());
289  v->set_attr();
290  }
291  }
292  if (f == 0) { CODE_SYNTAX_UNDEFINED(v->toString()); }
293  int save = align;
294  align = start_pos;
295  f->visit(this);
296  align = save;
297  if (local != 0)
298  local->pop_table();
299  listsep = _listsep; listelem=_listelem;
300  }
301  catch (Error err) {
302  std::cerr << " From unparsing code template " << SHORT(v->toString(),500) << "\n";
303  throw err;
304  }
305  }
306  void defaultVisit(POETCode* fc)
307  {
308  if (fc->get_enum() == SRC_PROGRAM) {
309  static_cast<POETProgram*>(fc)->write_to_file(out);
310  return;
311  }
312  POETCode* fc_eval = eval_AST(fc);
313  if (fc_eval == fc)
314  {
315  if (fc != EMPTY_LIST) {
316  std::string res = fc->toString(OUTPUT_NO_DEBUG);
317  output_content(res, res);
318  }
319  }
320  else if (fc_eval != EMPTY_LIST) {
321  fc_eval->visit(this);
322  }
323  }

325  void apply_list_restr(POETCode* cur)
326  {
327  switch (cur->get_enum()) {
328  case SRC_LIST:
329  case SRC_LVAR:
330  cur->visit(this) ; break;
331  default:
332  if (listelem == 0 || !apply_restr(cur,listelem))
333  cur->visit(this) ;
334  }
335  }

337  bool apply_restr(POETCode* code, POETCode* restr)
338  {
339  if (restr->get_enum() == SRC_CVAR) {
340  CodeVar* cvar = static_cast<CodeVar*>(restr);
341  if (code!=0 && !match_AST(code, cvar,MATCH_AST_PATTERN)) {
342 
343  cvar->get_entry().get_symTable()->push_table(false);
344  POETCode* params = cvar->get_entry().get_param();
345  if (params == 0 || match_AST(code, params,MATCH_AST_PATTERN)) {
346  cvar->get_entry().get_symTable()->pop_table();
347  CodeVar* tmp = ASTFactory::inst()->new_codeRef(cvar->get_entry(), code);
348  tmp->visit(this);
349  return true;
350  }
351  cvar->get_entry().get_symTable()->pop_table();
352  CODEGEN_MISMATCH(code, cvar);
353  }
354  }
355  else if (restr->get_enum() == SRC_OP) {
356  POETOperator* op = static_cast<POETOperator*>(restr);
357  if (op->get_op() == POET_OP_LIST || op->get_op()==POET_OP_LIST1) {
358  if (code == EMPTY || code == EMPTY_LIST) { return true; }
359  POETCode* _listsep = listsep;
360  CodeVar* _listelem = listelem;
361  int save = align;
362  align = start_pos;
363  listsep = op->get_arg(1);
364  listelem=dynamic_cast<CodeVar*>(op->get_arg(0));
365  apply_list_restr(code);
366  align = save; listsep = _listsep; listelem = _listelem;
367  return true;
368  }
369  }
370  return false;
371  }
