168  QString fid = m_currentScheme->id(i);
169  QFormat& fmt = m_currentScheme->formatRef(i);
170 
171  QTableWidgetItem *item;
172 
173  item = m_table->item(i, 1);
174  fmt.weight = item->checkState() == Qt::Checked ? QFont::Bold : QFont::Normal;
175 
176  item = m_table->item(i, 2);
177  fmt.italic = item->checkState() == Qt::Checked;
178 
179  item = m_table->item(i, 3);
180  fmt.underline = item->checkState() == Qt::Checked;
181 
182  item = m_table->item(i, 4);
183  fmt.overline = item->checkState() == Qt::Checked;
184 
185  item = m_table->item(i, 5);
186  fmt.strikeout = item->checkState() == Qt::Checked;
187 
188  item = m_table->item(i, 6);
189  fmt.waveUnderline = item->checkState() == Qt::Checked;
190 
191  QSimpleColorPicker *cp;
192 
193  cp = qobject_cast<QSimpleColorPicker*>(m_table->cellWidget(i, 7));
194  if ( cp )
195  fmt.foreground = cp->color();
196 
197  cp = qobject_cast<QSimpleColorPicker*>(m_table->cellWidget(i, 8));
198  if ( cp )

200

415
