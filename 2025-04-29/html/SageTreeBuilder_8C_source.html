81  if (isSgGlobal(node)) {

84  while ((token = tokens_->getNextToken()) && token->getStartLine() < pos.getStartLine()) {
85  info = SI::attachComment(node, token->getLexeme(), PreprocessingInfo::before, jovialStyle);
86  setCommentPositionAndConsumeToken(info);
87  }
88  // Comments same START line
89  while ((token = tokens_->getNextToken()) && token->getStartLine() == pos.getStartLine()) {
90  if (token->getStartCol() < pos.getStartCol()) {
91  // Before START
92  info = SI::attachComment(node, token->getLexeme(), PreprocessingInfo::before_syntax, jovialStyle);
93  }
94  else {
95  // After START
96  info = SI::attachComment(node, token->getLexeme(), PreprocessingInfo::after_syntax, jovialStyle);
97  }
98  setCommentPositionAndConsumeToken(info);
99  }
100 
101  // Comments same end line
102  while ((token = tokens_->getNextToken()) && token->getEndLine() == pos.getEndLine()) {
103  if (token->getEndCol() > pos.getEndCol()) {
104  info = SI::attachComment(node, token->getLexeme(), PreprocessingInfo::end_of, jovialStyle);
105  setCommentPositionAndConsumeToken(info);
106  }
107  }
108  // Comments after
109  while ((token = tokens_->getNextToken()) && token->getEndLine() > pos.getEndLine()) {
110  info = SI::attachComment(node, token->getLexeme(), PreprocessingInfo::after, jovialStyle);
111  setCommentPositionAndConsumeToken(info);
112  }
113  return;
114  }
115 
116  // Attach comments at end of a statement or expression
117  if (at_end && (isSgStatement(node) || isSgExpression(node))) {
118  boost::optional<const Token&> token{};

129  }

132  }

134  }

136  }

142  info = SI::attachComment(node, token->getLexeme(), PreprocessingInfo::before, jovialStyle);
143  setCommentPositionAndConsumeToken(info);
144  }
145  return;
146  }

174  }

800 void SageTreeBuilder::
801 Leave(SgFunctionDeclaration* declaration, SgScopeStatement* param_scope)
802 {
803  mlog[TRACE] << "SageTreeBuilder::Leave(SgFunctionDeclaration*) \n";
804 
805  ASSERT_not_null(declaration);
806 
807  SgName function_name = declaration->get_name();
808  SgVariableSymbol* result_symbol = param_scope->lookup_variable_symbol(declaration->get_name());
809  bool is_defining_decl = (isSgFunctionParameterScope(param_scope) == nullptr);
810 
811 // If this is a defining declaration then the function body has to be moved from the
812 // temporary parameter scope (param_scope is a SgBasicBlock*)
813  if (is_defining_decl)
814  {
815  SgBasicBlock* function_body = isSgBasicBlock(SageBuilder::topScopeStack());
816  ASSERT_not_null(function_body);
817 
818  // Move all of the statements temporarily stored in param_scope into the scope of the function body
819  if (isSgBasicBlock(param_scope)) {
820  SageInterface::moveStatementsBetweenBlocks (isSgBasicBlock(param_scope), function_body);
821  }
822 
823  // Connect the result SgInitializedName initially created in param_scope into the scope of the function body
824  if (result_symbol) {
825  auto proc = isSgProcedureHeaderStatement(declaration);
826  auto result = isSgInitializedName(result_symbol->get_declaration());
827  ASSERT_not_null(proc);
828  ASSERT_not_null(result);
829 
830  proc->set_result_name(result);
831  result->set_parent(declaration);
832  result->set_scope(function_body);
833  ASSERT_not_null(function_body->lookup_symbol(function_name));
834  }
835 
836  // The param_scope (SgBasicBlock) is still connected, so try to set its parent
837  // to nullptr and delete it.
838  param_scope->set_parent(nullptr);
839  delete param_scope;
840 
841  SageBuilder::popScopeStack(); // function body
842  SageBuilder::popScopeStack(); // function definition
843  } // is_def_decl
844  else
845  {
846  ASSERT_not_null(isSgFunctionParameterScope(param_scope));
847  ASSERT_require(declaration->get_functionParameterScope() == nullptr);
848 
849  declaration->set_functionParameterScope(isSgFunctionParameterScope(param_scope));
850  param_scope->set_parent(declaration);
851 
852  // The scope of the parameter names may end up connected incorrectly (gitlab-issue-287.jov)
853  if (auto proc = isSgProcedureHeaderStatement(declaration)) {
854  for (auto arg : declaration->get_parameterList()->get_args()) {
855  SgVariableSymbol* symbol = SageInterface::lookupVariableSymbolInParentScopes(arg->get_name(), param_scope);
856  if (symbol) {
857  mlog[WARN] << "potential duplicate initialized name, arg: " << arg << " scope: " << arg->get_scope() << "\n";
858  }
859  }
860  }
861 
862  if (result_symbol) {
863  SgProcedureHeaderStatement* proc_decl = isSgProcedureHeaderStatement(declaration);
864  SgInitializedName* result_name = isSgInitializedName(result_symbol->get_declaration());
865  ASSERT_not_null(proc_decl);
866  ASSERT_not_null(result_name);
867 
868  proc_decl->set_result_name(result_name);
869  result_name->set_parent(declaration);
870  }
871  }
872 
873  if (is_defining_decl) {
874  // Attach any remaining comments
875  auto scope = declaration->get_definition()->get_body();
876  attachComments(scope, /*at_end*/true);
877  }
878 
879  // Finished using the map for labels
880  labels_.clear();
881 
882  // Reset any dangling forward references to this function
883  if (!forward_var_refs_.empty()) {
884  reset_forward_var_refs(SageBuilder::topScopeStack());
885  }
886 
887  SageInterface::appendStatement(declaration, SageBuilder::topScopeStack());
888 }
