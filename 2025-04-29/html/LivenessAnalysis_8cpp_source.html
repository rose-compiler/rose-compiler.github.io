7 /******************************************
8  * Algorithm used:
9  * for all n, in[n] = out[n] = 0
10  * w = { set of all nodes }
11  * repeat until w empty
12  * n = w.pop()
13  * out[n] = OR (n' ELEM SUCC[n]) in[n']
14  * in[n] = use[n] OR (out[n] - def[n])
15  * if change to in[n]
16  * for all predecessors m of n, w.push(m)
17  *****************************************/

31 template<class T>
32 T LivenessAnalysis::merge_no_dups(T& v1, T& v2) {
33  T ret(v1);
34  for (typename T::const_iterator i = v2.begin(); i != v2.end(); ++i) {
35  bool notfound = true;
36  for (typename T::const_iterator z = ret.begin(); z != ret.end(); ++z) {
37  if (*z == *i) {
38  notfound = false;
39  break;
40  }
41  }
42  if (notfound) {
43  ret.push_back(*i);
44  }
45  }
46  return ret;
47 }
