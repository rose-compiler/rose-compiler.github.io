50 static void
51 deleteAttributeValue(AstAttribute *value, Sawyer::Attribute::Id id) {
52  if (value != NULL) {
53  switch (value->getOwnershipPolicy()) {
54  case AstAttribute::CONTAINER_OWNERSHIP:
55  delete value;
56  break;
57  case AstAttribute::NO_OWNERSHIP:
58  if (shouldEmit(id, HAS_MEMORY_LEAK))
59  mlog[WARN] <<attributeFullName(value, id) <<" is leaked\n";
60  break;
61  case AstAttribute::CUSTOM_OWNERSHIP:
62  // assume ownership is properly implemented by the subclass
63  break;
64  case AstAttribute::UNKNOWN_OWNERSHIP: {
65  if (shouldEmit(id, HAS_UNKNOWN_OWNERSHIP)) {
66  mlog[WARN] <<attributeFullName(value, id) <<" ownership is unknown and possibly leaked\n";
67

73  <<" memory is managed for this attribute type. If the author intended\n"
74  <<" attribute memory to be managed by ROSE's AstAttributeMechanism, then\n"
75  <<" the following definition should be added to the attribute's class:\n"
76  <<" virtual AstAttribute::OwnershipPolicy\n"

184 void
185 AstAttributeMechanism::assignFrom(const AstAttributeMechanism &other) {
186  if (this == &other)
187  return;
188  AstAttributeMechanism tmp; // for exception safety
189  BOOST_FOREACH (Sawyer::Attribute::Id id, other.attributes_.attributeIds()) { AstAttribute *attr = other.attributes_.attributeOrElse<AstAttribute*>(id, NULL);
191  ASSERT_not_null(attr);
192 
193  // Copy the attribute. This might throw, which is why we're using "tmp". If it throws, then we don't ever make it to
194  // the std::swap below, and the destination is unchanged and tmp will be cleaned up according to the ownership policy
195  // of each attribute. Also, as mentioned in the header file, be sure we invoke the non-const version of "copy" for the
196  // sake of subclasses that have not been modified to define the const version.
197  AstAttribute *copied = attr->copy();
198 
199  // Check various things about the attribute. Some of these might be important in places other than copying, but we
200  // report them here because there's a large amount of user code that doesn't follow the rules and we don't want to be
201  // completely obnoxious.
202  if (!copied && shouldEmit(id, HAS_NULL_COPY))
203  mlog[WARN] <<attributeFullName(attr, id) <<" was not copied; no virtual copy function?\n";
204  if (attr->attribute_class_name().compare("AstAttribute") == 0 && shouldEmit(id, HAS_NO_CLASS_NAME)) {
205  mlog[WARN] <<"attribute \"" <<StringUtility::cEscape(Sawyer::Attribute::name(id)) <<"\""
206  <<" does not implement attribute_class_name\n";
207  }
208  switch (attr->getOwnershipPolicy()) {
209  case AstAttribute::CONTAINER_OWNERSHIP:
210  ASSERT_require2(copied != attr,
211  "virtual copy function for \"" + Sawyer::Attribute::name(id) + "\" did not copy attribute");
212  break;
213  case AstAttribute::NO_OWNERSHIP:
214  // copy() is allowed to return itself; we don't care since we won't ever delete either attribute
215  // A leak will be reported instead of deleting the attribute.
216  break;
217  case AstAttribute::CUSTOM_OWNERSHIP:
218  // copy() is allowed to return itself; we don't care since we won't ever delete either attribute
219  // Assume attribute ownership is properly implemented without leaking memory.
220  break;
221  case AstAttribute::UNKNOWN_OWNERSHIP:
222  // Similar to CONTAINER_OWNERSHIP but only warn, don't assert.
223  if (copied != attr && shouldEmit(id, HAS_SELF_COPY))
224  mlog[WARN] <<attributeFullName(attr, id) <<" virtual copy function did not copy the attribute (unknown ownership)\n";
225  break;
226  }
227 
228  if (copied)
229  tmp.attributes_.setAttribute(id, copied);
230  }
231  std::swap(attributes_, tmp.attributes_);
232 }

415  if (type_arg_list == NULL && new_args_ptr == NULL) { // two null argument list? ... then they match.
