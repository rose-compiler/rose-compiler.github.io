78 CodeValues
79 Utf8CharacterEncodingForm::encode(CodePoint cp) {
80  CodeValues cvs;
81  if (cp <= 0x7f) {
82  cvs.push_back(cp);
83  } else {
84  unsigned prefixLength = 3; // 0b110x_xxxx
85  while (1) {
86  cvs.push_back(0x80 | (cp & 0x3f));
87  cp >>= 6;
88 
89  unsigned mask = ((unsigned)1 << (8-prefixLength)) - 1; // 0b0001_1111 if prefixLength==3
90  if ((cp & ~mask) == 0) {
91  unsigned prefix = ((unsigned)1 << prefixLength) - 2; // 0bxxxx_x110 if prefixLength==3
92  prefix <<= 8 - prefixLength; // 0b110x_xxxx if prefixLength==3
93  cvs.push_back(prefix | cp);
94  break;
95  }
96 
97  if (++prefixLength >= 8)
98  throw Exception("invalid code point for " + name() + ": " + StringUtility::addrToString(cp));
99  }
100  std::reverse(cvs.begin(), cvs.end());
101  }
102  return cvs;
103 }

105 State
106 Utf8CharacterEncodingForm::decode(CodeValue cv) {
107  switch (state_) {
108  case ERROR_STATE:
109  case FINAL_STATE:
110  case COMPLETED_STATE:
111  return state_ = ERROR_STATE;
112  case INITIAL_STATE:
113  if (cv <= 0x7f) { // 0b0xxx_xxxx
114  cp_ = cv;
115  return state_ = FINAL_STATE;
116  } else if ((cv & 0xe0) == 0xc0) { // 0x110x_xxxx
117  cp_ = cv & 0x1f;
118  return state_ = State(1);
119  } else if ((cv & 0xf0) == 0xe0) { // 0x1110_xxxx
120  cp_ = cv & 0x0f;
121  return state_ = State(2);
122  } else if ((cv & 0xf8) == 0xf0) { // 0x1111_0xxx
123  cp_ = cv & 0x07;
124  return state_ = State(3);
125  } else if ((cv & 0xfc) == 0xf8) { // 0x1111_10xx
126  cp_ = cv & 0x03;
127  return state_ = State(4);
128  } else if ((cv & 0xfe) == 0xfc) { // 0x1111_110x
129  cp_ = cv & 0x01;
130  return state_ = State(5);
131  } else { // invalid prefix
132  return state_ = ERROR_STATE;
133  }
134  default:
135  ASSERT_require(state_ >= 1 && state_ <= 5);
136  if ((cv & 0xc0) == 0x80) { // 0x10xx_xxxx
137  cp_ = (cp_ << 6) | (cv & 0x3f);
138  return state_ = (1==state_ ? FINAL_STATE : State(state_ - 1));
139  } else {
140  return state_ = ERROR_STATE; // invalid continuation octet
141  }
142  }
143 }

168 CodeValues
169 Utf16CharacterEncodingForm::encode(CodePoint cp) {
170  CodeValues cvs;
171  if (cp <= 0x0000d7ff || (cp >= 0x0000e000 && cp <= 0x0000ffff)) {
172  cvs.push_back(cp);
173  } else if (cp >= 0x00010000 && cp <= 0x0010ffff) {
174  CodePoint hiSurrogate = ((cp - 0x0001000) >> 10) + 0x0000d800; // a.k.a., leading surrogate
175  CodePoint loSurrogate = ((cp - 0x0001000) & 0x3ff) + 0x0000dc00; // a.k.a., trailing surrogate
176  ASSERT_require(hiSurrogate >= 0xd800 && hiSurrogate <= 0xdbff);
177  ASSERT_require(loSurrogate >= 0xdc00 && loSurrogate <= 0xdfff);
178  cvs.push_back(hiSurrogate);
179  cvs.push_back(loSurrogate);
180  } else {
181  throw Exception("attempt to encode a reserved Unicode code point");
182  }
183  return cvs;
184 }

186 State
187 Utf16CharacterEncodingForm::decode(CodeValue cv) {
188  switch (state_) {
189  case ERROR_STATE:
190  case FINAL_STATE:
191  case COMPLETED_STATE:
192  return state_ = ERROR_STATE;
193  case INITIAL_STATE:
194  if (cv <= 0xd7ff || (cv >= 0xe000 && cv <= 0xffff)) {
195  cp_ = cv;
196  return state_ = FINAL_STATE;
197  } else if (cv >= 0xd800 && cv <= 0xdbff) {
198  cp_ = (cv - 0xd800) << 10; // high/leasing surrogate
199  return state_ = State(1);
200  } else {
201  return state_ = ERROR_STATE;
202  }
203  case USER_DEFINED_1: // second of two 16-bit code values
204  if (cv >= 0xdc00 && cv <= 0xdfff) {
205  cp_ |= cv - 0xdc00;
206  return state_ = FINAL_STATE;
207  } else {
208  return state_ = ERROR_STATE; // expected a low/trailing surrogate
209  }
210  default:
211  ASSERT_not_reachable("invalid decoder state");
212  }

477  switch (state_) {
478  case ERROR_STATE:
479  case FINAL_STATE:
480  case COMPLETED_STATE:
481  return state_ = ERROR_STATE;
482  case INITIAL_STATE:
483  case USER_DEFINED_1: { // 1 means we're decodig the length
484  State st = les_->decode(octet);

495  State st = les_->decode(octet);

787 class StringSearcher {
788  typedef std::vector<StringEncodingScheme::Ptr> StringEncodingSchemes;
789  StringEncodingSchemes protoEncoders_;
790  std::vector<std::vector<Finding> > findings_;
791  std::vector<Finding> results_;
792  Address bufferVa_;
793  size_t minLength_, maxLength_; // limits on number of code points per string
794  bool discardCodePoints_; // throw away decoded code points?
795  size_t maxOverlap_; // allow one encoder to match overlapping strings?
796  Sawyer::Optional<Address> anchored_; // are strings anchored to starting address?
797  Sawyer::ProgressBar<size_t> progress_;
798 public:
799  StringSearcher(const std::vector<StringEncodingScheme::Ptr> &encoders,
800  size_t minLength, size_t maxLength, bool discardCodePoints, size_t maxOverlap,
801  size_t nBytesToCheck)
802  : protoEncoders_(encoders), bufferVa_(0), minLength_(minLength), maxLength_(maxLength),
803  discardCodePoints_(discardCodePoints), maxOverlap_(maxOverlap), progress_(mlog[MARCH], "scanned bytes") {
804  findings_.resize(encoders.size());
805  progress_.value(0, nBytesToCheck);
806  progress_.suffix(" addresses");
807  }
808 
809  // anchor the search to a particular address
810  void anchor(Address startVa) { anchored_ = startVa; }
811 
812  // obtain the final results
813  const std::vector<Finding>& results() const { return results_; }
814 
815  // search for strings
816  bool operator()(const MemoryMap::Super &map, const AddressInterval &interval) {
817  if (interval.least() > bufferVa_) {
818  // We skipped across some unmapped memory, so reap all decoders, saving strings for those decoders that are in a
819  // COMPLETED_STATE.
820  for (size_t i=0; i<findings_.size(); ++i) {
821  for (size_t j=0; j<findings_[i].size(); ++j) {
822  if (findings_[i][j].encoder->state() == COMPLETED_STATE &&
823  findings_[i][j].encoder->length() >= minLength_ &&
824  findings_[i][j].encoder->length() <= maxLength_) {
825  results_.push_back(findings_[i][j]);
826  }
827  }
828  findings_[i].clear();
829  }
830  }
831 
832  std::vector<uint8_t> buffer(4096); // arbitrary
833  Address bufferVa = interval.least();
834  while (1) {
835  size_t nread = map.at(bufferVa).atOrBefore(interval.greatest()).read(buffer).size();
836  progress_.value(progress_.value()+nread);
837  ASSERT_require(nread > 0);
838  for (size_t offset=0; offset<nread; ++offset) {
839 
840  // Create new encoders starting at this address. It the string searching is configured so as to find only those
841  // strings that start at a particular address, then terminate the search early once all those strings are done
842  // being parsed.
843  if (!anchored_ || *anchored_==bufferVa+offset) {
844  for (size_t i=0; i<findings_.size(); ++i) {
845  if (findings_[i].size() < maxOverlap_)
846  findings_[i].push_back(Finding(protoEncoders_[i], bufferVa + offset));
847  }
848  } else {
849  bool haveDecoders = false;
850  for (size_t i=0; i<findings_.size(); ++i) {
851  if (!findings_[i].empty()) {
852  haveDecoders = true;
853  break;
854  }
855  }
856  if (!haveDecoders)
857  return false;
858  }
859 
860  // Decode this next octet, removing decoders that encounter errors, and saving those which enter their final
861  // state. If a decoder enters the complete (but not final) state then save the string as it exists at that
862  // point, but do not remove the decoder.
863  Octet octet = buffer[offset];
864  for (size_t i=0; i<findings_.size(); ++i) {
865  for (size_t j=0; j<findings_[i].size(); ++j) {
866  State st = findings_[i][j].encoder->decode(octet);
867  ++findings_[i][j].nBytes;
868  if (discardCodePoints_)
869  findings_[i][j].encoder->consume();
870  if (ERROR_STATE == st || findings_[i][j].encoder->length() > maxLength_) {
871  findings_[i][j].encoder = StringEncodingScheme::Ptr();
872  } else if (FINAL_STATE == st) {
873  if (findings_[i][j].encoder->length() >= minLength_ &&
874  findings_[i][j].encoder->length() <= maxLength_) {
875  results_.push_back(findings_[i][j]);
876  }
877  findings_[i][j].encoder = StringEncodingScheme::Ptr();
878  } else if (COMPLETED_STATE == st &&
879  findings_[i][j].encoder->length() >= minLength_ &&
880  findings_[i][j].encoder->length() <= maxLength_) {
881  Finding fcopy = findings_[i][j];
882  fcopy.encoder = fcopy.encoder->clone();
883  results_.push_back(fcopy);
884  }
885  }
886  findings_[i].erase(std::remove_if(findings_[i].begin(), findings_[i].end(), hasNullEncoder),
887  findings_[i].end());
888  }
889  }
890  if (bufferVa + (nread-1) == interval.greatest())
891  break; // prevent possible overflow
892  bufferVa += nread;
893  ASSERT_forbid(bufferVa > interval.greatest());
894  }
895  return true; // keep going
896  }
897 };

899 StringFinder&
900 StringFinder::find(const MemoryMap::ConstConstraints &constraints, Sawyer::Container::MatchFlags flags) {
901  strings_.clear();
902  if (settings_.minLength > settings_.maxLength || encoders_.empty())
903  return *this;
904 
905  size_t nBytesToCheck = 0;
906  for (const MemoryMap::Node &node: constraints.nodes(Sawyer::Container::MATCH_NONCONTIGUOUS))
907  nBytesToCheck += node.key().size();
908 
909  StringSearcher stringFinder(encoders_, settings_.minLength, settings_.maxLength, discardingCodePoints_,
910  settings_.maxOverlap, nBytesToCheck);
911  if (constraints.isAnchored())
912  stringFinder.anchor(constraints.anchored().least());
913  constraints.traverse(stringFinder, flags);
914 
915  for (const Finding &finding: stringFinder.results())
916  strings_.push_back(EncodedString(finding.encoder, AddressInterval::baseSize(finding.startVa, finding.nBytes)));
917 
918  if (settings_.keepingOnlyLongest) {
919  AddressIntervalSet stringAddresses;
920  std::sort(strings_.begin(), strings_.end(), byDecreasingLength);
921  for (EncodedString &string: strings_) {
922  if (stringAddresses.overlaps(string.where())) {
923  string = EncodedString(); // mark for erasing
924  } else {
925  stringAddresses.insert(string.where());
926  }
927  }
928  strings_.erase(std::remove_if(strings_.begin(), strings_.end(), isNullString), strings_.end());
929  }
930 
931  return *this;
932 }
