267 void
268 RiscOperators::resetState() {
269  // Initialize registers so they correspond to the C global variables we'll generate, and then lock the register state so
270  // those registers don't change if we access subparts (like if we store EAX and write to AX).
271  currentState()->clear();
272  RegisterState::Ptr registers = RegisterState::promote(currentState()->registerState());
273  registers->initialize_large();
274  registers->accessModifiesExistingLocations(false);
275  RegisterState::RegPairs regpairs = registers->get_stored_registers();
276  for (RegisterState::RegPair &regpair: regpairs) {
277  std::string varName = registerVariableName(regpair.desc);
278  BaseSemantics::SValue::Ptr value = makeSValue(regpair.desc.nBits(), NULL, varName);
279  registers->writeRegister(regpair.desc, value, this);
280  }
281  registers->eraseWriters();
282  registers->eraseProperties();
283  executionHalted_ = false;
284 }

341 BaseSemantics::SValue::Ptr
342 RiscOperators::makeMask(size_t nBits, size_t nSet, size_t sa) {
343  if (sa >= nBits)
344  return number_(nBits, 0);
345  nSet = std::min(nSet, nBits-sa);
346  if (0 == nSet)
347  return number_(nBits, 0);
348 
349  ASSERT_require(nBits <= 64);
350  uint64_t i = IntegerOps::genMask<uint64_t>(sa, sa+nSet-1);
351  return number_(nBits, i);
352 }

407 RiscOperators::extract(const BaseSemantics::SValue::Ptr &a, size_t begin_bit, size_t end_bit) {
408  ASSERT_require(end_bit <= a->nBits());

417  return unsignedExtend(makeSValue(a->nBits(), NULL, ctext), newSize);

449  // Note: '<<' and '>>' are not well defined in C when the shift amount is >= the size of the left operand.
450  // ctext = ((((a) << ( (sa) % (8*sizeof(a))) |
451  // ((a) >> (8*sizeof(a)-((sa) % (8*sizeof(a)))))) & mask)

455  std::string mask = SValue::promote(makeMask(a->nBits(), a->nBits()))->ctext();
456  std::string ctext = "((((" + SValue::promote(a)->ctext() + ") << " + normalizedAmount + ") |"
457  " ((" + SValue::promote(a)->ctext() + ") >> " + complementAmount + "))"

499 BaseSemantics::SValue::Ptr
500 RiscOperators::shiftRightArithmetic(const BaseSemantics::SValue::Ptr &a, const BaseSemantics::SValue::Ptr &sa) {
501  // Do the right shift without any casting to signed types.
502  // signBits = (signBit ? ~(aType)0 : (aType)0)
503  // signExtension = (sa >= 8*sizeof(a) ? signBits : signBits << (8*sizeof(a) - sa))
504  // ctext = (sa >= 8*sizeof(a) ? signBits : (a >> sa) | signExtension)
505  std::string signBit = SValue::promote(extract(a, a->nBits()-1, a->nBits()))->ctext();
506  std::string signBits = "(" + signBit + " ? "
507  "~" + SValue::promote(number_(a->nBits(), 0))->ctext() + " : " +
508  SValue::promote(number_(a->nBits(), 0))->ctext() + ")";
509  std::string width = SValue::promote(number_(a->nBits(), a->nBits()))->ctext();
510  std::string signExtension = "(" + SValue::promote(sa)->ctext() + " >= " + width + " ? " +
511  signBits + " : " + signBits + " << (" + width + " - " + SValue::promote(sa)->ctext() + "))";
512  std::string ctext = "(" + SValue::promote(sa)->ctext() + " >= " + width + " ? " +
513  signBits + " : "
514  "(" + SValue::promote(a)->ctext() + " >> " + SValue::promote(sa)->ctext() + ") | " + signExtension +
515  ")";
516  return makeSValue(a->nBits(), NULL, ctext);
517 }

535 BaseSemantics::SValue::Ptr
536 RiscOperators::unsignedExtend(const BaseSemantics::SValue::Ptr &a, size_t newWidth) {
537  std::string ctext;
538  if (newWidth == a->nBits()) {
539  ctext = SValue::promote(a)->ctext();
540  } else if (newWidth < a->nBits()) {
541  SValue::Ptr mask = SValue::promote(makeMask(a->nBits(), newWidth));
542  std::string dstType = SValue::unsignedTypeNameForSize(newWidth);
543  // ctext = ((dstType)(a & mask))
544  ctext = "((" + dstType + ")(" + SValue::promote(a)->ctext() + " & " + mask->ctext() + "))";
545  } else {
546  ASSERT_require(newWidth > a->nBits());
547  // ctext = ((dstType)a)
548  std::string dstType = SValue::unsignedTypeNameForSize(newWidth);
549  ctext = "((" + dstType + ")" + SValue::promote(a)->ctext() + ")";
550  }
551  return makeSValue(newWidth, NULL, ctext);

555 RiscOperators::signExtend(const BaseSemantics::SValue::Ptr &a, size_t newWidth) {
556  std::string ctext;
557  if (newWidth == a->nBits()) {
558  ctext = SValue::promote(a)->ctext();
559  } else {
560  ASSERT_require(newWidth >= a->nBits());
561  // signBits = (signBit ? ~(dstType)0 : (dstType)0);
562  // signExtension = (signBits << (dstType)aWidth)
563  // ctext = (a | signExtension)
564  std::string signBit = SValue::promote(extract(a, a->nBits()-1, a->nBits()))->ctext();
565  std::string signBits = "(" + signBit + " ? "
566  "~" + SValue::promote(number_(newWidth, 0))->ctext() + " : " +
567  SValue::promote(number_(newWidth, 0))->ctext() + ")";
568  std::string signExtension = "(" + signBits + " << " +
569  SValue::promote(number_(newWidth, a->nBits()))->ctext() + ")";
570  ctext = "(" + SValue::promote(a)->ctext() + " | " + signExtension + ")";
571  }
572  return makeSValue(newWidth, NULL, ctext);
573 }

689 void
690 RiscOperators::writeRegister(RegisterDescriptor reg, const BaseSemantics::SValue::Ptr &value) {
691  RegisterState::Ptr registers = RegisterState::promote(currentState()->registerState());
692  RegisterState::BitRange wantLocation = RegisterState::BitRange::baseSize(reg.offset(), reg.nBits());
693  RegisterState::RegPairs regpairs = registers->overlappingRegisters(reg);
694  for (RegisterState::RegPair &regpair: regpairs) {
695  RegisterState::BitRange storageLocation = regpair.location();
696  RegisterState::BitRange overlapLocation = wantLocation & storageLocation;
697 
698  // Create the value to be written back to this storage location.
699  BaseSemantics::SValue::Ptr toWrite;
700  if (overlapLocation.least() > storageLocation.least()) {
701  size_t offset = 0;
702  size_t nbits = overlapLocation.least() - storageLocation.least();
703  toWrite = extract(regpair.value, offset, offset+nbits);
704  }
705  {
706  size_t offset = overlapLocation.least() - wantLocation.least();
707  size_t nbits = overlapLocation.size();
708  BaseSemantics::SValue::Ptr part = extract(value, offset, offset+nbits);
709  toWrite = toWrite ? concat(toWrite, part) : part;
710  }
711  if (overlapLocation.greatest() < storageLocation.greatest()) {
712  size_t offset = overlapLocation.greatest()+1 - storageLocation.least();
713  size_t nbits = storageLocation.greatest() - overlapLocation.greatest();
714  toWrite = concat(toWrite, extract(regpair.value, offset, offset+nbits));
715  }
716  ASSERT_require(toWrite->nBits() == regpair.value->nBits());
717  regpair.value = toWrite;
718  }
719 
720  // Substitute, and write substitution back to register state.
721  for (const RegisterState::RegPair &regpair: regpairs) {
722  BaseSemantics::SValue::Ptr regVar = makeSValue(regpair.desc.nBits(), NULL, registerVariableName(regpair.desc));
723  BaseSemantics::SValue::Ptr temp = saveSideEffect(regpair.value, regVar);
724  registers->writeRegister(regpair.desc, temp, this);
725  }
726 }

728 BaseSemantics::SValue::Ptr
729 RiscOperators::readMemory(RegisterDescriptor segreg, const BaseSemantics::SValue::Ptr &address,
730  const BaseSemantics::SValue::Ptr &dflt, const BaseSemantics::SValue::Ptr &/*cond*/) {
731  ASSERT_require2(dflt->nBits() % 8 == 0, "readMemory size must be a multiple of a byte");
732  size_t nBytes = dflt->nBits() >> 3;
733  BaseSemantics::SValue::Ptr retval;
734  BaseSemantics::MemoryState::Ptr mem = currentState()->memoryState();
735 
736  // Offset the address by the value of the segment register.
737  BaseSemantics::SValue::Ptr adjustedVa;
738  if (segreg.isEmpty()) {
739  adjustedVa = address;
740  } else {
741  BaseSemantics::SValue::Ptr segregValue = readRegister(segreg, undefined_(segreg.nBits()));
742  adjustedVa = add(address, signExtend(segregValue, address->nBits()));
743  }
744 
745  for (size_t byteNum=0; byteNum<nBytes; ++byteNum) {
746  size_t byteOffset = ByteOrder::ORDER_MSB==mem->get_byteOrder() ? nBytes-(byteNum+1) : byteNum;
747  std::string ctext = "mem[" + SValue::promote(adjustedVa)->ctext() +
748  "+" + StringUtility::numberToString(byteOffset) +
749  "]";
750  BaseSemantics::SValue::Ptr byte = makeSValue(8, NULL, ctext);
751  if (retval == NULL) {
752  retval = byte;
753  } else if (ByteOrder::ORDER_MSB == mem->get_byteOrder()) {
754  retval = concat(byte, retval);
755  } else if (ByteOrder::ORDER_LSB == mem->get_byteOrder()) {
756  retval = concat(retval, byte);
757  } else {
758  // See BaseSemantics::MemoryState::set_byteOrder
759  throw BaseSemantics::Exception("multi-byte read with memory having unspecified byte order", currentInstruction());
760  }
761  }
762  return retval;
763 }
