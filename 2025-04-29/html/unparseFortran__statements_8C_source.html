561  {
562  // If an array variable has an explicit variable declaration (in the code) then the dimension
563  // information will be output there. If not then we have to output the dimension statement
564  // and an entry for this variable.
565 
566  ASSERT_not_null(arrayReference);
567  SgVarRefExp* variableReference = isSgVarRefExp(arrayReference->get_lhs_operand());
568  ASSERT_not_null(variableReference);
569  SgVariableSymbol* variableSymbol = variableReference->get_symbol();
570  ASSERT_not_null(variableSymbol);
571  SgInitializedName* variableName = variableSymbol->get_declaration();
572  ASSERT_not_null(variableName);
573 
574  SgVariableDeclaration* variableDeclaration = isSgVariableDeclaration(variableName->get_parent());
575 
576  // If there is a SgVariableDeclaration then it is simpler to look for it in the scope,
577  // else we have to look at each variable declaration for the SgInitializedName (which
578  // is only more expensive).
579  bool foundArrayVariableDeclaration = false;
580  if (variableDeclaration != nullptr)
581  {
582  SgScopeStatement* variableScope = variableDeclaration->get_scope();
583  switch(variableScope->variantT())
584  {
585  case V_SgBasicBlock:
586  {
587  SgBasicBlock* basicBlock = isSgBasicBlock(variableScope);
588  SgStatementPtrList statementList = basicBlock->get_statements();
589  SgStatementPtrList::iterator i = find(statementList.begin(),statementList.end(),variableDeclaration);
590  foundArrayVariableDeclaration = (i != statementList.end());
591  break;
592  }
593 
594  default:
595  {
596  printf ("Default reached, variableScope = %p = %s \n",variableScope,variableScope->class_name().c_str());
597  ROSE_ABORT();
598  }
599  }
600  }
601  else
602  {
603  // There was no variable declaration found though the symbol, so we have to look
604  // for the SgInitializedName in each SgVariableDeclaration. However there will be
605  // at least two (and hopefully no more) SgInitializedName objects for a function
606  // parameter if it also has an explicit declaration in a SgVariableDeclaration.
607  // It would be cleaner to have one, and it could be consistant with old Style K&R C,
608  // however it is not clear if this would be a problem for where we would visit the
609  // IR node twice in the traversals. Need to look at the implementation of the old
