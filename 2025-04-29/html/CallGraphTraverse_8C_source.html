349 TraverseCallGraph::TraverseCallGraph(SgIncidenceDirectedGraph* graph)
350 {
351  this->graph = graph;
352 
353  set<SgGraphNode *> nodes = graph->computeNodeSet();
354  for(set<SgGraphNode*>::iterator itn = nodes.begin(); itn != nodes.end(); itn++) {
355  SgNode* n = (*itn)->get_SgNode();
356  ROSE_ASSERT(isSgFunctionDeclaration(n));
357  assert(!isSgTemplateFunctionDeclaration(n));
358  CGFunction func(isSgFunctionDeclaration(n), graph);
359 
360  // Skip functions that are compiler generated. Beware that under edg4x, an instantiated function template or
361  // member function template is compiler generated even when the template from whence it came is not compiler
362  // generated.
363  if (!n->get_file_info()->isCompilerGenerated()) {
364  functions.insert(func);
365  } else if (isSgTemplateInstantiationFunctionDecl(n)) {
366  SgTemplateInstantiationFunctionDecl *tpl = isSgTemplateInstantiationFunctionDecl(n);
367  if (!tpl->get_templateDeclaration()->get_file_info()->isCompilerGenerated())
368  functions.insert(func);
369  } else if (isSgTemplateInstantiationMemberFunctionDecl(n)) {
370  SgTemplateInstantiationMemberFunctionDecl *tpl = isSgTemplateInstantiationMemberFunctionDecl(n);
371  if (!tpl->get_templateDeclaration()->get_file_info()->isCompilerGenerated())
372  functions.insert(func);
373  }
374  }
375 
376 
377  // initially, put all the nodes into noPred
378  // initialized numCallers
379  for(set<CGFunction>::iterator it = functions.begin(); it != functions.end(); it++)
380  {
381  noPred.insert((&*it));
382  numCallers[(&*it)] = 0;
383  }
384 
385  // next, toss out all the nodes that have predecessor nodes
386  // and compute the correct numCallers mapping
387 
388  // set of edges that have been visited by the traversal
389  set<pair<const CGFunction*, const CGFunction*> > touchedEdges;
390  for(set<CGFunction>::iterator func = functions.begin(); func != functions.end(); func++)
391  {
392  for(CGFunction::iterator it = (*func).successors(); it != (*func).end(); it++)
393  {
394  const CGFunction* target = it.getTarget(functions);
395 
396  // if the target is compiler-generated, skip it
397  if(target==NULL) continue;
398 
399  // if we haven't yet touched this edge
400  pair<const CGFunction*, const CGFunction*> edge(&(*func), target);
401  if(touchedEdges.find(edge) == touchedEdges.end())
402  {
403  // increment the target function's number of callers
404  numCallers[target]++;
405 
406  // if the given callee is currently in noPred
407  set<const CGFunction*>::iterator pred_it = noPred.find(target);
408  if(pred_it != noPred.end())
409  noPred.erase(target);
410  }
411 
412  // add this edge to touchedEdges
413  touchedEdges.insert(edge);
414  }
415  }
416 }
