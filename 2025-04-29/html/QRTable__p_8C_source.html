68  eAssert(col != QRTable::Header && row != QRTable::Header, ("type cannot be set to header!"));
69  Qt::ItemFlags flags;

73  item->setData(Qt::CheckStateRole, QVariant()); // remove check-box
74  } else if (type == QRTable::Edit) {
