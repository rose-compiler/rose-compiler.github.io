190 void ReadWriteSets::recursiveInsert(std::set<AccessSetRecord>& set, const AccessSetRecord& record) {
191  std::set<AccessSetRecord>::iterator setIt = set.find(record);
192  //If the record of this name DOES NOT exist, we don't have to merge, just insert and return, which will include everything
193  if(setIt == set.end()) {
194  set.insert(record);
195  return;
196  }
197  //If the record DOES exist already we need to merge. First we verify that they have equal metadata (safety check they should)
198  ROSE_ASSERT(record.variableName == setIt->variableName &&
199  record.globality == setIt->globality &&
200  record.nodeId == setIt->nodeId);
201 
202 
203  //Insert each field of record into the found record field set
204  AccessSetRecord setCopy = *setIt;
205 
206  //When we're merging up the tree, sometimes a pointer rereference happens in one place on a given type, but not another
207  //We want to be sure to capture the worst case.
208  setCopy.accessType = std::max(record.accessType, setIt->accessType);
209  for(const AccessSetRecord &recordField : record.fields) {
210  recursiveInsert(setCopy.fields, recordField);
211  }
212 /* for(const AccessSetRecord &recordIndex : record.indexes) {
213  recursiveInsert(setCopy.indexes, recordIndex);
214  }*/
215  set.erase(setIt);
216  set.insert(setCopy);
217

679  return MEMBERS;
680  } else {
681  noteStr = VxUtilFuncs::makeNoteString("reference found to different class",
682  accessOrigin, coarseNameClass->get_declaration()->get_type()->unparseToString());
683  return OTHER_CLASS_MEMBERS; //Accessed from a different class, so it's a global access
684  }
685 
686  } else { //Otherwise, check that it's a local variable or an argument
687  //A little annoying, any parameter arguments will be associated with the declaration attached to
688  //the definition, not the first non-defining declaration. So get the parameter list that way
689  bool isDefAncestor = SageInterface::isAncestor(funcDef, coarseName); //Local variable
690  if(isDefAncestor) {
691  return LOCALS;
692  }
693  SgFunctionDeclaration* funcDecl =funcDef->get_declaration();
694  bool isDeclAncestor = SageInterface::isAncestor(funcDecl, coarseName); //Writing function arguments is OK.
695  if(isDeclAncestor) {
696  return PARAMETERS;
697  }
698 
699  noteStr = VxUtilFuncs::makeNoteString("write to non-local variable, give leek2 this example", accessOrigin, accessOrigin->unparseToString());
700  return GLOBALITY_UNKNOWN;
701 
702  }
703 
704  }
705 
706 
707  //2. It's something other than an SgInitailizedName, figure out what it is a deal with it.
708  //The below error messages might not make sense in this new usage, where we're expecting to see a log of
709  //undefined functions. We'll see.
710  if (isSgFunctionRefExp(current)) {
711  SgFunctionRefExp* funcRef = isSgFunctionRefExp(current);
712  SgFunctionDeclaration* funcDecl = funcRef->getAssociatedFunctionDeclaration();
713  if(funcDecl) {
714  VxUtilFuncs::mlog[Sawyer::Message::Common::INFO] << "Got function declaration: " << funcDecl->get_qualified_name().getString() << std::endl;
715  return GLOBALITY_UNKNOWN;
716  } else {
717  VxUtilFuncs::mlog[Sawyer::Message::Common::WARN] << VxUtilFuncs::makeNoteString("Got SgFunctionRef but couldn't resolve it", accessOrigin, accessOrigin->unparseToString()) << std::endl;
718  return GLOBALITY_UNKNOWN;
719  }
720  }
721  if (isSgMemberFunctionRefExp(current)) {
722  SgMemberFunctionRefExp* funcRef = isSgMemberFunctionRefExp(current);
723  SgMemberFunctionDeclaration* funcDecl = funcRef->getAssociatedMemberFunctionDeclaration();
724  if(funcDecl) {
725  VxUtilFuncs::mlog[Sawyer::Message::Common::INFO] << "Got Member function declaration: " << funcDecl->get_qualified_name().getString() << std::endl;
726  return GLOBALITY_UNKNOWN;
727  } else {
728  VxUtilFuncs::mlog[Sawyer::Message::Common::WARN] << VxUtilFuncs::makeNoteString("Got SgMemberFunctionRef but couldn't resolve it", accessOrigin, accessOrigin->unparseToString()) << std::endl;
729  return GLOBALITY_UNKNOWN;
730  }
731  }
732  if (isSgTemplateFunctionRefExp(current) || isSgTemplateMemberFunctionRefExp(current)) {
733  //I used to have a case to handle this, but it turns out that
734  //TemplateFunctionRefExp only happen when an uninstantiated
735  //template calls a templated function. Which should never been
736  //seen in the RWSets. So this is left here as a safetly check.
737  VxUtilFuncs::mlog[Sawyer::Message::Common::ERROR] << "We got an SgTemplateMemberFunctionRefExp, which shouldn't be possible. Contact ROSE team." << std::endl;
738  ROSE_ABORT();
739  }
740  if (isSgConstructorInitializer(current)) {
741  //This just tells us that an object is being constructed. It
742  //doesn't really tell us anything about the globality of that
743  //object. So call it LOCALS, and let it get upgraded later if necessary
744  return LOCALS;
745  }
746  if (isSgThisExp(current)) {
747  noteStr = VxUtilFuncs::makeNoteString("\'this\' reference found", accessOrigin, accessOrigin->unparseToString());
748  return MEMBERS;
749  }
750 
751  VxUtilFuncs::mlog[Sawyer::Message::Common::ERROR] <<
752  "meetsGlobalityConstraints: " <<
753  current->get_file_info()->get_filename() << ":" <<
754  current->get_file_info()->get_line() << "-" << current->get_file_info()->get_col()<<std::endl;
755  VxUtilFuncs::mlog[Sawyer::Message::Common::ERROR] <<
756  "In meetsGlobalityConstraints: unhandled reference type:" <<
757  current->class_name()<<std::endl;
758  ROSE_ABORT();
