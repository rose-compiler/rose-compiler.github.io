157 bool
158 X86::isFunctionCallSlow(const std::vector<SgAsmInstruction*> &insns, Address *target, Address *return_va) const {
159  if (isFunctionCallFast(insns, target, return_va))
160  return true;
161 
162  // The following stuff works only if we have a relatively complete AST.
163  static const size_t EXECUTION_LIMIT = 10; // max size of basic blocks for expensive analyses
164  if (insns.empty())
165  return false;
166  auto last = isSgAsmX86Instruction(insns.back());
167  ASSERT_not_null(last);
168  SgAsmFunction *func = SageInterface::getEnclosingNode<SgAsmFunction>(last);
169  SgAsmInterpretation *interp = SageInterface::getEnclosingNode<SgAsmInterpretation>(func);
170 
171  // Slow method: Emulate the instructions and then look at the EIP and stack. If the EIP points outside the current function and
172  // the top of the stack holds an address of an instruction within the current function, then this must be a function call.
173  if (interp && insns.size() <= EXECUTION_LIMIT) {
174  using namespace Rose::BinaryAnalysis;
175  using namespace Rose::BinaryAnalysis::InstructionSemantics;
176  using namespace Rose::BinaryAnalysis::InstructionSemantics::SymbolicSemantics;
177  const InstructionMap &imap = interp->get_instructionMap();
178  RegisterDictionary::Ptr regdict = registerDictionary();
179  SmtSolverPtr solver = SmtSolver::instance(Rose::CommandLine::genericSwitchArgs.smtSolver);
180  BaseSemantics::RiscOperators::Ptr ops = RiscOperators::instanceFromRegisters(regdict, solver);
181  ASSERT_not_null(ops);
182  DispatcherX86::Ptr dispatcher = DispatcherX86::promote(newInstructionDispatcher(ops));
183  SValue::Ptr orig_esp = SValue::promote(ops->peekRegister(dispatcher->REG_anySP));
184  try {
185  for (size_t i = 0; i < insns.size(); ++i)
186  dispatcher->processInstruction(insns[i]);
187  } catch (const BaseSemantics::Exception &e) {
188  return false;
189  }
190 
191  // If the next instruction address is concrete but does not point to a function entry point, then this is not a call.
192  SValue::Ptr eip = SValue::promote(ops->peekRegister(dispatcher->REG_anyIP));
193  if (auto target_va = eip->toUnsigned()) {
194  SgAsmFunction *target_func = SageInterface::getEnclosingNode<SgAsmFunction>(imap.get_value_or(*target_va, NULL));
195  if (!target_func || *target_va != target_func->get_entryVa())
196  return false;
197  }
198 
199  // If nothing was pushed onto the stack, then this isn't a function call.
200  const size_t spWidth = dispatcher->REG_anySP.nBits();
201  SValue::Ptr esp = SValue::promote(ops->peekRegister(dispatcher->REG_anySP));
202  SValue::Ptr stack_delta = SValue::promote(ops->add(esp, ops->negate(orig_esp)));
203  SValue::Ptr stack_delta_sign = SValue::promote(ops->extract(stack_delta, spWidth-1, spWidth));
204  if (stack_delta_sign->isFalse())
205  return false;
206 
207  // If the top of the stack does not contain a concrete value or the top of the stack does not point to an instruction in
208  // this basic block's function, then this is not a function call.
209  const size_t ipWidth = dispatcher->REG_anyIP.nBits();
210  SValue::Ptr top = SValue::promote(ops->peekMemory(dispatcher->REG_SS, esp, esp->undefined_(ipWidth)));
211  if (auto va = top->toUnsigned()) {
212  SgAsmFunction *return_func = SageInterface::getEnclosingNode<SgAsmFunction>(imap.get_value_or(*va, NULL));
213  if (!return_func || return_func!=func) {
214  return false;
215  }
216  } else {
217  return false;
218  }
219 
220  // Since EIP might point to a function entry address and since the top of the stack contains a pointer to an instruction in
221  // this function, we assume that this is a function call.
222  if (target)
223  eip->toUnsigned().assignTo(*target);
224  if (return_va)
225  top->toUnsigned().assignTo(*return_va);
226  return true;
227  }
228 
229  // Similar to the above method, but works when all we have is the basic block (e.g., this case gets hit quite a bit from the
230  // Partitioner). Returns true if, after executing the basic block, the top of the stack contains the fall-through address of
231  // the basic block. We depend on our caller to figure out if EIP is reasonably a function entry address.
232  if (!interp && insns.size() <= EXECUTION_LIMIT) {
233  using namespace Rose::BinaryAnalysis;
234  using namespace Rose::BinaryAnalysis::InstructionSemantics;
235  using namespace Rose::BinaryAnalysis::InstructionSemantics::SymbolicSemantics;
236  SmtSolverPtr solver = SmtSolver::instance(Rose::CommandLine::genericSwitchArgs.smtSolver);
237  SgAsmX86Instruction *x86insn = isSgAsmX86Instruction(insns.front());
238  ASSERT_not_null(x86insn);
239 #if 1 // [Robb P. Matzke 2015-03-03]: FIXME[Robb P. Matzke 2015-03-03]: not ready yet; x86-64 semantics still under construction
240  if (x86insn->get_addressSize() != x86_insnsize_32)
241  return false;
242 #endif
243  RegisterDictionary::Ptr regdict = registerDictionary();
244  const RegisterDescriptor SP = regdict->findLargestRegister(x86_regclass_gpr, x86_gpr_sp);
245  BaseSemantics::RiscOperators::Ptr ops = RiscOperators::instanceFromRegisters(regdict, solver);
246  DispatcherX86::Ptr dispatcher = DispatcherX86::promote(newInstructionDispatcher(ops));
247  try {
248  for (size_t i = 0; i < insns.size(); ++i)
249  dispatcher->processInstruction(insns[i]);
250  } catch (const BaseSemantics::Exception &e) {
251  return false;
252  }
253 
254  // Look at the top of the stack
255  const size_t ipWidth = dispatcher->REG_anyIP.nBits();
256  SValue::Ptr top = SValue::promote(ops->peekMemory(dispatcher->REG_SS, ops->peekRegister(SP),
257  ops->protoval()->undefined_(ipWidth)));
258  if (top->toUnsigned().orElse(0) == last->get_address() + last->get_size()) {
259  if (target) {
260  SValue::Ptr eip = SValue::promote(ops->peekRegister(dispatcher->REG_anyIP));
261  eip->toUnsigned().assignTo(*target);
262  }
263  if (return_va)
264  top->toUnsigned().assignTo(*return_va);
265  return true;
266  }
267  }
268 
269  return false;
270 }

325 AddressSet
326 X86::getSuccessors(SgAsmInstruction *insn_, bool &complete) const {
327  auto insn = isSgAsmX86Instruction(insn_);
328  ASSERT_not_null(insn);
329 
330  AddressSet retval;
331  complete = true; /*assume true and prove otherwise*/
332 
333  switch (insn->get_kind()) {
334  case x86_call:
335  case x86_farcall:
336  case x86_jmp:
337  case x86_farjmp:
338  /* Unconditional branch to operand-specified address. We cannot assume that a CALL instruction returns to the
339  * fall-through address. */
340  if (Sawyer::Optional<Address> va = branchTarget(insn)) {
341  retval.insert(*va);
342  } else {
343  complete = false;
344  }
345  break;
346 
347  case x86_ja:
348  case x86_jae:
349  case x86_jb:
350  case x86_jbe:
351  case x86_jcxz:
352  case x86_jecxz:
353  case x86_jrcxz:
354  case x86_je:
355  case x86_jg:
356  case x86_jge:
357  case x86_jl:
358  case x86_jle:
359  case x86_jne:
360  case x86_jno:
361  case x86_jns:
362  case x86_jo:
363  case x86_jpe:
364  case x86_jpo:
365  case x86_js:
366  case x86_loop:
367  case x86_loopnz:
368  case x86_loopz:
369  /* Conditional branches to operand-specified address */
370  if (Sawyer::Optional<Address> va = branchTarget(insn)) {
371  retval.insert(*va);
372  } else {
373  complete = false;
374  }
375  retval.insert(insn->get_address() + insn->get_size());
376  break;
377 
378  case x86_int: // assumes interrupts return
379  case x86_int1:
380  case x86_int3:
381  case x86_into:
382  case x86_syscall: {
383  retval.insert(insn->get_address() + insn->get_size()); // probable return point
384  complete = false;
385  break;
386  }
387 
388  case x86_ret:
389  case x86_iret:
390  case x86_rsm:
391  case x86_sysret:
392  case x86_ud2:
393  case x86_retf: {
394  /* Unconditional branch to run-time specified address */
395  complete = false;
396  break;
397  }
398 
399  case x86_hlt: {
400  /* Instructions having no successor. */
401  break;
402  }
403 
404  case x86_unknown_instruction: {
405  /* Instructions having unknown successors */
406  complete = false;
407  break;
408  }
409 
410  default: {
411  /* Instructions that always fall through to the next instruction */
412  retval.insert(insn->get_address() + insn->get_size());
413  break;
414  }
415  }
416  return retval;
417 }

419 AddressSet
420 X86::getSuccessors(const std::vector<SgAsmInstruction*>& insns, bool &complete, const MemoryMap::Ptr &initial_memory) const {
421  Sawyer::Message::Stream debug(mlog[DEBUG]);
422  using namespace Rose::BinaryAnalysis::InstructionSemantics;
423 
424  if (debug) {
425  debug <<"X86::getSuccessors(" <<StringUtility::addrToString(insns.front()->get_address())
426  <<" for " <<insns.size() <<" instruction" <<(1==insns.size()?"":"s") <<"):" <<"\n";
427  }
428 
429  AddressSet successors = Base::getSuccessors(insns, complete/*out*/);
430 
431  // If we couldn't determine all the successors, or a cursory analysis couldn't narrow it down to a single successor then we'll
432  // do a more thorough analysis now. In the case where the cursory analysis returned a complete set containing two successors, a
433  // thorough analysis might be able to narrow it down to a single successor. We should not make special assumptions about CALL
434  // and FARCALL instructions -- their only successor is the specified address operand.
435  if (!complete || successors.size() > 1) {
436  RegisterDictionary::Ptr regdict = registerDictionary();
437  PartialSymbolicSemantics::RiscOperators::Ptr ops = PartialSymbolicSemantics::RiscOperators::instanceFromRegisters(regdict);
438  ops->set_memory_map(initial_memory);
439  BaseSemantics::Dispatcher::Ptr cpu = newInstructionDispatcher(ops);
440 
441  try {
442  for (SgAsmInstruction *insn: insns) {
443  cpu->processInstruction(insn);
444  SAWYER_MESG(debug) <<" state after " <<insn->toString() <<"\n" <<*ops;
445  }
446  BaseSemantics::SValue::Ptr ip = ops->peekRegister(regdict->instructionPointerRegister());
447  if (auto ipval = ip->toUnsigned()) {
448  successors.clear();
449  successors.insert(*ipval);
450  complete = true;
451  }
452  } catch(const BaseSemantics::Exception &e) {
453  /* Abandon entire basic block if we hit an instruction that's not implemented. */
454  debug <<e <<"\n";
455  }
456  }
457 
458  if (debug) {
459  debug <<" successors:";
460  for (Address va: successors.values())
461  debug <<" " <<StringUtility::addrToString(va);
462  debug <<(complete?"":"...") <<"\n";
463  }
464 
465  return successors;
466 }

518 CallingConvention::Definition::Ptr
519 X86::cc_cdecl(const size_t bitsPerWord) const {
520  // Register prefix letter for things like "ax", "eax", "rax"
521  const std::string prefix = registerPrefix(bitsPerWord);
522  RegisterDictionary::Ptr regdict = registerDictionary();
523  auto cc = CC::Definition::instance("cdecl", "x86-" + boost::lexical_cast<std::string>(bitsPerWord) + " cdecl", constPtr());
524  const RegisterDescriptor SP = regdict->stackPointerRegister();
525 
526  //==== Address locations ====
527  cc->bitsPerWord(bitsPerWord);
528  cc->instructionPointerRegister(regdict->instructionPointerRegister());
529  cc->returnAddressLocation(ConcreteLocation(SP, 0, regdict));
530 
531  //==== Stack characteristics ====
532  cc->stackPointerRegister(SP);
533  cc->stackDirection(CC::StackDirection::GROWS_DOWN);
534  cc->nonParameterStackSize(bitsPerWord >> 3); // return address
535 
536  //==== Function parameters ====
537  // All parameters are passed on the stack.
538  cc->stackParameterOrder(CC::StackParameterOrder::RIGHT_TO_LEFT);
539  cc->stackCleanup(CC::StackCleanup::BY_CALLER);
540  {
541  auto pool = CC::StoragePoolStack::instance("args-on-stack", CC::isAnyType(), constPtr());
542  pool->initialOffset(bitsPerWord >> 3); // return address
543  cc->argumentValueAllocator()->append(pool);
544  }
545 
546  //==== Other inputs ====
547  // direction flag is always assumed to be valid
548  cc->nonParameterInputs().push_back(ConcreteLocation(regdict->findOrThrow("df"), regdict));
549  // code segment register is assumed to be valid
550  cc->nonParameterInputs().push_back(ConcreteLocation(regdict->findOrThrow("cs"), regdict));
551  // data segment register is assumed to be valid
552  cc->nonParameterInputs().push_back(ConcreteLocation(regdict->findOrThrow("ds"), regdict));
553  // stack segment register is assumed to be valid
554  cc->nonParameterInputs().push_back(ConcreteLocation(regdict->findOrThrow("ss"), regdict));
555 
556  //==== Return values ====
557  if (const RegisterDescriptor AX = regdict->findOrThrow(prefix + "ax")) {
558  auto nonFpReturns = CC::StoragePoolEnumerated::instance("non-fp return values", CC::isNonFpNotWiderThan(bitsPerWord));
559  nonFpReturns->append(ConcreteLocation(AX, regdict));
560  cc->returnValueAllocator()->append(nonFpReturns);
561  cc->appendOutputParameter(AX);
562  }
563 
564  if (const RegisterDescriptor ST0 = regdict->findLargestRegister(x86_regclass_st, x86_st_0)) {
565  auto fpReturns = CC::StoragePoolEnumerated::instance("fp return values", CC::isFpNotWiderThan(ST0.nBits()));
566  fpReturns->append(ConcreteLocation(ST0, regdict));
567  cc->returnValueAllocator()->append(fpReturns);
568  cc->appendOutputParameter(ST0);
569  }
570 
571  // The stack pointer is an output because it is changed by the callee (incremented by four when the function returns). It does
572  // not hold a return value in the normal sense of the word, and thus isn't added to the return value allocator functions.
573  cc->appendOutputParameter(SP);
574 
575  //==== Scratch registers ====
576  // (i.e., modified, not callee-saved, not return registers)
577  cc->scratchRegisters().insert(regdict->findOrThrow(prefix + "cx"));
578  cc->scratchRegisters().insert(regdict->findOrThrow(prefix + "dx"));
579  cc->scratchRegisters().insert(regdict->instructionPointerRegister());
580  cc->scratchRegisters().insert(regdict->findOrThrow(prefix + "flags"));
581  if (const RegisterDescriptor FPSTATUS = regdict->findLargestRegister(x86_regclass_flags, x86_flags_fpstatus))
582  cc->scratchRegisters().insert(FPSTATUS);
583 
584  //==== Callee-saved registers ====
585  // Everything else
586  RegisterParts regParts = regdict->getAllParts() - cc->getUsedRegisterParts();
587  std::vector<RegisterDescriptor> registers = regParts.extract(regdict);
588  cc->calleeSavedRegisters().insert(registers.begin(), registers.end());
589 
590  return cc;
591 }

604 
605  //==== Stack characteristics ====
606  cc->stackPointerRegister(SP);
607  cc->stackDirection(CC::StackDirection::GROWS_DOWN);
608  cc->nonParameterStackSize(bitsPerWord >> 3); // return address
609 
610  //==== Function parameters ====
611  // All parameters are passed on the stack
612  cc->stackParameterOrder(CC::StackParameterOrder::RIGHT_TO_LEFT);
613  cc->stackCleanup(CC::StackCleanup::BY_CALLEE);
614  {
615  auto pool = CC::StoragePoolStack::instance("args-on-stack", CC::isAnyType(), constPtr());
616  pool->initialOffset(bitsPerWord >> 3); // return address
617  cc->argumentValueAllocator()->append(pool);
618  }
619 
620  //==== Other inputs ====
621  // direction flag is always assumed to be valid
622  cc->nonParameterInputs().push_back(ConcreteLocation(regdict->findOrThrow("df"), regdict));
623  // code segment register is assumed to be valid
624  cc->nonParameterInputs().push_back(ConcreteLocation(regdict->findOrThrow("cs"), regdict));
625  // data segment register is assumed to be valid
626  cc->nonParameterInputs().push_back(ConcreteLocation(regdict->findOrThrow("ds"), regdict));
627  // stack segment register is assumed to be valid
628  cc->nonParameterInputs().push_back(ConcreteLocation(regdict->findOrThrow("ss"), regdict));
629 
630  //==== Return values ====
631  if (const RegisterDescriptor AX = regdict->findOrThrow(prefix + "ax")) {
632  auto nonFpReturns = CC::StoragePoolEnumerated::instance("non-fp return values", CC::isNonFpNotWiderThan(bitsPerWord));
633  nonFpReturns->append(ConcreteLocation(AX, regdict));
634  cc->returnValueAllocator()->append(nonFpReturns);
635  cc->appendOutputParameter(AX);
636  }
637 
638  if (const RegisterDescriptor ST0 = regdict->findLargestRegister(x86_regclass_st, x86_st_0)) {
639  auto fpReturns = CC::StoragePoolEnumerated::instance("fp return values", CC::isFpNotWiderThan(ST0.nBits()));
640  fpReturns->append(ConcreteLocation(ST0, regdict));
641  cc->returnValueAllocator()->append(fpReturns);
642  cc->appendOutputParameter(ST0);
643  }
644 
645  // The stack pointer is an output because it is changed by the callee (incremented by four when the function returns). It does
646  // not hold a return value in the normal sense of the word, and thus isn't added to the return value allocator functions.
647  cc->appendOutputParameter(SP);
648 
649  //==== Scratch registers ====
650  // Modified, not callee-saved, not return registers
651  cc->scratchRegisters().insert(regdict->findOrThrow(prefix + "cx"));
652  cc->scratchRegisters().insert(regdict->findOrThrow(prefix + "dx"));
653  cc->scratchRegisters().insert(regdict->instructionPointerRegister());
654  cc->scratchRegisters().insert(regdict->findOrThrow(prefix + "flags"));
655  if (const RegisterDescriptor FPSTATUS = regdict->findLargestRegister(x86_regclass_flags, x86_flags_fpstatus))
656  cc->scratchRegisters().insert(FPSTATUS);
657 
658  //==== Callee-saved registers ====
659  // Everything else
660  RegisterParts regParts = regdict->getAllParts() - cc->getUsedRegisterParts();
661  std::vector<RegisterDescriptor> registers = regParts.extract(regdict);
662  cc->calleeSavedRegisters().insert(registers.begin(), registers.end());
663 
664  return cc;
665 }
