140  void jserver_start(JvmT* je) {
141  JavaVMInitArgs jvm_args; /* VM initialization args. */
142  // Pei-Hung (7/6/2017): Bumped to latest(?), unsure why 1_4 was used
143  jvm_args.version = JNI_VERSION_1_6;
144  jvm_args.ignoreUnrecognized = JNI_FALSE;
145 
146  //----------------------------------------------------------------------------
147  // Add all our JVM options
148  //----------------------------------------------------------------------------
149  // TOO1 (2/11/2014): JVM options now stored in the Cmdline::Java::ECJ namespace.
150  std::list<std::string> jvm_options = Rose::Cmdline::Java::Ecj::jvm_options;
151 
152  std::string classpath = Rose::Cmdline::Java::Ecj::GetRoseClasspath();
153  jvm_options.push_back(classpath);
154 
155  // Pei-Hung (7/6/2017): Increasing stacksize fixes crashes on some rhel7 systems
156  std::string stack_option = "-Xss2m";
157  jvm_options.push_back(stack_option);
158 
159  jvm_args.nOptions = jvm_options.size();
160  jvm_args.options = new JavaVMOption[jvm_args.nOptions];
161  for(int i=0; i < jvm_args.nOptions; ++i) {
162  std::string jvm_option = jvm_options.front();
163  if (Rose::Cmdline::verbose > 0)
164  std::cout << "[INFO] [ECJ] jvm_option[" << i << "] = " << jvm_option << std::endl;
165  jvm_args.options[i].optionString = strdup(jvm_option.c_str());
166  jvm_options.pop_front();
167  }
168 
169  //----------------------------------------------------------------------------
170  // Create and load the Java VM.
171  //----------------------------------------------------------------------------
172  int res = JNI_CreateJavaVM(&(je->jvm), (void **)&(je->env), &jvm_args);
173 
174  if (res<0 || je->jvm==NULL || je->env==NULL) {
175  exit(1);
176  }
177 
178  return ;
179  }

210  jobjectArray jserver_getJavaStringArray(int argc, char **argv) {
211  jobjectArray argsStringArray = NULL;
212  jclass stringClass;
213  int i;
214  JNIEnv * env = get_env();
215 
216  /* We need the String class because that is the underlying type of the array. */
217  stringClass = jserver_getJavaStringClass();
218  if(stringClass == NULL)
219  return NULL;
220 
221  /* Build a new object array. Params are: env, length, class type of the array, initial object(?). */
222  argsStringArray = env->NewObjectArray((argc-1), stringClass, NULL);
223 
224  if(argsStringArray == NULL)
225  return NULL;
226 
227  /* Put all args from argv, after the first (which is this program's
228  name) into the array of Strings for JavaMain. The args array
229  for Java does not include the program name. */
230  for(i = 1; i < argc; i++) {
231  env->SetObjectArrayElement(argsStringArray, (jsize)i-1, jserver_getJavaString(argv[i]));
232  }
233  return argsStringArray;
234  }
