622 class IsModifiedBadlyVisitor: public AstSimpleProcessing {
623  SgInitializedName* initname;
624  bool& safe;
625 
626  public:
627  IsModifiedBadlyVisitor(SgInitializedName* initname, bool& safe):
628  initname(initname), safe(safe) {}
629 
630  virtual void visit(SgNode* n) {
631  SgVarRefExp* vr = isSgVarRefExp(n);
632  if (vr && vr->get_symbol()->get_declaration() == initname) {
633  switch (vr->get_parent()->variantT()) {
634  case V_SgReturnStmt:
635  case V_SgExprStatement:
636  case V_SgIfStmt:
637  case V_SgWhileStmt:
638  case V_SgDoWhileStmt:
639  case V_SgSwitchStatement:
640  case V_SgCaseOptionStmt:
641  case V_SgForStatement:
642  case V_SgForInitStatement:
643 #ifdef FD_DEBUG
644  cout << "Statement: Variable " << initname->get_name().getString() << " is safe" << endl;
645 #endif
646  // Safe
647  break;
648 
649  case V_SgPlusPlusOp:
650  case V_SgMinusMinusOp:
651 #ifdef FD_DEBUG
652  cout << "Inc/dec: Variable " << initname->get_name().getString() << " is safe" << endl;
653 #endif
654  // Safe
655  break;
656 
657  case V_SgAddOp:
658  case V_SgSubtractOp:
659  case V_SgMinusOp:
660  case V_SgUnaryAddOp:
661  case V_SgNotOp:
662  case V_SgPointerDerefExp:
663  case V_SgBitComplementOp:
664  case V_SgThrowOp:
665  case V_SgEqualityOp:
666  case V_SgLessThanOp:
667  case V_SgLessOrEqualOp:
668  case V_SgGreaterThanOp:
669  case V_SgGreaterOrEqualOp:
670  case V_SgNotEqualOp:
671  case V_SgMultiplyOp:
672  case V_SgDivideOp:
673  case V_SgIntegerDivideOp:
674  case V_SgModOp:
675  case V_SgAndOp:
676  case V_SgOrOp:
677  case V_SgBitAndOp:
678  case V_SgBitOrOp:
679  case V_SgBitXorOp:
680  case V_SgCommaOpExp:
681  case V_SgLshiftOp:
682  case V_SgRshiftOp:
683  case V_SgAssignInitializer:
684 #ifdef FD_DEBUG
685  cout << "Non-mutating: Variable " << initname->get_name().getString() << " is safe" << endl;
686 #endif
687  // Safe
688  break;
689 
690  case V_SgExprListExp:
691  if (isSgFunctionCallExp(n->get_parent()->get_parent())) {
692  if (isPotentiallyModified(vr, n->get_parent()->get_parent())) {
693 #ifdef FD_DEBUG
694  cout << "Function call: Variable " << initname->get_name().getString() << " is unsafe" << endl;
695 #endif
696  safe = false;
697  }
698  } else if (isSgConstructorInitializer(n->get_parent()->get_parent())) {
699 #ifdef FD_DEBUG
700  cout << "Constructor: Variable " << initname->get_name().getString() << " is unsafe" << endl;
701 #endif
702  safe = false;
703  // FIXME: constructors
704  } else {
705  cerr << n->get_parent()->get_parent()->sage_class_name() << endl;
706  assert (!"Unknown SgExprListExp case");
707  }
708  break;
709 
710  case V_SgAssignOp:
711  case V_SgPlusAssignOp:
712  case V_SgMinusAssignOp: {
713  SgBinaryOp* binop = isSgBinaryOp(vr->get_parent());
714  SgExpression* rhs = binop->get_rhs_operand();
715  bool lhs_good = (binop->get_lhs_operand() == vr);
716 #ifdef FD_DEBUG
717  cout << "Assign case for " << initname->get_name().getString() << endl;
718  cout << "lhs_good = " << (lhs_good ? "true" : "false") << endl;
719 #endif
720  SgAddOp* rhs_a = isSgAddOp(rhs);
721  SgExpression* rhs_a_lhs = rhs_a ? rhs_a->get_lhs_operand() : 0;
722  SgExpression* rhs_a_rhs = rhs_a ? rhs_a->get_rhs_operand() : 0;
723  if (lhs_good) {
724  if (isSgValueExp(binop->get_rhs_operand())) {
725  // Safe
726  } else if (isSgVarRefExp(rhs)) {
727  // Safe
728  } else if (isSgCastExp(binop->get_rhs_operand()) &&
729  isSgValueExp(isSgCastExp(binop->get_rhs_operand())
730  ->get_operand())) {
731  // Safe
732  } else if (isSgAssignOp(binop) &&
733  rhs_a &&
734  ((isSgVarRefExp(rhs_a_lhs) &&
735  isSgVarRefExp(rhs_a_lhs)->get_symbol()
736  ->get_declaration() == initname) ||
737  (isSgVarRefExp(rhs_a_rhs) &&
738  isSgVarRefExp(rhs_a_rhs)->get_symbol()
739  ->get_declaration() == initname))) {
740  // Safe
741  } else {
742 #ifdef FD_DEBUG
743  cout << "Assign: Variable " << initname->get_name().str() << " is unsafe because of " << binop->unparseToString() << ": " << binop->get_rhs_operand()->sage_class_name() << endl;
744 #endif
745  safe = false;
746  }
747  } else {
748  // Safe: RHS of assignment
749  }
750  }
751  break;
752 
753  default: {
754 #ifdef FD_DEBUG
755  cout << "Default: Variable " << initname->get_name().str() << " is unsafe because of " << vr->get_parent()->unparseToString() << ": " << vr->get_parent()->sage_class_name() << endl;
756 #endif
757  safe = false;
758  }
759  break;
760  }
761  }
762  }

767 void simpleIndexFiniteDifferencing(SgNode* root) {
768  moveForDeclaredVariables(root);
769  SgNode* proj = root;
770  while (proj && !isSgProject(proj)) proj = proj->get_parent();
771  ROSE_ASSERT (proj);
772  AstTests::runAllTests(isSgProject(proj));
773  rewrite(getAlgebraicRules(), root); // This might modify root
774  FdFindFunctionsVisitor ffv;
775  ffv.traverse(root, preorder);
776  for (unsigned int x = 0; x < ffv.functions.size(); ++x) {
777 #ifdef FD_DEBUG
778  cout << "Working on function " << ffv.functions[x]->get_declaration()->get_name().str() << endl;
779 #endif
780  SgBasicBlock* body = ffv.functions[x]->get_body();
781  FdFindInitnamesVisitor fiv;
782  fiv.traverse(ffv.functions[x]->get_declaration(), preorder);
783  vector<SgInitializedName*> initnames = fiv.initnames;
784  set<SgInitializedName*> safe_vars;
785  for (vector<SgInitializedName*>::iterator i = initnames.begin();
786  i != initnames.end(); ++i) {
787  SgInitializedName* initname = *i;
788 #ifdef FD_DEBUG
789  cout << "Found variable " << initname->get_name().str() << endl;
790 #endif
791  bool safe = true;

793  if (safe)
794  safe_vars.insert(initname);
795 #ifdef FD_DEBUG
796  cout << "Variable " << initname->get_name().str() << " is " << (safe ? "" : "not ") << "safe" << endl;
797 #endif
798  }
799  FdFindMultiplicationsVisitor fmv;
800  fmv.traverse(body, postorder);
801  vector<SgMultiplyOp*> mult_exprs;
802  for (unsigned int i = 0; i < fmv.exprs.size(); ++i) {
803  bool alreadyProcessed = false;

805  if (expressionTreeEqual(fmv.exprs[i], fmv.exprs[j])) {
806  alreadyProcessed = true;

815 #endif

818  SgExpression* expr2 = fmv.exprs[i]->get_rhs_operand();
819  bool isConst1 = isSgValueExp(expr1);
820  bool isSafeVar1 = isSgVarRefExp(expr1) &&
821  safe_vars.find(isSgVarRefExp(expr1)->get_symbol()
822  ->get_declaration())
823  != safe_vars.end();
824  bool isGood1 = isConst1 || isSafeVar1;
825  bool isConst2 = isSgValueExp(expr2);
826  bool isSafeVar2 = isSgVarRefExp(expr2) &&
827  safe_vars.find(isSgVarRefExp(expr2)->get_symbol()
828  ->get_declaration())
829  != safe_vars.end();
830  bool isGood2 = isConst2 || isSafeVar2;
831 #ifdef FD_DEBUG
832  cout << boolalpha << "isGood1 = " << isGood1 << ", isGood2 = " << isGood2 << endl;
833 #endif
834  if (isGood1 && isGood2) {
835 #ifdef FD_DEBUG
836  cout << "Expression is good to run FD on" << endl;
837 #endif
838  mult_exprs.push_back(fmv.exprs[i]);
839  } else {
840 #ifdef FD_DEBUG

842 #endif
843  }
844  }

846  doFiniteDifferencingOne(mult_exprs[i], body,
847  getFiniteDifferencingRules());
