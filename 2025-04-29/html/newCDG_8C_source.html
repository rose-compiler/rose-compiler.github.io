168 bool ControlDependenceGraph::checkCycle(const ControlFlowGraph& cfg)
169 {
170  // Add an edge from the exit to entry in the CFG, then count the number of
171  // strongly connected components. If the number is greater than 1, there is
172  // a cycle in the CFG which has no exit.
173  ControlFlowGraph cfgCopy = cfg;
174  boost::add_edge(cfgCopy.getExit(), cfgCopy.getEntry(), cfgCopy);
175 
176  std::vector<int> component(num_vertices(cfgCopy));
177 
178 #ifdef __INTEL_COMPILER
179  // DQ (8/28/2015): from Intel v14 icpc:
180  int num = 0;
181  printf ("Error: code not supported by Intel v14 compiler \n");
182  ROSE_ABORT();
183 #else
184  int num = boost::strong_components(cfgCopy, &component[0]);
185 #endif
186  return num == 1;
187 }

223  for (const VertexVertexMap& vv : iDom) {
224  children[vv.second].insert(vv.first);
225  }

228  ROSE_ASSERT(children.find(entry) != children.end());
229

231  std::vector<CFGVertexT> vertices;
232  appendSuccessors(entry, vertices, children);

237  // control dependence type (true, false or case).
238  DominanceFrontiersT domFrontiers;
239 
240  // Start to build the dominance frontiers.
241  while (!vertices.empty())
242  {
243  CFGVertexT v = vertices.back();
244  vertices.pop_back();
245 
246  for (const CFGEdgeT& e : boost::out_edges(v, cfg))
