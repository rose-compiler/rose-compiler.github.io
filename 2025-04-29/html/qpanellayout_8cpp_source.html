88 QString QPanelLayout::serialized() const
89 {
90  /*
91  Scheme :
92 
93  QPanelLayout::Position '{' comma-separated list of identifiers '}'
94  */
95 
96  QHash<int, QString> posMap;
97 
98  for ( int i = 0; i < m_list.size(); ++i )
99  {
100  PanelWrapper *wrapper = m_list.at(i);
101  Position position = wrapper->position;
102 
103  QPanel *panel = qobject_cast<QPanel*>(wrapper->item->widget());
104 
105  if ( !panel )
106  continue;
107 
108  if ( !posMap.contains(position) )
109  {
110  posMap[position] = QString::number(position) + "{" + panel->id() + "}";
111  } else {
112  QString& ref = posMap[position];
113 
114  ref.insert(ref.count() - 2, QString(",") + panel->id());
115  }
116  }
117 
118  return QStringList(posMap.values()).join("");
119 }

124 void QPanelLayout::addSerialized(const QString& layout)
125 {
126  //qDebug("layout : %s", qPrintable(layout));
127 
128  int last = 0, i = 0;
129  bool inList = false;
130  Position position = West;
131 
132  while ( i < layout.length() )
133  {
134  if ( inList )
135  {
136  if ( layout.at(i) == '}' )
137  inList = false;
138 
139  if ( !inList || (layout.at(i) == ',') )
140  {
141  QPanel *panel = QPanel::panel(layout.mid(last, i - last), m_parent);
142 
143  if ( panel )
144  {
145  panel->attach(m_parent);
146  addWidget(panel, position);
147 
148  //qDebug("\tpanel : %s", qPrintable(layout.mid(last, i - last)));
149  }
150 
151  last = i + 1;
152  }
153  } else if ( layout.at(i) == '{' ) {
154  inList = true;
155  position = Position(layout.mid(last, i - last).toInt());
156 
157  //qDebug("position : %i [%s]", position, qPrintable(layout.mid(last, i - last)));
158 
159  last = i + 1;
160  }
161 
162  ++i;
163  }
164 
165  update();
166 }

410 QSize QPanelLayout::calculateSize(SizeType sizeType) const
411 {
412  QSize totalSize;
413 
414  for ( int i = 0; i < m_list.size(); ++i )
415  {
416  QSize itemSize;
417  PanelWrapper *wrapper = m_list.at(i);
418  Position position = wrapper->position;
419 
420  if ( sizeType == MinimumSize )
421  itemSize = wrapper->item->minimumSize();
422  else // ( sizeType == SizeHint )
423  itemSize = wrapper->item->sizeHint();
424 
425  if ( (position == North) || (position == South) ) // || (position == Center) )
426  totalSize.rheight() += itemSize.height();
427 
428  if ( (position == West) || (position == East) ) // || (position == Center) )
429  totalSize.rwidth() += itemSize.width();
430 
431  }
432 
433  return totalSize;
434 }
