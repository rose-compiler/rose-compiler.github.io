86 std::string PtrAnal::
87 Get_VarName(AstInterface& fa, const AstNodePtr& rhs)
88 {
89  NameMap::const_iterator p = namemap.find(rhs.get_ptr());
90  if (p != namemap.end()) {
91  VarRef cur = (*p).second;
92  return cur.name;
93  }
94 
95  std::string readname, res;
96  AstNodePtr readscope;
97  AstNodePtr lhs1, rhs1;
98  bool readlhs;
99 
100  if (fa.IsVarRef(rhs, 0, &readname, &readscope)) {
101  assert(readname != "");
102  res = Local_GetVarName(fa, readscope, readname);
103  }
104  else if (fa.IsConstant(rhs,&res,&readname)) {
105  if (res == "string") {
106  readname = Local_GetVarName(fa, rhs);
107  Stmt stmt = allocate_x(readname);
108  stmts_pushback(stmts,stmt);
109  res= readname;
110  }
111  else
112  res = Local_GetConstName(fa, readname);
113  }
114  else if (fa.IsUnaryOp(rhs) || fa.IsBinaryOp(rhs) || fa.IsFunctionCall(rhs)) {
115  readname = Local_GetVarName(fa, rhs);
116  ProcessExpression(fa, readname, rhs);
117  res = readname;
118  }
119  else if (fa.IsAssignment(rhs,&lhs1,&rhs1,&readlhs)) {
120  ProcessAssign(fa, lhs1, rhs1, readlhs);
121  VarRef cur = namemap[lhs1.get_ptr()];

124  ROSE_ABORT();

126  namemap[rhs.get_ptr()] = cur;

128  }

131  ROSE_ABORT();
132  }

272 ProcessMod(AstInterface& fa, const std::string& readname,
273  std::list<std::string>& fields, const AstNodePtr& mod)
274 {
275  std::string modname;

278  modname = Get_VarName(fa,p);

282  stmts_pushback(stmts,stmt_last);
283  namemap[mod.get_ptr()] = VarRef(stmt_last,modname);
284  }
285  else {
286  AstInterface::OperatorEnum op;
287  AstNodePtr p2;

293  namemap[mod.get_ptr()] = VarRef(stmt_last, readname);
294  }
295  else if (fa.IsBinaryOp(mod,&op,&p,&p2)) {
296  if (op==AstInterface::BOP_DOT_ACCESS) {
297  std::string field = Local_GetFieldName(fa, p2);
298  fields.push_front(field);
299  ProcessMod(fa, readname, fields, p);
300  Stmt stmt_last = stmts_back(stmts);
301  namemap[mod.get_ptr()] = VarRef(stmt_last, readname);
302  }
303  else if (op==AstInterface::BOP_ARROW_ACCESS) {
304  std::string lhs = Get_VarName(fa, p), field = Local_GetFieldName(fa, p2);
305  fields.push_front(field);
306  Stmt stmt_last = deref_x_eq_y(lhs,fields,readname);
307  stmts_pushback(stmts,stmt_last);
308  namemap[p.get_ptr()] = VarRef(stmt_last,lhs);
309  namemap[mod.get_ptr()] = VarRef(stmt_last, readname);
310  }
311  else {
312  std::cerr << "can't handle " << AstInterface::AstToString(mod) << "\n";
313  ROSE_ABORT(); // other operations? to be handled later
314  }
315  }
316  else if (fa.IsFunctionCall(mod)) {
317  std::string lhs = Get_VarName(fa, mod);
318  Stmt stmt_last = deref_x_eq_y(lhs, fields, readname);
319  stmts_pushback(stmts,stmt_last);
320  namemap[mod.get_ptr()] = VarRef(stmt_last,lhs);
321  }
322  else {
323  std::cerr << "cannot process " << AstInterface::AstToString(mod) << "\n";
324  ROSE_ABORT(); // other operations? to be handled later
325  }
326  }
