82 SRecord
83 SRecord::parseMotorola(const std::string &input) {
84  using namespace StringUtility;
85 
86  // Header: 'S' + type
87  if (input.size() < 1+1+2) // 'S' + type + payloadsize
88  return SRecord().error("short input; Motorola S-Records must be at least 4 characters");
89  if (input[0]!='S')
90  return SRecord().error("not a Motorola S-Record since the line doesn't start with 'S'");
91  SRecord srec;
92  switch (unsigned t = parseBigEndianInteger(input, 1, 1).orElse(256)) {
93  case 0: srec.type_ = SREC_M_HEADER; break;
94  case 1: srec.type_ = SREC_M_DATA16; break;
95  case 2: srec.type_ = SREC_M_DATA24; break;
96  case 3: srec.type_ = SREC_M_DATA32; break;
97  case 4: srec.type_ = SREC_M_RESERVED; break;
98  case 5: srec.type_ = SREC_M_COUNT16; break;
99  case 6: srec.type_ = SREC_M_COUNT24; break;
100  case 7: srec.type_ = SREC_M_START32; break;
101  case 8: srec.type_ = SREC_M_START24; break;
102  case 9: srec.type_ = SREC_M_START16; break;
103  case 256:
104  return srec.error("type syntax error; expected a hexadecimal character");
105  default:
106  return srec.error("invalid record type " + numberToString(t));
107  }
108 
109  // Payload size: two hexadecimal characters
110  size_t payloadNBytes = 0;
111  if (!parseBigEndianInteger(input, 2, 2).assignTo(payloadNBytes))
112  return srec.error("payload size syntax error; expected two hexadecimal characters");
113  if (payloadNBytes < addressNBytes(srec.type()) + 1) // sizeof(address) + sizeof(checksum)
114  return srec.error("invalid record size; payload is too small for record type");
115  if (input.size() < 2 + 2 + 2*payloadNBytes) // sizeof(header)+sizeof(payloadsize)+sizeof(payload)
116  return srec.error("short input; line must contain at least " + plural(2+2+2*payloadNBytes, "characters"));
117 
118  // Address
119  if (!parseBigEndianInteger(input, 4, 2*addressNBytes(srec.type())).assignTo(srec.addr_))
120  return srec.error("address syntax error; expected four hexadecimal characters");
121  srec.addr_ = BitOps::signExtend(srec.addr_, 8*addressNBytes(srec.type()));
122 
123  // Data
124  ASSERT_require(payloadNBytes >= addressNBytes(srec.type()) + 1 /*checksum*/);
125  size_t dataNBytes = payloadNBytes - addressNBytes(srec.type()) - 1 /*checksum*/;
126  size_t at = 1 + 1 + 2 + 2*addressNBytes(srec.type());
127  for (size_t i = 0; i < dataNBytes; ++i, at += 2) {
128  if (!isxdigit(input[at]) || !isxdigit(input[at+1]))
129  return srec.error("data syntax error; data must contain only hexadecimal characters");
130  uint8_t byte = parseBigEndianInteger(input, at, 2).get();
131  srec.data_.push_back(byte);
132  }
133 
134  // Checksum
135  uint8_t parsedChecksum = 0;
136  if (!parseBigEndianInteger(input, 2 + 2 + 2*payloadNBytes - 2, 2).assignTo(parsedChecksum))
137  return srec.error("checksum syntax error; expected two hexadecimal characters");
138  uint8_t computedChecksum = checksum(SREC_MOTOROLA, input, 2, 2 + 2*payloadNBytes - 2); // sum address and data but not checksum
139  if (computedChecksum != parsedChecksum) {
140  return srec.error("invalid checksum; parsed " + numberToString(parsedChecksum) +
141  ", computed " + numberToString(computedChecksum));
142  }
143 
144  // Trailing garbage?
145  at = 2 + 2 + 2*payloadNBytes;
146  for (size_t eof = at; at < input.size(); ++at) {
147  if (!isspace(input[at]))
148  return srec.error("trailing garbage: \"" + cEscape(input.substr(eof)) + "\"");
149  }
150 
151 
152  return srec;
153 }

157 SRecord::parseIntel(const std::string &input) {
158  using namespace StringUtility;
159  static const size_t headerNChars = 1 + 2 + 4 + 2; // ':' + payload size + address + record type
160 
161  if (input.size() < headerNChars)
162  return SRecord().error("short input; Intel S-Records must be at least " + plural(headerNChars, "characters"));
163  if (input[0] != ':')
164  return SRecord().error("not an Intel S-Record ince the line doesn't start with ':'");
165 
166  // Payload size
167  size_t payloadNBytes = 0;
168  if (!parseBigEndianInteger(input, 1, 2).assignTo(payloadNBytes))
169  return SRecord().error("size syntax error; expected two hexadecimal characters");
170  size_t recordNChars = headerNChars + 2*payloadNBytes + 2;
171  if (input.size() < recordNChars)
172  return SRecord().error("short input; record needs at least " + plural(recordNChars, "characters"));
173 
174  // Address
175  SRecord srec;
176  if (!parseBigEndianInteger(input, 3, 4).assignTo(srec.addr_))
177  return srec.error("address syntax error; expected four hexadecimal characters");
178 
179  // Record type
180  switch (unsigned t = parseBigEndianInteger(input, 7, 2).orElse(65536)) {
181  case 0:
182  srec.type_ = SREC_I_DATA;
183  break;
184  case 1:
185  srec.type_ = SREC_I_END;
186  if (payloadNBytes != 0)
187  return srec.error("Intel end record should have a zero payload size");
188  if (srec.address() != 0)
189  return srec.error("Intel end record should have a zero address");
190  break;
191  case 2:
192  srec.type_ = SREC_I_EXTENDED_SA;
193  if (payloadNBytes != 2)
194  return srec.error("Intel extended segment address record should have a two-byte payload");
195  if (srec.address() != 0)
196  return srec.error("Intel extended segment address record should have a zero address");
197  break;
198  case 3:
199  srec.type_ = SREC_I_START_SA;
200  if (payloadNBytes != 2)
201  return srec.error("Intel start segment address record should have a two-byte payload");
202  if (srec.address() != 0)
203  return srec.error("Intel start segment address record should have a zero address");
204  break;
205  case 4:
206  srec.type_ = SREC_I_EXTENDED_LA;
207  if (payloadNBytes != 2)
208  return srec.error("Intel extended linear address record should have a two-byte payload");
209  if (srec.address() != 0)
210  return srec.error("Intel extended linear address record should have a zero address");
211  break;
212  case 5:
213  srec.type_ = SREC_I_START_LA;
214  if (payloadNBytes != 4)
215  return srec.error("Intel start linear address record should have a four-byte payload");
216  if (srec.address() != 0)
217  return srec.error("Intel start linear address record should have a zero address");
218  break;
219  case 65536:
220  return srec.error("type syntax error; expected two hexadecimal characters");
221  default:
222  return srec.error("invalid record type " + numberToString(t));
223  }
224 
225  // Payload
226  for (size_t i=0; i < payloadNBytes; ++i) {
227  uint8_t byte = 0;
228  if (!parseBigEndianInteger(input, headerNChars + 2*i, 2).assignTo(byte)) {
229  return srec.error("payload syntax error; payload must contain only hexadecimal characters");
230  } else {
231  srec.data_.push_back(byte);
232  }
233  }
234 
235  // Checksum
236  uint8_t parsedChecksum = 0;
237  if (!parseBigEndianInteger(input, recordNChars-2, 2).assignTo(parsedChecksum))
238  return srec.error("checksum syntax error; expected two hexadecimal characters");
239  uint8_t computedChecksum = checksum(SREC_INTEL, input, 1, recordNChars-3);
240  if (computedChecksum != parsedChecksum)
241  return srec.error("invalid checksum; parsed " + numberToString(parsedChecksum) +
242  ", computed " + numberToString(computedChecksum));
243 
244  // Trailing garbage?
245  for (size_t i = recordNChars; i<input.size(); ++i) {
246  if (!isspace(input[i]))
247  return srec.error("trailing garbage: \"" + cEscape(input.substr(recordNChars)) + "\"");
248  }
249 
250  return srec;
251 }

280  std::vector<SRecord> srecs;
281 
282  // Parse the input
283  RunLengthEncoding rle;
284  size_t nMotorola = 0, nIntel = 0;
285  while (input.good()) {
286  std::string line = rose_getline(input);
287  if (line.empty())
288  break;
289 
290  // Count syntax families
291  SRecord srec = SRecord::parse(line);
292  switch (srec.syntax()) {
293  case SREC_MOTOROLA: ++nMotorola; break;
294  case SREC_INTEL: ++nIntel; break;
295  default: break;
296  }
297 
298  // Check data record counts
299  if (srec.error().empty()) {
300  Sawyer::Optional<size_t> n;
301  switch (srec.type()) {
302  case SREC_M_COUNT16:
303  case SREC_M_COUNT24:
304  n = srec.address();
305  break;
306  default:
307  break;
308  }
309  if (n && *n != rle.nDataRecords()) {
310  srec.error("wrong data record count; record says " + StringUtility::numberToString(*n) +
311  " but expected " + StringUtility::numberToString(rle.nDataRecords()));
312  }
313  }
314 
315  // Save record
316  srecs.push_back(srec);
317  rle.insert(srec);
318  }
319 
320  // Input should consist of only one syntax style: either Motorola or Intel syntax
321  if (nMotorola > 0 && nIntel > 0) {
322  Syntax expected = nMotorola >= nIntel ? SREC_MOTOROLA : SREC_INTEL;
323  for (SRecord &srec: srecs) {
324  if (srec.syntax() != expected)
325  srec.error("wrong syntax family for S-Record collection");
326  }
327  }
328

431 AddressIntervalSet
432 SRecord::createSegments(const std::vector<SRecord> &srecs, const MemoryMap::Ptr &map, Address alignment_, unsigned accessPerms,
433  const std::string &name, MemoryMap::Clobber clobber) {
434  ASSERT_not_null(map);
435  const Address alignment = std::max(alignment_, (Address)1);
436 
437  // Given all srecord addresses, extend them so they're aligned, and merge all overlapping extensions together.
438  const AddressIntervalSet srecAddrs = dataAddresses(srecs);
439  AddressIntervalSet alignedAddresses;
440  for (const AddressInterval &interval: srecAddrs.intervals()) {
441  const Address begin = alignDown(interval.least(), alignment);
442  const Address end = alignUp(interval.greatest() + 1, alignment);
443  ASSERT_require(begin < end);
444  alignedAddresses |= AddressInterval::hull(begin, end-1);
445  }
446 
447  // Then create a temporary map that contains the minimum number of segments
448  auto tmp = MemoryMap::instance();
449  for (const AddressInterval &interval: alignedAddresses.intervals())
450  tmp->insert(interval, MemoryMap::Segment::anonymousInstance(interval.size(), accessPerms, name));
451 
452  // Finally, link all the new segments into the specified map
453  return map->linkTo(tmp, AddressInterval::whole(), clobber);
454 }

484 std::vector<SRecord>
485 SRecord::create(const MemoryMap::Ptr &map, Syntax syntax, size_t bytesPerRecord, size_t preferredAddrSize) {
486  std::vector<SRecord> retval;
487  uint8_t buffer[255];
488  ASSERT_require(bytesPerRecord > 0);
489  ASSERT_require(bytesPerRecord <= sizeof buffer);
490  Type dataRecordType = SREC_UNKNOWN_TYPE;
491  switch (syntax) {
492  case SREC_MOTOROLA: {
493  if (preferredAddrSize <= 2) {
494  preferredAddrSize = 2;
495  dataRecordType = SREC_M_DATA16;
496  } else if (preferredAddrSize <= 3) {
497  preferredAddrSize = 3;
498  dataRecordType = SREC_M_DATA24;
499  } else {
500  preferredAddrSize = 4;
501  dataRecordType = SREC_M_DATA32;
502  }
503  size_t maxBytesPerRecord = 255 - addressNBytes(dataRecordType);
504  bytesPerRecord = std::min(bytesPerRecord, maxBytesPerRecord);
505  break;
506  }
507  case SREC_INTEL: {
508  preferredAddrSize = 2;
509  dataRecordType = SREC_I_DATA;
510  bytesPerRecord = std::min(bytesPerRecord, size_t(255));
511  break;
512  }
513  default:
514  ASSERT_not_reachable("invalid syntax family");
515  }
516 
517  Address va = 0;
518  RunLengthEncoding rle;
519  while (map->atOrAfter(va).next().assignTo(va)) {
520  size_t nRead = map->at(va).limit(bytesPerRecord).read(buffer).size();
521  ASSERT_require(nRead > 0);
522 
523  switch (syntax) {
524  case SREC_MOTOROLA:
525  retval.push_back(SRecord(dataRecordType, va, buffer, nRead));
526  rle.insert(retval.back());
527  break;
528 
529  case SREC_INTEL: {
530  ASSERT_require(va >= rle.baseDataAddress()); // because va is increasing
531  Address offset = va - rle.baseDataAddress();
532  if (0 == rle.nDataRecords()) {
533  static uint8_t zeros[2];
534  retval.push_back(SRecord(SREC_I_EXTENDED_LA, 0, zeros, 2)); // good form to initialize the linear addresses
535  rle.insert(retval.back());
536  } else if (offset > 0xffff) {
537  uint8_t linear[2];
538  linear[0] = (va >> 24) & 0xff;
539  linear[1] = (va >> 16) & 0xff;
540  retval.push_back(SRecord(SREC_I_EXTENDED_LA, 0, linear, 2));
541  rle.insert(retval.back());
542  offset = va & 0xffff;
543  }
544  retval.push_back(SRecord(dataRecordType, offset, buffer, nRead));
545  rle.insert(retval.back());
546  break;
547  }
548 
549  default:
550  ASSERT_not_reachable("invalid syntax family");
551  }
552 
553  // Increment va safely
554  va += nRead - 1;
555  if (va == map->hull().greatest())
556  break; // avoid possible overflow
557  ++va;
558  }
559 
560  return retval;
561 }

563 std::string
564 SRecord::toString() const {
565  std::string s;
566  size_t checksumOffset = 0;
567  switch (syntax()) {
568  case SREC_MOTOROLA: {
569  switch (type_) {
570  case SREC_M_HEADER: s += "S0"; break;
571  case SREC_M_DATA16: s += "S1"; break;
572  case SREC_M_DATA24: s += "S2"; break;
573  case SREC_M_DATA32: s += "S3"; break;
574  case SREC_M_RESERVED: s += "S4"; break;
575  case SREC_M_COUNT16: s += "S5"; break;
576  case SREC_M_COUNT24: s += "S6"; break;
577  case SREC_M_START32: s += "S7"; break;
578  case SREC_M_START24: s += "S8"; break;
579  case SREC_M_START16: s += "S9"; break;
580  default:
581  ASSERT_not_reachable("invalid Motorola S-Record type " + StringUtility::numberToString(type_));
582  }
583 
584  size_t size = addressNBytes(type_) + data_.size() + 1 /*checksum*/;
585  if (size > 255)
586  throw Exception("S-Record size is too large");
587  s += (boost::format("%|02X|") % size).str();
588 
589  const Address addrMask = BitOps::lowMask<Address>(8 * addressNBytes(type_));
590  if (BitOps::signExtend(addr_ & addrMask, 8*addressNBytes(type_)) != addr_) {
591  throw Exception("S-Record address " + StringUtility::addrToString(addr_) +
592  " needs more than " + StringUtility::numberToString(addressNBytes(type_)) + " bytes");
593  }
594 
595  std::string addrFmt = "%|0" + StringUtility::numberToString(2*addressNBytes(type_)) + "x|";
596  s += (boost::format(addrFmt) % (addr_ & addrMask)).str();
597  checksumOffset = 2; // skip 'S' and the type
598  break;
599  }
600 
601  case SREC_INTEL: {
602  if (data_.size() > 255)
603  throw Exception("S-Record size is too large");
604  if (address() > 0xffff)
605  throw Exception("S-Record address is too large");
606  s = (boost::format(":%|02X|%|04X|") % data_.size() % (unsigned)address()).str();
