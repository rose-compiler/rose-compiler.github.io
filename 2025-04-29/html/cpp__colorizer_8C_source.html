139 void CppColorizer::loadDynamicKeywords()
140 {
141  return;
142 
143  QString strFileNameTag( "name" );
144  QString strClassNameTag( "name" );
145 
146  m_dynamicKeywords.clear();
147 
148  QString hlFileDir;// = KGlobal::dirs()->findResourceDir( "data", "highlighting.xml" );
149 
150  hlFileDir += "qeditorpart/highlight/";
151 
152  //kdDebug(9032) << "Highlighting Dir: " << hlFileDir << endl;
153 
154  if( hlFileDir.isNull() )
155  return;
156 
157  QDomDocument hlFile( "hlfile" ), curDoc ( "classlist" );
158  QFile hlRawFile( hlFileDir + "highlighting.xml" );
159  int keywordIndex = 0;
160  if( !hlRawFile.open( IO_ReadOnly ) )
161  return;
162  if( !hlFile.setContent( &hlRawFile ) ) {
163  hlRawFile.close();
164  return;
165  }
166  hlRawFile.close();
167 
168  QDomElement e = hlFile.documentElement();
169  QDomNode n = e.firstChild();
170  while( !n.isNull() ) {
171  e = n.toElement();
172  if( !e.isNull() ) {
173 
174  // kdDebug(9032) << "Loading classes-file: " << (hlFileDir + e.attribute( strFileNameTag )) << endl;
175 
176  QFile clsRawFile( hlFileDir + e.attribute( strFileNameTag ) );
177  if( clsRawFile.open( IO_ReadOnly ) && curDoc.setContent( &clsRawFile ) ) {
178 
179  QDomElement e = curDoc.documentElement();
180  QDomNode n = e.firstChild();
181  while( !n.isNull() ) {
182  e = n.toElement();
183  if( !e.isNull()) {
184  // kdDebug(9032) << "Adding dynamic keyword: '" << e.attribute( strClassNameTag ) << "'" << endl;
185  m_dynamicKeywords.insert( e.attribute( strClassNameTag ), keywordIndex++ );
186  }
187  n = n.nextSibling();
188  }
189 
190  }
191  clsRawFile.close();
192  }
193  n = n.nextSibling();
194  }

197 int CppColorizer::computeLevel( Q3TextParagraph* parag, int startLevel )
198 {
199  int level = startLevel;
200 
201  ParagData* data = (ParagData*) parag->extraData();
202  if( !data ){
203  return startLevel;
204  }
205 
206  data->setBlockStart( false );
207 
208  Q3ValueList<Symbol> symbols = data->symbolList();
209  Q3ValueList<Symbol>::Iterator it = symbols.begin();
210  while( it != symbols.end() ){
211  Symbol sym = *it++;
212  if( sym.ch() == '{' ){
213  ++level;
214  } else if( sym.ch() == '}' ){
215  --level;
216  }
217  }
218 
219  if( level > startLevel ){
220  data->setBlockStart( true );
221  }
222 
223  return level;
224 }
