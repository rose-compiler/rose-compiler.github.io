97 SAWYER_EXPORT std::string
98 Cursor::substr(const Location &limit1, const Location &limit2, const std::string &separator) const {
99  std::string retval;
100  Location begin = limit1, end = limit2;
101  if (end < begin)
102  std::swap(begin, end);
103  while (begin < end) {
104  if (begin.idx < end.idx) {
105  retval += rest(begin);
106  if (end.offset)
107  retval += separator;
108  ++begin.idx;
109  begin.offset = 0;
110  } else {
111  retval += rest(begin).substr(0, end.offset-begin.offset);
112  break;
113  }
114  }
115  return retval;
116 }

181 // Only called by match(). If the subclass doesn't override this, then we try calling the C-string version instead.
182 SAWYER_EXPORT ParsedValue
183 ValueParser::operator()(Cursor &cursor) {
184  std::string str = cursor.rest();
185  const char *s = str.c_str();
186  const char *rest = s;
187  try {
188  ParsedValue retval = (*this)(s, &rest, cursor.location());
189  if (NULL!=rest) {
190  ASSERT_require(rest>=s && rest<= s+strlen(s));
191  cursor.consumeChars(rest-s);
192  }
193  return retval;
194  } catch (const std::runtime_error&) {
195  // We must update the cursor location even if an exception is thrown because it indicates that a value
196  // was syntactically correct, but not semantically correct. E.g., a mathematical integer that could not be
197  // stored as an unsigned type.
198  if (NULL!=rest) {
199  ASSERT_require(rest>=s && rest<= s+strlen(s));
200  cursor.consumeChars(rest-s);
201  }
202  throw;
203  }
204 }

243 StringSetParser::operator()(Cursor &cursor) {
244  Location locStart = cursor.location();
245  std::string input = cursor.rest();
246  size_t bestMatchIdx = (size_t)(-1), bestMatchLen = 0;
247  for (size_t i=0; i<strings_.size(); ++i) {
248  if (boost::starts_with(input, strings_[i]) && ((size_t)(-1)==bestMatchIdx || strings_[i].size()>bestMatchLen)) {

254  throw std::runtime_error("specific word expected");
255  cursor.consumeChars(bestMatchLen);
256  return ParsedValue(strings_[bestMatchIdx], locStart, strings_[bestMatchIdx], valueSaver());

268 SAWYER_EXPORT ParsedValue
269 ListParser::operator()(Cursor &cursor) {
270  ASSERT_forbid(elements_.empty());
271  Location startLoc = cursor.location();
272  ExcursionGuard guard(cursor); // parsing the list should be all or nothing
273  ValueList values;
274  std::string sep = "";
275 
276  for (size_t i=0; i<maxLength_; ++i) {
277  const ParserSep &ps = elements_[std::min(i, elements_.size()-1)];
278 
279  // Advance over the value separator
280  if (0!=i) {
281  if (cursor.atArgBegin() || cursor.atEnd())
282  break; // we've advanced over the entire program argument
283  std::string str = cursor.rest();
284  const char *s = str.c_str();
285  boost::regex re("\\A(" + sep + ")");
286  boost::cmatch matched;
287  if (!regex_search(s, matched, re))
288  break;
289  cursor.consumeChars(matched.str().size());
290  }
291  sep = ps.second;
292 
293  // Find the next value separator so we can prevent from parsing through it
294  size_t endOfValue = cursor.rest().size();
295  {
296  boost::regex re(sep);
297  boost::cmatch matched;
298  std::string str = cursor.rest();
299  const char *s = str.c_str();
300  if (regex_search(s, matched, re))
301  endOfValue = matched.position();
302  }
303 
304  // Parse the value, stopping before the next separator
305  Cursor valueCursor(cursor.rest().substr(0, endOfValue));
306  ParsedValue value = ps.first->match(valueCursor);
307  value.valueLocation(cursor.location());
308  cursor.consumeChars(valueCursor.linearDistance());
309  values.push_back(value);
310  }
311 
312  if (values.size()<minLength_ || values.size()>maxLength_) {
313  std::ostringstream ss;
314  if (minLength_ == maxLength_) {
315  ss <<"list with " <<maxLength_ <<" element" <<(1==maxLength_?"":"s") <<" expected (got " <<values.size() <<")";
316  throw std::runtime_error(ss.str());
317  } else if (minLength_+1 == maxLength_) {
318  ss <<"list with " <<minLength_ <<" or " <<maxLength_ <<" element" <<(1==maxLength_?"":"s") <<" expected"
319  <<" (got " <<values.size() <<")";
320  throw std::runtime_error(ss.str());
321  } else {
322  std::ostringstream ss;
323  ss <<"list with " <<minLength_ <<" to " <<maxLength_ <<" elements expected (got " <<values.size() <<")";
324  throw std::runtime_error(ss.str());
325  }
326  }
327 
328  guard.cancel();
329  return ParsedValue(values, startLoc, cursor.substr(startLoc), valueSaver());
330 }

970 // cursor is initially at the first character of the first switch argument
971 SAWYER_EXPORT size_t
972 Switch::matchArguments(const std::string &switchString, const Location &endOfSwitch, Cursor &cursor /*in,out*/,
973  const ParsingProperties &props, ParsedValues &result /*out*/, bool finalAlignment) const {
974  ASSERT_forbid(arguments_.empty());
975 
976  ParsedValues parsedValues;
977  ExcursionGuard guard(cursor);
978  size_t nValuesParsed = 0;
979  size_t switchIdx = cursor.location().idx; // which program argument holds the switch name
980  if (switchIdx>0 && (cursor.atArgBegin() || cursor.atEnd()))
981  --switchIdx;
982 
983  // Parse arguments, or use defaults if allowed.
984  const SwitchArgument *lastParsedArgument = NULL;
985  ParsedValue lastParsedValue;
986  for (size_t argno=0; argno<arguments_.size(); ++argno) {
987  const SwitchArgument &sa = arguments_[argno];
988  Location valueLocation = cursor.location();
989  try {
990  ParsedValue value = sa.parser()->match(cursor);
991  parsedValues.push_back(value);
992  if (cursor.atArgEnd()) {
993  cursor.consumeArg();
994  } else if (finalAlignment || argno>0) {
995  throw extraTextAfterArgument(cursor, value);
996  }
997  ++nValuesParsed;
998  lastParsedArgument = &sa;
999  lastParsedValue = value;
1000  } catch (const std::runtime_error &e) {
1001  if (sa.isRequired()) {
1002  throw cursor.location()==valueLocation ?
1003  missingArgument(switchString, cursor, sa, e.what()) :
1004  malformedArgument(switchString, cursor, sa, e.what());
1005  }
1006  cursor.location(valueLocation);
1007  parsedValues.push_back(sa.defaultValue());
1008  }
1009  }
1010 
1011  // Regardless of whether we parsed arguments, we need to check final alignment. However, if the last value was a parsed
1012  // value from the loop above (as opposed to a default value) then we've already checked this and advanced to the next
1013  // program argument.
1014  bool lastArgWasParsed = !parsedValues.empty() && NOWHERE!=parsedValues.back().valueLocation();
1015  if (finalAlignment && !lastArgWasParsed && !cursor.atArgBegin() && !cursor.atEnd()) {
1016  if (lastParsedArgument) {
1017  throw extraTextAfterArgument(cursor, lastParsedValue);
1018  } else {
1019  throw extraTextAfterSwitch(switchString, endOfSwitch, cursor, props, parsedValues);
1020  }
1021  }
1022 
1023  explode(parsedValues);
1024  guard.cancel();
1025  result.insert(result.end(), parsedValues.begin(), parsedValues.end());
1026  return nValuesParsed;
1027 }

1029 // cursor is initially immediately after the switch name
1030 SAWYER_EXPORT void
1031 Switch::matchLongArguments(const std::string &switchString, Cursor &cursor /*in,out*/, const ParsingProperties &props,
1032  ParsedValues &result /*out*/) const {
1033  ExcursionGuard guard(cursor);
1034 
1035  // If the switch has no declared arguments use its intrinsic value.
1036  if (arguments_.empty()) {
1037  if (!cursor.atArgEnd())
1038  throw extraTextAfterSwitch(switchString, cursor.location(), cursor, props, result);
1039  result.push_back(intrinsicValue_);
1040  cursor.consumeArg();
1041  guard.cancel();
1042  return;
1043  }
1044 
1045  // Try to match the name/value separator. Advance the cursor to the first character of the first value.
1046  Location endOfSwitch = cursor.location();
1047  bool didMatchSeparator = false;
1048  if (cursor.atArgEnd()) {
1049  if (matchAnyString(props.valueSeparators, " ")) {
1050  didMatchSeparator = true;
1051  cursor.consumeArg();
1052  }
1053  } else {
1054  std::string s = cursor.rest();
1055  BOOST_FOREACH (const std::string &sep, props.valueSeparators) {
1056  if (boost::starts_with(s, sep)) {
1057  cursor.consumeChars(sep.size());
1058  didMatchSeparator = true;
1059  break;
1060  }
1061  }
1062  }
1063  if ((!didMatchSeparator || cursor.atEnd()) && nRequiredArguments()>0)
1064  throw noSeparator(switchString, cursor, props);
1065 
1066  // Parse the arguments for this switch now that we've consumed the prefix, switch name, and argument separators.
1067  matchArguments(switchString, endOfSwitch, cursor, props, result /*out*/, true /*finalAlignment*/);
1068  guard.cancel();
1069 }

1620  }
1621  }
1622

1624  bool inserted = false;
1625  BOOST_FOREACH (std::string prefix, inclusionPrefixes_) {
1626  const std::string &arg = cursor.arg();

1630  cursor.replace(args);

1632  }

1634  if (inserted)

1645  return result;
1646  }
1647  }

1660  }
1661  }
1662  }

1689 SAWYER_EXPORT bool
1690 Parser::parseOneSwitch(Cursor &cursor, const NamedSwitches &ambiguities, ParserResult &result) {
1691  ASSERT_require(cursor.atArgBegin());
1692  Optional<std::runtime_error> saved_error;
1693 
1694  // Single long switch
1695  ParsedValues values;
1696  if (const Switch *sw = parseLongSwitch(cursor, values, ambiguities, saved_error /*out*/)) {
1697  ASSERT_forbid(values.empty());
1698  ASSERT_require(cursor.atArgBegin() || cursor.atEnd());
1699  if (sw->skipping() != SKIP_STRONG)
1700  result.insertValuesForSwitch(values, this, sw);
1701  if (sw->skipping() != SKIP_NEVER) {
1702  result.skip(values[0].switchLocation());
1703  BOOST_FOREACH (const ParsedValue &pval, values)
1704  result.skip(pval.valueLocation());
1705  }
1706  return true;
1707  }
1708 
1709  if (!shortMayNestle_) {
1710  // Single short switch
1711  if (const Switch *sw = parseShortSwitch(cursor, values, ambiguities, saved_error, shortMayNestle_)) {
1712  ASSERT_forbid(values.empty());
1713  ASSERT_require(cursor.atArgBegin() || cursor.atEnd());
1714  if (sw->skipping() != SKIP_STRONG)
1715  result.insertValuesForSwitch(values, this, sw);
1716  if (sw->skipping() != SKIP_NEVER) {
1717  result.skip(values[0].switchLocation());
1718  BOOST_FOREACH (const ParsedValue &pval, values)
1719  result.skip(pval.valueLocation());
1720  }
1721  return true;
1722  }
1723  } else {
1724  // Or multiple short switches. If short switches are nestled, then the result is affected only if all the nesltled
1725  // switches can be parsed.
1726  typedef std::pair<const Switch*, ParsedValues> SwitchValues;
1727  std::list<SwitchValues> valuesBySwitch; // values for each nestled switch that was parsed
1728  ExcursionGuard guard(cursor);
1729  while (guard.startingLocation().idx == cursor.location().idx) {
1730  if (const Switch *sw = parseShortSwitch(cursor, values, ambiguities, saved_error, shortMayNestle_)) {
1731  ASSERT_forbid(values.empty());
1732  valuesBySwitch.push_back(SwitchValues(sw, values));
1733  values.clear();
1734  } else {
1735  break;
1736  }
1737  }
1738  if (!valuesBySwitch.empty() && (cursor.atArgBegin() || cursor.atEnd())) {
1739  BOOST_FOREACH (SwitchValues &svpair, valuesBySwitch) {
1740  const Switch *sw = svpair.first;
1741  const ParsedValues &values = svpair.second;
1742  if (sw->skipping() != SKIP_STRONG)
1743  result.insertValuesForSwitch(values, this, sw);
1744  if (sw->skipping() != SKIP_NEVER) {
1745  result.skip(values[0].switchLocation());
1746  BOOST_FOREACH (const ParsedValue &pval, values)
1747  result.skip(pval.valueLocation());
1748  }
1749  }
1750 
1751  guard.cancel();
1752  return true;
1753  }
1754  }
1755 
1756  // Throw or return zero?
1757  if (saved_error)
1758  throw *saved_error; // found at least one switch but couldn't ever parse arguments
1759  if (apparentSwitch(cursor))
1760  throw std::runtime_error("unrecognized switch: " + cursor.arg());
1761  return false;
1762 }

1950  std::vector<std::string> retval;
1951  boost::trim(line);
1952  size_t nchars = line.size();

1956  std::string word;
1957

1964  inQuote = ch;
