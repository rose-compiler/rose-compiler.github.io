172 // Statements are classified into one of the following types:
173 enum IRStmtType
174 {
175  SIMPLE, // Anything not covered below.
176  COMPOUND, // A block of statements.
177  LOOP, // Any type of top-tested, structured loop.
178  END_TESTED_LOOP, // Any type of end-tested, structured loop.
179  STRUCT_TWOWAY_CONDITIONAL, // Structured if-then-else.
180  STRUCT_MULTIWAY_CONDITIONAL, // Structured switch statement.
181  USTRUCT_TWOWAY_CONDITIONAL_T, // Unstructured branch (on true).
182  USTRUCT_TWOWAY_CONDITIONAL_F, // Unstructured branch (on false).
183  USTRUCT_MULTIWAY_CONDITIONAL, // Unstructured multiway branch
184  // (e.g., computed goto in Fortran or
185  // jump tables in low-level/assembly
186  // languages).
187  RETURN, // Return statement.
188  BREAK, // Break statement.
189  LOOP_CONTINUE, // Loop continuation statement.
190  ALTERNATE_PROC_ENTRY, // Alternate entry point (e.g., Fortran)
191  UNCONDITIONAL_JUMP, // GOTO in HLL, or unconditional direct
192  // jump in low-level/assembly languages.
193  UNCONDITIONAL_JUMP_I, // Assigned GOTO in HLL, or unconditional
194  // indirect jump in low-level/assembly
195  // languages.
196  NONE
197 };

264  // Given a loop statement, return the loop header statement. This
265  // would be the initialization statement in a C 'for' loop, for example.

271  // Given a loop statement, return:
272  //
273  // True: If the number of loop iterations is defined
274  // at loop entry (i.e. Fortran semantics). This causes the CFG builder
275  // to add the loop statement representative to the header node so that
276  // definitions from inside the loop don't reach the condition and increment
277  // specifications in the loop statement.
278  //
279  // False: If the number of iterations is not defined at
280  // entry (i.e. C semantics), we add the loop statement to a node that
281  // is inside the loop in the CFG so definitions inside the loop will
282  // reach uses in the conditional test. For C style semantics, the
283  // increment itself may be a separate statement. if so, it will appear
284  // explicitly at the bottom of the loop.
285  virtual bool LoopIterationsDefinedAtEntry(StmtHandle h) = 0;

287  //--------------------------------------------------------
288  // Invariant: a two-way conditional or a multi-way conditional MUST provide
289  // provide either a target, or a target label
290  //--------------------------------------------------------
291 
292  //--------------------------------------------------------
293  // Structured two-way conditionals
294  //
295  // Note: An important pre-condition for structured conditionals is
296  // that chains of else-ifs must be represented as nested elses. For
297  // example, this Matlab statement:
298  // if (c1)
299  // s1;
300  // elseif (c2)
301  // s2;
302  // else
303  // s3;
304  // end;
305  //
306  // would need be represented by the underlying IR as:
307  // if (c1)
308  // s1;
309  // else
310  // if (c2)
311  // s2;
312  // else
313  // s3;
314  // end;
315  // end;
316  //--------------------------------------------------------
317 
318  // Given a structured two-way conditional statement, return an
319  // IRStmtIterator* for the "true" part (i.e., the statements under
320  // the "if" clause). The user must free the iterator's memory via
321  // delete.
322  virtual IRStmtIterator *TrueBody(StmtHandle h) = 0;
323 
324  // Given a structured two-way conditional statement, return an
325  // IRStmtIterator* for the "else" part (i.e., the statements under
326  // the "else" clause). The user must free the iterator's memory via
327  // delete.
328  virtual IRStmtIterator *ElseBody(StmtHandle h) = 0;
329 
330  //--------------------------------------------------------
331  // Structured multiway conditionals
332  //--------------------------------------------------------
333 
334  // Given a structured multi-way branch, return the number of cases.
335  // The count does not include the default/catchall case.
336  virtual int NumMultiCases(StmtHandle h) = 0;
337 
338  // Given a structured multi-way branch, return an IRStmtIterator* for
339  // the body corresponding to target 'bodyIndex'. The n targets are
340  // indexed [0..n-1]. The user must free the iterator's memory via delete.
341  virtual IRStmtIterator *MultiBody(StmtHandle h, int bodyIndex) = 0;
342 
343  // Given a structured multi-way branch, return true if the cases have
344  // implied break semantics. For example, this method would return false
345  // for C since one case will fall-through to the next if there is no
346  // explicit break statement. Matlab, on the other hand, implicitly exits
347  // the switch statement once a particular case has executed, so this
348  // method would return true.
349  virtual bool IsBreakImplied(StmtHandle multicond) = 0;
350 
351  // Given a structured multi-way branch, return an IRStmtIterator*
352  // for the body corresponding to default/catchall case. The user
353  // must free the iterator's memory via delete.
354  virtual IRStmtIterator * GetMultiCatchall (StmtHandle h) = 0;
355

359 
360  // Given an unstructured two-way branch, return the label of the

372 
373  // Given an unstructured multi-way branch, return the label of the target

375  virtual StmtLabel GetUMultiTargetLabel(StmtHandle h, int targetIndex) = 0;
376 
377  // Given an unstructured multi-way branch, return label of the target
378  // corresponding to the optional default/catchall case. Return 0
379  // if there is no default target.
380  virtual StmtLabel GetUMultiCatchallLabel(StmtHandle h) = 0;
