18 /* Constructor reads symbol table entries beginning at entry 'i'. We can't pass an array of COFFSymbolEntry_disk structs
19  * because the disk size is 18 bytes, which is not properly aligned according to the C standard. Therefore we pass the actual
20  * section and table index. The symbol occupies the specified table slot and st_num_aux_entries additional slots.
21  *
22  * See http://www.skyfree.org/linux/references/coff.pdf */

27  ASSERT_not_null(symtab);

32  static uint8_t zeroRecord[COFFSymbol_disk_size];
33  memset(zeroRecord, 0, sizeof zeroRecord);
34  COFFSymbol_disk disk;
35  symtab->readContentLocal(idx * COFFSymbol_disk_size, &disk, COFFSymbol_disk_size);
36  if (0 == memcmp(&disk, zeroRecord, COFFSymbol_disk_size))
37  throw FormatError("zero symbol record");

39  if (disk.st_zero == 0) {

54  p_st_section_num = Rose::BinaryAnalysis::ByteOrder::leToHost(disk.st_section_num);
55  p_st_type = Rose::BinaryAnalysis::ByteOrder::leToHost(disk.st_type);
56  p_st_storage_class = Rose::BinaryAnalysis::ByteOrder::leToHost(disk.st_storage_class);
57  p_st_num_aux_entries = Rose::BinaryAnalysis::ByteOrder::leToHost(disk.st_num_aux_entries);

65  <<" is not bound to any section (section " <<p_st_section_num <<")\n";
66  }

75  case 2: p_binding = SYM_GLOBAL; break; /*extern*/
76  case 3: p_binding = SYM_GLOBAL; break; /*static*/
77  case 4: p_binding = SYM_LOCAL; break; /*register*/
78  case 5: p_binding = SYM_GLOBAL; break; /*extern def*/
79  case 6: p_binding = SYM_LOCAL; break; /*label*/
80  case 7: p_binding = SYM_LOCAL; break; /*label(undef)*/
81  case 8: p_binding = SYM_LOCAL; break; /*struct member*/
82  case 9: p_binding = SYM_LOCAL; break; /*formal arg*/
83  case 10: p_binding = SYM_LOCAL; break; /*struct tag*/
84  case 11: p_binding = SYM_LOCAL; break; /*union member*/
85  case 12: p_binding = SYM_GLOBAL; break; /*union tag*/
86  case 13: p_binding = SYM_GLOBAL; break; /*typedef*/
87  case 14: p_binding = SYM_GLOBAL; break; /*static(undef)*/
88  case 15: p_binding = SYM_GLOBAL; break; /*enum tag*/
89  case 16: p_binding = SYM_LOCAL; break; /*enum member*/
90  case 17: p_binding = SYM_GLOBAL; break; /*register param*/
91  case 18: p_binding = SYM_LOCAL; break; /*bit field*/
92  case 100: p_binding = SYM_GLOBAL; break; /*block(bb or eb)*/
93  case 101: p_binding = SYM_GLOBAL; break; /*function*/
94  case 102: p_binding = SYM_LOCAL; break; /*struct end*/
95  case 103: p_binding = SYM_GLOBAL; break; /*file*/
96  case 104: p_binding = SYM_GLOBAL; break; /*section*/
97  case 105: p_binding = SYM_WEAK; break; /*weak extern*/
98  case 107: p_binding = SYM_LOCAL; break; /*CLR token*/
99  case 0xff: p_binding = SYM_GLOBAL; break; /*end of function*/
100  }
101  switch (p_st_type & 0xf0) {
102  case 0x00: p_type = SYM_NO_TYPE; break; /*none*/
103  case 0x10: p_type = SYM_DATA; break; /*ptr*/
104  case 0x20: p_type = SYM_FUNC; break; /*function*/
105  case 0x30: p_type = SYM_ARRAY; break; /*array*/
106  }

110  <<": p_st_name_offset=" <<p_st_name_offset
111  <<", p_st_name=\"" <<StringUtility::cEscape(p_st_name) <<"\""
112  <<", p_st_section_num=" <<p_st_section_num
113  <<", p_st_type=" <<p_st_type
114  <<", p_st_storage_class=" <<p_st_storage_class

119  <<", p_type=" <<p_type <<"\n";

123  if (p_st_num_aux_entries > 0) {

125 
126  if (2 /*external*/ == p_st_storage_class && get_type() == SYM_FUNC && p_st_section_num > 0) {
127  // Auxiliary record format 1: Function definitions
128  unsigned bf_idx = Rose::BinaryAnalysis::ByteOrder::leToHost(*(uint32_t*)&(p_auxiliaryData[0]));
129  unsigned size = Rose::BinaryAnalysis::ByteOrder::leToHost(*(uint32_t*)&(p_auxiliaryData[4]));
130  unsigned lnum_ptr = Rose::BinaryAnalysis::ByteOrder::leToHost(*(uint32_t*)&(p_auxiliaryData[8]));
131  unsigned next_fn_idx = Rose::BinaryAnalysis::ByteOrder::leToHost(*(uint32_t*)&(p_auxiliaryData[12]));
132  unsigned res1 = Rose::BinaryAnalysis::ByteOrder::leToHost(*(uint16_t*)&(p_auxiliaryData[16]));

135  <<", size=" <<size <<", lnum_ptr=" <<StringUtility::addrToString(lnum_ptr)
136  <<", next_fn_idx=" <<next_fn_idx <<", res1=" <<res1 <<"\n";
137 
138  } else if (p_st_storage_class == 101 /*function*/ && (p_st_name == ".bf" || p_st_name == ".ef")) {
139  // Auxiliary record format 2: .bf and .ef symbols
140  unsigned res1 = Rose::BinaryAnalysis::ByteOrder::leToHost(*(uint32_t*)&(p_auxiliaryData[0]));
141  unsigned lnum = Rose::BinaryAnalysis::ByteOrder::leToHost(*(uint16_t*)&(p_auxiliaryData[4])); /*line num within source file*/
142  unsigned res2 = Rose::BinaryAnalysis::ByteOrder::leToHost(*(uint16_t*)&(p_auxiliaryData[6]));
143  unsigned res3 = Rose::BinaryAnalysis::ByteOrder::leToHost(*(uint32_t*)&(p_auxiliaryData[8]));
144  unsigned next_bf = Rose::BinaryAnalysis::ByteOrder::leToHost(*(uint32_t*)&(p_auxiliaryData[12])); /*only for .bf; reserved in .ef*/
145  unsigned res4 = Rose::BinaryAnalysis::ByteOrder::leToHost(*(uint16_t*)&(p_auxiliaryData[16]));
146  SAWYER_MESG(debug) <<"COFF aux " <<escapeString(p_st_name) <<": res1=" <<res1 <<", lnum=" <<lnum
147  <<", res2=" <<res2 <<", res3=" <<res3 <<", next_bf=" <<next_bf <<", res4=" <<res4 <<"\n";
148 
149  } else if (p_st_storage_class == 2/*external*/ && p_st_section_num == 0/*undef*/ && get_value()==0) {
150  // Auxiliary record format 3: weak externals
151  unsigned sym2_idx = Rose::BinaryAnalysis::ByteOrder::leToHost(*(uint32_t*)&(p_auxiliaryData[0]));
152  unsigned flags = Rose::BinaryAnalysis::ByteOrder::leToHost(*(uint32_t*)&(p_auxiliaryData[4]));
153  unsigned res1 = Rose::BinaryAnalysis::ByteOrder::leToHost(*(uint32_t*)&(p_auxiliaryData[8]));
154  unsigned res2 = Rose::BinaryAnalysis::ByteOrder::leToHost(*(uint32_t*)&(p_auxiliaryData[12]));
155  unsigned res3 = Rose::BinaryAnalysis::ByteOrder::leToHost(*(uint16_t*)&(p_auxiliaryData[16]));
156  SAWYER_MESG(debug) <<"COFF aux weak " <<escapeString(p_st_name) <<": sym2_idx=" <<sym2_idx
157  <<", flags=" <<flags <<", res1=" <<res1 <<", res2=" <<res2 <<", res3=" <<res3 <<"\n";
158 
159  } else if (p_st_storage_class == 103/*file*/ && 0 == p_st_name.compare(".file")) {
160  // Auxiliary record format 4: files. The file name is stored in the aux data as either the name itself or an offset
161  // into the string table. Replace the fake ".file" with the real file name.
162  const COFFSymbol_disk *d = (const COFFSymbol_disk*) &(p_auxiliaryData[0]);
163  if (0 == d->st_zero) {

165  if (fname_offset < 4)

167  set_name(new SgAsmBasicString(strtab->readContentLocalString(fname_offset)));
168  SAWYER_MESG(debug) <<"COFF aux file: offset=" <<fname_offset
169  <<", name=\"" <<get_name()->get_string(true) <<"\"\n";
170 
171  } else {
172  /* Aux data contains a NUL-padded name; the NULs (if any) are not part of the name. */
173  ASSERT_require(p_st_num_aux_entries == 1);
174  char fname[COFFSymbol_disk_size+1];
175  memcpy(fname, &(p_auxiliaryData[0]), COFFSymbol_disk_size);
176  fname[COFFSymbol_disk_size] = '\0';
177  set_name(new SgAsmBasicString(fname));
178  SAWYER_MESG(debug) <<"COFF aux file: inline-name=\"" <<get_name()->get_string(true) <<"\"\n";
179  }

181 
182  } else if (p_st_storage_class == 3/*static*/ && NULL != fhdr->get_file()->get_sectionByName(p_st_name, '$')) {
183  // Auxiliary record format 5: Section definition.
184  unsigned size = Rose::BinaryAnalysis::ByteOrder::leToHost(*(uint32_t*)&(p_auxiliaryData[0])); /*same as section header SizeOfRawData */
185  unsigned nrel = Rose::BinaryAnalysis::ByteOrder::leToHost(*(uint16_t*)&(p_auxiliaryData[4])); /*number of relocations*/
186  unsigned nln_ents = Rose::BinaryAnalysis::ByteOrder::leToHost(*(uint16_t*)&(p_auxiliaryData[6])); /*number of line number entries */
187  unsigned cksum = Rose::BinaryAnalysis::ByteOrder::leToHost(*(uint32_t*)&(p_auxiliaryData[8]));
188  unsigned sect_id = Rose::BinaryAnalysis::ByteOrder::leToHost(*(uint16_t*)&(p_auxiliaryData[12])); /*1-base index into section table*/
189  unsigned comdat = p_auxiliaryData[14]; /*comdat selection number if section is a COMDAT section*/
190  unsigned res1 = p_auxiliaryData[15];
191  unsigned res2 = Rose::BinaryAnalysis::ByteOrder::leToHost(*(uint16_t*)&(p_auxiliaryData[16]));
192  set_size(size);
193  set_type(SYM_SECTION);
194  SAWYER_MESG(debug) <<"COFF aux section: size=" <<size <<", nrel=" <<nrel <<", nln_ents=" <<nln_ents
195  <<", cksum=" <<cksum <<", sect_id=" <<sect_id <<", comdat=" <<comdat
196  <<", res1=" <<res1 <<", res2=" <<res2 <<"\n";
197 
198  } else if (p_st_storage_class==3/*static*/ && (p_st_type & 0xf)==0/*null*/ &&
199  get_value()==0 && NULL!=fhdr->get_file()->get_sectionByName(p_st_name)) {
200  // Auxiliary record for common data (COMDAT) sections
201  // FIXME[Robb P Matzke 2017-05-17]: The record format isn't documented in the reference listed above.
202  if (debug) {
203  debug <<"COFF aux comdat " <<escapeString(p_st_name) <<": aux record ignored\n";
204  Rose::BinaryAnalysis::hexdump(debug,
205  (Rose::BinaryAnalysis::Address) symtab->get_offset()+(idx+1)*COFFSymbol_disk_size,
206  " ",
207  p_auxiliaryData);
208  }
209 
210  } else {
211  if (mlog[WARN]) {
212  mlog[WARN] <<"COFF aux unknown " <<escapeString(p_st_name)
213  <<": st_storage_class=" <<p_st_storage_class
214  <<", st_type=" <<p_st_type <<", st_section_num=" <<p_st_section_num <<"\n";
215  Rose::BinaryAnalysis::hexdump(mlog[WARN], symtab->get_offset()+(idx+1)*COFFSymbol_disk_size, " ",
216  p_auxiliaryData);
217  }
218  }

222 /* Encode a symbol back into disk format */
223 void *
224 SgAsmCoffSymbol::encode(COFFSymbol_disk *disk) const
225 {
226  if (0 == p_st_name_offset) {
227  /* Name is stored in entry */
228  memset(disk->st_name, 0, sizeof(disk->st_name));
229  ROSE_ASSERT(p_st_name.size() <= sizeof(disk->st_name));
230  memcpy(disk->st_name, p_st_name.c_str(), p_st_name.size());
231  } else {
232  /* Name is an offset into the string table */
233  disk->st_zero = 0;
234  Rose::BinaryAnalysis::ByteOrder::hostToLe(p_st_name_offset, &(disk->st_offset));
235  }
236 
237  // Rose::BinaryAnalysis::ByteOrder::hostToLe(get_value(), &(disk->st_value));
238  Rose::BinaryAnalysis::ByteOrder::hostToLe(p_value, &(disk->st_value));
239  Rose::BinaryAnalysis::ByteOrder::hostToLe(p_st_section_num, &(disk->st_section_num));
240  Rose::BinaryAnalysis::ByteOrder::hostToLe(p_st_type, &(disk->st_type));
241  Rose::BinaryAnalysis::ByteOrder::hostToLe(p_st_storage_class, &(disk->st_storage_class));
242  Rose::BinaryAnalysis::ByteOrder::hostToLe(p_st_num_aux_entries, &(disk->st_num_aux_entries));
243  return disk;
244 }

356 SgAsmCoffSymbolTable*
357 SgAsmCoffSymbolTable::parse()
358 {
359  /* Set the section size according to the number of entries indicated in the header. */
360  SgAsmPEFileHeader *fhdr = dynamic_cast<SgAsmPEFileHeader*>(get_header());
361  ROSE_ASSERT(fhdr!=NULL);
362  set_offset(fhdr->get_e_coff_symtab());
363  set_size(fhdr->get_e_coff_nsyms()*SgAsmCoffSymbol::COFFSymbol_disk_size);
364 
365  SgAsmGenericSection::parse();
366 
367  /* The string table immediately follows the symbols. The first four bytes of the string table are the size of the
368  * string table in little endian. */
369  Rose::BinaryAnalysis::Address strtab_offset = get_offset() + fhdr->get_e_coff_nsyms() * SgAsmCoffSymbol::COFFSymbol_disk_size;
370  p_strtab = new SgAsmGenericSection(fhdr->get_file(), fhdr);
371  p_strtab->set_offset(strtab_offset);
372  p_strtab->set_size(sizeof(uint32_t));
373  p_strtab->set_synthesized(true);
374  p_strtab->set_name(new SgAsmBasicString("COFF Symbol Strtab"));
375  p_strtab->set_purpose(SP_HEADER);
376  p_strtab->parse();
377 
378  uint32_t word;
379  p_strtab->readContent(0, &word, sizeof word);
380  Rose::BinaryAnalysis::Address strtab_size = Rose::BinaryAnalysis::ByteOrder::leToHost(word);
381  if (strtab_size < sizeof(uint32_t))
382  throw FormatError("COFF symbol table string table size is less than four bytes");
383  p_strtab->extend(strtab_size - sizeof(uint32_t));
384 
385  /* Parse symbols until we've parsed the required number or we run off the end of the section. */
386  for (size_t i = 0; i < fhdr->get_e_coff_nsyms(); i++) {
387  try {
388  SgAsmCoffSymbol *symbol = new SgAsmCoffSymbol(fhdr, this, p_strtab, i);
389  i += symbol->get_st_num_aux_entries();
390  p_symbols->get_symbols().push_back(symbol);
391  } catch (const FormatError &e) {
392  mlog[WARN] <<"SgAsmCoffSymbolTable::parse: invalid symbol: " <<e.what() <<"\n"
393  <<" in section \"" <<get_name()->get_string(true) <<"\"[" <<get_id() <<"]\n"
394  <<" symbol #" <<i <<" at file offset "
395  <<StringUtility::addrToString(get_offset() + i*SgAsmCoffSymbol::COFFSymbol_disk_size) <<"\n"
396  <<" discarding this symbol\n";
397  } catch (const ShortRead &e) {
398  mlog[WARN] <<"SgAsmCoffSymbolTable::parse: read past end of section \"" <<get_name()->get_string(true) <<"\""
399  <<"[" <<get_id() <<"]\n"
400  <<" symbol #" <<i <<" at file offset " <<StringUtility::addrToString(e.offset) <<"\n"
401  <<" skipping " <<StringUtility::plural(fhdr->get_e_coff_nsyms()-i, "symbols")
402  <<" (including this one)\n";
403  break;
404  }
405  }
406 
407  return this;
408 }

410 /* Returns the number of COFF Symbol Table slots occupied by the symbol table. The number of slots can be larger than the
411  * number of symbols since some symbols might have auxiliary entries. */
412 size_t
413 SgAsmCoffSymbolTable::get_nslots() const
414 {
