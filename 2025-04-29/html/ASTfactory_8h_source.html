38 class MultiFactory
39 {
40  typedef std:: map<POETCode*, POETCode*> LastMap;
41  LastMap lastMap;
42  typedef std:: map<POETCode*, LastMap*> FirstMap;
43  FirstMap firstMap;
44  virtual POETCode* new_code(POETCode* first, POETCode* rest) = 0;
45  public:
46  POETCode* new_item(POETCode* first, POETCode* rest) {
47  LastMap* m = 0;
48  FirstMap::const_iterator p = firstMap.find(first);
49  if (p == firstMap.end()) {
50  firstMap[first] = m = new LastMap();
51  }
52  else
53  m = (*p).second;
54  LastMap::const_iterator p2 = m->find(rest);
55  if (p2 == m->end()) {
56  return (*m)[rest] = new_code(first, rest);
57  }
58  else
59  return (*p2).second;
60  }
61 };

74 class TupleTable : public MultiFactory {
75  virtual POETCode* new_code(POETCode* first, POETCode* last)
76  {
77  POETTuple* r1= dynamic_cast<POETTuple*>(first);
78  assert(r1 != 0);
79  int r1size = r1->size();
80  POETTuple* result = new POETTuple(r1size + 1);
81  for (int i = 0; i < r1size; ++i) {
82  result->impl[i] =r1->get_entry(i);
83  }
84  result->impl[r1size]=last;
85  return result;
86  }

92  TupleTable vectors;
93  POETTuple emptyVec;
94  public:
95  POETTuple* new_empty() { return &emptyVec; }
96  POETTuple* new_single(POETCode* item)
97  {
98  assert(item != 0);
99  std::map<POETCode*,POETTuple*>::const_iterator p = singles.find(item);
100  if (p != singles.end())
101  return (*p).second;
102  POETTuple* result = new POETTuple(1);
103  result->impl[0] = item;
104  singles[item] = result;
105  return result;
106  }
107  POETTuple* new_tuple(POETTuple* prev, POETCode* item)
108  {
109  assert(prev != 0 && item != 0);
110  return static_cast<POETTuple*>(vectors.new_item(prev, item));
111  }
