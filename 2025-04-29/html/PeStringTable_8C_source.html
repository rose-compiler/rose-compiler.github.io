124 /* Creates the storage item for the string at the specified offset. If "shared" is true then attempt to re-use a previous storage
125  * object, otherwise create a new one. Each storage object is considered to be a separate string, therefore when two strings
126  * share the same storage object, changing one string changes the other. */
127 SgAsmStringStorage *
128 SgAsmCoffStrtab::createStorage(Address offset, bool shared)
129 {
130  ROSE_ASSERT(offset!=SgAsmGenericString::unallocated);
131  SgAsmGenericSection *container = get_container();
132 
133  /* Has the string already been created? */
134  if (shared) {
135  for (referenced_t::iterator i=p_storageList.begin(); i!=p_storageList.end(); ++i) {
136  if ((*i)->get_offset()==offset && (*i)!=get_dontFree())
137  return *i;
138  }
139  }
140 
141  /* Read string length byte */
142  unsigned char byte;
143  container->readContentLocal(offset, &byte, 1);
144  unsigned len = byte;
145 
146  /* Make sure new storage isn't inside some other string. (We don't support nested strings in COFF where the length byte of
147  * the nested string is one of the characters of the outer string.) */
148  for (referenced_t::iterator i=p_storageList.begin(); i!=p_storageList.end(); ++i) {
149  ROSE_ASSERT((*i)->get_offset()==SgAsmGenericString::unallocated ||
150  offset + 1 + len <= (*i)->get_offset() ||
151  offset >= 1 + (*i)->get_string().size());
152  }
153 
154  /* Create storage object */
155  char *buf = new char[len];
156  container->readContentLocal(offset+1, buf, len);
157  SgAsmStringStorage *storage = new SgAsmStringStorage(this, std::string(buf, len), offset);
158  delete[] buf;
159 
160  /* It's a bad idea to free (e.g., modify) strings before we've identified all the strings in the table. Consider
161  * the case where two strings have the same value and point to the same offset (i.e., they share storage). If we modify one
162  * before we know about the other then (at best) we modify the other one also.
163  *
164  * The only time we can guarantee this is OK is when the new storage points to the same file location as "dont_free"
165  * since the latter is guaranteed to never be freed or shared. This exception is used when creating a new, unallocated
166  * string (see SgAsmStoredString(SgAsmGenericStrtab,const std::string&)). */
167  if (get_numberFreed()>0 && (!get_dontFree() || offset!=get_dontFree()->get_offset())) {
168  mlog[WARN] <<"SgAsmCoffStrtab::create_storage(" <<offset <<"): "
169  <<StringUtility::plural(p_numberFreed, "other strings")
170  <<" (of " <<p_storageList.size() <<" created) in [" <<container->get_id() <<"]"
171  <<" \"" <<container->get_name()->get_string(true) <<"\""
172  <<(1==get_numberFreed()?"has":"have") <<" been modified and/or reallocated\n";
173  ROSE_ASSERT(0==get_numberFreed());
174  }
175 
176  set_isModified(true);
177  p_storageList.push_back(storage);
178  return storage;
179 }

188 /* Write string table back to disk. Free space is zeroed out; holes are left as they are. */
189 void
190 SgAsmCoffStrtab::unparse(std::ostream &f) const
191 {
192  SgAsmGenericSection *container = get_container();
193 
194  /* Write length coded strings. Shared strings will be written more than once, but that's OK. */
195  for (size_t i=0; i<p_storageList.size(); i++) {
196  SgAsmStringStorage *storage = p_storageList[i];
197  ROSE_ASSERT(storage->get_offset()!=SgAsmGenericString::unallocated);
198  Address at = container->write(f, storage->get_offset(), storage->get_string());
199  container->write(f, at, '\0');
200  }
201 
202  /* Fill free areas with zero */
203  BOOST_FOREACH (const AddressInterval &interval, get_freeList().intervals())
204  container->write(f, interval.least(), std::string(interval.size(), '\0'));
205 }
