151 Linux::Specimen::prepareEnvAdjustments() const {
152  // Variables to be erased
153  std::vector<std::string> erasures;
154  for (char **entryPtr = environ; entryPtr && *entryPtr; ++entryPtr) {
155  char *eq = std::strchr(*entryPtr, '=');
156  ASSERT_not_null(eq);
157  std::string name(*entryPtr, eq);
158  for (const boost::regex &re: clearEnvVars_) {
159  if (boost::regex_search(name, re)) {
160  erasures.push_back(name);
161  break;
162  }
163  }
164  }
165 
166  // Return value should be a list of strings that are either variable names to be moved, or variables to be added. The
167  // variables to be added will have an '=' in the string.
168  char **retval = new char*[erasures.size() + setEnvVars_.size() + 1]();
169  char **entryPtr = retval;
170  for (const std::string &name: erasures) {
171  *entryPtr = new char[name.size()+1];
172  std::strcpy(*entryPtr, name.c_str());
173  ++entryPtr;
174  }
175  for (std::map<std::string, std::string>::const_iterator iter = setEnvVars_.begin(); iter != setEnvVars_.end(); ++iter) {
176  std::string var = iter->first + "=" + iter->second;
177  *entryPtr = new char[var.size()+1];
178  std::strcpy(*entryPtr, var.c_str());
179  ++entryPtr;
180  }
181  ASSERT_require((size_t)(entryPtr - retval) == erasures.size() + setEnvVars_.size());
182  ASSERT_require(NULL == *entryPtr);
183  return retval;
184 }

554 void
555 Linux::attach(const Specimen &specimen, Sawyer::Optional<DetachMode> onDelete) {
556  Sawyer::Message::Stream debug(mlog[DEBUG]);
557 
558  if (!specimen.program().empty()) {
559  // Attach to an executable program by running it.
560  detach();
561  specimen_ = specimen;
562  autoDetach_ = onDelete.orElse(DetachMode::KILL);
563  init();
564 
565  // Create the child exec arguments before the fork because heap allocation is not async-signal-safe.
566  char **argv = new char*[1 /*name*/ + specimen.arguments().size() + 1 /*null*/]();
567  argv[0] = new char[specimen.program().string().size()+1];
568  std::strcpy(argv[0], specimen.program().string().c_str());
569  for (size_t i = 0; i < specimen.arguments().size(); ++i) {
570  argv[i+1] = new char[specimen.arguments()[i].size()+1];
571  std::strcpy(argv[i+1], specimen.arguments()[i].c_str());
572  }
573 
574  // Prepare to close files when forking. This is a race because some other thread might open a file without the
575  // O_CLOEXEC flag after we've checked but before we reach the fork. And we can't fix that entirely within ROSE since we
576  // have no control over the user program or other libraries. Furthermore, we must do it here in the parent rather than
577  // after the fork because opendir, readdir, and strtol are not async-signal-safe and Linux does't have a closefrom
578  // syscall.
579  if (specimen.flags().isSet(Flag::CLOSE_FILES)) {
580  static const int minFd = 3;
581  if (DIR *dir = opendir("/proc/self/fd")) {
582  while (const struct dirent *entry = readdir(dir)) {
583  char *rest = nullptr;
584  errno = 0;
585  int fd = strtol(entry->d_name, &rest, 10);
586  if (0 == errno && '\0' == *rest && rest != entry->d_name && fd >= minFd)
587  fcntl(fd, F_SETFD, FD_CLOEXEC);
588  }
589  closedir(dir);
590  }
591  }
592 
593  if (debug) {
594  debug <<"attach: forking child process:\n";
595  for (int i = 0; argv[i]; ++i)
596  debug <<" argv[" <<i <<"]: \"" <<StringUtility::cEscape(argv[i]) <<"\n";
597  }
598 
599  char **envAdjustments = specimen.prepareEnvAdjustments();
600  child_ = fork();
601  if (0==child_) {
602  // Since the parent process may have been multi-threaded, we are now in an async-signal-safe context.
603  adjustEnvironment(envAdjustments);
604  if (specimen.flags().isSet(Flag::REDIRECT_INPUT))
605  devNullTo(0, O_RDONLY); // async-signal-safe
606 
607  if (specimen.flags().isSet(Flag::REDIRECT_OUTPUT))
608  devNullTo(1, O_WRONLY); // async-signal-safe
609 
610  if (specimen.flags().isSet(Flag::REDIRECT_ERROR))
611  devNullTo(2, O_WRONLY); // async-signal-safe
612 
613  // FIXME[Robb Matzke 2017-08-04]: We should be using a direct system call here instead of the C library wrapper because
614  // the C library is adjusting errno, which is not async-signal-safe.
615  if (-1 == ptrace(PTRACE_TRACEME, 0, 0, 0)) {
616  // errno is set, but no way to access it in an async-signal-safe way
617  const char *mesg= "Rose::BinaryAnalysis::Debugger::Linux::attach: ptrace_traceme failed\n";
618  if (write(2, mesg, strlen(mesg)) == -1)
619  abort();
620  _Exit(1); // avoid calling C++ destructors from child
621  }
622 
623  setPersonality(specimen.persona());
624  execv(argv[0], argv);
625 
626  // If failure, we must still call only async signal-safe functions.
627  const char *mesg = "Rose::BinaryAnalysis::Debugger::Linux::attach: exec failed: ";
628  if (write(2, mesg, strlen(mesg)) == -1)
629  abort();
630  mesg = strerror(errno);
631  if (write(2, mesg, strlen(mesg)) == -1)
632  abort();
633  if (write(2, "\n", 1) == -1)
634  abort();
635  _Exit(1);
636  }
637 
638  SAWYER_MESG(debug) <<" child PID is " <<child_ <<"\n";
639  argv = freeStringList(argv);
640  envAdjustments = freeStringList(envAdjustments);
641 
642  waitForChild();
643  if (isTerminated())
644  throw Exception("Rose::BinaryAnalysis::Debugger::Linux::attach: subordinate " +
645  howTerminated() + " before we gained control");
646  } else {
647  // Attach to an existing process.
648  if (-1 == specimen.process()) {
649  detach();
650  } else if (specimen.process() == child_) {
651  // do nothing
652  } else if (specimen.flags().isSet(Flag::ATTACH)) {
653  specimen_ = specimen;
654  init();
655  child_ = specimen.process();
656  SAWYER_MESG(debug) <<"attach: attaching to PID " <<child_ <<"\n";
657  sendCommand(PTRACE_ATTACH);
658  autoDetach_ = onDelete.orElse(DetachMode::DETACH);
659  waitForChild();
660  if (SIGSTOP==sendSignal_)
661  sendSignal_ = 0;
662  } else {
663  specimen_ = specimen;
664  init();
665  child_ = specimen.process();
666  SAWYER_MESG(debug) <<"attach: PID set to " <<child_ <<"\n";
667  autoDetach_ = onDelete.orElse(DetachMode::NOTHING);
668  }
669  }

1021  return 0;

1025  int fd;
1026  T(): fd(-1) {}
1027  ~T() {
1028  if (-1 != fd)
1029  close(fd);
1030  }
1031  } mem;

1034  // time. We'd also need to worry about alignment so we don't inadvertently read past the end of a memory region when we're

1053  nBytes -= nread;
1054  buffer += nread;
1055  totalRead += nread;
1056  }

1064  fmt.prefix = " ";

1181 Linux::findSystemCall() {

1183  std::vector<std::vector<uint8_t>> needles;
1184  if (as<const Architecture::X86>(guessSpecimenArchitecture())) {
1185  needles.push_back(std::vector<uint8_t>{0xcd, 0x80}); // x86: INT 0x80
1186  needles.push_back(std::vector<uint8_t>{0x0f, 0x05}); // x86: SYSCALL
1187  }
1188 
1189  // Make sure the syscall is still there if we already found it. This is reasonally fast.
1190  if (syscallVa_) {
1191  bool found = false;
1192  for (const auto &needle: needles) {
1193  std::vector<uint8_t> buf(needle.size());
1194  size_t nRead = readMemory(*syscallVa_, buf.size(), buf.data());
1195  if (nRead == buf.size() && std::equal(buf.begin(), buf.end(), needle.begin())) {
1196  found = true;
1197  break;
1198  }
1199  }
1200  if (!found)
1201  syscallVa_.reset();
1202  }
1203 
1204  // If we haven't found a syscall (even if we previously searched for one) then search now.
1205  std::vector<MemoryMap::ProcessMapRecord> segments = MemoryMap::readProcessMap(child_);
1206  for (const MemoryMap::ProcessMapRecord &segment: segments) {
1207  if (syscallVa_)
1208  break;
1209  if ("[vvar]" == segment.comment) {
1210  // Linux vvar segment cannot be read from /proc/*/mem even though it's marked readable in the map
1211  } else if ((segment.accessibility & MemoryMap::READ_EXECUTE) != MemoryMap::READ_EXECUTE) {
1212  // Don't try to read memory which is not readable and executable
1213  } else {
1214  auto map = MemoryMap::instance();
1215  map->insertProcessPid(child_, std::vector<MemoryMap::ProcessMapRecord>{segment});
1216  for (const auto &needle: needles) {
1217  if ((syscallVa_ = map->findAny(AddressInterval::whole(), needle, MemoryMap::EXECUTABLE)))
1218  break;
1219  }
1220  }
1221  }
1222 
1223  return syscallVa_;
1224 }

1264 Linux::remoteSystemCall(ThreadId tid, int syscallNumber, std::vector<uint64_t> args) {

1266  if (debug) {
1267  const auto decl = syscallDecls_.lookup(syscallNumber);
1268  debug <<"PID " <<child_ <<": remote system call " <<syscallNumber <<(decl ? " (" + decl->name + ")" : "") <<"\n";
1269  debug <<" arguments:\n";
1270  for (size_t i = 0; i < args.size(); ++i)
1271  debug <<" arg " <<i <<": " <<StringUtility::toHex(args[i]) <<"\n";
1272  }
1273 
1274  // Find a system call that we can hijack to do our bidding.
1275  Sawyer::Optional<Address> syscallVa = findSystemCall();
1276  if (!syscallVa) {
1277  SAWYER_MESG(mlog[ERROR]) <<"syscall failed: cannot find a system call instruction\n";
1278  return -1;
1279  } else {
1280  SAWYER_MESG(debug) <<" subverting system call instruction at " <<StringUtility::addrToString(*syscallVa) <<"\n";
1281  }
1282 
1283  // Registers that we'll need later, one per syscall argument.
1284  Architecture::Base::ConstPtr arch = guessSpecimenArchitecture();
1285  std::vector<RegisterDescriptor> regs;
1286  RegisterDescriptor syscallReg;
1287  if (as<const Architecture::X86>(arch)) {
1288  syscallReg = RegisterDescriptor(x86_regclass_gpr, x86_gpr_ax, 0, arch->bitsPerWord());
1289  if (arch->bitsPerWord() == 32) {
1290  switch (args.size()) {
1291  case 6: regs.push_back(RegisterDescriptor(x86_regclass_gpr, x86_gpr_bp, 0, 32)); // falls through
1292  case 5: regs.push_back(RegisterDescriptor(x86_regclass_gpr, x86_gpr_di, 0, 32)); // falls through
1293  case 4: regs.push_back(RegisterDescriptor(x86_regclass_gpr, x86_gpr_si, 0, 32)); // falls through
1294  case 3: regs.push_back(RegisterDescriptor(x86_regclass_gpr, x86_gpr_dx, 0, 32)); // falls through
1295  case 2: regs.push_back(RegisterDescriptor(x86_regclass_gpr, x86_gpr_cx, 0, 32)); // falls through
1296  case 1: regs.push_back(RegisterDescriptor(x86_regclass_gpr, x86_gpr_bx, 0, 32));
1297  }
1298  } else if (arch->bitsPerWord() == 64) {
1299  switch (args.size()) {
1300  case 6: regs.push_back(RegisterDescriptor(x86_regclass_gpr, 9, 0, 32)); // falls through
1301  case 5: regs.push_back(RegisterDescriptor(x86_regclass_gpr, 8, 0, 32)); // falls through
1302  case 4: regs.push_back(RegisterDescriptor(x86_regclass_gpr, 10, 0, 32)); // falls through
1303  case 3: regs.push_back(RegisterDescriptor(x86_regclass_gpr, x86_gpr_dx, 0, 32)); // falls through
1304  case 2: regs.push_back(RegisterDescriptor(x86_regclass_gpr, x86_gpr_si, 0, 32)); // falls through
1305  case 1: regs.push_back(RegisterDescriptor(x86_regclass_gpr, x86_gpr_di, 0, 32));
1306  }
1307  } else {
1308  ASSERT_not_reachable("invalid x86 word size: " + boost::lexical_cast<std::string>(arch->bitsPerWord()));
1309  }
1310  } else if (arch) {
1311  ASSERT_not_implemented("unsupported debug architecture: " + arch->name());
1312  }
1313  std::reverse(regs.begin(), regs.end());
1314 
1315  // Save registers we're about to overwrite
1316  std::vector<Sawyer::Container::BitVector> savedRegs(args.size());
1317  for (size_t i = 0; i < args.size(); ++i)
1318  savedRegs[i] = readRegister(tid, regs[i]);
1319  Sawyer::Container::BitVector savedSyscallReg = readRegister(tid, syscallReg);
1320 
1321  // Assign arguments to registers
1322  for (size_t i = 0; i < args.size(); ++i)
1323  writeRegister(tid, regs[i], args[i]);
1324  writeRegister(tid, syscallReg, syscallNumber);
1325 
1326  // Single step through the syscall instruction
1327  Address ip = executionAddress(tid);
1328  executionAddress(tid, *syscallVa);
1329  singleStep(tid);
1330  executionAddress(tid, ip);
1331  int64_t retval = readRegister(tid, syscallReg).toSignedInteger();
1332 
1333  // Restore registers
1334  for (size_t i = 0; i < args.size(); ++i)
1335  writeRegister(tid, regs[i], savedRegs[i]);
1336  writeRegister(tid, syscallReg, savedSyscallReg);
1337 
1338  if (debug) {
1339  const auto decl = syscallDecls_.lookup(syscallNumber);
1340  debug <<"system call " <<syscallNumber <<(decl ? " (" + decl->name + ")" : "")
1341  <<" returned " <<StringUtility::toHex(retval) <<"\n";
1342  }
1343  return retval;
1344 }
