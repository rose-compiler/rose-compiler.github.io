18 void ProgramDependenceGraph::build(const CFG& cfg)
19 {
20  std::map<CFGVertex, Vertex> cfgVerticesToPdgVertices;
21 
22  // Add the entry.
23  entry_ = boost::add_vertex(*this);
24 
25  // Add all CFG vertices to PDG.
26  for (CFGVertex cfgVertex : boost::vertices(cfg))
27  {
28  if (cfgVertex == cfg.getEntry() || cfgVertex == cfg.getExit())

30

44 void ProgramDependenceGraph::addControlDependenceEdges(
45  const std::map<CFGVertex, Vertex>& cfgVerticesToPdgVertices,
46  const CFG& cfg)
47 {
48  // Build the dominance frontiers of the reverse CFG, which represents the CDG
49  // of the original CFG.
50  CFG rvsCfg = cfg.makeReverseCopy();
51  DominanceFrontiersT domFrontiers = buildDominanceFrontiers(rvsCfg);
52 
53  for (const DominanceFrontiersT::value_type& vertices : domFrontiers)
54  {
55  Vertex src, tar;
56 
57  CFGVertex from = vertices.first;
58 
59  if (from == cfg.getEntry() || from == cfg.getExit())
60  continue;
61 
62  ROSE_ASSERT(cfgVerticesToPdgVertices.count(from));
63  src = cfgVerticesToPdgVertices.find(from)->second;
64 
65  typedef std::pair<CFGVertex, std::vector<CFGEdge> > VertexEdges;
66  for (const VertexEdges& vertexEdges : vertices.second)
67  {
68  CFGVertex to = vertexEdges.first;
69  const std::vector<CFGEdge>& cdEdges = vertexEdges.second;
70 
71  ROSE_ASSERT(cfgVerticesToPdgVertices.count(to));
72  tar = cfgVerticesToPdgVertices.find(to)->second;
73 
74  for (const CFGEdge& cdEdge : cdEdges)
75  {
76  // Add the edge.
77  Edge edge = boost::add_edge(tar, src, *this).first;
78  (*this)[edge].type = PDGEdge::ControlDependence;
79  (*this)[edge].condition = rvsCfg[cdEdge]->condition();
80  (*this)[edge].caseLabel = rvsCfg[cdEdge]->caseLabel();
81  }
82  }
83  }
84 
85  // Connect an edge from the entry to every node which does not have a control dependence.
86  for (Vertex pdgVertex : boost::vertices(*this))
87  {
88  if (pdgVertex == entry_) continue;
89  //if ((*this)[pdgVertex] == cfg[cfg.getExit()])
90  // continue;
91 
92  if (boost::in_degree(pdgVertex, *this) == 0)
93  {
94  Edge edge = boost::add_edge(entry_, pdgVertex, *this).first;
95  //(*this)[edge].cfgEdge = edgeTable[rvsCfg[cdEdge]];
96  (*this)[edge].type = PDGEdge::ControlDependence;
97  (*this)[edge].condition = VirtualCFG::eckTrue;
98  }
99  }
100 }

120 void ProgramDependenceGraph::writeGraphEdge(std::ostream& out, const Edge& edge) const
121 {
