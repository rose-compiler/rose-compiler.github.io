500 // Process the structure tree into a list of structure operations, creating
501 // the local variable entries along the way. Here is an example of how this
502 // is processed:
503 //
504 // on_entry { A --> B { width,
505 // height,
506 // data --> data1,
507 // more { stuff,
508 // things } } }
509 //
510 // This introduces the following series of structure annotations:
511 //
512 // Source Operator Target
513 // A --> B
514 // B .width B.width
515 // B .height B.height
516 // B .data B.data
517 // B.data --> data1
518 // B .more B.more
519 // B.more .stuff B.more.stuff
520 // B.more .things B.more.things
521 //
522 // This naming scheme is convenient because we can bind the names to
523 // actual memory blocks during the processing of the on_entry annotations,
524 // and then we never have to explicity process the "dot" operator because
525 // it's built into the name.
526 //
527 // These two public methods handle the top variables, which are required
528 // to be previously declared (either as variables from previous
529 // structures, formal paramters to the procedure, or global variables).
530 //
531 // These two routines also handle deletion of the structure trees.
532

625 void procedureAnn::add_structures(bool is_on_entry,
626  structure_list & the_list,
627  annVariable * parent_var,
628  structureTreeAnn * parent_node)
629 {
630  structureAnn * new_struc = 0;
631  annVariable * child_var = 0;
632 
633  // -- Determine if the introduction of the children of this node is
634  // allowed to create new local variables. This is true if we are in an
635  // on_entry annotation, or if the children are qualified by the "new"
636  // operator. There is a special case for the dot operator: if the parent
637  // was new, then the children (the fields) are also new.
638 
639  bool is_target_new = parent_node->is_target_new();
640  bool create = is_on_entry || is_target_new;
641 
642  // -- Visit each child of the current node, looking at the operator to
643  // determine what kind of entry to create in the structure list.
644 
645  structuretree_list * structures = parent_node->targets();
646 
647  for (structuretree_list_p p = structures->begin();
648  p != structures->end();
649  ++p)
650  {
651  structureTreeAnn * child_node = *p;
652 
653  new_struc = 0;
654 
655  // -- CASE 1: This is the target of an arrow operator. In this case,
656  // we look up the target and then create a new structureAnn.
657 
658  if (parent_node->op() == structureTreeAnn::Arrow) {
659  child_var = lookup(child_node->name(), create, is_target_new);
660  if (child_var)
661  new_struc = new structureAnn(parent_var, child_var, 0, child_node->line());
662  }
663 
664  // -- CASE 2: This is the target of a dot operator. The target name
665  // is a concatenation of the parents name with the dot, and the
666  // current node's name (which is basically the field name).
667 
668  // Special case: new structure implies new fields...
669 
670  create = create || parent_var->is_new();
671 
672  if (parent_node->op() == structureTreeAnn::Dot) {
673  string child_name = parent_var->name() + "." + child_node->name();
674  child_var = lookup(child_name, create, parent_var->is_new());
675  if (child_var)
676  new_struc = new structureAnn(parent_var, child_var,
677  & child_node->name(), child_node->line());
678  }
679 
680  // -- If we successfully created an entry, then add it to the
681  // appropriate list, and then work recursively on the tree below.
682 
683  if (new_struc) {
684 
685  // -- Error check: cannot use "new" inside an on_entry annotation
686 
687  if (is_on_entry && child_var->is_new())
688  _annotations->Error(child_node->line(),
689  string("Use of new operator on \"") + child_node->name() +
690  string("\" is not allowed inside on_entry."));
691 
692  // -- Error check: "null" cannot be a source
693 
694  if ( parent_var->name() == "null")
695  _annotations->Error(child_node->line(),
696  string("the \"null\" object cannot point to anything or be decomposed."));
697 
698  // -- Error check: "return" cannot be pointed to
699 
700  if ( child_var->name() == "return")
701  _annotations->Error(child_node->line(),
702  string("the \"return\" object cannot be a target."));
703 
704  // -- All check pass, add the new structure. Also record the fact
705  // that the source will be a use or a def, depending on whether we
706  // are processing on_entry or on_exit, respectively.
707 
708  the_list.push_back(new_struc);
709 
710  if (is_on_entry)
711  add_one_use(parent_var);
712  else
713  add_one_def(parent_var);
714 
715  if (child_node->targets())
716  add_structures(is_on_entry, the_list, child_var, child_node);
717  }
718  else {
719 
720  // -- ERROR: Couldn't find the target child
721 
722  _annotations->Error(child_node->line(),
723  string("Could not find target \"") + child_node->name() +
724  string("\""));
725 
726  return;
727  }
728  }
