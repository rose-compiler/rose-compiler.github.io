156 // Process (but do not consume) left and right parens, braces, and brackets.
157 //
158 // Left tokens push a token onto the stack and return true.
159 //
160 // Right tokens that match with the top of the stack pop a token from the stack and return true.
161 //
162 // All other situations return false.
163 static bool
164 adjustParens(const Ast::File::Ptr &file, size_t at, std::vector<Token> &nestingStack) {
165  ASSERT_not_null(file);
166  if (file->token(at).type() == TOK_LEFT) {
167  nestingStack.push_back(file->token(at));
168  return true;
169  } else if (file->token(at).type() == TOK_RIGHT && !nestingStack.empty()) {
170  const std::string right = file->lexeme(at);
171  const std::string expectedLeft = matching(right);
172  if (file->matches(nestingStack.back(), expectedLeft)) {
173  nestingStack.pop_back();
174  return true;
175  }
176  }
177  return false;
178 }

208 // Look at the C preprocessor and comment tokens in the specified input area and add relevant things to the AST.
209 static void
210 parsePriorRegion(const Ast::File::Ptr &file, const Ast::Definition::Ptr &defn, Ast::CppStack::Stack &runningCppStack,
211  size_t begin, size_t end) {
212  ASSERT_not_null(file);
213  ASSERT_not_null(defn);
214  if (begin >= end)
215  return;
216 
217  // Open a token stream to (re)parse part of the input
218  TokenStream tokens(file->tokenStream(), Sawyer::Language::Clexer::Indices::hull(begin, end - 1));
219  tokens.skipPreprocessorTokens(false);
220  tokens.skipCommentTokens(false);
221 
222  // Process all tokens
223  Token doxyToken;
224  while (tokens[0]) {
225  switch (tokens[0].type()) {
226  case TOK_CPP:
227  Ast::CppStack::process(file, tokens[0], runningCppStack);
228  tokens.consume();
229  break;
230 
231  case TOK_COMMENT:
232  if (isDoxygenComment(file, tokens[0]))
233  doxyToken = tokens[0];
234  tokens.consume();
235  break;
236 
237  case TOK_CHAR:
238  case TOK_STRING:
239  case TOK_LEFT:
240  case TOK_RIGHT:
241  doxyToken = Token(); // the Doxygen comment was for something other than `defn`
242  tokens.consume();
243  break;
244 
245  case TOK_WORD: {
246  const std::string lexeme = tokens.lexeme(tokens[0]);
247  if ("private" == lexeme || "protected" == lexeme || "public" == lexeme) {
248  // These do not cancel the Doxygen comment
249  } else {
250  doxyToken = Token(); // the Doxygen comment was for something other than `defn`
251  }
252  tokens.consume();
253  break;
254  }
255 
256  default:
257  tokens.consume();
258  break;
259  }
260  }
261 
262  // Save information such as the CPP conditional compilation stack and documentation
263  defn->cppStack->stack = runningCppStack;
264  if (doxyToken) {
265  defn->docToken = doxyToken;
266  const size_t n = tokens.location(doxyToken).second;
267  defn->doc = std::string(n, ' ') + tokens.lexeme(doxyToken);
268  }
269 }

372 // If at "<", consume and return this token and all tokens up to and including the balanced ">".
373 static std::string
374 parseTemplateArguments(const Ast::File::Ptr &file) {
375  ASSERT_not_null(file);
376  std::string retval;
377 
378  if (file->matches("<")) {
379  const std::string leftParens = "({[<";
380  const std::string rightParens = ">]})";
381  std::vector<char> stack; // expected closing characters
382  while (Token token = file->consume()) {
383  const std::string s = file->content(token.prior(), token.end());
384  retval += s;
385  for (size_t i = 0; i < s.size(); ++i) {
386  const char ch = s[i];
387  if (leftParens.find(ch) != std::string::npos) {
388  stack.push_back(matching(ch));
389  } else if (rightParens.find(ch) != std::string::npos) {
390  if (stack.empty()) {
391  message(ERROR, file, token, "no matching opening paren/brace");
392  return retval;
393  } else if (stack.back() != ch) {
394  message(ERROR, file, token, "mismatched closing paren/brace");
395  return retval;
396  } else {
397  stack.pop_back();
398  }
399  }
400 
401  if (ch == '>' && stack.empty()) {
402  if (i + 1 < s.size()) // probably a ">>" token, but too many closing brackets
403  message(ERROR, file, token, "mismatched closing paren/brace");
404  return retval;
405  }
406  }
407  }
408  }
409  return retval;
410 }

442 // Parse a list of zero or more tokens up to EOF or a token matching stopAt. Parens, curly braces, and square braces are
443 // honored, so the stopAt must appear outside those grouping tokens.
444 static Ast::TokenList::Ptr
445 parseBalancedTokens(const Ast::File::Ptr &file, size_t at, const std::vector<std::string> &stopAt) {
446  ASSERT_not_null(file);
447  std::vector<Token> nestingStack;
448  auto retval = Ast::TokenList::instance();
449 
450  while (Token token = file->token(at)) {
451  if (nestingStack.empty() && file->matchesAny(at, stopAt)) {
452  return retval;
453  } else if (adjustParens(file, at, nestingStack)) {
454  retval->tokens.push_back(token);
455  ++at;
456  } else if (token.type() == TOK_RIGHT) {
457  return retval;
458  } else {
459  retval->tokens.push_back(token);
460  ++at;
461  }
462  }
463  return retval;
464 }

471 // Parse an attribute, which is a name followed by an optional argument list. Do not consume tokens.
472 static std::pair<size_t, Ast::Attribute::Ptr>
473 parseAttribute(const Ast::File::Ptr &file, size_t at, const std::string &nameSpace) {
474  ASSERT_not_null(file);
475  const size_t begin = at;
476  const auto attrNameTokens = parseQualifiedName(file, at);
477  if (attrNameTokens.empty()) {
478  message(ERROR, file, file->token(at), "expected attribute name");
479  return {0, {}};
480  }
481  const std::string baseName = file->content(attrNameTokens, Expand::NONE);
482  const std::string fqName = nameSpace.empty() ? baseName : nameSpace + "::" + baseName;
483  at += attrNameTokens.size();
484  auto attribute = Ast::Attribute::instance(fqName, attrNameTokens);
485 
486  // Attribute argument list
487  if (file->matches(at, "(")) {
488  ++at;
489  attribute->arguments = Ast::ArgumentList::instance();
490  while (file->token(at) && !file->matches(at, ")")) {
491  if (const auto argument = parseBalancedTokens(file, at, ",")) {
492  attribute->arguments->elmts.push_back(argument);
493  at += argument->size();
494  }
495 
496  if (file->matches(at, ",")) {
497  ++at;
498  } else {
499  break;
500  }
501  }
502  if (file->matches(at, ")")) {
503  ++at;
504  } else {
505  message(ERROR, file, file->token(at), "expected \")\" after attribute argument list");
506  }
507  }
508 
509  ASSERT_require(at > begin);
510  ASSERT_not_null(attribute);
511  return std::make_pair(at - begin, attribute);
512 }

514 // Parse attributes. Add all the attributes to the `attributes` list argument and return the number of tokens parsed. Otherwise only
515 // return zero. Do not consume any tokens.
516 static size_t
517 parseOptionalAttributes(const Ast::File::Ptr &file, Ast::Attribute::EdgeVector<Ast::Attribute> &attributes) {
518  ASSERT_not_null(file);
519  size_t at = 0;
520 
521  while (file->matches(at, "[") && file->matches(at+1, "[")) {
522  at += 2;
523  std::string nameSpace;
524 
525  while (file->token(at)) {
526  // parse "using NAMESPACE:"
527  if (file->matches(at, "using") && file->token(at+1).type() == TOK_WORD && file->matches(at+2, ":")) {
528  if (!nameSpace.empty())
529  message(ERROR, file, file->token(at), "multiple \"using\" in attribute list");
530  if (!attributes.empty())
531  message(ERROR, file, file->token(at), "\"using\" must appear before attributes");
532  nameSpace = file->lexeme(at+1);
533  at += 3;
534  }
535 
536  // Parse an attribute
537  const std::pair<size_t, Ast::Attribute::Ptr> attrPair = parseAttribute(file, at, nameSpace);
538  if (attrPair.second) {
539  attributes.push_back(attrPair.second);
540  at += attrPair.first;
541  } else {
542  message(ERROR, file, file->token(at), "attribute expected");
543  return 0;
544  }
545 
546  if (file->matches(at, ",")) {
547  ++at;
548  } else {
549  break;
550  }
551  }
552 
553  // Closing "]]"
554  if (file->matches(at, "]") && file->matches(at+1, "]")) {
555  at += 2;
556  } else {
557  message(ERROR, file, file->token(at), "expected \"]]\" at end of attribute list");
558  return 0;
559  }
560  }
561  return at;
562 }

578 // Complain and return false if the attribute's number of arguments is incorrect.
579 static bool
580 checkNumberOfArguments(const Ast::File::Ptr &file, const Ast::Attribute::Ptr &attr, size_t minArgs, size_t maxArgs = 0) {
581  ASSERT_not_null(file);
582  ASSERT_not_null(attr);
583  if (0 == maxArgs)
584  maxArgs = minArgs;
585  ASSERT_require(minArgs <= maxArgs);
586 
587  const size_t nArgs = attr->arguments ? attr->arguments->elmts.size() : 0;
588  if (minArgs == maxArgs && nArgs != minArgs) {
589  message(ERROR, file, attr->nameTokens, "attribute \"" + attr->fqName + "\" has " +
590  boost::lexical_cast<std::string>(nArgs) + (1 == nArgs ? " argument" : " arguments") + " but needs " +
591  boost::lexical_cast<std::string>(minArgs) + (1 == minArgs ? " argument" : " arguments"));
592  return false;
593  } else if (minArgs < maxArgs && (nArgs < minArgs || nArgs > maxArgs)) {
594  message(ERROR, file, attr->nameTokens, "attribute \"" + attr->fqName + "\" has " +
595  boost::lexical_cast<std::string>(nArgs) + (1 == nArgs ? " argument" : " arguments") + " but needs between " +
596  boost::lexical_cast<std::string>(minArgs) + " and " + boost::lexical_cast<std::string>(maxArgs) +
597  (1 == maxArgs ? " argument" : " arguments"));
598  return false;
599  }
600  return true;
601 }

603 // Check class attributes
604 static void
605 checkAndApplyClassAttributes(const Ast::File::Ptr &file, const Ast::Class::Ptr &c) {
606  ASSERT_not_null(file);
607  ASSERT_not_null(c);
608  std::map<std::string, Ast::Attribute::Ptr> seen;
609 
610  for (const auto &attr: c->attributes) {
611  if (!checkRecognizedAttribute(attr->fqName, validClassAttrNames)) {
612  // Unknown name
613  message(ERROR, file, attr->nameTokens,
614  "\"" + attr->fqName + "\" is not a recognized Rosebud class attribute; did you mean \"" +
615  bestMatch(validClassAttrNames, attr->fqName) +"\"?");
616 
617  } else if (!seen.insert(std::make_pair(attr->fqName, attr())).second) {
618  // Used multiple times
619  message(ERROR, file, attr->nameTokens, "attribute \"" + attr->fqName + "\" is specified multiple times");
620  message(INFO, file, seen[attr->fqName]->nameTokens, "previously specified here");
621 
622  } else if ("Rosebud::abstract" == attr->fqName ||
623  "Rosebud::no_constructors" == attr->fqName ||
624  "Rosebud::no_destructor" == attr->fqName ||
625  "Rosebud::small_header" == attr->fqName ||
626  "Rosebud::suppress" == attr->fqName) {
627  checkNumberOfArguments(file, attr(), 0);

659  } else if (!seen.insert(std::make_pair(attr->fqName, attr())).second) {

670  checkNumberOfArguments(file, attr(), 0);

693  }

762 // Parse an optional property and return it if parsed and consumed from the input.
763 static Ast::Property::Ptr
764 parseOptionalProperty(const Ast::File::Ptr &file, Ast::CppStack::Stack&/*runningCppStack*/) {
765  ASSERT_not_null(file);
766  auto property = Ast::Property::instance();
767  property->startToken = file->token();
768 
769  // Look for attributes. At least one of them must be in the 'Rosebud' namespace in order for this to be a property.
770  size_t at = parseOptionalAttributes(file, property->attributes);
771  if (0 == at)
772  return {};
773 
774  // Parsing was a success only if we found at least one Rosebud attribute
775  bool isProperty = false;
776  for (const auto &attribute: property->attributes) {
777  if (boost::starts_with(attribute->fqName, "Rosebud::")) {
778  isProperty = true;
779  break;
780  }
781  }
782  if (!isProperty)
783  return {};
784 
785  //-----------------------------------------------------------------------------------------
786  // Everything beyond here is an error since we know by now that we have Rosebud attributes.
787  // However, don't consume any tokens if there's an error because they might be things we
788  // need to emit as non-Rosebud C++.
789  //-----------------------------------------------------------------------------------------
790 
791  // Parse the type and name of the data member, up to an "=" or ";". The type is the first N-1 tokens and the name is the last
792  // token.
793  property->cType = parseBalancedTokens(file, at, std::vector<std::string>{"=", ";"});
794  if (property->cType->size() < 2) {
795  message(ERROR, file, file->token(at), "property type and name expected");
796  return {};
797  }
798  at += property->cType->size();
799  property->nameToken = property->cType->tokens.back();
800  property->name = file->lexeme(property->nameToken);
801  property->cType->tokens.pop_back();
802 
803  // If the property name is followed by an "=" then parse the initialization expression up to the next ";" that occurs at this
804  // same paren nesting level.
805  if (file->matches(at, "=")) {
806  ++at;
807  property->cInit = parseBalancedTokens(file, at, ";");
808  if (property->cInit->empty()) {
809  message(ERROR, file, file->token(at), "property initialization expression expected");
810  return {};
811  }
812  at += property->cInit->size();
813  }
814 
815  // Semicolon at the end of the property definition
816  if (file->matches(at, ";")) {
817  ++at;
818  } else {
819  message(ERROR, file, file->token(at), "\";\" expected at end of property declaration");
820  return {};
821  }
822 
823  // Some of the Rosebud attributes cause changes to the AST
824  checkAndApplyPropertyAttributes(file, property);
825 
826  file->consume(at);
827  return property;
828 }

831 checkClassCppDirectives(const Ast::File::Ptr &file,
832  const Ast::CppStack::Stack &cstack, const Ast::CppStack::Stack &pstack,
833  const Token &classDefnStart, const Token &endToken) {
834  ASSERT_not_null(file);

840  // definition. Unfortunately this would require some redesign because when we're parsing CPP directives the #endif causes
841  // information about the nesting structure to be discarded. One way would be to move this function and its arguments inside
842  // the loop that processes the CPP directives.

851 static void
852 parseClassDefinitionBody(const Ast::File::Ptr &file, const Ast::Class::Ptr &c, Ast::CppStack::Stack &runningCppStack) {
853  ASSERT_not_null(file);
854  ASSERT_not_null(c);
855  std::vector<Token> nestingStack;
856  size_t filePos = file->token().prior();
857 
858  while (file->token() && (!nestingStack.empty() || !file->matches("}"))) {
859  const size_t startOfProperty = file->token().begin();
860  if (auto property = parseOptionalProperty(file, runningCppStack)) {
861  // Parse part of class definition before the property
862  parsePriorRegion(file, property, runningCppStack, filePos, startOfProperty);
863  checkClassCppDirectives(file, c->cppStack->stack, property->cppStack->stack, c->startToken, property->startToken);
864  property->priorText = file->trimmedContent(filePos, startOfProperty, property->docToken, property->priorTextToken);
865  c->properties.push_back(property);
866  filePos = file->token().prior(); // end of property
867 
868  // Check for property problems
869  if (settings.showingWarnings) {
870  ASSERT_forbid(property->name.empty());
871  if (property->name.find('_') != std::string::npos || std::isupper(property->name[0]))
872  message(WARN, file, property->nameToken, "property name should be camelCase");
873 
874  if (property->doc.empty() &&
875  ((property->accessorNames && !property->accessorNames->empty()) ||
876  (property->mutatorNames && !property->mutatorNames->empty()))) {
877  message(WARN, file, property->nameToken, "no Doxygen documentation for property \"" + property->name + "\"");
878  }
879 
880  if (!property->cInit) {
881  std::regex integralTypeRe("bool|char|short|int|long|long long"
882  "|unsigned"
883  "|unsigned (char|short|int|long|long long)"
884  "|signed"
885  "|signed (char|short|int|long|long long)"
886  "|u?int[0-9]+_t");
887  std::regex floatingTypeRe("float|double");
888  const std::string cType = property->cType->string(file);
889  if (boost::ends_with(cType, "*")) {
890  message(WARN, file, property->cType->tokens,
891  "no initializer for raw pointer (did you intend to initialize with \"= nullptr\"?)");
892  } else if (std::regex_match(cType, integralTypeRe)) {
893  message(WARN, file, property->cType->tokens,
894  "no initializer for integral type (did you intend to initialize with \"= 0\"?)");
895  } else if (std::regex_match(cType, floatingTypeRe)) {
896  message(WARN, file, property->cType->tokens,
897  "no initializer for floating-point type (did you intend to initialize with \"= NAN\"?)");
898  }
899  }
900  }
901 
902  } else {
903  if (!adjustParens(file, 0, nestingStack) && file->token().type() == TOK_RIGHT)
904  message(ERROR, file, file->token(), "unmatched \"" + file->lexeme() + "\"");
905  file->consume();
906  }
907  }
908 
909  // Check that the code after the last property but before the end of the class definition follows the rules for C preprocessor
910  // conditional compilation directives. I.e., conditional compilation directives that are opened before the beginning of the
911  // class definition are not closed until after the end of the class definition.
912  Ast::CppStack::Stack tempStack = runningCppStack;
913  std::vector<Token> tokens = parseCppAndComments(file, filePos, file->token().begin());
914  for (const Token &token: tokens)
915  Ast::CppStack::process(file, token, tempStack);
916  checkClassCppDirectives(file, c->cppStack->stack, tempStack, c->startToken, file->token());
917 
918  // Accumulate the input that appears inside the class definition after the last property but before the definition's final
919  // closing brace.
920  c->endText = file->trimmedContent(filePos, file->token().begin(), c->endTextToken);
921 }

923 // Parse an entire class definition if the token stream for the file is positioned at the beginning of a class definition. If it
924 // isn't, then return a null pointer.
925 static Ast::Class::Ptr
926 parseClassDefinition(const Ast::File::Ptr &file, Ast::CppStack::Stack &runningCppStack) {
927  ASSERT_not_null(file);
928  auto c = Ast::Class::instance();
929 
930  // A class definition can start with attributes.
931  const size_t at = parseOptionalAttributes(file, c->attributes);
932  if (!isAtClassDefinition(file, at))
933  return {}; // we're not at a class definition, but no error yet
934  file->consume(at);
935 
936  // The "class" keyword.
937  if (file->matches(0, "class")) {
938  c->startToken = file->consume();
939  } else {
940  message(ERROR, file, file->token(), "\"class\" expected at start of class definition");
941  return {};
942  }
943 
944  // Class name
945  if (file->token().type() == TOK_WORD) {
946  message(DEBUG, file, file->token(), "defining class \"" + file->lexeme() + "\"");
947  c->name = file->lexeme();
948  c->nameToken = file->consume();
949  } else {
950  message(ERROR, file, file->token(), "class name expected at start of class definition");
951  return {};
952  }
953 
954  // Class inheritance
955  if (file->matches(0, ":")) {
956  file->consume();
957  c->inheritance = parseClassInheritance(file);
958  }
959 
960  // Class body from the "{" to the matching "}"
961  if (file->matches(0, "{")) {
962  file->consume();
963  } else {
964  message(ERROR, file, file->token(), "expected \"{\" at start of class definition");
965  return {};
966  }
967  parseClassDefinitionBody(file, c, runningCppStack);
968  if (file->matches(0, "}")) {
969  file->consume();
970  } else {
971  message(ERROR, file, file->token(), "expected \"}\" at end of class definition");
972  return {};
973  }
974 
975  // Semicolon after the definition
976  if (file->matches(0, ";")) {
977  file->consume();
978  } else {
979  message(ERROR, file, file->token(), "expected semicolon after class definition");
980  return {};
981  }
982 
983  checkAndApplyClassAttributes(file, c);
984  return c;
985 }

987 static Ast::File::Ptr
988 parseFile(const std::string &fileName) {
989  auto file = Ast::File::instance(fileName);
990  if (!file)
991  return {};
992 
993  size_t filePos = 0;
994  Ast::CppStack::Stack runningCppStack;
995  std::vector<std::string> namespaceStack;
996 
997  while (file->token()) {
998  parseToNamespaceClassDefinitionOrAttributeList(file);
999  size_t startOfConstruct = file->token().begin();
1000 
1001  // Handle namespace declarations like "namespace Foo {" or "namespace Foo::Bar::...::Baz" {"
1002  const std::string ns = parseOptionalNamespaceDeclaration(file);
1003  if (!ns.empty()) {
1004  namespaceStack.push_back(ns);
1005  continue;
1006  }
1007 
1008  // Handle class declarations
1009  if (auto c = parseClassDefinition(file, runningCppStack)) {
1010  // Parse area before the class
1011  parsePriorRegion(file, c, runningCppStack, filePos, startOfConstruct);
1012  c->priorText = file->trimmedContent(filePos, startOfConstruct, c->docToken, c->priorTextToken);
1013  c->qualifiedNamespace = makeQualifiedName(namespaceStack);
1014  file->classes.push_back(c);
1015  filePos = file->token().prior();
1016 
1017  // Show warnings for questionable things about a class
1018  if (settings.showingWarnings) {
1019  if (c->doc.empty())
1020  message(WARN, file, c->nameToken, "no Doxygen documentation for class \"" + c->name + "\"");
1021  ASSERT_forbid(c->name.empty());
1022  if (!std::isupper(c->name[0]) || c->name.find('_') != std::string::npos)
1023  message(WARN, file, c->nameToken, "class name \"" + c->name + "\" should be PascalCase");
1024  }
1025  continue;
1026  }
1027 
1028  // Handle end of namespace curly brace.
1029  if (!namespaceStack.empty() && file->matches(0, "}")) {
1030  namespaceStack.pop_back();
1031  file->consume();
1032  continue;
1033  }
1034 
1035  // Handle anything else
1036  file->consume();
1037  }
1038 
1039  // Accumulate the part of the file that appears after the last class definition
1040  file->endText = file->trimmedContent(filePos, file->token().end(), file->endTextToken);
1041 
1042  // Warn about file problems
1043  if (file->classes.empty())
1044  message(WARN, file, "file contains to class definitions");
1045 
1046  return file;
1047 }
