168 /* Return true if the file looks like it might be an NE file according to the magic number. The file must contain what
169  * appears to be a DOS File Header at address zero, and what appears to be an NE File Header at a file offset specified in
170  * part of the DOS File Header (actually, in the bytes that follow the DOS File Header). */
171 bool
172 SgAsmNEFileHeader::isNe(SgAsmGenericFile *file)
173 {
174  /* Turn off byte reference tracking for the duration of this function. We don't want our testing the file contents to
175  * affect the list of bytes that we've already referenced or which we might reference later. */
176  bool was_tracking = file->get_trackingReferences();
177  file->set_trackingReferences(false);
178 
179  try {
180  /* Check DOS File Header magic number at beginning of the file */
181  unsigned char dos_magic[2];
182  file->readContent(0, dos_magic, sizeof dos_magic);
183  if ('M'!=dos_magic[0] || 'Z'!=dos_magic[1])
184  throw 1;
185 
186  /* Read four-byte offset of potential PE File Header at offset 0x3c */
187  uint32_t lfanew_disk;
188  file->readContent(0x3c, &lfanew_disk, sizeof lfanew_disk);
189  Rose::BinaryAnalysis::Address ne_offset = Rose::BinaryAnalysis::ByteOrder::leToHost(lfanew_disk);
190 
191  /* Look for the NE File Header magic number */
192  unsigned char ne_magic[2];
193  file->readContent(ne_offset, ne_magic, sizeof ne_magic);
194  if ('N'!=ne_magic[0] || 'E'!=ne_magic[1])
195  throw 1;
196  } catch (...) {
197  file->set_trackingReferences(was_tracking);
198  return false;
199  }
200 
201  file->set_trackingReferences(was_tracking);
202  return true;
203 }

960 /* Write section back to disk */
961 void
962 SgAsmNEEntryTable::unparse(std::ostream &f) const
963 {
964  Rose::BinaryAnalysis::Address spos=0; /*section offset*/
965 
966  for (size_t bi=0, ei=0; bi < p_bundle_sizes.size(); ei += p_bundle_sizes[bi++]) {
967  ROSE_ASSERT(p_bundle_sizes[bi] > 0 && p_bundle_sizes[bi] <= 0xff);
968  unsigned char n = p_bundle_sizes[bi];
969  spos = write(f, spos, n);
970

972  if (0 == p_entries[ei]->get_sectionIndex()) {
973  /* Unused entries */
974  spos = write(f, spos, '\0');
975  } else if (0 == p_entries[ei]->get_int3f()) {
976  /* Fixed entries */
977  ROSE_ASSERT(p_entries[ei]->get_sectionIndex() <= 0xff);
978  unsigned char n = p_entries[ei]->get_sectionIndex();
979  spos = write(f, spos, n);
980  for (size_t i = 0; i < p_bundle_sizes[bi]; i++) {
981  ROSE_ASSERT(p_entries[ei]->get_sectionIndex() == p_entries[ei+i]->get_sectionIndex());
982  ROSE_ASSERT(p_entries[ei+i]->get_int3f() == 0);
983  ROSE_ASSERT(p_entries[ei+i]->get_flags() <= 0xff);
984  n = p_entries[ei+i]->get_flags();
985  spos = write(f, spos, n);
986  uint16_t eoff_le;
987  Rose::BinaryAnalysis::ByteOrder::hostToLe(p_entries[ei+i]->get_sectionOffset(), &eoff_le);
988  spos = write(f, spos, sizeof eoff_le, &eoff_le);
989  }
990  } else {
991  /* Movable entries */
992  spos = write(f, spos, '\377');
993  for (size_t i = 0; i < p_bundle_sizes[bi]; i++) {
994  ROSE_ASSERT(p_entries[ei+i]->get_sectionIndex() > 0);
995  ROSE_ASSERT(p_entries[ei+i]->get_int3f() != 0);
996  ROSE_ASSERT(p_entries[ei+i]->get_flags() <= 0xff);
997  n = p_entries[ei+i]->get_flags();
998  spos = write(f, spos, n);
999  uint16_t word;
1000  Rose::BinaryAnalysis::ByteOrder::hostToLe(p_entries[ei+i]->get_int3f(), &word);
1001  spos = write(f, spos, sizeof word, &word);
1002  ROSE_ASSERT(p_entries[ei+i]->get_sectionIndex() <= 0xff);
1003  n = p_entries[ei+i]->get_sectionIndex();
1004  spos = write(f, spos, n);
1005  Rose::BinaryAnalysis::ByteOrder::hostToLe(p_entries[ei+i]->get_sectionOffset(), &word);
1006  spos = write(f, spos, sizeof word, &word);
1007  }
1008  }
1009  }
1010  write(f, spos, '\0');
1011 }

1040 /* Constructor. */
1041 SgAsmNERelocEntry::SgAsmNERelocEntry(SgAsmGenericSection *relocs, Rose::BinaryAnalysis::Address at,
1042  Rose::BinaryAnalysis::Address *rec_size) {
1043  initializeProperties();
1044 
1045  unsigned char byte;
1046  uint16_t u16_disk;
1047  uint32_t u32_disk;
1048 
1049  Rose::BinaryAnalysis::Address orig_at = at;
1050  ROSE_ASSERT(at == relocs->get_size()); /*the section is extended as we parse*/
1051 
1052  /* Only the low nibble is used for source type; the high nibble is modifier bits */
1053  relocs->extend(1);
1054  relocs->readContentLocal(at++, &byte, 1);
1055  unsigned n = byte;
1056  p_src_type = (SgAsmNERelocEntry::NERelocSrcType)(n & 0x0f);
1057  p_modifier = (SgAsmNERelocEntry::NERelocModifiers)(n>>8);
1058 
1059  /* The target type (3 bits), additive flag (1 bit), and target flags */
1060  relocs->extend(1);
1061  relocs->readContentLocal(at++, &byte, 1);
1062  n = byte;
1063  p_tgt_type = (SgAsmNERelocEntry::NERelocTgtType)(n & 0x03);
1064  p_flags = (SgAsmNERelocEntry::NERelocFlags)(n>>2);
1065 
1066  /* src_offset is the byte offset into the source section that needs to be patched. If this is an additive relocation then
1067  * the source will be patched by adding the target value to the value stored at the source. Otherwise the target value is
1068  * written to the source and the old contents of the source contains the next source offset, until we get 0xffff. */
1069  relocs->extend(2);
1070  relocs->readContentLocal(at, &u16_disk, 2);
1071  p_src_offset = Rose::BinaryAnalysis::ByteOrder::leToHost(u16_disk);
1072  at += 2;
1073 
1074  switch (p_tgt_type) {
1075  case RF_TGTTYPE_IREF:
1076  /* Internal reference */
1077  relocs->extend(4);
1078  relocs->readContentLocal(at++, &byte, 1);
1079  p_iref.sect_idx = byte;
1080  relocs->readContentLocal(at++, &byte, 1);
1081  p_iref.res1 = byte;
1082  relocs->readContentLocal(at, &u16_disk, 2);
1083  p_iref.tgt_offset = Rose::BinaryAnalysis::ByteOrder::leToHost(u16_disk);
1084  at += 2;
1085  break;
1086  case RF_TGTTYPE_IORD:
1087  /* Imported ordinal */
1088  relocs->extend(4);
1089  relocs->readContentLocal(at, &u16_disk, 2);
1090  p_iord.modref = Rose::BinaryAnalysis::ByteOrder::leToHost(u16_disk);
1091  relocs->readContentLocal(at+2, &u16_disk, 2);
1092  p_iord.ordinal = Rose::BinaryAnalysis::ByteOrder::leToHost(u16_disk);
1093  at += 4;
1094  if (p_flags & RF_2EXTRA) {
1095  if (p_flags & RF_32ADD) {
1096  relocs->extend(4);
1097  relocs->readContentLocal(at+8, &u32_disk, 4);
1098  p_iord.addend = Rose::BinaryAnalysis::ByteOrder::leToHost(u32_disk);
1099  at += 4;
1100  } else {
1101  relocs->extend(2);
1102  relocs->readContentLocal(at+8, &u16_disk, 2);
1103  p_iord.addend = Rose::BinaryAnalysis::ByteOrder::leToHost(u16_disk);
1104  at += 2;
1105  }
1106  } else {
1107  p_iord.addend = 0;
1108  }
1109  break;
1110  case RF_TGTTYPE_INAME:
1111  /* Imported name */
1112  relocs->extend(4);
1113  relocs->readContentLocal(at, &u16_disk, 2);
1114  p_iname.modref = Rose::BinaryAnalysis::ByteOrder::leToHost(u16_disk);
1115  relocs->readContentLocal(at+2, &u16_disk, 2);
1116  p_iname.nm_off = Rose::BinaryAnalysis::ByteOrder::leToHost(u16_disk);
1117  at += 4;
1118  if (p_flags & RF_2EXTRA) {
1119  if (p_flags & RF_32ADD) {
1120  relocs->extend(4);
1121  relocs->readContentLocal(at+8, &u32_disk, 4);
1122  p_iname.addend = Rose::BinaryAnalysis::ByteOrder::leToHost(u32_disk);
1123  at += 4;
1124  } else {
1125  relocs->extend(2);
1126  relocs->readContentLocal(at+8, &u16_disk, 2);
1127  p_iname.addend = Rose::BinaryAnalysis::ByteOrder::leToHost(u16_disk);
1128  at += 2;
1129  }
1130  } else {
1131  p_iname.addend = 0;
1132  }
1133  break;
1134  case RF_TGTTYPE_OSFIXUP:
1135  /* Operating system fixup */
1136  relocs->extend(4);
1137  relocs->readContentLocal(at, &u16_disk, 2);
1138  p_osfixup.type = Rose::BinaryAnalysis::ByteOrder::leToHost(u16_disk);
1139  relocs->readContentLocal(at+2, &u16_disk, 2);
1140  p_osfixup.res3 = Rose::BinaryAnalysis::ByteOrder::leToHost(u16_disk);
1141  at += 4;
1142  break;
1143  }
1144 
1145  if (rec_size)
1146  *rec_size = at - orig_at;
1147 }

1149 /* Write entry back to disk at the specified section and section offset, returning new offset */
1150 Rose::BinaryAnalysis::Address
1151 SgAsmNERelocEntry::unparse(std::ostream &f, const SgAsmGenericSection *section, Rose::BinaryAnalysis::Address spos) const
1152 {
1153  unsigned char byte;
1154  byte = (p_modifier << 8) | (p_src_type & 0x0f);
1155  spos = section->write(f, spos, byte);
1156  byte = (p_flags << 2) | (p_tgt_type & 0x03);
1157  spos = section->write(f, spos, byte);
1158 
1159  uint16_t word;
1160  uint32_t dword;
1161  Rose::BinaryAnalysis::ByteOrder::hostToLe(p_src_offset, &word);
1162  spos = section->write(f, spos, sizeof word, &word);
1163 
1164  switch (p_tgt_type) {
1165  case RF_TGTTYPE_IREF:
1166  Rose::BinaryAnalysis::ByteOrder::hostToLe(p_iref.sect_idx, &byte);
1167  spos = section->write(f, spos, byte);
1168  Rose::BinaryAnalysis::ByteOrder::hostToLe(p_iref.res1, &byte);
1169  spos = section->write(f, spos, byte);
1170  Rose::BinaryAnalysis::ByteOrder::hostToLe(p_iref.tgt_offset, &word);
1171  spos = section->write(f, spos, sizeof word, &word);
1172  break;
1173  case RF_TGTTYPE_IORD:
1174  Rose::BinaryAnalysis::ByteOrder::hostToLe(p_iord.modref, &word);
1175  spos = section->write(f, spos, sizeof word, &word);
1176  Rose::BinaryAnalysis::ByteOrder::hostToLe(p_iord.ordinal, &word);
1177  spos = section->write(f, spos, sizeof word, &word);
1178  if (p_flags & RF_2EXTRA) {
1179  if (p_flags & RF_32ADD) {
1180  Rose::BinaryAnalysis::ByteOrder::hostToLe(p_iord.addend, &dword);
1181  spos = section->write(f, spos, sizeof dword, &dword);
1182  } else {
1183  Rose::BinaryAnalysis::ByteOrder::hostToLe(p_iord.addend, &word);
1184  spos = section->write(f, spos, sizeof word, &word);
1185  }
1186  } else {
1187  ROSE_ASSERT(p_iord.addend==0);
1188  }
1189  break;
1190  case RF_TGTTYPE_INAME:
1191  Rose::BinaryAnalysis::ByteOrder::hostToLe(p_iname.modref, &word);
1192  spos = section->write(f, spos, sizeof word, &word);
1193  Rose::BinaryAnalysis::ByteOrder::hostToLe(p_iname.nm_off, &word);
1194  spos = section->write(f, spos, sizeof word, &word);
1195  if (p_flags & RF_2EXTRA) {
1196  if (p_flags & RF_32ADD) {
1197  Rose::BinaryAnalysis::ByteOrder::hostToLe(p_iname.addend, &dword);
1198  spos = section->write(f, spos, sizeof dword, &dword);
1199  } else {
1200  Rose::BinaryAnalysis::ByteOrder::hostToLe(p_iname.addend, &word);
1201  spos = section->write(f, spos, sizeof word, &word);
1202  }
1203  } else {
1204  ROSE_ASSERT(p_iname.addend==0);
1205  }
1206  break;
1207  case RF_TGTTYPE_OSFIXUP:
1208  Rose::BinaryAnalysis::ByteOrder::hostToLe(p_osfixup.type, &word);
1209  spos = section->write(f, spos, sizeof word, &word);
1210  Rose::BinaryAnalysis::ByteOrder::hostToLe(p_osfixup.res3, &word);
1211  spos = section->write(f, spos, sizeof word, &word);
1212  break;
1213  default:
1214  ROSE_ASSERT(!"unknown relocation target type");
1215  }
1216  return spos;
1217 }
