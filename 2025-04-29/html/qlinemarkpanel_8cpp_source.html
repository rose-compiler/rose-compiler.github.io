79 {

90 
91  const QFontMetrics fm( d->font() );
92 
93  int n, posY,

102  QLineMarksInfoCenter *mic = QLineMarksInfoCenter::instance();
103 
104  n = d->lineNumber(contentsY);

106 
107  //qDebug("first = %i; last = %i", first, last);
108  //qDebug("beg pos : %i", posY);

112  //qDebug("n = %i; pos = %i", n, posY);
113  QDocumentLine line = d->line(n);

116  break;
117 
118  if ( line.isHidden() )
119  continue;
120

123
