80 bool RoseAst::iterator::operator==(const iterator& x) const {
81  if(is_past_the_end() != x.is_past_the_end())
82  return false;
83  if(is_past_the_end() && x.is_past_the_end())
84  return true;
85  // this check ensures that 0 values work for trees. For DAGs we would need to compare the entire stack (i.e. context). Hence, this comparison
86  // is guaranteed to work for trees, but not for DAGs. For DAGs the comparison might be true although the traversal is at different positions at
87  // shared nodes. This is not problematic for comparisons with the past-the-end comparison, but may be problematic when different DAGs are compared.
88  if(_stack.top().node!=x._stack.top().node ||_stack.top().index!=x._stack.top().index)
89  return false;
90  // mode must be the same otherwise iterators are different as
91  // ++ may go to different elements. Therefore i==j -> *i==*j is not violated
92  // because in case of different modes we define == such that i==j does not hold.
93  if(_skipChildrenOnForward!=x._skipChildrenOnForward)
94  return false;
95  return true;
96 }

169 RoseAst::iterator&
170 RoseAst::iterator::operator++() {
171  // check if we are already past the end
172  if(is_past_the_end())
173  return *this;
174  assert(_stack.size()>0);
175  stack_element e=_stack.top();
176  _stack.pop();
177 
178  SgNode* new_node=0;
179  if(e.index==ROOT_NODE_INDEX) {
180  if(num_children(e.node)==0)
181  return *this;
182  else {
183  new_node=e.node;
184  goto root_node_has_children;
185  }
186  };
187  // check if we are at a null node (nothing to push and nothing to skip)
188  if(e.node==0) {
189  return *this;
190  }
191  new_node=access_node_by_parent_and_index(e.node,e.index);
192  root_node_has_children:
193  if(new_node==0) {
194  return *this;
195  } else if(!_withTemplates) {
196  if(isTemplateNode(new_node)) {
197  return *this;
198  }
199  } else if(!_skipChildrenOnForward) {
200  int numChildren=num_children(new_node);
201  // we need to push children in reverse order on the stack (if they are != null)
202  for(int new_index=numChildren-1;new_index>=0;new_index--) {
203  stack_element new_e;
204  new_e.node=new_node;
205  new_e.index=new_index;
206  if(new_e.node!=0) {
207  if(_withNullValues) {
208  // if we visit null-nodes we can push anything
209  _stack.push(new_e);
210  } else {
211  if(access_node_by_parent_and_index(new_e.node,new_e.index)!=0) {
212  _stack.push(new_e);
213  } else {
214  // we are not visiting null nodes therefore we do not push null nodes on the stack
215  }
216  }
217  } else {
218  // a null node has no children: nothing to do.
219  }
220  }
221  } else {
222  /* we have skipped children (because we do not put them on the stack)
223  since we do this only once, we set the flag back to false
224  */
225  _skipChildrenOnForward=false;
226  }
227  return *this;
228 }
