75  Access access) {
76  ASSERT_not_null(c);
77 
78  // Get the properties that are constructor arguments. If there are none, then we don't need this constructor.
79  std::vector<Ast::Property::Ptr> args = allConstructorArguments(c, h);
80  if (args.empty())
81  return false;
82 
83  //--------------------------------------------------------------------------------------------------------------------------
84  // Constructor declaration
85  //--------------------------------------------------------------------------------------------------------------------------
86 
87  // Constructor declaration having the names of all the Rosebud::ctor_arg properties as argument names.
88  header <<"\n"
89  <<THIS_LOCATION <<toString(access) <<":\n"
90  <<" /** Constructor. */\n"
91  <<" " <<(args.size() == 1 ? "explicit " : "") <<c->name <<"(";
92  for (const auto &p: args) {
93  auto argClass = p->findAncestor<Ast::Class>();
94  ASSERT_not_null(argClass);
95  header <<(p == args.front() ? "" : ",\n " + std::string(c->name.size() + 1, ' '))
96  <<constRef(propertyMutatorArgumentType(p)) <<" " <<p->name;
97  }
98  header <<");\n";
99 
100  //--------------------------------------------------------------------------------------------------------------------------
101  // Constructor definition
102  //--------------------------------------------------------------------------------------------------------------------------
103 
104  // Show information about how the constructor args map to classes. This comment is to help when debugging ROSE.
105  impl <<"\n"
106  <<THIS_LOCATION <<"// The association between constructor arguments and their classes:\n";
107  for (const auto &p: args) {
108  auto argClass = p->findAncestor<Ast::Class>();
109  ASSERT_not_null(argClass);
110  impl <<(boost::format("// property=%-16s class=%s\n") % p->name % argClass->name);
111  }
112 
113  // Implementation declaration, almost the same as above.
114  impl <<THIS_LOCATION <<c->name <<"::" <<c->name <<"(";
115  for (const auto &p: args) {
116  auto argClass = p->findAncestor<Ast::Class>();
117  impl <<(p == args.front() ? "" : ",\n" + std::string(2*c->name.size()+3, ' '))
118  <<constRef(propertyMutatorArgumentType(p)) <<" " <<p->name;
119  }
120  impl <<")";
121 
122  // Emit the initializer for the base class if there is a base class, and remove those properties from the `args` list.
123  size_t nInits = 0;
124  for (auto base: c->inheritance) {
125  if (base.first != "public")
126  continue;
127  const std::string baseClassName = base.second;
128  const auto baseClassVertex = h.findVertexKey(baseClassName);
129  if (baseClassVertex == h.vertices().end())
130  continue;
131 
132  Ast::Class::Ptr baseClass = baseClassVertex->value();
133  const size_t nBaseArgs = allConstructorArguments(baseClass, h).size();
134  ASSERT_require(nBaseArgs <= args.size());
135  if (nBaseArgs > 0) {
136  impl <<"\n : " <<baseClassName <<"(";
137  for (size_t i = 0; i < nBaseArgs; ++i) {
138  auto p = args.front();
139  auto argClass = p->findAncestor<Ast::Class>();
140  ASSERT_not_null(argClass);
141  impl <<(i ? ", " : "") <<p->name;
142  args.erase(args.begin());
143  }
144  impl <<")";
145  ++nInits;
146  }
147  }
148 
149  // Emit the property initializations, some of which come from arguments
150  for (const auto &p: c->properties) {
151  auto arg = std::find(args.begin(), args.end(), p());
152  auto argClass = p->findAncestor<Ast::Class>();
153  if (arg != args.end()) {
154  // Property is initialized from an argument
155  impl <<(nInits++ ? "\n , " : "\n : ")
156  <<propertyDataMemberName(p()) <<"(";
157  if (p->findAttribute("Rosebud::traverse")) {
158  impl <<"initParentPointer(" <<ctorInitializerExpression(p(), p->name) <<", this)";
159  } else {
160  impl <<ctorInitializerExpression(p(), p->name);
161  }
162  impl <<")";
163  } else {
164  // Property is initialized from the property's initial value expression
165  const std::string initExpr = ctorInitializerExpression(p(), initialValue(p()));
166  if (!initExpr.empty()) {
167  impl <<(nInits++ ? "\n , " : "\n : ")
168  <<propertyDataMemberName(p()) <<"(" <<initExpr <<")";
169  }
170  }
171  }
172 
173  impl <<" {";
174  genConstructorBody(impl, c);
175  impl <<"}\n";
176  return true;
177 }
