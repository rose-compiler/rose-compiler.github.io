57 bool
58 SgAsmElfFileHeader::isElf(SgAsmGenericFile *file)
59 {
60  /* Turn off byte reference tracking for the duration of this function. We don't want our testing the file contents to
61  * affect the list of bytes that we've already referenced or which we might reference later. */
62  bool was_tracking = file->get_trackingReferences();
63  file->set_trackingReferences(false);
64 
65  try {
66  unsigned char magic[4];
67  file->readContent(0, magic, sizeof magic);
68  if (0x7f!=magic[0] || 'E'!=magic[1] || 'L'!=magic[2] || 'F'!=magic[3])
69  throw 1;
70  } catch (...) {
71  file->set_trackingReferences(was_tracking);
72  return false;
73  }
74  file->set_trackingReferences(was_tracking);
75  return true;
76 }

137 SgAsmElfFileHeader*
138 SgAsmElfFileHeader::parse()
139 {
140  // The ELF File Header should have been constructed such that SgAsmElfFileHeader::ctor() was called.
141  SgAsmGenericHeader::parse();
142 
143  /* Read 32-bit header for now. Might need to re-read as 64-bit later. */
144  Elf32FileHeader_disk disk32;
145  if (sizeof(disk32)>get_size())
146  extend(sizeof(disk32)-get_size());
147  readContentLocal(0, &disk32, sizeof disk32, false); /*zero pad if we read EOF*/
148 
149  /* Check magic number early */
150  if (disk32.e_ident_magic[0]!=0x7f || disk32.e_ident_magic[1]!='E' ||
151  disk32.e_ident_magic[2]!='L' || disk32.e_ident_magic[3]!='F')
152  throw FormatError("Bad ELF magic number");
153 
154  /* File byte order should be 1 or 2. However, we've seen at least one example that left the byte order at zero, implying
155  * that it was the native order. We don't have the luxury of decoding the file on the native machine, so in that case we
156  * try to infer the byte order by looking at one of the other multi-byte fields of the file. */
157  Rose::BinaryAnalysis::ByteOrder::Endianness sex;
158  if (1 == disk32.e_ident_data_encoding) {
159  sex = Rose::BinaryAnalysis::ByteOrder::ORDER_LSB;
160  } else if (2==disk32.e_ident_data_encoding) {
161  sex = Rose::BinaryAnalysis::ByteOrder::ORDER_MSB;
162  } else if ((disk32.e_type & 0xff00)==0xff00) {
163  /* One of the 0xffxx processor-specific flags in native order */
164  if ((disk32.e_type & 0x00ff)==0xff)
165  throw FormatError("invalid ELF header byte order"); /*ambiguous*/
166  sex = Rose::BinaryAnalysis::ByteOrder::hostOrder();
167  } else if ((disk32.e_type & 0x00ff)==0x00ff) {
168  /* One of the 0xffxx processor specific orders in reverse native order */
169  sex = Rose::BinaryAnalysis::ByteOrder::hostOrder()==Rose::BinaryAnalysis::ByteOrder::ORDER_LSB ? Rose::BinaryAnalysis::ByteOrder::ORDER_MSB : Rose::BinaryAnalysis::ByteOrder::ORDER_LSB;
170  } else if ((disk32.e_type & 0xff00)==0) {
171  /* One of the low-valued file types in native order */
172  if ((disk32.e_type & 0x00ff)==0)
173  throw FormatError("invalid ELF header byte order"); /*ambiguous*/
174  sex = Rose::BinaryAnalysis::ByteOrder::hostOrder();
175  } else if ((disk32.e_type & 0x00ff)==0) {
176  /* One of the low-valued file types in reverse native order */
177  sex = Rose::BinaryAnalysis::ByteOrder::hostOrder() == Rose::BinaryAnalysis::ByteOrder::ORDER_LSB ? Rose::BinaryAnalysis::ByteOrder::ORDER_MSB : Rose::BinaryAnalysis::ByteOrder::ORDER_LSB;
178  } else {
179  /* Ambiguous order */
180  throw FormatError("invalid ELF header byte order");
181  }
182  ROSE_ASSERT(get_executableFormat() != NULL);
183  get_executableFormat()->set_sex(sex);
184  p_e_ident_data_encoding = disk32.e_ident_data_encoding; /*save original value*/
185 
186  /* Decode header to native format */
187  Rose::BinaryAnalysis::RelativeVirtualAddress entry_rva, sectab_rva, segtab_rva;
188  if (1 == disk32.e_ident_file_class) {
189  get_executableFormat()->set_wordSize(4);
190 
191  p_e_ident_padding.clear();
192  for (size_t i=0; i<sizeof(disk32.e_ident_padding); i++)
193  p_e_ident_padding.push_back(disk32.e_ident_padding[i]);
194 
195  p_e_ident_file_class = Rose::BinaryAnalysis::ByteOrder::diskToHost(sex, disk32.e_ident_file_class);
196  p_e_ident_file_version = Rose::BinaryAnalysis::ByteOrder::diskToHost(sex, disk32.e_ident_file_version);
197  p_e_type = Rose::BinaryAnalysis::ByteOrder::diskToHost(sex, disk32.e_type);
198  p_e_machine = Rose::BinaryAnalysis::ByteOrder::diskToHost(sex, disk32.e_machine);
199  get_executableFormat()->set_version(Rose::BinaryAnalysis::ByteOrder::diskToHost(sex, disk32.e_version));
200  entry_rva = Rose::BinaryAnalysis::ByteOrder::diskToHost(sex, disk32.e_entry);
201  segtab_rva = Rose::BinaryAnalysis::ByteOrder::diskToHost(sex, disk32.e_phoff);
202  sectab_rva = Rose::BinaryAnalysis::ByteOrder::diskToHost(sex, disk32.e_shoff);
203  p_e_flags = Rose::BinaryAnalysis::ByteOrder::diskToHost(sex, disk32.e_flags);
204  p_e_ehsize = Rose::BinaryAnalysis::ByteOrder::diskToHost(sex, disk32.e_ehsize);
205 
206  p_e_phnum = Rose::BinaryAnalysis::ByteOrder::diskToHost(sex, disk32.e_phnum);
207  if (p_e_phnum>0) {
208  p_phextrasz = Rose::BinaryAnalysis::ByteOrder::diskToHost(sex, disk32.e_phentsize);
209  ROSE_ASSERT(p_phextrasz>=sizeof(SgAsmElfSegmentTableEntry::Elf32SegmentTableEntry_disk));
210  p_phextrasz -= sizeof(SgAsmElfSegmentTableEntry::Elf32SegmentTableEntry_disk);
211  } else {
212  p_phextrasz = 0;
213  }
214 
215  p_e_shnum = Rose::BinaryAnalysis::ByteOrder::diskToHost(sex, disk32.e_shnum);
216  if (p_e_shnum>0) {
217  p_shextrasz = Rose::BinaryAnalysis::ByteOrder::diskToHost(sex, disk32.e_shentsize);
218  ROSE_ASSERT(p_shextrasz>=sizeof(SgAsmElfSectionTableEntry::Elf32SectionTableEntry_disk));
219  p_shextrasz -= sizeof(SgAsmElfSectionTableEntry::Elf32SectionTableEntry_disk);
220  } else {
221  p_shextrasz = 0;
222  }
223 
224  p_e_shstrndx = Rose::BinaryAnalysis::ByteOrder::diskToHost(sex, disk32.e_shstrndx);
225  } else if (2 == disk32.e_ident_file_class) {
226  /* We guessed wrong. This is a 64-bit header, not 32-bit. */
227  get_executableFormat()->set_wordSize(8);
228  Elf64FileHeader_disk disk64;
229  if (sizeof(disk64)>get_size())
230  extend(sizeof(disk64)-get_size());
231  readContentLocal(0, &disk64, sizeof disk64, false); /*zero pad at EOF*/
232 
233  p_e_ident_padding.clear();
234  for (size_t i=0; i<sizeof(disk64.e_ident_padding); i++)
235  p_e_ident_padding.push_back(disk64.e_ident_padding[i]);
236 
237  p_e_ident_file_class = Rose::BinaryAnalysis::ByteOrder::diskToHost(sex, disk64.e_ident_file_class);
238  p_e_ident_file_version = Rose::BinaryAnalysis::ByteOrder::diskToHost(sex, disk64.e_ident_file_version);
239  p_e_type = Rose::BinaryAnalysis::ByteOrder::diskToHost(sex, disk64.e_type);
240  p_e_machine = Rose::BinaryAnalysis::ByteOrder::diskToHost(sex, disk64.e_machine);
241  get_executableFormat()->set_version(Rose::BinaryAnalysis::ByteOrder::diskToHost(sex, disk64.e_version));
242  entry_rva = Rose::BinaryAnalysis::ByteOrder::diskToHost(sex, disk64.e_entry);
243  segtab_rva = Rose::BinaryAnalysis::ByteOrder::diskToHost(sex, disk64.e_phoff);
244  sectab_rva = Rose::BinaryAnalysis::ByteOrder::diskToHost(sex, disk64.e_shoff);
245  p_e_flags = Rose::BinaryAnalysis::ByteOrder::diskToHost(sex, disk64.e_flags);
246  p_e_ehsize = Rose::BinaryAnalysis::ByteOrder::diskToHost(sex, disk64.e_ehsize);
247 
248  p_e_phnum = Rose::BinaryAnalysis::ByteOrder::diskToHost(sex, disk64.e_phnum);
249  if (p_e_phnum>0) {
250  p_phextrasz = Rose::BinaryAnalysis::ByteOrder::diskToHost(sex, disk64.e_phentsize);
251  ROSE_ASSERT(p_phextrasz>=sizeof(SgAsmElfSegmentTableEntry::Elf64SegmentTableEntry_disk));
252  p_phextrasz -= sizeof(SgAsmElfSegmentTableEntry::Elf64SegmentTableEntry_disk);
253  } else {
254  p_phextrasz = 0;
255  }
256 
257  p_e_shnum = Rose::BinaryAnalysis::ByteOrder::diskToHost(sex, disk64.e_shnum);
258  if (p_e_shnum>0) {
259  p_shextrasz = Rose::BinaryAnalysis::ByteOrder::diskToHost(sex, disk64.e_shentsize);
260  ROSE_ASSERT(p_shextrasz>=sizeof(SgAsmElfSectionTableEntry::Elf64SectionTableEntry_disk));
261  p_shextrasz -= sizeof(SgAsmElfSectionTableEntry::Elf64SectionTableEntry_disk);
262  } else {
263  p_shextrasz = 0;
264  }
265 
266  p_e_shstrndx = Rose::BinaryAnalysis::ByteOrder::diskToHost(sex, disk64.e_shstrndx);
267  } else {
268  throw FormatError("invalid ELF header file class");
269  }
270 
271  /* Magic number. disk32 and disk64 have header bytes at same offset */
272  p_magic.clear();
273  for (size_t i=0; i<sizeof(disk32.e_ident_magic); i++)
274  p_magic.push_back(disk32.e_ident_magic[i]);
275 
276  /* File format */
277  get_executableFormat()->set_family(FAMILY_ELF);
278  switch (p_e_type) {
279  case 0:
280  get_executableFormat()->set_purpose(PURPOSE_UNSPECIFIED);
281  break;
282  case 1:
283  case 3:
284  get_executableFormat()->set_purpose(PURPOSE_LIBRARY);
285  break;
286  case 2:
287  get_executableFormat()->set_purpose(PURPOSE_EXECUTABLE);
288  break;
289  case 4:
290  get_executableFormat()->set_purpose(PURPOSE_CORE_DUMP);
291  break;
292  default:
293  if (p_e_type >= 0xff00 && p_e_type <= 0xffff) {
294  get_executableFormat()->set_purpose(PURPOSE_PROC_SPECIFIC);
295  } else {
296  get_executableFormat()->set_purpose(PURPOSE_OTHER);
297  }
298  break;
299  }
300  get_executableFormat()->set_isCurrentVersion(1 == get_executableFormat()->get_version());
301  get_executableFormat()->set_abi(ABI_UNSPECIFIED); /* ELF specifies a target architecture rather than an ABI */
302  get_executableFormat()->set_abiVersion(0);
303 
304  /* Target architecture */
305  set_isa(machineToIsa(p_e_machine));
306 
307  /* Read the optional section and segment tables and the sections to which they point. An empty section or segment table is
308  * treated as if it doesn't exist. This seems to be compatible with the loader since the 45-bit "tiny" ELF executable
309  * stores a zero in the e_shnum member and a completely invalid value in the e_shoff member. */
310  if (sectab_rva>0 && get_e_shnum()>0) {
311  SgAsmElfSectionTable *tab = new SgAsmElfSectionTable(this);
312  tab->set_offset(sectab_rva.rva());
313  tab->parse();
314  }
315  if (segtab_rva>0 && get_e_phnum()>0) {
316  SgAsmElfSegmentTable *tab = new SgAsmElfSegmentTable(this);
317  tab->set_offset(segtab_rva.rva());
318  tab->parse();
319  }
320 
321  /* Associate the entry point with a particular section. */
322  entry_rva.bindBestSection(this);
323  addEntryRva(entry_rva);
324 
325  return this;
326 }

387  Rose::BinaryAnalysis::ByteOrder::hostToDisk(sex, p_e_ident_data_encoding, &(disk->e_ident_data_encoding));
388  Rose::BinaryAnalysis::ByteOrder::hostToDisk(sex, p_e_ident_file_version, &(disk->e_ident_file_version));
389  ROSE_ASSERT(p_e_ident_padding.size() == ROSE_NELMTS(disk->e_ident_padding));
390  for (size_t i=0; i<ROSE_NELMTS(disk->e_ident_padding); i++)
391  disk->e_ident_padding[i] = p_e_ident_padding[i];
392  Rose::BinaryAnalysis::ByteOrder::hostToDisk(sex, p_e_type, &(disk->e_type));
393  Rose::BinaryAnalysis::ByteOrder::hostToDisk(sex, p_e_machine, &(disk->e_machine));
394  Rose::BinaryAnalysis::ByteOrder::hostToDisk(sex, get_executableFormat()->get_version(), &(disk->e_version));
395  Rose::BinaryAnalysis::ByteOrder::hostToDisk(sex, get_entryRva(), &(disk->e_entry));
396  if (get_segmentTable()) {
397  Rose::BinaryAnalysis::ByteOrder::hostToDisk(sex, get_segmentTable()->get_offset(), &(disk->e_phoff));
398  } else {
399  Rose::BinaryAnalysis::ByteOrder::hostToDisk(sex, 0, &(disk->e_phoff));
400  }
401  if (get_sectionTable()) {
402  Rose::BinaryAnalysis::ByteOrder::hostToDisk(sex, get_sectionTable()->get_offset(), &(disk->e_shoff));
403  } else {
404  Rose::BinaryAnalysis::ByteOrder::hostToDisk(sex, 0, &(disk->e_shoff));
405  }
406  Rose::BinaryAnalysis::ByteOrder::hostToDisk(sex, p_e_flags, &(disk->e_flags));
407  Rose::BinaryAnalysis::ByteOrder::hostToDisk(sex, p_e_ehsize, &(disk->e_ehsize));
408 
409  if (p_e_phnum>0) {

411  &(disk->e_phentsize));
412  } else {
413  Rose::BinaryAnalysis::ByteOrder::hostToDisk(sex, 0, &(disk->e_phentsize));
414  }

418  } else {
419  Rose::BinaryAnalysis::ByteOrder::hostToDisk(sex, 0, &(disk->e_shentsize));
420  }
421  Rose::BinaryAnalysis::ByteOrder::hostToDisk(sex, p_e_phnum, &(disk->e_phnum));
422  Rose::BinaryAnalysis::ByteOrder::hostToDisk(sex, p_e_shnum, &(disk->e_shnum));
423  Rose::BinaryAnalysis::ByteOrder::hostToDisk(sex, p_e_shstrndx, &(disk->e_shstrndx));

466 bool
467 SgAsmElfFileHeader::reallocate()
468 {
469  /* Reallocate superclass. This also calls reallocate() for all the sections associated with this ELF File Header. */
470  bool reallocated = SgAsmGenericHeader::reallocate();
471 
472  /* Resize header based on current word size */
473  Rose::BinaryAnalysis::Address need;
474  if (4==get_wordSize()) {
475  need = sizeof(Elf32FileHeader_disk);
476  } else if (8==get_wordSize()) {
477  need = sizeof(Elf64FileHeader_disk);
478  } else {
479  throw FormatError("unsupported ELF word size");
480  }
481  if (need < get_size()) {
482  if (isMapped()) {
483  ROSE_ASSERT(get_mappedSize()==get_size());
484  set_mappedSize(need);
485  }
486  set_size(need);
487  reallocated = true;
488  } else if (need > get_size()) {
489  get_file()->shiftExtend(this, 0, need-get_size(), SgAsmGenericFile::ADDRSP_ALL, SgAsmGenericFile::ELASTIC_HOLE);
490  reallocated = true;
491  }
492 
493  /* Update ELF-specific file class data member from generic data. */
494  switch(get_wordSize()) {
495  case 4:
496  p_e_ident_file_class = 1;
497  break;
498  case 8:
499  p_e_ident_file_class = 2;
500  break;
501  default:
502  ROSE_ASSERT(!"invalid word size");
503  break;
504  }
505 
506  /* Byte order. According to the spec, valid values are 1 (little-endian) and 2 (big-endian). However, we've seen cases
507  * where a value of zero is used to indicate "native" order (loader assumes words are in the order of the machine on which
508  * the loader is running, and the ROSE ELF parser determines the order by looking at other fields in the header). Any
509  * original value other than 1 or 2 will be written to the new output; otherwise we choose 1 or 2 based on the currently
510  * defined byte order. */
511  if (p_e_ident_data_encoding==1 || p_e_ident_data_encoding==2) {
512  p_e_ident_data_encoding = Rose::BinaryAnalysis::ByteOrder::ORDER_LSB==get_sex() ? 1 : 2;
513  }
514 
515  /* Update ELF-specific file type from generic data. */
516  switch (get_executableFormat()->get_purpose()) {
517  case PURPOSE_UNSPECIFIED:
518  case PURPOSE_PROC_SPECIFIC:
519  case PURPOSE_OS_SPECIFIC:
520  case PURPOSE_OTHER:
521  /* keep as is */
522  break;
523  case PURPOSE_LIBRARY:
524  if (p_e_type==1 || p_e_type==3) {
525  /* keep as is */
526  } else {
527  p_e_type = 1;
528  }
529  break;
530  case PURPOSE_EXECUTABLE:
531  p_e_type = 2;
532  break;
533  case PURPOSE_CORE_DUMP:
534  p_e_type = 4;
535  }
536 
537  /* Update ELF machine type. */
538  p_e_machine = isaToMachine(get_isa());
539 
540  /* The ELF header stores its own size */
541  p_e_ehsize = get_size();
542 
543  return reallocated;
544 }

546 void
547 SgAsmElfFileHeader::unparse(std::ostream &f) const
548 {
549  /* Write unreferenced areas back to the file before anything else. */
550  unparseHoles(f);
551 
552  /* Write the ELF segment table and segments first since they generally overlap with more specific things which may have
553  * been modified when walking the AST. (We generally don't modify segments, just the more specific sections.) */
554  if (get_segmentTable()) {
555  ROSE_ASSERT(get_segmentTable()->get_header()==this);
556  get_segmentTable()->unparse(f);
557  }
558 
559  /* Write the ELF section table and, indirectly, the sections themselves. */
560  if (get_sectionTable()) {
561  ROSE_ASSERT(get_sectionTable()->get_header()==this);
562  get_sectionTable()->unparse(f);
563  }
564 
565  /* Encode and write the ELF file header */
566  Elf32FileHeader_disk disk32;
567  Elf64FileHeader_disk disk64;
568  void *disk = NULL;
569  size_t struct_size = 0;
570  if (4 == get_wordSize()) {
571  disk = encode(get_sex(), &disk32);
572  struct_size = sizeof(disk32);
573  } else if (8 == get_wordSize()) {
574  disk = encode(get_sex(), &disk64);
575  struct_size = sizeof(disk64);
576  } else {
577  ROSE_ASSERT(!"unsupported word size");
578  }
579  write(f, 0, struct_size, disk);
580 }
