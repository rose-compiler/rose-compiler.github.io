70 void
71 SgAsmElfSymbol::parse_common()
72 {
73  /* Binding */
74  switch (get_elfBinding()) {
75  case STB_LOCAL: p_binding = SYM_LOCAL; break;
76  case STB_GLOBAL: p_binding = SYM_GLOBAL; break;
77  case STB_WEAK: p_binding = SYM_WEAK; break;
78  default: p_binding = SYM_NO_BINDING; break;
79  }
80 
81  /* Type */
82  switch (get_elfType()) {
83  case STT_NOTYPE: p_type = SYM_NO_TYPE; break;
84  case STT_OBJECT: p_type = SYM_DATA; break;
85  case STT_FUNC: p_type = SYM_FUNC; break;
86  case STT_SECTION: p_type = SYM_SECTION; break;
87  case STT_FILE: p_type = SYM_FILE; break;
88  case STT_COMMON: p_type = SYM_COMMON; break;
89  case STT_TLS: p_type = SYM_TLS; break;
90  case STT_IFUNC: p_type = SYM_IFUNC; break;
91  default: p_type = SYM_NO_TYPE; break;
92  }
93 
94  /* Definition state */
95  if (p_value || p_size) {
96  p_definitionState = SYM_DEFINED;
97  } else if (p_name->get_string().size() > 0 || get_elfType()) {
98  p_definitionState = SYM_TENTATIVE;
99  } else {
100  p_definitionState = SYM_UNDEFINED;
101  }
102 }

230 SgAsmElfSymbolSection *
231 SgAsmElfSymbolSection::parse()
232 {
233  SgAsmElfSection::parse();
234 
235  SgAsmElfFileHeader *fhdr = get_elfHeader();
236  ROSE_ASSERT(fhdr!=NULL);
237  SgAsmElfSectionTableEntry *shdr = get_sectionEntry();
238  ASSERT_always_require(shdr!=NULL);
239 
240  size_t entry_size, struct_size, extra_size, nentries;
241  calculateSizes(&entry_size, &struct_size, &extra_size, &nentries);
242  ROSE_ASSERT(entry_size==shdr->get_sh_entsize());
243 
244  /* Parse each entry */
245  for (size_t i=0; i<nentries; i++) {
246  SgAsmElfSymbol *entry=0;
247  if (4==fhdr->get_wordSize()) {
248  entry = new SgAsmElfSymbol(this); /*adds symbol to this symbol table*/
249  SgAsmElfSymbol::Elf32SymbolEntry_disk disk;
250  readContentLocal(i*entry_size, &disk, struct_size);
251  entry->parse(fhdr->get_sex(), &disk);
252  } else if (8==fhdr->get_wordSize()) {
253  entry = new SgAsmElfSymbol(this); /*adds symbol to this symbol table*/
254  SgAsmElfSymbol::Elf64SymbolEntry_disk disk;
255  readContentLocal(i*entry_size, &disk, struct_size);
256  entry->parse(fhdr->get_sex(), &disk);
257  } else {
258  throw FormatError("unsupported ELF word size");
259  }
260  if (extra_size>0)
261  entry->get_extra() = readContentLocalUcl(i*entry_size+struct_size, extra_size);
262  }
263  return this;
264 }

334 void
335 SgAsmElfSymbolSection::unparse(std::ostream &f) const
336 {
337  SgAsmElfFileHeader *fhdr = get_elfHeader();
338  ROSE_ASSERT(fhdr);
339  Rose::BinaryAnalysis::ByteOrder::Endianness sex = fhdr->get_sex();
340 
341  size_t entry_size, struct_size, extra_size, nentries;
342  calculateSizes(&entry_size, &struct_size, &extra_size, &nentries);
343 
344  /* Adjust the entry size stored in the ELF Section Table */
345  get_sectionEntry()->set_sh_entsize(entry_size);
346 
347  /* Write each entry's required part followed by the optional part */
348  for (size_t i=0; i<nentries; i++) {
349  SgAsmElfSymbol::Elf32SymbolEntry_disk disk32;
350  SgAsmElfSymbol::Elf64SymbolEntry_disk disk64;
351  void *disk=NULL;
352 
353  SgAsmElfSymbol *entry = p_symbols->get_symbols()[i];
354 
355  if (4==fhdr->get_wordSize()) {
356  disk = entry->encode(sex, &disk32);
357  } else if (8==fhdr->get_wordSize()) {
358  disk = entry->encode(sex, &disk64);
359  } else {
360  ROSE_ASSERT(!"unsupported word size");
361  }
362 
363  Rose::BinaryAnalysis::Address spos = i * entry_size;
364  spos = write(f, spos, struct_size, disk);
365  if (entry->get_extra().size()>0) {
366  ROSE_ASSERT(entry->get_extra().size()<=extra_size);
367  write(f, spos, entry->get_extra());
368  }
369  }
370 
371  unparseHoles(f);
372 }
