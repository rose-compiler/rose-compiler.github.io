95 class KeywordsHLItem: public HLItem{
96 public:
97  KeywordsHLItem( const char** keywords, int state, int ide_state, int context, bool finalize = true, bool ignoreCase = false )
98  : HLItem( state, context ), m_ok(false), m_state(state), m_ide_state(ide_state), m_finalize(finalize), m_ignoreCase(ignoreCase) {
99  int i = 1;
100  if ( ignoreCase ) {
101  while( *keywords )
102  m_keywords.insert( QString(*keywords++).lower(), i++ );
103  } else {
104  while( *keywords )
105  m_keywords.insert( QString(*keywords++), i++ );
106  }
107  }
108 
109  KeywordsHLItem( const QMap<QString, int> keywords, int state, int ide_state, int context, bool finalize = true, bool ignoreCase = false )
110  : HLItem( state, context ), m_ok(false), m_state(state), m_ide_state(ide_state), m_finalize(finalize), m_ignoreCase(ignoreCase) {
111  m_keywords = keywords;
112  }
113 
114  int attr() const { return m_ok ? m_state : m_ide_state; }
115 
116  int checkHL( const QChar* buffer, int pos, int length, int*, int* ){
117  //kdDebug(9032) << "KeywordsHLItem::checkHLItem" << endl;
118 
119  int start_pos = pos;
120 
121  while( (pos<length) && (buffer[pos].isLetterOrNumber() || buffer[pos] == '_') )
122  ++pos;
123 
124  if( start_pos != pos ) {
125  if ( m_ignoreCase ) {
126  m_ok = m_keywords.contains( QString(buffer+start_pos, pos-start_pos).lower() );
127  } else {
128  m_ok = m_keywords.contains( QString(buffer+start_pos, pos-start_pos) );
129  }
130  }
131 
132  return ( m_ok || m_finalize ) ? pos : start_pos;
133  }
134 
135 private:
136  QMap<QString, int> m_keywords;
137  bool m_ok;
138  int m_state;
139  int m_ide_state;
140  bool m_finalize;
141  bool m_ignoreCase;
