184 //-----------------------------------------------------------------------------
186 template<class VertexType, class EdgeType>
187 void stlGraph<VertexType, EdgeType>::cyclesAndConnectivity() {
188  int componentNumber = 0;
189  cycles = 0;
190 
191  // vertices to be visited
192  stack<int, vector<int> > verticesStack;
193 
194  // prevent multiple visits, assign not visited to all vertices
195  vector<vertexStatus> vertexState(size(), notVisited);
196 
197  // try to reach all other vertices starting from one
198  for( size_t i=0; i<size(); ++i) {
199  if(vertexState[i] == notVisited) {
200  componentNumber++;
201  // store on stack for further processing
202  verticesStack.push(i);
203 
204  // process stack
205  while(!verticesStack.empty()) {
206  int vert = verticesStack.top();
207  verticesStack.pop();
208  if(vertexState[vert] == visited) {
209  vertexState[vert] = processed;
210  } else {
211 
212  if(vertexState[vert] == notVisited) {
213  vertexState[vert] = visited;
214 
215  // new vertex, earmark for processed mark
216  verticesStack.push(vert);
217 
218  // if one of the new vertices has the processed flag we found a cycle
219  // earmark successors
220  Successor::iterator start = operator[](vert).second.begin();
221  Successor::iterator end = operator[](vert).second.end();
222  while(start != end) {
223  int succ = (*start).first;
224  if(vertexState[succ] == visited) {
225  // a cycle...
226  ++cycles;
227  //(*pOut) << " at least vertex " << operator[](succ).first << " lies in a cycle \n";
228  }
229 
230  // otherwise vertex was already processed or wasnt visited and is earmarked on the stack
231  if(vertexState[succ] == notVisited)
232  verticesStack.push(succ);
233 
234  ++start;
235  }
236  } // not visited
237 
238  }
239  } // stack not empty
240 
241  } // not visited
242  }
243 
244  if(isDirected()) {
245  if(componentNumber == 1) connectivity = strong;
246  else connectivity = weak;
247  } else {
248  connectivity = unknown;
249  }
250  components = componentNumber;
251 }
