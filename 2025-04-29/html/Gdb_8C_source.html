132  ASSERT_require(isAttached());
133  resetResponses();
134 
135  // Read asynchronous responses that are already waiting
136  while (true) {
137  const auto list = readOptionalResponses();
138  if (!list.empty()) {
139  if (mlog[DEBUG]) {
140  for (const GdbResponse &response: list)
141  mlog[DEBUG] <<response;
142  }
143  } else {
144  break;
145  }
146  }
147

166  }

184 std::list<GdbResponse>
185 Gdb::readResponseSet(bool required) {
186  std::list<GdbResponse> responses;
187 
188  // Get the first line of the response, which might be optional
189  Sawyer::Optional<std::string> line = required ? gdbOutput_.blockingNext() : gdbOutput_.nonblockingNext();
190  if (!line)
191  return responses;
192  responses.push_back(GdbResponse::parse(*line));
193  if (responses.back().atEnd)
194  return responses;
195 
196  // Since we had a first line of a response set, then keep reading until we get to the last line.
197  while ((line = gdbOutput_.blockingNext())) {
198  responses.push_back(GdbResponse::parse(*line));
199  if (responses.back().atEnd)
200  break;
201  }
202 
203  if (mlog[TRACE]) {
204  for (const GdbResponse &response: responses) {
205  if (!response.console.empty())
206  mlog[TRACE] <<response.console <<"\n";
207  }
208  }
209 
210  // Accumulate responses
211  responses_.insert(responses_.end(), responses.begin(), responses.end());
212 
213  return responses;
214 }

256  argv.push_back(boost::process::search_path(specimen.gdbName()).string());

270  readRequiredResponses();
271 
272  // Attach to remote target
273  sendCommand("-target-select remote " + specimen.remote().host + ":" + boost::lexical_cast<std::string>(specimen.remote().port));
274 
275  // Get an appropriate disassembler for this architecture
276  for (const GdbResponse &response: responses()) {
277  if (GdbResponse::AsyncClass::STOPPED == response.exec.aclass) {
278  if (const Yaml::Node &node = response.exec.results["frame"]["arch"]) {
279  const std::string arch = node.as<std::string>();
280  if ("m68k" == arch) {
281  disassembler_ = Architecture::findByName("nxp-coldfire").orThrow()->newInstructionDecoder();
282  } else if ("m68k:isa-a:nodiv" == arch) {
283  disassembler_ = Architecture::findByName("nxp-coldfire").orThrow()->newInstructionDecoder();
284  } else {
285  ASSERT_not_implemented("unrecognized architecture: " + arch);
286  }
287  }
288  }
289  }
290  ASSERT_not_null(disassembler_);
291 
292  // Get the list of register names for this architecture.
293  for (const GdbResponse &response: sendCommand("-data-list-register-names")) {
294  if (GdbResponse::ResultClass::DONE == response.result.rclass) {
295  if (Yaml::Node names = response.result.results["register-names"]) {
296  ASSERT_require(names.isSequence());
297  for (const auto &pair: names) {
298  const std::string name = pair.second.as<std::string>();

300  registers_.push_back(std::make_pair(name, reg));

303  <<" is not present in dictionary \"" <<registerDictionary()->name() <<"\"\n";
304  }
305  }
306  break;
307  }
308  }
309  }
310  ASSERT_forbid(registers_.empty());

356  if (!isAttached())
357  throw Exception("not attached to subordinate process");
358 
359  if (where.isSingleton() && gdbHandlesBreakPoints()) {
360  // We're adding a break point for a single address, and GDB is currently handling all the break points (if
361  // any). Therefore, make GDB handle this break point too.
362  if (gdbBreakPoints_.find(where.least()) == gdbBreakPoints_.end()) {

369  }
370  }
371  }
372  } else {

374  // handled by GDB. Therefore, handle the break points in ROSE.
375  for (const auto &gdbBreakPoint: gdbBreakPoints_)
376  sendCommand("-break-delete " + boost::lexical_cast<std::string>(gdbBreakPoint.second));
377  gdbBreakPoints_.clear();

491

493  // offsets[i] is the bit offset in retval for the start of register i.

496  offsets.reserve(registers_.size());

498  offsets.push_back(i > 0 ? offsets.back() + registers_[i-1].second.nBits() : 0);
499  totalBits += registers_[i].second.nBits();

501

503  Sawyer::Container::BitVector retval(totalBits);

549  // `reg` corresponds.

551  if (GdbResponse::ResultClass::DONE == response.result.rclass) {

563  return whole;
564  } else {
565  Sawyer::Container::BitVector retval(reg.nBits());

567  return retval;

578 Gdb::writeAllRegisters(ThreadId, const Sawyer::Container::BitVector &allValues) {
579  // Calculate the starting bit offset for each register, and the total number of bits needed.
580  // offsets[i] is the bit offset in retval for the start of register i

601 Gdb::writeRegister(ThreadId tid, RegisterDescriptor reg, const Sawyer::Container::BitVector &value) {
602  ASSERT_require(reg);
603  if (!isAttached())
604  throw Exception("not attached to subordinate process");
605  resetResponses();
606 
607  // Find the GDB register of which `reg` is a part.
608  Sawyer::Optional<size_t> idx = findRegisterIndex(reg);

611  const RegisterDescriptor gdbReg = registers_[*idx].second;
612  Sawyer::Container::BitVector gdbValue;

622  } else {

627  // create a variable first, then write to the variable, and finally delete the variable.
628  sendCommand("-var-create temp_reg * $" + registers_[*idx].first);
629  sendCommand("-var-assign temp_reg 0x" + gdbValue.toHex());

641 size_t
642 Gdb::readMemory(Address va, size_t nBytes, uint8_t *buffer) {
643  if (!isAttached())
644  throw Exception("not attached to subordinate process");
645  sendCommand("-data-read-memory-bytes " + boost::lexical_cast<std::string>(va) + " " + boost::lexical_cast<std::string>(nBytes));
646  for (const GdbResponse &response: responses()) {
647  if (GdbResponse::ResultClass::DONE == response.result.rclass) {
648  if (const Yaml::Node memoryList = response.result.results["memory"]) {
649  ASSERT_require(memoryList.isSequence());
650  for (const auto &elmt: memoryList) {
651  ASSERT_require(elmt.second.isMap());
652 
653  // When reading address 0xffffffff, GDB's "end" marker will overflow to zero. It would have been better if they
654  // had used a "least" and "greatest" address (akin to C++ library "front" and "back" iterators or Sawyer's
655  // "Interval" type), but they didn't. Therefore we need to handle this situation as a special case.
