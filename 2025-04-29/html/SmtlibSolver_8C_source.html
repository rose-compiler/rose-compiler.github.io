309 SmtlibSolver::outputComparisonFunctions(std::ostream &o, const std::vector<SymbolicExpression::Ptr> &exprs) {
310  struct T1: SymbolicExpression::Visitor {
311  Sawyer::Container::Map<SymbolicExpression::Operator, std::set<size_t> > widths;
312  std::ostream &o;
313 
314  T1(std::ostream &o): o(o) {}
315 
316  SymbolicExpression::VisitAction preVisit(const SymbolicExpression::Ptr &node) {
317  if (SymbolicExpression::InteriorPtr inode = node->isInteriorNode()) {
318  size_t w = inode->child(0)->nBits();
319  switch (inode->getOperator()) {
320  case SymbolicExpression::OP_ULT:
321  break; // SMT-LIB 2 defines "bvult" for this
322  case SymbolicExpression::OP_ULE:
323  if (widths.insertMaybeDefault(inode->getOperator()).insert(w).second) {
324  o <<"(define-fun bvule" <<w
325  <<" ((a (_ BitVec " <<w <<")) (b (_ BitVec " <<w <<")))"
326  <<" Bool"
327  <<" (or (bvult a b) (= a b)))\n";
328  }
329  break;
330  case SymbolicExpression::OP_UGT:
331  if (widths.insertMaybeDefault(inode->getOperator()).insert(w).second) {
332  o <<"(define-fun bvugt" <<w
333  <<" ((a (_ BitVec " <<w <<")) (b (_ BitVec " <<w <<")))"
334  <<" Bool"
335  <<" (not (bvule" <<w <<" a b)))\n";
336 
337  }
338  break;
339  case SymbolicExpression::OP_UGE:
340  if (widths.insertMaybeDefault(inode->getOperator()).insert(w).second) {
341  o <<"(define-fun bvuge" <<w
342  <<" ((a (_ BitVec " <<w <<")) (b (_ BitVec " <<w <<")))"
343  <<" Bool"
344  <<" (not (bvult a b)))\n";
345  }
346  break;
347  case SymbolicExpression::OP_SLT:
348  // signed(a) < signed(b) iff signed(b) - signed(a) > 0 (i.e., sign bit is clear)
349  if (widths.insertMaybeDefault(inode->getOperator()).insert(w).second) {
350  o <<"(define-fun bvslt" <<w
351  <<" ((a (_ BitVec " <<w <<")) (b (_ BitVec " <<w <<")))"
352  <<" Bool"
353  <<" (= #b0 ((_ extract " <<(w-1) <<" " <<(w-1) <<") (bvadd (bvneg a) b))))\n";
354  }
355  break;
356  case SymbolicExpression::OP_SLE:
357  // signed(a) <= signed(b) iff (signed(b) - signed(a) > 0 or a == b
358  if (widths.insertMaybeDefault(inode->getOperator()).insert(w).second) {
359  o <<"(define-fun bvsle" <<w
360  <<" ((a (_ BitVec " <<w <<")) (b (_ BitVec " <<w <<")))"
361  <<" Bool"
362  <<" (or (= a b) (= #b0 ((_ extract " <<(w-1) <<" " <<(w-1) <<") (bvadd (bvneg a) b)))))\n";
363  }
364  break;
365  case SymbolicExpression::OP_SGT:
366  // signed(a) > signed(b) iff (signed(a) - signed(b) > 0 (i.e., sign bit is clear)
367  if (widths.insertMaybeDefault(inode->getOperator()).insert(w).second) {
368  o <<"(define-fun bvsgt" <<w
369  <<" ((a (_ BitVec " <<w <<")) (b (_ BitVec " <<w <<")))"
370  <<" Bool"
371  <<" (= #b0 ((_ extract " <<(w-1) <<" " <<(w-1) <<") (bvadd a (bvneg b)))))\n";
372  }
373  break;
374  case SymbolicExpression::OP_SGE:
375  // signed(a) >= signed(b) iff (signed(a) - signed(b) > 0 || a == b
376  if (widths.insertMaybeDefault(inode->getOperator()).insert(w).second) {
377  o <<"(define-fun bvsge" <<w
378  <<" ((a (_ BitVec " <<w <<")) (b (_ BitVec " <<w <<")))"
379  <<" Bool"
380  <<" (or (= a b) (= #b0 ((_ extract " <<(w-1) <<" " <<(w-1) <<") (bvadd a (bvneg b))))))\n";
381  }
382  break;
383  default:
384  // not a comparison operation
385  break;
386  }
387  }
388  return SymbolicExpression::CONTINUE;
389  }

474 SmtSolver::SExprTypePair
475 SmtlibSolver::outputExpression(const SymbolicExpression::Ptr &expr) {
476  ASSERT_not_null(expr);
477  typedef std::vector<SExprTypePair> Etv;
478  SExprTypePair retval;
479 
480  SymbolicExpression::LeafPtr leaf = expr->isLeafNode();
481  SymbolicExpression::InteriorPtr inode = expr->isInteriorNode();
482 
483  // If we previously found a common subexpression, then use its name rather than re-emitting the same expression again. This
484  // can drastically reduce the size of the output file.
485  StringTypePair st;
486  if (termNames_.getOptional(expr).assignTo(st)) {
487  return SExprTypePair(SExpr::instance(st.first), st.second);
488  } else if (leaf) {
489  retval = outputLeaf(leaf);
490  } else {
491  ASSERT_not_null(inode);
492  switch (inode->getOperator()) {
493  case SymbolicExpression::OP_NONE:
494  ASSERT_not_reachable("not possible for an interior node");
495  case SymbolicExpression::OP_ADD:
496  retval = outputLeftAssoc("bvadd", inode);
497  break;
498  case SymbolicExpression::OP_AND: {
499  Etv children = outputExpressions(inode->children());
500  Type type = mostType(children);
501  children = outputCast(children, type);
502  if (BOOLEAN == type) {
503  retval = outputLeftAssoc("and", children);
504  } else {
505  ASSERT_require(BIT_VECTOR == type);
506  retval = outputLeftAssoc("bvand", children);
507  }
508  break;
509  }
510  case SymbolicExpression::OP_ASR:
511  retval = outputArithmeticShiftRight(inode);
512  break;
513  case SymbolicExpression::OP_XOR:
514  retval = outputXor(inode);
515  break;
516  case SymbolicExpression::OP_EQ:
517  retval = outputBinary("=", inode, BOOLEAN);
518  break;
519  case SymbolicExpression::OP_CONCAT: {
520  Etv children = outputCast(outputExpressions(inode->children()), BIT_VECTOR);
521  retval = outputLeftAssoc("concat", children);
522  break;
523  }
524  case SymbolicExpression::OP_CONVERT:
525  throw Exception("OP_CONVERT is not implemented yet");
526  case SymbolicExpression::OP_EXTRACT:
527  retval = outputExtract(inode);
528  break;
529  case SymbolicExpression::OP_FP_ABS:
530  case SymbolicExpression::OP_FP_NEGATE:
531  case SymbolicExpression::OP_FP_ADD:
532  case SymbolicExpression::OP_FP_MUL:
533  case SymbolicExpression::OP_FP_DIV:
534  case SymbolicExpression::OP_FP_MULADD:
535  case SymbolicExpression::OP_FP_SQRT:
536  case SymbolicExpression::OP_FP_MOD:
537  case SymbolicExpression::OP_FP_ROUND:
538  case SymbolicExpression::OP_FP_MIN:
539  case SymbolicExpression::OP_FP_MAX:
540  case SymbolicExpression::OP_FP_LE:
541  case SymbolicExpression::OP_FP_LT:
542  case SymbolicExpression::OP_FP_GE:
543  case SymbolicExpression::OP_FP_GT:
544  case SymbolicExpression::OP_FP_EQ:
545  case SymbolicExpression::OP_FP_ISNORM:
546  case SymbolicExpression::OP_FP_ISSUBNORM:
547  case SymbolicExpression::OP_FP_ISZERO:
548  case SymbolicExpression::OP_FP_ISINFINITE:
549  case SymbolicExpression::OP_FP_ISNAN:
550  case SymbolicExpression::OP_FP_ISNEG:
551  case SymbolicExpression::OP_FP_ISPOS:
552  throw Exception("floating point operations are not implemented yet");
553  case SymbolicExpression::OP_INVERT: {
554  ASSERT_require(inode->nChildren() == 1);
555  SExprTypePair child = outputExpression(inode->child(0));
556  retval = outputUnary((BOOLEAN==child.second?"not":"bvnot"), child);
557  break;
558  }
559  case SymbolicExpression::OP_ITE:
560  retval = outputIte(inode);
561  break;
562  case SymbolicExpression::OP_LET:
563  throw Exception("OP_LET not implemented");
564  case SymbolicExpression::OP_LSSB:
565  throw Exception("OP_LSSB not implemented");
566  case SymbolicExpression::OP_MSSB:
567  throw Exception("OP_MSSB not implemented");
568  case SymbolicExpression::OP_NE:
569  retval = outputNotEqual(inode);
570  break;
571  case SymbolicExpression::OP_NEGATE: {
572  ASSERT_require(inode->nChildren() == 1);
573  SExprTypePair child = outputCast(outputExpression(inode->child(0)), BIT_VECTOR);
574  retval = outputUnary("bvneg", child);
575  break;
576  }
577  case SymbolicExpression::OP_NOOP:
578  retval = outputExpression(SymbolicExpression::makeIntegerConstant(inode->nBits(), 0));
579  break;
580  case SymbolicExpression::OP_OR: {

971 // Logical right shift introducing one bits.
972 //
973 // ROSE: (shr1 AMOUNT EXPR)
974 // SMT-LIB: ((_ extract [EXPR.size-1] 0) ; the low-order half
975 // (bvlshr
976 // (concat ONES EXPR) ; thing to shift, ONES in high half, EXPR in low half
977 // EXTENDED_AMOUNT))
978 //
979 // SMT-LIB doesn't have a logical right-shift operation that introduces ones (only zeros), so we do it the hard way. We double
980 // the width of the value to be shifted by concatenating one bits in the high half. Then we shift the double-width value and
981 // return just the low half.
982 //
983 // The EXTENDED_AMOUNT is the original AMOUNT zero extended to be twice the width of the original EXPR (and the same width as
984 // the value that SMT-LIB is shifting.
985 SmtSolver::SExprTypePair
986 SmtlibSolver::outputLogicalShiftRight1(const SymbolicExpression::InteriorPtr &inode) {
987  ASSERT_not_null(inode);
988  ASSERT_require(inode->getOperator() == SymbolicExpression::OP_SHR1);
989  ASSERT_require(inode->nChildren() == 2);
990 
991  // Original arguments
992  const SymbolicExpression::Ptr sa = inode->child(0);
993  const SymbolicExpression::Ptr expr = inode->child(1);
994 
995  // Upper half of the bits to be shifted (all ones)
996  const SExpr::Ptr upper =
997  outputCast(outputExpression(SymbolicExpression::makeIntegerConstant(Sawyer::Container::BitVector(expr->nBits(), true))),
998  BIT_VECTOR).first;
999 
1000  // Lower half of the bits to be shifted
1001  const SExpr::Ptr lower = outputCast(outputExpression(expr), BIT_VECTOR).first;
1002 
1003  // The thing to be shifted
1004  const SExpr::Ptr shiftee = SExpr::instance(SExpr::instance("concat"), upper, lower);
1005 
1006  // The shift amount zero extended to the same width as the shiftee (twice the width of the original expr)
1007  const SExpr::Ptr extended_sa =
1008  outputCast(outputExpression(SymbolicExpression::makeExtend(SymbolicExpression::makeIntegerConstant(32, 2*expr->nBits()), sa)),
1009  BIT_VECTOR).first;
1010 
1011  // Shift the double-wide shiftee to the right using SMT-LIB's bvlshr operator that introduces zeros at the high end.
1012  const SExpr::Ptr shifted = SExpr::instance(SExpr::instance("bvlshr"), shiftee, extended_sa);
1013 
1014  // Extract just the low half of the shifted result.
1015  const SExpr::Ptr retval = SExpr::instance(SExpr::instance(SExpr::instance("_"),
1016  SExpr::instance("extract"),
1017  SExpr::instance(expr->nBits() - 1),
1018  SExpr::instance(0)),
1019  shifted);

1021 }

1053 // SMT-LIB: ((_ extract [2*EXPR.size-1] [EXPR.size]) ; the upper half
1054 // (bvshl
1055 // (concat EXPR ONES) ; i.e., the original value padded with zeros or ones in the low bits
1056 // EXTENDED_AMOUNT))
1057 //
1058 // Where EXTENDED_AMOUNT is the original AMOUNT extended to be the same width as either EXPR (in the shl0 case) or twice the
1059 // width of EXPR (in the shl1 case).
1060 //
1061 // What we're doing is concatenating EXPR|ONES to make a value that's twice as wide as EXPR with the EXPR part in the
1062 // high-order half. Then we shift left, introducing zeros (it doesn't matter that they're zeros, but that's all SMT-LIB can
1063 // do), and then we return the upper half of the bits which are the original EXPR bits and some of the ONES bits.

1071  const SymbolicExpression::Ptr expr = inode->child(1);

1074  const SExpr::Ptr upper = outputCast(outputExpression(expr), BIT_VECTOR).first;
1075

1077  const SExpr::Ptr lower =
1078  outputCast(outputExpression(SymbolicExpression::makeIntegerConstant(Sawyer::Container::BitVector(expr->nBits(), true))),
1079  BIT_VECTOR).first;
1080 
1081  // The thing to be shifted, twice as wide as the original ROSE argument, consisting of the original ROSE argument in the
1082  // high bits, and all ones in the low bits.
1083  const SExpr::Ptr shiftee = SExpr::instance(SExpr::instance("concat"), upper, lower);
1084 
1085  // The shift amount extended to the same width as the thing to be shifted since SMT-LIB requires that both arguments are
1086  // the same type.
1087  const SExpr::Ptr extended_sa =
1088  outputCast(outputExpression(SymbolicExpression::makeExtend(SymbolicExpression::makeIntegerConstant(32, 2*expr->nBits()), sa)),
1089  BIT_VECTOR).first;
1090 
1091  // Shift the double-wide value to the left using SMT-LIB's bvshl operator that introduces zeros at the low end.
1092  const SExpr::Ptr shifted = SExpr::instance(SExpr::instance("bvshl"), shiftee, extended_sa);

1095  const SExpr::Ptr retval = SExpr::instance(SExpr::instance(SExpr::instance("_"),
1096  SExpr::instance("extract"),
1097  SExpr::instance(2*expr->nBits() - 1),
1098  SExpr::instance(expr->nBits())),
