79  regs->resize("cr1", 64);
80  regs->resize("cr2", 64);

183 Amd64::cc_sysv() const {
184  RegisterDictionary::Ptr regdict = registerDictionary();
185  auto cc = CC::Definition::instance("sysv", "x86-64 sysv", constPtr());
186  const RegisterDescriptor SP = regdict->stackPointerRegister();
187 
188  //==== Address locations ====
189  cc->instructionPointerRegister(regdict->instructionPointerRegister());
190  cc->returnAddressLocation(ConcreteLocation(SP, 0, regdict));
191 
192  //==== Stack characteristics ====
193  cc->stackDirection(CC::StackDirection::GROWS_DOWN);
194  cc->nonParameterStackSize(bytesPerWord()); // return address
195 
196  //==== Function parameters ====
197 
198  // The first six integer or pointer arguments are passed in registers RDI, RSI, RDX, RCX, R8, and R9.
199  // These registers are also not preserved across the call.
200  {
201  auto pool = CC::StoragePoolEnumerated::instance("non-fp arguments", CC::isNonFpNotWiderThan(bitsPerWord()));
202  cc->argumentValueAllocator()->append(pool);
203  auto insert = [&cc, &regdict, &pool](const std::string &registerName) {
204  const RegisterDescriptor reg = regdict->findOrThrow(registerName);
205  cc->appendInputParameter(reg);
206  pool->append(ConcreteLocation(reg, regdict));
207  };
208  insert("rdi");
209  insert("rsi");
210  insert("rdx");
211  insert("rcx");
212  insert("r8");
213  insert("r9");
214  }
215 
216  // The first eight SSE arguments are passed in registers xmm0 through xmm7
217  if (const RegisterDescriptor XMM0 = regdict->findOrThrow("xmm0")) {
218  auto pool = CC::StoragePoolEnumerated::instance("fp arguments", CC::isFpNotWiderThan(XMM0.nBits()));
219  cc->argumentValueAllocator()->append(pool);
220  auto insert = [&cc, &regdict, &pool](const std::string &registerName) {
221  const RegisterDescriptor reg = regdict->findOrThrow(registerName);
222  cc->appendInputParameter(reg);
223  pool->append(ConcreteLocation(reg, regdict));
224  };
225  insert("xmm0");
226  insert("xmm1");
227  insert("xmm2");
228  insert("xmm3");
229  insert("xmm4");
230  insert("xmm5");
231  insert("xmm6");
232  insert("xmm7");
233  }
234 
235  // The AL register is an input register that stores the number of SSE registers used for variable argument calls. (It
236  // is also part of the first return value).
237  cc->appendInputParameter(regdict->findOrThrow("al")); // for varargs calls
238 
239  // Arguments that don't fit in the input registers are passed on the stack
240  cc->stackParameterOrder(CC::StackParameterOrder::RIGHT_TO_LEFT);
241  cc->stackCleanup(CC::StackCleanup::BY_CALLER);
242  {
243  auto pool = CC::StoragePoolStack::instance("args-on-stack", CC::isAnyType(), constPtr());
244  pool->initialOffset(bytesPerWord()); // return address
245  cc->argumentValueAllocator()->append(pool);
246  }
247 
248 
249  //==== Other inputs ====
250 
251  // direction flag is assuemd to be valid
252  cc->nonParameterInputs().push_back(ConcreteLocation(regdict->findOrThrow("df"), regdict));
253  // code segment register is assumed to be valid
254  cc->nonParameterInputs().push_back(ConcreteLocation(regdict->findOrThrow("cs"), regdict));
255  // data segment register is assumed to be valid
256  cc->nonParameterInputs().push_back(ConcreteLocation(regdict->findOrThrow("ds"), regdict));
257  // stack segment register is assumed to be valid
258  cc->nonParameterInputs().push_back(ConcreteLocation(regdict->findOrThrow("ss"), regdict));
259 
260  //==== Return values ====
261 
262  // Integer return values
263  if (const RegisterDescriptor AX = regdict->findOrThrow("rax")) {
264  auto pool = CC::StoragePoolEnumerated::instance("non-fp return values", CC::isNonFpNotWiderThan(bitsPerWord()));
265  cc->returnValueAllocator()->append(pool);
266 
267  pool->append(ConcreteLocation(AX, regdict));
268  cc->appendOutputParameter(AX);
269 
270  const RegisterDescriptor DX = regdict->findOrThrow("rdx");
271  pool->append(ConcreteLocation(DX, regdict));
272  cc->appendOutputParameter(DX);
273  }
274 
275  // FP return values
276  if (const RegisterDescriptor XMM0 = regdict->findOrThrow("xmm0")) {
277  auto pool = CC::StoragePoolEnumerated::instance("fp return values", CC::isFpNotWiderThan(XMM0.nBits()));
278  cc->returnValueAllocator()->append(pool);
279 
280  pool->append(ConcreteLocation(XMM0, regdict));
281  cc->appendOutputParameter(XMM0);
282 
283  const RegisterDescriptor XMM1 = regdict->findOrThrow("xmm1");
284  pool->append(ConcreteLocation(XMM1, regdict));
285  cc->appendOutputParameter(XMM1);
286  }
287 
288  // The stack pointer is an output because it is changed by the callee (incremented when the function returns). It does not hold
289  // a return value in the normal sense of the word, and thus isn't added to the return value allocator functions.
290  cc->appendOutputParameter(SP);
291

293  //cc->scratchRegisters().insert(regdict->findLargestRegister(x86_regclass_st, 1)); // dynamic st(1), overlaps mm<i+1>
294

296  // Modified, not callee-saved, not return values
297

299  cc->scratchRegisters().insert(regdict->findOrThrow("rdi"));
300  cc->scratchRegisters().insert(regdict->findOrThrow("rsi"));
301  //cc->scratchRegisters().insert(regdict->findOrThrow("rdx")); this is a return reg

304  cc->scratchRegisters().insert(regdict->findOrThrow("r9"));

308  cc->scratchRegisters().insert(regdict->findOrThrow("xmm2"));
309  cc->scratchRegisters().insert(regdict->findOrThrow("xmm3"));
310  cc->scratchRegisters().insert(regdict->findOrThrow("xmm4"));
311  cc->scratchRegisters().insert(regdict->findOrThrow("xmm5"));
312  cc->scratchRegisters().insert(regdict->findOrThrow("xmm6"));
313  cc->scratchRegisters().insert(regdict->findOrThrow("xmm7"));
314 
315  // These registers are almost always modified by a function
316  cc->scratchRegisters().insert(regdict->instructionPointerRegister());
317  cc->scratchRegisters().insert(regdict->findOrThrow("rflags"));
318  cc->scratchRegisters().insert(regdict->findOrThrow("fpstatus"));
319 
320  cc->scratchRegisters().insert(regdict->findOrThrow("r10")); //static chain ptr
321  cc->scratchRegisters().insert(regdict->findOrThrow("r11"));
322 
323  // Floating point registers are pretty much all scratch.
324  cc->scratchRegisters().insert(regdict->findOrThrow("xmm8"));
325  cc->scratchRegisters().insert(regdict->findOrThrow("xmm9"));
326  cc->scratchRegisters().insert(regdict->findOrThrow("xmm10"));
327  cc->scratchRegisters().insert(regdict->findOrThrow("xmm11"));
328  cc->scratchRegisters().insert(regdict->findOrThrow("xmm12"));
329  cc->scratchRegisters().insert(regdict->findOrThrow("xmm13"));
330  cc->scratchRegisters().insert(regdict->findOrThrow("xmm14"));
331  cc->scratchRegisters().insert(regdict->findOrThrow("xmm15"));
332 
333  cc->scratchRegisters().insert(regdict->findOrThrow("st0")); // i.e., statically mm0
334  cc->scratchRegisters().insert(regdict->findOrThrow("st1")); // mm1, etc. Since mm<i> could
335  cc->scratchRegisters().insert(regdict->findOrThrow("st2")); // overlap with st(0) or st(1),
336  cc->scratchRegisters().insert(regdict->findOrThrow("st3")); // two of these could also be
337  cc->scratchRegisters().insert(regdict->findOrThrow("st4")); // return values.
338  cc->scratchRegisters().insert(regdict->findOrThrow("st5"));
339  cc->scratchRegisters().insert(regdict->findOrThrow("st6"));
340  cc->scratchRegisters().insert(regdict->findOrThrow("st7"));
341 
342  //==== Callee-saved registers ====
343  // Everything else
344  RegisterParts regParts = regdict->getAllParts() - cc->getUsedRegisterParts();
345  std::vector<RegisterDescriptor> registers = regParts.extract(regdict);
346  cc->calleeSavedRegisters().insert(registers.begin(), registers.end());
347  return cc;
