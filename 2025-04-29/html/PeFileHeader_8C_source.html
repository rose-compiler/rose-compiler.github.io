111 bool
112 SgAsmPEFileHeader::isPe(SgAsmGenericFile *file)
113 {
114  /* Turn off byte reference tracking for the duration of this function. We don't want our testing the file contents to
115  * affect the list of bytes that we've already referenced or which we might reference later. */
116  bool was_tracking = file->get_trackingReferences();
117  file->set_trackingReferences(false);
118 
119  try {
120  /* Check DOS File Header magic number at beginning of the file */
121  unsigned char dos_magic[2];
122  file->readContent(0, dos_magic, sizeof dos_magic);
123  if ('M'!=dos_magic[0] || 'Z'!=dos_magic[1])
124  throw 1;
125 
126  /* Read four-byte offset of potential PE File Header at offset 0x3c */
127  uint32_t lfanew_disk;
128  file->readContent(0x3c, &lfanew_disk, sizeof lfanew_disk);
129  Address pe_offset = ByteOrder::leToHost(lfanew_disk);
130 
131  /* Look for the PE File Header magic number */
132  unsigned char pe_magic[4];
133  file->readContent(pe_offset, pe_magic, sizeof pe_magic);
134  if ('P'!=pe_magic[0] || 'E'!=pe_magic[1] || '\0'!=pe_magic[2] || '\0'!=pe_magic[3])
135  throw 1;
136  } catch (...) {
137  file->set_trackingReferences(was_tracking);
138  return false;
139  }
140 
141  file->set_trackingReferences(was_tracking);
142  return true;
143 }

596 /* Looks at the RVA/Size pairs in the PE header and creates an SgAsmGenericSection object for each one. This must be done
597  * after we build the mapping from virtual addresses to file offsets. */
598 void
599 SgAsmPEFileHeader::createTableSections()
600 {
601 
602  /* First, only create the sections. */
603  for (size_t i=0; i<get_rvaSizePairs()->get_pairs().size(); i++) {
604  SgAsmPERVASizePair *pair = get_rvaSizePairs()->get_pairs()[i];
605 
606  if (0==pair->get_e_size())
607  continue;
608 
609  /* Table names come from PE file specification and are hard coded by RVA/Size pair index */
610  const char *tabname_short;
611  std::string tabname = rvaSizePairName((PairPurpose)i, &tabname_short);
612 
613  /* Find the starting offset in the file.
614  * FIXME: We have a potential problem here in that ROSE sections are always contiguous in the file but a section created
615  * from an RVA/Size pair is not necessarily contiguous in the file. Normally such sections are in fact
616  * contiguous and we'll just ignore this for now. In any case, as long as these sections only ever read their
617  * data via the same MemoryMap that we use here, everything should be fine. [RPM 2009-08-17] */
618  Address pair_va = get_baseVa() + pair->get_e_rva();
619  MemoryMap::Ptr map = get_loaderMap();
620  ROSE_ASSERT(map!=NULL);
621  if (!map->baseSize(pair_va, pair->get_e_size()).exists(Sawyer::Container::MATCH_WHOLE)) {
622  mlog[WARN] <<"SgAsmPEFileHeader::createTableSections: pair-" <<i
623  <<", rva=" <<StringUtility::addrToString(pair->get_e_rva().rva())
624  <<", size=" <<StringUtility::plural(pair->get_e_size(), "bytes")
625  <<" \"" <<StringUtility::cEscape(tabname) <<"\":"
626  <<" unable to find a mapping for the virtual address (skipping)\n";
627  continue;
628  }
629  const MemoryMap::Node &me = *map->at(pair_va).findNode();
630  Address file_offset = me.value().offset() + pair_va - me.key().least();
631 
632  /* Create the new section */
633  SgAsmGenericSection *tabsec = NULL;
634  switch (i) {
635  case 0: {
636  /* Sometimes export sections are represented by a ".edata" section, and sometimes they're represented by an
637  * RVA/Size pair, sometimes both point to the same part of the file, and sometimes the RVA/Size pair points to
638  * a different part of the file. We don't want the exports duplicated in the AST, so we only create this table
639  * as exports if we haven't already seen some other export section. */
640  SgAsmGenericSectionPtrList &sections = get_sections()->get_sections();
641  bool seen_exports = false;
642  for (SgAsmGenericSectionPtrList::iterator si=sections.begin(); !seen_exports && si!=sections.end(); ++si)
643  seen_exports = isSgAsmPEExportSection(*si);
644  if (seen_exports) {
645  tabsec = new SgAsmGenericSection(get_file(), this);
646  } else {
647  tabsec = new SgAsmPEExportSection(this);
648  }
649  break;
650  }
651  case 1: {
652  /* Sometimes import sections are represented by a ".idata" section, and sometimes they're represented by an
653  * RVA/Size pair, and sometimes both point to the same part of the file. We don't want the imports duplicated
654  * in the AST, so we only create this table as imports if we haven't already seen some other import section. */
655  SgAsmGenericSectionPtrList &sections = get_sections()->get_sections();
656  bool seen_imports = false;
657  for (SgAsmGenericSectionPtrList::iterator si=sections.begin(); !seen_imports && si!=sections.end(); ++si)
658  seen_imports = isSgAsmPEImportSection(*si);
659  if (seen_imports) {
660  tabsec = new SgAsmGenericSection(get_file(), this);
661  } else {
662  tabsec = new SgAsmPEImportSection(this);
663  }
664  break;
665  }
666  case 14: {
667  /* 14==IMAGE_DIRECTORY_ENTRY_COM_DESCRIPTOR
668  * CLR metadata is stored in this section. It is used to indicate that the object file contains
669  * managed code. The format of the metadata is not documented, but can be handed to the CLR
670  * interfaces for handling metadata. https://docs.microsoft.com/en-us/windows/win32/debug/pe-format
671  */
672  tabsec = new SgAsmCliHeader(this);
673  break;
674  }
675  default: {
676  tabsec = new SgAsmGenericSection(get_file(), this);
677  break;
678  }
679  }
680  tabsec->set_name(new SgAsmBasicString(tabname));
681  tabsec->set_shortName(tabname_short);
682  tabsec->set_synthesized(true);
683  tabsec->set_purpose(SP_HEADER);
684 
685  tabsec->set_offset(file_offset);
686  tabsec->set_size(pair->get_e_size());
687  tabsec->set_fileAlignment(1);
688 
689  tabsec->set_mappedAlignment(1);
690  tabsec->set_mappedPreferredRva(pair->get_e_rva().rva());
691  tabsec->set_mappedActualVa(pair->get_e_rva().rva()+get_baseVa()); /*FIXME: not sure this is correct. [RPM 2009-09-11]*/
692  tabsec->set_mappedSize(pair->get_e_size());
693  tabsec->set_mappedReadPermission(true);
694  tabsec->set_mappedWritePermission(false);
695  tabsec->set_mappedExecutePermission(false);
696  pair->set_section(tabsec);
697  pair->get_e_rva().bindSection(tabsec);
698  }
699 
700  /* Now parse the sections */
701  for (size_t i=0; i<get_rvaSizePairs()->get_pairs().size(); i++) {
702  SgAsmPERVASizePair *pair = get_rvaSizePairs()->get_pairs()[i];
703  SgAsmGenericSection *tabsec = pair->get_section();
704  if (tabsec)
705  tabsec->parse();
706  }

709 /* Change size of PE header based on word size */
710 bool
711 SgAsmPEFileHeader::reallocate()
712 {
713  struct Resources {
714  unsigned char *oh;
715  Resources(): oh(NULL) {}
716  ~Resources() { delete[] oh; }
717  } r;
718 
719  bool reallocated = SgAsmGenericHeader::reallocate();
720 
721  /* Resize if necessary */
722  Address need = sizeof(PEFileHeader_disk);
723  if (4==get_wordSize()) {
724  need += sizeof(PE32OptHeader_disk);
725  } else if (8==get_wordSize()) {
726  need += sizeof(PE64OptHeader_disk);
727  } else {
728  throw FormatError("unsupported PE word size");
729  }
730  need += get_rvaSizePairs()->get_pairs().size() * sizeof(SgAsmPERVASizePair::RVASizePair_disk);
731  if (need<get_size()) {
732  if (isMapped()) {
733  ROSE_ASSERT(get_mappedSize()==get_size());
734  set_mappedSize(need);
735  }
736  set_size(need);
737  reallocated = true;
738  } else if (need>get_size()) {
739  get_file()->shiftExtend(this, 0, need-get_size(), SgAsmGenericFile::ADDRSP_ALL, SgAsmGenericFile::ELASTIC_HOLE);
740  reallocated = true;
741  }
742 
743  /* Make sure the RVA/Size pairs at the end of the header are consistent with the sections to which they point. Reallocate()
744  * has already been called recursively for the sections. */
745  updateRvaSizePairs();
746 
747  /* Make sure header is consistent with sections. Reallocate() has already been called recursively for the sections.
748  * Count the number of sections in the table and update the header's e_nsections member. */
749  if (get_sectionTable()) {
750  ROSE_ASSERT(get_sectionTable()->get_header()==this);
751  SgAsmGenericSectionList *all = get_sections();
752  p_e_nsections = 0;
753  for (size_t i=0; i<all->get_sections().size(); i++) {
754  SgAsmPESection *pesec = dynamic_cast<SgAsmPESection*>(all->get_sections()[i]);
755  if (pesec && pesec->get_section_entry()!=NULL)
756  p_e_nsections++;
757  }
758 
759  Address header_size = alignUp(get_sectionTable()->get_offset() + get_sectionTable()->get_size(),
760  (Address)(p_e_file_align>0 ? p_e_file_align : 1));
761 #if 1
762  /* The PE Specification regarding e_header_size (known as "SizeOfHeader" on page 14 of "Microsoft Portable Executable
763  * and Common Object File Format Specification: Revision 8.1 February 15, 2008" is not always followed. We recompute
764  * it here as being the minimum RVA from all the sections defined in the PE Section Table, but not smaller
765  * than the value according to the specification. This alternate value is kept if it's already in the parse tree,
766  * otherwise we use the correct value. (RPM 2008-10-21) */
767  Address min_offset = 0;
768  for (size_t i=0, nfound=0; i<all->get_sections().size(); i++) {
769  SgAsmPESection *pesec = dynamic_cast<SgAsmPESection*>(all->get_sections()[i]);
770  if (pesec && pesec->get_section_entry()!=NULL) {
771  if (0==nfound++) {
772  min_offset = pesec->get_offset();
773  } else {
774  min_offset = std::min(min_offset, pesec->get_offset() );
775  }
776  }
777  }
778 
779  Address header_size2 = std::max(header_size, min_offset);
780  if (p_e_header_size==header_size2)
781  header_size = header_size2;
782 
783  /* If the original header size was zero then don't change that--leave it at zero. Some tiny executables have a zero
784  * value here and as a result, since this is near the end of the NT Optional Header, they can truncate the file and
785  * the loader will fill the optional header with zeros when reading. (RPM 2008-11-11) */
786  if (p_e_header_size==0)
787  header_size = 0;
788 #endif
789  p_e_header_size = header_size;
790  }
791 
792  /* The size of the optional header. If there's a section table then we use its offset to calculate the optional header
793  * size in order to be compatible with the PE loader. Otherwise use the actual optional header size. */
794  if (get_sectionTable()) {
795  ROSE_ASSERT(get_sectionTable()->get_offset() >= get_offset() + sizeof(PEFileHeader_disk));
796  p_e_nt_hdr_size = get_sectionTable()->get_offset() - (get_offset() + sizeof(PEFileHeader_disk));
797  } else if (4==get_wordSize()) {
798  p_e_nt_hdr_size = sizeof(PE32OptHeader_disk);
799  } else if (8==get_wordSize()) {
800  p_e_nt_hdr_size = sizeof(PE64OptHeader_disk);
801  } else {
802  throw FormatError("invalid PE word size");
803  }
804 
805  /* Update COFF symbol table related data members in the file header */
806  if (get_coffSymbolTable()) {
807  ROSE_ASSERT(get_coffSymbolTable()->get_header()==this);
808  set_e_coff_symtab(get_coffSymbolTable()->get_offset());
809  set_e_coff_nsyms(get_coffSymbolTable()->get_nslots());
810  }
811 
812  /* Update some additional header fields */
813  set_e_num_rvasize_pairs(get_rvaSizePairs()->get_pairs().size());
814  set_e_opt_magic(4==get_wordSize() ? 0x010b : 0x020b);
815  set_e_lmajor((get_executableFormat()->get_version() >> 16) & 0xffff);
816  set_e_lminor(get_executableFormat()->get_version() & 0xffff);
817 
818  /* Adjust the COFF Header's e_nt_hdr_size to accommodate the NT Optional Header in such a way that EXEs from tinype.com
819  * don't change (i.e., don't increase e_nt_hdr_size if the bytes beyond it are zero anyway, and if they aren't then adjust
820  * it as little as possible. The RVA/Size pairs are considered to be part of the NT Optional Header. */
821  size_t oh_size = get_rvaSizePairs()->get_pairs().size() * sizeof(SgAsmPERVASizePair::RVASizePair_disk);
822  if (4==get_wordSize()) {
823  oh_size += sizeof(PE32OptHeader_disk);
824  } else if (8==get_wordSize()) {
825  oh_size += sizeof(PE64OptHeader_disk);
826  } else {
827  throw FormatError("unsupported PE word size");
828  }
829  r.oh = new unsigned char[oh_size];
830  memset(r.oh, 0, oh_size);
831  if (4==get_wordSize()) {
832  encode((PE32OptHeader_disk*)r.oh);
833  } else if (8==get_wordSize()) {
834  encode((PE64OptHeader_disk*)r.oh);
835  } else {
836  throw FormatError("unsupported PE word size");
837  }
838  while (oh_size>p_e_nt_hdr_size) {
839  if (0!=r.oh[oh_size-1]) break;
840  --oh_size;
841  }
842  set_e_nt_hdr_size(oh_size);
843  return reallocated;
844 }

846 /* Write the PE file header back to disk and all that it references */
847 void
848 SgAsmPEFileHeader::unparse(std::ostream &f) const
849 {
850  struct Resources {
851  unsigned char *oh;
852  Resources(): oh(NULL) {}
853  ~Resources() { delete[] oh; }
854  } r;
855 
856  /* Write unreferenced areas back to the file before anything else. */
857  unparseHoles(f);
858 
859  /* Write sections in the order of specialization, from least specialized to most specialized. This gives more specialized
860  * sections a chance to overwrite the less specialized sections. */
861  const SgAsmGenericSectionPtrList &sections = get_sections()->get_sections();
862  for (SgAsmGenericSectionPtrList::const_iterator si=sections.begin(); si!=sections.end(); ++si) {
863  if (V_SgAsmGenericSection==(*si)->variantT())
864  (*si)->unparse(f);
865  }
866  for (SgAsmGenericSectionPtrList::const_iterator si=sections.begin(); si!=sections.end(); ++si) {
867  if (V_SgAsmPESection==(*si)->variantT())
868  (*si)->unparse(f);
869  }
870  for (SgAsmGenericSectionPtrList::const_iterator si=sections.begin(); si!=sections.end(); ++si) {
871  if (V_SgAsmGenericSection!=(*si)->variantT() && V_SgAsmPESection!=(*si)->variantT())
872  (*si)->unparse(f);
873  }
874 
875  /* Encode the "NT Optional Header" before the COFF Header since the latter depends on the former. Adjust the COFF Header's
876  * e_nt_hdr_size to accommodate the NT Optional Header in such a way that EXEs from tinype.com don't change (i.e., don't
877  * increase e_nt_hdr_size if the bytes beyond it are zero anyway, and if they aren't then adjust it as little as possible.
878  * The RVA/Size pairs are considered to be part of the NT Optional Header. */
879  size_t oh_size = get_rvaSizePairs()->get_pairs().size() * sizeof(SgAsmPERVASizePair::RVASizePair_disk);
880  size_t rvasize_offset; /*offset with respect to "oh" buffer allocated below*/
881  if (4==get_wordSize()) {
882  oh_size += sizeof(PE32OptHeader_disk);
883  } else if (8==get_wordSize()) {
884  oh_size += sizeof(PE64OptHeader_disk);
885  } else {
886  throw FormatError("unsupported PE word size");
887  }
888  r.oh = new unsigned char[oh_size];
889  if (4==get_wordSize()) {
890  encode((PE32OptHeader_disk*)r.oh);
891  rvasize_offset = sizeof(PE32OptHeader_disk);
892  } else if (8==get_wordSize()) {
893  encode((PE64OptHeader_disk*)r.oh);
894  rvasize_offset = sizeof(PE64OptHeader_disk);
895  } else {
896  throw FormatError("unsupported PE word size");
897  }
898  for (size_t i=0; i<get_rvaSizePairs()->get_pairs().size(); i++, rvasize_offset+=sizeof(SgAsmPERVASizePair::RVASizePair_disk)) {
899  SgAsmPERVASizePair::RVASizePair_disk *rvasize_disk = (SgAsmPERVASizePair::RVASizePair_disk*)(r.oh+rvasize_offset);
900  get_rvaSizePairs()->get_pairs()[i]->encode(rvasize_disk);
901  }
902  while (oh_size>p_e_nt_hdr_size) {
903  if (0!=r.oh[oh_size-1]) break;
904  --oh_size;
905  }
906  ROSE_ASSERT(p_e_nt_hdr_size==oh_size); /*set in reallocate()*/
907 
908  /* Write the fixed-length COFF Header */
909  PEFileHeader_disk fh;
910  encode(&fh);
911  Address spos = write(f, 0, sizeof fh, &fh);
912 
913  /* Write the following "NT Optional Header" */
914  spos = write(f, spos, oh_size, r.oh);
915 }
