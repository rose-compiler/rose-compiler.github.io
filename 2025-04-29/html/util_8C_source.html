39  bool checkCycle(const ControlFlowGraph& cfg)
40  {
41  // Add an edge from the exit to entry in the CFG, then count the number of
42  // strongly connected components. If the number is greater than 1, there is
43  // a cycle in the CFG which has no exit.
44  ControlFlowGraph cfgCopy = cfg;
45  boost::add_edge(cfgCopy.getExit(), cfgCopy.getEntry(), cfgCopy);
46 
47  std::vector<int> component(num_vertices(cfgCopy));
48  int num = boost::strong_components(cfgCopy, &component[0]);
49 
50  return num == 1;
51  }
