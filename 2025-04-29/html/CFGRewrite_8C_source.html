137 // Replace the expression from with the expression to in the SgNode parent, which
138 // must be from's parent. Function checks to ensure that it is used properly.
139 // Note that if parent is an SgInitializedName, to must be SgInitializer in order
140 // for the replacement to work properly.
141 void replaceExpressionChecked(SgNode* parent, SgExpression* from, SgExpression* to) {
142  ROSE_ASSERT (from);
143  ROSE_ASSERT (to);
144  ROSE_ASSERT (parent);
145  ROSE_ASSERT (parent != from);
146  ROSE_ASSERT (parent != to);
147  vector<SgNode*> children = parent->get_traversalSuccessorContainer();
148  bool foundFrom = false;
149  // verify that from is indeed one of parent's children
150  for (size_t i = 0; i < children.size(); ++i) {
151  if (children[i] == from) {foundFrom = true; break;}
152  }
153  // if this is not the case, yell
154  if (!foundFrom) {
155  cerr << "foundFrom failed: looking for " << from->unparseToString() << " in " << parent->unparseToString() << endl;
156  ROSE_ASSERT (foundFrom);
157  }
158 
159  if (isSgExprStatement(parent)) {
160  // replace parent's expression (used to be from) with to
161  ROSE_ASSERT (isSgExprStatement(parent)->get_expression() == from);
162  isSgExprStatement(parent)->set_expression(to);
163  to->set_parent(parent);
164  } else if (isSgReturnStmt(parent)) {
165  // replace parent's expression (used to be from) with to
166  ROSE_ASSERT (isSgReturnStmt(parent)->get_expression() == from);
167  isSgReturnStmt(parent)->set_expression(to);
168  to->set_parent(parent);
169  } else if (isSgExpression(parent)) {
170  // replace parent's from sub-expression with to
171  isSgExpression(parent)->replace_expression(from, to);
172  to->set_parent(parent);
173  // verify that to is now a sub-expression of parent
174  vector<SgNode*> children = parent->get_traversalSuccessorContainer();
175  for (size_t i = 0; i < children.size(); ++i) {
176  if (children[i] == to) return;
177  }
178  ROSE_ASSERT (!"replace_expression didn't work");
179  } else if (isSgInitializedName(parent)) {
180  // replace parent's initializer with the to initializer
181  ROSE_ASSERT (isSgInitializer(to));
182  isSgInitializedName(parent)->set_initializer(isSgInitializer(to));
183  to->set_parent(parent);
184  } else if (isSgForStatement(parent)) {
185  // ensure that we're only trying to replace the for statement's increment expression as
186  // the for statement's other components are statements, not expressions
187  ROSE_ASSERT (isSgForStatement(parent)->get_increment() == from);
188  isSgForStatement(parent)->set_increment(to);
189  to->set_parent(parent);
190  } else {
191  cerr << "replaceExpressionChecked: parent is <" << parent->class_name() << " | " << parent->unparseToString() << ">, from is <" << from->class_name() << "> \n";
192  ROSE_ASSERT (!"Unhandled case");
193  }
194 }
