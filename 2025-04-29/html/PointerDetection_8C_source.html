94 // Insert a new MemoryTranfer into a list of transfers. Return true if inserted.
95 bool
96 insertMemoryTransfer(MemoryTransferList &list /*in,out*/, const MemoryTransfer &xfer) {
97  ASSERT_not_null(xfer.memoryVa);
98  uint64_t hash = xfer.memoryVa->hash();
99 
100  for (auto iter = list.begin(); iter != list.end(); ++iter) {
101  ASSERT_not_null(iter->memoryVa);
102  if (iter->insnVa == xfer.insnVa && iter->direction == xfer.direction && iter->memoryVa->hash() == hash)
103  return false; // already in the list
104  }
105  list.push_back(xfer);
106  return true;
107 }

117 // Insert a new result into the list of final results and returns true if anything was inserted.
118 bool
119 insertPointerDescriptor(PointerDescriptors &list /*in,out*/, const PointerDescriptor &next) {
120  bool inserted = false;
121  for (auto cur = list.begin(); cur != list.end(); ++cur) {
122  if (cur->nBits == next.nBits && cur->pointerVa->hash() == next.pointerVa->hash()) {
123  for (auto nextAccess: next.pointerAccesses) {
124  if (cur->pointerAccesses.insert(nextAccess).second)
125  inserted = true;
126  }
127  return inserted;
128  }
129  }
130 
131  list.push_back(next);
132  return true;
133 }

377 contains(const SymbolicExpression::Ptr &haystack, const SymbolicExpression::Ptr &needle) {
378  struct Visitor: SymbolicExpression::Visitor {
379  bool found = false;

381 
382  Visitor(const SymbolicExpression::Ptr &needle)
383  : needle(needle) {}
384 
385  virtual SymbolicExpression::VisitAction preVisit(const SymbolicExpression::Ptr &node) {
386  if (node->hash() == needle->hash()) {
387  found = true;
388  return SymbolicExpression::TRUNCATE;
389  } else {
390  return SymbolicExpression::CONTINUE;
391  }
392  }

395  return SymbolicExpression::CONTINUE;
396  }
397  };
398 
399  Visitor visitor(needle);
400  haystack->depthFirstTraversal(visitor);
401  return visitor.found;

404 static void
405 discoverDereferences(PointerDescriptors &pointers, const MemoryTransferMap &transferMap) {
406  Sawyer::Message::Stream debug(mlog[DEBUG]);
407  SAWYER_MESG(debug) <<"finding pointer dereferences\n";
408  for (PointerDescriptor &pointer: pointers) {
409  SAWYER_MESG(debug) <<" finding dereferences for pointer stored at " <<*pointer.pointerVa <<"\n";
410 
411  // Find all the values this pointer can have; i.e., all the addresses to which this pointer can point.
412  std::set<SymbolicExpression::Ptr, SymbolicLessp> pointerValues;
413  for (const PointerDescriptor::Access &access: pointer.pointerAccesses) {
414  SAWYER_MESG(debug) <<" at " <<StringUtility::addrToString(access.insnVa)
415  <<" pointer value = " <<*access.value <<"\n";
416  pointerValues.insert(access.value);
417  }
418 
419  // Now find all the memory accesses that have the pointer value as an address.
420  for (const SymbolicExpression::Ptr &pointerValue: pointerValues) {
421  for (const MemoryTransferMap::Node &node: transferMap.nodes()) {
422  for (const MemoryTransfer &xfer: node.value()) {
423  if (contains(xfer.memoryVa, pointerValue)) {
424  SAWYER_MESG(debug) <<" at " <<StringUtility::addrToString(xfer.insnVa)
425  <<" there is a " <<(PointerDescriptor::READ==xfer.direction?"read":"write") <<"-dereference "
426  <<" using pointer value " <<*pointerValue
427  <<" for memory address " <<*xfer.memoryVa
428  <<" having value " <<*node.key() <<"\n";
429  pointer.dereferences.insert(PointerDescriptor::Access(xfer.insnVa, xfer.direction, node.key()));
430  }
431  }
432  }
433  }
434  }

437 void
438 Analysis::pruneResults(PointerDescriptors &descriptors) {
439  // Discard things we don't want to save.
440  for (PointerDescriptor &desc: descriptors) {
441  if (!settings_.savePointerVas)
442  desc.pointerVa = SymbolicExpression::Ptr();
443  if (!settings_.savePointerAccesses) {
444  desc.pointerAccesses.clear();
445  } else if (!settings_.savePointerAccessValues) {
446  std::set<PointerDescriptor::Access> set;
447  for (const PointerDescriptor::Access &access: desc.pointerAccesses)

449  desc.pointerAccesses = set;
450  }
451  if (!settings_.savePointerDereferences) {
452  desc.dereferences.clear();
453  } else if (!settings_.savePointerDereferenceValues) {
454  std::set<PointerDescriptor::Access> set;
455  for (const PointerDescriptor::Access &access: desc.dereferences)
456  set.insert(PointerDescriptor::Access(access.insnVa, access.direction, SymbolicExpression::Ptr()));

458  }

462 void
463 Analysis::analyzeFunction(const P2::Partitioner::ConstPtr &partitioner, const P2::Function::Ptr &function) {
464  mlog[DEBUG] <<"analyzeFunction(" <<function->printableName() <<")\n";
465  printInstructionsForDebugging(partitioner, function);
466 
467  clearResults();
468  initialState_ = BaseSemantics::State::Ptr();
469  finalState_ = BaseSemantics::State::Ptr();
470 
471  // Build the CFG used by the dataflow: dfCfg. The dfCfg includes only those vertices that are reachable from the entry
472  // point for the function we're analyzing and which belong to that function. All return points in the function will flow
473  // into a special CALLRET vertex (which is absent if there are no returns).
474  typedef P2::DataFlow::DfCfg DfCfg;
475  DfCfg dfCfg = P2::DataFlow::buildDfCfg(partitioner, partitioner->cfg(), partitioner->findPlaceholder(function->address()));
476  size_t startVertexId = 0;
477  DfCfg::ConstVertexIterator returnVertex = dfCfg.vertices().end();
478  for (const DfCfg::Vertex &vertex: dfCfg.vertices()) {
479  if (vertex.value().type() == P2::DataFlow::DfCfgVertex::FUNCRET) {
480  returnVertex = dfCfg.findVertex(vertex.id());
481  break;
482  }
483  }
484  if (returnVertex == dfCfg.vertices().end()) {
485  mlog[DEBUG] <<" function CFG has no return vertex\n";
486  return;
487  }
488 
489  // Build the dataflow engine.
490  typedef DataFlow::Engine<DfCfg, BaseSemantics::State::Ptr, P2::DataFlow::TransferFunction, DataFlow::SemanticsMerge> DfEngine;
491  BaseSemantics::RiscOperators::Ptr ops = makeRiscOperators(partitioner);
492  ASSERT_always_not_null(ops);
493  BaseSemantics::Dispatcher::Ptr cpu = partitioner->newDispatcher(ops);
494  ASSERT_always_not_null(cpu);
495  P2::DataFlow::MergeFunction merge(cpu);
496  P2::DataFlow::TransferFunction xfer(cpu);
497  DfEngine dfEngine(dfCfg, xfer, merge);
498  dfEngine.name("pointer-detection");
499  dfEngine.maxIterations(dfCfg.nVertices() * settings_.maximumDataFlowIterationFactor);
500 
501  // Build the initial state
502  initialState_ = xfer.initialState();
503  BaseSemantics::RegisterStateGeneric::Ptr initialRegState =
504  BaseSemantics::RegisterStateGeneric::promote(initialState_->registerState());
505  initialRegState->initialize_large();
506  cpu->initializeState(initialState_);
507 
508  // Allow data-flow merge operations to create sets of values up to a certain cardinality. This is optional.
509  SymbolicSemantics::Merger::Ptr merger = SymbolicSemantics::Merger::instance(10 /*arbitrary*/);
510  initialState_->registerState()->merger(merger);
511  initialState_->memoryState()->merger(merger);
512 
513  // Run the data-flow analysis
514  bool converged = true;
515  try {
516  // Use this rather than runToFixedPoint because it lets us show a progress report
517  Sawyer::ProgressBar<size_t> progress(mlog[MARCH], function->printableName());
518  progress.suffix(" iterations");
519  dfEngine.reset(BaseSemantics::State::Ptr());
520  dfEngine.insertStartingVertex(startVertexId, initialState_);
521  while (dfEngine.runOneIteration())
522  ++progress;
523  } catch (const InstructionSemantics::BaseSemantics::NotImplemented &e) {
524  mlog[WHERE] <<e.what() <<"\n";
525  converged = false;
526  } catch (const InstructionSemantics::BaseSemantics::Exception &e) {
527  mlog[WARN] <<e.what() <<"\n";
528  converged = false;
529  } catch (const DataFlow::NotConverging &e) {
530  mlog[WARN] <<e.what() <<"\n";
531  converged = false; // didn't converge, so just use what we have
532  }
533  finalState_ = dfEngine.getInitialState(returnVertex->id());
534  if (!finalState_)
535  return;
536 
537  State::Ptr finalState = State::promote(finalState_);
538  SAWYER_MESG(mlog[DEBUG]) <<" " <<(converged ? "data-flow converged" : "DATA-FLOW DID NOT CONVERGE") <<"\n";
539 
540  if (mlog[DEBUG]) {
541  mlog[DEBUG] <<" memory accesses:\n";
542  for (const MemoryTransferMap::Node &node: finalState->memoryTransfers().nodes()) {
543  mlog[DEBUG] <<" value " <<*node.key() <<"\n";
544  for (const MemoryTransfer &xfer: node.value()) {
545  mlog[DEBUG] <<" " <<(PointerDescriptor::READ == xfer.direction ? "read" : "write")
546  <<" at insn " <<StringUtility::addrToString(xfer.insnVa)
547  <<" for pointer at " <<*xfer.memoryVa <<"\n";
548  }
549  }
550  }
551 
552  // Find data pointers
553  if (settings_.saveDataPointers) {
554  SAWYER_MESG(mlog[DEBUG]) <<" potential data pointers:\n";
555  Sawyer::Container::Set<SymbolicExpression::Hash> addrSeen;
556  for (size_t vertexId = 0; vertexId < dfCfg.nVertices(); ++vertexId) {
557  if (BaseSemantics::State::Ptr state = dfEngine.getFinalState(vertexId)) {
558  auto memState = BaseSemantics::MemoryCellState::promote(state->memoryState());
559  for (const BaseSemantics::MemoryCell::Ptr &cell: memState->allCells())
560  conditionallySavePointer(cell->address(), addrSeen, dataPointers_);
561  }
562  }
563  discoverDereferences(dataPointers_, finalState->memoryTransfers());
564  pruneResults(dataPointers_);
565  }
566 
567  // Find code pointers
568  if (settings_.saveCodePointers) {
569  SAWYER_MESG(mlog[DEBUG]) <<" potential code pointers:\n";
570  Sawyer::Container::Set<SymbolicExpression::Hash> addrSeen;
571  const RegisterDescriptor IP = partitioner->instructionProvider().instructionPointerRegister();
572  for (size_t vertexId = 0; vertexId < dfCfg.nVertices(); ++vertexId) {
573  if (BaseSemantics::State::Ptr state = dfEngine.getFinalState(vertexId)) {
574  SymbolicSemantics::SValue::Ptr ip =
575  SymbolicSemantics::SValue::promote(state->peekRegister(IP, ops->undefined_(IP.nBits()), ops.get()));
576  SymbolicExpression::Ptr ipExpr = ip->get_expression();
