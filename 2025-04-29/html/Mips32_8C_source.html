34 RegisterDictionary::Ptr
35 Mips32::registerDictionary() const {
36  static SAWYER_THREAD_TRAITS::Mutex mutex;
37  SAWYER_THREAD_TRAITS::LockGuard lock(mutex);
38 
39  if (!registerDictionary_.isCached()) {
40  auto regs = RegisterDictionary::instance(name());
41 
42  // 32 general purpose registers and hardware registers
43  for (size_t i=0; i<32; ++i) {
44  regs->insert("r"+StringUtility::numberToString(i), mips_regclass_gpr, i, 0, 32);
45  regs->insert("hw"+StringUtility::numberToString(i), mips_regclass_hw, i, 0, 32);
46  }
47 
48  // Alternate names for the general purpose registers
49  regs->insert("zero", mips_regclass_gpr, 0, 0, 32); // always equal to zero
50  regs->insert("at", mips_regclass_gpr, 1, 0, 32); // assembler temporary
51  regs->insert("v0", mips_regclass_gpr, 2, 0, 32); // return value from a function call
52  regs->insert("v1", mips_regclass_gpr, 3, 0, 32);
53  regs->insert("a0", mips_regclass_gpr, 4, 0, 32); // first four function arguments
54  regs->insert("a1", mips_regclass_gpr, 5, 0, 32);
55  regs->insert("a2", mips_regclass_gpr, 6, 0, 32);
56  regs->insert("a3", mips_regclass_gpr, 7, 0, 32);
57  regs->insert("t0", mips_regclass_gpr, 8, 0, 32); // temporary variables; need not be preserved
58  regs->insert("t1", mips_regclass_gpr, 9, 0, 32);
59  regs->insert("t2", mips_regclass_gpr, 10, 0, 32);
60  regs->insert("t3", mips_regclass_gpr, 11, 0, 32);
61  regs->insert("t4", mips_regclass_gpr, 12, 0, 32);
62  regs->insert("t5", mips_regclass_gpr, 13, 0, 32);
63  regs->insert("t6", mips_regclass_gpr, 14, 0, 32);
64  regs->insert("t7", mips_regclass_gpr, 15, 0, 32);
65  regs->insert("s0", mips_regclass_gpr, 16, 0, 32); // temporary variables; must be preserved
66  regs->insert("s1", mips_regclass_gpr, 17, 0, 32);
67  regs->insert("s2", mips_regclass_gpr, 18, 0, 32);
68  regs->insert("s3", mips_regclass_gpr, 19, 0, 32);
69  regs->insert("s4", mips_regclass_gpr, 20, 0, 32);
70  regs->insert("s5", mips_regclass_gpr, 21, 0, 32);
71  regs->insert("s6", mips_regclass_gpr, 22, 0, 32);
72  regs->insert("s7", mips_regclass_gpr, 23, 0, 32);
73  regs->insert("t8", mips_regclass_gpr, 24, 0, 32); // two more temporaries; need not be preserved
74  regs->insert("t9", mips_regclass_gpr, 25, 0, 32);
75  regs->insert("k0", mips_regclass_gpr, 26, 0, 32); // kernel use; may change unexpectedly
76  regs->insert("k1", mips_regclass_gpr, 27, 0, 32);
77  regs->insert("gp", mips_regclass_gpr, 28, 0, 32); // global pointer
78  regs->insert("sp", mips_regclass_gpr, 29, 0, 32); // stack pointer
79  regs->insert("s8", mips_regclass_gpr, 30, 0, 32); // temp; must be preserved (or "fp")
80  regs->insert("fp", mips_regclass_gpr, 30, 0, 32); // stack frame pointer (or "s8")
81  regs->insert("ra", mips_regclass_gpr, 31, 0, 32); // return address (link register)
82 
83  // Special purpose registers
84  regs->insert("hi", mips_regclass_spr, mips_spr_hi, 0, 32);
85  regs->insert("lo", mips_regclass_spr, mips_spr_lo, 0, 32);
86  regs->insert("pc", mips_regclass_spr, mips_spr_pc, 0, 32); // program counter
87 
88  // 32 floating point registers, each 32 bits
89  for (size_t i=0; i<32; ++i)
90  regs->insert("f"+StringUtility::numberToString(i), mips_regclass_fpr, i, 0, 32);
91 
92  // 16 floating point registers, each 64 bits. 64-bit FP are stored in pairs of 32-bit FP registers using the
93  // even register name. But since a register name in ROSE must be unique, we append the letter "d" (for "double").
94  for (size_t i=0; i<16; ++i)
95  regs->insert("f"+StringUtility::numberToString(i)+"d", mips_regclass_fpr, i*2, 0, 64);
96 
97  // Five FPU control registers are used to identify and control the FPU. The FCCR, FEXR, and FENR are portions
98  // (not necessarily contiguous) of the FCSR extended to 32 bits, and therefore all share a major number.
99  regs->insert("fir", mips_regclass_spr, mips_spr_fir, 0, 32); // FP implementation and revision
100  regs->insert("fcsr", mips_regclass_fcsr, mips_fcsr_all, 0, 32); // the entire FCSR register
101  regs->insert("fccr", mips_regclass_fcsr, mips_fcsr_fccr, 0, 32); // condition codes portion of FCSR
102  regs->insert("fexr", mips_regclass_fcsr, mips_fcsr_fexr, 0, 32); // FP exceptions
103  regs->insert("fenr", mips_regclass_fcsr, mips_fcsr_fenr, 0, 32); // FP enables
104 
105  // Names for the FCCR (condition code) bits in the FCSR register (cc0...cc7)
106  for (size_t i=0; i<8; ++i)
107  regs->insert("cc"+StringUtility::numberToString(i), mips_regclass_fcsr, mips_fcsr_fccr, 23+i, 1);
108 
109  // parts of the FIR (only those defined for MIPS32 release 1)
110  regs->insert("fir.d", mips_regclass_spr, mips_spr_fir, 17, 1); // is double-precision implemented?
111  regs->insert("fir.s", mips_regclass_spr, mips_spr_fir, 16, 1); // is single-precision implemented?
112  regs->insert("fir.processorid", mips_regclass_spr, mips_spr_fir, 8, 8); // identifies the FP processor
113  regs->insert("fir.revision", mips_regclass_spr, mips_spr_fir, 0, 8); // FP unit revision number
114 
115  // Additional registers for coprocessor 0 are not part of this dictionary. They use major number mips_regclass_cp0gpr.
116 
117  // Additional implementation-specific coprocessor 2 registers are not part of the dictionary. Coprocessor 2 may have up
118  // to 32 general purpose registers and up to 32 control registers. They use the major numbers mips_regclass_cp2gpr and
119  // mips_regclass_cp2spr.
120 
121  // Special registers
122  regs->instructionPointerRegister("pc");
123  regs->stackPointerRegister("sp");
124  regs->stackFrameRegister("fp");
125  regs->callReturnRegister("ra");
126 
127  registerDictionary_ = regs;
128  }
129 
130  return registerDictionary_.get();
131 }

150 Mips32::matchesHeader(SgAsmGenericHeader *header) const {

671 Mips32::branchTarget(SgAsmInstruction *insn_) const {

691  case mips_bgtz:
692  case mips_bgtzl:
693  case mips_blez:
694  case mips_blezl:
695  case mips_bltz:
696  case mips_bltzal:
697  case mips_bltzall:
698  case mips_bltzl:
699  // target address stored in the second argument
700  ASSERT_require(args.size() >= 2);
701  ASSERT_require(isSgAsmIntegerValueExpression(args[1]));
702  return isSgAsmIntegerValueExpression(args[1])->get_absoluteValue();
703 
704  case mips_beq:
705  case mips_beql:
706  case mips_bne:
707  case mips_bnel:
708  // target address stored in the third argument

719 AddressSet
720 Mips32::getSuccessors(SgAsmInstruction *insn_, bool &complete) const {
721  auto insn = isSgAsmMipsInstruction(insn_);
722  ASSERT_not_null(insn);
723 
724  complete = false;
725  AddressSet successors;
726 
727  switch (insn->get_kind()) {
728  case mips_b:
729  case mips_bal:
730  case mips_break:
731  case mips_j:
732  case mips_jal:
733  case mips_jalr:
734  case mips_jalx:
735  case mips_jr:
736  case mips_jr_hb:
737  case mips_syscall:
738  // unconditional branch
739  if (Sawyer::Optional<Address> target = branchTarget(insn)) {
740  successors.insert(*target);
741  complete = true;
742  }
743  break;
744 
745  case mips_beq:
746  case mips_beql:
747  case mips_bgez:
748  case mips_bgezal:
749  case mips_bgezall:
750  case mips_bgezl:
751  case mips_bgtz:
752  case mips_bgtzl:
753  case mips_blez:
754  case mips_blezl:
755  case mips_bltz:
756  case mips_bltzal:
757  case mips_bltzall:
758  case mips_bltzl:
759  case mips_bne:
760  case mips_bnel:
761  case mips_teq:
762  case mips_teqi:
763  case mips_tge:
764  case mips_tgei:
765  case mips_tgeiu:
766  case mips_tgeu:
767  case mips_tlt:
768  case mips_tlti:
769  case mips_tltiu:
770  case mips_tltu:
771  case mips_tne:
772  case mips_tnei:
773  // conditional branch
774  if (Sawyer::Optional<Address> target = branchTarget(insn)) {
775  successors.insert(*target);
776  complete = true;
777  }
778  successors.insert(insn->get_address() + insn->get_size()); // fall through address
779  break;
780 
781  default:
782  // fall through
783  successors.insert(insn->get_address() + insn->get_size());
784  complete = true;
785  break;
786  }
787  return successors;
