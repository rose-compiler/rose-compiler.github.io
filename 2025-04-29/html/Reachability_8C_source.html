43 // Command-line parsing
44 //
45 // Command-line parsing of switches whose arguments are lists of Reason enums is a little complicated because:
46 //
47 // 1. The token is a string but the value is an enum constant, so we need a translation from string to enum. Fortunately
48 // this is built into Sawyer::CommandLine as the EnumParser class template.
49 //
50 // 2. Since the settings being adjusted are sets of enum constants (BitFlags<T>), the documentation describing the defaults
51 // must be constructed from the set.
52 //
53 // 3. If the user specifies an argument (list of enum names) for the switch, then the default value of the switch should
54 // be discarded before the specified enums are unioned into the result. The clearing and unioning are handled by the
55 // ReasonSaver class.
56 //
57 // 4. The names of the enum constants in C++ are not the same as the names we want users to use on the command-line. Since
58 // the command-line is smaller than the API, we can use abbreviated terms in the command-line.
59 //
60 // 5. Since the switches are also used to turn features on and off, we need to support specifying the switch with no argument
61 // in which case a default should be used, but a different default than not specifying the switch at all. E.g., the
62 // switch-not-specified value might be the empty set, and the switch-specified-with-no-argument value is probably a set
63 // of one or more enum constants. The ability to not have an argument needs to be optional per switch.
64 //
65 // 6. We need some way for the user to turn off a feature. It's customary to use switches that start with "no-" to accomplish
66 // this, as in "--foo" and "--no-foo".
67 //
68 // 7. Large parts of the documentation must be dynamically generated and should be consistent between all switches.
70 
71 class ReasonSaver: public Sawyer::CommandLine::ValueSaver {
72  Reachability::ReasonFlags &storage_;
73  bool isDefault_;
74 protected:
75  ReasonSaver(Reachability::ReasonFlags &storage)
76  : storage_(storage), isDefault_(true) {}
77 public:
78  typedef Sawyer::SharedPointer<ReasonSaver> Ptr;
79  static Ptr instance(Reachability::ReasonFlags &storage) {

82  void save(const boost::any &value, const std::string &/*switchKey*/) override {
83  Reachability::Reason reason = boost::any_cast<Reachability::Reason>(value);
84  if (isDefault_)
85  storage_.clear();
86  isDefault_ = false;
87 
88  if (Reachability::NOT_REACHABLE == reason) {

90  } else {

92  }
93  }

341 size_t
342 Reachability::markEntryFunctions(const P2::Partitioner::ConstPtr &partitioner, ReasonFlags how) {
343  ASSERT_not_null(partitioner);
344  Sawyer::Message::Stream debug(mlog[DEBUG]);
345  SAWYER_MESG(debug) <<"marking entry functions";
346  Sawyer::Stopwatch timer;
347 
348  resize(partitioner);
349  size_t nChanges = 0;
350  for (const P2::Function::Ptr &func: partitioner->functions()) {
351  P2::ControlFlowGraph::ConstVertexIterator vertex = partitioner->findPlaceholder(func->address());
352  ASSERT_require(partitioner->cfg().isValidVertex(vertex));
353  if (0 != (func->reasons() & SgAsmFunction::FUNC_ENTRY_POINT)) {
354  ReasonFlags wasReachable = isIntrinsicallyReachable(vertex->id());
355  ReasonFlags reachable = wasReachable | how;
356  if (reachable != wasReachable) {
357  intrinsicallyReachable(vertex->id(), reachable);
358  ++nChanges;
359  }
360  }
361  }
362 
363  SAWYER_MESG(debug) <<"; " <<StringUtility::plural(nChanges, "changes") <<" in " <<timer <<"\n";
364  return nChanges;
365 }

544  std::set<size_t> vertexIds;

553 // Scan memory for reachable addresses. This is a little more complex than just reading one address at a time because:
554 // 1. We want it to be efficient, so we read a whole buffer at once and look for addresses in the buffer.
555 // 2. We need to handle alignment, byte order, and various sizes of addresses.
556 // 3. Depending on alignment and size, addresses can overlap each other in memory.
557 // 4. We need to watch out for addresses that would span two of our buffer-sized windows read from memory
558 // 5. We need to be careful of addresses that are 2^32 (or 2^64) to avoid overflow when incrementing
559 std::set<size_t>
560 Reachability::findExplicitMemoryReferents(const P2::Partitioner::ConstPtr &partitioner, const MemoryMap::Ptr &map,
561  size_t bytesPerWord, size_t alignment, ByteOrder::Endianness sex) {
562  ASSERT_not_null(partitioner);
563  ASSERT_not_null(map);
564 
565  if (0 == bytesPerWord)
566  bytesPerWord = settings_.addressNBytes;
567  if (0 == bytesPerWord)
568  bytesPerWord = partitioner->instructionProvider().instructionPointerRegister().nBits() / 8;
569 
570  if (0 == alignment)
571  alignment = settings_.addressAlignment;
572  if (0 == alignment)
573  alignment = bytesPerWord;
574 
575  if (ByteOrder::ORDER_UNSPECIFIED == sex)
576  sex = settings_.byteOrder;
577  if (ByteOrder::ORDER_UNSPECIFIED == sex)
578  sex = partitioner->instructionProvider().defaultByteOrder();
579 
580  ASSERT_require(bytesPerWord <= sizeof(Address));
581  ASSERT_not_null(map);
582  alignment = std::max(alignment, (size_t)1);
583  uint8_t buf[4096]; // arbitrary size
584  Address bufVa = map->hull().least();
585  std::set<size_t> vertexIds;
586 
587  while (map->atOrAfter(bufVa).next().assignTo(bufVa)) {
588  {
589  Address tmp = alignUp(bufVa, alignment);
590  if (tmp < bufVa)
591  break; // overflow
592  bufVa = tmp;
593  }
594  size_t bufSize = map->at(bufVa).limit(sizeof buf).read(buf).size();
595  if (bufSize < bytesPerWord) {
596  bufVa += bytesPerWord;
597  if (bufVa <= map->hull().least())
598  break; // overflow
599  continue;
600  }
601 
602  size_t bufOffset = 0;
603  while (1) {
604  bufOffset = alignUp(bufOffset, alignment);
605  if (bufOffset + bytesPerWord > bufSize)
606  break;
607 
608  Address targetVa = 0;
609  switch (sex) {
610  case ByteOrder::ORDER_UNSPECIFIED:
611  static bool messageShown = false;
612  if (!messageShown) {
613  mlog[WARN] <<"memory byte order is unspecified; assuming little endian\n";
614  messageShown = true;
615  }
616  // fall through
617  case ByteOrder::ORDER_LSB:
618  for (size_t i=0; i<bytesPerWord; i++)
619  targetVa |= buf[bufOffset+i] << (8*i);
620  break;
621  case ByteOrder::ORDER_MSB:
622  for (size_t i=0; i<bytesPerWord; i++)
623  targetVa = (targetVa << 8) | buf[bufOffset+i];
624  break;
625  }
626  P2::ControlFlowGraph::ConstVertexIterator targetVertex = partitioner->findPlaceholder(targetVa);
627  if (targetVertex != partitioner->cfg().vertices().end())
628  vertexIds.insert(targetVertex->id());
629  ++bufOffset;
630  }
631  bufVa += bufOffset;
632  }
633  return vertexIds;
634 }

718 size_t
719 Reachability::propagateImpl(const P2::Partitioner::ConstPtr &partitioner, std::vector<size_t> *vertexIds) {
720  ASSERT_not_null(partitioner);
721  Sawyer::Message::Stream debug(mlog[DEBUG]);
722  SAWYER_MESG(debug) <<"propagating";
723  Sawyer::Stopwatch timer;
724  resize(partitioner);
725 
726  typedef DataFlow::Engine<const P2::ControlFlowGraph, ReasonFlags, TransferFunction, MergeFunction> DfEngine;
727  TransferFunction xfer;
728  MergeFunction merge;
729  DfEngine dfEngine(partitioner->cfg(), xfer, merge);
730  dfEngine.name("reachability");
731  dfEngine.reset(NOT_REACHABLE);
732 
733  // Initialize the data-flow states for vertices with intrinsic reachability.
734  for (size_t i=0; i<partitioner->cfg().nVertices(); ++i) {
735  ReasonFlags r = intrinsicReachability_[i];
736  if (r.isAnySet())
737  dfEngine.insertStartingVertex(i, r);
738  }
739 
740  // Run a data-flow analysis to propagate reachability from intrinsically reachable vertices to other vertices.
741  dfEngine.runToFixedPoint();
742 
743  // Copy out all the vertex final states from the data-flow engine.
744  size_t nChanges = 0;
745  if (vertexIds != NULL)
746  vertexIds->reserve(partitioner->cfg().nVertices());
747  for (size_t vertexId = 0; vertexId < partitioner->cfg().nVertices(); ++vertexId) {
748  if (dfEngine.getFinalState(vertexId) != reachability_[vertexId]) {
749  reachability_[vertexId] = dfEngine.getFinalState(vertexId);
750  ++nChanges;
751  if (vertexIds)
752  vertexIds->push_back(vertexId);
753  }
754  }
755 
756  SAWYER_MESG(debug) <<"; " <<StringUtility::plural(nChanges, "changes") <<" in " <<timer <<"\n";
757  return nChanges;
758 }

764 size_t
765 Reachability::iterationMarking(const P2::Partitioner::ConstPtr &partitioner, const std::vector<size_t> &vertexIds) {
766  ASSERT_not_null(partitioner);
767  Sawyer::Message::Stream debug(mlog[DEBUG]);
768  size_t nMarked = 0;
769  std::set<P2::Function::Ptr> functionsToAnalyze;
770 
771  for (size_t vertexId: vertexIds) {
772  P2::ControlFlowGraph::ConstVertexIterator vertex = partitioner->cfg().findVertex(vertexId);
773  P2::BasicBlock::Ptr bblock;
774  if (vertex->value().type() == P2::V_BASIC_BLOCK)
775  bblock = vertex->value().bblock();
776 
777  // Explicit constants in instructions
778  if (settings_.markingExplicitInstructionReferents.isAnySet() && bblock)
779  nMarked += markExplicitInstructionReferents(partitioner, bblock, settings_.markingExplicitInstructionReferents);
780 
781  // Find functions that need to have the data-flow constants calculated
782  if (settings_.markingImplicitFunctionReferents.isAnySet() && bblock) {
783  std::vector<P2::Function::Ptr> functions = partitioner->functionsOwningBasicBlock(bblock);
784  for (const P2::Function::Ptr &function: functions) {
785  if (!dfReferents_.exists(function))
786  functionsToAnalyze.insert(function);
787  }
788  }
789  }
790 
791  // Implicit constants in function data-flow
792  cacheImplicitFunctionReferents(partitioner, functionsToAnalyze); // done in parallel
793  for (const P2::Function::Ptr &function: functionsToAnalyze) {
794  const std::set<size_t> referents = dfReferents_.get(function); // must exist by now, but may be empty
795  nMarked += intrinsicallyReachable(referents.begin(), referents.end(), settings_.markingImplicitFunctionReferents);
796  }
797  return nMarked;
798 }

800 void
801 Reachability::iterate(const P2::Partitioner::ConstPtr &partitioner) {
802  Sawyer::Message::Stream debug(mlog[DEBUG]);
803  resize(partitioner);
804 
805  // Run the data-flow to find constants in all the functions. We do this up front because we can do it in parallel,
806  // although if the reachability set is small, it might be better to calculate this on demand in the loop below.
807  if (settings_.precomputeImplicitFunctionReferents &&
808  settings_.markingImplicitFunctionReferents.isAnySet() &&
809  dfReferents_.isEmpty())
810  cacheAllImplicitFunctionReferents(partitioner);
811 
812  // Before starting, make sure that we've scanned the vertices that are already reachable.
813  SAWYER_MESG(debug) <<"iteration[-1] marking intrinsic reachability";
814  std::vector<size_t> ids = reachableVertices();
815  scannedVertexIds_.removeIfSeen(ids /*in,out*/);
816  size_t nMarked = iterationMarking(partitioner, ids);
817  SAWYER_MESG(debug) <<" changed " <<StringUtility::plural(nMarked, "vertices") <<"\n";
818 
819  // Iterative propgate and mark
820  for (size_t passNumber = 0; true; ++passNumber) {
821  SAWYER_MESG(debug) <<"iteration " <<passNumber <<"\n";
822 
823  // Propagate reachability
824  SAWYER_MESG(debug) <<"iteration[" <<passNumber <<"] propagating reachability";
825  propagate(partitioner, ids /*out*/);
826  SAWYER_MESG(debug) <<" changed " <<StringUtility::plural(ids.size(), "vertices") <<"\n";
827 
828  // Mark other vertices by scanning those changed by the propagate step
829  SAWYER_MESG(debug) <<"iteration[" <<passNumber <<"] marking intrinsic reachability";
830  scannedVertexIds_.removeIfSeen(ids /*in,out*/);
831  nMarked = iterationMarking(partitioner, ids);
832  SAWYER_MESG(debug) <<" changed " <<StringUtility::plural(nMarked, "vertices") <<"\n";
833 
834  if (0 == nMarked)
835  break;
836  }
