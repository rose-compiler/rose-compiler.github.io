30 SAWYER_EXPORT bool
31 initializeLibrary(size_t vmajor, size_t vminor, size_t vpatch, bool withThreads) {
32  // Make sure that the application has compiled the correct version and configuration of Sawyer header files. They
33  // must match the version and configuration expected by the Sawyer library.
34  if (vmajor != SAWYER_VERSION_MAJOR || vminor != SAWYER_VERSION_MINOR || vpatch != SAWYER_VERSION_PATCH)
35  throw std::runtime_error("inconsistent compiling/linking with libsawyer: version number mismatch");
36  if (withThreads != SAWYER_MULTI_THREADED)
37  throw std::runtime_error("inconsistent compiling/linking with libsawyer: thread support mismatch");
38 
39  Initializer init;
40 #if SAWYER_MULTI_THREADED
41  boost::call_once(initFlag, init);
42 #else
43  static bool initialized = false;
44  if (!initialized) {
45  init();
46  initialized = true;
47  }
48 #endif
49 
50  atexit(Sawyer::Message::shutdown);
51 
52  return true;
53 }

118 SAWYER_EXPORT void
119 checkBoost() {
120  // We do not support boost 1.54 with C++11 because of boost ticket #9215 [https://svn.boost.org/trac/boost/ticket/9215]. It
121  // is better if we check for this at runtime rather than compile time because many other features of boost 1.54 work
122  // fine. If we allow the user to continue, then boost::any's move constructor will enter infinite recursion eventually
123  // ending with a segmentation fault (although probably not occuring as a result of calling just Storage's c'tor).
124  ASSERT_always_forbid2(BOOST_VERSION == 105400 && __cplusplus >= 201103L,
125  "boost::any move constructor has infinite recursion in boost-1.54");
126 }
