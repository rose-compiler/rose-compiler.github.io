114 bool
115 SgAsmBlock::isFunctionCall(Address &target_va, Address &return_va)
116 {
117  static const Address INVALID_ADDR = (Address)(-1);
118  target_va = return_va = INVALID_ADDR;;
119  if (!isBasicBlock())
120  return false;
121  std::vector<SgAsmInstruction*> insns = AST::Traversal::findDescendantsTyped<SgAsmInstruction>(this);
122  assert(!insns.empty()); // basic blocks must have instructions
123 
124  // Check that all the successors point to functions entry addresses (other functions or this block's function). There
125  // might be one edge that points to the fall-through address of this block, and that's ok.
126  SgAsmFunction *func = AST::Traversal::findParentTyped<SgAsmFunction>(this);
127  SgAsmInterpretation *interp = AST::Traversal::findParentTyped<SgAsmInterpretation>(func);
128  std::set<Address> callee_vas;
129  if (interp) {
130  const InstructionMap &imap = interp->get_instructionMap();
131  const SgAsmIntegerValuePtrList &successors = get_successors();
132  for (SgAsmIntegerValuePtrList::const_iterator si=successors.begin(); si!=successors.end(); ++si) {
133  Address successor_va = (*si)->get_absoluteValue();
134  if (SgAsmInstruction *target_insn = imap.get_value_or(successor_va, NULL)) {
135  SgAsmFunction *target_func = AST::Traversal::findParentTyped<SgAsmFunction>(target_insn);
136  if (successor_va==target_func->get_entryVa()) {
137  callee_vas.insert(successor_va); // branches to a function entry point
138  } else if (return_va!=INVALID_ADDR) {
139  target_va = return_va = INVALID_ADDR;
140  return false; // multiple function-local CFG edges that are not this function's entry point
141  } else {
142  return_va = successor_va; // possible return address
143  }
144  }
145  }
146  }
147 
148  // Now for the architecture-dependent determination. This will not update target_va or return_va if they cannot be
149  // determined or are ambiguous; so we must reset them to INVALID_ADDR if we're about to return false.
150  bool retval = insns.front()->architecture()->isFunctionCallSlow(insns, &target_va, &return_va);
151  if (!retval) {
152  target_va = return_va = INVALID_ADDR;
153  } else if (INVALID_ADDR==target_va && 1==callee_vas.size()) {
154  target_va = *callee_vas.begin();
155  }
156  return retval;
157 }
