362  int value = wchar_val->get_value();
363  if (value < 256) {
364  if (value == '\b') {
365  curprint("\'\\b\'");
366  }
367  else if (value == '\t') {
368  curprint("\'\\t\'");
369  }
370  else if (value == '\n') {
371  curprint("\'\\n\'");
372  }
373  else if (value == '\f') {
374  curprint("\'\\f\'");
375  }
376  else if (value == '\r') {
377  curprint("\'\\r\'");
378  }
379  else if (value == '\"') {
380  curprint("\'\\\"\'");
381  }
382  else if (value == '\'') {
383  curprint("\'\\\'\'");
384  }
385  else if (value == '\\') {
386  curprint("\'\\\\\'");
387  }
388  else if (isprint(value)) {
389  string str = "\'";
390  str += (char) value;
391  str += "\'";
392  curprint(str.c_str());
393  }
394  else {
395  ostringstream octal; // stream used for the conversion
396  octal.fill('0');
397  octal.width(3);
398  octal << oct << value; // compute the octal character representation of the value
399  curprint("\'\\" + octal.str() + "\'");
400  }
401  }
402  else {
403  ostringstream unicode; // stream used for the conversion
404  unicode.fill('0');
405  unicode.width(4);
406  unicode << hex << value; // compute the Unicode character representation of the value
407  curprint("\'\\u" + unicode.str() + "\'");
408  }
409  }
410  else {
411  curprint(wchar_val->get_valueString());
412  }

603 // void Unparse_ExprStmt::unparseFuncCall
604 //
605 // This function is called whenever we unparse a function call. It is divided up
606 // into two parts. The first part unparses the function call and its arguments
607 // using an "in-order" tree traversal method. This is done when we have a binary
608 // operator overloading function and the operator overloading option is turned off.
609 // The second part unparses the function call directly in a list-like manner. This
610 // is done for non-operator function calls, or when the operator overloading option
611 // is turned on.
612 //-----------------------------------------------------------------------------------
613 void Unparse_Java::unparseFuncCall(SgExpression* expr, SgUnparse_Info& info) {
614  SgFunctionCallExp* func_call = isSgFunctionCallExp(expr);
615  ASSERT_not_null(func_call);
616 
617  if (func_call -> attributeExists("function_parameter_types")) {
618  AstRegExAttribute *attribute = (AstRegExAttribute *) func_call -> getAttribute("function_parameter_types");
619  curprint(attribute -> expression);
620  curprint(" ");
621  }
622 
623  if (func_call -> attributeExists("<init>")) {
624  AstRegExAttribute *constructor_attribute = (AstRegExAttribute *) func_call -> getAttribute("<init>");
625  curprint(constructor_attribute -> expression);
626  }
627  else unparseExpression(func_call->get_function(), info);
628  curprint("(");
629  unparseExpression(func_call->get_args(), info);
630  curprint(")");
