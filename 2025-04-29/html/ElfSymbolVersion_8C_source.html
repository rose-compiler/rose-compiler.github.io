4  * For more information see
5  * Sun: http://docs.sun.com/app/docs/doc/819-0690/chapter6-54676?a=view

8  *
9  * Related SageIII classes:
10  * SgAsmElfSymverSection -- GNU Symvol Version Table; The ".gnu.version" section, subclass of SgAsmElfSection. This

12  * SgAsmElfSymverEntry -- A single entry in an SgAsmElfSymverSection, usually an index into one of the other

71 SgAsmElfSymverSection *
72 SgAsmElfSymverSection::parse()
73 {
74  SgAsmElfSection::parse();
75 
76  SgAsmElfFileHeader *fhdr = get_elfHeader();
77  ROSE_ASSERT(fhdr!=NULL);
78  SgAsmElfSectionTableEntry *shdr = get_sectionEntry();
79  ASSERT_always_require(shdr!=NULL);
80 
81  size_t entry_size, struct_size, extra_size, nentries;
82  calculateSizes(&entry_size, &struct_size, &extra_size, &nentries);
83  ROSE_ASSERT(entry_size==shdr->get_sh_entsize());
84 
85  /* Parse each entry */
86  for (size_t i=0; i<nentries; ++i) {
87  SgAsmElfSymverEntry *entry=0;
88  entry = new SgAsmElfSymverEntry(this); /*adds symver to this symver table*/
89  uint16_t value;
90  readContentLocal(i*entry_size, &value, struct_size);
91  entry->set_value(diskToHost(fhdr->get_sex(), value));
92  }
93  return this;
94 }

112 void
113 SgAsmElfSymverSection::unparse(std::ostream &f) const
114 {
115  SgAsmElfFileHeader *fhdr = get_elfHeader();
116  ROSE_ASSERT(fhdr);
117  Rose::BinaryAnalysis::ByteOrder::Endianness sex = fhdr->get_sex();
118 
119  size_t entry_size, struct_size, extra_size, nentries;
120  calculateSizes(&entry_size, &struct_size, &extra_size, &nentries);
121 
122  /* Adjust the entry size stored in the ELF Section Table */
123  get_sectionEntry()->set_sh_entsize(entry_size);
124 
125  /* Write each entry's required part followed by the optional part */
126  for (size_t i=0; i<nentries; i++) {
127  uint16_t val;
128 
129  SgAsmElfSymverEntry *entry = p_entries->get_entries()[i];
130  Rose::BinaryAnalysis::ByteOrder::hostToDisk(sex,entry->get_value(),&val);
131 
132  Rose::BinaryAnalysis::Address spos = i * entry_size;
133  spos = write(f, spos, struct_size, &val);
134  }
135  unparseHoles(f);
136 }

273 SgAsmElfSymverDefinedSection *
274 SgAsmElfSymverDefinedSection::parse()
275 {
276  SgAsmElfSection::parse();
277 
278  SgAsmElfFileHeader *fhdr = get_elfHeader();
279  ROSE_ASSERT(NULL!=fhdr);
280 
281  Rose::BinaryAnalysis::Address entry_addr=0;
282 
283  Rose::BinaryAnalysis::ByteOrder::Endianness sex=fhdr->get_sex();
284  /* Parse each entry*/
285  while (entry_addr < this->get_size()) {
286  SgAsmElfSymverDefinedEntry *entry=new SgAsmElfSymverDefinedEntry(this); /*adds SymverDefinedEntry to this*/
287  SgAsmElfSymverDefinedEntry::ElfSymverDefinedEntry_disk entryDisk;
288  readContentLocal(entry_addr, &entryDisk, sizeof(entryDisk));
289  entry->parse(sex, &entryDisk);
290 
291  /* These are relative to the start of this entry - i.e. entry_addr */
292  size_t num_aux = Rose::BinaryAnalysis::ByteOrder::diskToHost(sex,entryDisk.vd_cnt);
293  size_t first_aux = Rose::BinaryAnalysis::ByteOrder::diskToHost(sex,entryDisk.vd_aux);
294  size_t next_entry = Rose::BinaryAnalysis::ByteOrder::diskToHost(sex,entryDisk.vd_next);
295 
296  Rose::BinaryAnalysis::Address aux_addr=entry_addr+first_aux;
297  for (size_t i=0; i < num_aux; ++i) {
298  SgAsmElfSymverDefinedAux *aux=new SgAsmElfSymverDefinedAux(entry,this); /*adds SymverDefinedAux to this entry*/
299  SgAsmElfSymverDefinedAux::ElfSymverDefinedAux_disk auxDisk;
300  readContentLocal(aux_addr, &auxDisk, sizeof(auxDisk));
301  aux->parse(fhdr->get_sex(), &auxDisk);
302 
303  size_t next_aux = Rose::BinaryAnalysis::ByteOrder::diskToHost(sex,auxDisk.vda_next);
304  if (next_aux == 0)
305  break;
306  aux_addr+=next_aux;
307  }
308 
309  if(next_entry == 0)
310  break;
311  entry_addr += next_entry;
312  }
313 
314  return this;
315 }

323 Rose::BinaryAnalysis::Address
324 SgAsmElfSymverDefinedSection::calculateSizes(size_t *entsize, size_t *required, size_t *optional, size_t *entcount) const
325 {
326  size_t struct_size = sizeof(SgAsmElfSymverDefinedEntry::ElfSymverDefinedEntry_disk);
327  size_t aux_size = sizeof(SgAsmElfSymverDefinedAux::ElfSymverDefinedAux_disk);
328 
329  size_t extra_size = 0;
330  size_t entry_size = 0;
331  size_t nentries = get_entries()->get_entries().size();
332 
333  size_t total_size=0;
334  /* Entire entry should be at least large enough for the required part. */
335  entry_size = struct_size;
336 
337  /* Each entry has a list of 'aux' structures */
338  const SgAsmElfSymverDefinedEntryPtrList &entries = get_entries()->get_entries();
339  for (size_t i=0; i<nentries; ++i) {
340  size_t numAux = entries[i]->get_entries()->get_entries().size();
341  extra_size = std::max(extra_size, numAux * aux_size);
342  total_size += (struct_size + (numAux* aux_size));
343  }
344  /* Note - we could try to reach into the dynamic section for nentries */
345 
346  /* Return values */
347  if (entsize)
348  *entsize = entry_size;
349  if (required)
350  *required = struct_size;
351  if (optional)
352  *optional = extra_size;
353  if (entcount)
354  *entcount = nentries;
355  return total_size;
356 }

358 void
359 SgAsmElfSymverDefinedSection::unparse(std::ostream &f) const
360 {
361  SgAsmElfFileHeader *fhdr = get_elfHeader();
362  ROSE_ASSERT(NULL != fhdr);
363  Rose::BinaryAnalysis::ByteOrder::Endianness sex = fhdr->get_sex();
364  size_t nentries;
365  calculateSizes(NULL,NULL,NULL,&nentries);
366 
367  /* Adjust the entry size stored in the ELF Section Table */
368  get_sectionEntry()->set_sh_entsize(0); /*This doesn't have consistently sized entries, zero it*/
369 
370  /* Write each entry's required part followed by the optional part */
371  Rose::BinaryAnalysis::Address entry_addr=0;/*as offset from section*/
372 
373  for (size_t ent=0; ent < nentries; ++ent) {
374  SgAsmElfSymverDefinedEntry *entry=get_entries()->get_entries()[ent];
375  SgAsmElfSymverDefinedEntry::ElfSymverDefinedEntry_disk entryDisk;
376  entry->encode(sex, &entryDisk);
377  const size_t entry_size=sizeof(SgAsmElfSymverDefinedEntry::ElfSymverDefinedEntry_disk);
378  const size_t aux_size=sizeof(SgAsmElfSymverDefinedAux::ElfSymverDefinedAux_disk);
379 
380  const SgAsmElfSymverDefinedAuxPtrList& auxes = entry->get_entries()->get_entries();
381  size_t num_aux = auxes.size();
382  size_t first_aux = entry_size;/*we always stick Aux's just after Entry*/
383  size_t next_entry = first_aux + ( num_aux * aux_size );
384  if (nentries -1 == ent) {
385  next_entry=0; /*final entry: next is null (0)*/
386  }
387 
388  Rose::BinaryAnalysis::ByteOrder::hostToDisk(sex, num_aux, &entryDisk.vd_cnt);
389  Rose::BinaryAnalysis::ByteOrder::hostToDisk(sex, first_aux, &entryDisk.vd_aux);
390  Rose::BinaryAnalysis::ByteOrder::hostToDisk(sex, next_entry, &entryDisk.vd_next);
391 
392  write(f, entry_addr, entry_size,&entryDisk);
393 
394  Rose::BinaryAnalysis::Address aux_addr=entry_addr+first_aux;
395  for (size_t i=0; i < num_aux; ++i) {
396  SgAsmElfSymverDefinedAux *aux=auxes[i];
397 
398  SgAsmElfSymverDefinedAux::ElfSymverDefinedAux_disk auxDisk;
399  aux->encode(sex, &auxDisk);
400 
401  size_t next_aux = aux_size;/*all auxes are contigious, so the next one is always just 8*/
402  if (num_aux-1 == i) {
403  next_aux=0;/* ... unless it is the final aux, then the next is null (0) */
404  }
405  Rose::BinaryAnalysis::ByteOrder::hostToDisk(sex,next_aux, &auxDisk.vda_next);
406 
407  write(f,aux_addr,aux_size, &auxDisk);
408  aux_addr+=next_aux;
409  }
410 
411  entry_addr += next_entry;
412  }
413 
414  unparseHoles(f);
415 }

630 SgAsmElfSymverNeededSection::calculateSizes(size_t *entsize, size_t *required, size_t *optional, size_t *entcount) const
631 {
632  size_t struct_size = sizeof(SgAsmElfSymverNeededEntry::ElfSymverNeededEntry_disk);
633  size_t aux_size = sizeof(SgAsmElfSymverNeededAux::ElfSymverNeededAux_disk);
634 
635  size_t extra_size = 0;
636  size_t entry_size = 0;
637  size_t nentries = get_entries()->get_entries().size();
638 
639  size_t total_size=0;
640  /* Entire entry should be at least large enough for the required part. */
641  entry_size = struct_size;
642 
643  /* Each entry has a list of 'aux' structures */
644  const SgAsmElfSymverNeededEntryPtrList &entries = get_entries()->get_entries();
645  for (size_t i=0; i<nentries; ++i) {
646  size_t numAux = entries[i]->get_entries()->get_entries().size();
647  extra_size = std::max(extra_size, numAux * aux_size);
648  total_size += (struct_size + (numAux* aux_size));
649  }
650  /* Note - we could try to reach into the dynamic section for nentries */
651 
652  /* Return values */
653  if (entsize)
654  *entsize = entry_size;
655  if (required)
656  *required = struct_size;
657  if (optional)
658  *optional = extra_size;
659  if (entcount)
660  *entcount = nentries;
661  return total_size;
662 }

664 void
665 SgAsmElfSymverNeededSection::unparse(std::ostream &f) const
666 {
667  SgAsmElfFileHeader *fhdr = get_elfHeader();
668  ROSE_ASSERT(NULL != fhdr);
669  Rose::BinaryAnalysis::ByteOrder::Endianness sex = fhdr->get_sex();
670  size_t nentries;
671  calculateSizes(NULL,NULL,NULL,&nentries);
672 
673  /* Adjust the entry size stored in the ELF Section Table */
674  get_sectionEntry()->set_sh_entsize(0); /* This doesn't have consistently sized entries, zero it */
675 
676  /* Write each entry's required part followed by the optional part */
677  /* Parse each entry*/
678  Rose::BinaryAnalysis::Address entry_addr=0; /* as offset from section */
679 
680  for (size_t ent=0; ent < nentries; ++ent) {
681  SgAsmElfSymverNeededEntry *entry=get_entries()->get_entries()[ent];
682  SgAsmElfSymverNeededEntry::ElfSymverNeededEntry_disk entryDisk;
683  entry->encode(sex, &entryDisk);
684  const size_t entry_size=sizeof(SgAsmElfSymverNeededEntry::ElfSymverNeededEntry_disk);
685  const size_t aux_size=sizeof(SgAsmElfSymverNeededAux::ElfSymverNeededAux_disk);
686 
687  const SgAsmElfSymverNeededAuxPtrList& auxes = entry->get_entries()->get_entries();
688  size_t num_aux = auxes.size();
689  size_t first_aux = entry_size; /* we always stick Aux's just after Entry */
690  size_t next_entry = first_aux + ( num_aux * aux_size );
691  if (nentries -1 == ent) {
692  next_entry=0; /* final entry: next is null (0) */
693  }
694 
695  Rose::BinaryAnalysis::ByteOrder::hostToDisk(sex, num_aux, &entryDisk.vn_cnt);
696  Rose::BinaryAnalysis::ByteOrder::hostToDisk(sex, first_aux, &entryDisk.vn_aux);
697  Rose::BinaryAnalysis::ByteOrder::hostToDisk(sex, next_entry, &entryDisk.vn_next);
698 
699  write(f, entry_addr, entry_size,&entryDisk);
700 
701  Rose::BinaryAnalysis::Address aux_addr=entry_addr+first_aux;
702  for (size_t i=0; i < num_aux; ++i) {
703  SgAsmElfSymverNeededAux *aux=auxes[i];
704 
705  SgAsmElfSymverNeededAux::ElfSymverNeededAux_disk auxDisk;
706  aux->encode(sex, &auxDisk);
707 
708  size_t next_aux = aux_size;/* all auxes are contigious, so the next one is always just 8 */
709  if (num_aux-1 == i) {
710  next_aux=0; /* ... unless it is the final aux, then the next is null (0) */
711  }
712  Rose::BinaryAnalysis::ByteOrder::hostToDisk(sex,next_aux, &auxDisk.vna_next);
713 
714  write(f,aux_addr,aux_size, &auxDisk);
715  aux_addr+=next_aux;
716  }
717  entry_addr += next_entry;
718  }
719  unparseHoles(f);
720 }
