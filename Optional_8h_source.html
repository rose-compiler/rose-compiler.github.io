397// These functions intentionally do not compile. They are to prevent comparisons and thus save users from making
398// mistakes like this:
399// Optional<int> x = 0;
400// int y = 1;
401// if (x == y) // won't compile
402// if (x && *x == y) // what they really meant
403// if (x.isEqual(y)) // another valid way to write it
404template<typename T, typename U>
405bool operator==(const Optional<T> &lhs, const U&) {
406 lhs.this_type_does_not_support_comparisons();
407 return false;
408}
409
410template<typename T, typename U>
411bool operator!=(const Optional<T> &lhs, const U&) {
412 lhs.this_type_does_not_support_comparisons();
413 return false;
414}
